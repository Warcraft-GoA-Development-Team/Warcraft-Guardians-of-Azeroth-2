# Port of vanilla heresy_historical.0011 without clerical regions.
# Root is the origin faith; the budget lives on scope:heresy_strict_capital (the seat barony).
wc_heresy_historical.0011 = {
	hidden = yes
	scope = faith

	trigger = {
		exists = scope:origin_faith
		exists = scope:heresy_faith
		exists = scope:heresy_strict_capital
		exists = scope:epicenter_ruler
		exists = scope:founder
		scope:heresy_strict_capital = {
			has_variable = wc_heresy_spread_remaining
			var:wc_heresy_spread_remaining > 0
		}
	}

	immediate = {
		clear_saved_scope = convert_ruler

		scope:origin_faith = {
			random_faith_ruler = {
				limit = {
					is_physically_able_ai = yes
					is_landed = yes
					highest_held_title_tier >= tier_county
					exists = capital_county
					NOR = {
						this = scope:founder
						this = scope:epicenter_ruler
						faith.religious_head ?= this
						any_held_title = { is_head_of_faith = yes }
						has_character_flag = recently_converted_by_heresy
						recently_converted_faith_trigger = yes
					}
					capital_county = {
						squared_distance = {
							target = scope:epicenter_ruler.capital_county
							value <= squared_distance_small
						}
					}
				}
				weight = {
					base = 100
					modifier = {
						factor = 1.5
						culture = { has_same_culture_heritage = scope:epicenter_ruler.culture }
					}
					modifier = {
						factor = 0.4
						highest_held_title_tier >= tier_kingdom
					}
					modifier = {
						factor = 1.5
						OR = {
							has_trait = zealous
							has_trait = lifestyle_mystic
							has_trait = lifestyle_scholar
							has_trait = theologian
						}
					}
					modifier = {
						factor = 0.001
						OR = {
							government_has_flag = government_is_nomadic
							government_has_flag = government_is_herder
						}
					}
				}
				save_scope_as = convert_ruler
			}
		}

		if = {
			limit = { exists = scope:convert_ruler }

			scope:convert_ruler = { wc_heresy_convert_ruler_effect = yes }

			scope:heresy_strict_capital = {
				if = {
					limit = { scope:convert_ruler = { highest_held_title_tier >= tier_kingdom } }
					change_variable = {
						name = wc_heresy_spread_remaining
						subtract = 4
					}
				}
				else_if = {
					limit = { scope:convert_ruler = { highest_held_title_tier = tier_duchy } }
					change_variable = {
						name = wc_heresy_spread_remaining
						subtract = 2
					}
				}
				else = {
					change_variable = {
						name = wc_heresy_spread_remaining
						subtract = 1
					}
				}
				if = {
					limit = { var:wc_heresy_spread_remaining <= 0 }
					remove_variable = wc_heresy_spread_remaining
				}
			}

			if = {
				limit = { scope:convert_ruler ?= { faith = scope:heresy_faith } }
				wc_heresy_send_spread_message_effect = yes
			}

			if = {
				limit = { scope:heresy_strict_capital = { has_variable = wc_heresy_spread_remaining } }
				trigger_event = {
					id = wc_heresy_historical.0011
					days = { 5 12 }
				}
			}
		}
		else = {
			scope:heresy_strict_capital = { remove_variable = wc_heresy_spread_remaining }
		}
	}
	# If the trigger fails for another reason (e.g. the founder scope is gone), the variable stays on the
	# seat barony. Harmless: the next outbreak of that heresy overwrites it, and budget effects only shrink it.
}
