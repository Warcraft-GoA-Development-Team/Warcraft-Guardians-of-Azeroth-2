namespace = forsaken_kingdom

# forsaken_kingdom.0001	Bandarion Keep and Tyr's Watch (liege at Garek's death)
# forsaken_kingdom.0002	Tyr's Watch founded (news: Holy Light players, rulers who hear from Lordaeron)

#############################################
# Bandarion Keep and Tyr's Watch
# Fired by wc_bandarion_keep_death_on_actions one day after Garek Bandarion dies
# holding b_autumnacre. Root: his liege at death, who sponsors the order.
# scope:garek is the dead captain, saved by the hook.
# Always renames the barony. If Evin Schmidtt can lead an order and the barony
# can be leased, Evin returns to the Light and founds Tyr's Watch there, and
# forsaken_kingdom.0002 goes out as news.
# Rename only (accepted): Evin dead, a child, landed or already converted; or the
# barony went to an heir whose primary title is above barony tier.
# The text never states how Garek died: in play the engine decides that.
#############################################
forsaken_kingdom.0001 = {
	type = character_event
	content_source = dlc_GOA
	title = forsaken_kingdom.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tyrs_watch_founded }
				desc = forsaken_kingdom.0001.desc.founded
			}
			desc = forsaken_kingdom.0001.desc.rename_only
		}
	}
	theme = death
	override_background = { reference = wc_undercity_throne_room }

	left_portrait = {
		character = root
		animation = sadness
	}
	# Only when Evin steps forward; the trigger draws nothing in the rename-only branch
	right_portrait = {
		trigger = { exists = scope:evin }
		character = scope:evin
		animation = prayer
	}
	lower_center_portrait = scope:garek

	immediate = {
		play_music_cue = "mx_cue_death"

		title:b_autumnacre = { set_title_name = bandarion_keep_custom_name }

		if = {
			limit = {
				character:evin_schmidtt ?= {
					is_alive = yes

					is_adult = yes

					is_landed = no

					has_character_flag = wc_awaits_the_light
				}
				title:b_autumnacre = {
					barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = root }
				}
			}
			save_scope_value_as = { name = tyrs_watch_founded value = yes }

			character:evin_schmidtt = {
				save_scope_as = evin

				hidden_effect = {
					set_variable = { name = wc_light_magic_lifestyle_additional_perks_variable value = wc_perks_needed_for_level_2_magic_trait_value }

					remove_character_flag = wc_awaits_the_light
				}

				set_character_faith = faith:holy_light

				hidden_effect = {
					if = {
						limit = { NOT = { has_perk = light_magic_tree_1_perk_1 } }
						add_perk = light_magic_tree_1_perk_1
					}
				}
			}
			# Only a founded order exempts the barony from lease revocation
			hidden_effect = {
				title:b_autumnacre = { set_variable = wc_tyrs_watch_seat }
			}
			wc_found_tyrs_watch_effect = yes

			# News: Holy Light players anywhere, and anyone who hears from Lordaeron
			hidden_effect = {
				every_player = {
					limit = {
						this != root

						OR = {
							faith = faith:holy_light
							get_news_from_region_trigger = { REGION = world_eastern_kingdoms_lordaeron }
						}
					}
					trigger_event = {
						id = forsaken_kingdom.0002
						days = { 3 7 }
					}
				}
			}
		}

		stress_impact = { base = minor_stress_impact_gain }
	}

	option = {	# Acknowledge; no choice by design
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { exists = scope:tyrs_watch_founded }
						desc = forsaken_kingdom.0001.a.founded
					}
					desc = forsaken_kingdom.0001.a.rename_only
				}
			}
		}
	}
}

#############################################
# Tyr's Watch founded (news)
# Fired by forsaken_kingdom.0001 three to seven days after the founding.
# Root: the viewer. Holy Light players anywhere, and anyone who gets news from
# Lordaeron, except the sponsor. Text varies by the viewer's faith.
# Purely informational: no effects.
#############################################
forsaken_kingdom.0002 = {
	type = character_event
	content_source = dlc_GOA
	title = forsaken_kingdom.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { faith = faith:holy_light }
				desc = forsaken_kingdom.0002.desc.light
			}
			triggered_desc = {
				trigger = { faith = faith:forsaken_cult }
				desc = forsaken_kingdom.0002.desc.forsaken
			}
			desc = forsaken_kingdom.0002.desc.other
		}
	}
	theme = faith
	override_background = { reference = temple_church }

	# Days have passed since the founding
	trigger = {
		exists = scope:evin
		scope:evin = { is_alive = yes }
	}

	left_portrait = {
		character = scope:evin
		animation = prayer
	}
	lower_center_portrait = scope:garek

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
	}

	option = {	# Close
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { faith = faith:holy_light }
						desc = forsaken_kingdom.0002.a.light
					}
					triggered_desc = {
						trigger = { faith = faith:forsaken_cult }
						desc = forsaken_kingdom.0002.a.forsaken
					}
					desc = forsaken_kingdom.0002.a.other
				}
			}
		}
	}
}
