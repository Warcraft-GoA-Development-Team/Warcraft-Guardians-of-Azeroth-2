# Any demon liege (Balnazzar included) auto-raises newly gained vassals as undead thralls.
# Reuses raise_undead_faith_only_effect (faith conversion, culture untouched) + loyal_undead opinion, same as wc_balnazzar_story.2000.
on_vassal_gained = {
	effect = {
		if = {
			limit = {
				root = { has_trait = being_demon }
				NOT = { scope:vassal = { has_trait = being_undead } }
			}
			scope:vassal = {
				raise_undead_faith_only_effect = { MASTER = root }
				add_opinion = { modifier = loyal_undead target = root }
				every_vassal_or_below = {
					limit = {
						is_alive = yes
						NOT = { has_trait = being_undead }
					}
					raise_undead_faith_only_effect = { MASTER = liege }
					add_opinion = { modifier = loyal_undead target = liege }
				}
			}
		}
	}
}

# Stratholme taken by conquest (not grant/inheritance) while still disguised - fires wc_balnazzar_story.1007.
on_title_gain = {
	effect = {
		if = {
			limit = {
				has_trait = nathrezim_disguise
				scope:title = title:c_stratholme
				OR = {
					scope:transfer_type = flag:conquest
					scope:transfer_type = flag:conquest_holy_war
					scope:transfer_type = flag:conquest_claim
					scope:transfer_type = flag:conquest_populist
				}
			}
			trigger_event = {
				id = wc_balnazzar_story.1007
				days = 3
			}
		}
	}
}

# wc_balnazzar_story.1002's own comment says it fires from on_title_lost when
# c_lordaeron is lost - that hook was never actually wired up. Redirect
# pattern (not a raw on_title_lost = { effect = {...} } block) matches
# wc_horde_invasion_on_actions.txt / wc_theramore_task_on_actions.txt, so this
# doesn't clobber their own on_title_lost hooks or the vanilla one.
on_title_lost = {
	on_actions = {
		wc_balnazzar_on_title_lost_actions
	}
}

wc_balnazzar_on_title_lost_actions = {
	effect = {
		if = {
			limit = {
				scope:title = title:c_lordaeron
				exists = global_var:balnazzar_dreadlord_story
				global_var:balnazzar_dreadlord_story = { story_owner = root }
				NOT = { has_trait = nathrezim_disguise }
			}
			trigger_event = {
				id = wc_balnazzar_story.1002
				days = { 3 10 }
			}
		}
	}
}
