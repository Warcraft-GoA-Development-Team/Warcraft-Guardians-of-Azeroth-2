WcBalnazzarCoverCrisis = {
	type = character
	text = {
		trigger = { has_variable = wc_nathrezim_cover_crisis }
		localization_key = WC_BALNAZZAR_COVER_CRISIS
	}
	text = {
		trigger = { NOT = { has_variable = wc_nathrezim_cover_crisis } }
		localization_key = WC_BALNAZZAR_NO_COVER_CRISIS
	}
}

WcBalnazzarWitnessPressure = {
	type = character
	text = {
		trigger = { has_variable = wc_nathrezim_cover_crisis }
		localization_key = WC_BALNAZZAR_PRESSURE_ENDED
	}
	text = {
		trigger = { wc_balnazzar_free_witness_count >= 2 }
		localization_key = WC_BALNAZZAR_PRESSURE_SEVERAL
	}
	text = {
		trigger = { wc_balnazzar_free_witness_count = 1 }
		localization_key = WC_BALNAZZAR_PRESSURE_ONE
	}
	text = {
		trigger = { wc_balnazzar_free_witness_count = 0 }
		localization_key = WC_BALNAZZAR_PRESSURE_NONE
	}
}
