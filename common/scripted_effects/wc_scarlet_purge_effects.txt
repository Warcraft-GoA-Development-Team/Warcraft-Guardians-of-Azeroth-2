# Local ruler scope: direct subjects only, so subordinate courts are not duplicated.
wc_scarlet_purge_scan_effect = {
	if = {
		limit = { wc_scarlet_purge_authority_trigger = yes }
		save_scope_as = wc_purge_authority
		every_courtier = {
			limit = { wc_scarlet_purge_target_trigger = { AUTHORITY = scope:wc_purge_authority } }
			wc_scarlet_purge_summon_effect = yes
		}
		every_vassal = {
			limit = { wc_scarlet_purge_target_trigger = { AUTHORITY = scope:wc_purge_authority } }
			wc_scarlet_purge_summon_effect = yes
		}
		every_prisoner = {
			limit = { wc_scarlet_purge_target_trigger = { AUTHORITY = scope:wc_purge_authority } }
			wc_scarlet_purge_summon_effect = yes
		}
	}
}

wc_scarlet_purge_summon_effect = {
	if = {
		limit = { wc_scarlet_purge_target_trigger = { AUTHORITY = scope:wc_purge_authority } }
		if = {
			limit = { NOT = { has_variable = wc_scarlet_purge_generation } }
			set_variable = { name = wc_scarlet_purge_generation value = 0 }
		}
		change_variable = { name = wc_scarlet_purge_generation add = 1 }
		save_scope_value_as = { name = wc_purge_generation value = var:wc_scarlet_purge_generation }
		set_variable = { name = wc_scarlet_purge_summons value = scope:wc_purge_authority years = 1 }
		add_character_flag = { flag = wc_scarlet_purge_cooldown years = 1 }
		if = {
			limit = { is_imprisoned_by = scope:wc_purge_authority }
			wc_scarlet_purge_open_trial_effect = yes
		}
		else = { trigger_event = wc_scarlet_purge.0001 }
	}
}

# Already imprisoned subject. Reuse the existing case generation and verdicts.
wc_scarlet_purge_open_trial_effect = {
	if = {
		limit = {
			is_alive = yes
			has_trait = being_undead
			NOT = { has_variable = wc_inquisition_stage }
			var:wc_scarlet_purge_summons ?= scope:wc_purge_authority
			is_imprisoned_by = scope:wc_purge_authority
			scope:wc_purge_authority = { wc_scarlet_purge_authority_trigger = yes }
		}
		# A local living judge replaces the religious head for this internal case.
		scope:wc_purge_authority = { save_scope_as = actor save_scope_as = inquisitor }
		set_variable = { name = wc_inquisition_purge_case value = scope:wc_purge_authority }
		wc_inquisition_open_effect = yes
		if = {
			limit = { has_variable = wc_inquisition_stage }
			remove_variable = wc_scarlet_purge_summons
		}
		else = { remove_variable = wc_inquisition_purge_case }
	}
}
