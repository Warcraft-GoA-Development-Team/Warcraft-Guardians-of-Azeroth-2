expose_to_radiation_interaction = {
	category = interaction_category_prison
	ignores_pending_interaction_block = yes
	icon = turn_leper
	common_interaction = yes #suppress error log
	
	interface_priority = -1

	ai_targets = {
		ai_recipients = prisoners
	}

	ai_frequency = 12

	is_shown = {
		scope:puppet_or_actor = {
			has_trait = irradiated
		}
		scope:recipient = {
			is_imprisoned_by = scope:puppet_or_actor
			NOR = {
				has_trait = irradiated
				has_trait = being_undead
			}
		}
	}

	on_accept = {
		scope:recipient = { 
			add_trait = irradiated
			set_culture = culture:leper_gnome
			set_character_faith = faith:glowbound		
		}
	}
	auto_accept = yes
}

# Send Prisoner to twisting nether
netherbind_interaction = {
	category = interaction_category_prison
	ignores_pending_interaction_block = yes
	icon = icon_spellbook
	common_interaction = yes #suppress error log
	
	interface_priority = -1

	ai_targets = {
		ai_recipients = prisoners
	}

	ai_frequency = 12

	is_shown = {
		scope:puppet_or_actor = {
			has_character_modifier = wc_eredruin_summon_modifier
		}
		scope:recipient = {
			is_imprisoned_by = scope:puppet_or_actor
			NOR = {
				is_in_prison_type = twisting_nether
				has_trait = being_demon
				has_trait = being_undead
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_description = {
				text = "currently_being_tortured"
				NOT = { has_character_flag = is_being_tortured }
			}
		}
		scope:puppet_or_actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }	# Warcraft
		custom_tooltip = {
			text = is_currently_being_purged_tt
			scope:recipient = {
				NOT = {
					has_character_flag = is_currently_being_purged
				}
			}
		}
		scope:recipient = {
			NOT = { has_trait = in_twisting_nether }
		}
	}

	on_accept = {
		if = {
			limit = { scope:recipient = { has_character_modifier = allowed_to_go_outside } }
			scope:recipient = { remove_character_modifier = allowed_to_go_outside }
		}
		scope:recipient = {
			hidden_effect = {
				change_prison_type = twisting_nether
			}
			trigger_event = WCBEI.100
			custom_tooltip = MOVE_PRISONER_TO_TWISTING_NETHER
			scope:recipient = {
				every_close_or_extended_family_member = {
					custom = all_family_members
					limit = {
						scope:puppet_or_actor != this
					}
					add_opinion = {
						modifier = family_moved_to_twisting_nether_opinion
						target = scope:puppet_or_actor
					}
				}
			}
		}
		scope:puppet_or_actor = {
			stress_impact = {
				compassionate = medium_stress_impact_gain
				forgiving = medium_stress_impact_gain
			}
			# set_variable = {
			# 	name = netherbind_prowess_bonus
			# 	value = 0
			# }
			hidden_effect = {
				send_interface_message = {
					type = event_prison_neutral
					title = msg_move_prisoner_to_twisting_nether
					right_icon = scope:recipient
					custom_tooltip = MOVE_PRISONER_TO_TWISTING_NETHER
				}
			}
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = medium_unity_loss
			DESC = clan_unity_move_to_dungeon.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}

	auto_accept = yes
}
