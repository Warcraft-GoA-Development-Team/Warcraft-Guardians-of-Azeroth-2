wc_titanforged_forge.0010 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_titanforged_forge.0010.title
	desc = wc_titanforged_forge.0010.desc
	theme = realm
	option = {
		name = wc_titanforged_forge.0010.pay
		trigger = { wc_titanforged_settlement_audience_trigger = yes scope:restoration = { var:wc_titanforged_waiting = flag:installation } gold >= 50 }
		ai_chance = { base = 50 }
		if = {
			limit = {
				wc_titanforged_settlement_audience_trigger = yes
				scope:restoration = {
					var:wc_titanforged_waiting = flag:installation
				}
				gold >= 50
			}
			remove_short_term_gold = 50
			scope:county = {
				set_variable = {
					name = wc_titanforged_installations
					value = yes
				}
				set_variable = {
					name = wc_titanforged_settlement_checkpoint
					value = 3
				}
			}
			scope:restoration = {
				remove_variable = wc_titanforged_waiting
			}
		}
		else = { wc_titanforged_settlement_retry_effect = { STAGE = installation } }
	}
	option = {
		name = wc_titanforged_forge.0010.requisition
		trigger = { wc_titanforged_settlement_audience_trigger = yes scope:restoration = { var:wc_titanforged_waiting = flag:installation }  }
		ai_chance = { base = 50 modifier = { factor = 3 has_trait = callous } modifier = { factor = 0.25 has_trait = compassionate } }
		if = {
			limit = {
				wc_titanforged_settlement_audience_trigger = yes
				scope:restoration = {
					var:wc_titanforged_waiting = flag:installation
				}
			}
			scope:county = {
				add_county_modifier = {
					modifier = wc_titanforged_requisitioned_installations_modifier
					years = 5
				}
			}
			scope:county = {
				set_variable = {
					name = wc_titanforged_installations
					value = yes
				}
				set_variable = {
					name = wc_titanforged_settlement_checkpoint
					value = 3
				}
			}
			scope:restoration = {
				remove_variable = wc_titanforged_waiting
			}
		}
		else = { wc_titanforged_settlement_retry_effect = { STAGE = installation } }
	}
	option = {
		name = wc_titanforged_restoration.wait
		trigger = { wc_titanforged_settlement_audience_trigger = no }
	}

}
