scripted_trigger study_faith_appropriate_lover_target = {
	faith = scope:target.faith
	can_set_relation_lover_trigger = { CHARACTER = root }
	NOR = {
		has_trait = chaste
	}
	is_available_ai_adult = yes
	might_cheat_on_every_partner_trigger = yes
	accepts_incest_with_trigger = { CHARACTER = root } #No incest or 100% ok with incest
}

scripted_effect study_faith_set_random_progress_effect = {
	if = {
		limit = {
			any_character_tenet = {
				knowledge_filter = unknown
				rite_filter = scope:target.rite
			}
		}
		random_character_tenet = {
			knowledge_filter = unknown
			rite_filter = scope:target.rite
			save_scope_as = discovered_tenet
		}
	}
	else = {
		limit = {
			any_character_doctrine = {
				knowledge_filter = unknown
				faith_filter = scope:target.faith
			}
		}
		random_character_doctrine = {
			knowledge_filter = unknown
			rite_filter = scope:target.rite
			save_scope_as = discovered_doctrine
		}
	}
}

scripted_effect study_faith_gain_random_progress_effect = {
	if = {
		limit = { exists = scope:discovered_tenet }
		add_known_tenet = scope:discovered_tenet
		pam_study_rite_tenet_popularity_effect = yes
	}
	if = {
		limit = { exists = scope:discovered_doctrine }
		add_known_doctrine = scope:discovered_doctrine
	}
}

# You find out that the target is a non-believer - now what?
study_faith_ongoing.1011 = {
	type = character_event
	title = study_faith_ongoing.1011.t
	desc = study_faith_ongoing.1011.desc
	theme = study_faith

	left_portrait = {
		character = root
		animation = disbelief
	}
	right_portrait = {
		character = scope:target
		animation = holding_scrolls
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}

	cooldown = { years = 50 }
	
	trigger = {
		scope:target = {
			OR = {
				intrigue > high_skill_rating
				has_trait = lifestyle_mystic
				any_secret = {
					type = secret_non_believer
				}
			}
			is_ai = yes 
			not = { has_variable = study_faith_nonbeliever_event }
		}
	}

	immediate = {
		# set the target as non believer
		scope:target = {
			if = {
				limit = {
					not = {
						any_secret = {
							type = secret_non_believer
						}
					}
				}
				give_non_believer_secret_effect = yes
			}
			random_secret = {
				type = secret_non_believer
				save_scope_as = nonbeliever_secret
			}
			set_variable = study_faith_nonbeliever_event
		}
	}

	option = { # I cannot rely on this source anymore
		name = study_faith_ongoing.1011.a
		
		scope:scheme = {
			end_scheme = yes
		}
		add_piety = minor_piety_gain

		scope:nonbeliever_secret ?= {
			reveal_to = root
		}

		stress_impact = {
			lazy = miniscule_stress_impact_loss
			impatient = major_stress_impact_gain
			diligent = major_stress_impact_gain
			zealous = major_stress_impact_loss
			fickle = minor_stress_impact_loss
		}

		ai_chance = {
			base = 80
		}
	}

	option = { # Maybe we can figure things out by how they're faking it
		name = study_faith_ongoing.1011.b
		
		scope:scheme = {
			add_scheme_modifier = {
				type = study_faith_learning_mannerism
			}
		}
		
		stress_impact = {
			impatient = medium_stress_impact_gain
			stubborn = minor_stress_impact_loss
			diligent = minor_stress_impact_loss
			zealous = major_stress_impact_gain
			fickle = minor_stress_impact_gain
		}
		ai_chance = {
			base = 20
		}
	}

	option = { # What do they believe in then?
		name = study_faith_ongoing.1011.c

		trigger = {
			# Warcraft - a hidden Warcraft cult instead of vanilla's pagan/hellenic culture groups (issue #1969)
			scope:target = { wc_study_faith_has_hidden_cult_trigger = { OWNER = root } }
		}

		custom_tooltip = study_faith_ongoing.1011.c.tt

		scope:scheme = {
			add_scheme_modifier = {
				type = study_faith_learning_mannerism
			}
			add_scheme_progress = scheme_progress_loss
		}
		
		stress_impact = {
			impatient = minor_stress_impact_gain
			zealous = medium_stress_impact_gain
			fickle = miniscule_stress_impact_loss
		}

		hidden_effect = {
			random_list = {
				# Warcraft - 20% they secretly follow a cult (issue #1969)
				20 = {
					trigger_event = {
						id = study_faith_ongoing.1012
						days = 120
					}
				}
				# else they are just cynical
				80 = {
					trigger_event = {
						id = study_faith_ongoing.1013
						days = 90
					}
				}
			}
		}

		ai_chance = {
			base = 0
		}
	}
}

scripted_effect study_faith_find_new_teacher_effect = {
	scope:scheme = {
		end_scheme = yes
	}
	set_variable = {
		name = study_faith_progress
		value = prev.scheme_progress
	}
	set_variable = study_faith_dissatisfied_with_current_teacher
	trigger_event = {
		id = study_faith_outcome.3000
		days = 3
	}
}

# Warcraft - rebuilt as a hidden cult reveal (issue #1969). Vanilla's hellenic/pagan faiths do not exist in GoA.
# The teacher secretly follows a forbidden cult: expose them, watch them, join them, or find another teacher.
study_faith_ongoing.1012 = {
	type = character_event
	title = wc_study_faith_ongoing.1012.t
	desc = {
		desc = wc_study_faith_ongoing.1012.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = { scope:actual_faith = faith:death_god }
				desc = wc_study_faith_ongoing.1012.desc.damned
			}
			triggered_desc = {
				trigger = { scope:actual_faith = faith:orcish_fel }
				desc = wc_study_faith_ongoing.1012.desc.burning_blade
			}
			triggered_desc = {
				trigger = { scope:actual_faith = faith:cult_of_hakkar }
				desc = wc_study_faith_ongoing.1012.desc.atalai
			}
			triggered_desc = {
				trigger = { scope:actual_faith = faith:ragnarism }
				desc = wc_study_faith_ongoing.1012.desc.dark_iron
			}
			desc = wc_study_faith_ongoing.1012.desc.twilight
		}
	}
	theme = study_faith

	left_portrait = {
		character = root
		animation = shock
	}
	right_portrait = {
		character = scope:target
		animation = scheme
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}

	trigger = {
		exists = scope:scheme
		scope:target ?= {
			is_alive = yes
			wc_study_faith_has_hidden_cult_trigger = { OWNER = root }
		}
	}

	# Warcraft - if the reveal no longer holds after the delay, fall back to vanilla's "just a cynic" outcome (issue #1969)
	on_trigger_fail = {
		if = {
			limit = {
				exists = scope:scheme
				scope:target ?= { is_alive = yes }
			}
			trigger_event = {
				id = study_faith_ongoing.1013
				days = 1
			}
		}
	}

	immediate = {
		scope:target = {
			wc_study_faith_pick_hidden_cult_effect = { OWNER = root }
			make_character_crypto_religionist_effect = { CRYPTO_RELIGION = scope:actual_faith }
			random_secret = {
				type = secret_crypto_religionist
				save_scope_as = cult_secret
			}
			scope:cult_secret ?= { reveal_to = root }
		}
	}

	# Expose them before the court
	option = {
		name = wc_study_faith_ongoing.1012.a
		custom_tooltip = wc_study_faith_ongoing.1012.a.tt

		add_piety = medium_piety_gain
		scope:cult_secret ?= {
			expose_secret = root
		}
		scope:scheme = { end_scheme = yes }

		stress_impact = {
			zealous = minor_stress_impact_loss
			compassionate = minor_stress_impact_gain
			honest = minor_stress_impact_gain
		}
		ai_chance = {
			base = 0
		}
	}

	# Keep them close, and keep studying
	option = {
		name = wc_study_faith_ongoing.1012.b

		if = {
			limit = {
				scope:cult_secret ?= {
					is_blackmailable_secret_trigger = {
						PARTICIPANT = scope:target
						BLACKMAILER = root
					}
				}
			}
			blackmail_target_with_secret_effect = {
				TARGET = scope:target
				SECRET = scope:cult_secret
			}
		}
		hidden_effect = {
			random = {
				chance = 15
				trigger_event = {
					id = wc_study_faith.0400
					days = { 150 210 }
				}
			}
		}

		stress_impact = {
			honest = minor_stress_impact_gain
			paranoid = minor_stress_impact_gain
			deceitful = minor_stress_impact_loss
		}
		ai_chance = {
			base = 0
		}
	}

	# Accept their initiation
	option = {
		name = wc_study_faith_ongoing.1012.c
		custom_tooltip = wc_study_faith_ongoing.1012.c.tt
		add_internal_flag = dangerous

		wc_study_faith_initiation_effect = yes

		stress_impact = {
			zealous = major_stress_impact_gain
			honest = minor_stress_impact_gain
			ambitious = minor_stress_impact_loss
			arbitrary = minor_stress_impact_loss
		}
		ai_chance = {
			base = 0
		}
	}

	# Find another teacher (the wc_ copy of vanilla's find-new-teacher effect; the witch secret vanilla gave the target is gone)
	option = {
		name = wc_study_faith_ongoing.1012.d
		custom_tooltip = study_faith_ongoing.1012.b.tt

		wc_study_faith_find_new_teacher_effect = yes

		stress_impact = {
			stubborn = minor_stress_impact_loss
			diligent = minor_stress_impact_loss
			zealous = minor_stress_impact_gain
			lazy = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 100
		}
	}
}
