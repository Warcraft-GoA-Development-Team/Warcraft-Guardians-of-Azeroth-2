# Evaluated in FAITH scope
petition_head_of_faith_faith_qualifies_trigger = {
	religion = religion:christianity_religion
	main_rite = {
		rite_has_parameter = spiritual_head_of_faith
	}
}

# Resolves the Head of Faith used by petitions, including an antipope where applicable.
petition_head_of_faith_has_religious_authority_trigger = {
	exists = religious_head_or_challenger
}

petition_head_of_faith_religious_authority_is_trigger = {
	religious_head_or_challenger ?= $CHARACTER$
}

petition_head_of_faith_has_usable_hook_on_religious_authority_trigger = {
	trigger_if = {
		limit = { petition_head_of_faith_has_religious_authority_trigger = yes }
		has_usable_hook = religious_head_or_challenger
	}
	trigger_else = { always = no }
}

# Evaluated in CHARACTER scope
# Follows petition_head_of_faith_recipient_still_valid_trigger so setup and arrival stay aligned
petition_head_of_faith_core_available_trigger = {
	has_pam_dlc_trigger = yes
	is_ai = no
	is_imprisoned = no
	custom_description = {
		text = petition_head_of_faith_decision_head_of_faith_exists
		petition_head_of_faith_has_religious_authority_trigger = yes
	}
	# Cannot be the recipient, at war with them, or have a petition pending or in flight
	trigger_if = {
		limit = { petition_head_of_faith_has_religious_authority_trigger = yes }
		NOR = {
			religious_head_or_challenger = this
			is_at_war_with = religious_head_or_challenger
		}
		religious_head_or_challenger = {
			save_temporary_scope_as = hof # for localization
			is_imprisoned = no
		}
		custom_tooltip = {
			text = NOT_IN_A_PETITION_TOOLTIP
			has_none_of_variables = {
				name = pending_petition_head_of_faith_type
				name = petition_head_of_faith_character_flag
			}
		}
	}
	# No recipient or pending-petition checks are needed when the vacancy requirement above fails.
}

# Must travel to the Head of Faith for absolution if:
# - King or emperor (always), OR
# - The Pope personally excommunicated you (pope_excom flag), at any rank.
# Mirrors request_excommunication: high-rank targets also require a Rome trip.
petition_head_of_faith_repentance_requires_petition_trigger = {
	faith = { petition_head_of_faith_faith_qualifies_trigger = yes }
	petition_head_of_faith_has_religious_authority_trigger = yes
	trigger_if = {
		limit = {
			NOR = {
				has_character_flag = pope_excom
				capital_county.clerical_region_title.holder ?= religious_head_or_challenger
				capital_county.clerical_region_title.holder ?= this
			}
		}
		highest_held_title_tier >= tier_kingdom
	}
	# A personal papal excommunication requires a petition at any rank.
}

petition_head_of_faith_repentance_available_trigger = {
	petition_head_of_faith_core_available_trigger = yes
	petition_head_of_faith_repentance_option_available_trigger = yes
}

petition_head_of_faith_repentance_option_available_trigger = {
	trigger_if = {
		limit = {
			NOR = {
				has_character_flag = pope_excom
				capital_county.clerical_region_title.holder ?= religious_head_or_challenger
				capital_county.clerical_region_title.holder ?= this
			}
		}
		highest_held_title_tier >= tier_kingdom
	}
	# A personal papal excommunication requires a petition at any rank.
	has_trait = excommunicated
	NOT = { has_character_modifier = recent_excommunication }
}

# Kingdom+ gate for the petition decision and setup interaction. Sub-kingdom rulers use request_excommunication_interaction instead.
petition_head_of_faith_excommunication_decision_available_trigger = {
	petition_head_of_faith_core_available_trigger = yes
	petition_head_of_faith_excommunication_option_available_trigger = yes
}

petition_head_of_faith_excommunication_option_available_trigger = {
	highest_held_title_tier >= tier_kingdom
	custom_description = {
		text = HOF_INTERACTION_UNREFORMED_FAITH
		NOT = { faith = { has_doctrine_parameter = unreformed } }
	}
}

petition_head_of_faith_any_available_trigger = {
	petition_head_of_faith_core_available_trigger = yes
	OR = {
		petition_head_of_faith_repentance_option_available_trigger = yes
		petition_head_of_faith_excommunication_option_available_trigger = yes
	}
}

# Piety reserved when preparing a petition. Waived if the petitioner holds a usable hook on the recipient.
petition_head_of_faith_can_afford_major_piety_cost_trigger = {
	OR = {
		petition_head_of_faith_has_usable_hook_on_religious_authority_trigger = yes
		piety >= major_piety_value
	}
}

petition_head_of_faith_can_afford_massive_piety_cost_trigger = {
	OR = {
		petition_head_of_faith_has_usable_hook_on_religious_authority_trigger = yes
		piety >= massive_piety_value
	}
}

petition_head_of_faith_recipient_still_valid_trigger = {
	is_alive = yes
	is_imprisoned = no
	has_character_flag = petition_head_of_faith_character_flag
	var:petition_head_of_faith_recipient ?= {
		is_alive = yes
		is_imprisoned = no
		location = default_location
	}
	petition_head_of_faith_religious_authority_is_trigger = { CHARACTER = var:petition_head_of_faith_recipient }
	faith = {
		faith_has_sacraments_central_trigger = yes
		petition_head_of_faith_faith_qualifies_trigger = yes
	}
	NOT = { is_at_war_with = var:petition_head_of_faith_recipient }
}

petition_head_of_faith_repentance_still_valid_trigger = {
	has_trait = excommunicated
	NOT = { has_character_modifier = recent_excommunication }
}

# Requires scope:secondary_recipient, and the target must share the interaction recipient
petition_head_of_faith_excommunication_target_requires_petition_trigger = {
	scope:secondary_recipient = {
		faith = { petition_head_of_faith_faith_qualifies_trigger = yes }
		petition_head_of_faith_religious_authority_is_trigger = { CHARACTER = scope:recipient }
	}
	trigger_if = {
		limit = { has_pam_dlc_trigger = yes }
		scope:secondary_recipient.highest_held_title_tier >= tier_kingdom
	}
	trigger_else = {
		scope:secondary_recipient.highest_held_title_tier >= tier_empire
	}
}

# Eligibility for an excommunication petition target.
# Evaluated in the candidate's scope; $ACTOR$ = the petitioning character.
petition_head_of_faith_excommunication_target_eligible_trigger = {
	highest_held_title_tier >= tier_kingdom
	NOT = { has_trait = excommunicated }
	age >= 12
	faith = $ACTOR$.faith
	petition_head_of_faith_religious_authority_is_trigger = {
		CHARACTER = $ACTOR$.religious_head_or_challenger
	}
	custom_tooltip = {
		text = cannot_excommunicate_antipope_protected_tt
		NOT = {
			pam_antipope_protected_from_excommunication_trigger = {
				EXCOMMUNICATOR = $ACTOR$.religious_head_or_challenger
			}
		}
	}
}

petition_head_of_faith_request_excommunication_still_valid_trigger = {
	custom_description = {
		text = HOF_INTERACTION_UNREFORMED_FAITH
		NOT = { faith = { has_doctrine_parameter = unreformed } }
	}
	var:petition_head_of_faith_target_character ?= {
		is_alive = yes
		petition_head_of_faith_excommunication_target_eligible_trigger = { ACTOR = prev }
		NOR = {
			this = prev
			this = prev.var:petition_head_of_faith_recipient
		}
	}
}

petition_head_of_faith_type_valid_trigger = {
	switch = {
		trigger = var:petition_head_of_faith_type
		flag:repentance = {
			petition_head_of_faith_repentance_still_valid_trigger = yes
		}
		flag:request_excommunication = {
			petition_head_of_faith_request_excommunication_still_valid_trigger = yes
		}
	}
}

petition_head_of_faith_valid_trigger = {
	petition_head_of_faith_recipient_still_valid_trigger = yes
	petition_head_of_faith_type_valid_trigger = yes
}

# Disposition triggers. Evaluated in PETITIONER scope.
# Uses opinion of petitioner (positive = good) and opinion of target (negative = good for petitioner)
# Requires var:petition_head_of_faith_recipient and var:petition_head_of_faith_target_character

petition_head_of_faith_disposition_favorable_trigger = {
	OR = {
		var:petition_head_of_faith_recipient = {
			opinion = {
				target = prev
				value >= petition_head_of_faith_favorable_opinion_threshold_value
			}
			OR = {
				NOT = { exists = prev.var:petition_head_of_faith_target_character }
				opinion = {
					target = prev.var:petition_head_of_faith_target_character
					value <= -10
				}
			}
		}
		# Alms and Endowments dynasty legacy: the Head of Faith looks favorably on the dynasty's petitions
		dynasty ?= { has_dynasty_perk = pam_legacy_2 }
	}
}

petition_head_of_faith_disposition_hostile_trigger = {
	OR = {
		var:petition_head_of_faith_recipient = {
			has_relation_rival = prev
		}
		AND = {
			exists = var:petition_head_of_faith_target_character
			var:petition_head_of_faith_recipient = {
				opinion = {
					target = prev.var:petition_head_of_faith_target_character
					value >= 50
				}
			}
		}
		var:petition_head_of_faith_recipient = {
			opinion = {
				target = prev
				value <= petition_head_of_faith_hostile_opinion_threshold_value
			}
		}
	}
}

petition_head_of_faith_disposition_skeptical_trigger = {
	NOR = {
		petition_head_of_faith_disposition_favorable_trigger = yes
		petition_head_of_faith_disposition_hostile_trigger = yes
	}
	OR = {
		var:petition_head_of_faith_recipient = {
			opinion = {
				target = prev
				value <= petition_head_of_faith_skeptical_opinion_threshold_value
			}
		}
		AND = {
			exists = var:petition_head_of_faith_target_character
			var:petition_head_of_faith_recipient = {
				opinion = {
					target = prev.var:petition_head_of_faith_target_character
					value >= 20
				}
			}
		}
	}
}

# Neutral is the fallback when not favorable, hostile, or skeptical.
petition_head_of_faith_disposition_neutral_trigger = {
	NOR = {
		petition_head_of_faith_disposition_favorable_trigger = yes
		petition_head_of_faith_disposition_hostile_trigger = yes
		petition_head_of_faith_disposition_skeptical_trigger = yes
	}
}

# Requires scope:petition_target and scope:recipient from petition_head_of_faith_save_scopes_effect
petition_head_of_faith_target_net_sinful_trigger = {
	exists = scope:petition_target
	exists = scope:recipient
	petition_head_of_faith_target_net_sin_value < 0
}

# Target has grave sins (VERY_SINFUL_TRAITS bar: criminal traits, murderer, despoiler). Guarantees Cite Sins success.
petition_head_of_faith_target_has_grave_sins_trigger = {
	exists = scope:petition_target
	exists = scope:recipient
	OR = {
		has_any_criminal_trait_in_rite_trigger = {
			CHARACTER = scope:petition_target
			RITE = scope:recipient.rite
		}
		scope:petition_target = {
			AND = {
				has_trait = murderer
				NOT = {
					scope:recipient.rite = {
						trait_is_virtue_rite = murderer
					}
				}
			}
		}
		# Warcraft - trait removed
		#scope:petition_target = {
		#	has_trait = despoiler_of_byzantium
		#}
	}
}

# Cite Sins is available when the target is net-sinful or has grave sins.
petition_head_of_faith_target_cite_sins_available_trigger = {
	OR = {
		petition_head_of_faith_target_net_sinful_trigger = yes
		petition_head_of_faith_target_has_grave_sins_trigger = yes
	}
}

# Penitent (scope:actor) has net negative sin value under the Head of Faith's rite.
petition_head_of_faith_actor_net_sinful_trigger = {
	exists = scope:actor
	exists = scope:recipient
	petition_head_of_faith_actor_net_sin_value < 0
}
