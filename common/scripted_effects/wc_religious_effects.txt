create_historical_holy_order_effect = {
	save_temporary_scope_as = leader
	
	$COUNTY$ = {
		random_county_province = {
			limit = { barony = { barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = $SPONSOR$ } } }
			barony = {
				save_temporary_scope_as = barony
				if = {
					limit = { NOT = { holder = $SPONSOR$ } }
					create_title_and_vassal_change = {
						type = leased_out
						save_scope_as = change
						add_claim_on_loss = no
					}
					change_title_holder_include_vassals = {
						holder = $SPONSOR$
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
				}
			}
		}
	}
	if = {
		limit = { exists = scope:barony }
		$SPONSOR$ = {
			create_holy_order = {
				leader = scope:leader
				capital = scope:barony
				save_scope_as = new_holy_order
			}
			wc_create_holy_order_with_gold_effect = { LEADER = scope:leader }
		}
		scope:new_holy_order.title = {
			# Figure out later when we get CoA
			# set_coa = title:$ORDER$
			set_title_name = $ORDER$
			set_title_prefix = $ORDER$_article
		}
	}
}
# we need this below if because root is not valid in vanilla's one
wc_create_holy_order_with_gold_effect = {
	create_holy_order_effect = yes

	# Warcraft
	$LEADER$ = {
		add_gold = 100 #So that they have some money to lend out
		add_piety_level = 2
		add_gold = holy_order_starting_gold
		every_courtier = {
			add_trait = order_member
		}
	}
}
historical_lease_out_effect = {
	$COUNTY$ = {
		random_county_province = {
			limit = { barony = { barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = $COUNTY$.holder } } }
			barony = {
				if = {
					limit = { NOT = { holder = $COUNTY$.holder } }
					create_title_and_vassal_change = {
						type = leased_out
						save_scope_as = change
						add_claim_on_loss = no
					}
					change_title_holder_include_vassals = {
						holder = $COUNTY$.holder
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
				}
				lease_out_to = $ORDER$
			}
		}
	}
}

# Founds Tyr's Watch at Bandarion Keep, led by Danitha Morr. Root: the sponsor,
# Garek Bandarion's liege at death. Called from forsaken_kingdom.0001, which has
# already checked that the barony can be leased and that Danitha can lead.
# scope:leader is deliberately not saved: create_holy_order_effect would then pay
# the leader a second time on top of wc_create_holy_order_with_gold_effect.
wc_found_tyrs_watch_effect = {
	if = {
		limit = { title:b_autumnacre.holder != root }
		create_title_and_vassal_change = {
			type = leased_out
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:b_autumnacre = {
			change_title_holder_include_vassals = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	create_holy_order = {
		leader = character:danitha_morr
		capital = title:b_autumnacre
		save_scope_as = new_holy_order
	}
	# Rename first and save the barony, so vanilla's founding messages read
	# "Tyr's Watch ... established in Bandarion Keep" instead of a generated name
	scope:new_holy_order.title = {
		set_title_name = holy_order_tyrs_watch
		set_title_prefix = holy_order_tyrs_watch_article
	}
	title:b_autumnacre = { save_temporary_scope_as = barony }
	wc_create_holy_order_with_gold_effect = { LEADER = character:danitha_morr }
}
