# called for when a casus belli resolves in one of the following ways.
# same scopes are available in these events as in the CBs themselves, this just fires for all CBs instead of a specific CB.
# note that any events/effects fired here WILL NOT show up in the war summary tooltip; they fire at the same time as the war resolution, but are not actually part of the war resolution itself.

wc_on_war_won_attacker = {
	effect = {
		# Warcraft - Path of the Damned: Digging up the Dead
		if = {
			limit = {
				scope:war = { using_cb = artifact_war }

				scope:attacker = {
					any_owned_story = {
						type = story_path_of_the_damned
						var:story_stage = 2 # Digging up the Dead stage
					}
					NOT = { has_title = title:h_scourge } # Not the Lich King
					NOT = { has_character_flag = post_digging_effect_fired_flag } # Not already done
				}
			}

			scope:attacker = { post_digging_up_the_dead_effect = yes }
		}
	}
}
