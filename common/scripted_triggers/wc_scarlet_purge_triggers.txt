# A local living authority; neither rank nor religious office exempts subjects.
wc_scarlet_purge_authority_trigger = {
	is_alive = yes
	is_adult = yes
	is_landed = yes
	is_imprisoned = no
	is_incapable_trigger = no
	has_faith = faith:scarletism
	NOT = { has_trait = being_undead }
	wc_scarlet_maa_member_trigger = yes
	wc_inquisition_court_available_trigger = yes
}

# Subject scope. Only a public trait is evidence; never infer hidden identities.
wc_scarlet_purge_target_trigger = {
	is_alive = yes
	has_trait = being_undead
	NOT = { has_variable = wc_inquisition_stage }
	NOT = { has_variable = wc_scarlet_purge_summons }
	NOT = { has_character_flag = wc_scarlet_purge_cooldown }
	NOT = { has_character_flag = wc_inquisition_cooldown }
	OR = { is_imprisoned = no is_imprisoned_by = $AUTHORITY$ }
	OR = {
		liege ?= $AUTHORITY$
		is_courtier_of = $AUTHORITY$
		is_imprisoned_by = $AUTHORITY$
	}
}

wc_scarlet_purge_summons_valid_trigger = {
	exists = scope:wc_purge_authority
	scope:wc_purge_authority = { wc_scarlet_purge_authority_trigger = yes }
	is_alive = yes
	has_trait = being_undead
	NOT = { has_variable = wc_inquisition_stage }
	var:wc_scarlet_purge_summons ?= scope:wc_purge_authority
	exists = scope:wc_purge_generation
	var:wc_scarlet_purge_generation = scope:wc_purge_generation
	OR = { is_imprisoned = no is_imprisoned_by = scope:wc_purge_authority }
	OR = {
		liege ?= scope:wc_purge_authority
		is_courtier_of = scope:wc_purge_authority
		is_imprisoned_by = scope:wc_purge_authority
	}
}
