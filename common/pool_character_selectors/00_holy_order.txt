	valid_character = {
		is_valid_auto_title_holder = yes
		is_valid_auto_title_holder_clergy = yes
	}
	
	character_score = {
		base = 1
		
		modifier = {
			add = 1000
			is_courtier_of = scope:base
		}
		compare_modifier = {
			value = martial
		}
		modifier = {
			dynasty.dynasty_prestige_level >= 2
			add = {
				value = dynasty.dynasty_prestige_level
				multiply = 20
				subtract = 20
			}
		}
		modifier = {
		piety_level >= 2
			add = {
				value = piety_level
				multiply = 5
			}
		}
		modifier = {
			add = 100
			has_same_culture_as = scope:base
		}
		modifier = {
			add = 15
			culture = { has_same_culture_heritage = scope:base.culture }
		}
		modifier = {
			add = 10
			culture = { has_same_culture_language = scope:base.culture }
		}
		modifier = {
			add = 5
			culture = { has_same_culture_ethos = scope:base.culture }
		}
		modifier = {
			add = 50
			age_in_relation_to_humans_more_than_trigger = { age = 35 }
		}
		modifier = {
			add = 100
			scope:base.rite = { rite_has_preferred_gender_for_clergy = root }
		}
	}
	
	config = {
		background = commander
		age = { 25 50 }
	}
	
	selection_count = 25
	
	pool = faith
	gender = clergy
	court = yes
}
