# Mantid Swarming cycle. Events: events/wc_events/wc_mantid_swarm_events.txt

on_game_start = { on_actions = { wc_swarm_game_start_on_action } }
yearly_global_pulse = { on_actions = { wc_swarm_yearly_on_action } }
on_war_won_attacker = { on_actions = { wc_swarm_war_won_attacker_on_action } }
on_war_won_defender = { on_actions = { wc_swarm_war_won_defender_on_action } }
on_war_invalidated = { on_actions = { wc_swarm_war_invalidated_on_action } }
on_great_holy_war_invalidation = { on_actions = { wc_swarm_ghw_invalidated_on_action } }

wc_swarm_game_start_on_action = {
	effect = {
		wc_swarm_init_clock_effect = yes
	}
}

# No root. Order: init, fail-safe, Swarm Stirs, launch.
wc_swarm_yearly_on_action = {
	effect = {
		wc_swarm_init_clock_effect = yes

		# Fail-safe: a Swarm whose war ended without any hook catching it
		if = {
			limit = {
				has_global_variable = wc_swarm_active
				faith:yshaarjism = {
					NOT = { exists = great_holy_war }
				}
			}
			wc_swarm_cleanup_effect = yes
		}

		# The Swarm Stirs, one year ahead
		if = {
			limit = {
				exists = title:d_seven_prophets.holder
				current_year = wc_swarm_stir_year_value
			}
			wc_swarm_save_scopes_effect = yes
			scope:swarm_head = { trigger_event = wc_mantid_swarm.0001 }
			if = {
				limit = {
					exists = scope:empress
					NOT = { scope:empress = scope:swarm_head }
				}
				scope:empress = { trigger_event = wc_mantid_swarm.0001 }
			}
		}

		# Launch
		if = {
			limit = {
				current_year >= global_var:wc_next_swarming_year
				NOT = { has_global_variable = wc_swarm_decision_pending }
				wc_swarm_can_launch_trigger = yes
			}
			title:d_seven_prophets.holder = {
				if = {
					limit = { is_ai = yes }
					trigger_event = wc_mantid_swarm.0010
				}
				else = {
					set_global_variable = {
						name = wc_swarm_decision_pending
						value = yes
						years = 1
					}
					trigger_event = wc_mantid_swarm.0011
				}
			}
		}
	}
}

# War-end hooks: logic stays in effect, which runs the same tick, while scope:war still exists.
wc_swarm_war_won_attacker_on_action = {
	trigger = { wc_swarm_is_swarm_war_trigger = yes }
	effect = {
		wc_swarm_record_war_end_effect = { OUTCOME = attacker_won }
	}
}

wc_swarm_war_won_defender_on_action = {
	trigger = { wc_swarm_is_swarm_war_trigger = yes }
	effect = {
		wc_swarm_record_war_end_effect = { OUTCOME = defender_won }
	}
}

# Mid-war invalidation (e.g. the defender converted): cleanup only, no aftermath.
wc_swarm_war_invalidated_on_action = {
	trigger = { wc_swarm_is_swarm_war_trigger = yes }
	effect = {
		wc_swarm_cleanup_effect = yes
	}
}

# Root is the great holy war, about to be destroyed before its war started.
wc_swarm_ghw_invalidated_on_action = {
	trigger = {
		has_global_variable = wc_swarm_active
		faith = faith:yshaarjism
	}
	effect = {
		wc_swarm_cleanup_effect = yes
	}
}
