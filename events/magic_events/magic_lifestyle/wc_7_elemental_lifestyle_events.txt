# Playing with Fire - Your fire elementals are setting crops on fire
elemental_magic_lifestyle.2011 = { 
	title = elemental_magic_lifestyle.2011.t
	desc = elemental_magic_lifestyle.2011.desc
	content_source = dlc_GOA
	theme = elemental_fire

	left_portrait = { 
		character = root
		animation = steward
	}

	right_portrait = { 
		character = scope:pleading_peasant
		animation = beg
	}

	trigger = { 
		is_landed = yes
	}

	immediate = { 
		create_character = {
			location = root.capital_province
			template = generic_peasant_character
			save_scope_as = pleading_peasant
		}
	}

	# I will rein in my elemental servants
	option = { 
		name = elemental_magic_lifestyle.2011.opt.a
		add_character_modifier = { 
			modifier = reined_in_fire_elementals
			years = 3
		} 
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_fire VALUE = 10 CHANGE = add }
		ai_chance = { 
			base = 25
			ai_value_modifier = {
				ai_compassion = 5
			}
		}
	}

	# My elementals shall go where they please
	option = { 
		name = elemental_magic_lifestyle.2011.opt.b
		add_character_modifier = {
			modifier = liberated_fire_elementals
			years = 3
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_boldness = 1
				ai_zeal = 1
				ai_greed = 1
			}
		}
	}

	# You dare question the spirits' will? You shall be burned alive!
	option = { 
		name = elemental_magic_lifestyle.2011.opt.c
		add_dread = minor_dread_gain
		add_piety = miniscule_piety_gain
		scope:pleading_peasant = {
			death = { 
				death_reason = death_burned_witch
				killer = root
			}
		}
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_fire VALUE = 10 CHANGE = subtract }
		capital_county = {
			add_county_modifier = {
				modifier = intrigue_arbitrary_justice_modifier
				years = 2
			}
		}
		ai_chance = { 
			base = 25
			modifier = { 
				add = 200
				OR = { 
					has_trait = callous has_trait = sadistic has_trait = arbitrary 
					religion = { has_doctrine = special_doctrine_end_justifies_means }
					has_dark_government_trigger = yes
				}
			}
		}
		stress_impact = {
			just = minor_stress_gain
			compassionate = minor_stress_gain
			forgiving = minor_stress_gain
			sadistic = minor_stress_loss
			arbitrary = minor_stress_loss
			wrathful = minor_stress_loss
		}
	}

	option = {
		name = elemental_magic_lifestyle.2011.opt.d
		trigger = {
			is_known_spell_trigger = { SPELL = flag:summon_water_elemental }
			has_sufficient_mana_value_trigger = { VALUE = wc_spell_summon_water_elemental_cost_mana_value }
		}
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_water VALUE = 10 CHANGE = add }
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_earth VALUE = 10 CHANGE = add }
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_air VALUE = 10 CHANGE = add }
		wc_change_mana_by_value_effect = {
			CHANGE = subtract
			VALUE = wc_spell_summon_water_elemental_cost_mana_value
		}
		stress_impact = {
			arbitrary = miniscule_stress_gain
			arrogant = minor_stress_gain
			just = minor_stress_loss
		}
	}

	after = { 
		hidden_effect = {
			scope:pleading_peasant = {
				trigger_event = { 
					id = harm.9601
					days = 5
				}
			}
		}
	}
}

# Totally Stoked - Blacksmiths ask you to help improve their forges
elemental_magic_lifestyle.2021 = {
	title = elemental_magic_lifestyle.2021.t
	desc = {
		first_valid = { 
			triggered_desc = {
				trigger = { is_landless_adventurer = no }
				desc = elemental_magic_lifestyle.2021.desc.landholder
			}
			desc = elemental_magic_lifestyle.2021.desc.adventurer
		}
	}
	content_source = dlc_GOA
	theme = elemental_fire

	left_portrait = { 
		character = root
		animation = steward
	}

	right_portrait = { 
		character = scope:blacksmith
		animation = inspect_weapon
	}

	trigger = { 
		is_landed = yes
	}

	immediate = { 
		create_character = {
			gender_female_chance = root_faith_dominant_gender_female_chance
			culture = root.culture
			faith = root.faith
			trait = diligent #it has a blacksmith's hammer!
			random_traits = yes
			dynasty = none
			employer = root
			save_scope_as = blacksmith
		}
		scope:blacksmith = {
			hidden_effect = {
				trigger_race_giving_no_gene_effect = yes			#Assigns race trait
			}
		}
		set_variable = {
			name = current_spell_rank
			value = 2
		}
	}

	# A fantastic idea! Take my servant, calcifer
	option = {
		trigger = {
			is_known_spell_trigger = { SPELL = flag:summon_fire_elemental }
			has_sufficient_mana_value_trigger = { VALUE = wc_spell_summon_fire_elemental_cost_mana_value }
		}
		name = elemental_magic_lifestyle.2021.opt.a
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_fire VALUE = 10 CHANGE = subtract }
		wc_change_mana_by_value_effect = {
			CHANGE = subtract
			VALUE = wc_spell_summon_fire_elemental_cost_mana_value
		}
		add_character_modifier = {
			modifier = fire_magic_improved_smithing
			years = 2
		}
		stress_impact = {
			greedy = miniscule_stress_gain
			arrogant = minor_stress_gain
			generous = minor_stress_loss
		}
	}

	# I shall personally assist, maybe I could learn something
	option = { 
		add_fire_elemental_magic_xp = 50
		random_list = { 
			45 = { 
				send_interface_toast = {
					title = elemental_magic_lifestyle.2021.opt.b.learn_nothing
					stress_impact = {
						base = minor_stress_gain
						arrogant = medium_stress_gain
					}
				}
			}
			50 = { 
				send_interface_toast = {
					title = elemental_magic_lifestyle.2021.opt.b.learn_something
					add_stewardship_lifestyle_xp = 100
				}
			}
			5 = { 
				modifier = { 
					add = 25
					OR = { has_trait = education_learning has_trait = education_stewardship has_trait = intellect_good has_trait = gregarious }
				}
				modifier = { 
					add = 10
					OR = { has_trait = diligent has_trait = stubborn has_trait = patient }
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2021.opt.b.learn_alot
					add_stewardship_lifestyle_perk_points = 1
				}
			}
		}
		add_character_modifier = { 
			modifier = elemental_assisted_smithing
			years = 2
		}
		stress_impact = {
			lazy = medium_stress_gain
			arbitrary = minor_stress_gain
			impatient = minor_stress_gain
			shy = minor_stress_gain
		}
		name = elemental_magic_lifestyle.2021.opt.b
	}

	# I will not debase the elements with this busywork
	option = { 
		name = elemental_magic_lifestyle.2021.opt.c
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_fire VALUE = 10 CHANGE = add }
		stress_impact = {
			generous = minor_stress_gain
			gregarious = minor_stress_loss
			lazy = minor_stress_loss
			arrogant = minor_stress_loss
		}
	}

	after = { 
		hidden_effect = {
			scope:blacksmith = {
				trigger_event = { 
					id = harm.9601
					days = 5
				}
			}
		}
	}
	remove_variable = current_spell_rank
}

# Fighting Fire with Fire Part 1 - Subjects ask you to use your fire elemental magic to tame a raging forest fire
elemental_magic_lifestyle.2031 = { # todo: see if we can find a way to get this to fire only in summer?
	title = elemental_magic_lifestyle.2031.t
	desc = {
		first_valid = { 
			triggered_desc = {
				trigger = { is_landless_adventurer = no }
				desc = elemental_magic_lifestyle.2031.desc.landholder
			}
			desc = elemental_magic_lifestyle.2031.desc.adventurer
		}
	}
	content_source = dlc_GOA
	theme = elemental_fire

	left_portrait = {
		character = root
		animation = stress
	}

	right_portrait = { 
		character = scope:steward_character
		animation = worry
	}

	trigger = { 
		any_realm_province = { 
			OR = { 
				terrain = forest
				terrain = mountain
			}
		}
	}

	immediate = {
		every_realm_province = { 
			limit = { 
				OR = { 
					terrain = forest
					terrain = mountain
				}
			}
			add_to_list = forest_fire_candidates
		}
		if = { 
			limit = { 
				OR = { 
					is_landless_adventurer = yes
					is_landless_administrative = yes
				}
			}
			domicile.domicile_location = { 
				add_to_list = forest_fire_candidates
			}
		}
		random_in_list = { 
			list = forest_fire_candidates
			add_province_modifier = {
				modifier = wc_forest_fires_province_modifier
				years = 2
			} 
			save_scope_as = affected_province
		}
		if = { 
			limit = { exists = cp:councillor_steward }
			cp:councillor_steward = {
				save_scope_as = steward_character
			}
		} else = { 
			random_courtier = {
				save_scope_as = steward_character
			}
		}
	}

	# i will deal with the forest fires using my elemental magic
	option = { 
		name = elemental_magic_lifestyle.2031.opt.a
		trigger_event = elemental_magic_lifestyle.2032
	}

	# help my subjects using conventional means
	option = { 
		name = elemental_magic_lifestyle.2031.opt.b
		scope:affected_province = {
			remove_province_modifier = wc_forest_fires_province_modifier
			add_province_modifier = {
				modifier = wc_mitigated_forest_fires_province_modifier
				years = 2
			}
		}
		stress_impact = {
			compassionate = miniscule_stress_impact_loss
			brave = minor_stress_impact_loss
			paranoid = miniscule_stress_impact_loss
			greedy = medium_stress_impact_gain
			callous = medium_stress_impact_gain
			arbitrary = medium_stress_impact_gain
		}
	}

	# do nothing
	option = { 
		name = elemental_magic_lifestyle.2031.opt.c
		random = { 
			50 = { 
				send_interface_toast = {
					title = elemental_magic_lifestyle.2031.toast.fires_worsen
					scope:affected_province = {
						remove_province_modifier = wc_forest_fires_province_modifier
						add_province_modifier = {
							modifier = wc_devastating_forest_fires_province_modifier
							years = 2
						}
					}
				}
			}
		}
		stress_impact = {
			arbitrary = miniscule_stress_impact_loss
			callous = minor_stress_impact_loss
			greedy = miniscule_stress_impact_loss
			paranoid = medium_stress_impact_gain
			brave = medium_stress_impact_gain
			compassionate = medium_stress_impact_gain
		}
	}
}

elemental_magic_lifestyle.2032 = { # todo: see if we can find a way to get this to fire only in summer?
	title = elemental_magic_lifestyle.2032.t
	desc = elemental_magic_lifestyle.2032.desc
	content_source = dlc_GOA
	theme = elemental_fire

	left_portrait = { 
		character = root
		animation = prayer
	}

	# persuade the fire elementals to leave
	option = { 
		name = elemental_magic_lifestyle.2032.opt.a
		duel ={ 
			skill = diplomacy
			value = decent_skill_rating
			# Success
			35 = { 
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2
					min = -49
				}
				modifier = {
					add = { 
						value = wc_elemental_fire_favour_current_value
						multiply = 0.5
					}
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.commune_success
					wc_change_amount_element_favour_effect = { ELEMENT = elemental_fire VALUE = 25 CHANGE = add }
					add_prestige = major_prestige_gain
					add_piety = major_piety_gain
					wc_clear_forest_fires_effect = yes
				}
			}
			# Failure
			65 = { 
				compare_modifier = {
					value = scope:duel_value
					multiplier = -2
					min = -49
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.commune_failure
				}
				stress_impact = {
					base = minor_stress_impact_gain
				}
			}
		}
		stress_impact = { 
			shy = minor_stress_impact_gain
			arbitrary = minor_stress_impact_gain
			gregarious = minor_stress_impact_loss
		}
	}

	# enslave the fire elementals
	option = { 
		name = elemental_magic_lifestyle.2032.opt.b
		wc_change_amount_element_favour_effect = { ELEMENT = elemental_fire VALUE = 20 CHANGE = subtract }
		spawn_army = {
            uses_supply = no
            inheritable = yes
            name = elemental_fire_slave_army
            men_at_arms = {
                type = fire_terrestrial
                stacks = 3
            }
            location = root.capital_province
        }
		wc_clear_forest_fires_effect = yes
		stress_impact = {
			just = minor_stress_gain
			compassionate = minor_stress_gain
			forgiving = minor_stress_gain
		}
	}

	# banish the elementals
	option = { 
		show_as_unavailable = { 
			always = yes
		}
		trigger = { 
			has_sufficient_mana_value_trigger = { VALUE = wc_spell_summon_fire_elemental_cost_mana_value }
		}
		name = elemental_magic_lifestyle.2032.opt.c
		wc_change_mana_by_value_effect = {
			CHANGE = subtract
			VALUE = 50
		}
		wc_clear_forest_fires_effect = yes
		add_elemental_fire_magic_lifestyle_xp = major_lifestyle_experience
	}

	# balance the elements
	option = { 
		trigger = { 
			OR = { 
				has_trait_with_flag = wc_lifestyle_elemental_water_magic_flag
				has_trait_with_flag = wc_lifestyle_elemental_earth_magic_flag
				has_trait_with_flag = wc_lifestyle_elemental_air_magic_flag
			}
		}
		name = elemental_magic_lifestyle.2032.opt.d
		add_character_modifier = {
			modifier = wc_lifestyle_balanced_the_elements_modifier
			years = 2
		}
		wc_clear_forest_fires_effect = yes
		stress_impact = {
			cynical = minor_stress_impact_gain
			arbitrary = minor_stress_impact_gain
			humble = minor_stress_impact_loss
			just = minor_stress_impact_loss
		}
	}
}
