wc_adventurer_potential_destination_trigger = {
	exists = location
	location = { NOT = { geographical_region = $REGION$ } }
	adventurer_potential_destination_distance_check = { REGION = $REGION$ }
}

wc_is_cold_climate_trigger = { 
	NOT = { is_desert_trigger = yes }
	NOT = { terrain = jungle }
	NOT = {
		geographical_region = world_northrend_sholazar_basin
	}
	OR = { 
		geographical_region = world_northrend
		geographical_region = world_kalimdor_north_winterspring
		geographical_region = world_pandaria_kun_lai_summit
	}
}
