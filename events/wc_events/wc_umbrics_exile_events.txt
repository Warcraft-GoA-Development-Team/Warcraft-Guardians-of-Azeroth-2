namespace = umbrics_exile

umbrics_exile.1 = {
    scope = none
    hidden = yes

    immediate = {
        hidden_effect = {
            add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:exile_umbric
            }

            if = {
                limit = { any_living_character = { is_umbric_trigger = yes } }
                character:magister_umbric = { save_scope_as = umbric }
            }

            scope:umbric = {
                if = {
                    limit = { is_imprisoned = yes }
                    release_from_prison = yes
                }

                become_an_adventurer = yes

                set_character_faith = faith:voidfarer
            }

            every_player = { trigger_event = { id = umbrics_exile.2 } }
        }
    }
}

umbrics_exile.2 = {
    type = character_event
    content_source = dlc_GOA
	title = umbrics_exile.2.t
	desc = umbrics_exile.2.desc
    theme = realm
    override_background = { reference = silvermoon }

    trigger = { is_ai = no }

    on_trigger_fail = {
        scope:umbric = {
            trigger_event = {
                id = umbrics_exile.3
                days = 1
            }
        }
    }

    left_portrait = {
        character = scope:umbric
        animation = writing
    }

    option = {
        name = umbrics_exile.2.a
        add_internal_flag = dangerous
        set_player_character = scope:umbric

        custom_tooltip = umbrics_exile.2.a.tt

        scope:umbric = {
            trigger_event = {
                id = umbrics_exile.3
                days = 1
            }
        }
    }
    option = {
        name = umbrics_exile.2.a

        scope:umbric = {
            trigger_event = {
                id = umbrics_exile.3
                days = 1
            }
        }
    }
}
