# Balnazzar - Dreadlord personal story
# Only fires when the player picks to play as character:430400 in the Frozen Throne
# bookmark (bm_tft_lordaeron, 605.6.6) - see common/on_action/game_start.txt.
# story_stage advances inline in each narrative event (events/story_cycles/
# wc_story_cycle_balnazzar_events.txt), not through effect_groups here - the
# chain is decision/event-driven, not a periodic pulse.

story_balnazzar_dreadlord = {
	visible = yes

	icon = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/icons/traits/creature_nathrezim.dds" # Story icon - nathrezim disguise trait now uses a mundane icon instead
	}

	background = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/illustrations/event_scenes/wc_undercity_throne_room.dds" # Where it all starts, .1000-.1002
	}

	visualization = {
		custom_string_key = "BALNAZZAR_DREADLORD_CUSTOM_STRING_KEY"
	}

	on_setup = {
		set_global_variable = { # Anchor, so the world can reference this playthrough's story
			name = balnazzar_dreadlord_story
			value = this
		}

		set_variable = {
			name = story_stage
			value = 1
		}
	}

	on_end = {
		remove_global_variable = balnazzar_dreadlord_story
	}

	on_owner_death = {
		# TODO: decide what happens if the player-Balnazzar dies mid-story
		end_story = yes
	}

	# CK3's age-related health traits (clouded_eyes, infirm, ...) trigger on
	# chronological age alone, ignoring life_expectancy entirely - even for
	# genuine nathrezim elsewhere in the mod. Keep stripping them from whoever
	# wears the disguise, instead of patching the shared vanilla health system.
	effect_group = {
		days = { 30 60 }

		trigger = {
			story_owner = { has_trait = nathrezim_disguise }
		}

		triggered_effect = {
			effect = {
				story_owner = {
					remove_trait = clouded_eyes
					remove_trait = faltering_heart
					remove_trait = fragile_bones
					remove_trait = infirm
					remove_trait = withering_mind
				}
			}
		}
	}

	# Trigger 1 of 3 for wc_balnazzar_story.2000 (see the design spec,
	# docs/superpowers/specs/2026-08-04-balnazzar-mask-ending-design.md) -
	# checks yearly once the story owner is aging with no legitimate heir,
	# and offers the .0100 confirmation event instead of letting a silent
	# death trigger a random succession crisis. balnazzar_shell_failing_notified
	# makes this one-shot - it should offer once, not nag every year.
	effect_group = {
		days = { 365 365 }

		trigger = {
			story_owner = {
				has_trait = nathrezim_disguise
				age >= 70
				NOT = { exists = primary_heir }
				NOT = { has_character_flag = balnazzar_shed_the_mask }
				NOT = { has_character_flag = balnazzar_shell_failing_notified }
			}
		}

		triggered_effect = {
			effect = {
				story_owner = {
					add_character_flag = balnazzar_shell_failing_notified
					trigger_event = {
						id = wc_balnazzar_story.0100
						days = { 5 15 }
					}
				}
			}
		}
	}
}
