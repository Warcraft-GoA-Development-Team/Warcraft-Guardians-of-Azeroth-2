invade_gnomeregan_decision = {
	title = {
		first_valid = {
			desc = invade_gnomeregan_decision
		}
	}
	desc = {
		first_valid = {
			desc = invade_gnomeregan_decision.desc
		}
	}
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 12
		empire = 0
		hegemony = 0
	}
	decision_group_type = major
	picture = { reference = "gfx/interface/illustrations/decisions/gnomeregan.dds" }
	selection_tooltip = invade_gnomeregan_decision_tt

	is_shown = {
		ROOT = {
			has_trait = creature_trogg
			is_landed = yes
		}

		trigger_if = {
			limit = {
				is_ai = yes
			}
			current_year > 603
		}
		trigger_else = {
			current_year > 602
		}

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_gnomeregan_decision
			}
		}

		game_start_date < 604.9.15
	}
	

	cost = {
		gold = { 
			value = 50
			if = { 
				limit = { is_ai = yes }
				multiply = 0
			}
		}
		prestige = { 
			value = 100
			if = { 
				limit = { is_ai = yes }
				multiply = 0
			}
		}
	}
	
	is_valid = {
		title:k_gnomeregan.holder = {
			has_trait = creature_gnome
		}
	}

	effect = {
		custom_tooltip = invade_gnomeregan_decision_effect.tt

		trigger_event = trogg_invasion.0001

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_gnomeregan_decision
			}
		}
		add_character_flag = gnomeregan_despoiler
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 15

		modifier = {
			add = -50
			title:k_gnomeregan.holder = {
				is_allied_to = title:k_ironforge.holder
			}
		}

		modifier = {
			add = 10
			title:k_gnomeregan.holder = {
				is_at_war = yes
			}
		}

		modifier = {
			add = 75
			title:k_gnomeregan.holder = {
				OR = {
					is_at_war_with = title:k_lordaeron.holder
					is_at_war_with = title:k_lordaeron.holder.liege
				}
			}
		}
	}
}

spread_radiation_decision = {
	title = {
		first_valid = {
			desc = spread_radiation_decision
		}
	}
	desc = {
		first_valid = {
			desc = spread_radiation_decision.desc
		}
	}
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 12
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}
	decision_group_type = major
	picture = { reference = "gfx/interface/illustrations/decisions/gnomeregan.dds" }
	selection_tooltip = spread_radiation_decision_tt

	is_shown = {
		has_trait = creature_gnome
		has_trait = irradiated
		is_landed = yes
	}
	
	is_valid = {
		OR = {
			title:c_gnomeregan.holder = ROOT
			liege = { 
				has_trait = creature_gnome
				has_trait = irradiated
				title:c_gnomeregan.holder = this
			}
		}
		has_sufficient_mana_value_trigger = { VALUE = massive_mana_value }
	}
	
	effect = {
		custom_tooltip = spread_radiation_decision_effect.tt

		wc_change_mana_by_value_effect = {
			CHANGE = subtract
			VALUE = massive_mana_value
		}

		hidden_effect = {
			every_courtier_or_guest = {
				limit = {
					NOT = {
						has_trait = being_undead
					}
				}
				add_trait = irradiated
				set_culture = culture:leper_gnome
				set_character_faith = faith:glowbound
			}

			every_prisoner = {
				limit = {
					NOT = {
						has_trait = being_undead
					}
				}
				add_trait = irradiated
				set_culture = culture:leper_gnome
				set_character_faith = faith:glowbound		
			}

			if = {
				limit = {
					any_realm_county = {
						OR = {
							has_county_modifier = wc_high_debris_damage_modifier
							has_county_modifier = wc_medium_debris_damage_modifier
							has_county_modifier = wc_small_debris_damage_modifier 
						}				
					}
				}

				every_realm_county = {
					remove_all_county_modifier_instances = wc_high_debris_damage_modifier
					remove_all_county_modifier_instances = wc_medium_debris_damage_modifier
					remove_all_county_modifier_instances = wc_small_debris_damage_modifier
				}
			}
			add_character_flag = gnomeregan_despoiler
		}
	}


	cost = {
		prestige = { 
			value = 75

            if = { # Free for AI
				limit = {
					is_ai = yes
				}
				multiply = 0
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}

	cooldown = { 
		years = 3
	}
}
