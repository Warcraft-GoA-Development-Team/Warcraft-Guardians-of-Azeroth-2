learn_language_success = {
	random_events = {
		100 = learn_language_outcome.1000
	}
	effect = {
		notify_puppeteer_of_scheme_outcome_effect = { OWNER = scope:owner TARGET = scope:target TYPE = event_toast_effect_good TITLE = scheme_outcome_learn_language_success_puppeteer }
		# Warcraft
		scope:owner = {
            ascension_checker_effect = { SOURCE = scheme }
        }
	}
}
