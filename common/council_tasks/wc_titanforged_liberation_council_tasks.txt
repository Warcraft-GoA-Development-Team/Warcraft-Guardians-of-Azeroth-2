task_reorder_county = {
	position = councillor_marshal

	task_type = task_type_county
	county_target = realm
	ai_county_target = realm
	task_progress = task_progress_value
	task_current_value = scope:councillor_liege.wc_titanforged_reorder_county_current_progress
	task_max_value = wc_titanforged_reorder_county_threshold
	highlight_own_realm = yes
	restart_on_finish = no

	asset = {
		icon = "gfx/interface/icons/council_task_types/task_increase_control.dds"
		background = "gfx/interface/buttons/button_round_bg.dds"
		frame = "gfx/interface/buttons/button_round_frame.dds"
		glow = "gfx/particles/halo.dds"
	}

	effect_desc = {
		desc = task_reorder_county_desc
	}

	potential_county = {
		scope:county = {
			faith = { is_chaos_aligned_faith_trigger = yes }
		}
		scope:councillor_liege = {
			faith = { is_titanforged_liberator_faith_trigger = yes }
		}
		NOT = {
			scope:county = {
				this = title:c_lordaeron
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:liberate_lordaeron_from_the_scourge_decision
				}
			}
		}
	}

	progress = {
		value = 0
		add = {
			value = wc_titanforged_reorder_county_base
			desc = WC_TITANFORGED_REORDER_BASE
		}
		add = {
			value = wc_titanforged_reorder_county_skill_bonus
			desc = WC_TITANFORGED_REORDER_SKILL_BONUS
		}
		if = {
			limit = {
				scope:councillor_liege = {
					has_relation_friend = scope:councillor
					NOT = { has_relation_best_friend = scope:councillor }
				}
			}
			add = {
				value = wc_titanforged_reorder_county_friend_bonus
				desc = WC_TITANFORGED_REORDER_FRIEND_BONUS
			}
		}
		if = {
			limit = {
				scope:councillor_liege = { has_relation_best_friend = scope:councillor }
			}
			add = {
				value = wc_titanforged_reorder_county_best_friend_bonus
				desc = WC_TITANFORGED_REORDER_BEST_FRIEND_BONUS
			}
		}
		if = {
			limit = {
				scope:councillor_liege = {
					has_relation_rival = scope:councillor
					NOT = { has_relation_nemesis = scope:councillor }
				}
			}
			add = {
				value = wc_titanforged_reorder_county_rival_bonus
				desc = WC_TITANFORGED_REORDER_RIVAL_BONUS
			}
		}
		if = {
			limit = {
				scope:councillor_liege = { has_relation_nemesis = scope:councillor }
			}
			add = {
				value = wc_titanforged_reorder_county_nemesis_bonus
				desc = WC_TITANFORGED_REORDER_NEMESIS_BONUS
			}
		}
		if = {
			limit = {
				scope:county = { culture = scope:councillor_liege.culture }
			}
			add = {
				value = wc_titanforged_reorder_county_same_culture_bonus
				desc = WC_TITANFORGED_REORDER_SAME_CULTURE_BONUS
			}
		}
		if = {
			limit = {
				scope:county = { faith = scope:councillor_liege.faith }
			}
			add = {
				value = wc_titanforged_reorder_county_same_faith_bonus
				desc = WC_TITANFORGED_REORDER_SAME_FAITH_BONUS
			}
		}
		if = {
			limit = { exists = scope:county }
			add = {
				value = wc_titanforged_reorder_county_development_penalty
				desc = WC_TITANFORGED_REORDER_DEVELOPMENT_PENALTY
			}
		}
		if = {
			limit = {
				scope:councillor_liege = { has_raised_armies = no }
			}
			add = {
				value = wc_titanforged_reorder_county_unraised_army_bonus
				desc = WC_TITANFORGED_REORDER_UNRAISED_ARMY_BONUS
			}
		}
	}

	on_start_task_county = {
		scope:county = {
			add_county_modifier = wc_titanforged_ongoing_reordering_county_modifier
		}
	}

	on_cancel_task_county = {
		scope:county = {
			remove_county_modifier = wc_titanforged_ongoing_reordering_county_modifier
			remove_variable = wc_titanforged_reorder_current_progress
		}
	}

	on_finish_task_county = {
		scope:county = {
			remove_county_modifier = wc_titanforged_ongoing_reordering_county_modifier
			remove_variable = wc_titanforged_reorder_current_progress
			reforge_liberated_county_effect = yes
			add_county_modifier = wc_titanforged_reordered_county_modifier
		}
		scope:councillor_liege = {
			trigger_event = wc_titanforged_reorder_task.0001
		}
	}

	on_monthly_county = {
		if = {
			limit = {
				NOT = {
					scope:county ?= {
						has_variable = wc_titanforged_reorder_current_progress
					}
				}
			}
			scope:county = {
				set_variable = {
					name = wc_titanforged_reorder_current_progress
					value = 0
				}
			}
		}

		scope:county = {
			change_variable = {
				name = wc_titanforged_reorder_current_progress
				add = wc_titanforged_reorder_county_total
			}
		}

		scope:councillor_liege = {
			trigger_event = {
				on_action = task_reorder_county_side_effects
			}
		}
	}

	ai_target_score = {
		value = 20
	}

	ai_will_do = {
		value = 50
	}
}
