# Any scope. Scarlet always advances on Stratholme.
wc_scarlet_initialize_campaign_effect = {
	if = {
		limit = {
			exists = global_var:wc_scarlet_crusade_title
			global_var:wc_scarlet_crusade_title = { NOT = { exists = var:wc_scarlet_campaign_resolved } }
		}
		global_var:wc_scarlet_crusade_title = {
			set_variable = { name = wc_scarlet_campaign_target value = title:c_stratholme }
			holder ?= { add_pressed_claim = title:d_stratholme }
		}
	}
}
