# Fel: the debt comes due
wc_study_faith.0111 = {
	type = character_event
	title = wc_study_faith.0111.t
	desc = wc_study_faith.0111.desc
	theme = disorder
	content_source = dlc_GOA

	left_portrait = {
		character = root
		animation = fear
	}
	right_portrait = {
		trigger = { exists = scope:victim }
		character = scope:victim
		animation = fear
	}

	trigger = {
		has_variable = wc_study_faith_fel_debt
	}

	immediate = {
		remove_variable = wc_study_faith_fel_debt

		random_prisoner = {
			limit = {
				is_ai = yes
				NOT = { is_close_or_extended_family_of = root }
			}
			save_scope_as = victim
		}
		if = {
			limit = { NOT = { exists = scope:victim } }
			random_courtier = {
				limit = {
					is_ai = yes
					is_lowborn = yes
					is_councillor = no
					NOT = { is_close_or_extended_family_of = root }
				}
				save_scope_as = victim
			}
		}
	}

	# Pay with another's life
	option = {
		name = wc_study_faith.0111.a
		trigger = { exists = scope:victim }

		add_dread = medium_dread_gain

		stress_impact = {
			base = medium_stress_impact_gain
			sadistic = medium_stress_impact_loss
			callous = medium_stress_impact_loss
		}
		scope:victim = {
			death = {
				death_reason = death_sacrificed_to_gods
				killer = root
			}
		}
		ai_chance = {
			base = 50
		}
	}

	# Offer your own blood
	option = {
		name = wc_study_faith.0111.b

		increase_wounds_no_death_effect = { REASON = burned }
		increase_wounds_no_death_effect = { REASON = burned }

		ai_chance = {
			base = 30
		}
	}

	# Refuse
	option = {
		name = wc_study_faith.0111.c

		remove_character_modifier = wc_study_faith_fel_insight
		remove_character_modifier = wc_study_faith_fel_insight_greater
		increase_wounds_no_death_effect = { REASON = burned }
		add_stress = major_stress_gain

		ai_chance = {
			base = 20
		}
	}
}
