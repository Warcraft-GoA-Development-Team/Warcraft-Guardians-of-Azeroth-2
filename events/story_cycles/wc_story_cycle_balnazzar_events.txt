# A Face Not His Own - Balnazzar fakes his death and takes Saidan Dathrohan's
# identity (character:60021), seizing the Scarlet Crusade / Order of the
# Silver Hand from within. Player transfer, mirrors the ck3-docs recipe-transfer pattern.
wc_balnazzar_story.1002 = {
	type = character_event
	title = wc_balnazzar_story.1002.title
	desc = wc_balnazzar_story.1002.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60021 # Saidan Dathrohan
		animation = worry
	}

	immediate = {
		character:60021 = { save_scope_as = dathrohan }
		global_var:balnazzar_dreadlord_story = {
			set_variable = { name = story_stage value = 3 }
		}
	}

	# ponytail: single option gated on Dathrohan being alive/playable - if he's dead
	# by this point the event has zero valid options. Add a fallback branch (delay
	# retrigger, or pick another host) once that's a real risk in playtesting.
	option = {
		name = wc_balnazzar_story.1002.a

		trigger = {
			scope:dathrohan = {
				is_alive = yes
				is_playable_character = yes
			}
		}

		show_as_tooltip = {
			death = { death_reason = death_vanished }
		}
		hidden_effect = {
			scope:dathrohan = {
				add_character_flag = balnazzar_wearing_dathrohan
				add_trait = balnazzar_disguise

				# The borrowed vitality burns off Dathrohan's own age-related
				# decline at the moment of possession. Ongoing protection while
				# he wears the trait lives in story_balnazzar_dreadlord's
				# effect_group (common/story_cycles/wc_story_cycle_balnazzar.txt).
				remove_trait = clouded_eyes
				remove_trait = faltering_heart
				remove_trait = fragile_bones
				remove_trait = infirm
				remove_trait = withering_mind

				# The truth itself, tracked as a real CK3 secret - exposure
				# (however it happens) resolves the story via .2000.
				add_dreadlord_disguise_secret_effect = yes
			}

			# Keep the story attached to whoever is actually playing Balnazzar's
			# arc. Without this, killing root below fires on_owner_death and
			# ends the story right here, even though the narrative continues.
			global_var:balnazzar_dreadlord_story = {
				make_story_owner = scope:dathrohan
			}

			# Move the player OFF root before killing it. Killing the still-played
			# root triggers the engine's death/succession screen immediately and
			# aborts the rest of this effect - set_player_character must go first.
			# Must be called unscoped (root = current player), NOT from inside
			# scope:dathrohan - the engine requires the CURRENT scope to already
			# be the player ("Scoped character is not a player" otherwise).
			set_player_character = scope:dathrohan

			death = { death_reason = death_vanished } # Balnazzar's old face "dies" - root, now an NPC

			scope:dathrohan = {
				trigger_event = {
					id = wc_balnazzar_story.1003
					days = { 10 30 }
				}
			}
		}
	}
}

# The Ambush at Stratholme - Renault leads his father and High Inquisitor
# Fairbanks into a Scourge trap. Fairbanks is buried alive under corpses;
# Alexandros fights alone until he drops the Ashbringer from exhaustion.
# Renault steps out of hiding and kills his own father with it - the
# kinslaying itself corrupts the blade (not a separate effect Balnazzar
# casts), but Renault can't bear to wield it afterward and leaves it behind.
# Canon has Kel'Thuzad reclaim it later when he raises Alexandros as a death
# knight - deferred, see the comment on create_artifact_corrupted_ashbringer_effect below.
wc_balnazzar_story.1005 = {
	type = character_event
	title = wc_balnazzar_story.1005.title
	desc = wc_balnazzar_story.1005.desc
	theme = death # A kinslaying/ambush, not a scheme unfolding at court

	override_background = wc_stratholme

	left_portrait = {
		character = character:60017 # Alexandros Mograine
		animation = personality_honorable
	}
	right_portrait = {
		character = character:60019 # Renault Mograine
		animation = disbelief
	}

	immediate = {
		character:60017 = { save_scope_as = alexandros }
		character:60019 = { save_scope_as = renault }
		global_var:balnazzar_dreadlord_story = {
			set_variable = { name = story_stage value = 6 }
		}
	}

	# ponytail: same single-option caveat as .1004 - no fallback if Alexandros is
	# already dead/missing by this point.
	option = {
		name = wc_balnazzar_story.1005.a

		trigger = {
			scope:alexandros = { is_alive = yes }
		}

		show_as_tooltip = {
			scope:alexandros = { death = { death_reason = death_battle killer = scope:renault } }
		}
		hidden_effect = {
			scope:alexandros = {
				death = { death_reason = death_battle killer = scope:renault }
			}

			scope:renault = {
				# Alexandros' original (uncorrupted) Ashbringer was inherited
				# on his death - destroy it before minting the corrupted one,
				# or Renault ends up carrying both swords.
				every_character_artifact = {
					limit = { has_variable = ashbringer }
					destroy_artifact = this
				}

				# Canon: the kinslaying corrupts the blade, but Renault cannot
				# bring himself to wield it and leaves it with his father's
				# body - it's Kel'Thuzad who later reclaims it when he raises
				# Alexandros as a death knight (Four Horsemen, Naxxramas).
				# ponytail: that reclaim is a deferred beat (needs a Kel'Thuzad/
				# Naxxramas system this mod doesn't have yet here), so for now
				# the artifact sits owned-but-unequipped on Renault rather than
				# literally on the corpse - CK3's inheritance would otherwise
				# just hand a dead man's artifact straight back to his heir.
				create_artifact_corrupted_ashbringer_effect = { OWNER = this }
			}

			trigger_event = {
				id = wc_balnazzar_story.1006
				days = { 10 25 }
			}
		}
	}
}

# Blood on Silver - with Alexandros dead and Renault broken by what he's done,
# Dathrohan-Balnazzar takes the Order of the Silver Hand for his own, soon to
# be reforged as the Scarlet Crusade.
wc_balnazzar_story.1006 = {
	type = character_event
	title = wc_balnazzar_story.1006.title
	desc = wc_balnazzar_story.1006.desc
	theme = intrigue

	override_background = sm_pa_scarlet_monastery_sm

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60019 # Renault Mograine
		animation = shame # Hollow, guilt-ridden - days removed from the kill itself
	}

	immediate = {
		character:60019 = { save_scope_as = renault }
		global_var:balnazzar_dreadlord_story = {
			set_variable = { name = story_stage value = 7 }
		}
	}

	# ponytail: same single-option caveat - no fallback if Renault is dead/missing.
	option = {
		name = wc_balnazzar_story.1006.a

		trigger = {
			scope:renault = { is_alive = yes }
		}

		hidden_effect = {
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = balnazzar_silver_hand_change
				add_claim_on_loss = yes
			}
			title:k_order_of_the_silver_hand = {
				change_title_holder_include_vassals = {
					holder = root
					change = scope:balnazzar_silver_hand_change
				}
			}
			resolve_title_and_vassal_change = scope:balnazzar_silver_hand_change

			# Alexandros' death lets normal succession run first, which can
			# briefly make Renault OUR liege (he inherits the Order before we
			# seize it back above). Force him back into vassalage now, or his
			# own duchy stays outside the realm we just took.
			if = {
				limit = {
					scope:renault = { NOR = { top_liege = root this = root } }
				}
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = balnazzar_renault_vassalize_change
				}
				scope:renault = {
					change_liege = {
						liege = root
						change = scope:balnazzar_renault_vassalize_change
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_renault_vassalize_change
			}

			# The Order becomes the Scarlet Crusade - crown, color and creed all at once.
			title:k_order_of_the_silver_hand = {
				set_title_name = scarlet_crusade_custom_name
				generate_coa = yes
				set_color_from_title = title:k_scarlet_monastery # The faith's own head title - authoritative Scarlet red
			}

			# Scoped to the Silver Hand realm only (root + vassals), not every
			# Holy Light believer on the map - a whole-world instant conversion
			# would be a much bigger swing than "the Crusade converts its own".
			if = {
				limit = { has_faith = faith:holy_light }
				set_character_faith_with_conversion = faith:scarletism
			}
			every_vassal_or_below = {
				limit = { has_faith = faith:holy_light }
				set_character_faith_with_conversion = faith:scarletism
			}

			# This IS "forming the Scarlet Crusade" - grant the same reward
			# form_the_scarlet_crusade_decision gives (wc_scarlet_and_argent_decisions.txt)
			# and mark it consumed, or that generic decision keeps showing
			# forever afterward (scarletism runs doctrine_pluralism_fundamentalist,
			# so its own "NOT = { exists = faith.religious_head }" is_shown
			# check is structurally always true - it can't self-suppress here).
			add_character_modifier = wc_scarlet_crusader
			add_character_flag = has_formed_the_scarlet_crusade
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:form_the_scarlet_crusade_decision
			}
			spawn_army = {
				uses_supply = no
				inheritable = yes
				name = scarlet_crusade
				men_at_arms = { type = trebuchet stacks = 5 }
				men_at_arms = { type = archmagi stacks = 5 }
				men_at_arms = { type = armored_horsemen stacks = 5 }
				men_at_arms = { type = armored_footmen stacks = 5 }
				men_at_arms = { type = light_footmen stacks = 5 }
				men_at_arms = { type = bowmen stacks = 5 }
				men_at_arms = { type = pikemen_unit stacks = 5 }
				location = capital_province
				origin = capital_province
			}

			# TODO: Varimathras arc (fake loyalty to Sylvanas, secretly still
			# Legion-bound) deliberately deferred - see conversation notes.
		}
	}
}

# The Mask Comes Off - shared resolution for all three "reject the mask"
# triggers: old age with no heir (.0100), secret exposure (on_expose in
# common/secret_types/wc_balnazzar_secret_types.txt, already wired to this
# event id), and paranoia (balnazzar_shed_the_mask_decision, Task 2).
# Whatever led here, the outcome is the same - the disguise ends, the
# Crusade burns, the Risen rise. Matches canon: Balnazzar drops "Dathrohan"
# and raises his own dead crusaders as the Risen, founding a new unholy
# empire at Stratholme.
wc_balnazzar_story.2000 = {
	type = character_event
	title = wc_balnazzar_story.2000.title
	desc = wc_balnazzar_story.2000.desc
	theme = death

	override_background = wc_stratholme

	left_portrait = {
		character = root
		animation = rage
	}

	immediate = {
		add_character_flag = balnazzar_shed_the_mask
		remove_trait = balnazzar_disguise
	}

	option = {
		name = wc_balnazzar_story.2000.a

		hidden_effect = {
			# Canon: Balnazzar's undead force ("the Risen") is a Stratholme-
			# regional force, not the whole Scarlet Crusade - the wider
			# Crusade survives him, fragmented (mirrors its real WoW split
			# into the mainline Crusade and Abbendis' Scarlet Onslaught).
			# capital_county, not primary_title.county, so this also catches
			# vassals whose capital sits here even if their primary title
			# is a barony (confirmed pattern: common/activities/guest_invite_rules/
			# activity_invite_rules.txt:1080-1082).
			every_vassal_or_below = {
				limit = {
					is_alive = yes
					NOT = { has_trait = being_undead }
					capital_county = root.capital_county
				}
				become_undead_effect = yes
			}

			every_courtier = {
				limit = {
					is_alive = yes
					NOT = { has_trait = being_undead }
				}
				become_undead_effect = yes
			}

			# Bring back Balnazzar's own nathrezim body (character:430400,
			# killed at .1002 - "his old face 'dies' - root, now an NPC" -
			# when he took Dathrohan's identity). Mirrors the exact
			# resurrection pattern already proven in this mod for Kel'Thuzad
			# (common/scripted_effects/wc_other_effects.txt:67), which copies
			# DNA/appearance/traits/name from the dead character - this is
			# what actually fixes the portrait, unlike set_race_effect
			# (traits/stats only, no gene/DNA change - creature_nathrezim's
			# look is baked in at creation, no runtime portrait_modifier
			# exists for it elsewhere in this mod).
			wc_resurrect_CHARACTER_with_STATUS_effect = {
				CHARACTER = character:430400
				STATUS = alive
			}

			# Only the direct capital territory breaks away with Balnazzar -
			# NOT every_held_title. The rest of the Scarlet Crusade (the
			# kingdom/duchy crown and its other counties) is deliberately
			# left behind on the dying shell below, to pass through normal
			# succession without him - it continues on its own, matching
			# canon. type = independence, not granted: nobody is handing
			# this to him, he's breaking it off by force. Sequence itself
			# (create_title_and_vassal_change / change_title_holder_include_vassals /
			# resolve_title_and_vassal_change) is the same one this mod
			# already uses for a live player-to-player handoff
