#On actions related to religion

# Code on-action: a new faith is created (character create/convert, or a rite detaching into its own faith)
# Root is the new faith; its head/leader is root.main_rite.head_of_rite (use it for faith-setup effects)
# scope:founder is the acting character that triggered the creation (creator/converter, heresy declarer, or on a divergence the head of the diverging rite)
# scope:founder can be absent - the game creates faiths with no acting character, so it is entered with ?=
# scope:old_faith is the faith they used to have
# scope:old_rite is the rite they used to have
# scope:diverged (true/false) optional scope, true if the faith was created because of a Rite diverging too much. Notice create a 100+ divergence heresy at the start does not count.
on_faith_created = {
	effect = {
		main_rite = { force_refresh_head_of_rite = yes }
		# Faiths created by a rite diverging too far start at 0 fervor
		if = {
			limit = {
				scope:diverged ?= yes
			}
			diverged_faith_starting_fervor_effect = yes
		}
		# Sanctification is per faith
		if = {
			limit = {
				exists = scope:old_faith
				scope:old_faith.religion = religion
			}
			every_artifact = {
				limit = { is_sanctified_for_faith = scope:old_faith }
				add_sanctified_for_faith = root
			}
		}
		# Remove Antipopes
		if = {
			limit = {
				scope:diverged ?= yes
				any_faith_character = { has_character_flag = antipope_flag }
			}
			ordered_faith_character = {
				limit = { has_character_flag = antipope_flag }
				order_by = head_of_faith_selection_weight
				save_scope_as = pam_diverged_antipope
			}
			every_faith_character = {
				limit = { has_character_flag = antipope_flag }
				pam_strip_antipope_office_effect = yes
			}
			scope:pam_diverged_antipope = {
				if = {
					limit = { faith = { has_doctrine = doctrine_temporal_head } }
					set_up_dynamic_temporal_hof_title_effect = { NEW_HOLDER = scope:pam_diverged_antipope }
				}
				else_if = {
					limit = { faith = { has_doctrine = doctrine_spiritual_head } }
					set_up_dynamic_spiritual_hof_title_effect = { CREATOR = scope:pam_diverged_antipope }
				}
			}
		}
		scope:founder ?= {
			if = {
				limit = { faith = root }
				pam_on_conversion_effect = yes
				remove_character_modifier = religious_reformer_modifier
				# If you have a legend maybe we change it?
				if = {
					limit = {
						has_legend_chapter_trigger = { CHAPTER = opening }
						promoted_legend ?= {
							exists = legend_property:faith
							exists = legend_property:religion
						}
						NOT = {
							promoted_legend ?= {
								has_legend_chapter = {
									name = opening
									localization_key = legend_chapter_opening_new_faith
								}
							}
						}
					}
					send_interface_toast = {
						title = legend_chapter_change_toast
						left_icon = scope:founder
						promoted_legend = {
							set_legend_chapter = {
								name = opening
								localization_key = legend_chapter_opening_new_faith
							}
						}
					}
				}
				else_if = {
					limit = {
						has_legend_chapter_trigger = { CHAPTER = opening }
					}
					random = {
						chance = 75
						send_interface_toast = {
							title = legend_chapter_change_toast
							left_icon = scope:founder
							promoted_legend = {
								set_legend_chapter = {
									name = opening
									localization_key = legend_chapter_opening_new_faith_generic
								}
							}
						}
					}
				}
			}
		}
		scope:old_faith = {
			if = {
				limit = { has_doctrine_parameter = unreformed }
				set_variable = { name = has_been_reformed }
				root = {
					if = {
						limit = { has_doctrine = unreformed_faith_doctrine }
						remove_doctrine = unreformed_faith_doctrine
					}
					if = {
						limit = { has_doctrine = west_african_unreformed_faith_doctrine }
						remove_doctrine = west_african_unreformed_faith_doctrine
					}
				}
				# Use the new faith's head for state faith, never scope:founder (may not follow)
				root.main_rite.head_of_rite ?= {
					if = {
						limit = {
							government_allows = state_faith
							liege = this
							primary_title.state_rite.faith = scope:old_faith
						}
						primary_title = {
							set_state_rite = rite
							pam_tenet_popularity_boost_rite_effect = { RITE = state_rite }
						}
					}
				}
				scope:founder ?= {
					if = {
						limit = {
							faith = root
							NOT = { has_character_flag = reformer_legend }
							has_dlc_feature = legends
							NOT = { has_game_rule = historical_legends_only }
						}
						add_character_flag = reformer_legend
						create_legend_seed = {
							type = holy
							quality = famed
							chronicle = new_faith_legend
							properties = {
								reformer = scope:founder
								faith = root
								religion = root.religion
								reason = flag:pagan_reformation
							}
						}
					}
				}
				every_faith_character = {
					limit = {
						has_character_flag = forced_conversion_on_reform
					}
					set_character_faith = root
					every_vassal_or_below = {
						limit = { faith = scope:old_faith }
						random = {
							chance = 75
							set_character_faith = root
						}
					}
					every_sub_realm_county = {
						limit = { faith = scope:old_faith }
						random = {
							chance = 75
							set_county_faith = root
						}
					}
				}
				# The founding wave (founder and vassals who already accepted) brings their courts, as with a new rite
				# Listed first, since the conversions below add characters to the faith being iterated
				if = {
					limit = {
						NOT = { scope:diverged ?= yes }
					}
					root = {
						every_faith_character = {
							limit = { is_ruler = yes }
							add_to_temporary_list = pam_reform_founding_wave
						}
					}
					every_in_list = {
						list = pam_reform_founding_wave
						pam_convert_court_to_faith_effect = {
							FAITH = root
							OLD_FAITH = scope:old_faith
						}
					}
				}
			}
			else = {
				scope:founder ?= {
					if = {
						limit = {
							faith = root
							NOT = { has_character_flag = reformer_legend }
							has_dlc_feature = legends
							NOT = { has_game_rule = historical_legends_only }
						}
						add_character_flag = reformer_legend
						create_legend_seed = {
							type = holy
							quality = famed
							chronicle = new_faith_legend
							properties = {
								reformer = scope:founder
								faith = root
								religion = root.religion
								reason = flag:new_faith
							}
						}
					}
				}
			}

			scope:founder ?= { play_music_cue = "mx_cue_faith_conversion" }
		}
		# A faith founded during the game may reshape its holy sites three times for free.
		set_variable = {
			name = free_holy_site_actions
			value = 3
		}
		# Clear the old faith's holy-site lock so it cannot use up the free actions
		# Locks sit on the founder, so swapping puppets cannot skip the cooldown
		scope:founder ?= {
			if = {
				limit = { faith = root }
				set_variable = has_created_a_faith
				remove_variable = holy_site_creation_cooldown
				remove_decision_cooldown = elevate_holy_site_decision
				remove_decision_cooldown = demote_holy_site_decision
				remove_decision_cooldown = abandon_holy_site_decision
				remove_decision_cooldown = adopt_holy_site_decision
				check_for_equal_doctrine_effect = yes
			}
		}
		# Set up diff-faith HoFs.
		# If we have a diff-faith HoF, we want to account for that.
		## Islamic use-case: we want same-Mohammad's Succession faiths to keep following their appropriate caliphs.
		if = {
			limit = {
				# Are we in the correct religion?
				religion = religion:islam_religion
				# Faiths with no HoF or a spiritual head aren't relevant for these calculations.
				has_doctrine = doctrine_temporal_head
			}
			# We save_scope_as = reformer for tooltip loc, because the events complain about root
			root.main_rite.head_of_rite ?= { save_scope_as = reformer }
			# Sunni path.
			if = {
				limit = { main_rite = { rite_has_doctrine = muhammad_succession_sunni_doctrine } }
				root.main_rite.head_of_rite ?= { trigger_event = head_of_faith.0001 }
			}
			# Shia path.
			else_if = {
				limit = { main_rite = { rite_has_doctrine = muhammad_succession_shia_doctrine } }
				root.main_rite.head_of_rite ?= { trigger_event = head_of_faith.0011 }
			}
			# Muhakkima path.
			else_if = {
				limit = { main_rite = { rite_has_doctrine = muhammad_succession_muhakkima_doctrine } }
				root.main_rite.head_of_rite ?= { trigger_event = head_of_faith.0021 }
			}
			# Zandaqa path.
			else_if = {
				limit = { main_rite = { rite_has_doctrine = muhammad_succession_zandaqa_doctrine } }
				root.main_rite.head_of_rite ?= { trigger_event = head_of_faith.0031 }
			}
		}
		# PAM - Christian Situation, add them to the correct group
		if = {
			limit = {
				exists = situation:the_christian_church
				religion = religion:christianity_religion
			}
			root.main_rite.head_of_rite ?= { recalculate_participant_group = situation:the_christian_church }
		}
		# Faiths with a College (Catholicism) need a seeded College or AI vacancy spam follows
		hidden_effect = {
			if = {
				limit = { has_doctrine_parameter = has_clerical_electors }
				pam_seed_cardinals_effect = yes
			}
		}
		# Achievements - evaluated on the acting character, as add_achievement_* gate on is_ai.
		scope:founder ?= {
			if = {
				limit = { faith = root }
				add_achievement_flag_effect = { FLAG = achievement_its_not_a_cult_flag }
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = west_african_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_siguic
						VALUE = yes
					}
				}
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = west_african_orisha_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_orisha
						VALUE = yes
					}
				}
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = west_african_bori_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_bori
						VALUE = yes
					}
				}
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = west_african_roog_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_roog_sene
						VALUE = yes
					}
				}
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = akom_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_akom
						VALUE = yes
					}
				}
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = kushitism_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_kushitism
						VALUE = yes
					}
				}
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = waaqism_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_waaqism
						VALUE = yes
					}
				}
				if = {
					limit = {
						scope:old_faith = {
							religion_tag = north_african_religion
							has_doctrine_parameter = unreformed
						}
					}
					add_achievement_global_variable_effect = {
						VARIABLE = achievement_mother_of_us_all_achamanism
						VALUE = yes
					}
				}
			}
		}
		# These events need a character, so they fire on the new faith's head, never a founder outside the faith
		root.main_rite.head_of_rite ?= {
			trigger_event = faith_conversion.0001 # Convert capital county
			trigger_event = faith_conversion.0006 # Same-religion capital joins the new faith
			trigger_event = faith_conversion.0003 # Convert close family
			trigger_event = faith_creation.0001 # Temporal rel head setup
			trigger_event = faith_creation.0002 # Spiritual rel head setup
			trigger_event = faith_creation.0003 # Faith flags
		}
		# Notifications go to the acting character.
		scope:founder ?= {
			if = {
				limit = { faith = root }
				trigger_event = faith_creation.1000 # Notification event
				trigger_event = faith_creation.1002 # Notification event - rite diverged into a new faith
			}
		}
	}
}

# Code on-action: character creates a rite
# Root is the creator
# scope:old_faith is the faith they used to have
# scope:old_rite is the rite they used to have
on_rite_created = {
	effect = {
		pam_on_conversion_effect = yes
		remove_character_modifier = religious_reformer_modifier
		# The rite is brand new, so everyone already holding it (creator, realm priest, AI vassals who accepted the code-sent offer) is part of the founding wave and brings their court and domain along
		faith = {
			every_faith_character = {
				limit = {
					rite = root.rite
				}
				pam_convert_court_to_rite_effect = { RITE = root.rite }
				if = {
					limit = {
						is_landed = yes
						highest_held_title_tier >= tier_county
					}
					pam_convert_domain_counties_to_rite_effect = { RITE = root.rite }
				}
			}
		}
		# Same grace var as rite_growth and heresy events, since create_dynamic_rite skips this on-action
		rite = {
			set_variable = {
				name = pam_rite_grace_period
				value = yes
				days = rite_growth_grace_period_days
			}
		}
		# Vassals answering the creation-time conversion offer later (players) also bring their court and domain; checked in pam_sway_to_rite_on_accept_effect
		rite = {
			set_variable = {
				name = pam_rite_founding_window
				value = yes
				years = 1
			}
		}
		# If you have a legend maybe we change it?
		if = {
			limit = {
				has_legend_chapter_trigger = { CHAPTER = opening }
				promoted_legend ?= {
					exists = legend_property:faith
					exists = legend_property:religion
				}
				NOT = {
					promoted_legend ?= {
						has_legend_chapter = {
							name = opening
							localization_key = legend_chapter_opening_new_faith
						}
					}
				}
			}
			send_interface_toast = {
				title = legend_chapter_change_toast
				left_icon = root
				promoted_legend = {
					set_legend_chapter = {
						name = opening
						localization_key = legend_chapter_opening_new_faith
					}
				}
			}
		}
		else_if = {
			limit = {
				has_legend_chapter_trigger = { CHAPTER = opening }
			}
			random = {
				chance = 75
				send_interface_toast = {
					title = legend_chapter_change_toast
					left_icon = root
					promoted_legend = {
						set_legend_chapter = {
							name = opening
							localization_key = legend_chapter_opening_new_faith_generic
						}
					}
				}
			}
		}
		scope:old_faith = {
			if = {
				limit = { has_doctrine_parameter = unreformed }
				set_variable = { name = has_been_reformed }
				root.faith = {
					if = {
						limit = { has_doctrine = unreformed_faith_doctrine }
						remove_doctrine = unreformed_faith_doctrine
					}
					#if = {
					#	limit = { has_doctrine = west_african_unreformed_faith_doctrine }
					#	remove_doctrine = west_african_unreformed_faith_doctrine
					#}
				}
				root = {
					if = {
						limit = {
							NOT = { has_character_flag = reformer_legend }
							has_dlc_feature = legends
							# Warcraft
						}
						add_character_flag = reformer_legend
						create_legend_seed = {
							type = holy
							quality = famed
							chronicle = new_faith_legend
							properties = {
								reformer = root
								faith = root.faith
								religion = root.religion
								reason = flag:pagan_reformation
							}
						}
					}
				}
				every_faith_character = {
					limit = {
						has_character_flag = forced_conversion_on_reform
					}
					set_character_faith = root.faith
					every_vassal_or_below = {
						limit = { faith = scope:old_faith }
						random = {
							chance = 75
							set_character_faith = root.faith
						}
					}
					every_sub_realm_county = {
						limit = { faith = scope:old_faith }
						random = {
							chance = 75
							set_county_faith = root.faith
						}
					}
				}
			}
			else = {
				root = {
					if = {
						limit = {
							NOT = { has_character_flag = reformer_legend }
							has_dlc_feature = legends
							# Warcraft
						}
						add_character_flag = reformer_legend
						create_legend_seed = {
							type = holy
							quality = famed
							chronicle = new_faith_legend
							properties = {
								reformer = root
								faith = root.faith
								religion = root.religion
								reason = flag:new_faith
							}
						}
					}
				}
			}

			root = { play_music_cue = "mx_cue_faith_conversion" }
		}
		set_variable = {
			name = has_created_a_faith
		}
		check_for_equal_doctrine_effect = yes

		# Warcraft - This is needed for our Union triggers
		# Set original faith this one was based on inside of the faith itself
		if = {
			limit = {
				scope:old_faith = { has_variable = parent_faith }
			}
			faith = {
				set_variable = {
					name = parent_faith
					value = scope:old_faith.var:parent_faith
				}
			}
		}
		else = {
			faith = {
				set_variable = {
					name = parent_faith
					value = scope:old_faith
				}
			}
		}

		# Set up diff-faith HoFs.
		faith = {
			# If we have a diff-faith HoF, we want to account for that.
			## Islamic use-case: we want same-Mohammad's Succession faiths to keep following their appropriate caliphs.
			# Warcraft - No Islam here
			## Faiths with the Rite tenet get to keep their old HoF.
			if = {
				limit = {
					# Assuming they exist.
					exists = scope:old_faith.religious_head
					# & have the appropriate doctrine parameter.
					has_doctrine_parameter = maintains_head_of_faith_on_creation
				}
				set_religious_head_title = scope:old_faith.religious_head_title
				# Christian faiths also get to keep ecumenism.
				if = {
					limit = {
						scope:old_faith = { has_doctrine = special_doctrine_ecumenical_christian }
					}
					add_doctrine = special_doctrine_ecumenical_christian
				}
			}
			## Otherwise, everyone else should lose their HoF, so we don't need to do anything.
		}
		# Achievements.
		root = {
			add_achievement_flag_effect = { FLAG = achievement_its_not_a_cult_flag }
		}
		# Warcraft - We don't use these achievements
	}

	events = {
		faith_conversion.0001 # Convert capital county
		faith_conversion.0003 # Convert close family
		faith_creation.1020 # Notification event - player founded a new rite
	}
}

# Code on-action: rite has been changed
# root is the rite being changed
# scope:changed_tenets_or_doctrines is set when tenets or doctrines changed (not set on aesthetic-only changes)
# scope:consume_rite_edit_token is set only when the player edited the rite via the interface (not by scripted effects)
# scope:added_tenets / scope:removed_tenets are lists of the tenets this change added to / removed from the rite (iterate with every_in_list = { list = added_tenets ... })
# scope:added_doctrines / scope:removed_doctrines are lists of the doctrines this change added to / removed from the rite
# Use these lists to inform adherents of exactly what changed in their rite
on_rite_updated = {
	events = {
	}

	effect = {
		root = {
			save_scope_as = rite
		}

		if = { # slient it for EC since EC has unique evvent to inform
			limit = {
				exists = scope:changed_tenets_or_doctrines
				OR = {
					NOT = {
						faith = { has_variable = ecumenical_council_ongoing }
					}
					exists = scope:consume_rite_edit_token
				}
			}

			# The message icon slot must never be left unset, hence the fall-through to the rite itself
			if = {
				limit = { any_in_list = { list = added_tenets count >= 1 } }
				every_in_list = {
					list = added_tenets
					save_scope_as = rite_change_icon
				}
			}
			else_if = {
				limit = { any_in_list = { list = added_doctrines count >= 1 } }
				every_in_list = {
					list = added_doctrines
					save_scope_as = rite_change_icon
				}
			}
			else_if = {
				limit = { any_in_list = { list = removed_tenets count >= 1 } }
				every_in_list = {
					list = removed_tenets
					save_scope_as = rite_change_icon
				}
			}
			else_if = {
				limit = { any_in_list = { list = removed_doctrines count >= 1 } }
				every_in_list = {
					list = removed_doctrines
					save_scope_as = rite_change_icon
				}
			}
			else = {
				root = { save_scope_as = rite_change_icon }
			}

			every_player = {
				limit = {
					OR = {
						rite = root
						AND = {
							faith = root.faith
							root.faith.main_rite = root
						}
					}
				}
				send_interface_message = {
					type = msg_rite_updated
					title = msg_rite_updated
					left_icon = scope:rite_change_icon

					show_as_tooltip = {
						every_in_list = {
							list = removed_doctrines
							root = { remove_rite_doctrine = prev }
						}
						every_in_list = {

	}

# Code on-action: when on_potential_great_holy_war_invalidation did not find a new target, and the GHW is about to be destroyed. no way out at this point

	events = {

	}
}

# Code on-action: when the GHW's war launch countdown hits zero. If a war is not started at this point, the GHW will invalidate, bypassing on_potential_great_holy_war_invalidation but not on_great_holy_war_invalidation
# Root is the great holy war
on_great_holy_war_countdown_end = {
	trigger = { #Only launch the holy war if attacker strength is above one third of defender.
		ghw_attackers_strength >= third_of_defenders_strength
	}
	events = {
		great_holy_war.0005
	}
}

update_main_rite_action = {
	# you should win your position via the antipope system if there is a religious head,
	trigger = {
		NOT = { exists = religious_head }
	}
	effect = {
		main_rite = {
			save_scope_as = main_rite
			save_scope_value_as = { name = rite_counties value = rite_counties }
			save_scope_value_as = { name = rite_holy_sites value = rite_holy_sites }
		}
		if = {
			limit = {
				any_faith_rite = {
					this != scope:main_rite
					rite_counties > scope:rite_counties
					rite_holy_sites >= scope:rite_holy_sites
				}
			}
			ordered_faith_rite = {
				limit = {
					this != scope:main_rite
					rite_holy_sites >= scope:rite_holy_sites
				}
				order_by = rite_counties
				save_scope_as = new_main_rite
				every_player = {
					limit = { faith = root }
					send_interface_message = {
						title = msg_faith_main_rite_changed.title
						desc = msg_faith_main_rite_changed.desc
						right_icon = scope:new_main_rite

						show_as_tooltip = {
							scope:new_main_rite = {
								set_main_rite = yes
							}
						}
					}
				}
				set_main_rite = yes
			}
		}
	}
}

# Faith Rite Growth System #
############################

# Faith scope. Once a month per faith.
# Slowly spawns new rites until the faith reaches its target rite count: floor(counties / X) - 1
# branched rites besides the mainline, so the first branched rite arrives at 2 * X counties.
# Cooldown and target count live in game/common/script_values/02_religion_values.txt.

faith_rite_growth_pulse = {
	trigger = {
		NOR = {
			has_doctrine_parameter = unreformed
			has_game_rule = none_branched_rite_frequency
			exists = var:rite_growth_recent_spawn
			exists = var:rite_growth_recent_heresy
		}
		rite_growth_rites_needed > 0
		years_from_game_start >= 2
	}
	random_events = {
		chance_to_happen = 3
		chance_of_no_event = {
			value = 0
			if = {
				limit = { fervor >= 25 }
				value = fervor
				max = 90
			}
			if = {
				limit = { rite_growth_rites_needed >= 5 }
				max = 40
			}
			else_if = {
				limit = { rite_growth_rites_needed >= 3 }
				max = 60
			}
		}
		100 = rite_growth.0001
	}
}

faith_rite_additional_growth_chance_pulse = {
	trigger = {
		NOR = {
			has_doctrine_parameter = unreformed
			has_game_rule = none_branched_rite_frequency
			# Shorter cooldown than the primary pulse so a faith at target can still trickle an extra rite
			exists = var:rite_growth_recent_spawn_fallback
			exists = var:rite_growth_recent_heresy
		}
		rite_growth_active_rite_count < rite_growth_rites_soft_max_cap
		num_county_followers >= rite_growth_fallback_min_counties
		years_from_game_start >= 2
	}
	random_events = {
		chance_to_happen = 1
		chance_of_no_event = 90
		100 = rite_growth.0001
	}
}

# Called after Holy Site is added for a faith
#
# Root scope: Faith holy site is added for
# scope:title - barony title of location where holy site is created
# scope:holy_site - holy site being added (warning! may not be set, if generating description, and holy site is new)
# scope:is_eminent_site - if the holy site is eminent (else it is regular)
# scope:is_new_holy_site - if the holy site was newly created (else it was adopted)
# scope:actor - character responsible, only set when a character drove the change
on_faith_holy_site_added = {
	effect = {
		if = {
			limit = {
				scope:holy_site ?= {
					any_holy_site_faith = { count = 1 }
				}
			}
			record_dynamic_holy_site_kingdom_effect = { BARONY = scope:title }
		}
		# Send Toasts
		if = {
			limit = { scope:is_new_holy_site = yes }
			notify_faith_of_holy_site_change_effect = {
				MESSAGE = msg_holy_site_created
				MESSAGE_BY_ACTOR = msg_holy_site_created_by_character
			}
		}
		else = {
			notify_faith_of_holy_site_change_effect = {
				MESSAGE = msg_holy_site_adopted
				MESSAGE_BY_ACTOR = msg_holy_site_adopted_by_character
			}
		}
		# CE3 achievement: Father's best side — record the founding dynasty on runtime-created holy sites
		if = {
			limit = {
				scope:is_new_holy_site = yes
				exists = scope:actor.dynasty
			}
			scope:title = {
				set_variable = {
					name = ce3_holy_site_founder_dynasty
					value = scope:actor.dynasty
				}
			}
		}
	}
}

# Called after Holy Site is removed for a faith
#
# Root scope: Faith holy site is removed for
# scope:title - barony title of location where holy site was
# scope:holy_site - holy site being removed
# scope:is_eminent_site - if the holy site was eminent (else it was regular)
# scope:actor - character responsible, only set when a character drove the change
on_faith_holy_site_removed = {
	effect = {
		release_dynamic_holy_site_kingdom_effect = { BARONY = scope:title }
		# We record the last faith left on a shared site here, since no add fires for them
		scope:holy_site ?= {
			if = {
				limit = { any_holy_site_faith = { count = 1 } }
				random_holy_site_faith = {
					record_dynamic_holy_site_kingdom_effect = { BARONY = scope:title }
				}
			}
		}
		# Send Toasts
		notify_faith_of_holy_site_change_effect = {
			MESSAGE = msg_holy_site_abandoned
			MESSAGE_BY_ACTOR = msg_holy_site_abandoned_by_character
		}
	}
}

# Called after a Holy Site of a faith becomes eminent, or stops being eminent
#
# Root scope: Faith the holy site belongs to
# scope:title - barony title of location where the holy site is
# scope:holy_site - holy site changing eminence
# scope:is_eminent_site - if the holy site is now eminent (else it is now regular)
# scope:actor - character responsible, only set when a character drove the change
on_faith_holy_site_eminence_changed = {
	effect = {
		if = {
			limit = { scope:is_eminent_site = yes }
			notify_faith_of_holy_site_change_effect = {
				MESSAGE = msg_holy_site_elevated
				MESSAGE_BY_ACTOR = msg_holy_site_elevated_by_character
			}
		}
		else = {
			notify_faith_of_holy_site_change_effect = {
				MESSAGE = msg_holy_site_demoted
				MESSAGE_BY_ACTOR = msg_holy_site_demoted_by_character
			}
		}
	}
}

# Called when a Character creates a new Holy Site.
# Note: Also used to show effects in GUI before creation
#
# Root scope: Character creating holy site
# scope:title - barony title of location where holy site is created
# scope:holy_site_type - holy site type being created
# scope:faith - Faith the holy site is created for
# scope:artifacts - List of artifacts being enshrined
on_character_holy_site_created = {
	effect = {
		# Same holy-site slot rules as game start
		scope:title.title_province = {
			add_holy_site_fallback_special_building_slot_effect = { FAITH = scope:faith }
		}
		hidden_effect = {
			if = {
				limit = { scope:faith = { faith_has_free_holy_site_action_trigger = yes } }
				scope:faith = { consume_free_holy_site_action_effect = yes }
			}
			else = {
				set_variable = {
					name = holy_site_creation_cooldown
					years = holy_site_action_cooldown_years
				}
			}
		}
		# CE3 achievement: Father's best side — a relic from a saint of the actor's family, enshrined in a holy site the actor's dynasty founded
		if = {
			limit = {
				exists = dynasty
				scope:title.var:ce3_holy_site_founder_dynasty ?= dynasty
				any_in_list = {
					list = artifacts
					exists = var:pam_saint_relic_saint
					save_temporary_scope_as = relic_temp
					root = {
						OR = {
							is_close_or_extended_family_of = scope:relic_temp.var:pam_saint_relic_saint
							scope:relic_temp.var:pam_saint_relic_saint.dynasty ?= dynasty
						}
					}
				}
			}
			add_achievement_global_variable_effect = {
				VARIABLE = ce3_06_fathers_best_side_achievement_unlocked
				VALUE = yes
			}
		}
	}
}
