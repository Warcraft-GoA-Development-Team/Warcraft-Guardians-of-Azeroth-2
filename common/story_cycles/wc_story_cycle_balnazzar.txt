# Balnazzar - Dreadlord personal story
# Only fires when the player picks to play as character:430400 in the Frozen Throne
# bookmark (bm_tft_lordaeron, 605.6.6) - see common/on_action/game_start.txt.
# TODO: everything below is a skeleton, fill in the real narrative beats

story_balnazzar_dreadlord = {
	visible = yes

	icon = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/icons/traits/lich_king.dds" # TODO: dedicated Dreadlord/Balnazzar icon
	}

	background = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/illustrations/event_scenes/wc_frostmourne_cave.dds" # TODO: dedicated background
	}

	visualization = {
		custom_string_key = "BALNAZZAR_DREADLORD_CUSTOM_STRING_KEY" # TODO: add loc key + decide what shows in the story UI
	}

	on_setup = {
		set_global_variable = { # Anchor, so the world can reference this playthrough's story
			name = balnazzar_dreadlord_story
			value = this
		}

		set_variable = {
			name = story_stage
			value = 1
		}
	}

	on_end = {
		remove_global_variable = balnazzar_dreadlord_story
	}

	on_owner_death = {
		# TODO: decide what happens if the player-Balnazzar dies mid-story
		end_story = yes
	}

	# CK3's age-related health traits (clouded_eyes, infirm, ...) trigger on
	# chronological age alone, ignoring life_expectancy entirely - even for
	# genuine nathrezim elsewhere in the mod. Keep stripping them from whoever
	# wears the disguise, instead of patching the shared vanilla health system.
	effect_group = {
		days = { 30 60 }

		trigger = {
			story_owner = { has_trait = balnazzar_disguise }
		}

		triggered_effect = {
			effect = {
				story_owner = {
					remove_trait = clouded_eyes
					remove_trait = faltering_heart
					remove_trait = fragile_bones
					remove_trait = infirm
					remove_trait = withering_mind
				}
			}
		}
	}

	# TODO: effect_group(s) for stage progression, mirroring wc_story_cycle_path_of_the_damned.txt
}
