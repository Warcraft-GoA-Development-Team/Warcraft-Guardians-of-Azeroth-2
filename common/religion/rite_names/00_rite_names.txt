# Culture Head Founder / Puppeteer is Culture Head / 50% General Chance
culture_head_founder_christian_rite = {
	
	# 'Culture' 'Religionname', i.e. Polish Christianity, Swedish Christianity
	name = d_culture_head_founder_christian_rite
	
	trigger = {
		scope:faith = { like_christianity_religion_trigger = yes }
		scope:founder = {
			capital_county.culture ?= culture
			OR = {
				culture.culture_head ?= this
				AND = {
					puppeteer.culture.culture_head ?= puppeteer
					puppeteer.culture ?= culture
				}
				static_group_filter = {
					group = culture_name_general_chance
					match = 0.5
				}
			}
			#To avoid double-naming with game start rites
			#culture = { #Warcraft add names for future rites
			#	NOR = {
			#		this = culture:greek
			#		this = culture:roman
			#		this = culture:georgian
			#	}
			#}
		}
	}
}

# Landless Founder, -ian family
landless_founder_christian_rite_ian = {

	# 'Firstname'ianism, examples: Nestorianism, Waldensianism
	name = d_landless_founder_christian_rite_ian

	trigger = {
		scope:faith = { like_christianity_religion_trigger = yes }
		scope:founder = {
			static_group_filter = {
				group = landless_founder_christian_rite_style
				match = 0.25
			}
			OR = {
				is_landed = no
				is_clergy = yes
				AND = {
					is_landed = yes
					highest_held_title_tier <= 2
				}
			}
		}
	}
}
