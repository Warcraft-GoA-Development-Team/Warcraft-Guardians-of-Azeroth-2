# Notification - Death/Defeat of the previous Champion
wc_path_of_the_damned.0100 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_path_of_the_damned.0100.title
	desc = wc_path_of_the_damned.0100.desc
	theme = plague

	left_portrait = {
		character = scope:old_champion
		animation = anger
	}

	immediate = {
		scope:old_champion = {
			every_character_artifact = {
				limit = { has_variable = frostmourne }
				save_scope_as = frostmourne
			}
		}

		if = {
			limit = { exists = scope:frostmourne }

			if = {
				limit = { scope:old_champion = { is_imprisoned_by = root } }
				scope:frostmourne = { set_owner = root }
			}
			else = {
				every_artifact = {
					limit = { has_variable = frostmourne }
					save_scope_as = frostmourne
				}
				if = {
					limit = { exists = scope:frostmourne }
					add_personal_artifact_claim = scope:frostmourne
				}
				else = { scope:story_path = { end_story = yes } }
			}
		}
	}

	option = { # It's bad :(
		name = wc_path_of_the_damned.0100.a
	}
}

# A new Champion
wc_path_of_the_damned.0101 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_path_of_the_damned.0101.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { this = scope:lich_king }
				desc = wc_path_of_the_damned.0101.desc.lich_king
			}
			triggered_desc = {
				trigger = { this = scope:new_champion }
				desc = wc_path_of_the_damned.0101.desc.new_champion
			}
			desc = wc_path_of_the_damned.0101.desc
		}
	}
	theme = plague

	override_background = {
		reference = wilderness_mountains
	}

	left_portrait = {
		character = scope:new_champion
		animation = personality_callous
	}
	lower_right_portrait = scope:lich_king
	artifact = {
		target = scope:frostmourne
		position = lower_left_portrait
		trigger = { exists = scope:frostmourne }
	}

	immediate = {
		if = {
			limit = { this = scope:new_champion }
			create_new_champion_effect = { FROSTMOURNE = scope:frostmourne }
		}
		else = {
			show_as_tooltip = {
				scope:new_champion = { create_new_champion_effect = { FROSTMOURNE = scope:frostmourne } }
			}
		}
	}

	option = { # Glory to the Lich King !
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { this = scope:lich_king }
						desc = wc_path_of_the_damned.0101.a.lich_king
					}
					triggered_desc = {
						trigger = { this = scope:new_champion }
						desc = wc_path_of_the_damned.0101.a.new_champion
					}
					desc = wc_path_of_the_damned.0101.a
				}
			}
		}

		if = {
			limit = { this = scope:new_champion }
			custom_tooltip = wc_path_of_the_damned.1001.a.tooltip
		}
	}
}

# Arthas meets Tichondrius - 1
wc_path_of_the_damned.1001 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_path_of_the_damned.1001.title
	desc = wc_path_of_the_damned.1001.desc
	theme = plague

	override_background = {
		reference = wc_androhal_by_dlikt
	}

	left_portrait = {
		character = root
		animation = rage
	}
	right_portrait = {
		character = scope:tichondrius
		animation = personality_callous
	}
	lower_right_portrait = scope:killed_malganis

	immediate = {
		scope:story = { # Save scope for dialogue
			var:malganis = { save_scope_as = killed_malganis }
			var:frostmourne = { save_scope_as = frostmourne }
			var:king_of_lordaeron = { save_scope_as = killed_king_of_lordaeron }
		}

		find_tichondrius_effect = yes
		find_nerzhul_effect = yes
	}

	option = { #Congratulate me?
		name = wc_path_of_the_damned.1001.a
		flavor = wc_path_of_the_damned.1001.a.flavor
		custom_tooltip = wc_path_of_the_damned.1001.a.tooltip

		trigger_event = wc_path_of_the_damned.1002

		following_lore_path_effect = yes

		ai_chance = {
			base = 100
		}
	}

	option = { #Die, demon!
		name = wc_path_of_the_damned.1001.b
		flavor = wc_path_of_the_damned.1001.b.flavor
		custom_tooltip = wc_path_of_the_damned.1001.b.tooltip

		root = { save_scope_as = champion }

		scope:nerzhul = { # He refused to obey
			trigger_event = {
				id = wc_path_of_the_damned.1003
				days = 5
			}
		}

		set_culture = culture:forsaken
		set_character_faith_with_conversion = faith:forsaken_cult

		stress_impact = {
			base = massive_stress_impact_gain
		}

		following_alternate_path_effect = yes

		ai_chance = {
			base = 1

			modifier = {
				if = {
					limit = { NOT = { has_religion = religion:deathbringer } }
					factor = 2
				}
				else_if = {
					limit = { has_religion = religion:tyrism_religion }
					factor = 12
				}
			}
		}
	}
}
