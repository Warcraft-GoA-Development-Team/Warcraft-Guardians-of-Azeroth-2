#Knights Templar
ho_christian_knights_templar = {
	
	type = military
	government = holy_order_government
	title = d_knights_templar
	name = "holy_order_knights_templar"
	coat_of_arms = "ho_knights_templar"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			scope:province.county ?= title:c_beirut
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
			}
		}
	}

	weight = {
		value = 10
		#Should be founded in jerusalem
		if = {
			limit = {
				scope:province.county ?= title:c_jerusalem
			}
			add = 1000
		}
		if = {
			limit = {
				scope:province = {
					geographical_region = world_middle_east_jerusalem
				}
			}
			add = 100
		}
		#More likely if first holy order
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							leader ?= {
								government_has_flag = government_is_military_holy_order
							}
						}
					}
				}
			}
			add = 100
		}
		
		#Less likely outside age of crusades
		if = {
			limit = {
				NOT = {
					scope:founder.religion = { exists = var:variable_ghw_unlocked }
				}
			}
			multiply = 0.1
		}
		#Trying to put them here on game start
		if = {
			limit = {
				years_from_game_start < 1
				
				scope:province.county ?= title:c_beirut
			}
			add = 1000
		}
	}

	men_at_arms = { order_knights }

}

			}

		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
				# founder is a hospital-building type
				ai_compassion >= 0
			}
		}
	}

	weight = {
		value = 10
		#Should be founded in jerusalem
		if = {
			limit = {

		}
		if = {
			limit = {
				scope:province = {
					geographical_region = world_middle_east_jerusalem
				}
			}
			add = 50
		}

		if = {
			limit = {
				scope:founder.faith = {
					any_faith_holy_order = {
						count < 2
						leader ?= {
							government_has_flag = government_is_military_holy_order
						}
					}
				}
			}
			add = 50
		}
		#More likely if founder is a hospital-building type

			limit = {
				scope:founder = {
					ai_compassion >= high_positive_ai_value
				}
			}
			add = 100
		}
		#Less likely outside age of crusades
		if = {
			limit = {
				NOT = {
					scope:founder.religion = { exists = var:variable_ghw_unlocked }
				}
			}
			multiply = 0.1
		}
		#Trying to put them here on game start
		if = {
			limit = {
				years_from_game_start < 1
				
				scope:province.county ?= title:c_jerusalem
			}
			add = 10000
		}
	}

	men_at_arms = { order_knights }

	effect = {
		scope:leader = {
			add_trait = lifestyle_physician
			add_trait_xp = {
				trait = lifestyle_physician
				value = { 25 50 }
			}
		}
	}
}

#Teutonic Knights
ho_christian_teutonic_order = {
	
	type = military
	government = holy_order_government
	title = d_teutonic_order
	name = "holy_order_teutonic_order"
	coat_of_arms = "ho_teutonic_order"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			always = no
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
				culture = {
					has_cultural_pillar = heritage_central_germanic
				}
			}
		}
	}

	weight = {
		value = 10
		#Areas of Teutonic interest
		if = {
			limit = {
				scope:province.county ?= title:c_malbork
			}
			add = 1000
		}
		if = {
			limit = {
				scope:province = {
					OR = {
						geographical_region = world_europe_west_germania
						geographical_region = world_middle_east_jerusalem
					}
				}
			}
			add = 100
		}
		if = {
			limit = {
				scope:province = {
					geographical_region = world_europe_east
				}
			}
			add = 50
		}
		
		#Less likely outside age of crusades
		if = {
			limit = {
				NOT = {
					scope:founder.religion = { exists = var:variable_ghw_unlocked }
				}
			}
			multiply = 0.1
		}
	}

	men_at_arms = { order_serjeants crossbowmen }

	effect = {
		scope:title = {
			set_variable = {
				name = order_heritage
				value = culture_pillar:heritage_central_germanic
			}
		}
		scope:leader = {
			if = {
				limit = {
					NOT = {
						has_realm_law = cultural_holy_order_succession_law
					}
				}
				add_realm_law = cultural_holy_order_succession_law
			}
		}
	}
}

	coat_of_arms = "ho_benedictine_order"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 867
			}
			scope:province.county ?= title:c_modena
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
				location ?= {
					OR = {
						geographical_region = world_europe_west
						geographical_region = world_europe_south_italy
					}
				}
			}
		}
	}

	weight = {
		value = 1
		#Should be founded in Italy
		if = {
			limit = {
				scope:province.county ?= title:c_modena
			}
			add = 5
		}
		#Trying to put benedictines here on game start
		if = {
			limit = {
				years_from_game_start < 1
				current_year >= 867
				
				scope:province.county ?= title:c_modena
			}
			add = 10000
		}
		if = {
			limit = {
				scope:province = {
					OR = {
						kingdom = title:k_sicily
						kingdom = title:k_romagna
					}
				}
			}
			add = 5
		}
	}

	effect = {
		if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 1178
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_simony }
		}
		else_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 1066
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_dulia }
		}
		else_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year >= 867
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_ora_et_labora }
		}
	}
}

#"Eremetic Benedictines" - the mixed pattern in Spain of older eremeticism and Benedictine 
ho_christian_eremetic_benedictines = {

	type = monastic
	government = monastic_holy_order_government
	title = d_eremetic_benedictines
	name = "holy_order_eremetic_benedictines"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = {
					has_variable = monastic_game_start_founder
					has_variable = 867_founder
				}
			}
			scope:province.county ?= title:c_burgos
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
			}
			#Pastoral-like
			scope:province.county = {
				any_county_province = {
					has_holding = no
				}
				OR = {
					scope:province.county = {
						county_control < 50
					}
					NOT = {
						scope:province.county = {
							save_temporary_scope_as = county_temp
							any_neighboring_county = {
								development_level < scope:county_temp.development_level
							}
						}
					}
				}
			}
		}
	}

	weight = {
		value = 10

		#More likely if the founder's rite values seclusion
		if = {
			limit = {
				scope:founder.rite ?= {
					OR = {
						rite_has_tenet = tenet_pastoral_isolation
						has_tenet_status = {
							tenet = tenet_pastoral_isolation
							status = permitted
						}
					}
				}
			}
			multiply = 2
		}

		#More likely if related to Benedictines
		if = {
			limit = {
				scope:founder = {
					has_variable = benedictine
				}
			}
			add = 10
		}
		#Eremetic tradition was v old
		if = {
			limit = {
				current_year > 1000
			}
			multiply = 0.2
		}
		#Trying to put them here on game start
		if = {
			limit = {
				years_from_game_start < 1
				
				scope:province.county ?= title:c_burgos
				scope:founder = {
					has_variable = 867_founder
				}
			}
			add = 10000
		}
	}

	effect = {
		if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
				current_year < 868
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_original_sin }
		}
	}
}

#"Pastoral Benedictines" - the lax and impoverished 9th century Anglo-Saxon Benedictines
ho_christian_pastoral_benedictines = {

	type = monastic
	government = monastic_holy_order_government
	title = d_pastoral_benedictines
	name = "holy_order_pastoral_benedictines"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
			}
			always = no
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
			}
			#Pastoral-like
			scope:province.county = {
				any_county_province = {
					has_holding = no
				}
				OR = {
					scope:province.county = {
						county_control < 50
					}
					NOT = {
						scope:province.county = {
							save_temporary_scope_as = county_temp
							any_neighboring_county = {
								development_level < scope:county_temp.development_level
							}
						}
					}
				}
			}
		}
	}

	weight = {
		value = 1

		#More likely if the founder's rite values seclusion
		if = {
			limit = {
				scope:founder.rite ?= {
					OR = {
						rite_has_tenet = tenet_pastoral_isolation
						has_tenet_status = {
							tenet = tenet_pastoral_isolation
							status = permitted
						}
					}
				}
			}
			multiply = 2
		}

		#More likely if related to Benedictines
		if = {
			limit = {
				scope:founder = {
					has_variable = benedictine
				}
			}
			add = 10
		}
		#Trying to put them here on game start
		if = {
			limit = {
				years_from_game_start < 1
				scope:province.county ?= title:c_hertfordshire
			}

		}
	}

}
