# Character scope; read-only, including the interval before old-save migration.
wc_balnazzar_suspicion_value = {
	value = 0
	if = {
		limit = {
			exists = global_var:balnazzar_dreadlord_story
			global_var:balnazzar_dreadlord_story = {
				story_owner = root
				exists = var:wc_balnazzar_suspicion
			}
		}
		add = global_var:balnazzar_dreadlord_story.var:wc_balnazzar_suspicion
		if = {
			limit = { global_var:balnazzar_dreadlord_story = { NOT = { has_variable = wc_balnazzar_suspicion_v2 } } }
			if = {
				limit = { global_var:balnazzar_dreadlord_story = { var:wc_balnazzar_suspicion >= 9 } }
				value = 100
			}
			else = { multiply = 10 }
		}
	}
	min = 0
	max = 100
}

wc_balnazzar_suspicion_fraction = {
	value = wc_balnazzar_suspicion_value
	divide = 100
}
