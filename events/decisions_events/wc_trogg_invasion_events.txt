	}

		character = title:k_gnomeregan.holder
		animation = beg
	}

	option = {
		name = {
			text = trogg_invasion.0007.option
		}

		title:k_gnomeregan.holder = {
			random_realm_county = {
				limit = {
					OR = {
						has_county_modifier = wc_high_debris_damage_modifier
						has_county_modifier = wc_medium_debris_damage_modifier
						has_county_modifier = wc_small_debris_damage_modifier 					
					}
				}
				if = {
					limit = {
						has_county_modifier = wc_high_debris_damage_modifier
					}

					remove_county_modifier = wc_high_debris_damage_modifier
					add_county_modifier = wc_medium_debris_damage_modifier
				}
				else_if = {
					limit = {
						has_county_modifier = wc_medium_debris_damage_modifier
					}

					remove_county_modifier = wc_medium_debris_damage_modifier
					add_county_modifier = wc_small_debris_damage_modifier
				}
				else_if = {
					limit = {
						has_county_modifier = wc_small_debris_damage_modifier
					}

					remove_county_modifier = wc_small_debris_damage_modifier
				}
			}
		}

		ai_chance = {
			base = 10

			modifier = {
				add = 40
				ROOT = {
					is_allied_to = title:k_gnomeregan.holder
				}
			}


			modifier = {
				add = 10
				ROOT = {
					opinion = { 
						target = title:k_gnomeregan.holder 
						value > 75
					}
				}
			}
		}
	}

	option = {
		name = {
			text = trogg_invasion.0007.optionb
		}

		ai_chance = {
			base = 40
		}

		add_prestige = -30
	}
}
