        empire = 12
        hegemony = 12
    }
    picture = {
        reference = "gfx/interface/illustrations/decisions/restore_arcandor_decision.dds"
    }
    decision_group_type = suramar

    is_shown = {
        OR = {
            AND = { 
                rite = rite:felweaver
                secret_faith = faith:faldorei_religion

        }
        NOT = { has_global_variable = moonweaver_restored_global_variable }
    }

        }
        any_vassal = { faith = faith:azji }
        is_independent_ruler = yes  
        OR = {
            AND = {
                rite = rite:felweaver
                secret_faith = faith:faldorei_religion
            }
            AND = {
                rite = rite:moonweaver 
                piety_level >= 3
            }
        }
    }

    is_valid_showing_failures_only = {
        is_imprisoned = no
        is_capable_adult = yes
        is_at_war = no
    }

    effect = {
        dynasty ?= { add_dynasty_prestige = medium_dynasty_prestige_gain }
        trigger_event = { id = wc_suramar_events.1011 }

        set_global_variable = moonweaver_restored_global_variable
    }

    ai_will_do = {
        base = 0
    }
