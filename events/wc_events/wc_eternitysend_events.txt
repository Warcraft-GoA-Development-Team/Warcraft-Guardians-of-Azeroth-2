namespace = eternitysend

#### eternitysend.0001 -> See if Cenarius is alive, if he is, then try different ways to kill him or if he is already dead start the invasion of Kaldorei
#### eternitysend.0100 -> Scope out three different ways to defeat Cenarius
#### eternitysend.0200 -> Canonical route, while harvesting wood for the new horde, warsong gets attacked by Cenarius
#### eternitysend.0201 -> Event for Cenarius and this is what declares war against the warsong clan
#### eternitysend.0202 -> Grommash is told about a well of power
#### eternitysend.0203 -> Grommash embraces the demonic powers, the rest of the clan follows
#### eternitysend.0204 -> Grommash wins the war, kill cenarius
#### eternitysend.0205 -> Grommash loses the war, start a duel with cenarius as a last ditch attempt
#### eternitysend.0206 -> Mannoroth reveals himself to be the mastermind

eternitysend.0001 = {
    scope = none
    hidden = yes

    immediate = {
        title:h_burning_legion.holder = { save_scope_as = archimonde }
        if = {
            limit = { any_living_character = { is_cenarius_trigger = yes } }
            character:410002 = { save_scope_as = cenarius }
        }

        if = {
            limit = { scope:cenarius = { is_alive = yes } }
            trigger_event = { id = eternitysend.0100 }
        }
        else = {
            scope:archimonde = {
                start_war = {
                    casus_belli = mongol_invasion_war
                    target = title:e_kaldorei.holder
                    target_title = title:k_hyjal
                }
            }
        }

        hidden_effect = {
            add_to_global_variable_list = {
                name = warcraft_invaders_story
                target = flag:eternitysend_story_started
            }
        }
    }
}

eternitysend.0100 = {
    scope = none
    hidden = yes

    immediate = {
        hidden_effect = {
            if = {
                limit = {
                    any_independent_ruler = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:warsong
                        this = culture.culture_head
                    }
                }
                random_independent_ruler = {
                    limit = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:warsong
                        this = culture.culture_head
                    }
                    save_scope_as = warsong_or_alternative
                }

                scope:warsong_or_alternative = {
                    trigger_event = { id = eternitysend.0200 }
                }
            }
            else_if = {
                limit = {
                    NOT = { exists = scope:warsong_or_alternative }
                    any_independent_ruler = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:durotaric
                        this = culture.culture_head
                    }
                }
                random_independent_ruler = {
                    limit = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:durotaric
                        this = culture.culture_head
                    }
                    save_scope_as = frostwolf_or_alternative
                }

                scope:frostwolf_or_alternative = {
                    trigger_event = { id = eternitysend.0300 }
                }
            }
            else = {
                limit = {
                    NAND = {
                        exists = scope:warsong_or_alternative
                        exists = scope:frostwolf_or_alternative
                    }
                }

                scope:archimonde = {
                    ordered_courtier = {
                        order_by = prowess
                        max = 1
                        limit = {
                            OR = {
                                has_trait = being_demon
                                has_trait = creature_fel_orc
                            }
                        }
                        save_scope_as = challenger_of_cenarius
                    }
                }

                scope:cenarius = {
                    trigger_event = { id = eternitysend.0400 }
                }
            }
        }
    }
}

eternitysend.0200 = { # Grommash and Warsong clan gets attacked
    type = character_event
	content_source = dlc_GOA
	title = eternitysend_0200_title
	desc = eternitysend_0200_desc
	theme = war
	override_background = {
		reference = wc_background_kalimdor_nature_1
	}

    immediate = {
        scope:cenarius = {
            trigger_event = { id = eternitysend.0201 }
        }
    }

    left_portrait = {
        character = scope:cenarius
        animation = personality_bold
    }
    right_portrait = {
        character = scope:warsong_or_alternative
        animation = fear
    }

    option = {
        name = eternitysend_0200_a

        scope:warsong_or_alternative = {
            trigger_event = {
                id = eternitysend.0202
                months = 9
            }

            if = {
                limit = { is_in_any_this_union_trigger = yes }
                leave_this_union_effect = yes
                add_character_modifier = { modifier = union_block_modifier }
            }
        }
    }
}

eternitysend.0202 = { # Grommash gets alerted about a well of power
    type = character_event
	content_source = dlc_GOA
	title = eternitysend_0202_title
	desc = eternitysend_0202_desc
	theme = war
	override_background = {
		reference = battlefield
	}

    immediate = {
        scope:warsong_or_alternative = {
            if = {
                limit = {
                    any_courtier = {
                        OR = {
                            has_trait = creature_orc
                            has_trait = creature_troll
                        }
                    }
                }
                random_courtier = {
                    limit = {
                        OR = {
                            has_trait = creature_orc
                            has_trait = creature_troll
                        }
                        learning >= high_skill_rating
                    }
                    save_scope_as = witch_doctor
                }
            }
        }
    }

    left_portrait = {
        character = scope:witch_doctor
        animation = storyteller
    }
    right_portrait = {
        character = scope:warsong_or_alternative
        animation = thinking
    }

    option = {
        name = eternitysend_0202_a

        custom_tooltip = {
            text = eternitysend_0202_a_tt
            scope:warsong_or_alternative = {
                trigger_event = {
                    id = eternitysend.0203
                    months = 3
                }
            }
        }
    }
}

eternitysend.0206 = {
    type = character_event
	content_source = dlc_GOA
	title = eternitysend_0206_title
	desc = eternitysend_0206_desc
	theme = war
	override_background = {
		reference = battlefield
	}

    immediate = {
        hidden_effect = {
            create_character = {
                employer = root
                template = mannoroth_template
                name = "Mannoroth"

                save_scope_as = mannoroth
            }
        }
    }

    left_portrait = {
        character = scope:mannoroth
        animation = manic
    }
    right_portrait = {
        character = root
        animation = fear
    }

    option = {
        name = eternitysend_0206_a

        custom_tooltip = { text = eternitysend_0206_a_tt }

        hidden_effect = {
            set_character_faith = faith:burning_legion_religion
            every_close_or_extended_family_member = {
                set_character_faith = faith:burning_legion_religion
            }
            every_vassal_or_below = {
                set_character_faith = faith:burning_legion_religion
            }
            every_courtier = { set_character_faith = faith:burning_legion_religion }

            every_ally = {
                break_alliance = root
            }

            if = {
                limit = { root = { highest_held_title_tier < tier_hegemony } }
                start_tributary_interaction_effect = {
                    TRIBUTARY = root
                    SUZERAIN = title:h_burning_legion.holder
                }
                title:h_burning_legion.holder = {
                    add_opinion = {
                        modifier = tributary_volunteered_opinion
                        target = root
                        opinion = 20
                    }
                    consume_all_criminal_reasons_effect = {
                        LIEGE = title:h_burning_legion.holder
                        CRIMINAL = root
                    }
                }
            }
        }
    }
}
