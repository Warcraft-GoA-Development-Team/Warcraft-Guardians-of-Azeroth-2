namespace = wc_titanforged_liberation

wc_titanforged_liberation.0002 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_titanforged_liberation.0002.title
	desc = wc_titanforged_liberation.0002.desc
	theme = realm

	immediate = {
		hidden_effect = {
			ordered_realm_de_jure_duchy = {
				limit = {
					root = { completely_controls = prev }
					any_de_jure_county = { has_county_modifier = wc_titanforged_reordered_county_modifier }
					NOT = { any_de_jure_county = { faith = { is_chaos_aligned_faith_trigger = yes } } }
					NOT = { any_de_jure_county = { development_level < 15 } }
					NOT = { any_de_jure_county = { county_control < 80 } }
					any_de_jure_county = { holder != root }
				}
				save_scope_as = ceded_duchy
			}
			scope:ceded_duchy = {
				random_de_jure_county = {
					limit = { holder != root }
					save_scope_as = ceded_vassal_county
				}
			}
			if = {
				limit = { exists = scope:ceded_vassal_county }
				scope:ceded_vassal_county = {
					holder = { save_scope_as = local_ruler }
				}
			}
		}
	}

	left_portrait = {
		character = root
		animation = emotion_thinking_scepter
	}

	option = { # Full independence, under a local ruler of the county's restored culture
		trigger = { exists = scope:local_ruler }
		name = wc_titanforged_liberation.0002.a
		give_land_to_settler = { TO_WHO = scope:local_ruler COUNTY_NAME = scope:ceded_duchy GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
		hidden_effect = {
			scope:local_ruler = {
				add_opinion = {
					modifier = wc_titanforged_grateful_for_liberation_opinion
					target = root
				}
			}
		}
	}

	option = { # Protectorate: independent under a local ruler, then bound as root's tributary
		trigger = { exists = scope:local_ruler }
		name = wc_titanforged_liberation.0002.b
		give_land_to_settler = { TO_WHO = scope:local_ruler COUNTY_NAME = scope:ceded_duchy GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
		hidden_effect = {
			scope:local_ruler = {
				start_tributary = {
					contract_group = tributary_settled
					suzerain = root
				}
				if = {
					limit = { is_tributary = yes } # suppress error logging
					tributary_contract_set_obligation_level = { type = default_tributary_taxes level = 0 }
					tributary_contract_set_obligation_level = { type = default_tributary_levies level = 0 }
					tributary_contract_set_obligation_level = { type = default_tributary_prestige level = 0 }
				}
				add_opinion = {
					modifier = wc_titanforged_grateful_for_liberation_opinion
					target = root
				}
			}
		}
	}

	option = { # Direct annexation: root keeps the duchy as ordinary crown land
		name = wc_titanforged_liberation.0002.c
		give_land_to_settler = { TO_WHO = root COUNTY_NAME = scope:ceded_duchy GIVE_INDEPENDENCY = no ADD_CLAIM_ON_LOSS = no }
	}
}
