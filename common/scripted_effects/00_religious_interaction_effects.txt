#Effects used for religion interactions

excommunicate_character = {
	if = {
		limit = {
			$REQUESTING_CHARACTER$.faith = $TARGET_CHARACTER$.faith
		}

		$EXCOMMUNICATOR$ = {
			save_scope_as = excommunicator
		}
		$REQUESTING_CHARACTER$ = {
			save_scope_as = requesting_character
		}
		$TARGET_CHARACTER$ = {
			save_scope_as = target_character
			set_variable = {
				name = excommunication_issuer
				value = scope:excommunicator
			}

			# If our religious head was convinced/persuaded to excommunicate us by another ruler, we want to make that clear.
			if = {
				limit = {
					scope:requesting_character != scope:excommunicator
				}
				add_excommunication_actual_effect = yes
				add_opinion = {
					modifier = excommunicated_opinion
					target = scope:excommunicator
					opinion = -30
				}
				add_opinion = {
					modifier = hate_opinion
					target = scope:requesting_character
					opinion = -50
				}
			}
			# Though if our head independently decided to excommunicate us, that should also be clear.
			else = {
				add_excommunication_actual_effect = yes
				add_opinion = {
					modifier = excommunicated_opinion
					target = scope:excommunicator
					opinion = -50
				}
			}

			add_piety_level = -1

			# LEGITIMACY LOSS FROM EXCOMMUNICATION
			excommunication_legitimacy_effect = yes

			if = {
				limit = {
					scope:requesting_character != scope:excommunicator
				}
				# Stored for the excommunication letter event: who asked the Pope to excommunicate you.
				set_variable = {
					name = requested_my_excommunication
					value = scope:requesting_character
				}
			}
			if = {
				limit = {
					empire_faith_gate_enabled_trigger = yes
					has_contested_empire_trigger = yes
					faith_has_existing_empire_holder_trigger = yes
				}
				# Stored for conditional letter text (empire faith gate reason).
				set_variable = {
					name = excommunication_reason
					value = flag:empire_faith_gate
				}
			}
			else_if = {
				limit = {
					scope:requesting_character = scope:excommunicator
					faith.religious_head ?= scope:excommunicator
					OR = {
						has_relation_rival = scope:excommunicator
						has_relation_nemesis = scope:excommunicator
						is_at_war_with = scope:excommunicator
					}
				}
				# Stored for conditional letter text (Pope excommunicated you on his own initiative).
				set_variable = {
					name = excommunication_reason
					value = flag:clergy_hostility
				}
			}

			# Notify other players of interest; the excommunicated player gets the recovery alert instead.
			every_player = {
				limit = {
					faith = scope:target_character.faith
					NOR = {
						this = scope:target_character # Excommunicated player uses action_excommunicated_recovery instead.
						this = scope:requesting_character # Requester already got a response from the Pope.
					}

					save_temporary_scope_as = player
					OR = {
						top_liege = scope:target_character # Our realm's top liege is a character of interest.
						top_liege = {
							any_vassal_or_below = { # Peers of the realm are also characters of interest...
								this = scope:target_character
								OR = {
									NOT = { highest_held_title_tier = 1 } #... unless they're someone else's baron (then they're too inconsequential).
									is_vassal_of = scope:player
								}
							}
						}
						any_close_family_member = {
							this = scope:target_character # Family members are characters of interest, even if not part of our realm.
						}
						any_neighboring_top_liege_realm_owner = {
							this = scope:target_character # Finally, any neighboring realm owners are characters of interest (as we get an excommunication war CB on them, as well as the ability to request claims).
						}
					}
				}
				save_scope_as = this_player
				send_interface_message = {
					type = excommunicated_other_message
					title = excommunicated_other_message
					desc = other_excommunicated_message
					left_icon = scope:target_character
					right_icon = scope:excommunicator	
				}
			}
		}

		# Head of Faith excommunication: heavier spiritual cost, and must petition Rome to be absolved (not a local bishop).
		if = {
			limit = {
				scope:target_character.faith.religious_head ?= {
					OR = {
						this = scope:requesting_character
						this = scope:excommunicator
					}
				}
			}
			scope:target_character = {
				if = {
					limit = { is_playable_character = yes }	# Only playable characters have spiritual fulfillment
					change_spiritual_fulfillment = massive_spiritual_fulfillment_loss
				}
				add_character_flag = {
					flag = pope_excom
				}

				custom_tooltip = non_hof_excom_lifting_punishment_tt
			}
		}
		else = {
			scope:target_character = {
				if = {
					limit = { is_playable_character = yes }	# Only playable characters have spiritual fulfillment
					change_spiritual_fulfillment = major_spiritual_fulfillment_loss
				}
			}
		} 

		# PAM - Great Schism: cross-rite excommunication is a grievance; anathematising the other rite's head has its own weight
		hidden_effect = {
			if = {
				limit = {
					pam_schism_head_of_rite_and_opposing_character_trigger = {
						HEAD = scope:target_character
						OTHER = scope:excommunicator
					}
				}
				pam_change_great_schism_progress_effect = {
					VALUE = pam_great_schism_progress_head_of_rite_excommunicated_value
					TT = pam_great_schism_progress_head_of_rite_excommunicated_tt
				}
			}
			else_if = {
				limit = {
					pam_schism_characters_in_opposite_pair_rites_trigger = {
						FIRST = scope:excommunicator
						SECOND = scope:target_character
					}
				}
				pam_change_great_schism_progress_effect = {
					VALUE = pam_great_schism_progress_excommunication_value
					TT = pam_great_schism_progress_excommunication_tt
				}
			}
		}
	}
}

# Clears letter-event tracking set in excommunicate_character when absolution happens.
clear_excommunication_tracking_effect = {
	remove_variable = excommunication_reason
	remove_variable = requested_my_excommunication
	remove_variable = excommunication_issuer
}

lift_character_excommunication_effect = {
	$REQUESTING_CHARACTER$ = {
		hidden_effect = {
			send_interface_toast = {	
				title = lift_excommunicate_interaction_desc.tt
				left_icon = $TARGET_CHARACTER$
				$TARGET_CHARACTER$ = {
					show_as_tooltip = { remove_trait = excommunicated }
				}
			}
		}
	}
	$TARGET_CHARACTER$ = {
		remove_trait = excommunicated
		clear_excommunication_tracking_effect = yes
		add_character_modifier = {
			modifier = excommunication_recently_lifted
			years = 10
		}
		add_opinion = {
			modifier = grateful_opinion
			target = $REQUESTING_CHARACTER$
			opinion = 30
		}
		# To nudge friendship.
		hidden_effect = {
			if = {
				limit = {
					NOR = {
						has_relation_friend = $REQUESTING_CHARACTER$
						has_relation_potential_friend = $REQUESTING_CHARACTER$
					}
				}
				set_relation_potential_friend = $REQUESTING_CHARACTER$
			}
		}
	}
}

# Saves the target kingdom to scope:target_kingdom, and the target character to scope:target_character
determine_ghw_target_effect = {
	# Find every *possible* kingdom
	$FAITH$ = {
		save_temporary_scope_value_as = {
			name = faith_strength
			value = faith_military_strength
		}
	}
	#Warcraft - remove historical event references
	
	every_kingdom = {
		limit = {
			valid_ghw_target_trigger = { FAITH = $FAITH$ }
			#Warcraft - remove historical event references
		}
		add_to_list = potential_kingdoms
	}
	#Warcraft - remove historical event references
	
	# Warcraft
	scope:the_faith = {
		random_holy_site = {
			limit = { always = yes }
			save_temporary_scope_as = holy_site
		}
	}
	
	# Pick out the 10 "best" ones
	ordered_in_list = {
		list = potential_kingdoms
		order_by = great_holy_war_target_kingdom_weight
		
		min = 0
		max = 10
		check_range_bounds = no
		
		add_to_list = filtered_potential_kingdoms
	}
	# Pick one of the top 10 at random (weighted)
	random_in_list = {
		list = filtered_potential_kingdoms
		weight = {
			base = 10
			modifier = {
				add = great_holy_war_target_kingdom_weight
			}
		}
		save_scope_as = target_kingdom
	}
	random_log_scopes = yes
	# Find a random top liege in the area
	scope:target_kingdom = {
		random_in_de_jure_hierarchy = {
			continue = { tier > tier_county }
			limit = { #Try to pick the largest ruler in the area.
				tier = tier_county
				$FAITH$.main_rite = {
					is_hostile_enough_for_holy_war_trigger = { RITE = prev.holder.rite }
				}
				holder = {
					highest_held_title_tier >= 4
					any_realm_county = { count >= 5 }
				}
			}
			alternative_limit = {
				tier = tier_county
				$FAITH$.main_rite = {
					is_hostile_enough_for_holy_war_trigger = { RITE = prev.holder.rite }
				}
				holder = {
					highest_held_title_tier >= 3
					any_realm_county = { count >= 5 }
				}
			}
			alternative_limit = {
				tier = tier_county
				$FAITH$.main_rite = {
					is_hostile_enough_for_holy_war_trigger = { RITE = prev.holder.rite }
				}
				holder = {
					highest_held_title_tier >= 2
					any_realm_county = { count >= 5 }
				}
			}
			alternative_limit = {
				tier = tier_county
				$FAITH$.main_rite = {
					is_hostile_enough_for_holy_war_trigger = { RITE = prev.holder.rite }
				}
			}
			alternative_limit = {
				tier = tier_kingdom
				$FAITH$.main_rite = {
					is_hostile_enough_for_holy_war_trigger = { RITE = prev.holder.rite }
				}
			}
			holder = {
				save_scope_as = target_character
			}
		}
	}
}

ghw_set_new_war_declarer_effect = {
	#Determine Sponsor. (effect fired within FAITH scope)
	if = {
		limit = {
			exists = religious_head
		}
		religious_head = {
			save_scope_as = ghw_sponsor
		}
		great_holy_war = {
			set_war_declarer = scope:ghw_sponsor
			if = {
				limit = {
					NOT = { has_pledged_attacker = scope:ghw_sponsor }
				}
				pledge_attacker = scope:ghw_sponsor
			}
		}
	}
}

# When someone is made to change their faith, find out which of their spouses/family will also convert.
#
# Expects:
#  scope:recipient - who's spouses/family may also convert.
#
# Returns:
#  List 'spouses_and_family_to_convert' - grabbed characters.
#
grab_spouses_and_family_to_convert_effect = {
	if = {
		limit = { NOT = { exists = scope:puppet_or_actor } }
		scope:actor = { save_scope_as = puppet_or_actor }
	}
	scope:recipient = {
		# Add yourself
		add_to_list = spouses_and_family_to_convert
		add_character_flag = spouse_or_family_to_convert
		# Spouses convert
		every_spouse = {
			limit = {
				OR = {
					is_courtier_of = scope:puppet_or_actor
					is_courtier_of = scope:recipient
					is_vassal_of = scope:puppet_or_actor
					is_vassal_of = scope:recipient
				}
				NOR = {
					has_character_flag = ai_will_not_convert
					rite = scope:puppet_or_actor.rite
					has_character_flag = spouse_or_family_to_convert 
				}
				trigger_if = {
					limit = {
						exists = scope:conversion_rite
					}
					rite != scope:conversion_rite
				}
			}
			add_to_list = spouses_and_family_to_convert
			add_character_flag = spouse_or_family_to_convert
		}
		# Family in recipient's court also convert
		if = {
			limit = {
				is_ruler = yes
			}
			every_close_or_extended_family_member = {
				custom = all_family_members_at_court
				limit = {
					employer ?= scope:recipient
					faith = scope:recipient.faith
					faith != scope:puppet_or_actor.faith
					is_ai = yes
					NOT = { has_character_flag = ai_will_not_convert }
					NOT = { has_character_flag = spouse_or_family_to_convert }
					# Don't add the recipient twice
					#root != scope:recipient
				}
				add_to_list = spouses_and_family_to_convert
				add_character_flag = spouse_or_family_to_convert
			}
		}
	}
	every_in_list = {
		list = spouses_and_family_to_convert
		remove_character_flag = spouse_or_family_to_convert
	}
}

# Default conversion effect; allows the recipient to demure whether to keep a secret faith or not
demand_conversion_interaction_effect = {
	if = {
		limit = { NOT = { exists = scope:puppet_or_actor } }
		scope:actor = { save_scope_as = puppet_or_actor }
	}
	scope:recipient.faith = {
		save_scope_as = old_faith
	}
	if = { # Confession penalty
		limit = {
			scope:puppet_or_actor = { var:pam_confession_penalty_var ?= scope:recipient }
		}
		scope:puppet_or_actor = {
			send_interface_toast = {
				title = pam_decision_events.0001.a.conversion.toast
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
				remove_variable = pam_confession_penalty_var
				remove_character_modifier = pam_confession_penalty_modifier
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					title = pam_decision_events.0001.a.conversion.toast_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
	}
	# Reward conversion
	scope:puppet_or_actor = { mandala_converter_piety_effect = yes }

	# Grab additional converters, and present them in the tooltip.
	grab_spouses_and_family_to_convert_effect = yes
	show_as_tooltip = {
		convert_family_to_faith_effect = {
			FALSE_CONVERSION = no
			FORCED = yes
		}
	}
	every_in_list = {
		list = spouses_and_family_to_convert
		# Don't say that people can convert when you know they can't
		if = {
			limit = {
				faith = scope:recipient.faith
				NOT = { has_crypto_religionist_secret_known_by_character_trigger = { CHARACTER = scope:puppet_or_actor } }
				NOT = { exists = secret_faith }
			}
			custom_tooltip = ask_for_conversion_interaction_accept_tooltip
		}
	}

	# Fire event that handles actual conversion.
	scope:recipient = {
		trigger_event = false_conversion.0900
	}
}

					flag = converted_by_forced_conversion_interaction

# Keeps the first valid bishop found by excommunicated_recovery_find_cleric_effect.

	if = {
		limit = {
			NOT = { exists = scope:repentance_cleric }

	}
}

excommunicated_recovery_find_cleric_effect = {
	# Try realm priest's superior first.
	if = {
		limit = {
			cp:councillor_court_chaplain ?= {
				highest_held_title_tier < tier_duchy
				superior ?= {
					highest_held_title_tier >= tier_duchy
					is_clergy = yes
					NOT = { this = root }
				}
			}
		}
		cp:councillor_court_chaplain.superior = {
			excommunicated_recovery_try_save_cleric_scope_effect = yes
		}
	}
	# Then the archbishop of your capital region.
	if = {
		limit = {
			NOT = { exists = scope:repentance_cleric }
			capital_county ?= {
				clerical_region_title ?= {
					holder ?= {
						highest_held_title_tier >= tier_duchy
						faith = root.faith
						NOT = { this = root }
					}
				}
			}
		}
		capital_county.clerical_region_title.holder = {
			excommunicated_recovery_try_save_cleric_scope_effect = yes
		}
	}
	# Then your religious superior if you are a theocratic ruler.
	if = {
		limit = {
			NOT = { exists = scope:repentance_cleric }
			government_has_flag = government_is_theocracy
			exists = superior
			NOT = { superior = root }
		}
		superior = {
			excommunicated_recovery_try_save_cleric_scope_effect = yes
		}
	}
	# Last resort: any valid bishop in your realm or de jure hierarchy.
	if = {
		limit = {
			NOT = { exists = scope:repentance_cleric }
		}
		every_vassal_or_below = {
			limit = {
				excommunicated_recovery_valid_cleric_for_actor_trigger = { ACTOR = root }
			}
			add_to_temporary_list = repentance_clerics
		}
		primary_title ?= {
			every_clerical_region_in_dejure_title = {
				holder ?= {
					if = {
						limit = {
							excommunicated_recovery_valid_cleric_for_actor_trigger = { ACTOR = root }
						}
						add_to_temporary_list = repentance_clerics
					}
				}
			}
		}
		every_in_list = {
			list = repentance_clerics
			excommunicated_recovery_try_save_cleric_scope_effect = yes
		}
	}
}
