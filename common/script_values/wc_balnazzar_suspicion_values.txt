# Character scope; read-only suspicion on the current story's 0-100 scale.
wc_balnazzar_suspicion_value = {
	value = 0
	if = {
		limit = {
			exists = global_var:balnazzar_dreadlord_story
			global_var:balnazzar_dreadlord_story = {
				story_owner = root
				exists = var:wc_balnazzar_suspicion
			}
		}
		add = global_var:balnazzar_dreadlord_story.var:wc_balnazzar_suspicion
	}
	min = 0
	max = 100
}

wc_balnazzar_suspicion_fraction = {
	value = wc_balnazzar_suspicion_value
	divide = 100
}

# Character scope. Only current, unresolved, free witnesses; two is the cap.
wc_balnazzar_free_witness_count = {
	value = 0
	every_courtier = {
		limit = {
			is_alive = yes
			is_imprisoned = no
			wc_balnazzar_pending_witness_trigger = yes
		}
		add = 1
	}
	max = 2
}
