namespace = wc_murder_save

# Owner POV
wc_murder_save.0002 = {
    type = character_event
    window = scheme_failed_event
    title = wc_murder_save.0002.t
    desc = wc_murder_save.0002.desc
    theme = murder_scheme
    left_portrait = {
        character = scope:target
        animation = anger
    }
    widget = {
        gui = "event_window_widget_scheme"
        container = "custom_widgets_container"
    }

    trigger = { scope:interception_unique = flag:demon_save }

    immediate = {
        murder_failure_effect = yes
    }

    option = {
        name = murder_save.0001.a
        scope:scheme = { end_scheme = yes }
    }

    option = {
        name = murder_save.0001.b
        trigger = {
            NOT = { exists = scope:scheme_discovered }
        }
        restart_murder_scheme_effect = yes
    }

    # Fire the rest of the outcome.
    after = {
        scope:target = { trigger_event = wc_murder_save.1002 }
    }
}
