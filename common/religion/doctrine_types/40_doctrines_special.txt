# Alcohol ladder
doctrine_alcohol_permitted = {
	doctrine_group_type = alcohol_restrictions
	index = 0
	divergence = 0

	icon = "special_doctrine_default"

	piety_cost = {
		value = faith_doctrine_cost_high
		if = {
			limit = { rite_has_doctrine = doctrine_alcohol_permitted }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = special_doctrine_alcohol_forbidden }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
	}

	parameters = {
		alcohol_permitted
	}
}

# Dietary ladder
# Parameters exist so a banned need can be swapped for an allowed one
# If adding more things, check how they relate to each other
special_doctrine_meat_only = {
	#Eats nothing but meat - cereals are replaced with meat rather than eaten
	doctrine_group_type = dietary_restrictions
	index = -1
	divergence = 12.5

	icon = "special_doctrine_default"

	piety_cost = {
		value = faith_doctrine_cost_high
		if = {
			limit = { rite_has_doctrine = special_doctrine_meat_only }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = special_doctrine_meat_forbidden_temples_fish }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { rite_has_doctrine = special_doctrine_meat_forbidden_temples }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
		else_if = {
			limit = {
				OR = {
					rite_has_doctrine = special_doctrine_meat_forbidden_fish
					rite_has_doctrine = special_doctrine_meat_forbidden
					rite_has_doctrine = special_doctrine_nothing_of_the_slain
				}
			}
			multiply = faith_changed_doctrine_cost_mult_four_step
		}
	}

	parameters = {
		cereals_forbidden
		meat_permitted
		leather_permitted
		diet_needs_meat
		meat_in_demand
		granary_buildings_forbidden
	}

}
