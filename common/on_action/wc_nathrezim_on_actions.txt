# Raw effect {} blocks for the same on_action don't merge across files - only the most-recently-loaded survives.
# Use on_actions = { list } to accumulate instead. This compatibility hook only raises
# newly gained vassals for a revealed Balnazzar who currently leads the Risen.
on_vassal_gained = {
	on_actions = {
		wc_nathrezim_on_vassal_gained_actions
	}
}

wc_nathrezim_on_vassal_gained_actions = {
	effect = {
		if = {
			limit = {
				root = {
					this = character:430400
					has_trait = creature_nathrezim
					has_title = title:d_wc_the_risen
					NOT = { has_character_flag = wc_balnazzar_disguise_active }
				}
				scope:vassal = {
					is_alive = yes
					NOT = { has_trait = being_undead }
				}
			}
			scope:vassal = {
				raise_undead_faith_only_effect = { MASTER = root }
				add_opinion = { modifier = loyal_undead target = root }
				# Use _no_notification variant: become_undead_effect leaks memory, can't cascade on entire vassal trees.
				every_vassal_or_below = {
					limit = {
						is_alive = yes
						NOT = { has_trait = being_undead }
					}
					raise_undead_faith_only_no_notification_effect = { MASTER = root }
					add_opinion = { modifier = loyal_undead target = root }
				}
			}
		}
	}
}
