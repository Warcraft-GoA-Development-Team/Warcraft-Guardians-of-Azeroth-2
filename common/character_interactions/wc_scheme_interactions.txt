# Balnazzar deliberately studying Dathrohan as a vessel - right-click Dathrohan
# after committing to the contingency at wc_balnazzar_story.1000.a. Restricted
# to Dathrohan specifically (see wc_balnazzar_contingency_scheme.txt's valid
# trigger) - the rest of Balnazzar's story chain hinges on this exact body.
start_wc_balnazzar_contingency_interaction = {
	category = interaction_category_hostile
	icon = icon_scheme_personal

	send_name = start_wc_balnazzar_contingency_interaction
	desc = start_wc_balnazzar_contingency_interaction_desc

	greeting = negative
	notification_text = start_wc_balnazzar_contingency_interaction_notification

	is_shown = {
		scope:actor = {
			has_character_flag = balnazzar_contingency_committed
			NOT = { has_character_flag = balnazzar_contingency_ready }
			NOT = { any_scheme = { type = wc_balnazzar_contingency } }
		}
		scope:recipient = character:60021 # Saidan Dathrohan
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { is_isolated_from_trigger = { target = scope:recipient } }
			can_start_scheme = {
				type = wc_balnazzar_contingency
				target_character = scope:recipient
			}
		}
	}

	auto_accept = yes
	common_interaction = yes
	ignores_pending_interaction_block = yes

	scheme = wc_balnazzar_contingency

	cooldown = { years = 2 }
}

# Interation to start a plague on another character
start_plague_interaction = {
	category = interaction_category_hostile
	is_highlighted = {
		always = yes
	}
	icon = death_disease

	send_name = start_plague_interaction

	interface_priority = 90
	use_diplomatic_range = {
		trigger_if = {
			limit = { scope:actor = { has_trait = in_ice_prison } }
			exists = scope:recipient.capital_province
			scope:recipient.capital_province = {
				NOR = {
					geographical_region = special_scourge_conquest_region_prio_1
					geographical_region = special_scourge_conquest_region_prio_2
				}
			}
		}
		trigger_else = {
			always = yes
		}
	}

	desc = start_plague_interaction_desc

	greeting = negative
	notification_text = start_plague_interaction_notification

	is_shown = {
		scope:actor = {
			has_trait_with_flag = plague_scheme_available
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }
		scope:actor = {
			has_dark_frenzy_trigger = { VALUE = 95 }
			
			can_start_scheme = {
				type = plague
				target_character = scope:recipient
			}
		}
	}

	on_accept = {
		scope:actor = {
			show_as_tooltip = {
				change_dark_frenzy_effect = { VALUE = -75 }
			}
			trigger_event = plague_maintenance.100
		}
	}

	auto_accept = yes

	cost = {
	}

	ignores_pending_interaction_block = yes
	common_interaction = yes

	scheme = plague

	cooldown_against_recipient = {
		years = 10
	}
	cooldown = {
		years = 5
	}
}
