namespace = felblood

#### felblood.0001 -> You ask your demon liege for a favour
#### felblood.0002 -> You end the life of your demon prisoner for their powers
#### felblood.0003 -> You wonder if you wanna make your family member and your vassals to do the same
#### felblood.0004 -> Hidden event to convert your family and your vassals to fel orcs

felblood.0001 = {
    type = character_event
    content_source = dlc_GOA
    title = felblood_0001_title
    desc = felblood_0001_desc
    theme = horde

    immediate = {
        if = {
            limit = {
                exists = liege
                liege = {
                    has_trait = being_demon
                    NOT = { has_trait = demon_hunter }
                }
            }
            liege = { save_scope_as = demon_liege }
        }
        else = {
            limit = {
                exists = top_liege
                top_liege = {
                    has_trait = being_demon
                    NOT = { has_trait = demon_hunter }
                }
            }
            top_liege = { save_scope_as = demon_liege }
        }
    }

    left_portrait = {
        character = root
        animation = throne_room_kneel_1
    }
    right_portrait = {
        character = scope:demon_liege
        animation = dismissal
    }

    option = {
        name = felblood_0001_a
        add_trait = creature_fel_orc
        hidden_effect = {
            trigger_event = felblood.0003
        }
    }
}

felblood.0002 = {
    type = character_event
    content_source = dlc_GOA
    title = felblood_0002_title
    desc = felblood_0002_desc
    theme = horde

    immediate = {
        random_prisoner = {
            limit = {
                has_trait = being_demon
			    NOT = { has_trait = demon_hunter }
            }
			save_scope_as = demon_prisoner
		}
    }

    left_portrait = {
        character = root
        animation = assassin
    }
    right_portrait = {
        character = scope:demon_prisoner
        animation = fear
    }

    option = {
        name = felblood_0002_a
        add_trait = creature_fel_orc
        scope:demon_prisoner = {
            override_death_killer_effect = {
                death_reason = death_sacrificed_to_gods
                killer = root
            }
        }
        hidden_effect = {
            trigger_event = felblood.0003
        }
    }
}

felblood.0003 = {
    type = character_event
    content_source = dlc_GOA
    title = felblood_0003_title
    desc = felblood_0003_desc
    theme = horde

    right_portrait = {
        character = root
        animation = worry
    }

    option = {
        name = felblood_0003_a
        custom_tooltip = felblood_0003_a_tt
        hidden_effect = {
            every_close_or_extended_family_member = {
                remove_trait = creature_orc
            }
            every_vassal_or_below = {
                limit = {
                    OR = {
                        has_faith = faith:burning_legion_religion
                        has_faith = faith:orcish_fel
                        has_faith = faith:black_harvest
                    }
                }
                remove_trait = creature_orc
            }

            trigger_event = felblood.0004
        }
    }
    option = {
        name = felblood_0003_b
    }
}

felblood.0004 = {
    hidden = yes

    immediate = {
        hidden_effect = {
            every_close_or_extended_family_member = {
                add_trait = creature_fel_orc
            }
            every_vassal_or_below = {
                limit = {
                    OR = {
                        has_faith = faith:burning_legion_religion
                        has_faith = faith:orcish_fel
                        has_faith = faith:black_harvest
                    }
                }
                add_trait = creature_fel_orc
            }
        }
    }
}
