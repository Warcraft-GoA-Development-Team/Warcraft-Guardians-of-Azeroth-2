scripted_effect become_landless_terenia = {
	save_scope_as = adventurer
	if = {
		limit = {
			player_heir ?= {
				NOT = { this = root }
				is_adult = yes
			}
		}
		player_heir = {
			save_scope_as = laamp_heir
			save_scope_as = laamp_inheritor
			#custom_tooltip = become_landless_adventurer_decision_title_scope_tt
		}
	}
	else_if = {
		limit = { is_independent_ruler = no }
		liege = {
			save_scope_as = laamp_liege
			save_scope_as = laamp_inheritor
			custom_tooltip = become_landless_adventurer_decision_title_scope_tt
		}
	}
	else = {
		custom_tooltip = become_landless_adventurer_decision_title_created_tt
		if = {
			limit = {
				NOT = { exists = scope:laamp_inheritor }
			}
			hidden_effect_new_object = {
				create_character = {
					template = laamp_inheritor_template
					culture = root.capital_county.culture
					faith = root.capital_county.faith
					dynasty = generate
					location = root.capital_province
					after_creation = {
						save_scope_as = laamp_inheritor
						add_character_flag = new_laamp_inheritor
					}
				}
			}
		}
	}
	create_landless_adventurer_title_tooltip_effect = yes
	show_as_tooltip = {
		add_prestige = medium_prestige_gain
		add_character_modifier = {
			modifier = ep3_voluntary_laamp_character_modifier
			years = 5
		}
	}
	trigger_event = { id = ep3_laamps.0030 }
	# Debug tracking
	if = {
		limit = {
			is_ai = yes
			debug_only = yes
		}
		if = {
			limit = { exists = global_var:ai_voluntary_laamp_count }
			change_global_variable = {
				name = ai_voluntary_laamp_count
				add = 1
			}
		}
		else = {
			set_global_variable = {
				name = ai_voluntary_laamp_count
				value = 1
			}
		}
		debug_log = "AI became adventurer voluntarily"
		debug_log_scopes = yes
	}
	if = {
		limit = { this = character:60677 NOT = { has_trait = eccentric } OR = { number_of_personality_traits < 4 has_trait = fickle has_trait = stubborn } }
		if = { limit = { has_trait = fickle } remove_trait = fickle }
		if = { limit = { has_trait = stubborn } remove_trait = stubborn }
		add_trait = eccentric
	}
}

scripted_effect education_update_terenia = {
	random_list = {
		1 = {
			modifier = {
				add = 2
				has_trait = education_$education_type$_1
			}
			modifier = {
				add = -1
				has_trait = education_$education_type$_5
			}
			root = { add_$education_type$_skill = 1 }
		}
		1 = {
			modifier = {
				add = 1
				has_trait = education_$education_type$_2
			}
			modifier = {
				add = 2
				has_trait = education_$education_type$_3
			}
			modifier = {
				add = 1
				has_trait = education_$education_type$_4
			}
			root = { add_$education_type$_skill = 2 }
		}
		1 = {
			modifier = {
				add = 1
				has_trait = education_$education_type$_3
			}
			modifier = {
				add = 3
				has_trait = education_$education_type$_4
			}
			modifier = {
				add = 5
				has_trait = education_$education_type$_5
			}
			root = { add_$education_type$_skill = 3 }
		}
	}
}
scripted_effect get_my_guardian_stats = {
	var:my_guardian = {
		if = {
			limit = { has_trait = education_diplomacy }
			education_update_terenia = { education_type = diplomacy }
		}
		else_if = {
			limit = { has_trait = education_martial }
			education_update_terenia = { education_type = martial }
		}
		else_if = {
			limit = { has_trait = education_stewardship }
			education_update_terenia = { education_type = stewardship }
		}
		else_if = {
			limit = { has_trait = education_intrigue }
			education_update_terenia = { education_type = intrigue }
		}
		else_if = {
			limit = { has_trait = education_learning }
			education_update_terenia = { education_type = learning }
		}
	}
}
scripted_effect check_my_guardian_variable = {
	if = {
		limit = {
			var:my_guardian = { is_alive = no }
		}
		if = {
			limit = {
				any_relation = {
					type = guardian
				}
			}
			random_relation = {
				type = guardian
				root = { set_variable = { name = my_guardian value = prev } }
			}
		}
		else = {
			random_courtier = {
				limit = { is_adult = yes }
				root = { set_variable = { name = my_guardian value = prev } }
			}
		}
	}
}
scripted_effect sm_pa_swear_fealty = {
	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = changing_liege
	}
	change_liege = {
		liege = $NEW_LIEGE$
		change = scope:changing_liege
	}
	resolve_title_and_vassal_change = scope:changing_liege
}
scripted_effect sm_pa_give_title = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = $ADD_CLAIM_ON_LOSS$
	}
	$COUNTY_NAME$ = {
		change_title_holder = {
			holder = $TO_WHO$
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change
	#make independent
	if = {
		limit = {
			always = $GIVE_INDEPENDENCY$
			$TO_WHO$ = { is_independent_ruler = no }
		}
		hidden_effect = {
			$TO_WHO$ = {
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = change
					add_claim_on_loss = no
				}
				becomes_independent = { change = scope:change }
				resolve_title_and_vassal_change = scope:change
			}
		}
	}
}
scripted_effect sm_remove_existing_guardian = {
	# Handle existing guardian
	$ward$ = {
		if = {
			limit = {
				any_relation = {
					type = guardian
				}
			}
			random_relation = {
				type = guardian
				save_scope_as = old_guardian
			}
			root = {
				remove_guardian_effect = {
					GUARDIAN = scope:old_guardian
					WARD = $ward$
					RETURN_WARD = yes
					HIDE_OPINION = no
				}
			}
		}
	}
}
scripted_effect sm_ward_depart_effect = {
	$guardian$ = {
		add_to_list = toast_list
		if = {
			limit = { is_ruler = yes }
			save_scope_as = guardian_liege
		}
		else_if = {
			limit = { exists = liege }
			liege = { save_scope_as = guardian_liege }
		}
		else = { save_scope_as = guardian_liege }
	}
	$ward$ = {
		add_to_list = toast_list
		if = {
			limit = { is_ruler = yes }
			save_scope_as = ward_liege
		}
		else_if = {
			limit = { exists = liege }
			liege = { save_scope_as = ward_liege }
		}
		else = { save_scope_as = ward_liege }
	}
	scope:guardian_liege = { add_to_list = toast_list }
	scope:ward_liege = { add_to_list = toast_list }
	# Set variables
	$ward$ = {
		set_variable = {
			name = character_making_education_request
			value = root
			years = 10
		}
		# Remove existing guardian if relevant
		if = {
			limit = {
				exists = scope:offer_guardianship_interaction
				any_relation = { type = guardian }
			}
			random_relation = {
				type = guardian
				save_scope_as = guardian_to_remove
			}
			remove_guardian_effect = {
				GUARDIAN = scope:guardian_to_remove
				WARD = $ward$
				RETURN_WARD = no
				HIDE_OPINION = no
			}
		}
		remove_character_flag = under_offer_as_ward_flag
	}
	if = { # No travel necessary
		limit = {
			OR = {
				$ward$ = $guardian$.host
				$guardian$ = $ward$.host
				$ward$.host = $guardian$.host
				#allow landed ruler to landed ruler education
				AND = {
					$ward$ = {
						is_landed_or_landless_administrative = yes
					}
					$guardian$ = {
						is_landed_or_landless_administrative = yes
					}
				}
			}
		}
		hidden_effect = { #to prevent doubling of "becomes guardian of x" effect in the interaction confirmation window
			$guardian$ = { set_relation_ward = $ward$ }
		}
		#Opinions
		guardian_add_opinion_effect = {
			GUARDIAN = $guardian$
			WARD_LIEGE = scope:ward_liege
			WARD = $ward$
		}
		# Convert culture setup
		$ward$ = {
			# Convert faith setup
			if = {
				limit = { has_character_flag = convert_faith }
				$guardian$ = {
					add_relation_flag = {
						relation = ward
						target = $ward$
						flag = convert_faith
					}
				}
				remove_character_flag = convert_faith
			}
		}
	}
	else = {
		if = { # Guardian heads to landed Ward
			limit = {
				$ward$ = { is_playable_character = yes }
			}
			$guardian$ = {
				set_variable = {
					name = guardian_travelling_to_ward
					years = 2
					value = $ward$
				}
				start_travel_plan = {
					destination = $ward$.capital_province
					on_start_on_action = on_guardian_depart_for_ward
					on_travel_planner_cancel_on_action = on_guardian_depart_travel_planner_exit
					on_arrival_on_action = on_guardian_arrive_at_ward
					on_arrival_destinations = last
					return_trip = no # One way
				}
			}
			hidden_effect = {
				every_in_list = {
					list = toast_list
					send_interface_toast = {
						title = guardian_departs_title
						left_icon = $guardian$
						right_icon = $ward$
						custom_tooltip = guardian_departs_tt
					}
				}
			}
			custom_tooltip = guardian_leaves_for_ward_tt
			$ward$ = {
				set_variable = {
					name = ward_waiting_for_guardian
					years = 2
					value = $guardian$
				}
			}
		}
		else = {
			$ward$ = {
				if = {
					limit = { # Ward heads to landed Guardian
						is_playable_character = no
						scope:guardian = { is_playable_character = yes }
						NOT = { scope:guardian = host }
					}
					$guardian$.capital_province = { save_scope_as = destination_province }
				}
				else_if = { # Ward heads to Guardian's host
					limit = { exists = $guardian$.host.capital_province }
					$guardian$.host.capital_province = { save_scope_as = destination_province }
				}
				else = {
					$guardian$.location = { save_scope_as = destination_province }
				}
				set_variable = {
					name = ward_travelling_to_guardian
					years = 2
					value = $guardian$
				}
				start_travel_plan = {
					destination = scope:destination_province
					on_start_on_action = on_ward_depart_for_guardian
					on_travel_planner_cancel_on_action = on_ward_depart_travel_planner_exit
					on_arrival_on_action = on_ward_arrive_at_guardian
					on_arrival_destinations = last
					return_trip = no # One way
				}
			}
			hidden_effect = {
				every_in_list = {
					list = toast_list
					send_interface_toast = {
						title = ward_departs_title
						left_icon = $ward$
						right_icon = $guardian$
						custom_tooltip = ward_departs_tt
					}
				}
			}
			custom_tooltip = ward_leaves_for_guardian_tt
			$guardian$ = {
				set_variable = {
					name = guardian_waiting_for_ward
					years = 2
					value = $ward$
				}
			}
		}
	}
}

scripted_effect sm_pa_declare_war = {
	hidden_effect = {
		if = {
			limit = { has_truce = $war_target$ }
			cancel_truce_both_ways = $war_target$
		}
	}
	if = {
		limit = {
			NOT = { faith = $war_target$.faith }
			faith = {
				faith_hostility_level = {
					target = $war_target$.faith
					value >= religious_cb_enabled_hostility_level
				}
			}
		}
		start_war = {
			casus_belli = terenia_major_religious_war
			target = $war_target$
			target_title = title:k_lordaeron
		}
	}
	else = {
		start_war = {
			cb = invasion_war
			target = $war_target$
			target_title = title:k_lordaeron
		}
	}
	$war_target$ = {
		random_sub_realm_county = {
			limit = { kingdom = title:k_lordaeron }
			title_province = { save_scope_as = spawn_army_location }
		}
	}
	if = {
		limit = { is_at_war_as_attacker = yes }
		random_character_war = {
			limit = { primary_attacker = root primary_defender = $war_target$ }
			save_temporary_scope_as = this_war
		}
		if = {
			limit = { exists = scope:this_war }
			spawn_army = {
				uses_supply = no
				inheritable = yes
				name = partisan_army_tt
				levies = {
					value = 300
				}
				location = scope:spawn_army_location
				origin = capital_province
				war = scope:this_war
			}
			hidden_effect = {
				if = {
					limit = {
						has_variable_list = supporters_of_terenia_menethil
					}
					every_in_list = {
						variable = supporters_of_terenia_menethil
						limit = { NOR = { is_at_war_with = root is_at_war_with = $war_target$ is_allied_to = $war_target$ } }
						scope:this_war = {
							set_called_to = prev
							add_attacker = prev
						}
					}
				}
			}
		}
	}
	else = {
		show_as_tooltip = {
			spawn_army = {
				uses_supply = no
				inheritable = yes
				name = partisan_army_tt
				levies = {
					value = 300
				}
				location = scope:spawn_army_location
				origin = capital_province
			}
		}
	}
}
