namespace = forsaken_kingdom

# forsaken_kingdom.0001	Bandarion Keep and Tyr's Watch (liege at Garek's death)
# forsaken_kingdom.0002	Tyr's Watch founded (news: Holy Light players, rulers who hear from Lordaeron)
# forsaken_kingdom.0003	The Light calls on us to protect the living (liege: Garek's Naxxramas sacrifice)

#############################################
# Bandarion Keep and Tyr's Watch
# Fired by wc_bandarion_keep_death_on_actions one day after Garek Bandarion dies
# holding b_autumnacre. Root: his liege at death, who sponsors the order.
# scope:garek is the dead captain, saved by the hook.
# The opening depends on wc_naxxramas_sacrifice, set by forsaken_kingdom.0003: the
# sacrifice text only if he burned at Naxxramas, a neutral text for any other death.
# Always renames the barony. If Danitha Morr can lead an order and the barony can be
# leased, she returns to the Light and founds Tyr's Watch there, forsaken_kingdom.0002
# goes out as news, and Evin Schmidtt, if he can, returns too and joins the order as
# keeper of Bandarion Keep.
# Rename only (accepted): Danitha dead, a child, landed or already converted; or the
# barony went to an heir whose primary title is above barony tier.
#############################################
forsaken_kingdom.0001 = {
	type = character_event
	content_source = dlc_GOA
	title = forsaken_kingdom.0001.t
	desc = {
		first_valid = {	# Opening: how he died
			triggered_desc = {
				trigger = { has_global_variable = wc_naxxramas_sacrifice }
				desc = forsaken_kingdom.0001.desc.opening.sacrifice
			}
			desc = forsaken_kingdom.0001.desc.opening.neutral
		}
		first_valid = {	# Body: Danitha steps forward, or no one does
			triggered_desc = {
				trigger = { exists = scope:tyrs_watch_founded }
				desc = forsaken_kingdom.0001.desc.founded
			}
			desc = forsaken_kingdom.0001.desc.rename_only
		}
		triggered_desc = {	# Keeper, only when Evin joined
			trigger = { exists = scope:evin }
			desc = forsaken_kingdom.0001.desc.keeper
		}
	}
	theme = death
	override_background = { reference = wc_undercity_throne_room }

	left_portrait = {
		character = root
		animation = sadness
	}
	# Only when Danitha steps forward; the trigger draws nothing in the rename-only branch
	right_portrait = {
		trigger = { exists = scope:danitha }
		character = scope:danitha
		animation = prayer
	}
	lower_center_portrait = scope:garek
	# Only when Evin joins as keeper
	lower_right_portrait = {
		trigger = { exists = scope:evin }
		character = scope:evin
		animation = prayer
	}

	immediate = {
		play_music_cue = "mx_cue_death"

		title:b_autumnacre = { set_title_name = bandarion_keep_custom_name }

		if = {
			limit = {
				character:danitha_morr ?= {
					is_alive = yes

					is_adult = yes

					is_landed = no

					has_character_flag = wc_awaits_the_light
				}
				title:b_autumnacre = {
					barony_is_valid_for_holy_order_lease_trigger = { CHARACTER = root }
				}
			}
			save_scope_value_as = { name = tyrs_watch_founded value = yes }

			character:danitha_morr = {
				save_scope_as = danitha

				hidden_effect = {
					set_variable = { name = wc_light_magic_lifestyle_additional_perks_variable value = wc_perks_needed_for_level_2_magic_trait_value }

					remove_character_flag = wc_awaits_the_light
				}

				set_character_faith = faith:holy_light

				hidden_effect = {
					if = {
						limit = { NOT = { has_perk = light_magic_tree_1_perk_1 } }
						add_perk = light_magic_tree_1_perk_1
					}
				}
			}
			# Evin returns to the Light too and keeps the keep for the order, if he can
			if = {
				limit = {
					character:evin_schmidtt ?= {
						is_alive = yes

						is_adult = yes

						is_landed = no

						has_character_flag = wc_awaits_the_light
					}
				}
				character:evin_schmidtt = {
					save_scope_as = evin

					hidden_effect = {
						set_variable = { name = wc_light_magic_lifestyle_additional_perks_variable value = wc_perks_needed_for_level_2_magic_trait_value }

						remove_character_flag = wc_awaits_the_light
					}

					set_character_faith = faith:holy_light

					hidden_effect = {
						if = {
							limit = { NOT = { has_perk = light_magic_tree_1_perk_1 } }
							add_perk = light_magic_tree_1_perk_1
						}
					}
				}
			}
			# Only a founded order exempts the barony from lease revocation
			hidden_effect = {
				title:b_autumnacre = { set_variable = wc_tyrs_watch_seat }
			}
			wc_found_tyrs_watch_effect = yes

			# Danitha now holds the order and has a court. Evin could not be her courtier
			# before, so the founding effect's every_courtier pass missed him: add him here.
			if = {
				limit = { exists = scope:evin }
				hidden_effect = {
					scope:evin = { add_trait = order_member }
					scope:danitha = { add_courtier = scope:evin }
				}
			}

			# News: Holy Light players anywhere, and anyone who hears from Lordaeron
			hidden_effect = {
				every_player = {
					limit = {
						this != root

						OR = {
							faith = faith:holy_light
							get_news_from_region_trigger = { REGION = world_eastern_kingdoms_lordaeron }
						}
					}
					trigger_event = {
						id = forsaken_kingdom.0002
						days = { 3 7 }
					}
				}
			}
		}

		stress_impact = { base = minor_stress_impact_gain }
	}

	option = {	# Acknowledge; no choice by design
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { exists = scope:tyrs_watch_founded }
						desc = forsaken_kingdom.0001.a.founded
					}
					desc = forsaken_kingdom.0001.a.rename_only
				}
			}
		}
	}
}

#############################################
# Tyr's Watch founded (news)
# Fired by forsaken_kingdom.0001 three to seven days after the founding.
# Root: the viewer. Holy Light players anywhere, and anyone who gets news from
# Lordaeron, except the sponsor. Text varies by the viewer's faith; a keeper line
# follows when Evin joined and is still alive.
# Purely informational: no effects.
#############################################
forsaken_kingdom.0002 = {
	type = character_event
	content_source = dlc_GOA
	title = forsaken_kingdom.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { faith = faith:holy_light }
				desc = forsaken_kingdom.0002.desc.light
			}
			triggered_desc = {
				trigger = { faith = faith:forsaken_cult }
				desc = forsaken_kingdom.0002.desc.forsaken
			}
			desc = forsaken_kingdom.0002.desc.other
		}
		triggered_desc = {	# Days have passed; Evin may have died
			trigger = { scope:evin ?= { is_alive = yes } }
			desc = forsaken_kingdom.0002.desc.keeper
		}
	}
	theme = faith
	override_background = { reference = temple_church }

	# Days have passed since the founding
	trigger = {
		exists = scope:danitha
		scope:danitha = { is_alive = yes }
	}

	left_portrait = {
		character = scope:danitha
		animation = prayer
	}
	lower_center_portrait = scope:garek

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
	}

	option = {	# Close
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { faith = faith:holy_light }
						desc = forsaken_kingdom.0002.a.light
					}
					triggered_desc = {
						trigger = { faith = faith:forsaken_cult }
						desc = forsaken_kingdom.0002.a.forsaken
					}
					desc = forsaken_kingdom.0002.a.other
				}
			}
		}
	}
}

#############################################
# The Light calls on us to protect the living
# Fired by wc_naxxramas_sacrifice_on_action on 1 April 607 while Garek Bandarion
# holds Autumnacre (or a later April, if his liege changed while it was queued).
# Root: Garek's liege. scope:garek is the captain, alive until immediate runs.
# Dark Ranger Anya tells it if she is alive; otherwise the Deathguard report it.
# Garek takes the Light and dies (death_burned_magic). His death runs
# wc_bandarion_keep_death_on_actions, which fires forsaken_kingdom.0001 a day later;
# wc_naxxramas_sacrifice tells 0001 to use the sacrifice opening.
#############################################
forsaken_kingdom.0003 = {
	type = character_event
	content_source = dlc_GOA
	title = forsaken_kingdom.0003.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:anya }
				desc = forsaken_kingdom.0003.desc.anya
			}
			desc = forsaken_kingdom.0003.desc.deathguard
		}
	}
	theme = death
	override_background = { reference = wc_undercity_throne_room }

	# Ninety days have passed since the on_action queued this
	trigger = {
		exists = scope:garek
		scope:garek = {
			is_alive = yes

			has_title = title:b_autumnacre
			liege = root

			has_character_flag = wc_captain_of_the_deathguard
		}
	}

	# Anya tells it; drawn only when she is alive to tell it
	left_portrait = {
		trigger = { exists = scope:anya }
		character = scope:anya
		animation = sadness
	}
	right_portrait = {
		character = root
		animation = shock
	}
	lower_center_portrait = {
		character = scope:garek
		animation = prayer
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"

		hidden_effect = {
			character:dark_ranger_anya ?= {
				if = {
					limit = { is_alive = yes }
					save_scope_as = anya
				}
			}

			# Marker for forsaken_kingdom.0001's opening: death_burned_magic alone is not
			# proof, the Malevolence spell kills with the same reason
			set_global_variable = wc_naxxramas_sacrifice
		}

		scope:garek = {
			set_character_faith = faith:holy_light
			add_trait = holy_warrior

			death = { death_reason = death_burned_magic }
		}
	}

	option = {	# Acknowledge; no choice by design
		name = forsaken_kingdom.0003.a
	}
}
