wc_culture_tradition_events.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		set_variable = is_performing_a_dark_ritual
		trigger_event = { id = wc_culture_tradition_events.0002 days = 1 }
	}
}

scripted_effect random_injury_fate_effect = {
	random_list = {
		35 = { # Wounded
			show_as_tooltip = { add_trait_force_tooltip = wounded_1 }
			save_scope_value_as = { name = injury_fate value = 5 }
		}
		15 = { # Severely Injured
			show_as_tooltip = { add_trait_force_tooltip = wounded_2 }
			save_scope_value_as = { name = injury_fate value = 4 }
		}
		4 = { # Brutally Mauled
			show_as_tooltip = { add_trait_force_tooltip = wounded_3 }
			save_scope_value_as = { name = injury_fate value = 3 }
		}
		1 = { # Die
			show_as_tooltip = { override_death_effect = { death_reason = death_suicide } }
			save_scope_value_as = { name = injury_fate value = 2 }
		}
		45 = {
			save_scope_value_as = { name = injury_fate value = 1 }
		} # Nothing
	}
}

scripted_effect apply_random_injury_effect = {
	switch = {
		trigger = scope:injury_fate

		5 = { add_trait_force_tooltip = wounded_1 } # Wounded
		4 = { add_trait_force_tooltip = wounded_2 } # Severly Injured
		3 = { add_trait_force_tooltip = wounded_3 } # Brutally Mauled
	}
}

scripted_effect sacrifice_progression_effect = {
	if = {
		limit = {
			has_trait = blind
			has_trait = one_legged
		}
		add_trait_force_tooltip = maimed
		save_scope_value_as = { name = sacrifice_fate value = 4 }
	}
	else_if = {
		limit = {
			has_trait = blind
			NOT = { has_trait = one_legged }
		}
		add_trait_force_tooltip = one_legged
		save_scope_value_as = { name = sacrifice_fate value = 3 }
	}
	else_if = {
		limit = {
			has_trait = one_eyed
			NOT = { has_trait = blind }
		}

		remove_trait = one_eyed
		add_trait_force_tooltip = blind
		save_scope_value_as = { name = sacrifice_fate value = 2 }
	}
	else = {
		add_trait_force_tooltip = one_eyed
		save_scope_value_as = { name = sacrifice_fate value = 1 }
	}
}

scripted_effect random_consequences_effect = {
	random_list = {
		10 = { # No vision
			save_scope_value_as = { name = vision_fate value = 1 }
		}
		25 = { # Weak vision
			save_scope_value_as = { name = vision_fate value = 2 }
		}
		60 = { # Normal vision
			save_scope_value_as = { name = vision_fate value = 3 }
		}
		5 = { # Strong vision
			save_scope_value_as = { name = vision_fate value = 4 }
		}
	}
}

wc_culture_tradition_events.0002 = {
	type = character_event
	title = wc_culture_tradition_events.0002.t
	desc = {
		first_valid = {
			triggered_desc = { # Chieftain
				trigger = { culture.culture_head = root }
				desc = wc_culture_tradition_events.0002.desc.chieftain
			}
			triggered_desc = { # Vassal
				trigger = { culture.culture_head.primary_heir = root }
				desc = wc_culture_tradition_events.0002.desc.heir
			}
			desc = wc_culture_tradition_events.0002.desc.others # Others
		}
	}

	theme = bleeding_hollow
	override_background = {
		trigger = { is_travelling = yes }
		reference = terrain
	}
	override_background = {
		trigger = { is_travelling = no }
		reference = corridor_night
	}

	left_portrait = {
		character = root
		animation = thinking
	}

	option = { # Accept
		name = wc_culture_tradition_events.0002.a
		custom_tooltip = wc_culture_tradition_events.0002.a_tt

		show_as_tooltip = { sacrifice_progression_effect = yes }
		random_injury_fate_effect = yes

		# Vision
		hidden_effect = { random_consequences_effect = yes }

		trigger_event = {
			id = wc_culture_tradition_events.0003
			days = 10
		}

		stress_impact = {
			brave = medium_stress_impact_loss
			craven = major_stress_impact_gain
			zealous = minor_stress_impact_loss
		}

		ai_chance = {
			base = 50

			modifier = {
				health = fine_health
				factor = 1.5
			}
			modifier = {
				health >= good_health
				factor = 2
			}
			modifier = {
				add = 50
				has_trait = brave
			}
		}
	}

	option = { # Coward
		name = wc_culture_tradition_events.0002.b
		custom_tooltip = wc_culture_tradition_events.0002.b_tt

		add_prestige = major_prestige_loss

		stress_impact = {
			brave = medium_stress_impact_gain
			craven = medium_stress_impact_loss
			zealous = minor_stress_impact_gain
		}

		remove_variable = is_performing_a_dark_ritual

		ai_chance = {
			base = 50

			modifier = {
				health <= poor_health
				factor = 0.5
			}
			modifier = {
				health <= death_chance_starts_health
				factor = 0.1
			}
			modifier = {
				add = 50
				has_trait = craven
			}
		}
	}
}

scripted_effect strong_vision_effect = {
	if = { # Saw death
		limit = { NOT = { has_perk = know_thyself_perk } }
		add_perk = know_thyself_perk 
	}

	# Become stronger in mystic
	if = {
		limit = { has_trait = lifestyle_mystic }

		add_trait_xp = {
			trait = lifestyle_mystic
			value = 70
		}
	}
	else = {
		add_trait_force_tooltip = lifestyle_mystic

		add_trait_xp = {
			trait = lifestyle_mystic
			value = 50
		}
	}
}

scripted_effect normal_vision_effect = {
	if = { # Saw death
		limit = { NOT = { has_perk = know_thyself_perk } }
		add_perk = know_thyself_perk
	}

	# Become stronger in mystic
	if = {
		limit = { has_trait = lifestyle_mystic }

		add_trait_xp = {
			trait = lifestyle_mystic
			value = 25
		}
	}
	else = { add_trait_force_tooltip = lifestyle_mystic }
}

scripted_effect weak_vision_effect = {
	if = { # Saw death
		limit = { NOT = { has_perk = know_thyself_perk } }
		add_perk = know_thyself_perk
	}

	stress_impact = { base = medium_stress_impact_gain }

	add_prestige = medium_prestige_loss
}

scripted_effect no_vision_effect = {
	stress_impact = { base = major_stress_impact_gain }

	add_prestige = major_prestige_loss
}

wc_culture_tradition_events.0003 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:vision_fate = 1 }
				desc = wc_culture_tradition_events.0003.t.no_vision
			}
			desc = wc_culture_tradition_events.0003.t
		}
	}
	desc = {
		# 1 - What part is sacrificed
		first_valid = {
			triggered_desc = {
				trigger = { scope:sacrifice_fate = 1 }
				desc = wc_culture_tradition_events.0002.desc.sac_eye
			}
			triggered_desc = {
				trigger = { scope:sacrifice_fate = 2 }
				desc = wc_culture_tradition_events.0002.desc.sac_blind
			}
			triggered_desc = {
				trigger = { scope:sacrifice_fate = 3 }
				desc = wc_culture_tradition_events.0002.desc.sac_leg
			}
			desc = wc_culture_tradition_events.0002.desc.maimed
		}

		# 2 - What vision do you have
		first_valid = {
			triggered_desc = { # Strong vision
				trigger = { scope:vision_fate = 4 }
				desc = wc_culture_tradition_events.0002.desc.strong_vision
			}
			triggered_desc = { # Normal vision
				trigger = { scope:vision_fate = 3 }
				desc = wc_culture_tradition_events.0002.desc.normal_vision
			}
			triggered_desc = { # Weak vision
				trigger = { scope:vision_fate = 2 }
				desc = wc_culture_tradition_events.0002.desc.weak_vision
			}
			desc = wc_culture_tradition_events.0002.desc.no_vision # No vision
		}

		# 3 - What injury do you have
		first_valid = {
			triggered_desc = { # Die, no vision
				trigger = {
					scope:injury_fate = 2
					scope:vision_fate = 1
				}
				desc = wc_culture_tradition_events.0002.desc.dying_no_vision
			}
			triggered_desc = { # Die
				trigger = { scope:injury_fate = 2 }
				desc = wc_culture_tradition_events.0002.desc.dying
			}
			triggered_desc = { # Brutally mauled
				trigger = { scope:injury_fate = 3 }
				desc = wc_culture_tradition_events.0002.desc.brutally_mauled
			}
			triggered_desc = { # Severely injured
				trigger = { scope:injury_fate = 4 }
				desc = wc_culture_tradition_events.0002.desc.severely_injured
			}
			triggered_desc = { # Wounded
				trigger = { scope:injury_fate = 5 }
				desc = wc_culture_tradition_events.0002.desc.wounded
			}
			desc = wc_culture_tradition_events.0002.desc.no_injury # You're totally safe
		}
	}

	theme = bleeding_hollow
	override_background = {
		trigger = { is_travelling = yes }
		reference = terrain
	}
	override_background = {
		trigger = { is_travelling = no }
		reference = corridor_night
	}

	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { scope:vision_fate = 4 }
			animation = prayer
		}
		triggered_animation = {
			trigger = {
				OR = {
					scope:injury_fate = 2
					scope:injury_fate = 3
				}
			}
			animation = severelywounded
		}
		triggered_animation = {
			trigger = {
				OR = {
					scope:vision_fate = 2
					scope:vision_fate = 1
				}
			}
			animation = stress
		}
		triggered_animation = {
			trigger = { always = yes }
			animation = chaplain
		}
	}

	immediate = {
		# Sacrifice
		sacrifice_progression_effect = yes

		# Random injury
		apply_random_injury_effect = yes

		## Vision consequences ##
		switch = {
			trigger = scope:vision_fate

			4 = { strong_vision_effect = yes } # Strong vision - Canonically what Kilrogg got
			3 = { normal_vision_effect = yes } # Normal vision
			2 = { weak_vision_effect = yes } # Weak vision
			1 = { no_vision_effect = yes } # No vision
		}

		# Tradition modifier
		add_character_modifier = performed_a_dark_ritual_modifier
	}

	option = { # Ok
		name = {
			text = {
				first_valid = {
					triggered_desc = { # Dying
						trigger = { scope:injury_fate = 2 }
						desc = wc_culture_tradition_events.0003.a.dying
					}
					triggered_desc = { # Dying - No vision
						trigger = {
							scope:injury_fate = 2
							scope:vision_fate = 1
						}
						desc = wc_culture_tradition_events.0003.a.dying_no_vision
					}
					triggered_desc = { # No vision
						trigger = { scope:vision_fate = 1 }
						desc = wc_culture_tradition_events.0003.a.no_vision
					}
					desc = wc_culture_tradition_events.0003.a # Fallback
				}
			}
		}

		if = {
			limit = { scope:injury_fate = 2 }
			override_death_effect = { death_reason = death_suicide }
		}
		else_if = {
			limit = { scope:vision_fate = 1 }

			random_list = {
				50 = { add_trait_force_tooltip = depressed_1 }
				50 = {} #Nothing hapenned
			}
		}
		else_if = {
			limit = { scope:vision_fate = 3 }

			random_list = { # Become Fearless - normal vision
				50 = {
					if = {
						limit = { has_trait = craven }

						remove_trait = craven
						add_trait_force_tooltip = brave
					}
					else_if = {
						limit = {
							number_of_personality_traits <= 3
							NOT = { has_trait = brave }
						}

						add_trait_force_tooltip = brave
					}
					else = { add_prestige = medium_prestige_gain }
				}
				50 = {} # Nothing happened
			}
		}
		else_if = {
			limit = { scope:vision_fate = 4 }

			if = { # Become fearless - strong vision
				limit = { has_trait = craven }

				remove_trait = craven
				add_trait_force_tooltip = brave
			}
			else_if = {
				limit = { 
					number_of_personality_traits <= 3
					NOT = { has_trait = brave }
				}

				add_trait_force_tooltip = brave
			}
			else = { add_prestige = major_prestige_gain }

		}

		remove_variable = is_performing_a_dark_ritual

		ai_chance = { base = 50 }
	}
}
