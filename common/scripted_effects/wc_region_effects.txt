get_lost_in_region_effect = {
	random_county_in_region = {
		region = $REGION$
		limit = { 
			NOT = { holder = root }
		}
		save_scope_as = adventuring_location
	}
	save_scope_value_as = {
		name = $REGION$_location
		value = yes
	}
}

update_locked_region_effect = {


	if = {
		limit = { has_variable = is_isolated }

		remove_variable = is_isolated

		set_variable = { 
			name = isolated_region
			value = 0
		}

		set_variable = { 
			name = in_suramar 
			value = no 
		}
		set_variable = { 
			name = in_uldum 
			value = no 
		}
		set_variable = { 
			name = in_pandaria 
			value = no 
		}
	}

	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = mists_of_pandaria_parted_variable }
				global_var:mists_of_pandaria_parted_variable = no
			}
			is_in_pandaria_trigger = yes
		}
		set_variable = is_isolated
		set_variable = { 
			name = in_pandaria 
			value = yes 
		}
		set_variable = { 
			name = isolated_region
			value = 1
		}
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_global_variable = uldum_has_opened_variable }
				global_var:uldum_has_opened_variable = no
			}
			is_in_uldum_trigger = yes
		}
		set_variable = is_isolated
		set_variable = { 
			name = in_uldum 
			value = yes 
		}
		set_variable = { 
			name = isolated_region
			value = 2
		}
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_global_variable = suramar_barrier_lowered_variable }
				global_var:suramar_barrier_lowered_variable = no
			}
			is_in_suramar_trigger = yes
		}
		set_variable = is_isolated
		set_variable = { 
			name = in_suramar 
			value = yes 
		}
		set_variable = { 
			name = isolated_region
			value = 3
		}
	}
}
