		}

		holder = {
			if = {
				limit = { NOT = { this = scope:liadrin } }
				add_opinion = {
					target = scope:liadrin
					modifier = revoked_title
					opinion = -30
				}
			}
		}
		change_title_holder = {
			holder = scope:liadrin
			change = scope:change
		}
	}
	title:c_dawnstar = {
		holder = {
			if = {
				limit = { NOT = { this = scope:liadrin } }
				add_opinion = {
					target = scope:liadrin
					modifier = revoked_title
					opinion = -30
				}
			}
		}
		change_title_holder = {
			holder = scope:liadrin
			change = scope:change
		}

		change_liege = {
			liege = title:e_quelthalas.holder

prepare_sunblade_event_attacker_effect = {
	# Independent landed candidates already have a legal realm context.
	# Everyone else receives the same dynamic landless duchy used by proven
	# GoA2 adventurer/event-invasion patterns.
	scope:sunblade_leader = {
		if = {
			limit = {
				OR = {
					is_landed = no
					NOT = { top_liege = this }
				}
			}
			if = {
				limit = { NOT = { has_trait = adventurer } }
				add_trait = adventurer
				add_character_flag = sunwell_phase3_temporary_adventurer_trait
			}
		}
	}

	if = {
		limit = {
			scope:sunblade_leader = {
				OR = {
					is_landed = no
					NOT = { top_liege = this }
				}
			}
		}
		create_dynamic_title = {
			tier = duchy
			name = d_sunblade_expedition
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = sunblade_title_change
			add_claim_on_loss = no
		}
		scope:new_title = {
			set_capital_county = title:c_sunwell_plateau
			set_landless_title = yes
			set_destroy_on_gain_same_tier = yes
			set_no_automatic_claims = yes
			set_can_be_named_after_dynasty = no
			change_title_holder = {
				holder = scope:sunblade_leader
				change = scope:sunblade_title_change
			}
		}
		resolve_title_and_vassal_change = scope:sunblade_title_change
		scope:new_title = {
			generate_coa = yes
			set_variable = {
				name = sunwell_phase3_temporary_expedition_title
				value = yes
			}
		}
	}
}

cleanup_sunblade_event_attacker_effect = {
	save_scope_as = sunblade_event_attacker
	every_held_title = {

		}

		remove_trait = adventurer
		remove_character_flag = sunwell_phase3_temporary_adventurer_trait
	}
}

	remove_global_variable = sunwell_phase3_anveena_coerced
	remove_global_variable = sunwell_phase3_muru_darkened

			exists = title:e_quelthalas.holder
			title:e_quelthalas.holder = { is_alive = yes }
		}
		set_global_variable = {
			name = sunwell_phase3_ruler
			value = title:e_quelthalas.holder

	}
	else_if = {

			exists = title:k_quelthalas.holder

			name = sunwell_phase3_ruler

			name = sunwell_phase3_ruler
			value = root
		}
	}

	get_anveena_character_effect = yes
	get_kalecgos_character_effect = yes
	get_rommath_character_effect = yes
	get_muru_character_effect = yes

	set_global_variable = { name = sunwell_phase3_anveena value = scope:anveena_character }
	set_global_variable = { name = sunwell_phase3_kalecgos value = scope:kalecgos_character }
	set_global_variable = { name = sunwell_phase3_rommath value = scope:rommath_character }
	set_global_variable = { name = sunwell_phase3_muru value = scope:muru_character }

		if = { limit = { is_landed = no } set_employer = global_var:sunwell_phase3_ruler }
		apply_sunwell_phase3_protection_effect = yes
	}
	scope:muru_character = {
		if = { limit = { is_imprisoned = yes } release_from_prison = yes }
		if = { limit = { is_landed = no } set_employer = global_var:sunwell_phase3_ruler }
		apply_sunwell_phase3_protection_effect = yes
	}

	global_var:sunwell_phase3_ruler = { apply_sunwell_phase3_protection_effect = yes }

	if = {
		limit = {
			exists = title:h_burning_legion.holder
			title:h_burning_legion.holder = { is_alive = yes }
		}

	}
}

		value = current_military_strength
	}
	scope:sunblade_target_ruler = {
		save_scope_value_as = {
			name = sunblade_defender_strength_at_declaration
			value = current_military_strength
		}

			scope:sunblade_attacker_strength_before_reinforcement < {

		spawn_army = {
			name = sunblade_expedition_army
			uses_supply = no
			inheritable = no
			levies = {
				value = scope:sunblade_defender_strength_at_declaration
				multiply = 0.95
				subtract = scope:sunblade_attacker_strength_before_reinforcement
				min = 500
				max = 10000
			}

		}
	}

		limit = { exists = global_var:sunwell_phase3_legion_present }
		spawn_army = {
			name = sunblade_legion_reinforcements
			uses_supply = no
			inheritable = no

				multiply = 0.1
				min = 250
				max = 1500
			}
			location = title:c_sunwell_plateau.title_province
		}
	}
}

		limit = {
			exists = title:h_burning_legion.holder

		}
		title:h_burning_legion.holder = {
			save_scope_as = legion_foothold_holder
		}
		set_global_variable = sunwell_phase3_legion_present
	}
	else_if = {
		limit = {
			exists = global_var:sunwell_phase3_sunblade_leader
			global_var:sunwell_phase3_sunblade_leader = { is_alive = yes }
		}
		global_var:sunwell_phase3_sunblade_leader = {
			save_scope_as = legion_foothold_holder
		}
	}

	if = {
		limit = { exists = scope:legion_foothold_holder }

		title:d_queldanas_isle = {
			every_de_jure_county = {
				limit = {
					OR = {
						this = title:c_sunwell_plateau
						holder.top_liege = global_var:sunwell_phase3_ruler
					}
				}
				save_temporary_scope_as = sunblade_queldanas_county
				give_land_to_settler = {
					TO_WHO = scope:legion_foothold_holder
					COUNTY_NAME = scope:sunblade_queldanas_county
					GIVE_INDEPENDENCY = yes
					ADD_CLAIM_ON_LOSS = no
				}
			}
		}

		title:c_sunwell_plateau = {
			change_county_control = 100
		}

		scope:legion_foothold_holder = {
			spawn_army = {
				name = sunblade_legion_vanguard
				uses_supply = no
				inheritable = no
				men_at_arms = { type = fel_guard stacks = 30 }
				location = title:c_sunwell_plateau.title_province
			}
		}

		if = {
			limit = {
				exists = global_var:sunwell_phase3_legion_present
				NOT = { exists = global_var:sunwell_phase3_legion_reinforced }
			}
			scope:legion_foothold_holder = { spawn_burning_legion_troops_effect = yes }
			set_global_variable = sunwell_phase3_legion_reinforced
		}
	}
}

transform_quelthalas_to_void_effect = {
	root = { save_scope_as = void_quelthalas_ruler }

	root = {
		if = {
			limit = { NOT = { has_trait = being_void } }
			become_being_void_effect = yes
		}
		set_culture = culture:void_elf
		if = {
			limit = {
				NOT = { any_held_title = { is_head_of_faith = yes } }
			}
			set_character_faith = faith:voidfarer
		}
	}

	every_living_character = {
		limit = {
			this != scope:void_quelthalas_ruler
			top_liege = scope:void_quelthalas_ruler
			location = { geographical_region = world_eastern_kingdoms_quelthalas }
			can_become_voidfarer_trigger = yes
			OR = {
				has_culture = culture:blood_elf
				has_culture = culture:high_elf
			}
		}
		set_character_faith = faith:voidfarer
		set_culture = culture:void_elf
	}

	every_county_in_region = {
		region = world_eastern_kingdoms_quelthalas
		limit = {
			holder.top_liege = scope:void_quelthalas_ruler
			OR = {
				culture = culture:blood_elf
				culture = culture:high_elf
			}
		}
		set_county_faith = faith:voidfarer
		set_county_culture = culture:void_elf
	}

	set_global_variable = sunwell_void_quelthalas_established
}
