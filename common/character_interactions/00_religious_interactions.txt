# Interaction used to convert people when you reform or create a faith
ask_for_conversion_interaction = {
	category = interaction_category_religion
	special_interaction = request_conversion_interaction
	popup_on_receive = yes

	desc = compel_conversion_interaction_desc

	hidden = yes

	is_shown = {
		scope:actor.faith != scope:recipient.faith
		scope:recipient.liege ?= scope:actor
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }	# Warcraft
		valid_ask_for_conversion_conditions_trigger = yes
	}

	ai_accept = {
		base = 0

		# Warcraft
		master_thrall_modifier = yes
		# End Warcraft

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:actor
			RECIPIENT = scope:recipient
		}

		religion_demand_conversion_default_modifier = yes
		religion_demand_conversion_christian_situation_modifier = yes

		# We add a few extra modifiers if you have a state faith, to make the initial conversion rate more immersive and plausible
		modifier = {
			scope:actor = {
				government_allows = state_faith

				# If you fulfill a minimum amount of the below triggers, we consider that you hold enough sway within the realm to make it easier for others to abandon the state faith
				calc_true_if = {
					amount >= 4

					diplomacy >= excellent_skill_level
					learning >= excellent_skill_level
					has_trait = august # You finished the August lifestyle tree
					has_trait = theologian # You finished the Theologian lifestyle tree
					AND = {
						top_liege = this
						legitimacy_level >= 4
					}
					influence_level >= 3
					house = {
						OR = {
							is_dominant_family = yes
							is_powerful_family = yes
						}
					}
				}
			}
			add = {
				value = 50
				if = {
					limit = {
						scope:actor.influence_level = 5
					}
					add = 50
				}
				else_if = {
					limit = {
						scope:actor.influence_level >= 4
					}
					add = 25
				}
				if = {
					limit = {
						scope:actor = {
							top_liege = this
							legitimacy_level >= 5
						}
					}
					add = 50
				}
			}
			desc = STATE_FAITH_CREATE_FAITH_REASON
		}
		modifier = { # If you have the prophet perk, let's increase acceptance equal to half your diplomacy and learning
			scope:actor = {
				government_allows = state_faith
				has_perk = prophet_perk
			}
			add = {
				value = scope:actor.learning
				add = scope:actor.diplomacy
				multiply = 0.5
			}
			desc = HAS_PERK_PROPHET_REASON
		}

	}

	ai_min_reply_days = 1
	ai_max_reply_days = 5

	on_accept = {
		new_faith_created_conversion_effect = yes

		# If we're a clan this interaction affects unity
		accept_faith_conversion_add_clan_unity_effect = yes

		# If character is converting to the state faith, we add influence
		state_faith_conversion_add_piety_effect = yes
		# Warcraft - remove struggle reward
	}

	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = ask_for_conversion_decline_notification
				right_icon = scope:recipient
			}
		}

		# If we're a clan this interaction affects unity
		refuse_faith_conversion_add_clan_unity_effect = yes

		state_faith_refuse_conversion_add_piety_effect = yes
	}

	#Use hook
	send_option = {
		is_valid = {
			always = no # Only here to ensure that religion_demand_conversion_default_modifier doesn't error
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	send_options_exclusive = no
}

ask_for_conversion_courtier_interaction = {
	category = interaction_category_religion
	icon = religious

	special_interaction = request_conversion_interaction
	ai_maybe = yes
	can_send_despite_rejection = yes

	desc = {
		desc = ask_for_conversion_interaction_desc
		triggered_desc = {
			trigger = {
				scope:actor = {
					refusing_conversion_is_crime_trigger = {
						CHARACTER = scope:recipient
					}
				}
				scope:recipient = {
					target_is_liege_or_above = scope:actor
				}
			}
			desc = demand_conversion_vassal_ruler_is_crime
		}
	}

	ai_targets = {
		ai_recipients = courtiers
		chance = 0.25
	}
	ai_target_quick_trigger = {
		adult = yes
	}
	ai_frequency_by_tier = {
		barony = 0
		county = 36
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			is_adult = yes
		}
	}

	is_shown = {
		scope:actor.faith != scope:recipient.faith
		scope:recipient = { is_courtier_of = scope:actor }
		scope:recipient = { is_ruler = no }
		scope:recipient = { is_imprisoned = no }
	}

	cooldown_against_recipient = { years = 15 }

	is_valid_showing_failures_only = {
		valid_demand_conversion_conditions_trigger = yes
	}

	ai_accept = {
		base = 0

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:actor
			RECIPIENT = scope:recipient
		}

		# Increase base chance
		modifier = {
			add = 50
			desc = EDUCATE_CHILD_ACTOR_IS_MY_LIEGE
		}
		religion_demand_conversion_default_modifier = yes
		
		# Warcraft
		scourge_interaction_modifiers = yes
		master_thrall_modifier = yes
		# End Warcraft
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
			scope:actor = { has_strong_hook = scope:recipient }
			NOT = { scope:recipient = { has_character_flag = ai_will_not_convert } }
		}
	}

	ai_min_reply_days = 1
	ai_max_reply_days = 5

	on_accept = {
		demand_conversion_interaction_effect = yes
		hidden_effect = {
			scope:actor = {
				send_interface_message = {
					type = event_religious_good
					title = ask_for_conversion_courtier_interaction.tt
					left_icon = scope:recipient
					scope:recipient = {
						show_as_tooltip = { demand_conversion_interaction_effect = yes }
					}
				}
			}
		}
		if = {
			limit = { scope:hook = yes }
			scope:actor = {
				if = {
					limit = {
						has_usable_hook = scope:recipient
					}
					use_hook = scope:recipient
				}
			}
			scope:recipient = {
				save_scope_value_as = {
					name = forcibly_converted
					value = yes
				}
			}
		}

		# If we're a clan this interaction affects unity
		accept_faith_conversion_add_clan_unity_effect = yes

		# If character is converting to the state faith, we add influence
		state_faith_conversion_add_piety_effect = yes
		# Warcraft - remove struggle reward
		# If you have the Mendicant Preaching personal tenet we give you SP
		conversion_tenet_mendicant_preachers_effect = yes
	}

	on_decline = {
		#Negative opinions
		scope:recipient = {
			if = {
				limit = {
					scope:actor = {
						refusing_conversion_is_crime_trigger = {
							CHARACTER = scope:recipient
						}
					}
					scope:recipient = {
						target_is_liege_or_above = scope:actor
					}
				}
				reverse_add_opinion = {
					target = scope:actor
					modifier = illegal_resisted_conversion_opinion
					years = 10
				}
			}
			else = {
				add_opinion = {
					modifier = demanded_my_conversion_opinion
					target = scope:actor
					opinion = -10
				}
			}
		}
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = ask_for_conversion_decline_notification
				left_icon = scope:recipient
			}
		}

		# If we're a clan this interaction affects unity
		refuse_faith_conversion_add_clan_unity_effect = yes

		state_faith_refuse_conversion_add_piety_effect = yes
	}

	cost = {
		influence = {
			value = 0
			if = {
				limit = { scope:influence_send_option = yes }
				add = scope:actor.medium_influence_value
				desc = INFLUENCE_INTERACTION_ACCEPTANCE_SEND_OPTION
			}
		}
	}

	# Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
			NOT = {
				scope:recipient = { is_imprisoned_by = scope:actor }
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_option = { # EP3 Influence
		is_shown = { # Actor must have a government that uses influence
			scope:actor = {
				government_has_flag = government_has_influence
			}
		}
		is_valid = {
			# Actor has enough influence
			scope:actor = { influence >= medium_influence_value }
			# Both characters are within the same top realm
			custom_tooltip = {
				text = not_same_realm_tt
				scope:recipient.top_liege = scope:actor.top_liege
			}
		}
		flag = influence_send_option
		localization = TRADE_INFLUENCE_FOR_BETTER_AI_ACCEPTANCE
	}
	send_options_exclusive = no


	ai_will_do = {
		base = 100

		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}

		modifier = { # The AI doesn't really want to convert Diasporic culture courtiers
			# Warcraft
			scope:actor = {
				faith = {
					faith_hostility_level = {
						target = scope:recipient.faith
						value <= faith_astray_level
					}
				}
			}
			scope:recipient = {
				# like_judaism_religion_trigger = yes
				culture = {
					has_cultural_tradition = tradition_diasporic
				}
			}
			factor = 0
		}

		modifier = { # Do not convert Righteous faith subjects, unless disturbingly zealous
			factor = 0
			scope:actor = {
				ai_zeal < 100
				rite = {
					rite_hostility_level = {
						target = scope:recipient.rite
						value <= faith_fully_accepted_level
					}
				}
			}
		}

		modifier = { # Astray faiths require specific circumstances to demand conversion of
			factor = 0
			scope:actor = {
				has_tolerant_faith_or_culture_trigger = yes
				ai_zeal < 100
				rite = {
					rite_hostility_level = {
						target = scope:recipient.rite
						value <= faith_astray_level
					}
				}
			}
			NOT = {
				scope:recipient = {
					is_close_family_of = scope:actor
				}
			}
		}

		modifier = { # Astray faiths require specific circumstances to demand conversion of
			factor = 0
			scope:actor = {
				rite = { 
					OR = {
						rite_has_doctrine = doctrine_pluralism_fundamentalist
						rite_has_doctrine = doctrine_pluralism_dark # Warcraft
					}
				}
				ai_zeal < -10
				rite = {
					rite_hostility_level = {
						target = scope:recipient.rite
						value <= faith_astray_level
					}
				}
			}
			NOT = {
				scope:recipient = {
					is_close_family_of = scope:actor
				}
			}
		}

		modifier = { # Astray faiths require specific circumstances to demand conversion of
			factor = 0
			scope:actor = {
				NOR = {
					rite = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
					has_tolerant_faith_or_culture_trigger = yes
				}
				ai_zeal < 25
				rite = {
					rite_hostility_level = {
						target = scope:recipient.rite
						value <= faith_astray_level
					}
				}
			}
			NOT = {
				scope:recipient = {
					is_close_family_of = scope:actor
				}
			}
		}

		modifier = { # Rulers with State Faith won't try to convert people with rites that are astray or better
			factor = 0
			scope:actor = {
				government_allows = state_faith
				primary_title.state_rite ?= {
					rite_hostility_level = {
						target = scope:recipient.rite
						value <= faith_astray_level
					}
				}
			}
		}
	}
}

# House head demand conversion
demand_conversion_interaction = {
	icon = religious
	category = interaction_category_religion
	popup_on_receive = yes
	pause_on_receive = yes
	can_send_despite_rejection = yes
	ai_maybe = yes

	notification_text = HOUSE_HEAD_DEMAND_CONVERSION

	is_shown = {
		scope:puppet_or_actor ?= {
			this != scope:recipient
			is_house_head = yes
			faith != scope:recipient.faith
			OR = {
				house ?= scope:recipient.house
				AND = { # House Bloc
					house.house_confederation ?= {
						scope:recipient.house.house_confederation ?= this
						has_cohesion_level_parameter = bloc_members_join_leaders_faith
					}
				}
			}
		}
		NOT = {
			scope:recipient = { # Use the vassal-liege interaction instead.
				is_ruler = yes
				target_is_liege_or_above = scope:puppet_or_actor
			}
		}
		scope:recipient = { # To avoid you converting someone's entire family, and then asking them. Restricting it to landed rulers at least gives them a chance to create their own cadet house.
			is_ruler = yes
		}
	}

	cooldown_against_recipient = { years = 15 }

	desc = {
		desc = demand_conversion_interaction_desc
		triggered_desc = {
			trigger = {
				scope:puppet_or_actor = {
					refusing_conversion_is_crime_trigger = {
						CHARACTER = scope:recipient
					}
				}
				scope:recipient = {
					target_is_liege_or_above = scope:puppet_or_actor
				}
			}
			desc = demand_conversion_vassal_ruler_is_crime
		}
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }	# Warcraft
		valid_demand_conversion_conditions_trigger = yes
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				influence >= demand_conversion_influence_cost_value
			}
		}
	}

	cost = {
		influence = {
			value = demand_conversion_influence_cost_value
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				if = {
					limit = { demand_conversion_influence_cost_value > 0 }
					change_influence = {
						value = {
							value = demand_conversion_influence_cost_value
							max = influence
							multiply = -1
						}
					}
				}
			}
		}
		demand_conversion_interaction_effect = yes
		if = {
			limit = { scope:hook = yes }
			scope:puppet_or_actor = {
				use_hook = scope:recipient
			}
			scope:recipient = {
				save_scope_value_as = {
					name = forcibly_converted
					value = yes
				}
			}
		}
		scope:puppet_or_actor = {
			trigger_event = char_interaction.0181
		}

		# Warcraft - remove struggle reward

		# If we're a clan this interaction affects unity
		accept_faith_conversion_add_clan_unity_effect = yes

		# If character is converting to the state faith, we add influence
		state_faith_conversion_add_piety_effect = yes

		conversion_tenet_acts_of_the_apostles_effect = yes

		# If you have the Mendicant Preaching personal tenet we give you SP
		conversion_tenet_mendicant_preachers_effect = yes
	}

	on_decline = {
		scope:recipient = {
			execute_decision = create_cadet_branch_decision
			show_as_tooltip = { create_cadet_branch = {} }
		}
		#Negative opinions
		scope:recipient = {
			if = {
				limit = {
					scope:puppet_or_actor = {
						refusing_conversion_is_crime_trigger = {
							CHARACTER = scope:recipient
						}
					}
					scope:recipient = {
						target_is_liege_or_above = scope:puppet_or_actor
					}
				}
				reverse_add_opinion = {
					target = scope:puppet_or_actor
					modifier = illegal_resisted_conversion_opinion
					years = 10
				}
			}
			else = {
				add_opinion = {
					modifier = demanded_my_conversion_opinion
					target = scope:puppet_or_actor
					opinion = -10
				}
			}
		}
		scope:puppet_or_actor = {
			trigger_event = char_interaction.0182
		}

		# If we're a clan this interaction affects unity
		refuse_faith_conversion_add_clan_unity_effect = yes

		state_faith_refuse_conversion_add_piety_effect = yes
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:puppet_or_actor
			object = scope:recipient
			scope:hook = yes
			scope:puppet_or_actor = { has_strong_hook = scope:recipient }
		}
	}

	#For loc
	on_auto_accept = {
		scope:recipient = {
			save_scope_value_as = {
				name = forcibly_converted
				value = yes
			}
		}
	}

	ai_accept = {
		base = 0

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:puppet_or_actor
			RECIPIENT = scope:recipient
		}

		# Increase base chance
		modifier = {
			add = 50
			desc = EDUCATE_CHILD_ACTOR_IS_MY_LIEGE
		}
		modifier = {
			add = 100
			NOT = {
				scope:recipient = {
					can_execute_decision = create_cadet_branch_decision
				}
			}
			desc = ASK_FOR_CONVERSION_CAN_NOT_CREATE_CADET_BRANCH
		}
		modifier = {
			add = -10
			scope:recipient = {
				can_execute_decision = create_cadet_branch_decision
			}
			desc = ASK_FOR_CONVERSION_CAN_CREATE_CADET_BRANCH
		}
		religion_demand_conversion_default_modifier = yes
		religion_demand_conversion_christian_situation_modifier = yes
		
		# Warcraft
		scourge_interaction_modifiers = yes
		master_thrall_modifier = yes
		# End Warcraft
	}

	can_send_despite_rejection = yes
	force_notification = yes

	#Use hook
	send_option = {
		is_valid = {
			scope:puppet_or_actor = {
				has_usable_hook = scope:recipient
			}
			NOT = {
				scope:recipient = { is_imprisoned_by = scope:puppet_or_actor }
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:puppet_or_actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_option = { # EP3 Influence
		is_shown = { # Actor must have a government that uses influence
			scope:puppet_or_actor ?= {
				government_has_flag = government_has_influence
			}
		}
		is_valid = {
			# Actor has enough influence
			scope:puppet_or_actor = { influence >= medium_influence_value }
			# Both characters are within the same top realm
			custom_tooltip = {
				text = not_same_realm_tt
				scope:recipient.top_liege = scope:puppet_or_actor.top_liege
			}
		}
		flag = influence_send_option
		localization = TRADE_INFLUENCE_FOR_BETTER_AI_ACCEPTANCE
	}
	send_options_exclusive = no
}

				target_is_liege_or_above = scope:recipient
				is_tributary_of = scope:recipient

			NAND = {

				is_target_in_variable_list = {
					name = humsacd_hofs

			limit = {
				scope:puppet_or_actor = {
					government_allows = state_faith
				}

				open_view_data = {
					view = decision_detail
					player = scope:puppet_or_actor
				}
			}
		}

		else = {
			scope:recipient.rite = {
				open_view_data = {
					view = faith_conversion
					player = scope:puppet_or_actor # interaction effects are executed for all clients, make sure to open the window only for the actor

			}
		}

		base = 0
	}

temporal_condemnation_interaction = {
	category = interaction_category_religion
	desc = {
		first_valid = {
			#triggered_desc = {
			#	trigger = { scope:puppet_or_actor.faith = faith:quranist }
			#	desc = temporal_condemnation_interaction_desc_fasiq
			#}
			desc = temporal_condemnation_interaction_desc
		}
	}
	icon = religious

	is_shown = {
		scope:puppet_or_actor != scope:recipient
		scope:puppet_or_actor.rite ?= {
			rite_has_parameter = temporal_condemnation_enabled
		}
		scope:recipient = {
			liege ?= scope:puppet_or_actor
			is_playable_character = yes
			is_ruler = yes
		}
	}

	cooldown = { years = 5 }

	is_valid = {
		scope:puppet_or_actor = {
			piety >= medium_piety_value
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient.religion = scope:puppet_or_actor.religion
		custom_description = {
			text = temporal_condemnation_has_been_condemned
			object = scope:recipient
			scope:recipient = {
				NOT = {
					has_character_modifier = temporal_condemnation_modifier
				}
			}
		}
		NAND = {
			exists = scope:puppet_or_actor.faith.great_holy_war
			scope:puppet_or_actor.faith.great_holy_war = {
				has_pledged_attacker = scope:recipient
			}
		}
		scope:recipient = {
			is_adult = yes
		}
		scope:recipient = { NOT = { has_strong_hook = scope:puppet_or_actor } }
	}

	auto_accept = yes
	force_notification = yes
	notification_text = temporal_condemnation_interaction.notification

	on_accept = {
		scope:puppet_or_actor = {
			add_piety = medium_piety_loss
		}

		scope:puppet_or_actor = {
			duel = {
				skill = learning
				value = religious_legal_pronouncement_learning_requirement
				50 = {
					desc = temporal_condemnation_interaction_success
					compare_modifier = {
						value = scope:duel_value
						multiplier = 3
						min = -45
					}
					modifier = {
						add = 25
						scope:puppet_or_actor = {
							has_personal_tenet_flag = tenet_religious_law_condemnation_bonus
						}
					}
					scope:puppet_or_actor = {
						add_opinion = {
							target = scope:recipient
							modifier = temporal_condemnation_opinion
							years = 10
						}
						custom_tooltip = temporal_condemnation_revoke_reason
						hidden_effect = {
							send_interface_toast = {
								type = event_toast_effect_good
								title = temporal_condemnation_interaction.tt
								left_icon = scope:recipient
								show_as_tooltip = { custom_tooltip = temporal_condemnation_got_revoke_reason }
								if = {
									limit = {
										has_personal_tenet_flag = tenet_religious_law_scholar_trait
									}
									add_trait_xp = {
										trait = lifestyle_scholar
										value = 20
									}
								}
							}
						}
						stress_and_fulfillment_impact = {
							cynical = minor_stress_impact_gain
						}
					}
					if = {
						limit = {
							scope:is_puppet_action ?= yes
							scope:actor != scope:recipient
						}
						scope:actor = {
							send_interface_toast = {
								type = event_toast_effect_good
								title = temporal_condemnation_interaction.tt_puppeteer
								left_icon = scope:puppet_or_actor
								right_icon = scope:recipient
							}
						}
					}
					scope:recipient = {
						add_character_modifier = {
							modifier = temporal_condemnation_modifier
							years = 15
						}
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = temporal_condemnation_condemned_me_opinion
						}
						send_interface_toast = {
							type = event_toast_effect_bad
							title = temporal_condemnation_interaction.recipient_success
							left_icon = scope:puppet_or_actor
							custom_tooltip = temporal_condemnation_interaction.recipient_success_desc
						}
					}

					# If we're a clan this interaction affects unity
					add_clan_unity_interaction_effect = {
						CHARACTER = scope:puppet_or_actor
						TARGET = scope:recipient
						VALUE = medium_unity_gain
						DESC = clan_unity_condemn_sins.desc
						REVERSE_NON_HOUSE_TARGET = no
					}
				}
				50 = {
					desc = temporal_condemnation_interaction_failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = -3
						min = -45
					}
					scope:puppet_or_actor = {
						send_interface_toast = {
							type = event_toast_effect_bad
							title = temporal_condemnation_interaction.failure
							left_icon = scope:recipient
							custom_tooltip = temporal_condemnation_interaction.failure_desc
						}
					}
					if = {
						limit = {
							scope:is_puppet_action ?= yes
							scope:actor != scope:recipient
						}
						scope:actor = {
							send_interface_toast = {
								type = event_toast_effect_bad
								title = temporal_condemnation_interaction.failure
								left_icon = scope:recipient
								custom_tooltip = temporal_condemnation_interaction.failure_desc
							}
						}
					}
				}
			}
		}

host_honored_guest_interaction = {
	icon = religious
	category = interaction_category_friendly

	desc = host_honored_guest_interaction_desc

	ai_targets = {
		ai_recipients = vassals
	}
	ai_target_quick_trigger = {
		adult = yes
	}
	ai_frequency_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	cooldown = { years = 1 }

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			piety >= major_piety_value
		}
	}

	is_shown = {
		scope:recipient != scope:puppet_or_actor
		scope:puppet_or_actor ?= {
			rite ?= {
				rite_has_parameter = host_honored_guests_active
			}
			is_playable_character = yes
			is_adult = yes
		}
	}

	is_valid_showing_failures_only = {
		# Target faith is Hostile.
		custom_description = {
			text = host_honored_guest_is_hostile
			object = scope:recipient
			NOT = {
				scope:recipient = {
					rite = {
						rite_hostility_level = {
							target = scope:puppet_or_actor.rite
							value = faith_hostile_level
						}
					}
				}
			}
		}
		# Target faith is Evil.
		custom_description = {
			text = host_honored_guest_is_evil
			object = scope:recipient
			NOT = {
				scope:recipient = {
					rite = {
						rite_hostility_level = {
							target = scope:puppet_or_actor.rite
							value = faith_evil_level
						}
					}
				}
			}
		}
		# Target is sick
		custom_description = {
			text = host_honored_guest_is_unwell
			object = scope:recipient
			scope:recipient = {
				is_healthy = yes
			}
		}
		# Target is commanding an army
		custom_description = {
			text = host_honored_guest_is_commanding
			object = scope:recipient
			scope:recipient = {
				is_commanding_army = no
			}
		}
		# Target is imprisoned
		custom_description = {
			text = host_honored_guest_is_imprisoned
			object = scope:recipient
			scope:recipient = {
				is_imprisoned = no
			}
		}
		# At war with target
		custom_description = {
			text = host_honored_guest_is_at_war
			object = scope:recipient
			scope:recipient = {
				NOT = { is_at_war_with = scope:puppet_or_actor }
			}
		}
		# Is my courtier
		custom_description = {
			text = host_honored_guest_is_my_courtier
			object = scope:recipient
			scope:recipient = {
				NOT = { is_courtier_of = scope:puppet_or_actor }
			}
		}
	}

	auto_accept = yes
	force_notification = yes
	notification_text = host_honored_guest_interaction.notification
	#answer_accept_key = host_honored_guest_interaction.response

	on_accept = {
		scope:puppet_or_actor = {
			stress_and_fulfillment_impact = {
				cynical = minor_stress_impact_gain
				shy = minor_stress_impact_gain
			}
		}
		scope:recipient = {
			hidden_effect = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = host_honored_guest_interaction_recipient.notification
					left_icon = scope:puppet_or_actor
				}
			}
			# Spend piety based on recipient's tier.
			scope:puppet_or_actor = {
				add_piety = {
					value = 0
					subtract = scope:recipient.tier_scaled_piety_value
				}
			}

			# Increase opinion of recipient.
			add_opinion = {
				modifier = pleased_opinion
				target = scope:puppet_or_actor
				opinion = {
					value = 20

					# Zealous characters more likely to appreciate the religious ceremony (if of your faith)
					if = {
						limit = {
							has_trait = zealous
							faith = scope:puppet_or_actor.faith
						}
						add = 10
					}
					# Cynical characters, or characters who do not believe in the hospitality ritual, are less likely to be appreciative.
					else_if = {
						limit = {
							OR = {
								has_trait = cynical
								NOT = {
									rite = {
										rite_has_parameter = host_honored_guests_active
									}
								}
							}
						}
						subtract = 10
					}

					# But arrogant characters like it more regardless.
					if = {
						limit = {
							has_trait = arrogant
						}
						add = 10
					}
				}
			}
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = miniscule_unity_gain
			DESC = clan_unity_honored_guest.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					is_powerful_vassal_of = scope:puppet_or_actor
					liege = root
				}
				opinion = {
					target = scope:puppet_or_actor
					value = {-15 0 }
				}
			}
		}

		modifier = {
			add = 75
			scope:recipient = {
				OR = {
					is_powerful_vassal_of = scope:puppet_or_actor
					liege = root
				}
				opinion = {
					target = scope:puppet_or_actor
					value = { 0 20 }
				}
			}
		}

		modifier = {
			add = 40
			scope:recipient = {
				OR = {
					is_powerful_vassal_of = scope:puppet_or_actor
					liege = root
				}
				opinion = {
					target = scope:puppet_or_actor
					value = { 20 60 }
				}
			}
		}

		modifier = {
			add = 10
			scope:recipient = {
				OR = {
					is_powerful_vassal_of = scope:puppet_or_actor
					liege = root
				}
				opinion = {
					target = scope:puppet_or_actor
					value = { 60 100 }
				}
			}
		}
	}
}

# Spiritual Head of Faith Actions

hof_ask_for_gold_interaction = {
	icon = icon_clergy_ask_money
	category = interaction_category_religion
	common_interaction = yes

	desc = hof_ask_for_gold_interaction_desc

	notification_text = RECIPIENT_HOF_ASK_FOR_GOLD

	ai_targets = {
		ai_recipients = head_of_faith
		ai_recipients = religious_head_challenger
		ai_recipients = realm_priest_superior
		ai_recipients = superior
		ai_recipients = realm_priest_top_superior
		ai_recipients = councillors
	}
	ai_target_quick_trigger = {
		adult = yes
	}
	ai_frequency_by_tier = {
		barony = 0
		county = 36
		duchy = 24
		kingdom = 6
		empire = 6
		hegemony = 6
	}

	cooldown_against_recipient = { years = 3 }

	redirect = {
		scope:actor = {
			spiritual_hof_or_realm_priest_superior_recipient_redirect_effect = yes
		}
	}

	is_available = {
		trigger_if = {
			limit = {
				is_ai = yes
			}
			piety >= major_piety_value
			NOR = {
				has_trait = excommunicated
				has_character_flag = hof_request_cooldown
			}
			OR = {
				is_landed = yes
				government_has_mechanic = theocracy
			}
		}
	}

	is_shown = {
		scope:recipient != scope:puppet_or_actor
		scope:puppet_or_actor ?= {
			faith ?= {
				OR = {
					has_doctrine = doctrine_spiritual_head
					has_doctrine_parameter = has_clerical_regions
				}
			}
			is_playable_character = yes
			NOT = {
				# Ecclesiasticals have the 'request_ecclesiastical_treasury_interaction' instead
				government_has_flag = government_is_ecclesiastical
			}
			broader_clergy_show_hof_or_superior_interactions_trigger = yes
		}
		scope:recipient = {
			faith = scope:puppet_or_actor.faith
			# Antipope uses petition_antipope_gold_interaction instead
			trigger_if = {
				limit = { has_pam_dlc_trigger = yes }
				pam_is_antipope_trigger = no
			}
		}
		# Those under an antipope ask them for gold instead of the real Head of Faith
		trigger_if = {
			limit = {
				scope:recipient ?= scope:puppet_or_actor.faith.religious_head
			}
			scope:puppet_or_actor ?= { is_following_religious_head_challenger = no }
		}
		scope:actor = {
			trigger_if = {
				limit = {
					cp:councillor_court_chaplain ?= scope:recipient
					pam_is_archbishop_or_higher_trigger = no
					actor_has_capital_archbishop_trigger = yes
				}
				capital_county.clerical_region_title.holder ?= scope:recipient
			}
		}
		# Warcraft
		trigger_if = {
			limit = { scope:puppet_or_actor = { is_ai = yes } }
			scope:recipient = {
				NOT = { has_variable = 1_year_cd_claim_kal }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }	# Warcraft
		scope:recipient = {
			trigger_if = {
				limit = {
					has_treasury = yes
				}
				treasury >= hof_ask_for_gold_request_value
			}
			trigger_else = {
				gold >= hof_ask_for_gold_request_value
			}
		}

		scope:actor = {
			spiritual_hof_or_realm_priest_superior_recipient_trigger = yes
			piety >= major_piety_value
			trigger_if = {
				limit = {
					has_pam_dlc_trigger = yes
				}
				clergy_interaction_generic_piety_level_or_puppet_req_trigger = yes
			}
			NOR = {
				has_trait = excommunicated
				is_at_war_with = scope:recipient
				has_character_modifier = promised_pilgrimage_to_clergy_modifier
			}
			custom_tooltip = {
				text = HOF_INTERACTION_UNREFORMED_FAITH
				NOT = {
					faith = { has_doctrine_parameter = unreformed }
				}
			}
			pam_antipope_clerical_request_trigger = { RECIPIENT = scope:recipient }
		}
	}

	cost = {
		influence = {
			value = 0
			if = {
				limit = { scope:influence_send_option = yes }
				add = scope:actor.medium_influence_value
				desc = INFLUENCE_INTERACTION_ACCEPTANCE_SEND_OPTION
			}
		}
	}

	ai_accept = {
		base = -40

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:actor
			RECIPIENT = scope:recipient
		}

		modifier = {
			add = 25
			scope:influence_send_option = yes
			desc = INFLUENCE_INTERACTION_ACCEPTANCE_SEND_OPTION
		}

		modifier = {
			add = pam_offer_pilgrimage_acceptance_value
			scope:offer_pilgrimage = yes
			desc = OFFER_PILGRIMAGE_INTERACTION_OPTION_REASON
		}

		ask_for_gold_acceptance_modifier = yes

		# Warcraft
		scourge_interaction_modifiers = yes
		master_thrall_modifier = yes
		# End Warcraft
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
		}
	}

	send_option = {
		is_shown = {
			scope:actor != scope:recipient
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_option = { # EP3 Influence
		is_shown = { # Actor must have a government that uses influence
			scope:actor = {
				government_has_flag = government_has_influence
			}
		}
		is_valid = {
			# Actor has enough influence
			scope:actor = { influence >= medium_influence_value }
			# Both characters are within the same top realm
			custom_tooltip = {
				text = not_same_realm_tt
				scope:recipient.top_liege = scope:actor.top_liege
			}
		}
		flag = influence_send_option
		localization = TRADE_INFLUENCE_FOR_BETTER_AI_ACCEPTANCE
	}

	send_option = {
		is_shown = {
			has_pam_dlc_trigger = yes
			pam_offer_pilgrimage_option_shown_trigger = { ACTOR = scope:actor }
		}
		is_valid = {
			pam_offer_pilgrimage_option_valid_trigger = { ACTOR = scope:actor }
		}
		flag = offer_pilgrimage
		localization = OFFER_PILGRIMAGE_INTERACTION_OPTION
	}

	send_options_exclusive = no

	ai_min_reply_days = 1
	ai_max_reply_days = 5

	can_send = {
		# Anti-spam AI blockers
		ai_religious_request_blocker_trigger = { INTERACTION = hof_ask_for_gold_interaction }
	}

	on_send = {
		# Anti-spam AI blockers
		ai_religious_request_blocker_effect = { INTERACTION = hof_ask_for_gold_interaction }
	}

	on_accept = {
		hof_ask_for_gold_interaction_effect = yes

		pam_offer_pilgrimage_effect = {
			ACTOR = scope:actor
			CLERIC = scope:recipient
		}

		scope:actor = {
			trigger_event = religious_interaction.3
		}
		# Warcraft
		scope:recipient = {
			if = {
				limit = {
					OR = {
						is_ai = no
						has_title = title:e_kaldorei
					}
				}
				set_variable = {
					name = 1_year_cd_claim_kal
					value = yes
					years = 3
				}
			}
		}
	}

	on_decline = {
		scope:recipient = {
			add_piety = miniscule_piety_loss
		}
		scope:actor = {
			show_as_tooltip = {
				hof_ask_for_gold_refusal_effect = yes
			}
			trigger_event = religious_interaction.32
		}
	}

	ai_will_do = {
		base = 0

		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
		#Make AI clerics do this less - they should use Request Treasury Funds instead
		modifier = {
			is_clergy = yes
			add = -25
		}
		modifier = { # Counts get a bad 'exchange rate' due to their low income.
			highest_held_title_tier < tier_county
			scope:recipient.treasury_or_gold < 3000
			add = -50
		}
		modifier = { # Dukes are also somewhat reluctant to request gold if not independent.
			highest_held_title_tier = tier_duchy
			top_liege != this
			scope:recipient.treasury_or_gold < 3000
			add = -25
		}
		modifier = { # dont spam on player HoF
			scope:recipient = {
				is_ai = no
				faith.religious_head ?= this
			}
			add = -50
		}

		modifier = { # I am low on money while at war
			add = 50
			scope:actor = {
				is_at_war = yes
				gold < medium_gold_value
			}
		}
		modifier = { # I am in debt
			add = 50
			scope:actor = {
				gold < 0
			}
			#Pope demanding hook in return
			NOT = {
				scope:recipient = {
					OR = {
						ai_honor < 0
						ai_zeal < low_negative_ai_value
					}
				}
			}
		}
		modifier = { # I am badly in debt
			add = 50
			scope:actor = {
				gold < massive_gold_value_negative
			}
		}
		modifier = { # Piety to burn, and no reason to keep it.
			add = 25
			scope:actor = {
				piety > massive_piety_value
				NOR = {
					has_character_modifier = vow_of_poverty_modifier
					has_trait = generous
				}
			}
		}
		modifier = { # Or when greedy
			add = scope:actor.ai_greed
			scope:actor = {
				ai_greed > 0
			}
		}

		modifier = { # If the Pope has overflowing coffers, it'd be rather impious to NOT ask him to give some to a fellow, poor and struggling christian!
			add = {
				value = scope:recipient.treasury_or_gold
				subtract = 1000
				divide = 100
				max = 50
			}
			scope:actor = {
				OR = {
					ai_has_economical_boom_personality = yes
					piety > major_piety_value
					AND = {
						ai_has_warlike_personality = yes
						war_chest_gold < war_chest_gold_maximum
					}
				}
			}
			scope:recipient.treasury_or_gold >= 1000
		}

		modifier = { #That's _my_ piety
			add = -100
			scope:actor = { ai_has_builder_or_pious_builder_personality = yes }
		}

		modifier = { #Go on! Git!
			add = -100
			scope:actor = { government_has_flag = government_is_mandala }
		}
	}
}

hof_ask_for_claim_interaction = {
	category = interaction_category_diplomacy
	icon = icon_clergy_claim_title
	interface_priority = 35

	desc = hof_ask_for_claim_interaction_desc
	notification_text = hof_ask_for_claim_interaction_notification
	target_type = title
	target_filter = secondary_recipient_domain_titles
	redirect = {
		if = {
			limit = {
				exists = scope:actor.religious_head_or_challenger
			}
			scope:recipient = {
				save_scope_as = secondary_recipient
			}
			scope:actor.religious_head_or_challenger = {
				save_scope_as = recipient
			}
		}
	}

	has_valid_target_showing_failures_only = {
		scope:target = {
			tier >= tier_county
			is_landless_type_title = no
		}
		trigger_if = { # Admin can't request claim on other admin titles, only the top liege's primary title is up for grabs
			limit = {
				scope:secondary_recipient = {
					government_has_mechanic = administrative
					top_liege = this
				}
				scope:actor = {
					government_has_mechanic = administrative
				}
			}
			scope:target = {
				this = scope:secondary_recipient.primary_title
			}
		}
	}

	cooldown_against_recipient = { years = 3 }

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			piety >= major_piety_value
			exists = religious_head_or_challenger
			is_playable_character = yes
			OR = {
				ai_greed >= 50
				ai_has_warlike_personality = yes
			}
			NOR = {
				has_trait = excommunicated
				has_character_flag = hof_request_cooldown
			}
		}
		NOT = {
			has_character_modifier = promised_pilgrimage_to_clergy_modifier
		}
	}

	is_shown = {
		#Pam DLC has new version
		has_pam_dlc_trigger = no
		exists = scope:actor.religious_head_or_challenger
		scope:actor != scope:actor.religious_head_or_challenger
		scope:recipient.religious_head_or_challenger ?= scope:actor.religious_head_or_challenger
		scope:actor.faith ?= scope:secondary_recipient.faith
		NOR = {
			scope:actor = scope:secondary_recipient
			scope:secondary_recipient ?= scope:recipient
		}
		scope:actor.faith ?= {
			has_doctrine = doctrine_spiritual_head
		}
		scope:secondary_recipient ?= {
			is_landed_or_landless_administrative = yes
			primary_title ?= { tier >= tier_county }
			NOT = { government_has_flag = government_is_holy_order }
		}
		# Warcraft
		trigger_if = {
			limit = { scope:actor = { is_ai = yes } }
			scope:recipient = {
				NOT = { has_variable = 1_year_cd_claim_kal }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			pam_antipope_clerical_request_trigger = { RECIPIENT = scope:recipient }
		}
		pam_not_promised_pilgrimage_trigger = { ACTOR = scope:actor }
		trigger_if = { # Can't request on titles you already have a pressed claim on
			limit = {
				exists = scope:target
				scope:actor = {
					has_strong_claim_on = scope:target
				}
			}
			custom_description = {
				text = "hof_ask_for_claim_has_pressed_claim_already"
				always = no
			}
		}
		trigger_if = { # Can't request claims on baronies
			limit = {
				OR = {
					scope:recipient.highest_held_title_tier = tier_barony
					AND = {
						exists = scope:target
						scope:target = {
							tier = tier_barony
						}
					}
				}
			}
			custom_description = {
				text = "hof_ask_for_claim_no_baronies"
				always = no
			}
		}
		trigger_if = { # Can't request claims on theocratic lessees or theocracies (= Prince-Bishops)
			limit = {
				scope:secondary_recipient = {
					government_has_flag = government_is_theocracy
				}
			}
			custom_description = {
				text = "hof_ask_for_claim_theocratic_lessee"
				always = no
			}
		}
		trigger_if = { # Admin can't request claims on other admin titles
			limit = {
				scope:secondary_recipient = {
					government_has_mechanic = administrative
					NOT = { top_liege = this }
				}
				scope:actor = {
					government_has_mechanic = administrative
				}
			}
			custom_description = {
				text = "hof_ask_for_claim_on_admin_titles"
				always = no
			}
		}
		scope:actor = { # Variable piety cost depending on tier
			trigger_if = {
				limit = {
					exists = scope:target
				}
				piety >= claim_piety_value
			}
			custom_description = {
				text = "hof_ask_for_claim_excommunicated"
				NOT = { has_trait = excommunicated }
			}
			NOT = {
				is_at_war_with = scope:recipient
			}
		}
		scope:actor = {
			custom_tooltip = {
				text = HOF_INTERACTION_UNREFORMED_FAITH
				NOT = {
					faith = { has_doctrine_parameter = unreformed }
				}
			}
		}

		scope:recipient = {
			#Any member of your own house. Lowborn clerics have none, hence the safe scope switches below.
			house ?= {
				every_house_member = {
					limit = {
						# Cannot select them self as beneficiary
						this != scope:recipient
					}
					add_to_list = characters
				}
			}
			#Any nearby family member who is of your dynasty, even if they're a separate house.
			every_close_or_extended_family_member = {
				limit = {
					dynasty ?= scope:recipient.dynasty
					NOT = { house ?= scope:recipient.house }
				}
				add_to_list = characters
			}
			#And then any landless heads of minor houses in your dynasty.
			dynasty ?= {
				every_dynasty_member = {
					limit = {
						this = this.house.house_head
						NOR = {
							this.house = scope:recipient.house
							any_close_or_extended_family_member = { this = scope:recipient }
						}
						is_ruler = no
					}
					add_to_list = characters
				}

			if = {
				limit = { faith.great_holy_war.ghw_war_declarer ?= scope:recipient }

					limit = {
						is_adult = yes
						is_ruler = no
					}
					add_to_list = characters
				}
			}
			# If you're a laamp, add you too.
			if = {
				limit = { government_has_flag = government_is_landless_adventurer }
