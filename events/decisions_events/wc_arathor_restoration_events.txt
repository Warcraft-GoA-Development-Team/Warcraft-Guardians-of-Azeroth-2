arathor_restoration.0100 = {
    type = character_event
    content_source = dlc_GOA
	window = fullscreen_event
	title = arathor_restoration.0100.t
	desc = arathor_restoration.0100.desc
	theme = emperor
	override_background = { reference = ep3_fullscreen_restore_rome }

	immediate = {
		save_scope_as = arathor_hegemon

        play_music_cue = mx_cue_epic_sacral_moment
	}

	option = { # Stay peaceful and rule peacefully
		name = arathor_restoration.0100.a
        if = {
            limit = { has_dlc_feature = legends }
            create_legend_seed = {
                type = legitimizing
                quality = illustrious
                chronicle = new_title
                properties = {
                    title = title:h_arathor
                    founder = root
                }
            }
        }
        ai_chance = {
            base = 40
            modifier = {
                add = 15
                has_trait = humble
            }
            modifier = {
                add = 30
                has_trait = craven
            }
            modifier = {
                add = 15
                has_trait = content
            }
        }
	}

    option = { # Lay the claim on all of Humanity's Kingdoms!
        name = arathor_restoration.0100.b
        add_pressed_claim = title:k_kul_tiras
        add_pressed_claim = title:k_mechagon
        add_pressed_claim = title:k_plunder_isle
        add_pressed_claim = title:k_tol_barad
        add_pressed_claim = title:k_stormwind
        ai_chance = {
            base = 60
            modifier = {
                add = 10
                has_trait = greedy
            }
            modifier = {
                add = 15
                has_trait = arrogant
            }
            modifier = {
                add = 15
                has_trait = ambitious
            }
        }
    }

	after = {
        create_arathor_empire_scripted_effect = yes
		set_nickname_effect = { NICKNAME = nick_the_glorious }
		# Send narrative fluff to other players (if any).
		every_player = {
			limit = { this != root }
			trigger_event = arathor_restoration.0101
		}
	}
}

arathor_restoration.0101 = {
    type = character_event
    content_source = dlc_GOA
	window = fullscreen_event
	title = arathor_restoration.0101.t
	desc = arathor_restoration.0101.desc
	theme = emperor
	override_background = { reference = ep3_fullscreen_restore_rome }
	
	immediate = {
		#Same-faith non-empire tier rulers are over-awed.

		}
		#Otherwise, this is a tacit statement of imperial threat.
		else = { play_music_cue = "mx_cue_combat_2" }
	}

	option = { 
		name = arathor_restoration.0101.a
	}
}
