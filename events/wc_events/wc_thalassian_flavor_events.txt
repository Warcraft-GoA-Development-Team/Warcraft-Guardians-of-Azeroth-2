### THALASSIAN FLAVOR EVENTS ###
# 0010 - A House in Decline

# 0010 - An old Thalassian house sells part of its ancestral library
wc_thalassian_flavor_events.0010 = {
    type = character_event
    content_source = dlc_GOA
    title = wc_thalassian_flavor_events.0010.t
    desc = wc_thalassian_flavor_events.0010.d
    theme = order
    override_background = {
        reference = wc_background_silvermoon
    }

    trigger = {
        NOT = { has_character_flag = wc_thalassian_flavor_0010_flag }
        has_culture = culture:high_elf
        has_trait = user_of_order_magic_1
        capital_province = {
            geographical_region = world_eastern_kingdoms_quelthalas_quelthalas
        }
        NOT = {
            title:b_sunwell_plateau.title_province = {
                has_building = sunwell_inert
            }
        }
    }

    immediate = {
        add_character_flag = wc_thalassian_flavor_0010_flag
    }

    left_portrait = {
        character = root
        animation = thinking
    }

    # One volume should be enough.
    option = {
        name = wc_thalassian_flavor_events.0010.a
        trigger = { gold >= 50 }
        show_as_unavailable = { always = yes }
        remove_short_term_gold = 50
        add_order_magic_lifestyle_perk_points = 1
    }

    # I will take the entire collection.
    option = {
        name = wc_thalassian_flavor_events.0010.b
        trigger = { gold >= 150 }
        show_as_unavailable = { always = yes }
        remove_short_term_gold = 150
        add_order_magic_lifestyle_perk_points = 3
    }

    # I know collectors who will pay handsomely.
    option = {
        name = wc_thalassian_flavor_events.0010.c
        add_gold = 25
    }
}

# 0020 - The Ancient Magister

# 0020 - An aged High Elf magister seeks a final appointment as Court Mage
wc_thalassian_flavor_events.0020 = {
    type = character_event
    content_source = dlc_GOA
    title = wc_thalassian_flavor_events.0020.t
    desc = wc_thalassian_flavor_events.0020.d
    theme = order
    override_background = {
        reference = wc_background_sunwell_active
    }

    trigger = {
        NOT = { has_character_flag = wc_thalassian_flavor_0020_flag }
        has_culture = culture:high_elf
        exists = liege
        NOT = {
            title:b_sunwell_plateau.title_province = {
                has_building = sunwell_inert
            }
        }
    }

    immediate = {
        add_character_flag = wc_thalassian_flavor_0020_flag

        create_character = {
            template = wc_thalassian_ancient_magister_character
            location = root.location
            save_scope_as = ancient_magister
        }

        scope:ancient_magister = {
            hidden_effect = {
                add_character_modifier = {
                    modifier = wc_thalassian_ancient_magister_lifespan
                }
            }
        }
    }

    left_portrait = {
        character = scope:ancient_magister
        animation = thinking
    }

    # Offer him a place at court. Never replaces an existing Court Mage.
    option = {
        name = wc_thalassian_flavor_events.0020.a
        trigger = {
            custom_description = {
                text = wc_thalassian_flavor_events.0020.a_tt_no_court_mage
                NOT = { any_court_position_holder = { type = court_caster_court_position count > 0 } }
            }
            custom_description = {
                text = wc_thalassian_flavor_events.0020.a_tt_can_employ
                can_employ_court_position_type = court_caster_court_position
            }
        }
        show_as_unavailable = { always = yes }
        root = {
            add_courtier = scope:ancient_magister
            appoint_court_position = {
                recipient = scope:ancient_magister
                court_position = court_caster_court_position
            }
        }
        reverse_add_opinion = {
            target = root.liege
            modifier = disappointed_opinion
            opinion = -20
        }
    }

    # Send him to my liege.
    option = {
        name = wc_thalassian_flavor_events.0020.b
        root = {
            liege = {
                add_courtier = scope:ancient_magister
                if = {
                    limit = {
                        can_appoint_char_to_court_position = {
                            CHAR = scope:ancient_magister
                            COURT_POS = court_caster_court_position
                        }
                    }
                    appoint_court_position = {
                        recipient = scope:ancient_magister
                        court_position = court_caster_court_position
                    }
                }
            }
        }
        reverse_add_opinion = {
            target = root.liege
            modifier = grateful_opinion
            opinion = 25
        }
    }

    # Let him go his own way. He stays unaffiliated and wanders.
    option = {
        name = wc_thalassian_flavor_events.0020.c
        add_prestige = 50
    }
}

# 0030 - The Old Border

# 0030 - Amani warbands begin coordinating along the southern border of Quel'Thalas
wc_thalassian_flavor_events.0030 = {
    type = character_event
    content_source = dlc_GOA
    title = wc_thalassian_flavor_events.0030.t
    desc = wc_thalassian_flavor_events.0030.d
    theme = realm
    override_background = {
        reference = wc_background_eversong
    }

    trigger = {
        NOT = { has_character_flag = wc_thalassian_flavor_0030_flag }
        has_culture = culture:high_elf
        is_landed = yes
        capital_province = { geographical_region = world_eastern_kingdoms_quelthalas_quelthalas }
        current_date >= 583.1.1
        current_date <= 602.12.31
        NOT = {
            any_war_enemy = {
                capital_province = { geographical_region = world_eastern_kingdoms_quelthalas_zulaman }
            }
        }
    }

    immediate = {
        add_character_flag = wc_thalassian_flavor_0030_flag
    }

    # The Farstriders will remind them where the border lies.
    option = {
        name = wc_thalassian_flavor_events.0030.a
        add_character_modifier = {
            modifier = wc_thalassian_old_border_counter_tactics_modifier
            years = 5
        }
        culture:high_elf = {
            change_cultural_acceptance = {
                target = culture:amani
                value = -3
                desc = wc_thalassian_flavor_events.0030.a_cultural_acceptance
            }
        }
    }

    # Strengthen the passes. Let them come.
    option = {
        name = wc_thalassian_flavor_events.0030.b
        trigger = { gold >= 50 }
        show_as_unavailable = { always = yes }
        remove_short_term_gold = 50
        hidden_effect = {
            every_sub_realm_county = {
                limit = {
                    OR = {
                        de_jure_liege = title:d_thalassian_pass
                        de_jure_liege = title:d_tranquillien
                        de_jure_liege = title:d_shalandis
                        this = title:c_farstrider_enclave
                    }
                }
                add_county_modifier = {
                    modifier = wc_thalassian_old_border_fortifications_modifier
                }
            }
        }
    }

    # Do not provoke them. I want to know who is organizing these raids.
    option = {
        name = wc_thalassian_flavor_events.0030.c
        add_character_modifier = {
            modifier = wc_thalassian_old_border_intelligence_modifier
            years = 5
        }
    }
}

            add_county_modifier = {
