GetCrisisStatusGUI = {
	type = landed_title
	
	text = {
		trigger = {
			exists = holder
			exists = var:legions_puppet
		}
		localization_key = "CRISIS_STATUS_LEGIONS_PUPPET"
	}
	text = {
		trigger = {
			OR = {
				AND = {
					exists = holder
					NOT = { root = title:e_pandaria_empire }
				}
				title:e_pandaria_empire.holder ?= {
                	is_mogu_thunder_king_trigger = yes
            	}
			}
		}
		localization_key = "CRISIS_STATUS_ACTIVE"
	}
	text = {
		trigger = { always = no }
		fallback = yes
		localization_key = "CRISIS_STATUS_UNACTIVE"
	}
}
