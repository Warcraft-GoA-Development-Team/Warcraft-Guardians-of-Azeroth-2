is_valid_auto_title_holder = {
	is_capable_adult = yes
	is_imprisoned = no
	NOT = { always = no } # Warcraft - merchant_var is never set (Paiza merchants removed)
}

is_valid_auto_title_holder_clergy = {
	scope:base.rite = { rite_has_allowed_gender_for_clergy = root }
	is_married = no
	is_betrothed = no
	NOT = { has_trait = excommunicated }
}

is_valid_auto_title_holder_non_clergy = {
	trigger_if = {
		limit = {
			scope:base = { has_realm_law = male_only_law }
		}
		is_male = yes
	}
	trigger_else_if = {
		limit = {
			scope:base = { has_realm_law = female_only_law }
		}
		is_female = yes
	}
	trigger_else = {}
}
