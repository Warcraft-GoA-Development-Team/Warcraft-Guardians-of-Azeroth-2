		is_adult = yes
		faith = scope:top_liege_temp.faith # Must be of the same faith

			text = can_be_appointed_interaction_tt
			OR = {
				can_be_granted_theocratic_titles_by = { RULER = scope:top_liege_temp } # Gender checks
				AND = {
					is_diarch_of_target = scope:top_liege_temp
					scope:top_liege_temp = { has_diarchy_active_parameter = diarchy_is_co_rulership }
				}
			}
		}
		custom_tooltip = {
			text = can_be_appointed_interaction_clergy_tt
			OR = { # Clergy or Lay Clergy
				is_clergy = yes
				rite = { rite_has_parameter = lay_clergy } # HoF cannot be temporal if you have Theocratic clergy, so no need to check for that
				this = scope:top_liege_temp.player_heir # Designated Theocratic Heir
			}
		}
		custom_tooltip = { # Only theocracies
			text = only_theocratic_tt
			OR = {
				government_has_flag = government_is_theocracy
				is_landed = no
			}
		}
		custom_tooltip = { # Barred from holding titles
			text = can_be_appointed_interaction_can_inherit_tt
			NOR = {
				has_trait = order_member
				has_trait = gallivanter
				has_trait = disinherited
			}
		}
	}
}

		is_at_war_with = scope:puppet_or_actor
		has_trait = excommunicated
	}

		limit = { exists = scope:recipient }

			this = scope:recipient

		}

		incapable = no

		scope:puppeteer ?= scope:puppet_or_actor
	}
}
clergy_claim_title_responsibility_trigger = {
	OR = {
		#Clerical region overlaps
		AND = {
			any_held_title = {
				has_clerical_region = yes
				any_county_in_clerical_region = {
					any_this_title_or_de_jure_above = {
						holder ?= scope:recipient
					}
				}
			}
		}
		#Landed title overlaps
		AND = {
			is_landed = yes
			any_held_title = {
				OR = {
					#they have a title beneath recipient's titles
					any_this_title_or_de_jure_above = {
						holder ?= scope:recipient
					}
					#they have a title that presides over recipient's titles
					any_in_de_jure_hierarchy = {
						holder ?= scope:recipient
					}
				}
			}
		}
	}
}

#When AI should be bothering player clergy with requests
pam_should_bother_clergy_interaction_trigger = {
	$PRIEST$ = {
		trigger_if = {
			limit = {
				scope:puppet_or_actor = { is_ai = yes }
				is_ai = no
			}
			OR = {
				scope:puppet_or_actor.highest_held_title_tier >= $PRIEST$.highest_held_title_tier
				AND = {
					scope:puppet_or_actor = { is_vassal_of = $PRIEST$ }
					scope:puppet_or_actor.highest_held_title_tier >= $PRIEST$.highest_held_title_tier_minus_one
				}
				is_allied_to = scope:puppet_or_actor
				AND = {
					exists = house
					house ?= scope:puppet_or_actor.house
				}
				is_heir_of = scope:puppet_or_actor
				any_heir = {
					this = scope:puppet_or_actor
				}
			}
		}
	}
}

# $ACTOR$ and $RECIPIENT$ dioceses border or are nearby, and the scan is bounded by the actor's diocese
pam_rite_sway_adjacent_diocese_trigger = {
	OR = {
		$ACTOR$.capital_county.clerical_region_title ?= {
			any_county_in_clerical_region = {
				any_neighboring_county = {
					clerical_region_title ?= $RECIPIENT$.capital_county.clerical_region_title
				}
			}
		}
		AND = {
			# No diocese seat on the far end means there is no adjacent diocese to measure
			exists = $RECIPIENT$.capital_county.clerical_region_title.title_capital_county
			$ACTOR$.capital_county.clerical_region_title.title_capital_county ?= {
				squared_distance = {
					target = $RECIPIENT$.capital_county.clerical_region_title.title_capital_county
					value <= squared_distance_small
				}
			}
		}
	}
}

# We put quick checks first and leave the bordering realm scan last, behind a capital distance check
pam_rite_sway_clergy_reach_trigger = {
	OR = {
		custom_tooltip = {
			text = sway_ruler_to_my_rite_interaction_flock_reach_tt
			OR = {
				$RECIPIENT$ = {
					superior ?= $ACTOR$
				}
				trigger_if = {
					limit = { $RECIPIENT$ = { is_ruler = yes } }
					$RECIPIENT$.capital_county.clerical_region_title.holder ?= $ACTOR$
				}
				trigger_else = {
					$RECIPIENT$.location.county.clerical_region_title.holder ?= $ACTOR$
				}
			}
		}
		custom_tooltip = {
			text = sway_ruler_to_my_rite_interaction_prelate_reach_tt
			pam_rite_sway_peer_prelate_lane_trigger = {
				ACTOR = $ACTOR$
				RECIPIENT = $RECIPIENT$
			}
		}
		custom_tooltip = {
			text = sway_ruler_to_my_rite_interaction_liege_reach_tt
			pam_rite_sway_liege_lane_trigger = {
				ACTOR = $ACTOR$
				RECIPIENT = $RECIPIENT$
			}
		}
		custom_tooltip = {
			text = sway_ruler_to_my_rite_interaction_border_reach_tt
			# A landless recipient has no capital county for the distance pre-gate to measure
			exists = $RECIPIENT$.capital_county
			$ACTOR$.capital_county ?= {
				squared_distance = {
					target = $RECIPIENT$.capital_county
					value <= squared_distance_large
				}
			}
			$ACTOR$ = {
				any_neighboring_realm_owner = { this = $RECIPIENT$ }
			}
		}
	}
}

#The recipient of the "Give me land" interaction will back down without any convincing required
#The crit, the instant success
pam_instant_negotiate_land_success_trigger = {
	scope:recipient = {
		is_ai = yes
	}
	OR = {
		#They don't have the legitimacy or piety
		scope:recipient = {
			piety_level <= 0
			legitimacy_level <= 0
		}
		#You have the pope in your pocket
		trigger_if = {
			limit = {
				exists = scope:secondary_recipient
			}
			pam_negotiate_land_instant_pope_hook_trigger = yes
		}
		#You intimidate them militarily or through dread
		OR = {
			#Mostly milstrength
			AND = {
				#Milstrength 4x as high
				half_current_military_strength >= scope:recipient.twice_current_military_strength
				has_dread_level_towards = {
					target = scope:recipient
					level >= 1
				}
			}
			#Milstrength and intimidation
			AND = {
				#Milstrength twice as high
				current_military_strength >= scope:recipient.twice_current_military_strength
				has_dread_level_towards = {
					target = scope:recipient
					level >= 2
				}
			}
		}
	}
}

pam_negotiate_land_instant_pope_hook_trigger = {
	has_strong_hook = scope:secondary_recipient
	highest_held_title_tier >= scope:recipient.highest_held_title_tier
	#Won't give away primary title or its aboves
	scope:target = {
		NOR = {
			this = scope:recipient.primary_title
			any_in_de_jure_hierarchy = {
				this = scope:recipient.primary_title
			}
		}
	}
	scope:target.tier < tier_kingdom
	faith.religious_head = scope:secondary_recipient
	scope:recipient = {
		NOR = {
			has_trait = cynical
			has_trait = excommunicated
		}
		rite = faith.main_rite
	}
}

# Cheap pre-filter for HoF split secular ruler lists (since split_clerical_region_size might not be performant)
pam_potential_secular_clerical_split_ruler_trigger = {
	$RULER$ = {
		is_alive = yes
		is_landed = yes
		faith = scope:actor.faith
		NOR = {
			this = scope:recipient
			trigger_if = {
				limit = { exists = scope:recipient.liege }
				this = scope:recipient.liege
			}
		}
		highest_held_title_tier >= tier_duchy
		NOT = { pam_government_is_religious_playable_trigger = yes }
		capital_county = {
			exists = clerical_region_title
			clerical_region_title.holder = scope:recipient
			# We always keep the region capital, because splitting it off would leave the new region with no capital
			NOT = { this = clerical_region_title.title_preferred_capital_county }
		}
	}
	# any_sub_realm_county, not any_realm_county: for a vassal the latter is the liege's whole realm
	trigger_if = {
		limit = { scope:recipient = scope:actor }
		$RULER$ = {
			NOT = {
				any_sub_realm_county = {
					any_neighboring_county = { this = scope:recipient.capital_county }
				}
			}
		}
	}
}

		save_temporary_scope_as = clerical_region_title_temp

							count >= all

									holder = {
										faith = scope:puppet_or_actor.faith
									}
								}
							}
						}
					}
				}
			}
			trigger_else_if = {
				limit = {
					scope:target = {
						tier = tier_duchy
					}
				}
				any_neighboring_county = {
					duchy = scope:target
					NOT = { exists = scope:clerical_region_title }
				}
				scope:target = {
					#All counties must have no clerical region
					#All counties must be within recipient's realm OR held by a same-faith ruler

						count >= all

							}
						}

				}

				}

					any_direct_de_jure_vassal_title = {
						#All counties must have no clerical region

							}

								}
							}

	tier = tier_duchy
	save_temporary_scope_as = duchy_temp
    NOR = {
        $NEW_KING$ = {
            completely_controls = prev
        }
        holder ?= $NEW_KING$
        holder ?= {
            target_is_liege_or_above = $NEW_KING$
        }
    }
    NOT = {
    	any_de_jure_county = {
    		kingdom.title_capital_county ?= this
    	}
    }
    #Is not only duchy in kingdom
    kingdom = {
    	any_direct_de_jure_vassal_title = {
    		tier = tier_duchy
    		NOR = {
    			this = scope:duchy_temp
    			$NEW_KING$ = {
        		    completely_controls = prev
        		}
        		holder ?= $NEW_KING$
        		holder ?= {
        		    target_is_liege_or_above = $NEW_KING$
        		}
    		}
    	}
    }
    #Is not controlled by de jure liege
   	trigger_if = {
   		limit = {
   			exists = kingdom.holder
   		}
   		kingdom.holder = {
   			NOR = {
   				this = scope:duchy_temp.holder
   				any_vassal_or_below = {
   					this = scope:duchy_temp.holder
   				}
   				completely_controls = scope:duchy_temp
   			}
   		}
   	}
    #Not bordering any other county of de jure kingdom (not held by new king)
    NOT = {
        any_de_jure_county = {
            save_temporary_scope_as = county_temp
            
            any_neighboring_county = {
                kingdom = scope:county_temp.kingdom
                duchy = {
                    NOR = {
                        this = scope:county_temp.duchy
                        $NEW_KING$ = {
                            completely_controls = prev
                        }
                        holder ?= {
                            target_is_same_character_or_above = $NEW_KING$
                        }
                    }
                }
            }
        }
    }
    any_de_jure_county = {
        any_neighboring_county = {
            duchy = {
            	OR = {
            		holder ?= {
                		target_is_same_character_or_above = $NEW_KING$
                	}
                	$NEW_KING$ = {
                	    completely_controls = prev
                	}
                }
            }
        }
    }
}

	exists = var:converted_development
	var:converted_development >= 1
	NOT = { rite_has_parameter = unreformed }
	exists = var:associated_culture
	exists = var:associated_county
	save_temporary_scope_as = rite_scope
	trigger_if = {
		limit = {
			root = { exists = capital_province }
			exists = var:associated_county
		}
		"root.capital_province.squared_distance(scope:rite_scope.var:associated_county.title_province)" <= squared_distance_mega
	}
	root = {
		trigger_if = {
			limit = {
				has_variable_list = rites_converted_in_counties
			}
			NOT = {
				is_target_in_variable_list = {
					name = rites_converted_in_counties
					target = scope:rite_scope
				}
			}
		}
		any_sub_realm_county = {
			faith = root.faith
			trigger_if = {
				limit = {
					has_variable_list = rites_converted_in_the_county
				}
				NOT = {
					is_target_in_variable_list = {
						name = rites_converted_in_the_county
						target = scope:rite_scope
					}
				}
			}
		}
	}
	faith = {
		pam_personal_syncretism_faith_matches_trigger = yes
	}
}
