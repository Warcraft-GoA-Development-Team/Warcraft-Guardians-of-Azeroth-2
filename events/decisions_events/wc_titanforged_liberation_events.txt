wc_titanforged_liberation.0002 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_titanforged_liberation.0002.title
	desc = wc_titanforged_liberation.0002.desc
	theme = realm

	immediate = {
		hidden_effect = {
			random_in_list = {
				variable = wc_titanforged_liberated_duchies
				limit = {
					NOT = { any_de_jure_county = { development_level < 15 } }
				}
				save_scope_as = ceded_duchy
			}
			if = {
				limit = {
					exists = faith.religious_head
				}
				faith.religious_head = {
					save_scope_as = my_head_of_faith
				}
			}
		}
	}

	left_portrait = {
		character = root
		animation = emotion_thinking_scepter
	}

	option = {
		name = wc_titanforged_liberation.0002.a
		give_land_to_settler = { TO_WHO = root COUNTY_NAME = scope:ceded_duchy GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
		remove_list_variable = {
			name = wc_titanforged_liberated_duchies
			target = scope:ceded_duchy
		}
	}

	option = {
		trigger = {
			exists = faith.religious_head
			faith = { has_doctrine_parameter = spiritual_head_of_faith }
		}
		name = wc_titanforged_liberation.0002.b
		give_land_to_settler = { TO_WHO = scope:my_head_of_faith COUNTY_NAME = scope:ceded_duchy GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
		remove_list_variable = {
			name = wc_titanforged_liberated_duchies
			target = scope:ceded_duchy
		}
	}

	option = {
		name = wc_titanforged_liberation.0002.c
		give_land_to_settler = { TO_WHO = root COUNTY_NAME = scope:ceded_duchy GIVE_INDEPENDENCY = no ADD_CLAIM_ON_LOSS = no }
		remove_list_variable = {
			name = wc_titanforged_liberated_duchies
			target = scope:ceded_duchy
		}
	}
}
