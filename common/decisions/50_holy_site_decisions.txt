elevate_holy_site_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_mentor_sp.dds"
	}
	decision_group_type = religious

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24 # Head of faiths are Duchy level titles
		kingdom = 24
		empire = 12
		hegemony = 12
	}

	# Opened via OpenGameViewData from the Holy Site window or theological puppet action
	is_invisible = yes

	is_valid_showing_failures_only = {
		exists = scope:selected_item
	}

	is_valid = {
		save_temporary_scope_as = root
		exists = scope:selected_item

		can_manage_faith_holy_sites_trigger = yes

		scope:selected_item ?= {
			root.faith = {
				is_faith_holy_site = prev
				NOT = { is_faith_eminent_holy_site = prev }
				save_temporary_scope_as = faith
				custom_tooltip = {
					text = elevate_holy_site_decision_under_eminent_holy_sites_max
					list_size:eminent_holy_site < root.faith.religion.eminent_holy_sites_max
				}
			}
			custom_tooltip = {
				text = elevate_holy_site_decision_target_in_realm_or_head_of_faith
				OR = {
					root.faith.religious_head ?= root
					holy_site_barony.holder = {
						target_is_same_character_or_above = root
					}
				}
			}
			custom_tooltip = {
				text = elevate_holy_site_decision_county_holder_correct_faith
				holy_site_county.holder.faith = root.faith
			}
		}

		# Show usage of free holy site action, if appropriate
		faith = {
			holy_site_free_action_show_usage_trigger = yes
		}
	}

	cost = {
		piety = {
			value = 1000
			if = {
				limit = { faith = { faith_has_free_holy_site_action_trigger = yes } }
				multiply = 0
			}
		}
	}

	cooldown = { years = holy_site_decision_cooldown_value }

	effect = {
		scope:selected_item.holy_site_county.holder = {
			save_scope_as = county_holder
		}
		root.faith = {
			make_holy_site_eminent = {
				target = scope:selected_item
				actor = root
			}
			consume_free_holy_site_action_effect = yes
		}
		if = {
			limit = {
				exists = scope:county_holder
				NOT = { scope:county_holder = root }
			}
			scope:county_holder = {
				add_opinion = {
					modifier = holy_site_elevated_county_holder_opinion
					target = root
				}
			}
		}
		add_character_modifier = {
			modifier = holy_site_elevated_modifier
			years = 10
		}
	}

	ai_potential = {
		is_independent_ruler = yes
		can_manage_faith_holy_sites_trigger = yes
		faith = {
			list_size:eminent_holy_site < religion.eminent_holy_sites_max
		}
		OR = {
			highest_held_title_tier >= tier_kingdom
			faith.religious_head = this
		}
	}

	ai_will_do = {
		base = 100
	}

	widget = {
		gui = "decision_view_widget_select_holy_site"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT_HOLY_SITE"

		setup_items = {
			scope:actor = {
				faith = {
					every_holy_site = {
						is_eminent = no
						add_to_list = item_list
					}
				}
			}
		}

		default_item = {
			scope:actor = {
				faith = {
					random_holy_site = {
						is_eminent = no
						save_scope_as = selected_item
					}
				}
			}
		}

		ai_select_item = {
			scope:actor = {
				faith = {
					random_holy_site = {
						is_eminent = no
						limit = {
							OR = {
								scope:actor.faith.religious_head =  scope:actor
								holy_site_barony.holder = {
									target_is_same_character_or_above = scope:actor
								}
							}
						}
						save_scope_as = selected_item
					}
				}
			}
		}
	}
}

# AI only - decision to enshrine holy relics into holy sites they have access to.
ai_enshrine_artifact_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/temple.dds"
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 24
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	is_shown = {
		is_ai = yes
	}

	is_valid = {
		is_available_ai_adult = yes
		has_any_artifact = yes

		any_character_artifact = {
			is_holy_relic = yes
			is_holy_relic_for = root.faith

			save_temporary_scope_as = artifact_to_enshrine
			root.faith = {
				any_holy_site = {
					root = {
						can_enshrine_holy_site_artifact = {
							holy_site = prev
							artifact = scope:artifact_to_enshrine
						}
					}
				}
			}
		}
	}

	effect = {
		# Get highest rarity relic owned
		ordered_character_artifact = {
			limit = {
				is_holy_relic_for = root.faith
			}
			max = 1
			order_by = artifact_rarity
			save_scope_as = artifact_to_enshrine
		}

		if = {
			limit = { exists = scope:artifact_to_enshrine }

			# Empty Eminent site preferred
			root.faith = {
				random_eminent_holy_site = {
					limit = {
						total_active_artifact_rarity_sum = 0
						root = {
							can_enshrine_holy_site_artifact = {
								holy_site = prev
								artifact = scope:artifact_to_enshrine
							}
						}
					}
					save_scope_as = target_holy_site
				}

				# Fall back to random mix of eminent/regular sites
				if = {
					limit = {
						NOT = { exists = scope:target_holy_site }
					}
					random_holy_site = {
						limit = {
							root = {
								can_enshrine_holy_site_artifact = {
									holy_site = prev
									artifact = scope:artifact_to_enshrine
								}
							}
						}
						save_scope_as = target_holy_site
					}
				}
			}
		}

		enshrine_holy_site_artifact = {
			holy_site = scope:target_holy_site
			artifact = scope:artifact_to_enshrine
		}
	}

	ai_potential = {
		has_any_artifact = yes
	}
	ai_will_do = { base = 100 }
}

# AI only - decision for zealots to strip hostile-faith relics out of holy sites in counties they directly hold.
#
# Sweep removes every matching artifact in one firing.
ai_remove_enshrined_artifact_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/temple.dds"
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 24
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	is_shown = {
		is_ai = yes
	}

	is_valid = {
		is_available_ai_adult = yes

		# Faith-aligned with top liege (independents pass: top_liege is themselves)
		faith = top_liege.faith
		# Don't mess with top lieges/faiths where player has a say
		top_liege = {
			is_ai = yes
		}
		faith = {
			OR = {
				NOT = { exists = religious_head }
				religious_head = { is_ai = yes }
			}
		}

		any_held_county = {
			any_county_holy_site = {
				holy_site_has_artifact = yes
				save_temporary_scope_as = target_holy_site

				OR = {
					# Case A: own-faith site holds a non-relic-for-us artifact.
					AND = {
						any_holy_site_faith = {
							this = root.faith
						}
						any_holy_site_enshrined_artifact = {
							NOT = { is_holy_relic_for = root.faith }
							root = {
								can_remove_holy_site_artifact = {
									holy_site = scope:target_holy_site
									artifact = prev
								}
							}
						}
					}
					# Case B: hostile-faith site holds a hostile-faith relic.
					AND = {
						any_holy_site_faith = {
							this != root.faith
							num_county_followers > 0
							save_temporary_scope_as = target_faith
							root.faith = {
								faith_hostility_level = {
									target = scope:target_faith
									value >= faith_hostile_level
								}
							}
							scope:target_holy_site = {
								any_holy_site_enshrined_artifact = {
									is_holy_relic_for = scope:target_faith
									root = {
										can_remove_holy_site_artifact = {
											holy_site = scope:target_holy_site
											artifact = prev
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}

	ai_potential = {
		# Zealot-flavored
		OR = {
			has_trait = zealous
			ai_zeal >= 25
		}
	}

	effect = {
		every_held_county = {
			save_temporary_scope_as = held_county
			every_county_holy_site = {
				limit = {
					holy_site_has_artifact = yes
				}
				save_temporary_scope_as = target_holy_site
				every_holy_site_faith = {
					save_temporary_scope_as = target_faith

					# Case A: our faith — remove any artifact that is not a
					# relic for our faith.
					if = {
						limit = {
							this = root.faith
						}
						scope:target_holy_site = {
							every_holy_site_enshrined_artifact = {
								limit = {
									NOT = { is_holy_relic_for = root.faith }
								}
								save_temporary_scope_as = target_artifact
								root = {
									remove_holy_site_artifact = {
										holy_site = scope:target_holy_site
										artifact = scope:target_artifact
									}
								}
							}
						}
					}
					# Case B: not our faith — remove artifacts that are
					# relics for a hostile faith the site recognizes.
					else_if = {
						limit = {
							num_county_followers > 0
							root.faith = {
								faith_hostility_level = {
									target = scope:target_faith
									value >= faith_hostile_level
								}
							}
						}
