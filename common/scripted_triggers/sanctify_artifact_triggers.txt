# Triggers used by sanctify_artifact_interaction.
# Root scope: the artifact being considered.
# RITE = Rite to consider it for
#
# Whether the actor's faith permits sanctifying this artifact.
# The first branch is the baseline open to every Rite; each later branch matches one
# sanctify_artifacts_* tenet parameter to a further category of artifact.
sanctify_artifact_is_eligible_trigger = {
	# A Rite sworn to poverty may never consecrate an illustrious piece, whatever else it permits
	trigger_if = {
		limit = { $RITE$ = { rite_has_parameter = sanctify_artifacts_humble } }
		NOT = { rarity = illustrious }
	}

	OR = {
		# Shrine furniture and reliquary vessels
		AND = {
			rarity_level >= masterwork
			OR = {
				artifact_slot_type = book
				artifact_type = goblet
				artifact_type = jug
				artifact_type = bowl
				artifact_type = plate
				artifact_type = urn
				artifact_type = sculpture
				artifact_type = chest
			}
		}

		# Weapons, armor, war-shields (e.g. tenet_warmonger)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_weapons }
			rarity_level >= masterwork
			OR = {
				artifact_slot_type = primary_armament
				artifact_slot_type = armor
				artifact_type = wall_shield
				artifact_type = wall_shield_special
			}
		}

		# Items made by prestigious dead close family / dynasty members, and cabinets (e.g. tenet_ancestor_worship, tenet_filial_piety)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_ancestral }
			rarity_level >= masterwork
			OR = {
				creator ?= {
					is_alive = no
					OR = {
						prestige_level >= 3 # Illustrious or better
						highest_held_title_tier >= tier_empire # Mostly for historical characters
					}
					OR = {
						is_close_or_extended_family_of = prev.artifact_owner
						dynasty = prev.artifact_owner.dynasty
					}
				}
				artifact_type = cabinet
			}
		}

		# Animal artifacts (e.g. tenet_sanctity_of_nature)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_natural }
			rarity_level >= masterwork
			OR = {
				artifact_type = animal_trinket
				artifact_type = bird_feather
				artifact_type = animal_skull
				artifact_type = animal_hide
				artifact_type = animal_hide_big
			}
		}

		# Symbols of the Earth: Artifacts & sculptures made out of stone, clay, gems
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_earth }
			rarity_level >= masterwork
			OR = {
				has_artifact_feature_group = generic_material_earthware
				has_artifact_feature_group = decoration_material_gem
				# Specific Unique artifacts with big gems
				always = no # Warcraft - real-world artifact variable, never set in GoA
				always = no # Warcraft - real-world artifact variable, never set in GoA
				always = no # Warcraft - real-world artifact variable, never set in GoA
				always = no # Warcraft - real-world artifact variable, never set in GoA
			}
		}

		# Symbols of the people: thrones, tapestries, crowns (e.g. tenet_communal_identity)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_communal }
			rarity_level >= masterwork
			OR = {
				artifact_slot_type = throne
				artifact_slot_type = helmet # aka 'crown'
				artifact_type = tapestry
			}
		}

		# Journals, chronicles, scientific instruments (e.g. tenet_pursuit_of_knowledge)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_knowledge }
			rarity_level >= masterwork
			OR = {
				artifact_type = journal
				artifact_type = chronicle
				artifact_type = scientific_apparatus
			}
		}

		# Celestial instruments (e.g. tenet_astrology)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_celestial }
			rarity_level >= masterwork
			artifact_type = scientific_apparatus
		}

		# Alchemy and occult preparations (e.g. tenet_esotericism)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_occult }
			rarity_level >= masterwork
			OR = {
				artifact_type = elixir
				artifact_type = panacea
				artifact_type = philosophers_stone
			}
		}

		# Humble vessels and texts are allowed at any quality, in trade for the illustrious ban above
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_humble }
			OR = {
				artifact_type = bowl
				artifact_type = plate
				artifact_type = urn
				artifact_slot_type = book
			}
		}

		# Jewelry, goblets, crowns, gems, ornate regalia (e.g. tenet_hedonistic)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_luxurious }
			rarity_level >= masterwork
			OR = {
				artifact_type = brooch
				artifact_type = brooch_pedestal
				artifact_type = ring
				artifact_type = necklace
				artifact_type = necklace_pedestal
				artifact_slot_type = regalia
				artifact_slot_type = helmet # aka crown

				has_artifact_feature_group = decoration_material_gem
				# Specific Unique artifacts with big gems
				always = no # Warcraft - real-world artifact variable, never set in GoA
				always = no # Warcraft - real-world artifact variable, never set in GoA
				always = no # Warcraft - real-world artifact variable, never set in GoA
				always = no # Warcraft - real-world artifact variable, never set in GoA
			}
		}

		# Artifacts of lust, fertility, and the flesh (e.g. tenet_carnal_exaltation)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_debauched }
			OR = {
				# Books on seduction, lovemaking, fertility, and the human body
				has_artifact_feature = book_topic_intrigue_seduction
				has_artifact_feature = book_topic_intrigue_the_art_of_love
				has_artifact_feature = book_topic_intrigue_fertility
				has_artifact_feature = book_topic_learning_anatomy
				has_artifact_feature = book_topic_soulmate
				has_artifact_feature = book_topic_alchemy_the_great_book_of_mercy
				has_artifact_feature = book_topic_alchemy_mercury
				# Specific unique artifacts
				has_variable = naked_tapestry
				has_variable = virility_badge
				has_variable = grotesque_wooden_phallus
				has_variable = commissioned_wooden_phallus
				always = no # Warcraft - real-world artifact variable, never set in GoA
				has_variable = prince_archivist_love_poem
				has_variable = witch_queen_love_poem
				has_variable = lover_keepsake
				has_variable = religious_fertility_statue
				has_variable = de_materia_medica
			}
		}

		# Large reliquary cabinets (e.g. tenet_megaliths)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_megalithic }
			rarity_level >= masterwork
			artifact_type = cabinet
		}

		# Ritual blades (e.g. tenet_human_sacrifice, tenet_fp3_fedayeen)
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_sacrificial }
			rarity_level >= masterwork
			OR = {
				artifact_type = dagger
				artifact_type = sword
			}
		}

		# Regalia and gilded artifacts (e.g. tenet_sun_worship )
		AND = {
			$RITE$ = { rite_has_parameter = sanctify_artifacts_regal }
			rarity_level >= masterwork
			OR = {
				artifact_slot_type = regalia
				artifact_slot_type = helmet
				has_artifact_feature_group = decoration_material_wire
				has_artifact_feature_group = decoration_material_gem
			}
		}

		# Has had a Relic added to it of a specific faith (see event artifact.4090)
		AND = {
			has_variable = added_relic_faith
			var:added_relic_faith = $RITE$.rite_faith
		}
	}
}
