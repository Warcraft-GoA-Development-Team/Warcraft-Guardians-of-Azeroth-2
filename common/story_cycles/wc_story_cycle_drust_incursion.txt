story_drust_incursion = {	
	on_setup = {
		if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:summon_gorak_tul 
					}
				}
			}
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:summon_gorak_tul
			}
		}
		story_owner = {
			save_scope_as = gorak_tul
			var:summoner ?= {
				save_scope_as = summoner
			}
		}

		# Spawn troops
		story_owner = {
			trigger_event = {
				id = drust_crisis.5
				days = 2
			}
		}
	}

	on_owner_death = {
		end_story = yes
		debug_log_date = yes
		set_global_variable = {
			name = drust_incursion_has_ended
			value = yes
		}
	}
	
	# Fire event for AI Gorak Tul to invade
	effect_group = {
		days = { 180 360 }

		trigger = {
			story_owner = {
				is_ai = yes
				top_liege = this
				is_able_to_wage_war_trigger = yes
			}
		}

		triggered_effect = {
			trigger = {
				always = yes
			}
			effect = {
				story_owner = {
					trigger_event =  drust_crisis.8
				}
			}
		}
	}
	
	# Story handling in case of succession weirdness
	effect_group = {
		days = 30

		trigger = {
			story_owner = {
				is_ai = yes
				OR = {
					is_landed = no
					top_liege != this					
				}
			}
		}

		triggered_effect = {
			trigger = {
				always = yes
			}
			effect = {
				story_owner = {
					override_death_effect = { death_reason = death_battle }
				}
				end_story = yes
			}
		}
	}
}
