GetDeathGodDivinityName = {
	type = character

	# Death
	text = {
		localization_key = deathbringer_high_god_name
		trigger = {
			has_title = title:k_frozen_throne
		}
	}

	# Lich Queen
	text = {
		localization_key = death_god_lich_queen_name
		trigger = {
			title:k_frozen_throne.holder ?= { is_female = yes }
		}
	}

	# Lich King
	text = {
		localization_key = death_god_lich_king_name
		trigger = {
			exists = title:k_frozen_throne.holder
		}
	}

	# Death
	text = {
		localization_key = deathbringer_high_god_name
		fallback = yes
	}
}
