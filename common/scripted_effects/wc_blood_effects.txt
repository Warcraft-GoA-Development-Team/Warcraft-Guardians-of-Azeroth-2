try_to_set_blood_effect = {
	# When both parents exist and have race traits
	if = {
		limit = {
			exists = father
			exists = mother
			father = { has_race_trait_trigger = yes }
			mother = { has_race_trait_trigger = yes }
		}
		if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = human }
			}
			set_blood_effect = { BLOOD = blood_human }
		}
		else_if = {
			limit = {
				NOR = {
					has_trait = creature_orc
					has_trait = creature_fel_orc #Fel Orcs give normal orc blood
				}
				OR = {
					AND = {
						exists = real_father
						real_father = { has_trait = creature_orc }
					}
					AND = {
						NOT = { exists = real_father }
						father ?= { has_trait = creature_orc }
					}
					mother = { has_trait = creature_orc }
					AND = {
						exists = real_father
						real_father = { has_trait = creature_fel_orc }
					}
					AND = {
						NOT = { exists = real_father }
						father ?= { has_trait = creature_fel_orc }
					}
					mother = { has_trait = creature_fel_orc }
					AND = {
						OR = {
							AND = {
								exists = real_father
								real_father = { has_trait = blood_orc }
							}
							AND = {
								NOT = { exists = real_father }
								father ?= { has_trait = blood_orc }
							}
						}
						mother = { has_trait = blood_orc }
					}
				}
			}
			set_blood_effect = { BLOOD = blood_orc }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = ogre }
			}
			set_blood_effect = { BLOOD = blood_ogre }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = high_elf }
			}
			set_blood_effect = { BLOOD = blood_high_elf }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = draenei }
			}
			set_blood_effect = { BLOOD = blood_draenei }
		}
		else_if = {
			limit = {
				NOT = {
					has_trait = creature_vrykul
					has_trait = creature_kvaldir #Kvaldir give normal Vrykul blood. Not very likely but just in case.
				}
				OR = {
					AND = {
						exists = real_father
						real_father = { has_trait = creature_vrykul }
					}
					AND = {
						NOT = { exists = real_father }
						father ?= { has_trait = creature_vrykul }
					}
					mother = { has_trait = creature_vrykul }
					AND = {
						exists = real_father
						real_father = { has_trait = creature_kvaldir }
					}
					AND = {
						NOT = { exists = real_father }
						father ?= { has_trait = creature_kvaldir }
					}
					mother = { has_trait = creature_kvaldir }
					AND = {
						OR = {
							AND = {
								exists = real_father
								real_father = { has_trait = blood_vrykul }
							}
							AND = {
								NOT = { exists = real_father }
								father ?= { has_trait = blood_vrykul }
							}
						}
						mother = { has_trait = blood_vrykul }
					}
				}
			}
			set_blood_effect = { BLOOD = blood_vrykul }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = magnataur }
			}
			set_blood_effect = { BLOOD = blood_magnataur }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = taunka }
			}
			set_blood_effect = { BLOOD = blood_taunka }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = night_elf }
			}
			set_blood_effect = { BLOOD = blood_night_elf }
		}
		else_if = {
			limit = {
				NOR = {
					has_trait = creature_tauren
					has_trait = creature_highmountain_tauren #Highmountain Tauren can give normal Tauren blood
				}
				OR = {
					AND = {
						exists = real_father
						real_father = { has_trait = creature_tauren }
					}
					AND = {
						NOT = { exists = real_father }
						father ?= { has_trait = creature_tauren }
					}
					mother = { has_trait = creature_tauren }
					AND = {
						exists = real_father
						real_father = { has_trait = creature_highmountain_tauren }
					}
					AND = {
						NOT = { exists = real_father }
						father ?= { has_trait = creature_highmountain_tauren }
					}
					mother = { has_trait = creature_highmountain_tauren }
					AND = {
						OR = {
							AND = {
								exists = real_father
								real_father = { has_trait = blood_tauren }
							}
							AND = {
								NOT = { exists = real_father }
								father ?= { has_trait = blood_tauren }
							}
						}
						mother = { has_trait = blood_tauren }
					}
				}
			}
			set_blood_effect = { BLOOD = blood_tauren }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = centaur }
			}
			set_blood_effect = { BLOOD = blood_centaur }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = dryad }
			}
			set_blood_effect = { BLOOD = blood_dryad }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = frostnymph }
			}
			set_blood_effect = { BLOOD = blood_frostnymph }
		}
		else_if = {
			limit = {
				can_have_blood_trait_trigger = { RACE = highborne }
			}
			set_blood_effect = { BLOOD = blood_highborne }
		}
		# Same-race blood trait inheritance
		# When parents share a race but one or both carry blood traits from prior mixing
		if = {
			limit = {
				has_blood_trait_trigger = no # No blood assigned by cross-race logic above
				OR = {
					father = { has_blood_trait_trigger = yes }
					mother = { has_blood_trait_trigger = yes }
				}
			}
			random_list = {
				50 = {
					trigger = {
						father = { has_blood_trait_trigger = yes }
					}
					copy_blood_effect = { CHARACTER = father }
				}
				50 = {
					trigger = {
						mother = { has_blood_trait_trigger = yes }
					}
					copy_blood_effect = { CHARACTER = mother }
				}
				50 = {
					# No blood trait inherited - blood diluted over generations
				}
			}
		}
	}
	# Fallback based on the blood gfx
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_human } }
		set_blood_effect = { BLOOD = blood_human }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_orc } }
		set_blood_effect = { BLOOD = blood_orc }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_ogre } }
		set_blood_effect = { BLOOD = blood_ogre }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_high_elf } }
		set_blood_effect = { BLOOD = blood_high_elf }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_draenei } }
		set_blood_effect = { BLOOD = blood_draenei }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_vrykul } }
		set_blood_effect = { BLOOD = blood_vrykul }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_magnataur } }
		set_blood_effect = { BLOOD = blood_magnataur }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_night_elf } }
		set_blood_effect = { BLOOD = blood_night_elf }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_tauren } }
		set_blood_effect = { BLOOD = blood_tauren }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_centaur } }
		set_blood_effect = { BLOOD = blood_centaur }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_dryad } }
		set_blood_effect = { BLOOD = blood_dryad }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_frostnymph } }
		set_blood_effect = { BLOOD = blood_frostnymph }
	}
	else_if = {
		limit = { is_blood_no_gene_trigger = { BLOOD = blood_highborne } }
		set_blood_effect = { BLOOD = blood_highborne }
	}
	else_if = { #Small chance for Kul Tirans to have Vrykul blood
		limit = {
			has_trait = creature_human
			culture = { has_clothing_gfx = small_blood_vrykul_gfx } #a gfx is used and not culture for potential culture hybrids
		}
		random = {
			chance = 5
			set_blood_effect = { BLOOD = blood_vrykul }
		}
	}
}

remove_blood_traits_effect = {
	if = {
		limit = { any_character_trait = { has_trait_flag = blood_trait } }
		random_character_trait = {
			limit = { has_trait_flag = blood_trait }
			prev = { remove_trait = prev }
		}
	}
}

set_blood_effect = {
	remove_blood_traits_effect = yes
	add_trait = $BLOOD$

	add_character_flag = blood_giving_fired_flag
}

copy_blood_effect = {
	# Add new blood always at the end
	$CHARACTER$ = {
		random_character_trait = {
			limit = { has_trait_flag = blood_trait }
			save_temporary_scope_as = this_trait
		}
	}
	if = {
		limit = { exists = scope:this_trait }
		set_blood_effect = { BLOOD = scope:this_trait }
	}
}
