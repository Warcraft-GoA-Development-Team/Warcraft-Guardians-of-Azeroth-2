random_yearly_playable_pulse = {
	on_actions = {
		wc_magic_flavor_events_pulse
	}
}

three_year_playable_pulse = {
	on_actions = {
		court_caster_events_pulse
	}
}

court_caster_events_pulse = {
    random_events = {
        chance_of_no_event = {
			value = 60 # Lower this once we add more events
            if = {
                limit = { is_ai = yes }
                add = 10
            }
            if = {
                limit = {
                    exists = court_position:court_caster_court_position
                    court_position:court_caster_court_position.aptitude:court_caster_court_position <= 2
                }
                add = 20
            }
        }

        # Court caster events
        100 = wc_magic_flavor_events_pulse.0001 # Get some mana
        100 = wc_magic_flavor_events_pulse.0002 # Gain some lifestyle XP
        100 = wc_magic_flavor_events_pulse.0003 # Gain some spell power
        100 = wc_magic_flavor_events_pulse.0004 # Mismanaged casters
        100 = wc_magic_flavor_events_pulse.0005 # Mana Leak
    }
}

# General flavor events for all classes
wc_magic_flavor_events_pulse = {
    random_events = {
        chance_of_no_event = {
			value = 70 # Lower this once we add more events
            if = {
                limit = { is_ai = yes }
                add = 15
            }
        }

        # Flavor events
        100 = wc_magic_flavor_events.0010
        100 = wc_magic_flavor_events.0020
        15 = wc_magic_flavor_events.0030
        10 = all_magic_lifestyle.1000

        # Arcane magic events
        100 = arcane_magic_lifestyle_events.1031 # Time Stops
        10 = arcane_magic_lifestyle_events.1041 # Rift
        100 = arcane_magic_lifestyle_events.1001
        100 = arcane_magic_lifestyle_events.1011
        50 = arcane_magic_lifestyle_events.1021

        # Elemental magic events
        100 = elemental_magic_lifestyle.2011 # playing with fire
        100 = elemental_magic_lifestyle.2021 # totally stoked
        100 = elemental_magic_lifestyle.2031 # fighting with fire
    }
}
