	effect = {
		# Warcraft - Path of the Damned
		root.side_commander = { save_scope_as = attacker }
		enemy_side.side_commander = { save_scope_as = defender }

			limit = {
				scope:attacker = {
					any_owned_story = {
						type = story_path_of_the_damned
						var:story_stage = 2 # Digging up the Dead stage
					}

					NOT = { has_character_flag = post_digging_effect_fired_flag } # Not already done
				}
			}

			battle_against_paladin_digging_up_the_dead_effect = yes
		}
	}
}
