namespace = scarlet

## Events for creation of Scarlet and Argent religion


### Event is intended to proc when murder_terenas_effect is fired during arthas activity 
scarlet.1000 = {
	type = character_event
	content_source = dlc_GOA
	#window = fullscreen_event
	title = SCARL.000_TITLE
	desc = SCARL.000_DESC
	theme = plague
	left_portrait = {
		character = root
		animation = personality_cynical
	}

	immediate = {
		hidden_effect = {
			title:k_order_of_the_silver_hand.holder = { save_scope_as = scoped_silverhand }
			title:k_lordaeron.holder = { save_scope_as = scoped_betrayer }
		}
	}

	option = {
		name = SCARL.000_OPT_A
		every_ruler = {
			limit = {
				any_character_to_title_neighboring_county = {
					empire = title:e_lordaeron
				}	
				NOR = {
					has_trait = being_undead 
			  		has_title = title:k_order_of_the_silver_hand
			  	}
				has_trait = creature_human
				has_religion = religion:tyrism_religion
			}
			trigger_event = {
				id = scarlet.1001
				days = { 3 5 }
			}
		}
		ai_chance = {
			base = 95
		}
	}

	option = {
		name = SCARL.000_OPT_B
		custom_tooltip = SCARL.000_OPT_tt

		ai_chance = {
			base = 5
		}
	}
}

scarlet.1001 = {
	type = letter_event
	content_source = dlc_GOA
	opening = SCARL.001_TITLE
	desc = SCARL.001_DESC
	sender = {
		character = scope:scoped_silverhand
		animation = personality_bold
	}

	trigger = {
		NOR = {
			is_at_war_with = scope:scoped_silverhand
		 	this = scope:scoped_silverhand 
		}
	}

	immediate = {
		save_scope_as = invitation_recipient
	}

	option = { # This sounds lovely, I will come along
		name = SCARL.001_OPT_A
		custom_tooltip = SCARL.001_OPT_A_TT
		scope:scoped_silverhand = {
			add_opinion = {
				modifier = respect_opinion
				target = scope:invitation_recipient
				opinion = 10
			}
		}
		hidden_effect = {
			scope:scoped_silverhand = {
				trigger_event = scarlet.1002 ##Acceptance event
			}
		}
		ai_chance = {
			base = 85
			modifier = {
				add = -1000
				has_trait = craven
			}
			modifier = { ##If they manage to become undead somehow
				add = -1000
				has_trait = being_undead
			}
		}
		trigger_event = {
			id = scarlet.1004
			days = 7
		}
	}

	option = { # I'd rather get rolled by the pack of candle lovers on my border with my 300 stack of levies and no income, thanks
		custom_tooltip = SCARL.001_OPT_B_TT
		name = SCARL.001_OPT_B
		scope:scoped_silverhand = {
			add_opinion = {
				modifier = insult_opinion
				target = scope:invitation_recipient
				opinion = -50
			}
		}
		add_internal_flag = dangerous
		hidden_effect = {
			scope:scoped_silverhand = {
				trigger_event = scarlet.1003 ##Denial
			}
		}
		ai_chance = {
			base = 15
		}
		stress_impact = {
			base = medium_stress_gain
		}
	}
}
