passive_rite_learning_candidate_tenet_is_core_trigger = {
	save_temporary_scope_as = passive_rite_learning_candidate_tenet
	$RITE$ = {
		rite_has_tenet = scope:passive_rite_learning_candidate_tenet
	}
}

passive_rite_learning_candidate_doctrine_is_core_trigger = {
	save_temporary_scope_as = passive_rite_learning_candidate_doctrine
	$RITE$ = {
		rite_has_doctrine = scope:passive_rite_learning_candidate_doctrine
	}
}

passive_rite_learning_doctrine_is_learnable_trigger = {
	# Warcraft
	NOR = {
		this = doctrine:special_doctrine_is_mortal_faith
		this = doctrine:special_doctrine_is_primitive_faith
		this = doctrine:special_doctrine_legions_puppet
		this = doctrine:special_doctrine_eastern_kingdoms_syncretism
		this = doctrine:special_doctrine_is_troll_faith
		this = doctrine:special_doctrine_elunic_gender_stratification
		this = doctrine:special_doctrine_earth_mother_mercy
	}
}

# this = candidate source character
# $LEARNER$ = the learner
is_valid_passive_learning_source_for_learner_trigger = {
	is_available_adult = yes
	faith != $LEARNER$.faith
	save_temporary_scope_as = passive_rite_learning_candidate_source
	$LEARNER$ = {
		OR = {
			any_character_tenet = {
				knowledge_filter = unknown
				passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:passive_rite_learning_candidate_source.rite }
				count >= 1
			}
			any_character_doctrine = {
				knowledge_filter = unknown
				passive_rite_learning_candidate_doctrine_is_core_trigger = { RITE = scope:passive_rite_learning_candidate_source.rite }
				passive_rite_learning_doctrine_is_learnable_trigger = yes
				count >= 1
			}
		}
	}
}

is_valid_passive_learning_rite_for_learner_trigger = {
	this != $LEARNER$.rite
	save_temporary_scope_as = passive_rite_learning_candidate_rite
	$LEARNER$ = {
		OR = {
			any_character_tenet = {
				knowledge_filter = unknown
				passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:passive_rite_learning_candidate_rite }
				count >= 1
			}
			any_character_doctrine = {
				knowledge_filter = unknown
				rite_filter = scope:passive_rite_learning_candidate_rite
				passive_rite_learning_doctrine_is_learnable_trigger = yes
				count >= 1
			}
		}
	}
}

is_apex_partner_passive_learning_source_trigger = {
	is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
	opinion = {
		target = root
		value >= 20
	}
}

is_close_partner_passive_learning_source_trigger = {
	is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
	opinion = {
		target = root
		value >= -20
	}
}
