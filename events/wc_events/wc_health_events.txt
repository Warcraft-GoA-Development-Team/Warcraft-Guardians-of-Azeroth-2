#You become undead via Plague
WCHEA.1001 = {
	type = character_event
	content_source = dlc_GOA
	title = WCHEA_1001_TITLE
	desc = WCHEA_1001_DESC
	theme = healthcare
	left_portrait = {
		character = root
		animation = sick
	}

	trigger = {
		is_vulnerable_to_plague_character_trigger = yes
		NOT = {
			has_character_modifier = important_lore_character
		}
	}
	

	weight_multiplier = {
		base = 1

		common_disease_modifier = yes
	}

	immediate = {
		play_music_cue = "mx_cue_illness"
		
		# For graphics
		hidden_effect = { add_trait = ill }
	}

	# Must feed!
	option = {
		name = WCHEA_1001_OPT_A

		raise_undead_by_move_to_pool_effect = { MASTER = scope:owner }
		
		# For graphics
		hidden_effect = { remove_trait = ill }
	}
}

# Gamestart Historical Ruler Setup
WCHEA.2000 = { # Age assignment for ancient historical rulers # by juketm
	scope = none
	hidden = yes
	
	immediate = {
		# Assigns appropriate Ages to historical characters before our timeframe
		# Usually born around 1.1.1
		
		character:lineofthunderking1 = { # Lei Shen
			change_age = 50
			dynasty = { add_dynasty_prestige = 10000 add_dynasty_prestige_level = 9 }
		}
	}
}

scripted_effect elder_undead_vanish_effect = { # takes one parameter, REASON, a valid death reason
	if = {
		limit = {
			any_vassal = {
				count >= 1
			}
		}
		every_vassal = {
			custom = all_vassals
			add_opinion = {
				modifier = suicide_opinion
				target = root
				opinion = -15
			}
		}
	}
	add_dynasty_prestige = medium_dynasty_prestige_loss
	add_piety = major_piety_loss
	override_death_effect = { death_reason = $REASON$ } 
}

# World Weary - Vanish without a trace
WCHEA.9000 = { 
	type = character_event
	content_source = dlc_GOA
	title = WCHEA.9000.title
	desc = WCHEA.9000.desc

	left_portrait = { 
		character = root
		animation = depressed
	}

	# Do it
	option = { 
		name = WCHEA.9000.opt.a
		random_list = {  
			40 = { 
				modifier = { 
					add = learning
				}
				modifier = { 
					has_trait = reclusive
					add = 10
				}
				elder_undead_vanish_effect = { REASON = death_vanished }
			}
			60 = { 
				modifier = { 
					add = { 
						value = intrigue
						factor = -0.5
					}
				}
				send_interface_toast = { 
					type = event_toast_effect_neutral
					text = world_weary_fail_elder_undead_tt
					left_icon = scope:root
					add_prestige = medium_prestige_loss
				}
			}
		}
	}

	# Back out
	option = { 
		name = WCHEA.9000.opt.b
	}
}
