### Tier: severe negative - capital popular_opinion -51 and below
republic_popular_opinion_event.1001 = {
	type = character_event
	title = republic_popular_opinion_event.1001.t
	desc = republic_popular_opinion_event.1001.desc
	theme = republic

	cooldown = { years = 10 }

	left_portrait = {
		character = root
		animation = worry
		camera = camera_body_left_slight
	}
	right_portrait = {
		character = scope:relation_01
		animation = disapproval
		camera = camera_body
	}

	
	trigger = {
		government_has_flag = government_is_republic
		is_healthy = yes
		is_available_quick = {
			adult = yes
			imprisoned = no
			incapable = no
			in_army = no
			travel = no
		}
		OR = {
			any_relation = {
				type = friend
				type = lover
				this != liege
				is_available_quick = {
					adult = yes
					imprisoned = no
					incapable = no
					in_army = no
				}
			}
			any_close_or_extended_family_member = {
				is_available_quick = {
					adult = yes
					imprisoned = no
					incapable = no
					in_army = no
				}
			}
			any_spouse = {
				this != root.liege
				is_available_quick = {
					adult = yes
					imprisoned = no
					incapable = no
					in_army = no
				}
			}
		}
	}

	immediate = {
		location = { save_scope_as = event_location_01 }
		every_relation = {
			type = friend
			type = lover
			limit = {
				is_available_quick = {
					adult = yes
					imprisoned = no
					incapable = no
					in_army = no
				}
				this != root.liege
				}
			add_to_list = relation_candidates
		}
		every_close_or_extended_family_member = {
			limit = {
				this != root
				is_available_quick = {
					adult = yes
					imprisoned = no
					incapable = no
					in_army = no
				}
				this != root.liege
			}
			add_to_list = relation_candidates
		}
		every_spouse = {
			limit = {
				is_available_quick = {
					adult = yes
					imprisoned = no
					incapable = no
					in_army = no
				}
				this != root.liege
			}
			add_to_list = relation_candidates
		}
		random_in_list = {
			list = relation_candidates
			weight = {
				base = 1
				modifier = {
					add = 30
					has_relation_soulmate = root
				}
				modifier = {
					add = 15
					OR = {
						has_relation_friend = root
						has_relation_lover = root
						is_primary_heir_of = root
					}
				}
				modifier = {
					add = 5
					OR = {
						is_child_of = root
						is_grandchild_of = root
					}
				}
				modifier = {
					factor = 1.5
					opinion = {
						target = root
						value >= 50
					}
				}
				modifier = {
					factor = 0.5
					opinion = {
						target = root
						value < 0
					}
				}
			}
			save_scope_as = relation_01
		}
		if = {
			limit = {
				can_employ_court_position_type = bodyguard_court_position
				short_term_gold >= medium_gold_value
			}
			create_character = {
				template = bodyguard_court_position_template
				location = scope:event_location_01
				culture = root.culture
				faith = root.faith
				save_scope_as = hired_bodyguard
				after_creation = { add_character_flag = created trigger_race_giving_no_gene_effect = yes } # Warcraft
			}
		}
	}
	
	## "You are right. We need to hold our heads high."
	option = {
		name = republic_popular_opinion_event.1001.a
		trigger = {
			NOT = { has_trait = craven } 
		}
		add_character_modifier = {
			modifier = gdp_head_held_high_modifier
			years = 5
		}

		stress_and_fulfillment_impact = {
			# Gain
			humble = minor_stress_impact_gain
			# Loss
			arrogant = minor_stress_impact_loss
		}

		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_trait = humble
			}
		}
	}
	
	## "Might be a good time to get myself a bodyguard."
	option = {
		name = republic_popular_opinion_event.1001.b
		trigger = {
			can_employ_court_position_type = bodyguard_court_position
		}
		remove_short_term_gold = medium_gold_value
		add_courtier = scope:hired_bodyguard
		scope:hired_bodyguard = { remove_character_flag = created }
		appoint_court_position = {
			recipient = scope:hired_bodyguard
			court_position = bodyguard_court_position
		}

		stress_and_fulfillment_impact = {
			# Gain
			trusting = minor_stress_impact_gain
		}

		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_trait = trusting
			}
			modifier = {
				factor = 0
				NOT = { short_term_gold >= medium_gold_value }
			}
		}
	}
	
	## "The mob are going to kill us all!"
	option = {
		name = republic_popular_opinion_event.1001.c
		add_character_modifier = {
			modifier = gdp_fear_of_the_mob_modifier
			years = 5
		}

		stress_and_fulfillment_impact = {
			# Loss
			paranoid = major_stress_impact_loss
			craven = major_stress_impact_loss
		}

		ai_chance = { base = 100 }
	}
	
	## "They won't find me easy prey."
	option = {
		name = republic_popular_opinion_event.1001.d
		trigger = {
			OR = {
				has_trait = brave
				has_trait = wrathful
				has_lifestyle = martial_lifestyle
			}
		}
		add_character_modifier = {
			modifier = gdp_training_for_the_worst_modifier
			years = 5
		}

		stress_and_fulfillment_impact = {
			# Loss
			diligent = minor_stress_impact_loss
			brave = minor_stress_impact_loss
			wrathful = minor_stress_impact_loss
		}

		ai_chance = { base = 100 }
	}

	after = {
		scope:hired_bodyguard = {
			silent_disappear_ai_if_not_hired_with_flag = yes
		}
	}
}

### Tier: mild negative - capital popular_opinion -50 to -1
scripted_effect patrician_aggrandizement_signorial_opinion_effect = {
	every_vassal = {
		limit = {
			this != root
			this.domicile ?= {
				is_domicile_type = merchant_company
			}
			this = {
				government_has_flag = government_is_republic
			}
			this.house ?= {
				OR = {
					has_house_aspiration_parameter = republic_aggrandizement
					has_house_aspiration_parameter = republic_signorial
				}
			}
		}
		custom = custom.every_patrician_aggrandizement_signorial
		this = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 20
			}
		}
	}
}

}

			}
		}
	}
