# Support is a bargain requiring both parties' consent.
wc_nathrezim_cover.0003 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = wc_nathrezim_public_nature }
				desc = wc_nathrezim_cover.0003.title.public
			}
			desc = wc_nathrezim_cover.0003.title
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = wc_nathrezim_public_nature scope:wc_cover_witness = { wc_nathrezim_scarlet_character_trigger = yes } has_strong_hook = scope:wc_cover_witness }
				desc = wc_nathrezim_cover.0003.scarlet_coerced
			}
			triggered_desc = {
				trigger = { has_variable = wc_nathrezim_public_nature scope:wc_cover_witness = { wc_nathrezim_scarlet_character_trigger = yes }  }
				desc = wc_nathrezim_cover.0003.scarlet_collusion
			}
			triggered_desc = {
				trigger = { has_variable = wc_nathrezim_public_nature }
				desc = wc_nathrezim_cover.0003.public
			}
			desc = wc_nathrezim_cover.0003.desc
		}
	}
	theme = intrigue
	left_portrait = { character = scope:wc_cover_witness animation = scheme }
	trigger = { wc_nathrezim_cover_case_valid_trigger = yes }
	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { has_variable = wc_nathrezim_public_nature }
						desc = wc_nathrezim_cover.0003.accept.public
					}
					desc = wc_nathrezim_cover.0003.accept
				}
			}
		}
		trigger = {
			wc_nathrezim_cover_case_valid_trigger = yes
			gold >= 150
			scope:wc_cover_witness = { wc_nathrezim_scarlet_rejection_trigger = no can_add_hook = { target = scope:wc_cover_body type = favor_hook } }
		}
		if = {
			limit = {
				wc_nathrezim_cover_case_valid_trigger = yes
				gold >= 150
				scope:wc_cover_witness = { wc_nathrezim_scarlet_rejection_trigger = no can_add_hook = { target = scope:wc_cover_body type = favor_hook } }
			}
			pay_treasury_or_gold = { target = scope:wc_cover_witness value = 150 }
			scope:wc_cover_witness = { add_hook = { target = scope:wc_cover_body type = favor_hook } }
			wc_nathrezim_settle_witness_effect = yes
		}
		else = { wc_nathrezim_retry_cover_case_effect = yes }
		ai_chance = { base = 70 }
	}
	option = {
		name = wc_nathrezim_cover.0003.refuse
		trigger = { wc_nathrezim_cover_case_valid_trigger = yes }
		trigger_event = wc_nathrezim_cover.0002
		ai_chance = { base = 30 modifier = { add = 50 has_trait = arrogant } }
	}
	option = {
		name = wc_nathrezim_cover.escape
		trigger = { wc_nathrezim_cover_case_valid_trigger = yes wc_nathrezim_can_escape_cover_trigger = yes }
		wc_balnazzar_escape_cover_effect = yes
		ai_chance = { base = 10 }
	}
	option = {
		name = wc_nathrezim_cover.dismiss
		trigger = { wc_nathrezim_cover_case_valid_trigger = no }
		scope:wc_cover_body = { wc_nathrezim_retry_cover_case_effect = yes }
	}
}
