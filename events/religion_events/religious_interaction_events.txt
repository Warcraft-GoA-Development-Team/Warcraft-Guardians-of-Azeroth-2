religious_interaction.1 = { # Faith head divorce event (divorce_character_rel_head interaction)
	type = letter_event
	opening = {
		desc = religious_interaction.1.opening
	}
	desc = {
		desc = religious_interaction.1.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						scope:secondary_recipient = { exposed_cheating_on_spouse_trigger = { SPOUSE = scope:actor } }
						scope:secondary_recipient = {
							any_relation = {
								type = lover
								NOR = {
									this = scope:actor
									this = scope:recipient
								}
								NOT = { has_secret_relation_lover = scope:secondary_recipient }
								relation_with_character_is_incestuous_in_my_rite_trigger = { CHARACTER = scope:secondary_recipient	}
							}
						}
					}
				}
				desc = religious_interaction.1.adultery
			}
			triggered_desc = {
				trigger = {
					scope:actor = {
						OR = {
							is_close_or_extended_family_of = scope:secondary_recipient
							is_cousin_of = scope:secondary_recipient
							dynasty = scope:secondary_recipient.dynasty
						}
					}
				}
				desc = religious_interaction.1.consanguinity
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient = { is_eunuch_trigger = yes }
				}
				desc = religious_interaction.1.eunuch
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient.faith != scope:actor.faith
				}
				desc = religious_interaction.1.different_faith
			}
			desc = religious_interaction.1.fallback
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:price_feud
				}
				desc = religious_interaction.1.price_feud
			}
			triggered_desc = {
				trigger = {
					exists = scope:price_lease
				}
				desc = religious_interaction.1.price_lease
			}
			triggered_desc = {
				trigger = {
					exists = scope:price_hook
				}
				desc = religious_interaction.1.price_hook
			}
			triggered_desc = {
				trigger = {
					exists = scope:price_vows
				}
				desc = religious_interaction.1.price_vows
			}
			triggered_desc = {
				trigger = {
					exists = scope:price_none
				}
				desc = religious_interaction.1.price_none
			}
		}
	}
	sender = scope:recipient

	immediate = {
		scope:divorcer = {
			save_scope_as = actor
		}
	}

	option = {
		name = {
			trigger = {
				exists = scope:price_feud
			}
			text = religious_interaction.1.a_feud
		}
		name = {
			trigger = {
				exists = scope:price_hook
			}
			text = religious_interaction.1.a_hook
		}
		name = {
			trigger = {
				NOR = {
					exists = scope:price_feud
					exists = scope:price_hook
				}
			}
			text = religious_interaction.1.a
		}
		scope:actor = {
			divorce_effect = {
				DIVORCER = scope:actor
				DIVORCEE = scope:secondary_recipient
			}
			scope:recipient = {
				if = {
					limit = {
						exists = scope:price_feud
					}
					divorce_house_feud_effect = yes
				}
				else_if = {
					limit = {
						exists = scope:price_lease
					}
					divorce_holy_order_lease_effect = yes
				}
				else_if = {
					limit = {
						exists = scope:price_hook
					}
					divorce_get_hook_effect = yes
				}
				else_if = {
					limit = {
						exists = scope:price_vows
					}
					divorce_put_spouse_in_monastery_effect = yes
				}
			}
		}
		
		hidden_effect = { # Notify the one being divorced
			scope:secondary_recipient = {
				send_interface_toast = {
					title = X_DIVORCED_ME
					left_icon = scope:recipient
					right_icon = scope:actor
					#custom_tooltip = X_DIVORCED_ME_DESC
				}
			}
		}
	}
}

religious_interaction.1020 = {
	type = letter_event
	opening = {
		desc = religious_interaction.excommunication.opening
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					var:excommunication_reason ?= flag:empire_faith_gate
				}
				desc = religious_interaction.1020.empire
			}
			triggered_desc = {
				trigger = {
					exists = scope:my_sinful_trait
				}
				desc = religious_interaction.1020.sin
			}
			desc = religious_interaction.1020.requested
		}
	}
	sender = {
		character = scope:recipient
		animation = schadenfreude
	}

	trigger = {
		exists = var:excommunication_issuer
		exists = var:requested_my_excommunication
	}

	immediate = {
		var:excommunication_issuer = {
			save_scope_as = recipient
			save_scope_as = excommunication_issuer
		}
		var:requested_my_excommunication = {
			save_scope_as = actor
		}
		add_excommunication_tooltip_only_effect = yes
		# Pick a sinful trait to mention in the letter, if any.
		random_character_trait = {
			limit = {
				save_temporary_scope_as = temp_trait
				root.rite = {
					trait_is_sin_rite = scope:temp_trait
				}
			}
			save_scope_as = my_sinful_trait
		}
	}

	option = {
		name = religious_interaction.1020.a
		flavor = {
			first_valid = {
				triggered_desc = {
					trigger = {
						petition_head_of_faith_repentance_requires_petition_trigger = yes
					}
					desc = religious_interaction.1020.a.tt_petition
				}
				desc = religious_interaction.1020.a.tt_declaration
			}
		}
		add_piety = minor_piety_gain
		stress_and_fulfillment_impact = {
			zealous = minor_stress_impact_loss
			humble = minor_stress_impact_loss
			arrogant = minor_stress_impact_gain
			wrathful = minor_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				OR = {
					has_trait = arrogant
					has_trait = wrathful
					has_trait = vengeful
					has_trait = cynical
				}
			}
		}
	}
