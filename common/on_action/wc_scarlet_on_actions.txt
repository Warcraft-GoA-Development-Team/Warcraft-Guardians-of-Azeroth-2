#### To Keep DNAs intact
on_game_start = { on_actions = { unprune_characters } }

unprune_characters = {
	effect = {
		character:60572 = { make_unprunable = yes } ##Whitemane
	}
}

#### starts checking if scarlet can form when terenas is killed in arthas expedition activity
yearly_global_pulse = { on_actions = { scarlet_start_check } }

scarlet_start_check = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = scarlet_event_chain_started } 
			{ has_global_variable = scourge_lordaeron_complete } 
		}
			random = {
				chance = 0
				modifier = {
					add = 30 
					current_date >= 606.1.1 
				}
				modifier = {
					add = 100 
					current_date >= 610.1.1 
				}
				set_global_variable = scarlet_event_chain_started
				trigger_event = { on_action = scarlet_setup months = { 2 4 } }
			}
		}
	}
}

#### Triggers Scarlet Setup - survivors of lordaeron plague get an option to link up
scarlet_setup = {
	effect = {
		limit = {
			geographical_region = world_eastern_kingdoms_lordaeron
			NOT = { has_trait = being_undead }
			has_trait = creature_human
			has_religion = religion:tyrism_religion
		}
		every_ruler = { trigger_event = scarlet.1000 }
	}
}
