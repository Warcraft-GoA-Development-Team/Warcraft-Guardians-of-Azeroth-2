
# Outcome events for the wc_mograine_assassination scheme
# root here is the scheme owner, Balnazzar (same convention as
# wc_balnazzar_contingency) - he's pulling the strings and the one who sees
# the scheme UI. Renault, the actual instrument, is referenced explicitly by
# character:60019 wherever he needs to appear.

namespace = wc_mograine_assassination_outcome

##################################
# MAINTENANCE EVENTS 0000 - 0099
##################################

# Rolls success/failure when the scheme completes (on_phase_completed pulse)
wc_mograine_assassination_outcome.0001 = {
	hidden = yes

	immediate = {
		random = {
			chance = scope:scheme.scheme_success_chance

			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}

		hidden_effect = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = wc_mograine_assassination_outcome.2001
			}
			else = {
				trigger_event = wc_mograine_assassination_outcome.4001
			}
		}
	}
}

# Failure retry: the flag set by .1004 only lets the decision fire once, so a
# failed scheme re-attempts itself here instead of waiting on the decision to
# reappear - scheduled from .4001's hidden_effect, well past the scheme's own
# cooldown so start_scheme doesn't reject it.
wc_mograine_assassination_outcome.0002 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				root = { is_alive = yes } # Balnazzar
				character:60017 = { is_alive = yes } # Alexandros Mograine
				character:60019 = { is_alive = yes } # Renault Mograine
			}
			start_scheme = {
				type = wc_mograine_assassination
				target_character = character:60017
			}
		}
	}
}

# crime reason) - Alexandros's trust runs too deep to act on suspicion alone,
# and the scheme itself is untouched, still running.

	type = character_event
	title = wc_mograine_assassination_outcome.1004.title
	desc = wc_mograine_assassination_outcome.1004.desc

		exists = character:60017 # Alexandros Mograine

	}

	left_portrait = {
		character = character:60017 # Alexandros Mograine
		animation = worry

		character = root # Balnazzar
		animation = scheme
	}

	immediate = {
		character:60017 = {
			add_opinion = {
				modifier = wc_suspects_scheming_against_me
				target = root
			}
			add_stress = medium_stress_gain
		}
	}

	option = {
		name = wc_mograine_assassination_outcome.1004.a
	}
}

# The Ambush at Stratholme - Renault kills Alexandros with the dropped Ashbringer; kinslaying corrupts it.
# Kel'Thuzad reclaiming it later (death knight arc) is deferred, same as before this rework.
wc_mograine_assassination_outcome.2001 = {
	type = character_event
	window = scheme_successful_event
	title = wc_mograine_assassination_outcome.2001.title
	desc = wc_mograine_assassination_outcome.2001.desc
	theme = death

	override_background = wc_stratholme

	left_portrait = {
		character = character:60017 # Alexandros Mograine
		animation = personality_honorable
	}
	right_portrait = {
		character = character:60019 # Renault Mograine
		animation = disbelief
	}

	immediate = {
		character:60017 = { save_scope_as = alexandros }
	}

	option = {
		name = wc_mograine_assassination_outcome.2001.a

		show_as_tooltip = {
			scope:alexandros = { death = { death_reason = death_battle killer = character:60019 } }
		}
		hidden_effect = {
			scope:alexandros = {
				death = { death_reason = death_battle killer = character:60019 }
			}

			character:60019 = {
				# Destroy inherited uncorrupted Ashbringer before minting corrupted one,
				# or Renault ends up carrying both.
				every_character_artifact = {
					limit = { has_variable = ashbringer }
					destroy_artifact = this
				}

				# Kinslaying corrupts blade; Renault leaves it behind, Kel'Thuzad reclaims it later (deferred, no system yet).
				# ponytail: sits owned-unequipped on Renault instead - inheritance would hand it straight back to his heir.
				create_artifact_corrupted_ashbringer_effect = { OWNER = this }

				# add_hook's strong_blackmail_hook requires an actual secret behind
				# it (confirmed via error.log: "requires a secret but none was set").
				# Renault owns the murder secret - he's the one who did it.
				add_secret = {
					type = secret_murder
					target = scope:alexandros
				}
				random_secret = {
					limit = {
						secret_type = secret_murder
						secret_target = scope:alexandros
					}
					save_scope_as = renault_murder_secret
				}
			}

			# Balnazzar ordered it too - his own secret, not just Renault's. Without
			# this, the only thing exposure/discovery can ever implicate is Renault;
			# the mastermind who set the whole thing up carries no risk of his own.
			add_secret = {
				type = secret_murder
				target = scope:alexandros
			}

			# Balnazzar holds this over Renault now - real leverage, not just narration.
			add_hook = {
				type = strong_blackmail_hook
				target = character:60019 # Renault Mograine
				secret = scope:renault_murder_secret
			}

			if = {
				limit = { exists = scope:scheme }
				scope:scheme = { end_scheme = yes }
			}

			# root = Balnazzar throughout this file (scheme owner) - .1006 assumes
			# the same, so a plain trigger_event is enough, no rerouting needed.
			trigger_event = {
				id = wc_balnazzar_story.1006
				days = { 10 25 }
			}

			trigger_event = {
				id = wc_mograine_assassination_outcome.2002
				days = { 5 15 }
			}
		}
	}
}
