
character_gender_can_rule_title_trigger = {
	trigger_if = {
		limit = {
			$GENDER$ = { is_male = yes }
			exists = $TITLE$
			$TITLE$ = {
				NOT = {
					has_title_law = female_only_law
				}
			}
		}
		always = yes
	}
	trigger_else_if = {
		limit = {
			$GENDER$ = { is_female = yes }
			exists = $TITLE$
			$TITLE$ = {
				NOT = {
					has_title_law = male_only_law
				}
			}
		}
		always = yes
	}
	trigger_else = {
		always = no
	}
}

}

can_increase_government_authority = {
	realm_law_authority_level_max = no
	realm_law_use_government_authority = yes
	trigger_if = {
		limit = {
			has_realm_law_flag = has_requirement_for_authority_level
		}
		trigger_if = {
			limit = {
				realm_law_use_church_authority = yes
			}
			switch = {
				trigger = has_realm_law_flag
				authority_level_2 = { can_pass_law_church_authority_3_trigger = yes }
				authority_level_1 = { can_pass_law_church_authority_2_trigger = yes }
				authority_level_0 = { can_pass_law_church_authority_1_trigger = yes }
				fallback = { always = no }
			}
		}
		trigger_else = {
			switch = {
				trigger = has_realm_law_flag
				authority_level_2 = { can_pass_law_ca3_trigger = yes }
				authority_level_1 = { can_pass_law_ca2_trigger = yes }
				authority_level_0 = { can_pass_law_ca1_trigger = yes }
				fallback = { always = no }
			}
		}
	}
}

should_synchronize_liege_law = {
	top_liege != this
	OR = {
		trigger_if = {
			limit = { realm_law_use_japanese_bureaucracy = yes }
			liege = { realm_law_use_japanese_bureaucracy = yes }
		}
		trigger_else = {
			government_has_mechanic = administrative
			top_liege = {
				government_has_mechanic = administrative
			}
		}
	}
	use_same_authority_with_liege = yes
}

use_same_authority_with_liege = {
	OR = {
		government_type = liege.government_type
		trigger_if = {
			limit = { realm_law_use_crown_authority = yes }
			liege = { realm_law_use_crown_authority = yes }
		}
		trigger_else_if = {
			limit = { realm_law_use_tribal_authority = yes }
			liege = { realm_law_use_tribal_authority = yes }
		}
		trigger_else_if = {
			limit = { realm_law_use_imperial_bureaucracy = yes }
			liege = { realm_law_use_imperial_bureaucracy = yes }
		}
		trigger_else_if = {
			limit = { realm_law_use_nomadic_authority = yes }
			liege = { realm_law_use_nomadic_authority = yes }
		}
		trigger_else_if = {
			limit = { realm_law_use_celestial_bureaucracy = yes }
			liege = { realm_law_use_celestial_bureaucracy = yes }
		}
		trigger_else_if = {
			limit = { realm_law_use_meritocratic_bureaucracy = yes }
			liege = { realm_law_use_meritocratic_bureaucracy = yes }
		}
		trigger_else_if = {
			limit = { realm_law_use_japanese_bureaucracy = yes }
			liege = { realm_law_use_japanese_bureaucracy = yes }
		}
		trigger_else_if = {
			limit = { realm_law_use_church_authority = yes }
			liege = { realm_law_use_church_authority = yes }
		}
		trigger_else = {
			always = no
		}
	}
}

can_have_single_heir_succession_law_trigger = {
	# Warcraft
	can_have_dark_succession_law_trigger = no
	trigger_if = { # Celestial/Meritocratic/Soryo
		limit = {
			OR = {
				government_allows = merit
				government_has_flag = government_is_japan_feudal
			}
		}
		custom_tooltip = { # You are independent, have the innovation or are Ceremonial Monarch
			text = can_have_single_heir_succession_law_trigger_merit_or_japan_tt
			OR = {
				is_independent_ruler = yes
				culture = {
					has_innovation = innovation_primogeniture
				}
				tgp_has_ceremonial_liege_title_trigger = yes
			}
		}
	}
	trigger_else_if = {
		limit = { tgp_realm_has_ceremonial_liege_trigger = yes }
		tgp_has_ceremonial_liege_title_trigger = yes
	}
	trigger_else = {
		culture = {
			has_innovation = innovation_primogeniture
		}
	}
	# Non-tribals only
	trigger_if = { # We split the tooltip into two, to make sure we don't show admin unless you actually have it
		limit = {
			has_ep3_dlc_trigger = yes
		}
		custom_tooltip = {
			text = must_be_feudal_clan_or_admin
			OR = {
				government_has_flag = government_is_feudal
				government_has_flag = government_is_clan
				government_has_mechanic = administrative
				government_has_flag = government_is_japan_feudal
				tgp_has_ceremonial_liege_title_trigger = yes
			}
		}
	}
	trigger_else = {
		custom_description = {
			text = succession_laws_must_have_organised_government_tooltip
			OR = {
				government_has_flag = government_is_feudal
				government_has_flag = government_is_clan
				government_allows = merit
				government_has_flag = government_is_japan_feudal
				tgp_has_ceremonial_liege_title_trigger = yes
				# Warcraft
				has_monarchy_government_trigger = yes
			}
		}
	}
	# Admin rules
	trigger_if = {
		limit = { government_has_mechanic = administrative }
		trigger_if = {
			limit = { tgp_realm_has_ceremonial_liege_trigger = yes }
			tgp_has_ceremonial_liege_title_trigger = yes
		}
		trigger_else = { is_independent_ruler = yes }
	}
}

# Scope's $AUTHORITY$ authority law level is at least $CHARACTER$'s
has_higher_or_equal_authority_trigger = {
	OR = {
		has_realm_law = $AUTHORITY$_authority_3
		AND = {
			has_realm_law = $AUTHORITY$_authority_2
			$CHARACTER$ = {
				OR = {
					has_realm_law = $AUTHORITY$_authority_0
					has_realm_law = $AUTHORITY$_authority_1
					has_realm_law = $AUTHORITY$_authority_2
				}
			}
		}
		AND = {
			has_realm_law = $AUTHORITY$_authority_1
			$CHARACTER$ = {
				OR = {
					has_realm_law = $AUTHORITY$_authority_0
					has_realm_law = $AUTHORITY$_authority_1
				}
			}
		}
		AND = {
			has_realm_law = $AUTHORITY$_authority_0
			$CHARACTER$ = {
				has_realm_law = $AUTHORITY$_authority_0
			}
		}
	}
}
