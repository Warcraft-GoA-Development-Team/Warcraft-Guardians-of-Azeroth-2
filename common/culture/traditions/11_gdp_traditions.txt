	cost = {
		prestige = {
			add = {
				value = tradition_base_cost

				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}

			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 100
		if = {
			limit = {
				OR = {

tradition_saharan_traders = {

				}
			}
			NOR = {
				has_cultural_tradition = tradition_caravaneers
				has_cultural_tradition = tradition_silk_road_traders

		}

	can_pick_for_hybridization = {
		custom_tooltip = {
			text = only_one_caravan_trade_tradition

	character_modifier = {
		trade_category_precious_metals_xp_gain_mult = 0.2
		trade_category_salt_xp_gain_mult = 0.2

		trait_track_caravaneer_xp_gain_mult = 0.2

		trade_good_gold_sell_price_mult = 0.0075
	}

		trade_good_gold_production_mult = 0.15
		trade_good_salt_production_mult = 0.1
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			# Ethos requirement.
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_stoic = { is_in_list = traits }
						culture_pillar:ethos_spiritual = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_stoic_or_spiritual_desc
				}
			}
			# Oasis requirement.
			if = {
				limit = {
					NOT = {
						any_culture_county = {
							count >= 2
							county_has_province_with_terrain = {
								terrain = oasis
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = oasis_count_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_saharan_traders }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}

			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 100
		if = {
			limit = {
				NOT = {
					scope:character = {
						any_sub_realm_county = {

					}

		}

			limit = {
				any_culture_county = {
					count >= 3

						terrain = oasis
					}
				}
			}
			multiply = 4
		}
	}
}
