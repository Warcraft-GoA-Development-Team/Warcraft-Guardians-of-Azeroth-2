request_elevation_interaction = {
	category = interaction_category_religion
	interface_priority = 10
	desc = request_elevation_interaction_desc
	notification_text = request_elevation_interaction_notification
	icon = religious
	target_type = title
	target_filter = clerical_region
	show_all_targets = no

	cost = {
		piety = {
			value = pam_request_elevation_title_cost_value
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	is_available = {
		has_pam_dlc_trigger = yes
		trigger_if = {
			limit = { is_ai = yes }
			is_clergy = yes
			primary_title ?= { tier < tier_duchy }
		}
	}

	is_shown = {
		has_pam_dlc_trigger = yes
		scope:puppet_or_actor ?= {
			this != scope:recipient
			religious_head_or_challenger ?= scope:recipient
			OR = {
				is_clergy = yes
				has_council_position = councillor_court_chaplain
			}
			trigger_if = {
				limit = {
					is_landed = yes
				}
				highest_held_title_tier = tier_county
			}
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				piety >= pam_request_elevation_title_cost_value
			}
		}
		scope:puppet_or_actor = {
			is_available_quick = {
				imprisoned = no
				incapable = no
			}
		}
		pam_not_antipope_when_recipient_is_hof_tt = { CHARACTER = scope:puppet_or_actor }
	}

	can_be_picked_title = {
		scope:target.title_capital_county.holder.faith = scope:puppet_or_actor.faith
		scope:target.title_capital_county.holder.religious_head_or_challenger = scope:puppet_or_actor.religious_head_or_challenger
		# Only the clerical region that contains the petitioner's capital (or their liege's, if unlanded)
		trigger_if = {
			limit = { exists = scope:puppet_or_actor.capital_county }
			scope:puppet_or_actor.capital_county.clerical_region_title ?= scope:target
		}
		trigger_else = {
			scope:puppet_or_actor.liege.capital_county.clerical_region_title ?= scope:target
		}
		custom_tooltip = {
			text = request_clerical_region_title_cannot_request_hof_see_tt
			NOT = { scope:target.holder ?= scope:recipient }
		}
		custom_tooltip = {
			text = request_elevation_cannot_unseat_player_tt
			trigger_if = {
				limit = { exists = scope:target.holder }
				scope:target.holder = { is_ai = yes }
			}
		}
		NOT = { scope:target.holder ?= scope:puppet_or_actor }
	}

	# can_be_picked_title is picker-only
	has_valid_target_showing_failures_only = {
		scope:target.title_capital_county.holder.faith = scope:puppet_or_actor.faith
		scope:target.title_capital_county.holder.religious_head_or_challenger = scope:puppet_or_actor.religious_head_or_challenger
		custom_tooltip = {
			text = request_clerical_region_title_capital_region_tt
			trigger_if = {
				limit = { exists = scope:puppet_or_actor.capital_county }
				scope:puppet_or_actor.capital_county.clerical_region_title ?= scope:target
			}
			trigger_else = {
				scope:puppet_or_actor.liege.capital_county.clerical_region_title ?= scope:target
			}
		}
		custom_tooltip = {
			text = request_clerical_region_title_cannot_request_hof_see_tt
			NOT = { scope:target.holder ?= scope:recipient }
		}
		custom_tooltip = {
			text = request_elevation_cannot_unseat_player_tt
			trigger_if = {
				limit = { exists = scope:target.holder }
				scope:target.holder = { is_ai = yes }
			}
		}
		NOT = { scope:target.holder ?= scope:puppet_or_actor }
	}

	cooldown_against_recipient = { years = 10 } # Spam prevention

	is_highlighted = {
		scope:puppet_or_actor.piety_level > scope:recipient.piety_level
	}

	on_accept = {
		# A vacant see has nobody to unseat, so the displacement consequences only apply when it is held
		if = {
			limit = { exists = scope:target.holder }
			scope:target.holder = {
				save_scope_as = displaced_incumbent
				# Make sure your kicked realm priest is pissed
				add_prestige = medium_prestige_loss
			}
			scope:puppet_or_actor = {
				reverse_add_opinion = {
					target = scope:displaced_incumbent
					modifier = fp3_resentful_opinion
					opinion = -40
				}
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = {
					has_council_position = councillor_court_chaplain
				}
			}
			scope:puppet_or_actor = {
				save_scope_as = original_councillor
			}
			scope:puppet_or_actor.liege = {
				save_scope_as = original_councillor_liege
			}
		}
		scope:puppet_or_actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = request_elevation_interaction_toast_title
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
				add_prestige = major_prestige_gain
				scope:puppet_or_actor = {
					if = {
						limit = {
							piety > 0
							scope:is_puppet_action ?= yes
							pam_request_elevation_title_cost_value > 0
						}
						add_piety = {
							value = {
								add = pam_request_elevation_title_cost_value
								max = piety
								multiply = -1
							}
						}
					}
				}
			}
			# Every clerical region the incumbent shepherds transfers
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
			}
			if = {
				limit = { exists = scope:displaced_incumbent }
				scope:displaced_incumbent = {
					every_held_title = {
						limit = { has_clerical_region = yes }
						change_title_holder_include_vassals = {
							holder = scope:puppet_or_actor
							change = scope:change
						}
					}
				}
			}
			else = {
				scope:target = {
					change_title_holder_include_vassals = {
						holder = scope:puppet_or_actor
						change = scope:change
					}
				}
			}
			resolve_title_and_vassal_change = scope:change
			# situation:the_christian_church = { # Warcraft
			# 	if = {
			# 		limit = {
			# 			situation_top_has_catalyst = catalyst_clergy_elevated_to_archbishopric
			# 		}
			# 		trigger_situation_catalyst = catalyst_clergy_elevated_to_archbishopric
			# 	}
			# }
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = request_elevation_interaction_toast_title_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
	}

	on_decline = {
		scope:puppet_or_actor = {
			if = {
				limit = { is_ai = yes }
				add_opinion = {
					target = scope:recipient
					modifier = annoyed_opinion
					opinion = -25
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = head_of_faith
	}

	# is_available caps AI petitioners below duchy tier, so the higher tiers are unreachable by design.
	ai_frequency_by_tier = {
		barony = 0
		county = 60
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	ai_accept = {
		# The Church does not give away a see lightly
		base = -75

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:puppet_or_actor
			RECIPIENT = scope:recipient
		}

		clerical_petition_to_authority_acceptance_modifier = yes

		clerical_petition_displace_incumbent_acceptance_modifier = {
			INCUMBENT = scope:target.holder
			HELD_TITLE = scope:target
			DISPLACEMENT_DESC = TITLE_ALREADY_HELD
			CULTURE_DESC = SAME_CULTURE_AS_SEAT_REASON
		}

		modifier = { # Taking a head of rite's own see is all but unthinkable
			add = -200
			exists = scope:target.holder
			NOT = { scope:target.holder ?= scope:recipient }
			scope:target.holder ?= { rite.head_of_rite ?= this }
			desc = REQUEST_HEAD_OF_RITE_SEE_REASON
		}

		modifier = { # Less pious than the cleric they would replace
			add = -40
			scope:target.holder ?= { piety_level > scope:puppet_or_actor.piety_level }
			desc = LOWER_PIETY_LEVEL_THAN_HOLDER_REASON
		}

		modifier = { # Not yet a figure of standing in the Church
			add = -40
			scope:puppet_or_actor = { piety_level < 4 }
			desc = BELOW_PIETY_LEVEL_4_REASON
		}

		modifier = { # The sitting cleric has given no cause to be removed
			add = -40
			scope:target.holder ?= {
				is_incapable = no
				is_imprisoned = no
				NOT = { has_trait = excommunicated }
			}
			desc = INCUMBENT_IS_BLAMELESS_REASON
		}

		modifier = { # An unlanded chaplain asking to be handed a whole see
			add = -25
			scope:puppet_or_actor = { is_landed = no }
			desc = PETITIONER_IS_UNLANDED_REASON
		}
	}

	ai_will_do = {
		# base = 25 so the AI still petitions when the opinion drivers cannot fire
		base = 25

		# I dislike my superior
		opinion_modifier = { 
			trigger = { exists = scope:puppet_or_actor.superior }
			opinion_target = scope:puppet_or_actor.superior
			who = scope:puppet_or_actor
			multiplier = -1
		}
		# My superior dislikes me
		opinion_modifier = { 
			trigger = { exists = scope:puppet_or_actor.superior }
			opinion_target = scope:puppet_or_actor
			who = scope:puppet_or_actor.superior
			multiplier = -1
		}
	}
}

# Petition the HoF to create a new ecclesiastical title: secular kings pick the holder (realm priest default), clergy become the holder themselves
request_ecclesiastical_title_creation_interaction = {
	category = interaction_category_religion
	interface_priority = 10
	desc = request_ecclesiastical_title_creation_interaction_desc
	notification_text = request_ecclesiastical_title_creation_interaction_notification
	icon = request_clerical_title_creation # TODO_PAM_ALI TIT-73245
	use_diplomatic_range = no

	# The picker chooses the seat, not the extent, and create_clerical_region fills the de jure kingdom
	target_type = title
	target_filter = domain_titles
	target_source = puppet_or_actor
	show_all_targets = no

	cost = {
		piety = {
			# Free Christian de jure county count of the anchor's de jure kingdom * piety
			add = {
				value = pam_new_clerical_region_cost_value
				if = { # AI pays less
					limit = { scope:actor = { is_ai = yes } }
					multiply = 0.75
				}
			}
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	# Sets the new-holder slot (secondary_recipient): secular → realm priest (picker can override); clergy → themselves. Title is resolved in on_accept (redirect can't save titles)
	redirect = {
		if = {
			limit = {
				scope:puppet_or_actor ?= { has_government = ecclesiastical_government }
			}
			scope:puppet_or_actor = {
				save_scope_as = secondary_recipient
			}
		}
		else = {
			scope:puppet_or_actor = {
				# A character may hold only one clerical region, so an existing archbishop cannot default here
				if = {
					limit = {
						exists = cp:councillor_court_chaplain
						cp:councillor_court_chaplain = {
							faith = scope:puppet_or_actor.faith
							NOT = { any_held_title = { has_clerical_region = yes } }
						}
					}
					cp:councillor_court_chaplain = {
						save_scope_as = secondary_recipient
					}
				}
				else = {
					ordered_courtier = {
						limit = {
							NOT = { any_held_title = { has_clerical_region = yes } }
							can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
						}
						order_by = learning
						save_scope_as = secondary_recipient
					}
				}
			}
		}
	}

	# Holder picker (secular actors only) — realm priest is the pre-selected default via redirect
	populate_recipient_list = {
		scope:puppet_or_actor = {
			cp:councillor_court_chaplain ?= {
				if = {
					limit = {
						faith = scope:puppet_or_actor.faith
						NOT = { any_held_title = { has_clerical_region = yes } }
					}
					add_to_list = characters
				}
			}
			every_courtier = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
				}
				add_to_list = characters
			}
			every_vassal = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
					has_government = ecclesiastical_government
				}
				add_to_list = characters
			}
		}
	}

	# can_be_picked_title handles title eligibility, so we keep is_shown fast for the AI
	is_shown = {
		scope:puppet_or_actor.faith ?= {
			has_doctrine_parameter = has_clerical_regions
			OR = {
				NOT = { has_doctrine_parameter = temporal_head_of_faith }
				NOT = { exists = religious_head }
			}
		}
		OR = {
			scope:puppet_or_actor ?= {
				highest_held_title_tier >= tier_duchy
				is_available_quick = {
					adult = yes
					alive = yes
					imprisoned = no
					incapable = no
				}
				NOT = { pam_government_is_religious_playable_trigger = yes }
				any_held_title = {
					tier >= tier_duchy
					is_landless_type_title = no
					is_head_of_faith = no
					title_capital_county ?= {
						NOT = { exists = clerical_region_title }
					}
				}
			}
			scope:puppet_or_actor ?= {
				has_government = ecclesiastical_government
				is_landed = yes
				liege.primary_title ?= {
					tier = tier_kingdom
					is_landless_type_title = no
				}
				is_available_quick = {
					adult = yes
					alive = yes
					imprisoned = no
					incapable = no
				}
				primary_title ?= { tier < tier_duchy }
				NOT = { any_held_title = { has_clerical_region = yes } }
				any_held_title = {
					tier >= tier_county
					is_landless_type_title = no
					is_head_of_faith = no
					title_capital_county ?= {
						NOT = { exists = clerical_region_title }
					}
				}
			}
		}
		has_pam_dlc_trigger = yes
		scope:recipient != scope:puppet_or_actor
		OR = {
			pam_recipient_is_clerical_title_authority_trigger = yes # HoF, or the Head of Rites when the faith has no HoF
			# A realm under an antipope asks them instead
			scope:puppet_or_actor.religious_head_or_challenger ?= scope:recipient
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				piety >= pam_new_clerical_region_cost_value
			}
		}
		scope:puppet_or_actor = {
			is_available_quick = {
				imprisoned = no
				incapable = no
			}
			pam_antipope_clerical_request_trigger = { RECIPIENT = scope:recipient }
			NOT = {
				has_character_modifier = promised_pilgrimage_to_clergy_modifier
			}
		}
		custom_tooltip = {
			text = pam_request_ecclesiastical_title_creation_needs_free_cleric_tt
			exists = scope:secondary_recipient
			scope:secondary_recipient = {
				faith = scope:puppet_or_actor.faith
				NOT = { any_held_title = { has_clerical_region = yes } }
			}
		}
		trigger_if = {
			limit = { scope:actor = scope:puppet_or_actor }
			custom_tooltip = {
				text = pam_request_ecclesiastical_title_creation_no_eligible_anchor_tt
				scope:puppet_or_actor = {
					any_held_title = {
						pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
					}
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = pam_request_ecclesiastical_title_creation_no_eligible_anchor_tt_puppeteer
				scope:puppet_or_actor = {
					any_held_title = {
						pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
					}
				}
			}
		}
	}

	has_valid_target_showing_failures_only = {
		custom_tooltip = {
			text = pam_new_clerical_region_invalid_anchor_tt
			scope:target ?= {
				pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
			}
		}
	}

	can_be_picked_title = {
		scope:target = {
			pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
		}
	}

	is_highlighted = {
		scope:puppet_or_actor.piety_level > scope:recipient.piety_level
	}

	# Creates a clerical region on the capital of the picked title; appoints the new holder as court chaplain of that title's holder
	on_accept = {
		scope:secondary_recipient = {
			save_scope_as = new_ecclesiastical_holder
		}
		# The picked title seats the region; create_clerical_region fills it from that seat's de jure kingdom
		scope:target = {
			save_scope_as = target_kingdom
			holder = { save_scope_as = kingdom_holder }
		}
		# We bring the new holder into court as chaplain unless they already hold the seat
		if = {
			limit = { scope:kingdom_holder != scope:new_ecclesiastical_holder }
			scope:kingdom_holder = {
				if = {
					limit = {
						scope:new_ecclesiastical_holder = {
							NOT = { is_courtier_of = scope:kingdom_holder }
							is_ruler = no
						}
					}
					add_courtier = scope:new_ecclesiastical_holder
				}
				if = {
					limit = {
						scope:new_ecclesiastical_holder = {
							NOT = { is_councillor_of = scope:kingdom_holder }
						}
					}
					assign_councillor_type = {
						type = councillor_court_chaplain
						remove_existing_councillor = yes
						target = scope:new_ecclesiastical_holder
					}
				}
			}
		}
		# Warcraft
		# hidden_effect = {
		# 	if = {
		# 		limit = {
		# 			has_pam_dlc_trigger = yes
		# 			scope:puppet_or_actor = {
		# 				religion = religion:christianity_religion
		# 				any_character_situation = {
		# 					situation_type = the_christian_church
		# 					situation_top_has_catalyst = catalyst_the_christian_church_clerical_region_created
		# 				}
		# 			}
		# 		}
		# 		situation:the_christian_church = {
		# 			trigger_situation_catalyst = catalyst_the_christian_church_clerical_region_created
		# 		}
		# 	}
		# }
		# Create the clerical region on the kingdom's capital and give the new title to the new holder
		custom_tooltip = {
			text = request_create_new_ecclesiastical_title_effect_tt
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:target_kingdom = {
				title_capital_county = {
					save_scope_as = clerical_region_anchor_county
					random_county_province = {
						limit = { is_county_capital = yes }
						save_scope_as = region_capital
						create_clerical_region = {
							domicile_type = ecclesiastical_domicile
							name = CLERICAL_REGION_PROVINCE_TITLE_NAME
							save_scope_as = new_title
						}
						scope:new_title ?= { # ?= avoids a tooltip-time error since the title does not exist yet when the description is built
							set_coa = scope:target_kingdom
							set_color_from_title = scope:target_kingdom
							set_capital_county = scope:clerical_region_anchor_county
							set_landless_title = yes
							set_no_automatic_claims = yes
							set_always_follows_primary_heir = yes
							change_title_holder = {
								holder = scope:new_ecclesiastical_holder
								change = scope:change
							}
							resolve_title_and_vassal_change = scope:change
						}
					}
				}
			}
		}
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = request_ecclesiastical_title_creation_accepted_tt
				left_icon = scope:new_title
				right_icon = scope:new_ecclesiastical_holder
				# New holder is grateful — skipped when the petitioner appointed themselves (clergy path), to avoid a self-opinion bonus
				if = {
					limit = {
						scope:new_ecclesiastical_holder != scope:puppet_or_actor
					}
					reverse_add_opinion = {
						target = scope:new_ecclesiastical_holder
						modifier = grateful_opinion
						opinion = 50
					}
				}
			}
			if = {
				limit = {
					NOT = {
						has_government = ecclesiastical_government
					}
				}
				add_hook = {
					type = loyalty_hook
					target = scope:new_ecclesiastical_holder
				}
			}
		}
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = request_ecclesiastical_title_creation_accepted_tt
					left_icon = scope:new_title
					right_icon = scope:new_ecclesiastical_holder
				}
			}
		}
		if = {
			limit = { scope:recipient != scope:actor }
			scope:recipient = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = request_ecclesiastical_title_creation_accepted_tt
					left_icon = scope:new_title
					right_icon = scope:new_ecclesiastical_holder
				}

					limit = {
						is_ruler = yes
						NOT = { government_has_flag = government_is_ecclesiastical }
					}
					change_government = ecclesiastical_government
				}
			}
			# When a puppeteer initiates, the puppet also drains their own piety up to their current maximum
			scope:puppet_or_actor = {
				if = {
					limit = {
						piety > 0
						scope:is_puppet_action ?= yes
						pam_request_ecclesiastical_title_split_cost_value > 0
					}
					add_piety = {
						value = {
							add = pam_request_ecclesiastical_title_split_cost_value
							max = piety
							multiply = -1
						}
					}
				}
			}
		}
		# Shared — the superior who lost territory resents the new holder (and the petitioner, if they are not the same person)
		scope:clerical_region_holder = {
			add_opinion = {
				target = scope:new_ecclesiastical_holder
				modifier = fp3_resentful_opinion
				opinion = -50
			}
			if = {
				limit = {
					scope:puppet_or_actor != scope:new_ecclesiastical_holder
				}
				add_opinion = {
					target = scope:puppet_or_actor
					modifier = fp3_resentful_opinion
					opinion = -50
				}
			}
		}
		# If recipient is a player - let's add a global AI cooldown so you don't get spammed with requests
		scope:recipient = {
			if = {
				limit = { is_ai = no }
				set_variable = {
					name = pam_request_ecclesiastical_title_split_ai_cooldown
					years = 3
				}
			}
		}
	}

	on_decline = {
		scope:puppet_or_actor = {
			if = {
				limit = { is_ai = yes }
				add_opinion = {
					target = scope:recipient
					modifier = annoyed_opinion
					opinion = -10
				}
			}
		}
		# If recipient is a player - let's add a global AI cooldown so you don't get spammed with requests
		scope:recipient = {
			if = {
				limit = { is_ai = no }
				set_variable = {
					name = pam_request_ecclesiastical_title_split_ai_cooldown
					years = 3
				}
			}
		}
	}

		clerical_petition_displace_incumbent_acceptance_modifier = {
			INCUMBENT = scope:secondary_recipient

		opinion_modifier = { 
			opinion_target = scope:secondary_recipient
			who = scope:puppet_or_actor
			multiplier = -1
		}
		# The affected superior/region holder dislikes me
		opinion_modifier = { 
			opinion_target = scope:puppet_or_actor
			who = scope:secondary_recipient
			multiplier = -1
		}
		modifier = {
			has_strong_hook = scope:secondary_actor
			add = 50
		}
		modifier = {
			has_hook = scope:secondary_actor
			add = 25
		}
		# Clergy actors take the new title themselves, so there is no third party to weigh opinion against
		opinion_modifier = { 
			trigger = { scope:secondary_actor != scope:actor }
			opinion_target = scope:secondary_actor
			who = scope:actor
			multiplier = 0.1
		}
		opinion_modifier = { 
			trigger = { scope:secondary_actor != scope:actor }
			opinion_target = scope:actor
			who = scope:secondary_actor
			multiplier = 0.1
		}
		modifier = {
			scope:secondary_actor = {
				is_clergy = yes
			}
			add = 25
		}
		modifier = {
			add = {
				value = scope:secondary_actor.learning
			}
		}
		# Ambitious
		modifier = {
			trigger = {
				scope:puppet_or_actor = { has_trait = ambitious }
			}
			add = 25
		}
		# Greedy
		modifier = {
			trigger = {
				scope:puppet_or_actor = { has_trait = greedy }
			}
			add = 10
		}
		# Different Rite
		modifier = {
			add = 20
			scope:puppet_or_actor.rite != scope:secondary_actor.rite
			desc = I_HAVE_A_DIFFERENT_RITE
		}
