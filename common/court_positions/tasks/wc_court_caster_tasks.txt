task_cast_spells = {
    court_position_types = { court_caster_court_position }

    cost = {
		round = no
		gold = {
			value = 0.5
			if = {
				limit = {
					scope:liege = {
						has_treasury = no
					}
				}
				add = {
					value = monthly_court_position_task_cost
					desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
					format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
				}
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					scope:liege = {
						has_treasury = yes
					}
				}
				add = {
					value = monthly_court_position_task_cost
					desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
					format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
				}
			}
		}
	}

    on_start = {
        scope:liege = { set_variable = court_caster_task_cast_spells }
	}

    on_end = {
        scope:liege = { remove_variable = court_caster_task_cast_spells }
    }

    ai_will_do = {
		value = {
			add = 150
            add = ai_rationality
		}
        if = {
			limit = {
				has_treasury = no
				monthly_character_income < monthly_court_position_task_cost_double
			}
			add = -1000
		}
		else_if = {
			limit = {
				has_treasury = yes
				monthly_character_treasury_income < monthly_court_position_task_cost_double
			}
			add = -1000
		}
    }
}

task_manage_mana = {
    court_position_types = { court_caster_court_position }

    cost = {
		round = no
		gold = {
			value = 0
			if = {
				limit = {
					scope:liege = {
						has_treasury = no
					}
				}
				add = {
					value = monthly_court_position_task_cost
					desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
					format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
				}
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					scope:liege = {
						has_treasury = yes
					}
				}
				add = {
					value = monthly_court_position_task_cost
					desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
					format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
				}
			}
		}
	}

    on_start = {
        scope:liege = { set_variable = court_caster_task_manage_mana }
	}

    on_end = {
        scope:liege = { remove_variable = court_caster_task_manage_mana }
    }
    

    ai_will_do = {
		value = {
			add = 25
		}
        if = {
			limit = {
				has_treasury = no
				monthly_character_income < monthly_court_position_task_cost_double
			}
			add = -1000
		}
		else_if = {
			limit = {
				has_treasury = yes
				monthly_character_treasury_income < monthly_court_position_task_cost_double
			}
			add = -1000
		}
        if = {
		    limit = { 
                exists = var:wc_current_mana_variable
				var:wc_current_mana_variable <= wc_twenty_percent_of_max_mana_value
			}
			add = 50
		}
        if = {
		    limit = { 
                exists = var:wc_current_mana_variable
				var:wc_current_mana_variable <= wc_fourty_percent_of_max_mana_value
            }
			add = 25
		}
        if = {
            limit = {
                NOT = { exists = var:wc_current_mana_variable }
            }
            add = -1000
        }
    }
}
