# Voluntary entry point: stage the canonical defeat before Sylvanas attacks.
wc_balnazzar_stage_death_decision = {
	# AI must clear the entry before Alexandros's scripted death (605.9.9); the
	# canonical chain is impossible to observe at a 12-month recheck.
	ai_check_interval = 1
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = major

	desc = wc_balnazzar_stage_death_decision_desc

	is_shown = {
		this = character:430400 # Balnazzar
		is_alive = yes
		is_landed = yes
		has_title = title:k_lordaeron
		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = { story_owner = root }
		NOT = { has_character_flag = wc_balnazzar_staged_death }
		NOT = { has_character_flag = wc_balnazzar_disguise_active }
		NOT = { has_character_flag = balnazzar_became_dathrohan }
		exists = character:430100 # Varimathras
		character:430100 = { is_alive = yes }
		exists = character:42076 # Sylvanas Windrunner
		character:42076 = { is_alive = yes }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
		is_at_war = no
	}

	is_valid = {
		custom_tooltip = {
			text = wc_balnazzar_stage_death_decision_requirement_witnesses
			wc_balnazzar_can_stage_death_trigger = yes
		}
	}

	effect = {
		custom_tooltip = wc_balnazzar_stage_death_decision_effect_tooltip
		hidden_effect = {
			trigger_event = wc_balnazzar_story.0002
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

# Adventurer decision after losing Lordaeron; its controller chooses when to
# take Dathrohan's place.
balnazzar_become_dathrohan_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	# is_landless_ruler = yes below means only an adventurer ever sees this - group it
	# with the other adventurer-only major decisions instead of the generic major list.
	decision_group_type = adventurer

	# Every tier gets a short recheck: 0 means "never considered"
	# (_decisions.info), and a landless ruler's effective tier is not proven to
	# be duchy, so leaving barony/county at 0 could silently skip the possession.
	# The decision is is_shown only for the landless story owner regardless.
	ai_check_interval_by_tier = {
		barony = 1
		county = 1
		duchy = 1
		kingdom = 1
		empire = 1
		hegemony = 1
	}

	desc = balnazzar_become_dathrohan_decision_desc

	is_shown = {
		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = { story_owner = root }
		NOT = { has_character_flag = wc_balnazzar_disguise_active }
		NOT = { has_character_flag = balnazzar_became_dathrohan }
		is_landless_ruler = yes
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	is_valid = {
		custom_tooltip = {
			text = balnazzar_become_dathrohan_decision_requirement_available
			wc_balnazzar_can_take_dathrohan_trigger = yes
		}
	}

	effect = {
		custom_tooltip = balnazzar_become_dathrohan_decision_effect_tooltip
		hidden_effect = { balnazzar_become_dathrohan_effect = yes }
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

balnazzar_move_against_alexandros_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	# Major decision - starts assassination scheme, point-of-no-return
	decision_group_type = major

	# Continue checking after promotion, including a county-tier Dathrohan.
	# Two months so the AI can open Renault's murder before Alexandros dies.
	ai_check_interval_by_tier = {
		barony = 2
		county = 2
		duchy = 2
		kingdom = 2
		empire = 2
		hegemony = 2
	}

	desc = balnazzar_move_against_alexandros_decision_desc

	is_shown = {
		has_character_flag = wc_balnazzar_disguise_active
		this = character:60021
		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = { story_owner = root }
		has_character_flag = balnazzar_kelthuzad_pact_sealed
		NOT = { has_character_flag = wc_balnazzar_renault_solicited }
		character:60019 = { is_alive = yes is_adult = yes }
		exists = character:60017 # Alexandros Mograine
		character:60017 = { is_alive = yes }
		# Hide only once Balnazzar's murder scheme against Alexandros is already running.
		# An unrelated murder scheme must not block this story decision.
		NOT = {
			any_scheme = {
				type = murder
				scheme_target_character = character:60017
				count > 0
			}
		}
	}

	is_valid = {
		custom_tooltip = {
			text = balnazzar_move_against_alexandros_decision_requirement_project
			wc_balnazzar_murder_project_valid_trigger = yes
			NOT = { has_variable = wc_balnazzar_renault_request }
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		custom_tooltip = balnazzar_move_against_alexandros_decision_effect_tooltip
		if = {
			limit = { wc_balnazzar_murder_project_valid_trigger = yes NOT = { has_variable = wc_balnazzar_renault_request } }
			trigger_event = { id = wc_balnazzar_story.1005 }
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

# Voluntary upkeep for an active disguise under suspicion.
wc_balnazzar_maintain_mask_decision = {
	ai_check_interval = 6
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = decisions

	desc = wc_balnazzar_maintain_mask_decision_desc

	cooldown = { years = 1 }

	is_shown = {
		has_character_flag = wc_balnazzar_disguise_active
		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = {
			story_owner = root
			exists = var:wc_balnazzar_suspicion
			var:wc_balnazzar_suspicion > 0
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		add_stress = medium_stress_gain
		hidden_effect = {
			if = {
				limit = { wc_balnazzar_free_witness_count > 0 }
				wc_balnazzar_change_suspicion_effect = { CHANGE = -2 }
			}
			else = { wc_balnazzar_change_suspicion_effect = { CHANGE = -10 } }
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 25
		modifier = {
			add = 25
			global_var:balnazzar_dreadlord_story = { var:wc_balnazzar_suspicion >= 30 }
		}
		modifier = {
			add = 25
			global_var:balnazzar_dreadlord_story = { var:wc_balnazzar_suspicion >= 60 }
		}
		modifier = {
			add = 15
			has_trait = paranoid
		}
		modifier = {
			factor = 0.5
			stress_level >= 2
		}
	}
}
