summon_burning_legion_decision = { # By Ikarimonogatari
	picture = { reference = "gfx/interface/illustrations/decisions/wc_summon_burning_legion.dds" }
	decision_group_type = major

	ai_check_interval = 1

	desc = summon_burning_legion_decision_desc

	is_shown = {
		AND = { # For arthas
			scourge_quelthalas_frostmourne_check_trigger = yes
			is_landed = yes
		}
		game_start_date < 605.6.6

		is_target_in_global_variable_list = { # Only make this available before kelthuzad is spawned
			name = unavailable_unique_decisions
			target = flag:resurrect_kelthuzad_decision
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:summon_burning_legion_decision
			}
		}
	}

	is_valid = {
		AND = { # For arthas
			scourge_quelthalas_frostmourne_check_trigger = yes
			is_landed = yes
		}
		OR = {
			any_courtier = {
				has_trait = creature_lich
			}
			any_vassal = {
				has_trait = creature_lich
			}
		}
		NOT = { # Don't allow this if Nordrassil has already been destroyed
			title:b_nordrassil.title_province = { has_building = nordrassil_destroyed }
		}
	}

	cost = {}

	cooldown = { years = 2 }

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = {
			text = summon_burning_legion_decision.tt
			if = { # If the Lich King or a fellow vassal holds it, give the title to Arthas
				limit = {
					OR = {
						title:c_dalaran.holder = { has_title = title:h_scourge }
						title:c_dalaran.holder = { is_vassal_of = title:h_scourge.holder }
					}
				}
				trigger_event = burninglegion.0001
			}
			else_if = { # Invade Dalaran
				limit = { NOT = { title:c_dalaran.holder = root } }
				trigger_event = burninglegion.0002
			}
			else = { # Start searching for the Book of Medivh and Summon the Legion
				limit = { title:c_dalaran.holder = root }
				trigger_event = burninglegion.0003
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}

embrace_felblood_decision = { # By Ikarimonogatari
	picture = { reference = "gfx/interface/illustrations/decisions/embrace_felblood.dds" }
	decision_group_type = major

	ai_check_interval_by_tier = {

		county = 12
		duchy = 24
		kingdom = 36
		empire = 0 # Will only do it if you are weak and desperate
		hegemony = 0 # Will only do it if you are weak and desperate
	}

	desc = embrace_felblood_decision_desc

	is_shown = {
		has_trait = creature_orc
		OR = {
			has_faith = faith:burning_legion_religion
			has_faith = faith:orcish_fel
			has_faith = faith:black_harvest
		}
		NOT = { has_trait = creature_fel_orc }
	}

	is_valid = {
		has_trait = creature_orc
		custom_tooltip = {
			text = you_are_a_fel_religion_fangirl
			OR = {
				has_faith = faith:burning_legion_religion
				has_faith = faith:orcish_fel
				has_faith = faith:black_harvest
			}
		}
		OR = {
			custom_tooltip = {
				text = a_non_demon_hunter_demon_is_your_prisoner

					NOT = { has_trait = demon_hunter } # Demon Hunters are not REAL demons
					has_trait = being_demon
					is_imprisoned_by = root
				}
			}
			custom_tooltip = {
				text = a_non_demon_hunter_demon_is_your_liege
				is_independent_ruler = no
				OR = {
					liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
					top_liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
				}
			}
		}
	}

	cost = {
		prestige = 500
		piety = 1000
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
	}

	effect = {
		hidden_effect = {
			remove_trait = creature_orc
		}
		if = {
			limit = {
				is_independent_ruler = no
				OR = {
					liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
					top_liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
				}
			}
			custom_tooltip = {
				text = demon_liege_tt
				trigger_event = felblood.0001
			}
		}
		else = {
			limit = {
				any_prisoner = {
					has_trait = being_demon
					NOT = { has_trait = demon_hunter }
					is_imprisoned_by = root
				}
			}
			custom_tooltip = {
				text = demon_prisoner_tt
				trigger_event = felblood.0002
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
	}

	ai_will_do = {
		base = 25
		modifier = {
			add = 50
			top_liege = { has_trait = being_demon }
		}
		modifier = {
			add = 25
			liege = { has_trait = being_demon }
		}
		modifier = {
			add = 25
			any_prisoner = { has_trait = being_demon }
		}
		modifier = {
			add = 25
			has_trait = ambitious
		}
	}
}

become_legion_commander_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/wc_summon_burning_legion.dds" }
	decision_group_type = adventurer

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	desc = become_legion_commander_decision_desc

	is_shown = {
		has_ep3_dlc_trigger = yes
		government_has_flag = government_is_landless_adventurer
		can_become_legion_commander_trigger = yes
		NOR = {
			has_trait = legion_commander
			has_character_flag = had_laamp_LC_decision
			has_character_flag = used_to_be_legion_commander
		}
		trigger_if = {
			limit = {
				is_ai = yes
				OR = {
					exists = location.county.holder
					exists = location.duchy.holder
					exists = location.county.holder.top_liege
				}
			}
			NOT = {
				any_player = {
					OR = {
						this = location.county.holder
						this = location.duchy.holder
						this = location.county.holder.top_liege
					}
				}
			}
		}
	}

	is_valid = {
		has_religion = religion:doombringer # Only those who serve Sargeras
		calc_true_if = {
			amount >= 8
			martial >= 22
			intrigue >= 30
			piety_level >= 5
			OR = {
				AND = {
					has_trait = strategist
					has_trait = gallant
				}
				AND = {
					has_trait = schemer
					has_trait = torturer
				}
			}
			custom_tooltip = {
				text = become_conqueror_decision_war_participation
				any_memory = {
					count >= 20
					OR = {
						has_memory_type = offensive_war
						has_memory_type = defensive_war
						has_memory_type = joined_allys_war
					}
				}
			}
			custom_tooltip = {
				text = become_conqueror_decision_war_participation_kingdom
				any_memory = {
					count >= 5
					OR = {
						has_memory_type = offensive_war
						has_memory_type = defensive_war
						has_memory_type = joined_allys_war
					}
					any_memory_participant = {
						primary_title.tier >= tier_kingdom
					}
				}
			}
			calc_true_if = {
				amount >= 1
				custom_tooltip = {
					text = become_conqueror_decision_two_commander_traits
					any_character_trait = {
						count >= 2
						has_trait_category = commander
					}
				}
				martial >= 28
			}

			custom_tooltip = {
				text = become_legion_commander_decision_enough_battle_kills
				any_killed_character = {
					count >= 20
				}
			}
			custom_tooltip = {
				text = become_conqueror_decision_enough_war_contracts
				has_variable = war_contract_tracker
				var:war_contract_tracker >= 20
			}
			calc_true_if = {
				amount >= 1
				custom_tooltip = {
					text = become_conqueror_decision_enough_high_level_knights
					any_knight = {
						count >= 6
						prowess >= 18
					}
				}
				custom_tooltip = {
					text = become_conqueror_decision_enough_mediocre_knights
					any_knight = {
						count >= 12
						prowess >= 12
					}
				}
			}
		}
	}

	cost = {
		piety = 5000
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
		is_incapable_trigger = no
	}

	effect = {
		trigger_event = ep3_laamp_decision_event.1110
		show_as_tooltip = {
			if = {
				limit = {
					exists = location.kingdom.holder
				}
				add_pressed_claim = location.kingdom
				start_war = {
					cb = ep3_laamp_adventurer_kingdom_invasion
					target = location.kingdom.holder
					claimant = root
					target_title = location.kingdom
					target_title = location.kingdom.title_capital_county
				}
			}
			else_if = {
				limit = {
					exists = location.duchy.holder
				}
				add_pressed_claim = location.duchy
				start_war = {
					cb = claim_cb
					target = location.duchy.holder
					claimant = root
					target_title = location.duchy
					target_title = location.duchy.title_capital_county
				}
			}
			else_if = {
				limit = {
					exists = location.county.holder
				}
				add_pressed_claim = location.county
				start_war = {
					cb = claim_cb
					target = location.county.holder
					claimant = root
					target_title = location.county
				}
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_at_war = no
		is_incapable_trigger = no
	}

	ai_will_do = {
		base = 100
	}
}

destroy_the_world_tree = {
	picture = { reference = "gfx/interface/illustrations/decisions/wc_destroyed_nordrassil.dds" }
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 30
		duchy = 24
		kingdom = 18
		empire = 12
		hegemony = 6
	}

	desc = destroy_the_world_tree_desc

	is_shown = {
		has_trait = being_demon
		is_independent_ruler = yes
		NOR = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:destroy_the_world_tree
			}

			exists = title:h_burning_legion.holder # Don't allow this if burning legion has already spawned
		}
	}

	is_valid = {
		has_religion = religion:doombringer # Only those who serve Sargeras
		has_trait = being_demon
		is_independent_ruler = yes
		completely_controls = title:c_nordrassil

		title:b_nordrassil.title_province = { # Is there even a Nordrassil to be destroyed?
			OR = {
				has_building = nordrassil_sacrificed
				has_building = nordrassil_restored
				has_building = nordrassil
			}
		}
	}

	cost = {}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
		is_incapable_trigger = no
	}

	effect = {
		save_scope_as = destroyer
		if = {
			limit = {
				title:b_nordrassil.title_province = {
					OR = {
						has_building = nordrassil_sacrificed
						has_building = nordrassil_restored
					}
				}
			}
			custom_tooltip = {
				text = weaker_nordrassil_being_destroyed
				trigger_event = nordrassilsdestruction.0001
			}
		}
		else = {
			limit = { title:b_nordrassil.title_province = { has_building = nordrassil } }
			custom_tooltip = {
				text = prime_nordrassil_being_destroyed
				trigger_event = nordrassilsdestruction.0002
			}
		}
		scope:destroyer = {
			create_story = dynamic_story_burning_legion
			save_scope_as = story_owner
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:destroy_the_world_tree
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_at_war = no
		is_incapable_trigger = no
	}

	ai_will_do = {
		base = 100
	}
}
