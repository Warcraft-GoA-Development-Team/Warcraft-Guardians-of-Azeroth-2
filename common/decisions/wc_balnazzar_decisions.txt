# Manual fallback into wc_balnazzar_story.1001's own "go to ground" choice - that event only
# ever auto-fires from on_title_lost (c_lordaeron specifically changing hands) or on_vassal_change
# (losing independence some other way, e.g. conquered down into vassalage while a lesser vassal
# keeps holding c_lordaeron itself - see wc_balnazzar_on_actions.txt). A player whose story never
# trips either hook had no way into the rest of the chain at all - the whole story silently
# softlocked for them. Shares balnazzar_go_to_ground_effect
# (common/scripted_effects/wc_balnazzar_effects.txt) with .1001.a, so all paths do the exact same
# title-transfer + landless-adventurer conversion.
balnazzar_go_to_ground_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = decisions

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	desc = balnazzar_go_to_ground_decision_desc

	is_shown = {
		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = { story_owner = root }
		NOT = { has_trait = nathrezim_disguise }
		NOT = { has_character_flag = balnazzar_went_to_ground }
		is_landless_ruler = no
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		balnazzar_go_to_ground_effect = yes
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 10
	}
}

balnazzar_move_against_alexandros_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	# major (not decisions) - this starts the Mograine assassination scheme, the arc's
	# real point-of-no-return, matches its weight in the "Major Decisions" panel group.
	decision_group_type = major

	# Dathrohan is duchy-tier (Tyr's Hand) here - Order isn't seized until .1006.
	# 0 = skip check, matching become_legion_commander_decision's pattern.
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	desc = balnazzar_move_against_alexandros_decision_desc

	is_shown = {
		has_trait = nathrezim_disguise
		has_character_flag = balnazzar_kelthuzad_pact_sealed
		exists = character:60017 # Alexandros Mograine
		character:60017 = { is_alive = yes }
		# The scheme's own presence is the real signal it's already in motion - same check
		# balnazzar_press_renault_decision's is_shown already uses.
		NOT = {
			any_scheme = {
				type = wc_mograine_assassination
				count > 0
			}
		}
	}

	# ck3 bug: bare is_alive=yes on a directly-scoped character mis-renders in Requirements panel.
	# Wrap in custom_tooltip with an explicit line instead.
	is_valid = {
		custom_tooltip = {
			text = balnazzar_move_against_alexandros_decision_requirement_alive
			exists = character:60017
			character:60017 = { is_alive = yes }
			exists = character:60019 # Renault Mograine
			character:60019 = { is_alive = yes }
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		custom_tooltip = balnazzar_move_against_alexandros_decision_effect_tooltip
		trigger_event = {
			id = wc_balnazzar_story.1004
			days = 3
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 50
	}
}

# Grand Crusader's authority, spent on the realm rather than the man wearing it - vassals
# still holding the old faith are made to pay for the privilege of practicing it, or convert
# outright. Real gold/piety now, real vassal resentment (natural faction risk through opinion,
# not a scripted count-threshold) as the cost.
balnazzar_demand_purity_tribute_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = decisions

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	desc = balnazzar_demand_purity_tribute_decision_desc

	is_shown = {
		has_trait = nathrezim_disguise
		# Only makes sense once the Crusade itself exists (.1006.a) - collecting a
		# "purity tribute" for a creed he hasn't founded yet has nothing to demand it in the name of.
		has_character_flag = has_formed_the_scarlet_crusade
		any_vassal = {
			count > 0
			NOT = { faith = faith:scarletism }
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = balnazzar_demand_purity_tribute_decision.cooldown_desc
			NOT = { has_character_flag = balnazzar_demanded_purity_tribute_cooldown }
		}
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		add_gold = medium_gold_value
		add_piety = minor_piety_value
		add_character_flag = { flag = balnazzar_demanded_purity_tribute_cooldown years = 3 }

		every_vassal = {
			limit = { NOT = { faith = faith:scarletism } }
			add_opinion = { modifier = wc_purity_tribute_demanded target = root }
			random = {
				chance = 30
				set_character_faith_with_conversion = faith:scarletism
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 20
	}
}
