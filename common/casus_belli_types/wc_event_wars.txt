plague_war = {
	group = event
	icon = struggle
	# Lok'tar ogar!
	white_peace_possible = no

	allowed_for_character = { NOT = { scope:attacker = { has_character_modifier = crisis_surrender_modifier } } # Warcraft
		always = yes
	}
	
	allowed_against_character = { NOT = { scope:defender = { has_character_modifier = crisis_surrender_modifier } } # Warcraft
		always = yes
	}
	
	valid_to_start = {
		scope:attacker = { NOT = { is_isolated_from_trigger = { target = scope:defender } } } # Warcraft
		always = no
	}

	should_invalidate = {
		scope:attacker = { is_isolated_from_trigger = { target = scope:defender } } # Warcraft
	
		scope:attacker = { NOT = { has_trait_with_flag = plague_scheme_available } }
	}

	on_invalidated_desc = plague_war_on_invalidated_desc
	
	on_invalidated = {
	}

	on_declaration = {
		on_declared_war = yes
	}	

	on_victory_desc = plague_war_on_victory_desc
	
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		
		war_reparation_victory_effect = {
			PRESTIGE = shatter_realm_prestige_value
		}
		
		# Warcraft
		title_and_vassal_plague_taker_effect = { WINNER = scope:attacker LOSER = scope:defender }
		
		# Warcraft
		title_and_vassal_resolver_invasion_effect = { WINNER = scope:attacker LOSER = scope:defender }

		resolve_title_and_vassal_change = scope:change

		# Truce
		add_truce_attacker_victory_effect = yes

		# FP1: note the victory for future memorialisation via stele (if applicable).
		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }

		add_legitimacy_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = invasion_war_white_peace_desc_defender
			}
			desc = invasion_war_white_peace_desc
		}
	}
	
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		
		war_reparation_defeat_effect = {
			GOLD = 0
			PRESTIGE = medium_prestige_value
		}

		add_truce_white_peace_effect = yes

		# EP3: note gold gained from military assistance/join war contracts and their war contribution threshold
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = invasion_war_white_peace_desc_defender
			}
			desc = invasion_war_white_peace_desc
		}
	}
	
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		
		# Legitimacy
		add_legitimacy_attacker_defeat_effect = yes

		#EP2 accolade glory gain for winning against higher ranked enemy
		scope:defender = { accolade_defender_war_end_glory_gain_med_effect = yes }

		war_reparation_defeat_effect = {
			GOLD = 2
			PRESTIGE = major_prestige_value
		}
		
		# Warcraft
		scope:attacker = {
			shatter_realm_effect = { WINNER = scope:defender LOSER = scope:attacker }
		}
		
		# Truce
		add_truce_attacker_defeat_effect = yes
		
		scope:attacker = {
			save_temporary_scope_as = loser
		}
		on_lost_aggression_war_discontent_loss = yes

		add_legitimacy_attacker_defeat_effect = yes
	}
	
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	
	war_name = "PLAGUE_WAR_NAME"
	cb_name = "PLAGUE_CB_NAME"
	
	interface_priority = 200
	
	ticking_war_score_targets_entire_realm = yes
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	attacker_ticking_warscore = 0
	attacker_wargoal_percentage = 0.8
}

				limit = {
					any_vassal = {
						has_trait = creature_high_elf
						has_trait = being_undead
						highest_held_title_tier >= tier_duchy
					}
				}
				random_vassal = {
					limit = {
						has_trait = creature_high_elf
						has_trait = being_undead
						highest_held_title_tier >= tier_duchy
					}

					save_scope_as = new_holder
				}
			}
			else = {
				create_character = {
					culture = scope:attacker.culture
					faith = scope:attacker.faith
					employer = scope:attacker
					trait = creature_high_elf
					trait = being_undead
					gender_female_chance = {
						if = {
							limit = { root.faction_leader.culture = { has_cultural_parameter = martial_custom_male_only_combatant } }
							add = 0
						}
						else_if = {
							limit = { root.faction_leader.culture = { has_cultural_parameter = martial_custom_female_only_combatant } }
							add = 100
						}
						else = {
							add = 50
						}
					}
					save_scope_as = new_holder
					after_creation = {
						trigger_race_giving_no_gene_effect = yes # Warcraft, Assigns race trait
					}
				}
			}
		}

				}
			}

					change = scope:change
					take_baronies = no
				}
			}

				change_title_holder_include_vassals = {
					holder = scope:new_holder
					change = scope:change
					take_baronies = no
				}
			}
			title:c_sunwell_plateau = {
				change_title_holder_include_vassals = {
					holder = scope:new_holder
					change = scope:change
					take_baronies = no
				}
			}
		}

		title:k_zulaman.holder = { # Trolls regain their lands held by the elves

				usurp_entire_title_effect = { TARGET = scope:defender TITLE = title:k_zulaman ONLY_DIRECT = no TAKE_TITLE = no }
			}
		}

		resolve_title_and_vassal_change = scope:change

		if = {
			limit = {
				NOT = {
					scope:new_holder.liege = title:e_scourge.holder
				}
			}

			create_title_and_vassal_change = {

			}

			scope:new_holder = {
				change_liege = {
					liege = title:e_scourge.holder
					change = scope:change
				}
			}
		}

		resolve_title_and_vassal_change = scope:change

		# Attacker gets Prestige Experience, Defender loses Prestige, all other participants gain Prestige based on their war contribution.
		modify_all_participants_fame_values = {
			WINNER = scope:attacker
			LOSER = scope:defender
			FAME_BASE = 100 # Set manually because I'm a rebel
			IS_RELIGIOUS_WAR = no
			WINNER_FAME_SCALE = 10
			LOSER_FAME_SCALE = -10
			WINNER_ALLY_FAME_SCALE = 10
			LOSER_ALLY_FAME_SCALE = 10
		}
		
		# Truce
		add_truce_attacker_victory_effect = yes

		# FP1: note the victory for future memorialisation via stele (if applicable).
		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }

		# EP3: note gold gained from military assistance/join war contracts and their war contribution threshold
		laamp_as_mercenary_payout_tooltip_effect = yes
	}

		desc = WHITE_PEACE_DISABLED
	}

	on_white_peace = {
	}

	on_defeat_desc = {

			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }

			}

		}
	}

		# Legitimacy
		add_legitimacy_attacker_defeat_effect = yes

		scope:attacker = { show_pow_release_message_effect = yes }

		# Prestige loss for the attacker

			pay_short_term_gold = {
				gold = 5
				target = scope:defender
				yearly_income = yes
			}
			add_prestige = {
				value = massive_prestige_value
				multiply = -1.0
			}
		}
		
		setup_invasion_cb = {
			titles = target_titles
			attacker = scope:attacker
			defender = scope:defender
			claimant = scope:claimant
			victory = no
		}

		# Attacker loses Prestige, all other war participants gain Prestige (Defender gains full prestige, all allies on both sides gain based on war contribution).
		modify_all_participants_fame_values = {
			WINNER = scope:defender
			LOSER = scope:attacker

			IS_RELIGIOUS_WAR = no
			WINNER_FAME_SCALE = 10
			LOSER_FAME_SCALE = -10
			WINNER_ALLY_FAME_SCALE = 10

	group = religious
	ai = no

	defender_faith_can_join = yes

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	# Additional requirements to the defines in 00_casus_belli_groups.txt
	allowed_for_character = {always = no}

	allowed_for_character_display_regardless = {}

	allowed_against_character = {
		scope:attacker = {
			faith = {
				faith_hostility_level = {
					target = scope:defender.faith
					value >= religious_cb_enabled_hostility_level
				}
			}
			ALL_FALSE = {
				top_liege = scope:defender.top_liege
				liege = scope:defender
			}
		}
	}
	target_titles = neighbor_land_or_water
	target_title_tier = all
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder

	valid_to_start = {
		#scope:target = {
		#	tier = tier_kingdom
		#}
		always = no
	}

	should_invalidate = {
		OR = {
			NOT = {
				any_in_list = {
					list = target_titles
					any_in_de_jure_hierarchy = {
						tier = tier_county
						holder = {
							OR = {
								this = scope:defender
								target_is_liege_or_above = scope:defender
							}
						}
					}
				}
			}
			#Faith change does not immediately invalidate, it's run through event war_event.3100
			scope:war = { exists = var:invalidate_defender_faith_change }
			scope:war = { exists = var:invalidate_attacker_faith_change }
		}
	}

	on_invalidated_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = {
						any_in_list = {
							list = target_titles
							any_in_de_jure_hierarchy = {
								tier = tier_county
								holder = {
									OR = {
										this = scope:defender
										target_is_liege_or_above = scope:defender
									}
								}
							}
						}
					}
				}
				desc = msg_religious_war_invalidation_region_message
			}
			triggered_desc = {
				trigger = {
					scope:war = {
						OR = {
							exists = var:invalidate_defender_faith_change
							exists = var:invalidate_attacker_faith_change
						}
					}
				}
				desc = msg_religious_war_invalidated_hostility_level_message
			}
		}
	}

	on_invalidated = {

	}

		piety = {
			value = 0
		}
	}

		scope:attacker = { add_character_flag = declared_major_religious_war_flag }
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker.faith = {
						has_doctrine = doctrine_pluralism_pluralistic
					}
				}
				desc = religious_war_victory_desc_doctrine_pluralism
			}
			desc = religious_war_victory_desc
		}
		triggered_desc = {
			trigger = { scope:attacker = { is_local_player = yes } }
			desc = major_religious_war_warning_desc
		}
	}

	on_victory = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			if = {
				limit = {
					has_variable_list = supporters_of_terenia_menethil
				}
				every_in_list = {
					variable = supporters_of_terenia_menethil
					limit = { is_alive = no }
					prev = { remove_list_variable = { name = supporters_of_terenia_menethil target = prev } }
				}
				every_in_list = {
					variable = supporters_of_terenia_menethil
					limit = { is_alive = yes }
					if = {
						limit = { scope:defender = { highest_held_title_tier = tier_empire } }
						add_gold = 400
					}
					else_if = {
						limit = { scope:defender = { highest_held_title_tier = tier_kingdom } }
						add_gold = 300
					}
					else_if = {
						limit = { scope:defender = { highest_held_title_tier = tier_duchy } }
						add_gold = 200
					}
					else = { add_gold = 100 }
				}
			}
		}

		scope:attacker = { accolade_attacker_war_end_glory_gain_high_effect = yes }

		create_title_and_vassal_change = {
			type = conquest_holy_war
			save_scope_as = change
			add_claim_on_loss = yes
		}

			list = target_titles
			custom_tooltip = RELIGIOUS_CB_TITLE

				RELIGIOUS_WAR = yes
			}
		}

			list = vassals_taken
			change_liege = {
				liege = scope:attacker
				change = scope:change
			}
		}

			list = titles_taken
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
			hidden_effect = {
				if = {
					limit = { tier = tier_county NOT = { this = title:c_lordaeron } NOT = { culture = culture:lordaeronian } }

						chance = 60
						set_county_culture = culture:lordaeronian

				}

					limit = { this = title:c_lordaeron NOT = { culture = culture:lordaeronian } }

					set_county_faith = scope:attacker.faith
					change_development_level = { value = development_level divide = 10 multiply = -7 }
				}
			}

			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		every_in_list = {
			list = target_titles
				scope:attacker = {
				add_piety_experience = {
					add = religious_cb_piety_gain_kingdom
					if = {
						limit = {
							scope:attacker.faith = {
								has_doctrine_parameter = bonus_holy_war_piety_active
							}
						}
						multiply = 1.5
					}
				}
			}
		}

			list = target_titles
			order_by = tier

				value = -5
				desc = fervor_loss_religious_war
			}
		}
		scope:defender.faith = {
			change_fervor = {
				value = 10
				desc = fervor_gain_religious_war
			}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.

			FAME_BASE = religious_cb_ally_prestige_kingdom
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}

		scope:attacker = {
			add_piety = religious_cb_piety_white_peace
			stress_impact = {
				ambitious = medium_stress_impact_gain

			stress_impact = {
				arrogant = medium_stress_impact_gain
			}
		}

	}

	}
