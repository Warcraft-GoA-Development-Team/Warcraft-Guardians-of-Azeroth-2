vanish_historical_character_effect = {
	override_death_effect = { death_reason = death_vanished }
}
purge_historical_character_effect = {
	set_to_lowborn = yes
	# Basically reset parents
	if = {
		limit = { exists = father }
		create_character = {
			age = current_year
			gender = male
			location = title:c_stormwind.title_province
			faith = faith:drogbar_religion
			culture = culture:drogbar
			random_traits = no
			dynasty = none
			save_scope_as = dummy_male
		}
		set_father = scope:dummy_male
		scope:dummy_male = {
			trigger_race_giving_no_gene_effect = yes # Warcraft, Assigns race trait
			override_death_effect = { death_reason = death_vanished }
		}
	}
	if = {
		limit = { exists = real_father }
		create_character = {
			age = current_year
			gender = male
			location = title:c_stormwind.title_province
			faith = faith:drogbar_religion
			culture = culture:drogbar
			random_traits = no
			dynasty = none
			save_scope_as = dummy_male
		}
		set_real_father = scope:dummy_male
		scope:dummy_male = {
			trigger_race_giving_no_gene_effect = yes # Warcraft, Assigns race trait
			override_death_effect = { death_reason = death_vanished }
		}
	}
	if = {
		limit = { exists = mother }
		create_character = {
			age = current_year
			gender = female
			location = title:c_stormwind.title_province
			faith = faith:drogbar_religion
			culture = culture:drogbar
			random_traits = no
			dynasty = none
			save_scope_as = dummy_female
		}
		set_mother = scope:dummy_female
		scope:dummy_female = {
			trigger_race_giving_no_gene_effect = yes # Warcraft, Assigns race trait
			override_death_effect = { death_reason = death_vanished }
		}
	}
	if = {
		limit = { is_alive = yes }
		override_death_effect = { death_reason = death_vanished }
	}
}

}
