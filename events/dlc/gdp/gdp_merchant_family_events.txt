scripted_effect gdp_merchant_family_0001_marry_and_start_camp_effect = {
	scope:house_member = {
		if = {
			limit = {
				is_female = yes
				is_lowborn = no
				scope:wedding_interest = {
					is_lowborn = yes
				}
			}
			marry_matrilineal = scope:wedding_interest
		}
		else_if = {
			limit = {
				is_male = yes
				is_lowborn = yes
				scope:wedding_interest = {
					is_lowborn = no
				}
			}
			marry_matrilineal = scope:wedding_interest
		}
		else = {
			marry = scope:wedding_interest
		}
		if = {
			limit = { has_playable_adventurer_dlc_trigger = yes }
			create_landless_adventurer_title_effect = {
				REASON = flag:runaway_allowed
				FLAVOR_CHAR = root
			}
			add_courtier = scope:wedding_interest
		}
		else = {
			select_and_move_to_pool_effect = yes
			scope:wedding_interest = { select_and_move_to_pool_effect = yes }
		}
	}
}

# House member of root who is available to run away
scripted_trigger gdp_merchant_family_0001_valid_house_member_trigger = {
	is_available_ai_adult = yes
	NOR = {
		is_married = yes
		has_trait = chaste
		has_trait = celibate
		has_sexuality = asexual
	}
	is_attracted_to_gender_of = $WEDDING_INTEREST$
	is_courtier_of = root
}

# Available to run away
scripted_trigger gdp_merchant_family_0001_valid_wedding_interest_trigger = {
	is_available_ai_adult = yes
	is_house_head = no
	liege ?= {
		is_ai = yes
	}
	NOR = {
		is_married = yes
		has_trait = chaste
		has_trait = celibate
		has_sexuality = asexual
	}
	save_temporary_scope_as = gdp_merchant_family_0001_wedding_interest_check
	root = {
		any_close_family_member = {
			gdp_merchant_family_0001_valid_house_member_trigger = { WEDDING_INTEREST = scope:gdp_merchant_family_0001_wedding_interest_check }
		}
	}
}

# Star-Crossed Lovers
gdp_merchant_family.0001 = {
	type = character_event
	title = gdp_merchant_family.0001.t
	desc = {
		desc = gdp_merchant_family.0001.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:house_member = {
						OR = {
							has_trait = callous
							has_trait = stubborn
							has_trait = arrogant
						}
					}
				}
				desc = gdp_merchant_family.0001.desc.stubborn_outro
			}
			desc = gdp_merchant_family.0001.desc.fallback_outro
		}
	}
	theme = marriage
	override_background = corridor_night
	left_portrait = {
		character = scope:house_member
		triggered_animation = {
			trigger = {
				scope:house_member = {
					OR = {
						has_trait = callous
						has_trait = stubborn
						has_trait = arrogant
					}
				}
			}
			animation = war_over_win
		}
		animation = beg
		camera = camera_event_right
	}
	right_portrait = {
		character = scope:wedding_interest
		animation = worry
	}

	trigger = {
		is_house_head = yes
		government_has_flag = government_is_merchant
		is_available_adult = yes
		# We only want this to happen once in a campaign
		primary_title ?= {
			NOT = { has_variable = had_event_gdp_merchant_family_0001 }
		}
		house ?= {
			OR = {
				has_house_unity_stage = rapacious_cs
				has_house_unity_stage = opportunistic_cs
			}
			any_house_relation = {
				OR = {
					has_house_relation_level = quarrel
					has_house_relation_level = rivalry
					has_house_relation_level = feud
				}
				any_relation_house = {
					this != root.house
					any_house_member = {
						gdp_merchant_family_0001_valid_wedding_interest_trigger = yes
					}
				}
			}
		}
	}

	weight_multiplier = {
		modifier = {
			factor = 2
			house ?= {
				any_house_relation = {
					count >= 2
					OR = {
						has_house_relation_level = quarrel
						has_house_relation_level = rivalry
						has_house_relation_level = feud
					}
				}
			}
		}
	}

	immediate = {
		primary_title ?= {
			set_variable = had_event_gdp_merchant_family_0001
		}
		house ?= {
			random_house_relation = {
				limit = {
					OR = {
						has_house_relation_level = quarrel
						has_house_relation_level = rivalry
						has_house_relation_level = feud
					}
					any_relation_house = {
						this != root.house
						any_house_member = {
							gdp_merchant_family_0001_valid_wedding_interest_trigger = yes
						}
					}
				}
				random_relation_house = {
					limit = {
						this != root.house
						any_house_member = {
							gdp_merchant_family_0001_valid_wedding_interest_trigger = yes
						}
					}
					save_scope_as = wedding_interest_house
					house_head = { save_scope_as = wedding_interest_house_head }
					random_house_member = {
						limit = {
							gdp_merchant_family_0001_valid_wedding_interest_trigger = yes
						}
						save_scope_as = wedding_interest
					}
				}
			}
		}
		random_close_family_member = {
			limit = {
				gdp_merchant_family_0001_valid_house_member_trigger = { WEDDING_INTEREST = scope:wedding_interest }
			}
			save_scope_as = house_member
		}
	}

	option = { # Take the gold, and go with my blessing.
		name = gdp_merchant_family.0001.a
		pay_short_term_gold = {
			target = scope:house_member
			gold = {
				add = scope:wedding_interest.sum_of_all_skills_value
				add = scope:wedding_interest.prowess
			}
		}
		gdp_merchant_family_0001_marry_and_start_camp_effect = yes
		ai_chance = {
			base = 10
			
			modifier = {
				factor = 0
				OR = {
					has_trait = greedy
					gold <= {
						add = scope:wedding_interest.sum_of_all_skills_and_prowess_value
						multiply = 2
					}
				}
			}
		}
	}

	option = { # Convince both to stay.
		name = gdp_merchant_family.0001.b
		skill = intrigue

		duel = {
			skill = intrigue
			value = high_skill_rating
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				desc = gdp_merchant_family.0001.b.success
				send_interface_toast = {
					title = gdp_merchant_family.0001.b.success
					left_icon = scope:house_member
					right_icon = scope:wedding_interest
					gdp_merchant_family_0001_marry_and_start_camp_effect = yes
					scope:house_member = {
						progress_towards_friend_effect = {
							CHARACTER = root
							OPINION = 0
							REASON = friend_helped_me_marry
						}
						add_opinion = {
							target = root
							modifier = grateful_opinion
							opinion = 50
						}
					}
					scope:wedding_interest = { save_scope_as = relationship_reason_involved_character }
					clear_saved_scope = relationship_reason_involved_character
				}
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				desc =  gdp_merchant_family.0001.b.failure
				send_interface_toast = {
					title =  gdp_merchant_family.0001.b.failure
					left_icon = scope:house_member
					right_icon = scope:wedding_interest
					scope:house_member = {
						add_stress = medium_stress_gain
						add_opinion = {
							target = root
							modifier = disappointed_opinion
							opinion = -30
						}
					}
				}
			}
		}

		stress_and_fulfillment_impact = {
			generous = medium_stress_impact_gain
		}

		ai_chance = {
			base = 10
			
			modifier = {
				factor = 0
				has_trait = generous
			}
		}
	}

	option = { # Refuse them
		name = gdp_merchant_family.0001.c

		stress_and_fulfillment_impact = {
			compassionate = medium_stress_impact_gain
			generous = medium_stress_impact_gain
			callous = minor_stress_impact_loss
			sadistic = minor_stress_impact_loss
			chaste = minor_stress_impact_loss
		}
		scope:house_member = {
			add_opinion = {
				target = root
				modifier = heartbroken_opinion
				opinion = -30
			}
		}
		ai_chance = {
			base = 10
			
			modifier = {
				factor = 0
				OR = {
					has_trait = compassionate
					has_trait = generous
				}
			}
		}
	}
}

# Thicker than Water
# water is an adopted house member or associate who outdoes a blood relative
# We weigh this up when they are adopted, since the stakes are clearer
gdp_merchant_family.0002 = {
	type = character_event
	title = gdp_merchant_family.0002.t
	desc = {
		desc = gdp_merchant_family.0002.desc_intro
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:water = { gdp_merchant_family_0002_adopted_trigger = yes }
				}
				desc = gdp_merchant_family.0002.desc_adopted
			}
			desc = gdp_merchant_family.0002.desc_associate
		}
		desc = gdp_merchant_family.0002.desc_outro
	}
	theme = family
	left_portrait = {
		character = root
		animation = stressed_teacher
	}
	right_portrait = {
		character = scope:blood
		animation = debating
	}
	lower_right_portrait = {
		character = scope:water
	}

	trigger = {
		is_house_head = yes
		government_has_flag = government_is_merchant
		# We only want this to happen once in a campaign
		primary_title ?= {
			NOT = { has_variable = had_event_gdp_merchant_family_0002 }
		}
		OR = {
			AND = {
				any_close_family_member = {
					is_available_ai_adult = yes
					is_courtier_of = root
					gdp_merchant_family_0002_adopted_trigger = yes
					save_temporary_scope_as = temp_water_adopted
				}
				any_close_family_member = {
					is_available_ai_adult = yes
					is_courtier_of = root
					NOT = { gdp_merchant_family_0002_adopted_trigger = yes }
					sum_of_all_skills_value < scope:temp_water_adopted.sum_of_all_skills_value
				}
			}
			AND = {
				any_courtier = {
					is_available_ai_adult = yes
					NOR = {
						is_spouse_of = root
						is_close_family_of = root
						gdp_merchant_family_0002_adopted_trigger = yes
					}
					save_temporary_scope_as = temp_water_associate
				}
				any_close_family_member = {
					is_available_ai_adult = yes
					is_courtier_of = root
					NOT = { gdp_merchant_family_0002_adopted_trigger = yes }
					sum_of_all_skills_value < scope:temp_water_associate.sum_of_all_skills_value
				}
			}
		}
	}

	weight_multiplier = {
		modifier = {
			add = 2
			any_close_family_member = {
				is_available_ai_adult = yes
				is_courtier_of = root
				gdp_merchant_family_0002_adopted_trigger = yes
				sum_of_all_skills_value >= sum_of_all_skills_threshold_good
			}
		}
	}

	immediate = {
		primary_title ?= {
			set_variable = had_event_gdp_merchant_family_0002
		}
		# Prefer an adopted house member
		if = {
			limit = {
				any_close_family_member = {
					is_available_ai_adult = yes
					is_courtier_of = root
					gdp_merchant_family_0002_adopted_trigger = yes
					save_temporary_scope_as = temp_water_adopted
				}
				any_close_family_member = {
					is_available_ai_adult = yes
					is_courtier_of = root
					NOT = { gdp_merchant_family_0002_adopted_trigger = yes }
					sum_of_all_skills_value < scope:temp_water_adopted.sum_of_all_skills_value
				}
			}
			random_close_family_member = {
				limit = {
					is_available_ai_adult = yes
					is_courtier_of = root
					gdp_merchant_family_0002_adopted_trigger = yes
					# Only an adopted member who actually outdoes a blood relative, so scope:blood can always be saved
					save_temporary_scope_as = temp_water
					root = {
						any_close_family_member = {
							is_available_ai_adult = yes
							is_courtier_of = root
							NOT = { gdp_merchant_family_0002_adopted_trigger = yes }
							sum_of_all_skills_value < scope:temp_water.sum_of_all_skills_value
						}
					}
				}
				# They better be good
				weight = {
					base = 1
					modifier = {
						add = sum_of_all_skills_value
					}
				}
				save_scope_as = water
			}
		}
		# Fallback to any skilled associate who isn't a house member
		else = {
			random_courtier = {
				limit = {
					is_available_ai_adult = yes
					NOR = {
						is_spouse_of = root
						is_close_family_of = root
						gdp_merchant_family_0002_adopted_trigger = yes
					}
					save_temporary_scope_as = temp_water
					root = {
						any_close_family_member = {
							is_available_ai_adult = yes
							is_courtier_of = root
							NOT = { gdp_merchant_family_0002_adopted_trigger = yes }
							sum_of_all_skills_value < scope:temp_water.sum_of_all_skills_value
						}
					}
				}
				weight = {
					base = 1
					modifier = {
						add = sum_of_all_skills_value
					}
				}
				save_scope_as = water
			}
		}
		random_close_family_member = {
			limit = {
				is_available_ai_adult = yes
				is_courtier_of = root
				NOT = { gdp_merchant_family_0002_adopted_trigger = yes }
				sum_of_all_skills_value < scope:water.sum_of_all_skills_value
			}
			save_scope_as = blood
		}
	}

	# Side with blood relative / family loyalty
	option = {
		name = gdp_merchant_family.0002.a
		house = {
			add_unity_value = {
				value = minor_unity_gain
				character = root
				desc = gdp_merchant_family.0002.a.unity_gain.desc
			}
		}
		add_character_modifier = {
			modifier = gdp_nepotism_modifier
			years = 10
		}
		scope:blood = {
			add_stewardship_skill = 1
		}
		stress_and_fulfillment_impact = {
			arbitrary = minor_stress_impact_loss
			callous = minor_stress_impact_loss
			greedy = minor_stress_impact_gain
		}

		ai_chance = {
			base = 10
			ai_value_modifier = {
				ai_greed = medium_chance_impact_negative_ai_value
			}
			modifier = {
				add = 20
				has_trait = arbitrary
			}
			modifier = {
				add = 20
				has_trait = callous
			}
			modifier = {
				add = -15
				has_trait = greedy
			}
		}
	}

	# Side with the higher-performing water candidate
	option = {
		name = gdp_merchant_family.0002.b
		house = {
			add_unity_value = {
				value = minor_unity_loss
				character = root
				desc = gdp_merchant_family.0002.b.unity_loss.desc
			}
		}
		add_character_modifier = {
			modifier = gdp_merit_modifier
			years = 10
		}
		stress_and_fulfillment_impact = {
			cynical = minor_stress_impact_loss
			greedy = minor_stress_impact_loss
		}

		ai_chance = {
			base = 10
			ai_value_modifier = {
				ai_greed = medium_chance_impact_positive_ai_value
			}
			modifier = {
				add = 10
				has_trait = cynical
			}
			modifier = {
				add = 10
				has_trait = greedy
			}
		}
	}

	# Figure it out
	option = {
		name = gdp_merchant_family.0002.c
		
		stress_and_fulfillment_impact = {
			base = minor_stress_impact_loss

			lazy = minor_stress_impact_loss
			cynical = minor_stress_impact_loss
			compassionate = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
			modifier = {
				add = 25
				has_trait = lazy
			}
			modifier = {
				add = 20
				has_trait = cynical
			}
			modifier = {
				add = -20
				has_trait = compassionate
			}
		}
	}
}
