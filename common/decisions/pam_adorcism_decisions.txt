### Curse Enemy ###
pam_curse_enemy_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/disaster.dds"
	}
	decision_group_type = religious

	# Warcraft - Add troll / loa flavor
	title = { 
		first_valid = { 
			triggered_desc = {
				trigger = { 
					limit = { religion = religion:loa_group }
					desc = pam_curse_enemy_decision_loa
				}
			}
			desc = pam_curse_enemy_decision
		}
	}
	desc = { 
		first_valid = { 
			triggered_desc = {
				trigger = { 
					limit = { 
						AND = { 
							religion = religion:loa_group
							culture = { has_cultural_pillar = language_zandali }
						}
					}
					desc = pam_curse_enemy_decision_desc_loa_troll
				}
			}
			triggered_desc = {
				trigger = { 
					limit = { religion = religion:loa_group }
					desc = pam_curse_enemy_decision_desc_loa
				}
			}
			desc = pam_curse_enemy_decision_desc 
		} 
	}

	selection_tooltip = pam_curse_enemy_decision_tooltip
	confirm_text = pam_curse_enemy_decision_confirm

	sort_order = 26
	cooldown = { years = 5 }

	is_shown = {
		has_pam_dlc_trigger = yes
		has_personal_tenet_flag = tenet_adorcism_curse_enemy
		pam_adorcism_has_available_curse_target_trigger = yes
	}

	is_valid = {
		custom_description = {
			text = pam_curse_enemy_decision_valid_target
			pam_adorcism_has_available_curse_target_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		custom_description = {
			text = pam_curse_enemy_decision_selected_target_valid
			pam_adorcism_curse_target_is_valid_trigger = {
				TARGET = scope:selected_item
				ACTOR = root
			}
		}
	}

	cost = {
		piety = major_piety_value
	}

	effect = {
		scope:selected_item ?= {
			save_scope_as = pam_adorcism_curse_target
		}
		show_as_tooltip = {
			add_character_modifier = {
				modifier = pam_adorcism_curse_focus_modifier
				years = 3
			}
			add_stress = major_stress_loss
			change_spiritual_fulfillment = minor_spiritual_fulfillment_value
			if = {
				limit = { exists = scope:selected_item }
				scope:selected_item = {
					if = {
						limit = {
							rite = root.rite
						}
						add_character_modifier = {
							modifier = pam_adorcism_ill_omen_modifier
							years = 1
						}
					}
					if = {
						limit = {
							OR = {
								rite = root.rite
								probably_superstitious = yes
							}
						}
						add_character_modifier = {
							modifier = pam_adorcism_curse_anxiety_modifier
							years = 1
						}
					}
				}
			}
		}
		trigger_event = pam_adorcism_events.0001
	}

	widget = {
		gui = "decision_view_widget_select_character"
		controller = select_scope_object

		setup_items = {
			scope:actor = {
				every_relation = {
					type = rival
					limit = { pam_adorcism_curse_target_is_living_adult_trigger = yes }
					add_to_list = item_list
				}
				every_war_enemy = {
					limit = {
						pam_adorcism_curse_target_is_living_adult_trigger = yes
						NOR = {
							has_relation_rival = scope:actor
							has_relation_nemesis = scope:actor
						}
					}
					add_to_list = item_list
				}
			}
		}

		default_item = {
			scope:actor = {
				if = {
					limit = {
						any_relation = {
							type = rival
							pam_adorcism_curse_target_is_living_adult_trigger = yes
						}
					}
					ordered_relation = {
						type = rival
						limit = { pam_adorcism_curse_target_is_living_adult_trigger = yes }
						order_by = max_military_strength
						position = 0
						save_scope_as = selected_item
					}
				}
				else_if = {
					limit = {
						any_war_enemy = {
							pam_adorcism_curse_target_is_living_adult_trigger = yes
						}
					}
					ordered_war_enemy = {
						limit = { pam_adorcism_curse_target_is_living_adult_trigger = yes }
						order_by = max_military_strength
						position = 0
						save_scope_as = selected_item
					}
				}
			}
		}

		is_item_valid = {
			pam_adorcism_curse_target_is_valid_trigger = {
				TARGET = root
				ACTOR = scope:actor
			}
		}

		ai_select_item = {
			scope:actor = {
				if = {
					limit = {
						any_relation = {
							type = rival
							pam_adorcism_curse_target_is_living_adult_trigger = yes
						}
					}
					ordered_relation = {
						type = rival
						limit = { pam_adorcism_curse_target_is_living_adult_trigger = yes }
						order_by = max_military_strength
						position = 0
						save_scope_as = selected_item
					}
				}
				else = {
					ordered_war_enemy = {
						limit = { pam_adorcism_curse_target_is_living_adult_trigger = yes }
						order_by = max_military_strength
						position = 0
						save_scope_as = selected_item
					}
				}
			}
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 60
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	ai_potential = {
		ai_vengefulness >= 0
		pam_adorcism_has_available_curse_target_trigger = yes
	}

	ai_will_do = {
		base = 35
		modifier = {
			add = 35
			OR = {
				has_trait = wrathful
				has_trait = vengeful
			}
		}
		modifier = {
			add = -35
			OR = {
				has_trait = calm
				has_trait = forgiving
			}
		}
		ai_value_modifier = {
			ai_vengefulness = 0.5
			ai_zeal = 0.25
		}
	}
}
