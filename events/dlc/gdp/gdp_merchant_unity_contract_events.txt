scripted_trigger gdp_merchant_unity_contract_1010_would_rip_off_customers_trigger = {
	NOR = {
		# Honest chars never lie.
		has_trait = honest
		# Just characters won't lie unless they're actively deceitful.
		AND = {
			has_trait = just
			NOT = { has_trait = deceitful }
		}
	}
}

scripted_effect gdp_merchant_unity_contract_1010_find_partner_effect = {
	# Clean the partner if they exist (in case they die between events).
	clear_saved_scope = partner
	# Now nab 'em afresh.
	court_position:partner_court_position ?= {
		if = {
			limit = {
				is_available_allow_travelling = yes
				location = root.location
				# Not subject of investigation
				NOT = { this = scope:task_contract.task_contract_employer }
			}
			save_scope_as = partner
		}
	}
}

scripted_effect gdp_merchant_unity_contract_1010_person_scope_effect = {
	if = {
		limit = {
			NOT = { exists = scope:person_c }
		}
		save_scope_as = person_c
	}
	else_if = {
		limit = {
			NOT = { exists = scope:person_d }
		}
		save_scope_as = person_d
	}
}

scripted_trigger gdp_merchant_unity_contract_1010_person_scope_trigger = {
	is_available_quick = {
		ai = yes
		adult = yes
		imprisoned = no
		incapable = no
	}
	NOR = {
		scope:person_c ?= this
		scope:person_d ?= this
	}
}

#	Arrival at market event.
gdp_merchant_unity_contract.1010 = {
	type = character_event
	title = gdp_merchant_unity_contract.1010.t
	window = visit_settlement_window
	desc = {
		desc = gdp_merchant_unity_contract.1010.desc.intro
		triggered_desc = {
			trigger = { exists = scope:partner }
			desc = gdp_merchant_unity_contract.1010.desc.intro.has_partner
		}
		desc = gdp_merchant_unity_contract.1010.desc.midtro
		desc = gdp_merchant_unity_contract.1010.desc.outro
	}
	theme = trade
	left_portrait = {
		character = root
		animation = thinking
	}
	center_portrait = {
		character = scope:local_ruler
		animation = dismissal
	}
	right_portrait = {
		character = scope:trader
		animation = disapproval
	}
	lower_left_portrait = scope:person_c
	lower_center_portrait = scope:person_d
	override_background = { reference = alley_day }

	immediate = {
		scope:trader = {
			random_list = {
				# Trader is a bad'n
				100 = {
					trigger = { gdp_merchant_unity_contract_1010_would_rip_off_customers_trigger = yes }
					save_scope_value_as = {
						name = irregularity
						value = flag:missing
					}
				}
				# Trader is only human
				100 = {
					trigger = { gdp_merchant_unity_contract_1010_would_rip_off_customers_trigger = yes }
					save_scope_value_as = {
						name = irregularity
						value = flag:quality
					}
				}
				# Trader is a good'n
				100 = {
					save_scope_value_as = {
						name = irregularity
						value = flag:dues
					}
				}
			}
		}
		# Find merchants
		scope:market = {
			every_character_in_market = {
				limit = {
					government_has_flag = government_is_merchant
					NOT = { this = root }
					gdp_merchant_unity_contract_1010_person_scope_trigger = yes
				}
				add_to_temporary_list = potential_merchants_1
			}
			every_trade_route_in_market = {
				limit = {
					trade_route_owner = {
						government_has_flag = government_is_merchant
						NOR = {
							this = root
							is_in_list = potential_merchants_1
						}
						gdp_merchant_unity_contract_1010_person_scope_trigger = yes
					}
				}
				trade_route_owner = { add_to_temporary_list = potential_merchants_2 }
			}
			every_in_global_list = {
				variable = merchants_tally
				limit = {
					holder = {
						NOR = {
							this = root
							is_in_list = potential_merchants_1
							is_in_list = potential_merchants_2
						}
						gdp_merchant_diplo_range_trigger = {
							MERCHANT = this
							TARGET = root
						}
						gdp_merchant_diplo_range_trigger = {
							MERCHANT = this
							TARGET = scope:local_ruler
						}
						gdp_merchant_unity_contract_1010_person_scope_trigger = yes
					}
				}
				holder = { add_to_temporary_list = potential_merchants_3 }
			}
		}
		hidden_effect = {
			while = {
				limit = {
					NOT = { exists = scope:person_d }
				}
				random_list = {
					7 = {
						trigger = {
							list_size = { name = potential_merchants_1 value > 0 }
						}
						random_in_list = {
							list = potential_merchants_1
							limit = { gdp_merchant_unity_contract_1010_person_scope_trigger = yes }
							gdp_merchant_unity_contract_1010_person_scope_effect = yes
							remove_from_list = potential_merchants_1
						}
					}
					7 = {
						trigger = {
							list_size = { name = potential_merchants_1 value < 1 }
							list_size = { name = potential_merchants_2 value > 0 }
						}
						random_in_list = {
							list = potential_merchants_2
							limit = { gdp_merchant_unity_contract_1010_person_scope_trigger = yes }
							gdp_merchant_unity_contract_1010_person_scope_effect = yes
							remove_from_list = potential_merchants_2
						}
					}
					7 = {
						trigger = {
							list_size = { name = potential_merchants_1 value < 1 }
							list_size = { name = potential_merchants_2 value < 1 }
							list_size = { name = potential_merchants_3 value > 0 }
						}
						random_in_list = {
							list = potential_merchants_3
							limit = { gdp_merchant_unity_contract_1010_person_scope_trigger = yes }
							gdp_merchant_unity_contract_1010_person_scope_effect = yes
							remove_from_list = potential_merchants_3
						}
					}
					10 = {
						scope:local_ruler = {
							ordered_courtier = {
								limit = {
									OR = {
										location = scope:destination
										location = scope:local_ruler.capital_province
									}
									NOR = {
										is_close_or_extended_family_of = scope:local_ruler
										is_consort_of = scope:local_ruler
									}
									gdp_merchant_unity_contract_1010_person_scope_trigger = yes
								}
								order_by = {
									value = 0
									add = stewardship
									if = {
										limit = { is_councillor = yes }
										add = 10
									}
									if = {
										limit = { has_any_court_position = yes }
										add = 10
									}
								}
								gdp_merchant_unity_contract_1010_person_scope_effect = yes
							}
						}
					}
					10 = {
						create_character = {
							template = merchant_template
							location = scope:destination
							culture = scope:destination.culture
							faith = scope:destination.faith
							after_creation = {
								add_character_flag = created
								gdp_merchant_unity_contract_1010_person_scope_effect = yes
								trigger_race_giving_no_gene_effect = yes # Warcraft
							}
						}
					}
				}
			}
		}
		### Append them to the task contract: if they die too early, we invalidate.
		#### Largely a concern for MP.
		scope:task_contract = {
			set_variable = {
				name = person_c
				value = scope:person_c
			}
			set_variable = {
				name = person_d
				value = scope:person_d
			}
		}
		# Note that we'll be taking our background cues from scope:person_b (local_ruler).
		scope:local_ruler = { save_scope_as = bg_override_char }
		# Try to grab our partner.
		gdp_merchant_unity_contract_1010_find_partner_effect = yes
		# Plus, stop us whilst we're here.
		hidden_effect = {
			current_travel_plan = { pause_travel_plan = yes }
		}
	}
	
	option = { # Meet scope:local_ruler (person_b).
		name = gdp_merchant_unity_contract.1010.a
		scope:local_ruler = {
			custom_tooltip = gdp_merchant_unity_contract.1010.a.tt
			save_scope_as = char_current
		}
		trigger_event = gdp_merchant_unity_contract.1011
	}
	
	option = { # Meet scope:person_c.
		name = gdp_merchant_unity_contract.1010.b
		trigger = { exists = scope:person_c }
		scope:person_c = {
			custom_tooltip = gdp_merchant_unity_contract.1010.a.tt
			save_scope_as = char_current
		}
		trigger_event = gdp_merchant_unity_contract.1011
	}
	
	option = { # Meet scope:person_d.
		name = gdp_merchant_unity_contract.1010.c
		trigger = { exists = scope:person_d }
		scope:person_d = {
			custom_tooltip = gdp_merchant_unity_contract.1010.a.tt
			save_scope_as = char_current
		}
		trigger_event = gdp_merchant_unity_contract.1011
	}
	
	option = { # Fast resolution.
		name = gdp_merchant_unity_contract.1010.d
		custom_tooltip = gdp_merchant_unity_contract.1012.tt.boilerplate.end_investigation
		trigger_event = gdp_merchant_unity_contract.1021
	}
}

scripted_effect gdp_merchant_unity_contract_1011_duel_generic_effect = {
	save_scope_value_as = {
		name = method
		value = flag:$SKILL$
	}
	duel = {
		skill = $SKILL$
		target = scope:char_current
		# Victory!
		50 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = 3.5
				min = -49
			}
			min = 5
			desc = gdp_merchant_unity_contract.1011.$SKILL$.tt.success
			gdp_merchant_unity_contract_1011_duel_success_effect = yes
		}
		# Defeat!
		50 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = -3.5
				min = -49
			}
			min = 5
			desc = gdp_merchant_unity_contract.1011.$SKILL$.tt.failure
			gdp_merchant_unity_contract_1011_duel_failure_effect = yes
		}
	}
}
