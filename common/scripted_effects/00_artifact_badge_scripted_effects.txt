## UNIQUE BADGES ##
setup_unique_pilgrim_badge_vars_effect = {
	scope:newly_created_artifact ?= {
		set_variable = is_$HOLY_SITE$_pilgrim_badge
		set_variable = is_unique_pilgrim_badge
		set_variable = {
			name = is_pilgrim_badge
			value = flag:yes
		}
		set_variable = {
			name = pilgrim_badge_location
			value = scope:badge_location
		}		
	}
}

#For use when you visit a holy site
create_artifact_badge_effect = {
	$RECEIVER$ = { save_scope_as = receiving_char }
	hidden_effect_new_object = {
		scope:receiving_char.location = { save_scope_as = badge_location }
		# Santiago de Compostela
		# if = {
		# 	limit = { scope:badge_location.barony = title:b_santiago }
		# 	create_artifact_badge_santiago_effect = { ACQUIRER = scope:receiving_char }
		# }
		# #Jerusalem
		# else_if = {
		# 	limit = { scope:badge_location.barony = title:b_jerusalem }
		# 	create_artifact_badge_jerusalem_effect = { ACQUIRER = scope:receiving_char }
		# }
		# #Rome
		# else_if = {
		# 	limit = { scope:badge_location.barony = title:b_vaticano }
		# 	create_artifact_badge_rome_effect = { ACQUIRER = scope:receiving_char }
		# }
		#Generic
		#else = {
			create_artifact_badge_generic_effect = { ACQUIRER = scope:receiving_char }
		#}
		#Make the history look neat
		scope:newly_created_artifact ?= {
			add_artifact_history = {
				type = created
				location = scope:badge_location
				actor = scope:receiving_char
				recipient = scope:receiving_char
			}
			save_scope_as = created_pilgrim_badge
		}
	}
}

#For use in events, outside of going on an actual pilgrimage to the holy site
generate_artifact_badge_effect = {
	$OWNER$ = { save_scope_as = current_owner }
	hidden_effect_new_object = {
		random_list = {
			1 = {
				trigger = {
					NOT = {
						scope:current_owner = {
							any_character_artifact = { has_variable = is_santiago_pilgrim_badge }
						}
					}
				}
				create_artifact_badge_santiago_effect = { ACQUIRER = scope:current_owner }
			}
			1 = {
				trigger = {
					NOT = {
						scope:current_owner = {
							any_character_artifact = { has_variable = is_jerusalem_pilgrim_badge }
						}
					}
				}
				create_artifact_badge_jerusalem_effect = { ACQUIRER = scope:current_owner }
			}
			1 = {
				trigger = {
					NOT = {
						scope:current_owner = {
							any_character_artifact = { has_variable = is_rome_pilgrim_badge }
						}
					}
				}
				create_artifact_badge_rome_effect = { ACQUIRER = scope:current_owner }
			}
			10 = {
				create_artifact_badge_generic_effect = { ACQUIRER = scope:current_owner }
			}
		}
		#Make the history look neat
		random_pool_character = {
			province = scope:badge_location
			limit = { 
				age = { 20 40 }
				is_lowborn = yes
			}
			save_scope_as = dummy_vendor
		}
		if = {
			limit = {
				NOT = { exists = scope:dummy_vendor }
			}
			create_character = {
				location = scope:badge_location
				random_traits = yes
				dynasty = none
				faith = scope:badge_location.faith
				rite = scope:badge_location.rite
				culture = scope:badge_location.culture
				gender_female_chance = 50
				save_scope_as = dummy_vendor
			}
			scope:dummy_vendor = { add_character_flag = created_vendor }
		}
	}
	scope:newly_created_artifact = {
		add_artifact_history = {
			type = purchased
			location = scope:badge_location
			actor = scope:dummy_vendor
			recipient = scope:current_owner
		}
		save_scope_as = generated_pilgrim_badge 
	}
	scope:dummy_vendor = {
		if = {
			limit = { has_character_flag = created_vendor }
			silent_disappearance_effect = yes
		}
	}
	clear_saved_scope = badge_location
}
