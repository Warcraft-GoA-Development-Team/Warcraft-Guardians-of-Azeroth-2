# Called every January 1st
# Root does not exist
yearly_global_pulse = {
    # Warcraft
	events = {
		wc_horde_invasion.0001 # Horde Invasion: Setup Blackhand and the associated story
	}
    effect = {
        # Warcraft
		# Sends notifications about the Dark Portal reopening
		if = {
			limit = {
				is_target_in_global_variable_list = {
					name = available_unique_events
					target = flag:reopens_dark_portal
				}
				
				dark_portal_is_opened_trigger = yes
				title:c_dark_portal.holder.top_liege = {
					has_draenor_culture_trigger = yes
					draenor_faith_trigger = yes
				}
			}
			reopen_dark_portal_effect = yes
		}
    }
    on_actions = {
        lich_king_invasion_on_action # Happens between 590 - 601
        orcish_invasion_of_kalimdor_on_action # Happens between 592 - 602
		theramore_survivors_story_on_action # Happens between 605 - 610
		burning_legion_invasion_on_action # Happens between 604 - 624
		eternitysend_on_action # Happens a few months after Burning Legion has spawned
		burning_legion_invasion_of_kaldorei_on_action # Happens after Cenarius dies
		illidan_stormrage_on_action # Happens when Cenarius dies
        thunder_king_invasion_on_action # Happens between 600 - 650 or after 650 depending on game rules

		# Warcraft
		delay = { days = { 1 360 } }
		draenor_events_pulse
    }
}

lich_king_invasion_on_action = {
    trigger = {
        current_date > 590.1.1
        game_start_date < 603.1.1 # All of this is only for 583 start date and what not
        NOT = {
            is_target_in_global_variable_list = {
                name = warcraft_invaders_story
                target = flag:lich_king_story_started
            }
        }
    }
    effect = {
        random = {
			chance = 0
			modifier = {
				add = 1
				current_date > 590.1.1
			}
			modifier = {
				add = 4
				current_date > 592.1.1
			}
			modifier = {
				add = 5
				current_date > 594.1.1
			}
			modifier = {
				add = 15
				current_date > 596.1.1
			}
			modifier = {
				add = 30
				current_date > 598.1.1
			}
            modifier = {
                add = 45
                current_date > 600.1.1
            }
			debug_log = "attempting to spawn Lich King soon"
			trigger_event = {
				id = WCCSC.01
				days = { 2 240 } # keep this smaller than 1 year if the global variable is set in event .0001
			}
		}
    }
}

burning_legion_invasion_on_action = {
	trigger = {
		current_date > 604.1.1
		game_start_date < 605.6.6
		NOT = { has_game_rule = wc_burning_legion_crisis_arthas_dependant }
		NOT = {
            is_target_in_global_variable_list = {
                name = warcraft_invaders_story
                target = flag:burning_legion_story_started
            }
        }
	}
	effect = {
		random = {
			chance = 0
			modifier = { # If Arthas or Arthas' alternative exists, let them cook
				add = -25
				any_living_character = { scourge_quelthalas_frostmourne_check_trigger = yes }
			}
			modifier = {
				add = -100
				title:b_nordrassil.title_province = { has_building = nordrassil_destroyed }
			}
			modifier = {
				add = 1
				current_date > 604.1.1
			}
			modifier = {
				add = 4
				current_date > 608.1.1
			}
			modifier = {
				add = 5
				current_date > 612.1.1
			}
			modifier = {
				add = 15
				current_date > 616.1.1
			}
			modifier = {
				add = 30
				current_date > 620.1.1
			}
            modifier = {
                add = 45
                current_date > 624.1.1
				has_game_rule = wc_burning_legion_crisis_guaranteed
            }
			debug_log = "Trying to spawn Burning Legion and Archimonde soon"
			summon_burning_legion_effect = yes
			trigger_event = { # This starts the subjugation content and asks doombringer faiths to join the Burning Legion
				id = burning_legion.0100
				days = 2
			}
		}
	}
}

thunder_king_invasion_on_action = {
    trigger = {
        current_date > 600.1.1
        NOT = { has_game_rule = wc_thunder_king_crisis_never }
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:revive_lei_shen_decision
            }
        }
    }
    effect = {
        ### Thunder King Crisis by Juke
		# Based on Mongol Invasion Code
		if = {
			limit = { has_game_rule = wc_thunder_king_crisis_random }
			random = {
				chance = 1
				modifier = {
					factor = 0.1
					current_date < 600.1.1
				}
				modifier = {
					add = 3
					current_date > 660.1.1
				}
				modifier = {
					add = 5
					current_date > 690.1.1
				}
				modifier = {
					add = 5
					current_date > 715.1.1
				}
				debug_log = "Attempting to spawn Thunder King within 240 days"
				trigger_event = {
					id = pandaria_unification.0400
					days = { 2 240 }
				}
			}
		}
		else = {
            limit = {
                current_date > 650.1.1
                current_date < 750.1.1
            }
            debug_log = "Checking for chance of spawning Thunder King after 650"
            random = {
                chance = 1
                modifier = {
                    add = 9
                    current_date > 660.1.1
                }
                modifier = {
                    add = 5
                    current_date > 690.1.1
                }
                modifier = {
                    add = 10
                    current_date > 715.1.1
                }
                modifier = {
                    add = 50
                    current_date > 650.1.1
                    has_game_rule = wc_thunder_king_crisis_guaranteed
                }
                debug_log = "Attempting to spawn Thunder King within 240 days"
                trigger_event = {
                    id = pandaria_unification.0400
                    days = { 2 240 }
                }
            }
		}
    }
}
