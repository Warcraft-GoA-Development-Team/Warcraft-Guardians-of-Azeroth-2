# Notify players
middle_east_decisions.1101 = {
	type = character_event
	window = big_event_window
	title = middle_east_decisions.1100.t
	desc = {
		desc = middle_east_decisions.1101.intro
		desc = newline.desc
		first_valid = {
			triggered_desc = {
				trigger = { faith.religion = religion:islam_religion }
				desc = middle_east_decisions.1101.islam_desc
			}
			desc = middle_east_decisions.1101.other_desc
		}
	}
	theme = crown
	override_background = {
		reference = ep2_holy_site_mena #Warcraft
	}
	override_effect_2d = { reference = legend_glow }
	left_portrait = {
		character = scope:dar_al_islam_ruler
		animation = jockey_wave
		camera = camera_event_table_right
	}
	
	immediate = {
		show_as_tooltip = {
			scope:dar_al_islam_ruler = {
				get_title = title:h_dar_al_islam
			}
		}
	}
	
	option = {
		name = {
			trigger = {
				faith.religion = religion:islam_religion
				top_liege = scope:dar_al_islam_ruler
			}
			text = middle_east_decisions.1101.a
		}
		name = {
			trigger = {
				faith.religion = religion:islam_religion
				top_liege != scope:dar_al_islam_ruler
			}
			text = middle_east_decisions.1101.b
		}
		name = {
			trigger = { faith.religion != religion:islam_religion }
			text = middle_east_decisions.1101.c
		}
	}
}

# Notify other Noorist head of faiths, and destroy their (now obsolete) head of faith title
middle_east_decisions.1102 = {
	type = character_event
	window = big_event_window
	title = middle_east_decisions.1100.t
	desc = {
		desc = middle_east_decisions.1101.intro
		desc = newline.desc
		desc = middle_east_decisions.1102.desc
	}
	theme = crown
	override_background = {
		reference = ep2_holy_site_mena #Warcraft
	}
	override_effect_2d = { reference = legend_glow }
	left_portrait = {
		character = scope:dar_al_islam_ruler
		animation = jockey_wave
		camera = camera_event_table_right
	}
	
	immediate = {
		show_as_tooltip = {
			scope:dar_al_islam_ruler = {
				get_title = title:h_dar_al_islam
			}
		}
		if = {
			limit = { exists = var:old_hof_title }
			destroy_title = var:old_hof_title
			remove_variable = old_hof_title
		}
	}
	
	option = {
		name = middle_east_decisions.1102.a
	}
}
