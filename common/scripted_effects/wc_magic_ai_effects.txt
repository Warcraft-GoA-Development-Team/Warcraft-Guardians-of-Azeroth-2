decide_to_cast_spell_effect = { # Stage 2
    random_list = {
        50 = { # Yes
            # TODO_MAGIC_AI: Add modifiers for wether or not they want to cast
            if = {
                limit = {
                    debug_only = yes 
                }
                change_global_variable = {
                    name = ai_spells_pre_cast_stage_2
                    add = 1
                }
            }
            select_spell_class_ai_effect = yes
        }
        50 = { # Nahh
            # Come back later
            trigger_event = {
                on_action = wc_on_ai_spellcasting
                days = { 7 35 } # So it will not cause a major lag on every 1st of each month
            } 
        }
    }
}

set_last_elemental_type_effect = {
    if = {
        limit = {
            exists = var:current_spell_class
            OR = {
                var:current_spell_class = flag:elemental_air
                var:current_spell_class = flag:elemental_water
                var:current_spell_class = flag:elemental_fire
                var:current_spell_class = flag:elemental_earth
            }
            exists = var:last_elemental_type
            is_ai = no
        }
        set_variable = {
            name = elemental_type
            value = var:last_elemental_type
        }
    }
    if = {
        limit = {
            is_ai = yes
        }
        # TODO_MAGIC_AI: Add more modifiers for if they'll pick spirit or decay
        random_list = {
            10 = {
                modifier = {
                    add = 25
                    exists = var:last_elemental_type
                    var:last_elemental_type = flag:spirit
                }
                set_variable = {
                    name = elemental_type
                    value = flag:spirit
                }
                set_variable = {
                    name = last_elemental_type
                    value = flag:spirit
                }
            }
            10 = {
                modifier = {
                    add = 25
                    exists = var:last_elemental_type
                    var:last_elemental_type = flag:decay
                }
                set_variable = {
                    name = elemental_type
                    value = flag:decay
                }
                set_variable = {
                    name = last_elemental_type
                    value = flag:decay
                }
            }
        }
    }
}
