RelationDescriptor = {
	type = character

	text = {
		trigger = {
			opinion = {
				target = scope:second
				value > 0
			}
		}
		localization_key = positive_desc_of_other
	}

	text = {
		trigger = {
			opinion = {
				target = scope:second
				value <= 0
			}
		}
		localization_key = negative_desc_of_other
	}
}

GetCrime = {
	type = character

	text = {
		trigger = { has_trait = murderer }
		localization_key = crime_murderer
	}

	text = {
		trigger = {
			liege.rite = {
				NOR = {
					rite_has_doctrine = doctrine_kinslaying_accepted
					rite_has_doctrine = doctrine_kinslaying_shunned
				}
			}
			OR = {
				has_trait = kinslayer_1
				has_trait = kinslayer_2
				has_trait = kinslayer_3
			}
		}
		localization_key = crime_kinslayer
	}

	text = {
		trigger = {
			liege.rite = { rite_has_parameter = deviancy_illegal }
			has_trait = deviant
		}
		localization_key = crime_deviant
	}

	text = {
		trigger = { 
			NOT = {
				liege.rite = { rite_has_tenet = tenet_ritual_cannibalism }
			}
			has_trait = cannibal
		}
		localization_key = crime_cannibal
	}

	text = {
		trigger = {
			liege.rite = { rite_has_parameter = homosexuality_illegal }
			is_male = yes
			has_trait = sodomite
		}
		localization_key = crime_sodomite
	}

	text = {
		trigger = {
			NOT = {
				liege.rite = { rite_has_parameter = allows_unrestricted_marriage }
			}
			has_trait = incestuous
		}
		localization_key = crime_incestuous
	}

	text = {
		trigger = {
			OR = {
				AND = {
					liege.rite = { rite_has_parameter = adultery_female_crime }
					is_female = yes
				}
				AND = {
					liege.rite = { rite_has_parameter = adultery_male_crime }
					is_male = yes
				}
			}
			has_trait = adulterer
		}
		localization_key = crime_adulterer
	}

	text = {
		trigger = {
			liege.rite = { rite_has_parameter = witchcraft_illegal }
			has_trait = witch
		}
		localization_key = crime_witch
	}

	text = {
		trigger = { has_trait = fornicator }
		localization_key = crime_fornicator
	}

	text = {
		trigger = { has_trait = denounced }
		localization_key = crime_denounced
	}

	text = {
		trigger = {  # Warcraft
			OR = {
				has_trait = excommunicated
				has_trait = scarlet_mark
			}
		}
		localization_key = crime_excommunicated
	}

	text = {
		trigger = {
			religion = ROOT.religion
			faith != ROOT.faith
		}
		localization_key = crime_heretic
	}

	text = {
		trigger = {
			NOT = {
        		ROOT = { like_pagan_religion_trigger = yes }
        	}
        	religion != ROOT.religion
        	like_pagan_religion_trigger = yes
		}
		localization_key = crime_heathen
	}

	text = {
		trigger = {
			religion != ROOT.religion
		}
		localization_key = crime_infidel
	}

	text = {
		trigger = { always = yes }
		localization_key = crime_fallback
	}
}
