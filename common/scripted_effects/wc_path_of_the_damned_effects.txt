funeral_of_terenas_effect = {
	hurt_scourge_jailor_consequences_effect = { HURT_JAILOR = root SCOURGE = scope:killer }

	# Save potential terenas scope
	global_var:path_of_the_damned_story.var:king_of_lordaeron = { save_scope_as = terenas }

	# Uther Stress Event
	title:k_order_of_the_silver_hand.holder = {
		save_scope_as = uther
		scope:killer = { save_scope_as = arthas }
		trigger_event = { 
			id = order_of_the_silver_hand_events.001
			days = 3
		}
	}
	# Sally Whitemane Conversion
	character:60572 = { set_character_faith_with_conversion = faith:scarletism }

	# Terenas Funeral
	# Compile list of people who should come.
	every_close_or_extended_family_member = { add_to_temporary_list = funeral_attendees }
	every_vassal_or_below = { add_to_temporary_list = funeral_attendees }
	every_courtier = { add_to_temporary_list = funeral_attendees }
	character:60036 = { add_to_temporary_list = funeral_attendees } # Uther
	every_relation = { type = lover add_to_temporary_list = funeral_attendees }
	every_relation = { type = friend add_to_temporary_list = funeral_attendees }

	# We carry the flame
	ordered_in_list = {
		list = funeral_attendees

		limit = { is_landed = yes } # Attackable

		order_by = {
			if = {
				limit = { has_title = title:k_order_of_the_silver_hand }
				value = 2
			}
			else = {
				value = 1
			}
		}

		add_character_flag = { flag = carry_urn_of_the_king_of_lordaeron }
	}

	every_in_list = {
		list = funeral_attendees
		# have to save these now because it wont work after he dies
		if = {
			limit = { # Terenas had a good opinion of them or vv
				is_alive = yes 
				OR = {
					opinion = {
						value > 0
						target = scope:terenas 
					}
					scope:terenas = {
						opinion = {
							value > 0
							target = prev
						}
					}
				}
			}
			add_character_flag = {
				flag = terenas_good
				days = 30
			}
		}

		trigger_event = {
			id = wc_path_of_the_damned.0002
			days = 6
		}
	}
}

find_tichondrius_effect = {
	find_nerzhul_effect = yes

	scope:nerzhul = {
		every_relation = {
			type = scourge_jailor
			add_to_temporary_list = valid_tichondrius
		}
	}
	ordered_in_list = {
		limit = { is_ai = yes }
		list = valid_tichondrius
		order_by = find_tichondrius_value
		max = 1
			
		save_scope_as = tichondrius
	}
	if = {
		limit = { NOT = { exists = scope:tichondrius } }

		create_character = {
			template = tichondrius_character_template
			employer = scope:nerzhul
				
			after_creation = { save_scope_as = tichondrius }
		}
	}
}

		}
	}

			limit = { has_variable = urn_of_king_of_lordaeron }
			save_scope_as = enchanted_urn
		}
	}

	scope:defender = { trigger_event = wc_path_of_the_damned.2004 }

		limit = { scope:attacker = { is_ai = no } }

	}
}
