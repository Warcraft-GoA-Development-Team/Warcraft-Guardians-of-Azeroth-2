#Scripted Modifiers based on a character's Religion or Faith.
religion_scaled_virtuous_traits_modifier = {
	modifier = {
		add = {
			add = $BASE_VALUE$
			multiply = $CHARACTER$.num_virtuous_traits
		}
		$CHARACTER$ = {
			num_virtuous_traits > 0
		}
	}
}
