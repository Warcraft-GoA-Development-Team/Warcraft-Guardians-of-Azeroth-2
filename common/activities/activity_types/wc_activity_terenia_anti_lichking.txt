activity_anti_lichking = {
	is_shown = {
		exists = character:60677
		this = character:60677
		is_playable_character = yes
		NOT = { has_global_variable = terenia_tried_lich_king }
		is_incapable = no
		OR = {
			has_variable = conquer_lordaeron_first
			has_variable = kill_lich_king_first
		}
		# DLC locked
		has_dlc_feature = tours_and_tournaments
		has_dlc_feature = advanced_activities
	}

	can_start = {
		trigger_if = {
			limit = { has_variable = conquer_lordaeron_first }
			OR = {
				has_title = title:e_lordaeron
				has_title = title:k_lordaeron
			}
		}
	}

	can_start_showing_failures_only = {
		is_available_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	ai_select_num_provinces = {
		value = 0
	}

	is_valid = {
		any_attending_character = {
			is_alive = yes
		}
		trigger_if = {
			limit = { NOT = { has_variable = killed_lich_king } }
			title:c_icecrown.holder = {
				OR = {
					has_trait = lich_king
					has_title = title:e_scourge
				}
			}
		}
	}

	guest_join_chance = {
		base = 300
	}

	wait_time_before_start = {
		months = 2
	}
	max_guest_arrival_delay_time = {
		months = 2
	}
	on_invalidated = {
		if = {
			limit = {
				scope:host = { is_alive = yes }
			}
			every_attending_character = {
				limit = {
					is_alive = yes
				}
				dynasty ?= {
					root = {
						add_to_variable_list = {
							name = dynasties_for_renown_reward
							target = prev
						}
					}
				}
				trigger_event = personal_adventure.100
			}
		}
		add_reward_dynasty_alk_effect = yes
		
		if = {
			limit = { character:60677 = { is_alive = yes } }
			character:60677 = { trigger_event = { id = personal_adventure.055 years = { 1 2 } } }
		}
	}

	on_host_death = {
		#Make the player as a new host, so he can fight a duel too, other way lets all go home, fruck lich king
		activity_special_type_progression_major = yes
		if = {
			limit = { 
				has_current_phase = alk_phase_killing_lich_king
				any_attending_character = { is_alive = yes is_ai = no NOT = { this = scope:host } }
			}
			scope:host = { save_scope_as = old_host }
			random_attending_character = {
				limit = {
					is_ai = no
					is_alive = yes
					NOT = { this = scope:old_host }
				}
				save_scope_as = new_host
			}
			#Also inherit the travel plan
			if = {
				limit = {
					scope:host.current_travel_plan = {
						any_entourage_character = { this = scope:new_host }
					}
				}
				scope:host.current_travel_plan = { set_travel_plan_owner = scope:new_host }
			}
			set_activity_host = scope:new_host
			scope:new_host = {
				trigger_event = { id = personal_adventure.106 days = { 6 8 } }
			}
		}
		#50% someone decides to keep fighting lich king, what a brave person, 50% all decide to run away
		else_if = {
			limit = {
				has_current_phase = alk_phase_killing_lich_king
				any_attending_character = { is_alive = yes NOT = { this = scope:host } }
			}
			random_list = {
				1 = {
					scope:host = { save_scope_as = old_host }
					random_attending_character = {
						limit = {
							is_alive = yes
							NOT = { this = scope:old_host }
						}
						save_scope_as = new_host
					}
					#Also inherit the travel plan
					if = {
						limit = {
							scope:host.current_travel_plan = {
								any_entourage_character = { this = scope:new_host }
							}
						}
						scope:host.current_travel_plan = { set_travel_plan_owner = scope:new_host }
					}
					set_activity_host = scope:new_host
					scope:new_host = {
						trigger_event = { id = personal_adventure.106 days = { 6 8 } }
					}
				}
				1 = {
					scope:host = {
						current_travel_plan = {
							every_entourage_character = { set_location_to_default = yes }
						}
					}
				}
			}
		}
		#Otherwise indirectly cancel the activity by not setting a new host
		else = {
			# Send your entourage home
			scope:host = {
				current_travel_plan = {
					every_entourage_character = { set_location_to_default = yes }
				}
			}
		}
	}
