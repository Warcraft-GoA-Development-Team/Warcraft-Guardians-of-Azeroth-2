			exists = father
			father = { has_character_flag = wc_balnazzar_raised_alexandros }
		}

			exists = father
			father = { has_character_flag = wc_balnazzar_raised_alexandros }

	}

		is_imprisoned = no

	wc_balnazzar_renault_owner_trigger = yes
	global_var:balnazzar_dreadlord_story = {

		NOT = { exists = var:wc_balnazzar_renault_delay }
		NOT = { exists = var:wc_balnazzar_renault_trial_resolved }
	}
	OR = {
		character:60019 = { is_imprisoned_by = root }
		AND = {
			wc_balnazzar_renault_present_trigger = yes
			OR = {
				wc_balnazzar_renault_darion_trigger = yes
				AND = {
					character:60019 = { opinion = { target = root value <= -25 } }
					has_hook_of_type = { target = character:60019 type = favor_hook }
				}
			}
		}
	}
}

# Recheck the presented cause; a changed situation closes the stale scene.
wc_balnazzar_renault_trial_choice_trigger = {
	wc_balnazzar_renault_trial_trigger = yes
	global_var:balnazzar_dreadlord_story = { exists = var:wc_balnazzar_renault_cause }
	trigger_if = {
		limit = { global_var:balnazzar_dreadlord_story = { var:wc_balnazzar_renault_cause = flag:prison } }
		character:60019 = { is_imprisoned_by = root }
	}
	trigger_else_if = {
		limit = { global_var:balnazzar_dreadlord_story = { var:wc_balnazzar_renault_cause = flag:darion } }
		wc_balnazzar_renault_present_trigger = yes
		wc_balnazzar_renault_darion_trigger = yes
	}
	trigger_else = {
		wc_balnazzar_renault_present_trigger = yes
		wc_balnazzar_renault_darion_trigger = no
		character:60019 = { opinion = { target = root value <= -25 } }
		has_hook_of_type = { target = character:60019 type = favor_hook }
	}
}

		wc_balnazzar_parricide_trigger = yes
	}
	OR = {
		AND = {
			wc_balnazzar_parricide_trigger = yes
