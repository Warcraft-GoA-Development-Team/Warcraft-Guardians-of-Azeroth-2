



# Set one of your children as heir
#This interaction is referenced in code! If it's renamed you have to ping a coder
designate_heir_interaction = {
	icon = heir

	category = interaction_category_diplomacy

	desc = designate_heir_interaction_desc

	is_shown = {
		# Is candidate (recipient) valid?
		scope:recipient = {
			is_child_of = scope:puppet_or_actor
		}
		NOR = {
			scope:recipient = scope:puppet_or_actor.designated_heir
			scope:recipient = scope:puppet_or_actor
		}

		scope:puppet_or_actor = {
			trigger_if = {
				limit = {
					always = yes # Warcraft
					#is_roman_emperor_primary_title_trigger = no
				}
				custom_description = {
					text = "dynasty_member_leads_election"
					NAND = {
						primary_title ?= {
							has_title_law_flag = elective_succession_law
						}
						dynasty ?= primary_heir.dynasty
					}
				}
			}
		}
	}

	cost = {
		prestige = {
			value = {
				# this not initalized in GUI
				if = {
					limit = { NOT = { exists = scope:puppet_or_actor } }
					scope:actor = { save_temporary_scope_as = puppet_or_actor }
				}
				if = {
					limit = {
						scope:puppet_or_actor = { government_has_flag = government_is_mandala }
					}
					multiply = 0
				}
				else_if = {
					limit = {
						scope:puppet_or_actor = { government_has_flag = government_is_landless_adventurer }
					}
					multiply = {
						value = 0
						desc = "[adventurer|E]"
					}
				}
				else = {
					add = {
						if = {
							limit = {
								scope:puppet_or_actor = {
									government_has_mechanic = administrative
									government_has_flag = government_has_influence
								}
							}
							value = designate_heir_admin_prestige_cost
						}
						else = {
							value = designate_heir_cost
						}
						desc = BASE
					}
				}
			}
			if = {
				limit = { scope:puppet_or_actor != scope:actor }
				multiply = 0
			}
		}
		influence = {
			value = {
				add = 0
				# this not initalized in GUI
				if = {
					limit = { NOT = { exists = scope:puppet_or_actor } }
					scope:actor = { save_temporary_scope_as = puppet_or_actor }
				}
				if = {
					limit = {
						scope:puppet_or_actor = {
							government_has_mechanic = administrative
							government_has_flag = government_has_influence
						}
					}
					add = {
						value = designate_heir_admin_influence_cost
						desc = "[government_bureaucratic_gov|E]"
					}
				}
			}
			if = {
				limit = { scope:puppet_or_actor != scope:actor }
				multiply = 0
			}
		}
		piety = {
			value = {
				add = 0
				# this not initalized in GUI
				if = {
					limit = { NOT = { exists = scope:puppet_or_actor } }
					scope:actor = { save_temporary_scope_as = puppet_or_actor }
				}
				if = {
					limit = {
						scope:puppet_or_actor = { government_has_flag = government_is_mandala }
					}
					add = designate_heir_cost
				}
			}
			if = {
				limit = { scope:puppet_or_actor != scope:actor }
				multiply = 0
			}
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				is_eunuch_trigger = no
				NOR = {
					has_trait = disinherited
					has_trait = devoted
					has_trait = order_member
					has_trait = bastard
				}
				house ?= scope:puppet_or_actor.house
			}
		}
		scope:puppet_or_actor = {
			can_designate_heir_trigger = yes
			trigger_if = {
				limit = {
					government_has_mechanic = administrative
					government_has_flag = government_has_influence
				}
				prestige >= designate_heir_admin_prestige_cost
				influence >= designate_heir_admin_influence_cost

				custom_tooltip = {
					text = valid_co_emperor_candidate_liege_trigger
					scope:recipient.top_liege = scope:puppet_or_actor.top_liege
				}
			}
			trigger_else_if = {
				limit = {
					NOR = {
						government_has_flag = government_is_mandala
						government_has_flag = government_is_landless_adventurer
					}
				}
				prestige >= designate_heir_cost
			}
			trigger_else = {}
			trigger_if = {
				limit = {
					government_has_flag = government_is_landless_adventurer
					exists = scope:recipient
				}
				scope:recipient = {
					is_adult = yes
				}
			}
			trigger_if = {
				limit = {
					government_has_flag = government_is_mandala
				}
				piety >= designate_heir_cost
			}
			trigger_if = {
				limit = { NOT = { scope:puppet_or_actor.primary_heir ?= scope:recipient } }
				custom_tooltip = {
					text = designate_heir_interaction.tt.currently_coruling
					NOT = { has_diarchy_active_parameter = diarchy_is_co_rulership }
				}
			}
		}
		# Warcraft
		scope:actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }
	}

	on_auto_accept = {
		scope:recipient = {
			trigger_event = char_interaction.0120
		}
	}

	on_accept = {
		scope:puppet_or_actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = designate_heir_interaction_notification

				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient

				set_designated_heir = scope:recipient
			}
		}
		notify_puppeteer_of_outcome_effect = {
			TYPE = event_toast_effect_neutral
			TITLE = designate_heir_interaction_notification_puppeteer
		}
	}

	auto_accept = yes
}
