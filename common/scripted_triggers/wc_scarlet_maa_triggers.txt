# Character scope: the registered Crusade holder and every level of vassal.
wc_scarlet_maa_member_trigger = {
	exists = global_var:wc_scarlet_crusade_title
	exists = global_var:wc_scarlet_crusade_title.holder
	OR = {
		this = global_var:wc_scarlet_crusade_title.holder
		is_vassal_or_below_of = global_var:wc_scarlet_crusade_title.holder
	}
}
