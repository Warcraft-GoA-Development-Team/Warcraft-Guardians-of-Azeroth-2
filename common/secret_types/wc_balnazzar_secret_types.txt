# Balnazzar's hidden nature (added via add_dreadlord_disguise_secret_effect).

secret_dreadlord_disguise = {
	category = religious

	is_valid = {
		scope:secret_owner = { has_character_flag = wc_balnazzar_disguise_active }
	}

	is_criminal = {
		always = yes
	}

	# A dreadlord unmasked as a fraud plausibly loses standing, same as any other secret.
	is_shunned = {
		always = yes
	}

	on_expose = {
		# Only reachable via console/debug in normal play, but match the vanilla pattern
		# (e.g. secret_deviant) of guarding against a character exposing their own secret.
		save_scope_as = secret
		scope:secret_owner = {
			if = {
				limit = { this != scope:secret_exposer }
				trigger_event = {
					id = wc_balnazzar_story.2000
					days = 2
				}
			}
		}
		secret_exposed_notification_effect = yes
	}
}
