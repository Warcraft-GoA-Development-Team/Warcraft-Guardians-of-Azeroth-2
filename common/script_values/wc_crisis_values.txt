coalition_strength_value = {
	value = 0
	every_in_list = {
		variable = coalition_members
		add = max_military_strength
	}
}
crisis_power_value = {
	value = 0
	if = {
		limit = {
			coalition_strength_value > 0
			max_military_strength > 0
		}
		add = coalition_strength_value
		divide = max_military_strength
	}
}
coalition_members_number = {
	value = 0
	every_in_list = {
		variable = coalition_members
		add = 1
	}
}
crisis_power_gui_value = {
	value = 0 
	if = {
		limit = {
			crisis_power_value > 0
		}
		value = crisis_power_value
		multiply = 0.75
	}
}
counties_value = {
	every_county = {
		add = 1
	}
}
counties_controlled_value = {
	every_county = {
		limit = {
			root = { completely_controls = prev }
		}
		add = 1
	}
}
counties_controlled_percentage_value = {
	value = counties_controlled_value
	divide = counties_value
}
