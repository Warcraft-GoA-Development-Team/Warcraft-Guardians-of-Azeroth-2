can_be_impotent_trigger = {
	can_have_children = yes
	NOT = { has_trait = impotent }
	NOT = { has_trait_with_flag = dragon_family }
	is_female = yes
	age > age_45_value
}

		has_trait = giant
		has_trait = dire
	}

does_not_need_food_trigger = {
	OR = {
		scope:character ?= {
			is_being_gene_trigger = { BEING = being_void }
			is_being_gene_trigger = { BEING = being_demon }
			is_being_gene_trigger = { BEING = being_undead }
		}
		is_being_gene_trigger = { BEING = being_void }
		is_being_gene_trigger = { BEING = being_demon }
		is_being_gene_trigger = { BEING = being_undead }
	}
}


can_get_world_weary_perk = { 
	OR = { 	
		has_trait = being_order 	has_trait = being_light
		has_trait = being_life		has_trait = being_void
		has_trait = being_undead	has_trait = immortal
		
		culture = { has_cultural_parameter = elder_culture }
	}
}

demon_suicide_trigger = { 
	has_trait = being_demon
	faith = { religion = religion:doombringer }
	age >= 120 # do not change this to one of our values
	has_perk = know_thyself_perk
	NOT = { has_character_modifier = important_lore_character }
	exists = player_heir
}

can_become_immortality_deprived = {
	AND = {
		OR = {
			is_target_in_global_variable_list = {
				name = elven_lifespan
				target = flag:nordrassil_was_sacrificed_to_kill_archimonde
			}
			is_target_in_global_variable_list = {
				name = elven_lifespan
				target = flag:nordrassil_was_destroyed_by_a_demon
			}
		}
		OR = {
			has_trait = creature_night_elf
			has_trait = creature_dryad
		}
		NOR = {
			has_character_modifier = important_lore_character # Don't want Lore characters to look like Grannys

			has_character_modifier = world_tree_sacrificed_for_night_elf
			has_character_modifier = world_tree_sacrificed_for_dryad_and_keeper
			has_character_modifier = world_tree_destroyed_for_night_elf
			has_character_modifier = world_tree_destroyed_for_dryad_and_keeper
		}
	}
}

health_event_is_worried_trigger = {
	OR = {
		ai_compassion >= medium_positive_ai_value
		dread_modified_ai_boldness = {
			dreaded_character = $CHARACTER$
			value <= medium_negative_ai_value
		}
		has_relation_friend = $CHARACTER$
		has_relation_lover = $CHARACTER$
		opinion = {
			target = $CHARACTER$
			value >= high_positive_opinion
		}
	}
	$CHARACTER$.health < fine_health
}
