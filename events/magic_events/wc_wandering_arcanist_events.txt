# The Tirisfal Fragment
wandering_arcanist.0001 = {
	type = character_event
	title = wandering_arcanist.0001.t
	desc = wandering_arcanist.0001.desc
	content_source = dlc_GOA
	theme = order
	override_background = { reference = location_study }

	left_portrait = {
		character = root
		animation = page_flipping
	}

	trigger = {
		wc_wandering_arcanist_valid_trigger = yes

		NOR = {
			has_character_flag = wc_wandering_arcanist_tome_seen
			has_character_flag = wc_magic_flavor_0030_seen
			has_character_flag = wc_arcane_1011_seen
		}

		wc_legion_attention_eligible_trigger = yes
	}

	immediate = {
		add_character_flag = { flag = wc_wandering_arcanist_cd years = 3 }
		add_character_flag = wc_wandering_arcanist_tome_seen
	}

	# WHAT - Study the fragment, whatever it calls to
	option = {
		name = wandering_arcanist.0001.a

		wc_change_legion_attention_effect = { VALUE = 20 }
		if = {
			limit = { order_magic_lifestyle_perks >= 36 }
			wc_change_amount_mp_effect = {
				CHANGE = add
				VALUE = 3
			}
		}
		else = {
			add_order_magic_lifestyle_xp = medium_lifestyle_xp
		}
		wc_event_chance_to_expose_secret_effect = { MAGICTYPE = order VALUE = 10 }

		stress_impact = {
			ambitious = minor_stress_impact_loss
			arrogant = minor_stress_impact_loss
			craven = minor_stress_impact_gain
		}

		ai_chance = {
			base = 50
			ai_value_modifier = {
				ai_boldness = 0.5
				ai_rationality = -0.25
			}
			stress_impact_ai_chance_modifier = { TRAIT = ambitious STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = arrogant STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = craven STRESS = minor_stress_impact_gain }
		}
	}

	# WHAT - Sell it to the Kirin Tor, who know what it is
	option = {
		name = wandering_arcanist.0001.b
		trigger = {
			exists = title:d_kirin_tor.holder
			NOT = { title:d_kirin_tor.holder = root }
		}
		show_as_unavailable = { always = yes }

		wc_change_kirin_tor_standing_effect = { VALUE = 15 }
		if = {
			limit = { is_landless_adventurer = yes }
			domicile ?= { change_provisions = medium_provisions_gain }
		}
		else = {
			add_gold = medium_gold_value
		}

		stress_impact = {
			greedy = minor_stress_impact_loss
			ambitious = minor_stress_impact_gain
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_greed = 1
				ai_rationality = 0.5
			}
			stress_impact_ai_chance_modifier = { TRAIT = greedy STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = ambitious STRESS = minor_stress_impact_gain }
		}
	}

	# WHAT - Burn it
	option = {
		name = wandering_arcanist.0001.c

		wc_change_legion_attention_effect = { VALUE = -5 }
		wc_change_kirin_tor_standing_effect = { VALUE = 5 }
		add_stress = minor_stress_loss

		stress_impact = {
			ambitious = medium_stress_impact_gain
			paranoid = minor_stress_impact_loss
		}

		ai_chance = {
			base = 50
			ai_value_modifier = {
				ai_boldness = -0.5
			}
			stress_impact_ai_chance_modifier = { TRAIT = ambitious STRESS = medium_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = paranoid STRESS = minor_stress_impact_loss }

		}
	}
	content_source = dlc_GOA
	theme = order

		character = root

	}

		trigger = { exists = scope:karazhan_holder }

	}

		wc_wandering_arcanist_valid_trigger = yes

		base = 0.5

		}
		modifier = {
			is_ai = yes
			factor = 0.1
		}
	}

	immediate = {
		title:c_karazhan.holder ?= {
			if = {
				limit = { NOT = { this = root } }
				save_scope_as = karazhan_holder
			}
		}

	# WHAT - Take the key

		name = wandering_arcanist.0007.a

			ambitious = minor_stress_impact_loss
			craven = minor_stress_impact_gain
		}

		ai_chance = {
			base = 100
			stress_impact_ai_chance_modifier = { TRAIT = ambitious STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = craven STRESS = minor_stress_impact_gain }

	}

		name = wandering_arcanist.0007.b

			location = root.location
			culture = culture:nathrezim
			faith = faith:burning_legion_religion
			age = { 16 500 }
			gender_female_chance = 50
			dynasty = none

			after_creation = {
				add_character_flag = created
				trigger_race_giving_no_gene_effect = yes
				set_age = { age_16_value age_500_value }
			}
		}
	}

	# WHAT - Accept the offer (the fall)

		name = wandering_arcanist.0005.a
		add_internal_flag = dangerous

		set_variable = { name = wc_kirin_tor_standing value = 0 }
		add_character_flag = wc_wandering_arcanist_fallen

			limit = { NOT = { has_perk = disorder_magic_tree_1_perk_1 } }

		}
		else = {
			add_disorder_magic_lifestyle_perk_points = 1
		}

			limit = {
				NOT = { can_openly_learn_magic_type_trigger = { MAGICTYPE = disorder } }
				NOT = { any_secret = { secret_type = secret_disorder_magic_user } }
			}
			add_secret = { type = secret_disorder_magic_user }
		}

		random_owned_story = {
			limit = { story_type = story_cycle_wc_legion_attention }
			hidden_effect = { end_story = yes }
		}

		stress_impact = {
			ambitious = medium_stress_impact_loss
			arrogant = minor_stress_impact_loss
			zealous = major_stress_impact_gain
			compassionate = medium_stress_impact_gain
		}

		ai_chance = {
			base = 10
			ai_value_modifier = {
				ai_boldness = 0.5
				ai_zeal = -0.5
			}
			modifier = {
				is_ai = yes
				factor = 0.5
			}
			stress_impact_ai_chance_modifier = { TRAIT = ambitious STRESS = medium_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = arrogant STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = zealous STRESS = major_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = compassionate STRESS = medium_stress_impact_gain }
		}
	}

	option = {
		name = wandering_arcanist.0005.b
		skill = prowess

		duel = {
			skill = prowess
			value = high_skill_rating
			40 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
					min = -39
				}
				min = 5
				send_interface_toast = {
					title = wandering_arcanist.0005.b_good

					left_icon = root

				}
			}

					value = scope:duel_value
					multiplier = -3
					min = -59
				}
				min = 5

			craven = major_stress_impact_gain

			}

			text = {
				first_valid = {
					triggered_desc = {
						trigger = {
							exists = title:d_kirin_tor.holder
							NOT = { title:d_kirin_tor.holder = root }
						}
						desc = wandering_arcanist.0005.c_kirin_tor
					}
					desc = wandering_arcanist.0005.c_priest

			remove_short_term_gold = major_gold_value
		}
		add_prestige = medium_prestige_loss

		stress_impact = {
			arrogant = medium_stress_impact_gain
			humble = minor_stress_impact_loss
		}

		ai_chance = {
			base = 75
			ai_value_modifier = {
				ai_rationality = 0.5
			}
			stress_impact_ai_chance_modifier = { TRAIT = arrogant STRESS = medium_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = humble STRESS = minor_stress_impact_loss }
		}
