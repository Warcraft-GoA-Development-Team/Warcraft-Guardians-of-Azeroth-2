join_coalition_interaction = {
	interface_priority = 60
	category = interaction_category_diplomacy
	desc = action_join_coalition_combined_group_description
	icon = maintab_crises

	is_shown = {
		scope:actor = {
			NOT = { is_in_coalition_against_trigger = { CRISIS_HOLDER = scope:recipient } }
		}
		scope:recipient = {
			is_crisis_trigger = yes
		}
		NOT = { scope:recipient = scope:actor }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			NOR = {
				has_trait = lich_king
				has_trait = thunder_king
				has_trait = conqueror
				has_trait = legion_commander
			}
			is_confederation_member = no # Already in a defensive Confederation
			NOT = { is_isolated_from_trigger = { target = scope:recipient } } # Your lands are in Isolation
			custom_tooltip = {
				text = cannot_into_coalition_since_already_in_union
				leads_any_this_union_trigger = no # You are already in a defensive alliance
				is_in_any_this_union_trigger = no # You are already in a defensive alliance
			}
			has_crisis_coalition_cooldown_trigger = no
			can_be_in_coalition_trigger = yes
			can_be_in_coalition_against_trigger = { CRISIS_HOLDER = scope:recipient }
			custom_tooltip = {
				text = can_into_coalition_since_crisis_on_continent
				has_crisis_on_same_continent_trigger = yes
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			join_crisis_coalition_effect = { CRISIS_HOLDER = scope:recipient }
		}
	}
	auto_accept = yes

	ai_potential = {
		always = yes
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	ai_frequency = 12
	ai_will_do = {
		base = 0
		join_coalition_scripted_modifier = { CRISIS_HOLDER = scope:recipient }
	}
}
leave_coalition_interaction = {
	interface_priority = 60
	category = interaction_category_diplomacy
	desc = leave_coalition_interaction_desc

	is_shown = {
		scope:actor = {
			is_in_coalition_against_trigger = { CRISIS_HOLDER = scope:recipient }
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			is_at_war = no
		}
	}
	
	on_accept = {
		scope:actor = {
			leave_crisis_coalition_effect = { CRISIS_HOLDER = scope:recipient }
		}
	}
	auto_accept = yes

	ai_potential = {
		always = yes
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			can_be_in_coalition_trigger = no
		}
	}
}
