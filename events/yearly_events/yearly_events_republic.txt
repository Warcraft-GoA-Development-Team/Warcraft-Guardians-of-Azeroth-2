scripted_trigger gdp_potential_street_brawler_trigger = {
	is_male = yes
	is_married = no
	is_clergy = no
	age = { 16 $MAX_AGE$ }
	is_available_quick = {
		adult = yes
		imprisoned = no
		incapable = no
		in_army = no
	}
	top_liege = root.top_liege
}

scripted_trigger gdp_house_is_hostile_to_root_house_trigger = {
	house ?= {
		this != root.house
		any_house_relation = {
			any_relation_house = { this = root.house }
			OR = {
				has_house_relation_level = quarrel
				has_house_relation_level = rivalry
				has_house_relation_level = feud
			}
		}
	}
}

scripted_trigger gdp_has_personal_animosity_trigger = {
	has_relation_rival = $CHARACTER$
}

scripted_trigger gdp_can_supply_rival_house_brawler_trigger = {
	any_close_or_extended_family_member = {
		NOR = {
			is_close_or_extended_family_of = root
			house ?= root.house
		}
		gdp_potential_street_brawler_trigger = { MAX_AGE = 25 }
		house.house_head.top_liege ?= root.top_liege
		OR = {
			gdp_house_is_hostile_to_root_house_trigger = yes
			gdp_has_personal_animosity_trigger = { CHARACTER = $BRAWLER$ }
		}
	}
}

scripted_effect vote_intimidation_response_effect = {
	if = {
		limit = {
			has_trait = vengeful
		}
		set_relation_rival = { reason = rival_intimidation target = root }
	}
	else_if = {
		limit = {
			has_trait = brave
		}
		progress_towards_rival_effect = {
			REASON = rival_intimidation
			CHARACTER = root
			OPINION = -15
		}
	}
	else_if = {
		limit = {
			has_trait = craven
		}
		add_opinion = {
			modifier = threatened_opinion
			target = root
			opinion = -5
		}
	}
	else = {
		add_opinion = {
			modifier = threatened_opinion
			target = root
			opinion = -10
		}	
	}
}
