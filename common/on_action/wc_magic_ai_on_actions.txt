wc_on_ai_spellcasting = {
    trigger = { 
        is_ai = yes
        is_magic_user_trigger = yes
    }
    effect = { # Stage 1
        if = {
            limit = {
                debug_only = yes 
            }
            change_global_variable = {
                name = ai_spells_pre_cast_stage_1
                add = 1
            }
        }
        # Can we even cast a spell?
        if = {
            limit = {
                is_any_spell_castable_trigger = yes 
            }
            decide_to_cast_spell_effect = yes
        }
        else = {
            # Come back later
            trigger_event = {
                on_action = wc_on_ai_spellcasting
                days = { minimum_days_before_casting_value maximum_days_before_casting_value } # So it will not cause a major lag on every 1st of each month
            }
        }
    }
}
