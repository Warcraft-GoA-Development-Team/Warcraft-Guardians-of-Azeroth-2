# Who the Legion can notice: arcane users not already bound to the Legion or summoning demons on purpose
wc_legion_attention_eligible_trigger = {
	is_adult = yes

	NOT = { any_relation = { type = summoned } }

	wc_legion_attention_permanently_excluded_trigger = no

	is_magic_class_user_trigger = { MAGIC = order }
}

# Permanent exits from the meter; a temporary one (an active summon) blocks new gains via the eligibility checks in the cast hook and events, while the value keeps decaying
wc_legion_attention_permanently_excluded_trigger = {
	OR = {
		has_character_flag = wc_wandering_arcanist_fallen
		has_trait = being_demon
		has_trait = legion_commander
		faith = faith:burning_legion_religion
		secret_faith ?= { this = faith:burning_legion_religion }
	}
}

wc_legion_attention_noticed_trigger = {
	wc_legion_attention_current_value >= wc_legion_attention_noticed_value
}

wc_legion_attention_hunted_trigger = {
	wc_legion_attention_current_value >= wc_legion_attention_hunted_value
}

wc_legion_attention_marked_trigger = {
	wc_legion_attention_current_value >= wc_legion_attention_marked_value
}

# Shared gate for the pulse events (.0001, .0002, .0007)
wc_wandering_arcanist_valid_trigger = {
	is_adult = yes
	is_incapable = no

	NOT = { has_character_flag = wc_wandering_arcanist_cd }

	is_magic_class_user_trigger = { MAGIC = order }
}

# Shared gate for the stage events (.0003-.0005) fired by the story cycle
wc_legion_attention_stage_event_trigger = {
	is_incapable = no

	wc_legion_attention_eligible_trigger = yes

	# The Legion is a real army during an invasion; the quiet "something notices you" track pauses
	NOT = { exists = title:h_burning_legion.holder }
}
