request_clerical_chastity_interaction = {
	category = interaction_category_religion
	icon = celibate
	interface_priority = 35
	common_interaction = no

	use_diplomatic_range = yes
	can_send_despite_rejection = yes
	ai_maybe = yes
	ai_min_reply_days = 1
	ai_max_reply_days = 5
	popup_on_receive = yes

	desc = request_clerical_chastity_interaction_desc
	greeting = negative
	notification_text = request_clerical_chastity_notification_text

	cost = {
		piety = {
			value = request_clerical_chastity_interaction_cost_value
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	cooldown_against_recipient = { years = 5 }

	is_shown = {
		scope:puppet_or_actor ?= {
			this != scope:recipient
			clerical_chastity_actor_can_demand_chastity_trigger = yes
		}
		scope:recipient = {
			scope:puppet_or_actor.faith ?= faith
			trigger_if = {
				limit = {
					is_clergy = yes
				}
				OR = {
					is_married = yes
					has_trait = lustful
					has_trait = rakish
					has_trait = seducer
					has_trait = deviant
				}
			}
			NOR = {
				has_trait = celibate
				has_trait = eunuch
				has_character_modifier = clerical_chastity_vow_modifier
			}
			trigger_if = {
				limit = { scope:puppet_or_actor = { is_ai = yes } }
				NOT = { has_variable = request_clerical_chastity_ai_cooldown }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor = {
			NOT = {
				is_at_war_with = scope:recipient
			}
			pam_not_antipope_when_recipient_is_hof_tt = { CHARACTER = scope:puppet_or_actor }
			#Gnostic seculars can do this with high learning
			trigger_if = {
				limit = {
					is_clergy = no
				}
				learning >= very_high_skill_rating
			}
		}
		pam_not_promised_pilgrimage_trigger = { ACTOR = scope:puppet_or_actor }
		scope:recipient = {
			NOT = {
				has_character_modifier = clerical_chastity_vow_modifier
			}
			is_physically_able_adult = yes
			trigger_if = {
				limit = { scope:actor = scope:puppet_or_actor }
				custom_tooltip = {
					text = clerical_chastity_recipient_must_be_in_reach_tt
					clerical_chastity_recipient_valid_for_actor_scope_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = clerical_chastity_recipient_must_be_in_reach_tt_puppeteer
					clerical_chastity_recipient_valid_for_actor_scope_trigger = yes
				}
			}
		}
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				piety >= request_clerical_chastity_interaction_cost_value
			}
		}
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:puppet_or_actor
			object = scope:recipient
			scope:hook = yes
		}
	}

	send_option = {
		is_valid = {
			scope:puppet_or_actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	send_option = {
		flag = clerical_chastity_piety_stipend
		localization = clerical_chastity_offer_piety
		starts_enabled = { always = no }
	}
	should_use_extra_icon = {
		scope:puppet_or_actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
	send_options_exclusive = no

	on_accept = {
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				add_piety = {
					value = {
						add = request_clerical_chastity_interaction_cost_value
						max = piety
						multiply = -1
					}
				}
			}
		}
		scope:puppet_or_actor = {
			if = {
				limit = { scope:hook = yes }
				use_hook = scope:recipient
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = request_clerical_chastity_actor_toast
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
				scope:recipient = {
					trigger_event = {
						id = clerical_chastity_vow.0001
						days = 1825
					}
					if = {
						limit = {
							is_available_quick = {
								advanced_ruler = yes
							}
						}
						change_spiritual_fulfillment = {
							value = minor_spiritual_fulfillment_value
							if = {
								limit = { scope:clerical_chastity_piety_stipend = yes }
								add = medium_spiritual_fulfillment_value
							}
						}
					}
					if = {
						limit = { scope:clerical_chastity_piety_stipend = yes }
						add_piety = medium_piety_gain
					}
					send_interface_toast = {
						type = event_toast_effect_good
						title = request_clerical_chastity_recipient_toast
						left_icon = scope:recipient
						right_icon = scope:puppet_or_actor
						add_character_modifier = {
							modifier = clerical_chastity_vow_modifier
							years = 5
						}
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = resented_forced_chastity_opinion
						}
						stress_impact = {
							lustful = medium_stress_impact_gain
						}
					}
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = request_clerical_chastity_actor_toast_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { is_ai = no }
				set_variable = {
					name = request_clerical_chastity_ai_cooldown
					years = 3
				}
			}
		}
	}

	on_decline = {
		scope:puppet_or_actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = request_clerical_chastity_decline_actor_toast
				right_icon = scope:recipient
				scope:recipient = {
					add_piety = medium_piety_loss
					add_stress = minor_stress_gain
					add_opinion = {
						target = scope:puppet_or_actor
						modifier = spurned_clerical_counsel_opinion
					}
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = request_clerical_chastity_decline_actor_toast_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { is_ai = no }
				set_variable = {
					name = request_clerical_chastity_ai_cooldown
					years = 3
				}
			}
		}
	}

	ai_target_quick_trigger = {
		adult = yes
	}

	ai_frequency_by_tier = {
		barony = 0
		county = 200
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	ai_targets = {
		ai_recipients = rulers_in_clerical_region
	}

	ai_targets = {
		ai_recipients = vassals
		max = 5
	}

	ai_targets = {
		ai_recipients = peer_vassals
		max = 10
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 5
			scope:puppet_or_actor = {
				OR = {
					ai_zeal > 0
					faith = { trait_is_virtue = chaste }
					has_trait = zealous
				}
			}
		}
		modifier = {
			add = 35
			scope:puppet_or_actor = {
				has_trait = greedy
			}
			scope:recipient = {
				NOT = {
					any_child = {
						is_alive = yes
					}
				}
			}
		}
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					has_trait = adulterer
					has_trait = fornicator
				}
			}
		}
		modifier = {
			add = 35
			scope:recipient = {
				OR = {
					has_trait = lustful
					has_trait = rakish
				}
				any_child = {
					count >= 3
					is_alive = yes
				}
			}
		}
		modifier = {
			add = 30
			scope:recipient = {
				has_relation_rival = scope:puppet_or_actor
			}
		}
		modifier = {
			add = 25
			recipient_in_actor_clerical_region_trigger = yes
		}
	}

	ai_accept = {
		base = -25

		# Warcraft
		master_thrall_modifier = yes
		# End Warcraft

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:puppet_or_actor
			RECIPIENT = scope:recipient
		}

		modifier = {
			add = 25
			scope:recipient = {
				has_trait = zealous
			}
			desc = TAKE_THE_VOWS_ZEALOUS
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_value_1
			desc = ESTATE_CLERICAL_CHASTITY_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_easier_clerical_chastity_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_value_2
			desc = ESTATE_CLERICAL_CHASTITY_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_easier_clerical_chastity_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_value_3
			desc = ESTATE_CLERICAL_CHASTITY_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_easier_clerical_chastity_3
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_1
			desc = ESTATE_VISITATION_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_chastity_visitation_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_2
			desc = ESTATE_VISITATION_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_chastity_visitation_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_3
			desc = ESTATE_VISITATION_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_chastity_visitation_3
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_1
			desc = ESTATE_CHANCERY_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_chancery_court_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_2
			desc = ESTATE_CHANCERY_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_chancery_court_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_3
			desc = ESTATE_CHANCERY_REASON
			scope:puppet_or_actor ?= {
				domicile ?= {
					has_domicile_parameter = estate_chancery_court_3
				}
			}
		}
		ecclesiastical_domicile_cathedral_awe_modifier = yes
		modifier = {
			add = -35
			scope:recipient = {
				has_trait = lustful
			}
			desc = TAKE_THE_VOWS_LUSTFUL
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:puppet_or_actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = -35
			scope:recipient = {
				has_relation_rival = scope:puppet_or_actor
			}
			desc = clerical_chastity_ai_rival
		}
		modifier = {
			add = 35
			recipient_in_actor_clerical_region_trigger = yes
			desc = clerical_chastity_ai_clerical_region
		}
		modifier = {
			add = 25
			scope:clerical_chastity_piety_stipend = yes
			desc = clerical_chastity_ai_piety_stipend
		}
		modifier = {
			add = -20
			scope:recipient = {
				highest_held_title_tier = tier_kingdom
			}
			desc = clerical_chastity_ai_tier_rank
		}
		modifier = {
			add = -30
			scope:recipient = {
				highest_held_title_tier = tier_empire
			}
			desc = clerical_chastity_ai_tier_rank
		}
		modifier = {
			add = -50
			scope:recipient = {
				highest_held_title_tier = tier_hegemony
			}
			desc = clerical_chastity_ai_tier_rank
		}
	}
