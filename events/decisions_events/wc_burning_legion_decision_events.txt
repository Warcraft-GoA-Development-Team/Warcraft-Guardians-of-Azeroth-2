namespace = burninglegion

#### burninglegion.0001 -> Give Arthas Dalaran
#### burninglegion.0002 -> If Dalaran is held by Antonidas, declare war
#### burninglegion.0003 -> Kill Antonidas and take Book of Medivh
#### burninglegion.0004 -> Kelthuzad sucessfully brings the Burning Legion to Azeroth

burninglegion.0001 = {
    hidden = yes

    immediate = {
        hidden_effect = {
            create_title_and_vassal_change = {
                type = granted
                save_scope_as = change
                add_claim_on_loss = no
            }
            title:c_dalaran = {
                change_title_holder = {
                    holder = root
                    change = scope:change
                }
            }
            resolve_title_and_vassal_change = scope:change
        }

        trigger_event = {
            id = burninglegion.0003
            days = 1
        }
    }
}

burninglegion.0002 = {
    hidden = yes

    immediate = {
        start_war = {
			casus_belli = invade_dalaran_cb
			target = title:c_dalaran.holder
			target_title = title:c_dalaran
		}
        arthas_scourge_spawn_troops_effect = yes
    }
}

burninglegion.0003 = {
    type = character_event
    content_source = dlc_GOA
    title = burninglegion_0003_title
    desc = burninglegion_0003_desc
    theme = death
    override_background = { reference = sm_dalaran_background }

    immediate = {
        hidden_effect = {
            if = {
                limit = { any_living_character = { is_antonidas_trigger = yes } }
                character:59000 = {
                    save_scope_as = antonidas
                }
            }
            else = {
                title:k_dalaran.holder = { save_scope_as = antonidas }
            }
        }

        random_courtier = {
            limit = { has_trait = creature_lich }
            save_scope_as = summonerofburninglegion
        }
    }

    left_portrait = {
		character = scope:antonidas
		animation = pain
	}
    right_portrait = {
		character = root
		animation = duel_wield_weapon
	}
    lower_center_portrait = {
        character = scope:summonerofburninglegion
    }

    option = {
        name = burninglegion_0003_a
        if = { # If that character is somehow not dead, kill em now
            limit = {
                scope:antonidas = {
                    is_alive = yes
                }
            }
            scope:antonidas = {
                override_death_killer_effect = {
                    death_reason = death_battle
                    killer = root
                }
            }
        }

        hidden_effect = {
            create_title_and_vassal_change = {
                type = conquest
                save_scope_as = change
                add_claim_on_loss = no
            }
            title:c_dalaran = {
                change_title_holder = {
                    holder = root
                    change = scope:change
                }
            }
            resolve_title_and_vassal_change = scope:change
        }

        create_artifact_book_of_medivh_effect = {
            OWNER = scope:summonerofburninglegion
        }

        custom_tooltip = {
            text = burninglegion_0003_a_tt
            trigger_event = {
                id = burninglegion.0004
                days = 3
            }
        }
    }
}
