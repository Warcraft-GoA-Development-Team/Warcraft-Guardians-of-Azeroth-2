wc_titanforged_ordering_duchy_cb = {
	icon = duchy_conquest_cb
	group = conquest
	combine_into_one = no
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }
	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150
	allowed_for_character = {
		wc_titanforged_implantation_ruler_trigger = yes
		is_independent_ruler = yes
		highest_held_title_tier >= tier_duchy

	}
	allowed_against_character = {
		scope:attacker = { NOT = { top_liege = scope:defender.top_liege } }
	}
	target_titles = neighbor_land
	target_title_tier = duchy
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder
	ai_score_mult = {
		value = viking_conquest_ai_score_value
		add = house_relation_ai_score_value
		multiply = struggle_wars_prioritise_struggle_targets_value
	}
	valid_to_start = {
		scope:attacker = {
			piety >= 500
			trigger_if = {
				limit = { scope:defender.faith = { is_chaos_aligned_faith_trigger = no } }
				prestige >= 500
			}
		}
		scope:target = {
			is_landless_type_title = no
			any_de_jure_county = {
				wc_titanforged_implantation_county_trigger = yes
				holder = { OR = { this = scope:defender is_vassal_or_below_of = scope:defender } }
			}
			any_de_jure_county = {
				holder = { OR = { this = scope:defender is_vassal_or_below_of = scope:defender } }
				any_neighboring_county = { holder = { OR = { this = scope:attacker is_vassal_or_below_of = scope:attacker } } }
			}
		}
	}
	should_invalidate = {
		NOT = {
			any_in_list = {
				list = target_titles
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = {
						target_is_same_character_or_above = scope:defender
					}
				}
			}
		}
	}
	on_invalidated_desc = msg_religious_war_invalidation_region_message
	on_invalidated = {}
	cost = {
		piety = { value = 500 }
		prestige = {
			value = 0
			if = { limit = { scope:defender.faith = { is_chaos_aligned_faith_trigger = no } } add = 500 }
		}
	}
	on_declaration = {
		on_declared_war = yes
	}
	on_victory_desc = wc_titanforged_ordering_on_victory_desc_tt
	on_victory = { wc_titanforged_ordering_victory_effect = yes }
	on_white_peace_desc = wc_titanforged_ordering_on_white_peace_desc_tt
	on_white_peace = { wc_titanforged_ordering_white_peace_effect = yes }
	on_defeat_desc = wc_titanforged_ordering_on_defeat_desc_tt
	on_defeat = { wc_titanforged_ordering_defeat_effect = yes }
	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = "WC_TITANFORGED_ORDERING_WAR_NAME"
	war_name_base = "WC_TITANFORGED_ORDERING_WAR_NAME_BASE"
	cb_name = "WC_TITANFORGED_ORDERING_DUCHY_CB_NAME"
	interface_priority = 80
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
