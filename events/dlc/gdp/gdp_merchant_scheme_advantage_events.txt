namespace = gdp_merchant_scheme_advantage

scripted_effect merchant_scheme_advantage_save_scopes_effect = {
	scope:scheme = {
		task_contract = {
			save_scope_as = task_contract
			task_contract_taker = { save_scope_as = merchant }
			task_contract_employer = {
				save_scope_as = trader
				assigned_trade_route ?= { save_scope_as = route }
			}
			task_contract_location = {
				save_scope_as = task_contract_location
				barony.trade_market = {
					save_scope_as = market
					every_character_in_market = {
						limit = {
							government_has_flag = government_is_merchant
							NOT = { this = scope:merchant }
						}
						add_to_list = potential_rivals
					}
				}
			}
			task_contract_target ?= { save_scope_as = task_contract_target }
		}
		every_scheme_agent_character = {
			limit = {
				NOT = { this = scope:trader }
			}
			add_to_list = potential_agents
		}
	}
	hidden_effect = {
		# Save portrait scope
		random_list = {
			10 = {
				trigger = {
					any_in_list = { list = potential_agents }
				}
				random_in_list = {
					list = potential_agents
					limit = {
						is_available_quick = {
							imprisoned = no
							incapable = no
						}
						NOT = { this = scope:trader }
					}
					save_scope_as = agent
					save_scope_as = portrait
				}
			}
			10 = {
				trigger = {
					court_position:partner_court_position ?= {
						is_available_quick = {
							imprisoned = no
							incapable = no
						}
						NOT = { this = scope:trader }
					}
				}
				court_position:partner_court_position = {
					save_scope_as = partner
					save_scope_as = portrait
				}
			}
			30 = {
				scope:trader = { save_scope_as = portrait }
			}
		}
		# Save rival scope
		random_in_list = {
			list = potential_rivals
			save_scope_as = rival
		}
		# Save malus
		random_list = {
			10 = {
				modifier = {
					add = -10
					scope:scheme = { has_scheme_modifier = gdp_merchant_scheme_advantage_success_chance_potential_modifier }
				}
				save_scope_value_as = { name = malus value = flag:potential }
			}
			10 = {
				modifier = {
					add = -10
					scope:scheme = { has_scheme_modifier = gdp_merchant_scheme_advantage_success_chance_growth_modifier }
				}
				save_scope_value_as = { name = malus value = flag:growth }
			}
			10 = {
				modifier = {
					add = -10
					scope:scheme = { has_scheme_modifier = gdp_merchant_scheme_advantage_success_chance_length_modifier }
				}
				save_scope_value_as = { name = malus value = flag:length }
			}
			10 = {
				save_scope_value_as = { name = malus value = flag:opportunity }
			}
			10 = {
				trigger = {	NOT = { scope:portrait = scope:trader } }
				save_scope_value_as = { name = malus value = flag:quit }
			}
			10 = {
				trigger = { NOT = { scope:task_contract = { has_task_contract_type = promote_goods } } }
				save_scope_value_as = { name = malus value = flag:competitiveness }
			}
			10 = {
				trigger = {
					exists = scope:rival.house
					NOT = { scope:merchant.house ?= scope:rival.house }
				}
				save_scope_value_as = { name = malus value = flag:relation }
			}
		}
		# Save duel type
		random_list = {
			10 = {
				trigger = { house.house_unity_value >= 33 }
				save_scope_value_as = { name = duel_type value = flag:unity }
			}
			10 = {
				save_scope_value_as = { name = duel_type value = flag:stress }
			}
			10 = {
				save_scope_value_as = { name = duel_type value = flag:associate }
			}
		}
		# Save spend type
		random_list = {
			10 = {
				trigger = {
					house.house_unity_value >= 5
					NOT = { scope:duel_type = flag:unity }
				}
				save_scope_value_as = { name = spend_type value = flag:unity }
			}
			10 = {
				trigger = { NOT = { scope:duel_type = flag:stress } }
				save_scope_value_as = { name = spend_type value = flag:stress }
			}
			10 = {
				trigger = { NOT = { scope:duel_type = flag:associate } }
				save_scope_value_as = { name = spend_type value = flag:associate }
			}
		}
	}
}

scripted_effect merchant_scheme_advantage_upset_outcome_effect = {
	save_scope_value_as = { name = opinion value = $OPINION_LOSS$ }
	save_scope_value_as = { name = stress value = $STRESS_GAIN$ }
	save_scope_value_as = { name = unity value = $UNITY_LOSS$ }
	if = {
		limit = { scope:opinion != 0 }
		scope:portrait = {
			add_opinion = {
				target = root
				modifier = upset_opinion
				opinion = scope:opinion
			}
		}
	}
	scope:portrait = {
		if = {
			limit = { scope:stress != 0 }
			stress_and_fulfillment_impact = { base = scope:stress }
		}
	}
	scope:merchant.house = {
		if = {
			limit = { scope:unity != 0 }
			add_unity_value = {
				value = scope:unity
				character = root
				desc = gdp_merchant_scheme_advantage.desc
			}
		}
	}
}

scripted_effect gdp_merchant_scheme_advantage_add_opportunity_effect = {
	scope:merchant = {
		send_interface_toast = {
			type = event_toast_effect_good
			title = gdp_merchant_scheme_advantage_outcome
			right_icon = scope:trader
			custom_tooltip = {
				text = gdp_merchant_scheme_advantage_add_2_opportunities_tt
				scope:scheme = { change_opportunities = 2 }
			}
		}
	}
}

scripted_effect gdp_merchant_scheme_advantage_malus_outcome_effect = {
	scope:merchant = {
		send_interface_toast = {
			type = event_toast_effect_bad
			title = gdp_merchant_scheme_advantage_outcome
			switch = {
				trigger = scope:malus
				flag:potential = {
					scope:scheme = {
						add_scheme_modifier = { type = gdp_merchant_scheme_advantage_success_chance_potential_modifier }
					}
				}
				flag:growth = {
					scope:scheme = {
						add_scheme_modifier = { type = gdp_merchant_scheme_advantage_success_chance_growth_modifier }
					}
				}
				flag:length = {
					scope:scheme = {
						add_scheme_modifier = { type = gdp_merchant_scheme_advantage_success_chance_length_modifier }
					}
				}
				flag:opportunity = {
					custom_tooltip = {
						text = gdp_merchant_scheme_advantage_subtract_opportunity_tt
						scope:scheme = { change_opportunities = -1 }
					}
				}
				flag:quit = {
					scope:portrait = { move_to_pool = yes }
				}
				flag:competitiveness = {
					scope:market = {
						add_competitiveness_modifier_for_character = {
							character = scope:merchant
							modifier = gdp_merchant_scheme_advantage_market_competitiveness_modifier
						}
					}
				}
				flag:relation = {
					scope:merchant.house = {
						change_house_relation_effect = {
							HOUSE = scope:rival.house
							VALUE = house_relation_damage_minor_value 
							REASON = scheme_push_for_advantage
							CHAR = scope:merchant
							TARGET_CHAR = scope:rival
							TITLE = dummy_female
						}
					}
				}
			}
		}
	}
}
