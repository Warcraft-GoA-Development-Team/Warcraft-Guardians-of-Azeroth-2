# Raw effect {} blocks for the same on_action don't merge across files - only the most-recently-loaded survives.
# Use on_actions = { list } to accumulate instead. A nathrezim liege auto-raises newly-gained
# vassals as undead thralls - a general mechanic, not specific to any one character's arc.
# being_demon alone is too broad (annihilan/eredruin also carry it, per history/characters/) -
# raising the dead is a dreadlord thing, not shared by every demon race in this mod.
on_vassal_gained = {
	on_actions = {
		wc_nathrezim_on_vassal_gained_actions
	}
}

wc_nathrezim_on_vassal_gained_actions = {
	effect = {
		if = {
			limit = {
				root = { has_trait = creature_nathrezim }
				NOT = { scope:vassal = { has_trait = being_undead } }
			}
			scope:vassal = {
				raise_undead_faith_only_effect = { MASTER = root }
				add_opinion = { modifier = loyal_undead target = root }
				# Use _no_notification variant: become_undead_effect leaks memory, can't cascade on entire vassal trees.
				every_vassal_or_below = {
					limit = {
						is_alive = yes
						NOT = { has_trait = being_undead }
					}
					raise_undead_faith_only_no_notification_effect = { MASTER = liege }
					add_opinion = { modifier = loyal_undead target = liege }
				}
			}
		}
	}
}
