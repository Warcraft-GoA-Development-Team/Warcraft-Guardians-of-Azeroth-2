init_disorder_spells_effect = {
    add_to_global_variable_list = { name = disorder_spells target = flag:subjugate }
    add_to_global_variable_list = { name = disorder_spells target = flag:summon }
}

cast_subjugate_effect = {
    if = {
        limit = {
            NOT = { exists = scope:cancel }
        }
        get_recipient_effect = { SPELL = subjugate }

        scope:this_recipient ?= {
            set_relation_master = root
        }
    }
    else = {
        var:subjugate_recipient = {
            if = {
                limit = {
                    has_relation_master = root
                }
                remove_relation_master = root
            }
        }
    }
}

cast_summon_effect = {
    if = {
        limit = {
            NOT = { exists = scope:cancel }
        }
        get_recipient_effect = { SPELL = summon }

        custom_tooltip = cast_summon_effect_tt

        if = {
            limit = {
                has_variable = summoning_demon
            }
            switch = {
                trigger = var:summoning_demon

                flag:imp = {
                    add_character_modifier = wc_imp_summon_modifier

                    create_character = {
                        # culture = culture:imp
                        culture = culture:sayaadi
                        faith = faith:burning_legion_religion
                        employer = root
                        trait = creature_sayaadi
                        # trait = creature_imp
                        trait = being_demon
                        gender_female_chance = root_faith_dominant_gender_adjusted_female_chance
                        dynasty = none
                        random_traits = yes

                        save_scope_as = demon
                    }
                }
                flag:voidwalker = {
                    add_character_modifier = wc_voidwalker_summon_modifier

                    create_character = {
                        culture = culture:sayaadi
                        # culture = culture:voidwalker
                        random_faith = {
                            faith:cthunism = {}
                            faith:yoggsaronism = {}
                            faith:nzothism = {}
                            faith:yshaarjism = {}
                            faith:ghuunism = {}
                        }
                        employer = root
                        trait = creature_sayaadi
                        # trait = creature_voidwalker
                        trait = being_void
                        gender_female_chance = root_faith_dominant_gender_adjusted_female_chance
                        dynasty = none
                        random_traits = yes

                        save_scope_as = demon
                    }
                }
                flag:sayaad = {
                    add_character_modifier = wc_sayaad_summon_modifier

                    create_character = {
                        culture = culture:sayaadi
                        faith = faith:burning_legion_religion
                        employer = root
                        trait = creature_sayaadi
                        trait = being_demon
                        gender = female
                        dynasty = none
                        random_traits = yes

                        save_scope_as = demon
                    }
                }
                flag:felhound = {
                    add_character_modifier = wc_felhound_summon_modifier

                    create_character = {
                        culture = culture:sayaadi
                        # culture = culture:felhound
                        faith = faith:burning_legion_religion
                        employer = root
                        trait = creature_sayaadi
                        # trait = creature_felhound
                        trait = being_demon
                        gender_female_chance = root_faith_dominant_gender_adjusted_female_chance
                        dynasty = none
                        random_traits = yes

                        save_scope_as = demon
                    }
                }
                flag:darkglare = {
                    add_character_modifier = wc_darkglare_summon_modifier

                    create_character = {
                        culture = culture:observer
                        faith = faith:burning_legion_religion
                        employer = root
                        # trait = creature_sayaadi
                        trait = creature_observer
                        trait = being_demon
                        gender_female_chance = root_faith_dominant_gender_adjusted_female_chance
                        dynasty = none
                        random_traits = yes

                        save_scope_as = demon
                    }
                }
            }

            scope:demon = {
                set_relation_summoner = root
            }

            remove_variable = summoning_demon

            trigger_event = {
                on_action = wc_summon_demon_pulse
                months = 1
            }
        }
    }
    else = {
        remove_modifier_if_present = { TYPE = character MODIFIER = wc_imp_summon_modifier }
        remove_modifier_if_present = { TYPE = character MODIFIER = wc_voidwalker_summon_modifier }
        remove_modifier_if_present = { TYPE = character MODIFIER = wc_sayaad_summon_modifier }
        remove_modifier_if_present = { TYPE = character MODIFIER = wc_felhound_summon_modifier }
        remove_modifier_if_present = { TYPE = character MODIFIER = wc_darkglare_summon_modifier }

        every_relation = {
            type = summon
            limit = {
                is_alive = yes
                NOT = {
                    has_character_flag = dispelled
                }
            }
            add_character_flag = dispelled
            override_death_effect = { death_reason = death_vanished }
        }
    }
}

sayaad_seduction_effect = {
    random_$TYPE$ = {
        limit = {
            is_attracted_to_women = yes
            NOT = {
                has_opinion_modifier = {
                    modifier = magic_charmed_opinion
                    target = root
                }
            }
        }
        add_opinion = {
            modifier = magic_charmed_opinion
            target = root
            opinion = $OPINION$
        }
    }
}

felhound_secret_effect = {
    random_$TYPE$ = {
        limit = {
            any_secret = {
                is_magic_secret_trigger = yes
                NOT = {
                    is_known_by = root
                }
            }
        }
        random_secret = {
            limit = {
                is_magic_secret_trigger = yes
                NOT = {
                    is_known_by = root
                }
            }
            expose_secret = root
        }
    }
}

darkglare_observation_effect = {
    var:casted_target = {
        if = {
            limit = {
                0 < {
                    add = $TYPE$_magic_lifestyle_perk_points
                    add = $TYPE$_magic_lifestyle_perks
                }
            }

            add_$TYPE$_magic_lifestyle_xp = 50
        }
        else = {
            switch = {
                trigger = has_lifestyle

                light_magic_lifestyle = { add_light_magic_lifestyle_xp = 25 }
                shadow_magic_lifestyle = { add_shadow_magic_lifestyle_xp = 25 }
                disorder_magic_lifestyle = { add_disorder_magic_lifestyle_xp = 25 }
                order_magic_lifestyle = { add_order_magic_lifestyle_xp = 25 }
                life_magic_lifestyle = { add_life_magic_lifestyle_xp = 25 }
                death_magic_lifestyle = { add_death_magic_lifestyle_xp = 25 }
                elemental_fire_magic_lifestyle = { add_elemental_fire_magic_lifestyle_xp = 25 }
                elemental_water_magic_lifestyle = { add_elemental_water_magic_lifestyle_xp = 25 }
                elemental_air_magic_lifestyle = { add_elemental_air_magic_lifestyle_xp = 25 }
                elemental_earth_magic_lifestyle = { add_elemental_earth_magic_lifestyle_xp = 25 }
            }
        }
    }
}
