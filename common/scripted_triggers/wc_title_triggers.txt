is_arathor_emperor_trigger = {
	OR = {
		has_title = title:e_arathor
		has_title = title:h_arathor
	}
}

is_arathor_emperor_primary_title_trigger = {
	OR = {
		primary_title ?= title:e_arathor
		primary_title ?= title:h_arathor
	}
}

is_orcish_clan_title_trigger = { 
	OR = { 
		this = title:d_stormreaver_clan
		this = title:k_shattered_hand_clan
		this = title:k_bleeding_hollow_clan
		this = title:d_twilights_hammer_clan
		this = title:k_frostwolf_clan
		this = title:d_dragonmaw_clan
		this = title:d_burning_blade_clan
		this = title:d_blacktoothgrin_clan
		this = title:k_warsong_clan
		this = title:d_laughing_skull_clan
		this = title:d_hammerfall
	}
}
