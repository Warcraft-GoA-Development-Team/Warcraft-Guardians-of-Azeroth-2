# Spawns the Scourge, Ner'zhul and its dreadlords
WCCSC.01 = {
	scope = none
	hidden = yes

	immediate = {
		debug_log = "The Scourge has appeared!"
		debug_log_date = yes
		if = {
			limit = {
				character:10014 = { 
					is_alive = yes 
				}
			}
			character:10014 = { 
				add_trait = being_undead
				remove_trait = creature_orc
				add_trait = creature_lich
				add_trait = lich_king
				add_trait = in_ice_prison
				remove_trait = education_learning_4
				add_trait = education_learning_5
				set_location = title:c_icecrown.title_province
				set_culture = culture:scourge
				set_character_faith = faith:death_god
				save_scope_as = nerzhul
				create_story = story_scourge_invasion
			}
		}
		else = {
			create_character = {
				save_scope_as =  nerzhul
				location = title:c_icecrown.title_province
				template = lich_nerzhul_character_template
				
				name = "Ner'zhul"
				after_creation = { 
					create_story = story_scourge_invasion
				}
			}
		}
		scope:nerzhul = {
			create_title_and_vassal_change = {
				type = created
				save_scope_as = created
				add_claim_on_loss = no
			}
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = conquest
				add_claim_on_loss = yes
			}
			title:e_scourge = {
				change_title_holder = {
					holder = scope:nerzhul
					change = scope:created
				}
			}
			title:d_death_god = {
				change_title_holder = {
					holder = scope:nerzhul
					change = scope:created
				}
			}
			title:c_icecrown = {
				change_title_holder = {
					holder = scope:nerzhul
					change = scope:conquest
					take_baronies = yes
				}
			}
			resolve_title_and_vassal_change = scope:conquest
			resolve_title_and_vassal_change = scope:created
			
			# Delayed army spawn
			trigger_event = WCCSC.2
			
			every_ruler = {
				limit = { get_news_from_region_trigger = { REGION = world_northrend } }
				trigger_event = WCCSC.5
			}
		}
	}
}

# Notification: the Scourge appears
WCCSC.5 = {
	type = character_event
	content_source = dlc_GOA
	title = WCCSC_5_TITLE
	desc = WCCSC_5_DESC
	theme = war
	override_background = {
		reference = wc_background_northrend
	}
	window = fullscreen_event
	
	option = {
		name = {
			text = {
				first_valid = {
					# All are equal in death
					triggered_desc = {
						trigger = { faith = faith:orcish_fel }
						desc = WCCSC_5_OPT_A_death
					}
					# Better serve the Legion well, Ner'zhul
					triggered_desc = {
						trigger = { faith = faith:orcish_fel }
						desc = WCCSC_5_OPT_A_legion
					}
					# Death comes into play
					triggered_desc = { # Corrupted orcs option
						trigger = { faith = faith:orcish_fel }
						desc = WCCSC_5_OPT_A_evil
					}
					# God save us all!
					desc = WCCSC_5_OPT_A_neutral
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
	option = {
		name = WCCSC_5_OPT_B
		trigger = {
			can_be_in_coalition_trigger = yes
			NOT = { is_in_coalition_against_trigger = { CRISIS_HOLDER = title:e_scourge.holder } }
		}
		join_crisis_coalition_effect = { CRISIS_HOLDER = title:e_scourge.holder }
		ai_chance = {
			base = 0
			join_coalition_scripted_modifier = { CRISIS_HOLDER = title:e_scourge.holder  }
		}
	}
}

# Dreadlord jailors comes to finish you
WCCSC.15 = {
	type = character_event
	content_source = dlc_GOA
	title = WCCSC_15_TITLE
	desc = WCCSC_15_DESC
	theme = death
	override_background = {
		reference = wc_background_northrend
	}
	left_portrait = {
		character = root
		animation = fear
	}
	right_portrait = {
		character = scope:scourge_jailor
		animation = rage
	}
	
	trigger = {
		any_scourge_jailor_is_avaliable_trigger = yes
		NOT = {
			any_secret = {
				secret_target = scope:hurt_jailor
				secret_type = secret_murder
			}
		}
	}
	
	immediate = {
		find_scourge_jailor_and_save_to_scope_effect = yes
	}

	option = {
		name = RULKAN_OPT
		
		end_scourge_effect = yes
		
		death = { death_reason = death_execution killer = scope:scourge_jailor }
	}
}
