caster_attribute_trigger = {
    prowess > 15
    OR = {
        learning > 20
        has_trait = magic_good
        has_trait = open_terrain_expert
        has_trait = lifestyle_mystic
    }
    custom_tooltip = {
		text = single_maa_attribute_allowance
		trigger_if = {
			limit = {
				exists = accolade
			}
			accolade = {
				NOT = { has_accolade_category = men_at_arms }
			}
		}
	}
}
caster_attribute_acceptable_trigger = {
    prowess > 10
    OR = {
        learning > 15
        has_trait = magic_good
        has_trait = open_terrain_expert
        has_trait = lifestyle_mystic
    }
}

massive_infantry_attribute_trigger = {
    prowess > 20
    OR = {
        has_trait = strong
        has_trait = athletic
        has_trait = physique_good
        has_trait_xp = {
            trait = tourney_participant
            track = foot
            value >= 40
        }
        has_trait = rough_terrain_expert
        has_dire_or_giant_trigger = yes
    }
    culture ?= {
        OR = {
            has_innovation = innovation_giants
            has_innovation = innovation_troll_tactics
            has_innovation = innovation_drakonid
