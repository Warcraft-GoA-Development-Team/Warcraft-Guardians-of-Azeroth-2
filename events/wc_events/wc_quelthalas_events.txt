# M'uru arrives from Outland
wc_quelthalas_events.105 = {
    type = character_event
	content_source = dlc_GOA
	title = wc_quelthalas_events.105.t
	desc = wc_quelthalas_events.105.desc
	theme = emperor
	override_background = {
		reference = wc_background_silvermoon
	}

	immediate = {
		# Resolve the persistent M'uru reference first so this event can never
		# create a second M'uru for an existing campaign save.
		get_muru_character_effect = yes
		get_rommath_character_effect = yes
		if = {
			limit = { character:42077 = { is_alive = yes } }
			character:42077 = { save_scope_as = halduron_character }
		}
	}

    left_portrait = {
        character = root
		animation = interested
    }
	
    right_portrait = {
        character = scope:muru_character
        animation = fear 
    }

	# Turn M'uru over to Rommath and the Magisterium.
    option = {
        name = wc_quelthalas_events.105.a
		scope:rommath_character = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}
		}
		scope:halduron_character ?= {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -15
			}
		}
		stress_impact = {
			compassionate = minor_stress_impact_gain
			just = minor_stress_impact_gain
			callous = minor_stress_impact_loss
			ambitious = minor_stress_impact_loss
		}
		trigger_event = {
			id = wc_quelthalas_events.106
			days = { 20 35 }
		}
		ai_chance = {
			base = 120
			ai_value_modifier = {
				ai_rationality = 1
				ai_compassion = -1
			}
			stress_impact_ai_chance_modifier = { TRAIT = compassionate STRESS = minor_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = just STRESS = minor_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = callous STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = ambitious STRESS = minor_stress_impact_loss }
		}
	}

	# Keep the captive under guard while its nature is studied.
	option = {
		name = wc_quelthalas_events.105.b
		trigger = { gold >= 50 }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = 50
		scope:rommath_character = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -5
			}
		}
		scope:halduron_character ?= {
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 10
			}
		}
		stress_impact = {
			paranoid = minor_stress_impact_loss
		}
		trigger_event = {
			id = wc_quelthalas_events.106
			days = { 45 70 }
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_rationality = 1
				ai_boldness = -0.5
				ai_greed = -0.5
			}
			stress_impact_ai_chance_modifier = { TRAIT = paranoid STRESS = minor_stress_impact_loss }
		}
	}

	# Treat M'uru as a sentient captive rather than a resource.
	option = {
		name = wc_quelthalas_events.105.c
		add_prestige = 50
		scope:rommath_character = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -15
			}
		}
		scope:halduron_character ?= {
			add_opinion = {
				target = root
				modifier = grateful_opinion

			}
		}

			sadistic = medium_stress_impact_gain

		get_sunblade_leader_effect = yes

	}

		character = scope:sunblade_leader
		animation = war_attacker
	}

	option = {
		name = wc_quelthalas_events.404.a

		if = {
			limit = {
				NOT = { exists = global_var:sunwell_phase3_war_started }
				exists = scope:sunblade_leader
				exists = scope:sunblade_war_defender
				title:c_sunwell_plateau.holder.top_liege = scope:sunblade_war_defender
				scope:sunblade_leader = {
					is_alive = yes
					is_imprisoned = no
					OR = {
						is_landed = yes
						any_held_title = {
							has_variable = sunwell_phase3_temporary_expedition_title
						}
					}
					top_liege = this
					NOT = { is_at_war_with = scope:sunblade_war_defender }
					NOT = { this = scope:sunblade_war_defender }
				}
			}
			scope:sunblade_leader = {
				start_war = {
					casus_belli = sunblade_queldanas_cb
					target = scope:sunblade_war_defender
					target_title = title:c_sunwell_plateau
				}
			}
			if = {
				limit = {
					scope:sunblade_leader = {
						is_at_war_with = scope:sunblade_war_defender
					}
				}
				set_global_variable = sunwell_phase3_war_started
				scope:sunblade_leader = {
					spawn_sunblade_event_army_effect = yes
				}
			}
			else = {
				set_global_variable = sunwell_phase3_war_start_failed
				set_global_variable = sunwell_phase3_war_resolved
				custom_tooltip = sunblade_war_start_failed_tt
				scope:sunblade_leader = {
					cleanup_sunblade_event_attacker_effect = yes
				}
				global_var:sunwell_phase3_ruler = {
					trigger_event = { id = wc_quelthalas_events.405 }
				}
			}
		}
		else = {
			if = {
				limit = { NOT = { exists = global_var:sunwell_phase3_war_started } }
				set_global_variable = sunwell_phase3_war_start_failed
				set_global_variable = sunwell_phase3_war_resolved
				custom_tooltip = sunblade_war_start_failed_tt
				if = {
					limit = { exists = scope:sunblade_leader }
					scope:sunblade_leader = {
						cleanup_sunblade_event_attacker_effect = yes
					}
				}
				global_var:sunwell_phase3_ruler = {
					trigger_event = { id = wc_quelthalas_events.405 }
				}
			}
		}
	}
}
