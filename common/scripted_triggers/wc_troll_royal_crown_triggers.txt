wc_troll_royal_crown_trigger = {
    exists = this
    has_trait = creature_troll
    has_no_portrait_trigger = no
    peasant_clothing_trigger = no
    portrait_can_use_noble_headgear_trigger = yes
    portrait_wear_no_crown_trigger = no
    NOR = {
        highest_held_title_tier >= tier_empire
        AND = {
            takes_spouse_clothing_trigger = yes
            primary_spouse ?= { highest_held_title_tier >= tier_empire }
        }
    }
    OR = {
        king_clothing_trigger = yes
        AND = {
            takes_spouse_clothing_trigger = yes
            primary_spouse ?= { king_clothing_trigger = yes }
        }
        AND = {
            is_diarch = yes
            liege ?= {
                highest_held_title_tier = tier_kingdom
                has_diarchy_active_parameter = diarchy_is_co_rulership
                king_clothing_trigger = yes
            }
        }
    }
}
