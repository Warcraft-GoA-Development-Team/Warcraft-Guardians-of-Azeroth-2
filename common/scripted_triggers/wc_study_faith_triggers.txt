# Study Faith adaptation (issue #1969)
# Index:
#   Hidden cult reveal (study_faith_ongoing.1011 / 1012): wc_study_faith_has_hidden_cult_trigger and its rows
#   Families, archives, zealot and judge (wc_study_faith events 0100-0313) are further down

# Character scope. Mirrors the guard inside make_character_crypto_religionist_effect,
# which silently does nothing when this is false.
wc_study_faith_can_take_secret_faith_trigger = {
	NOT = { any_secret = { type = secret_crypto_religionist } }
	NOT = { exists = secret_faith }
}

# Teacher scope. CULT = faith:x, OWNER = the studier. A cult row is skipped when either side already follows it.
wc_study_faith_cult_usable_trigger = {
	NOT = { faith = $CULT$ }
	$OWNER$ = {
		NOT = { faith = $CULT$ }
	}
}

# Teacher scope. Row 1: Cult of the Damned, among Light faiths and human cultures
wc_study_faith_cult_damned_row_trigger = {
	OR = {
		religion = religion:tyrism_religion
		culture = { has_human_heritage_trigger = yes }
	}
}

# Teacher scope. Row 2: Burning Blade, among orcs
wc_study_faith_cult_burning_blade_row_trigger = {
	culture = { has_cultural_pillar = heritage_orcish }
}

# Teacher scope. Row 3: Atal'ai, among loa worshippers
wc_study_faith_cult_atalai_row_trigger = {
	religion = religion:loa_group
}

# Teacher scope. Row 4: Dark Iron, among dwarves
wc_study_faith_cult_dark_iron_row_trigger = {
	culture = { has_cultural_pillar = heritage_dwarven }
}

# Teacher scope. OWNER = the studier. True when wc_study_faith_pick_hidden_cult_effect will find a cult.
# Keep the rows in the same order as the effect.
wc_study_faith_has_hidden_cult_trigger = {
	wc_study_faith_can_take_secret_faith_trigger = yes
	$OWNER$ = {
		wc_study_faith_can_take_secret_faith_trigger = yes
	}

	OR = {
		AND = {
			wc_study_faith_cult_damned_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:death_god OWNER = $OWNER$ }
		}
		AND = {
			wc_study_faith_cult_burning_blade_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:orcish_fel OWNER = $OWNER$ }
		}
		AND = {
			wc_study_faith_cult_atalai_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:cult_of_hakkar OWNER = $OWNER$ }
		}
		AND = {
			wc_study_faith_cult_dark_iron_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:ragnarism OWNER = $OWNER$ }
		}
		wc_study_faith_cult_usable_trigger = { CULT = faith:twilights_hammer_cult OWNER = $OWNER$ }
	}
}

# Shared gate for the player-only paths fired from study_faith_ongoing (0100-0300)

# Studier scope. VAR = the scheme variable that marks "already happened in this scheme".
wc_study_faith_player_path_trigger = {
	exists = scope:scheme
	exists = scope:target
	scope:target = { is_alive = yes }

	is_ai = no

	scope:scheme = {
		NOT = { has_variable = $VAR$ }
	}
}

# Rejection and ruling (0300-0313)

# Character scope (a candidate). TEACHER = the teacher. Alive, follows the teacher's faith, and is neither the teacher nor root.
wc_study_faith_cleric_candidate_trigger = {
	is_alive = yes

	NOT = { this = $TEACHER$ }
	NOT = { this = root }

	faith = $TEACHER$.faith
}

# Character scope. A courtier zealot: a cleric candidate who is zealous or very pious.
wc_study_faith_zealot_candidate_trigger = {
	wc_study_faith_cleric_candidate_trigger = { TEACHER = $TEACHER$ }
	OR = {
		has_trait = zealous
		piety_level >= 2
	}
}

# Teacher scope. Same order as wc_study_faith_pick_zealot_effect.
wc_study_faith_has_zealot_trigger = {
	OR = {
		cp:councillor_court_chaplain ?= { wc_study_faith_cleric_candidate_trigger = { TEACHER = $TEACHER$ } }
		liege ?= {
			cp:councillor_court_chaplain ?= { wc_study_faith_cleric_candidate_trigger = { TEACHER = $TEACHER$ } }
		}
		any_courtier = { wc_study_faith_zealot_candidate_trigger = { TEACHER = $TEACHER$ } }
	}
}

# Character scope (a candidate). Needs scope:target and scope:zealot.
wc_study_faith_judge_candidate_trigger = {
	is_alive = yes

	NOR = {
		this = root
		this = scope:target
		this = scope:zealot
	}

	faith = scope:target.faith
}

# Teacher scope. Same order as wc_study_faith_pick_judge_effect.
wc_study_faith_has_judge_trigger = {
	OR = {
		faith.religious_head ?= { wc_study_faith_judge_candidate_trigger = yes }
		top_liege ?= {
			cp:councillor_court_chaplain ?= { wc_study_faith_judge_candidate_trigger = yes }
		}
		liege ?= {
			cp:councillor_court_chaplain ?= { wc_study_faith_judge_candidate_trigger = yes }
		}
	}
}
