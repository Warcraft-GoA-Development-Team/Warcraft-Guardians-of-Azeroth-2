set_story_cycle_path_of_the_damned_effect = {
	if = {
		limit = { NOT = { exists = global_var:path_of_the_damned_story } } # No duplication

		$STORY_OWNER$ = {
			# If Frostmourne don't exist, create it
			if = {
				limit = { frostmourne_can_be_found_trigger = yes }

				create_artifact_frostmourne_effect = { OWNER = $STORY_OWNER$}
				set_variable = { name = frostmourne value = scope:newly_created_artifact }
			}
			else_if = {
				limit = { NOT = { has_variable = frostmourne } }

				random_artifact = {
					limit = { has_variable = frostmourne }
					save_temporary_scope_as = existing_frostmourne
				}
				if = {
					limit = { exists = scope:existing_frostmourne }
					set_variable = { name = frostmourne value = scope:existing_frostmourne }
				}
			}

			create_story = story_path_of_the_damned
		}
	}
}

check_kelthuzad_never_died_effect = {
	if = {
		limit = {
			exists = var:necromancer
			OR = {
				var:necromancer = { # Never died 
					is_alive = yes
					NOT = { has_character_flag = kel_thuzad_lich }
				}
				var:necromancer = { # Already resurrected
					has_character_flag = kel_thuzad_lich
					OR = {
						NOT = { is_alive = yes }
						NOT = {
							OR = {
								liege = story_owner
								employer = story_owner
							}
						}
					}
				}
			}
		}

		var:necromancer = { save_scope_as = necromancer }
		story_owner = {
			trigger_event = {
				id = wc_path_of_the_damned.0102 # Skip to Dalaran Stage
				days = 20
			}
		}
	}
}

funeral_of_plagued_king_effect = {
	save_scope_as = plagued_king

	# Compile list of people who should come.
	scope:plagued_king = {
		every_close_or_extended_family_member = { add_to_temporary_list = funeral_attendees }
		every_vassal_or_below = { add_to_temporary_list = funeral_attendees }
		every_courtier = { add_to_temporary_list = funeral_attendees }
		title:k_order_of_the_silver_hand.holder = { add_to_temporary_list = funeral_attendees } # Uther or Highlord of the Silver Land
		every_relation = { type = lover add_to_temporary_list = funeral_attendees }
		every_relation = { type = friend add_to_temporary_list = funeral_attendees }
	}

	# We carry the flame
	ordered_in_list = {
		list = funeral_attendees

		limit = { is_landed = yes } # Attackable

		order_by = {
			if = {
				limit = { has_title = title:k_order_of_the_silver_hand }
				value = 2
			}
			else = {
				value = 1
			}
		}

		add_character_flag = { flag = carry_urn_of_the_king_of_lordaeron }
	}

	every_in_list = {
		list = funeral_attendees
		# have to save these now because it wont work after he dies
		if = {
			limit = { # Terenas had a good opinion of them or vv
				is_alive = yes 
				OR = {
					opinion = {
						value > 0
						target = scope:plagued_king 
					}
					scope:plagued_king  = {
						opinion = {
							value > 0
							target = prev
						}
					}
				}
			}
			add_character_flag = {
				flag = terenas_good
				days = 30
			}
		}

		trigger_event = {
			id = wc_path_of_the_damned.0002
			days = 14
		}
	}
}

resolve_lost_artifact_effect = {
	random_artifact = {
		limit = { has_variable = $ARTIFACT_VARIABLE$ }
		save_scope_as = lost_artifact
	}

	if = {
		limit = { exists = scope:lost_artifact }

		scope:lost_artifact = { artifact_owner = { save_scope_as = artifact_holder } }
		story_owner = { save_scope_as = champion }

		if = {
			limit = {
				exists = scope:artifact_holder
				scope:artifact_holder = { artifact_holder_is_loyal_to_lich_king_trigger = yes }
			}

			scope:lost_artifact = { set_owner = scope:champion }
		}
		else = {
			scope:champion = {
				send_interface_message = {
					type = $CLAIM_MSG$
					left_icon = scope:lost_artifact
					add_personal_artifact_claim = scope:lost_artifact
				}
			}
		}
	}
	else = { end_story = yes }
}

		every_relation = {
			type = scourge_jailor
			add_to_temporary_list = valid_tichondrius
		}
	}
	ordered_in_list = {
		limit = { is_ai = yes }
		list = valid_tichondrius
		order_by = find_tichondrius_value
		max = 1
			
		save_scope_as = tichondrius
	}
	if = {
		limit = { NOT = { exists = scope:tichondrius } }

		create_character = {
			template = tichondrius_character_template
			employer = scope:nerzhul
				
			after_creation = { save_scope_as = tichondrius }
		}
	}
}

battle_against_paladin_digging_up_the_dead_effect = {
	if = {
		limit = {
			scope:defender = {
				has_trait = order_member # TODO: Replace with paladin trait when class content is made
				NOT = { any_character_artifact = { this = global_var:path_of_the_damned_story.var:enchanted_urn } }
			}
		}

		scope:defender = { trigger_event = wc_path_of_the_damned.2000 }
	}
			
	if = {
		limit = {
			scope:defender = {
				any_character_artifact = { this = global_var:path_of_the_damned_story.var:enchanted_urn }
				is_at_war_with = scope:attacker
			}
		}
		scope:attacker = { post_digging_up_the_dead_effect = yes }
	}
}
