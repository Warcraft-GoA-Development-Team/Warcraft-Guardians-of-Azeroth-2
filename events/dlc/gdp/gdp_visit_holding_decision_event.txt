#Find common and masterwork artifacts to improve
scripted_effect 1111_set_artifact_effect = {
	every_character_artifact = {
		limit = {
			OR = {
				rarity = common
				rarity = masterwork
			}
			# Animals are living creatures and cannot be improved by the Curator
			NOT = { artifact_type = creature }
		}
		add_to_list = owned_artifacts
	}
	random_in_list = {
		list = owned_artifacts
		weight = {
			base = 1
			modifier = {
				add = 9
				rarity = common
			}
			modifier = {
				add = 5
				rarity = masterwork
			}
		}
		save_scope_as = artifact_target_1
	}
	random_in_list = {
		list = owned_artifacts
		limit = { this != scope:artifact_target_1 }
		weight = {
			base = 1
			modifier = {
				add = 9
				rarity = common
			}
			modifier = {
				add = 5
				rarity = masterwork
			}
		}
		save_scope_as = artifact_target_2
	}
	random_in_list = {
		list = owned_artifacts
		limit = {
			AND = {
				this != scope:artifact_target_1
				this != scope:artifact_target_2		 
			}
		}
		weight = {
			base = 1
			modifier = {
				add = 9
				rarity = common
			}
			modifier = {
				add = 5
				rarity = masterwork
			}
		}
		save_scope_as = artifact_target_3
	}			
}
#Discount if you are good at stewardship or learning
scripted_effect 1111_artifact_cost_effect = {
	save_scope_value_as = {
		name = $COST_NAME$
		value = {
			value = $GOLD_VALUE$
			if = {
				limit = {
					scope:liege = { stewardship >= average_skill_level }
				}
				subtract = {
					value = $GOLD_VALUE$
					multiply = 0.1
				}	
			}
			if = {
				limit = {
					scope:liege = { learning >= average_skill_level }
				}
				subtract = {
					value = $GOLD_VALUE$
					multiply = 0.1
				}	
			}
			if = {
				limit = {
					scope:liege = { learning >= good_skill_level }
				}
				subtract = {
					value = $GOLD_VALUE$
					multiply = 0.1
				}	
			}
		}
	}
}
scripted_effect 1111_artifact_value_effect = {
	#Expensive
	if = {
		limit = { rarity = masterwork }
		1111_artifact_cost_effect = { GOLD_VALUE = root.major_gold_value COST_NAME = $COST_NAME$ }
	}
	#OK price
	else_if = {
		limit = { rarity = common }
		1111_artifact_cost_effect = { GOLD_VALUE = root.medium_gold_value COST_NAME = $COST_NAME$ }
	}	
}
scripted_effect 1111_artifact_increase_rarity_effect = {
	switch = {
		trigger = rarity
		common = { set_artifact_rarity = masterwork }
		masterwork = { set_artifact_rarity = famed }
	}
}

gdp_visit_holding_decision_event.1111 = {
	type = character_event 
	title = gdp_visit_holding_decision_event.1111.t

	desc = gdp_visit_holding_decision_event.1111.desc
	
	theme = merchant
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_generic" }
	override_background = { reference = study }

	artifact = {
		target = scope:artifact_target_1
		position = lower_left_portrait
	}
	artifact = {
		target = scope:artifact_target_2
		position = lower_center_portrait
	}
	artifact = {
		target = scope:artifact_target_3
		position = lower_right_portrait
	}
	left_portrait = {
		character = root
		animation = chess_certain_win
		camera = camera_event_scheme_center_look_right
	}
	right_portrait = {
		character = scope:curator
		animation = eccentric
	}

	immediate = {
		1111_set_artifact_effect = yes
		root = { save_scope_as = liege }
		suzerain = { save_scope_as = patron }
		if = {
			limit = {
				NOT = { exists = scope:curator } 
		  	}
		 	create_character = {
		 		template = curator_template
		 		location = root.location
		        culture = root.location.culture
		        faith = root.location.faith
				save_scope_as = curator
				after_creation = { trigger_race_giving_no_gene_effect = yes } # Warcraft
		 	}
		}
		scope:curator = { save_scope_as = antiquarian_from_task }
		#Pre-compute each offered artifact's cost so options can be gated on available gold
		if = {
			limit = { exists = scope:artifact_target_1 }
			scope:artifact_target_1 = { 1111_artifact_value_effect = { COST_NAME = 1111_artifact_cost_1 } }
		}
		else = { save_scope_value_as = { name = 1111_artifact_cost_1 value = 0 } }
		if = {
			limit = { exists = scope:artifact_target_2 }
			scope:artifact_target_2 = { 1111_artifact_value_effect = { COST_NAME = 1111_artifact_cost_2 } }
		}
		else = { save_scope_value_as = { name = 1111_artifact_cost_2 value = 0 } }
		if = {
			limit = { exists = scope:artifact_target_3 }
			scope:artifact_target_3 = { 1111_artifact_value_effect = { COST_NAME = 1111_artifact_cost_3 } }
		}
		else = { save_scope_value_as = { name = 1111_artifact_cost_3 value = 0 } }
	}

	option = { 
		name = gdp_visit_holding_decision_event.1111.a
		trigger = {
			short_term_gold >= scope:1111_artifact_cost_1
		}
		show_as_unavailable = { short_term_gold < scope:1111_artifact_cost_1 }
		scope:artifact_target_1 = {
			save_scope_as = artifact_target
			1111_artifact_increase_rarity_effect = yes
			hidden_effect = { antiquarian_improve_artifact_effect = yes }
		}
		custom_tooltip = gdp_visit_holding_decision_event.1111.a.tt
		pay_short_term_gold = {
			target = scope:patron
			gold = scope:1111_artifact_cost_1
		}
		visit_holding_set_option_cooldown_guilds_effect = { VAR = 1111_has_improved_artifact_recently }
		trigger_event = gdp_visit_holding_decision_event.1110
	}

	option = { 
		name = gdp_visit_holding_decision_event.1111.b
		trigger = {
			exists = scope:artifact_target_2
			short_term_gold >= scope:1111_artifact_cost_2
		}
		show_as_unavailable = { exists = scope:artifact_target_2 short_term_gold < scope:1111_artifact_cost_2 }
		scope:artifact_target_2 = {
			save_scope_as = artifact_target
			1111_artifact_increase_rarity_effect = yes
			hidden_effect = { antiquarian_improve_artifact_effect = yes }
		}
		custom_tooltip = gdp_visit_holding_decision_event.1111.b.tt
		pay_short_term_gold = {
			target = scope:patron
			gold = scope:1111_artifact_cost_2
		}
		visit_holding_set_option_cooldown_guilds_effect = { VAR = 1111_has_improved_artifact_recently }
		trigger_event = gdp_visit_holding_decision_event.1110
	}

	option = { 
		name = gdp_visit_holding_decision_event.1111.c
		trigger = {
			exists = scope:artifact_target_3
			short_term_gold >= scope:1111_artifact_cost_3
		}
		show_as_unavailable = { exists = scope:artifact_target_3 short_term_gold < scope:1111_artifact_cost_3 }
		scope:artifact_target_3 = {
			save_scope_as = artifact_target
			1111_artifact_increase_rarity_effect = yes
			hidden_effect = { antiquarian_improve_artifact_effect = yes }
		}
		custom_tooltip = gdp_visit_holding_decision_event.1111.c.tt
		pay_short_term_gold = {
			target = scope:patron
			gold = scope:1111_artifact_cost_3
		}
		visit_holding_set_option_cooldown_guilds_effect = { VAR = 1111_has_improved_artifact_recently }
		trigger_event = gdp_visit_holding_decision_event.1110
	}
	#Go Back To Previous
	option = { 
		name = gdp_visit_holding_decision_event.1111.d
		custom_tooltip = gdp_visit_holding_decision_event.1111.d.tt
		trigger_event = gdp_visit_holding_decision_event.1110
	}

scripted_effect 1112_artifact_arms_armor_creation_effect = {
	if = {
		limit = {
			NOT = { exists = scope:arms_armor_artifact_1 }
		}
		1112_artifact_arms_armor_rarity_effect = yes
		#Generate weapon 
		create_artifact_weapon_effect = {
			OWNER = scope:blacksmith
			CREATOR = scope:blacksmith
			SET_WEAPON_TYPE = flag:no
		}
		#Finalization
		scope:newly_created_artifact ?= { save_scope_as = arms_armor_artifact_1 }
	}
	if = {
		limit = {
			NOT = { exists = scope:arms_armor_artifact_2 }
		}
		1112_artifact_arms_armor_rarity_effect = yes
		#Generate armor
		create_artifact_armor_effect = {
			OWNER = scope:blacksmith
			CREATOR = scope:blacksmith
			SET_ARMOR_TYPE = flag:no
		}
		#Finalization
		scope:newly_created_artifact ?= { save_scope_as = arms_armor_artifact_2 }
	}
	if = {
		limit = {
			NOT = { exists = scope:arms_armor_artifact_3 }
		}
		1112_artifact_arms_armor_rarity_effect = yes
		random_list = {
		#Generate weapon OR armor
			50 = { 
				create_artifact_weapon_effect = {
					OWNER = scope:blacksmith
					CREATOR = scope:blacksmith
					SET_WEAPON_TYPE = flag:no
				}
			}
			50 = { 
				create_artifact_armor_effect = {
					OWNER = scope:blacksmith
					CREATOR = scope:blacksmith
					SET_ARMOR_TYPE = flag:no
				}
			}
		}
		#Finalization
		scope:newly_created_artifact ?= { save_scope_as = arms_armor_artifact_3 }
	}
	scope:arms_armor_artifact_1 = { 1112_artifact_arms_armor_value_effect = yes }
	scope:arms_armor_artifact_2 = { 1112_artifact_arms_armor_value_effect = yes }
	scope:arms_armor_artifact_3 = { 1112_artifact_arms_armor_value_effect = yes }

scripted_effect visit_holding_try_spawn_monk_effect = {
	if = {
		limit = {
			NOT = { has_variable = visit_holding_checked_for_monk }
			NOT = { exists = scope:monk }
			NOT = { visit_holding_rare_spawn_already_seen_trigger = { SCENE = monk } }
		}
		random_list = {
			100 = {
				#Search for monk in pool
				if = {
					limit = {
						NOT = { exists = scope:monk }
					}
					random_pool_character = {
						province = scope:visiting_location
						limit = {
							is_available_healthy_adult = yes
							has_trait = devoted
							faith = scope:visiting_location.faith
							learning >= high_skill_rating
						}
						save_scope_as = monk
					}
				}
				#Otherwise create them
				if = {
					limit = {
						NOT = { exists = scope:monk }
					}
					create_character = {
						template = monk_character_template
						location = scope:visiting_location
						dynasty = none
						culture = scope:visiting_location.culture
						faith = scope:visiting_location.faith
						learning = { high_skill_rating very_high_skill_rating }
						save_scope_as = monk
						after_creation = {
							add_character_flag = created
						}
					}
				}
				if = {
					limit = { exists = scope:monk }
					visit_holding_mark_rare_spawn_seen_effect = { SCENE = monk }
				}
			}
		}
		set_variable = visit_holding_checked_for_monk
	}
}

#Used to generate random recruits from templates
scripted_trigger 1320_generate_character_trigger = {
	NOR = {
        AND = {
            exists = scope:market_first_recruit
            scope:market_first_recruit = { has_character_flag = is_$ARCHETYPE$_recruit }
        }
        AND = {
            exists = scope:market_second_recruit
            scope:market_second_recruit = { has_character_flag = is_$ARCHETYPE$_recruit }
        }
        AND = {
            exists = scope:market_third_recruit
            scope:market_third_recruit = { has_character_flag = is_$ARCHETYPE$_recruit }
        }
    }
}

	create_character = {
        template = $ARCHETYPE$_recruit_template
        gender_female_chance = 50
        location = root.location
        dynasty = none 
        culture = $CULTURE$
        faith = $FAITH$
        save_scope_as = 1320_new_character
        after_creation = { add_character_flag = is_$ARCHETYPE$_recruit }
    }
    scope:1320_new_character ?= {

    		limit = { location != root.location }

    	}
    }
