# Building types are not event scopes, so the great project is saved for [cathedral_building.GetName]
scripted_effect pam_cathedral_save_finished_building_scope_effect = {
	if = {
		limit = { exists = scope:great_project }
		scope:great_project = {
			save_scope_as = cathedral_building
		}
	}
	else = {
		random_great_project = {
			limit = {
				OR = {
					great_project_type = pam_cathedral_01
					great_project_type = pam_cathedral_02
					great_project_type = pam_cathedral_03
				}
				OR = {
					great_project_founder = root
					great_project_owner = root
				}
			}
			save_scope_as = great_project
			save_scope_as = cathedral_building
		}
	}
}

#Owner (archbishop) Events (the owner can also be the founder, but this is to insure some theocratic rulers always get a couple events during the course of the cathedral being built)
scripted_effect pam_cathedral_apply_archdiocese_sermon_effect = {
	scope:archdiocese ?= {
		every_county_in_clerical_region = {
			limit = {
				faith = scope:owner.faith
			}
			add_county_modifier = {
				modifier = $COUNTY_MODIFIER$
				years = $DURATION$
			}
			holder = {
				if = {
					limit = {
						is_landed = yes
						faith = scope:owner.faith
						NOT = {
							this = scope:owner
						}
					}
					hidden_effect = {
						add_character_modifier = {
							modifier = $RULER_MODIFIER$
							years = $DURATION$
						}
						if = {
							limit = {
								is_ai = no
							}
							send_interface_toast = {
								type = event_toast_effect_neutral
								title = pam_cathedral_west_portal_sermon_toast
								left_icon = scope:owner
								right_icon = scope:province.barony
							}
						}
					}
				}
			}
		}
	}
}

						limit = {
