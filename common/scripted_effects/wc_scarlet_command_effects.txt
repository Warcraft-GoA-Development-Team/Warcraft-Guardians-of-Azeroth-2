# Scarlet institutional mechanics. State belongs to the founding title.

wc_scarlet_remove_stratholme_effect = {
	global_var:wc_scarlet_crusade_title.var:wc_scarlet_campaign_target ?= {
		if = {
			limit = { has_county_modifier = wc_balnazzar_scarlet_command_modifier }
			remove_county_modifier = wc_balnazzar_scarlet_command_modifier
		}
	}
}

wc_scarlet_grant_stratholme_effect = {
	global_var:wc_scarlet_crusade_title.var:wc_scarlet_campaign_target ?= {
		if = {
			limit = { NOT = { has_county_modifier = wc_balnazzar_scarlet_command_modifier } }
			add_county_modifier = { modifier = wc_balnazzar_scarlet_command_modifier }
		}
	}
}

# Character scope. Breaking a personal guarantee leaves the institution intact.
# Body scope. Political withdrawal does not rescind Darion's standing refusal
# to remain in the court of a revealed demon. Captives cannot escape this way.
wc_scarlet_darion_departure_effect = {
	if = {
		limit = {
			character:60020 = {
				var:wc_scarlet_civil_commander ?= prev
				is_alive = yes is_imprisoned = no is_landed = no
				NOT = { has_trait = being_undead }
				is_courtier_of = prev
			}
		}
		character:60020 = { move_to_pool = yes }
		save_scope_value_as = { name = wc_darion_departed value = yes }
	}
	wc_scarlet_end_personal_agreements_effect = yes
}

wc_scarlet_end_personal_agreements_effect = {
	remove_variable = wc_foundation_pending
	remove_variable = wc_balnazzar_renault_request
	remove_variable = wc_lordaeron_offer
	remove_variable = wc_lordaeron_consent
	global_var:wc_scarlet_crusade_title ?= {
		if = {
			limit = { var:wc_scarlet_darion_request_owner ?= prev }
			remove_variable = wc_scarlet_darion_pending
			remove_variable = wc_scarlet_darion_request_owner
			if = {
				limit = { var:wc_scarlet_stratholme_open ?= prev }
				remove_variable = wc_scarlet_stratholme_open
			}
		}
		if = {
			limit = { var:wc_scarlet_darion_agreed ?= prev }
			remove_variable = wc_scarlet_darion_agreed
		}
	}
	if = {
		limit = { character:60020 = { var:wc_scarlet_civil_commander ?= prev } }
		character:60020 = {
			remove_variable = wc_scarlet_civil_commander
			remove_variable = wc_scarlet_civil_withdrawn
		}
	}
	if = {
		limit = { global_var:wc_scarlet_crusade_title ?= { var:wc_scarlet_sponsor_body ?= prev } }
		global_var:wc_scarlet_crusade_title = { remove_variable = wc_scarlet_sponsor_body }
	}
}

# Character scope. Both foundation routes use this, without generic costs/armies.
wc_scarlet_register_crusade_effect = {
	if = {
		limit = { NOT = { exists = global_var:wc_scarlet_crusade_title } }
		set_global_variable = { name = wc_scarlet_crusade_title value = $TITLE$ }
		$TITLE$ = { set_variable = wc_scarlet_legacy_migrated }
	}
	if = {
		limit = {
			NOT = { is_target_in_global_variable_list = { name = unavailable_unique_decisions target = flag:form_the_scarlet_crusade_decision } }
		}
		add_to_global_variable_list = { name = unavailable_unique_decisions target = flag:form_the_scarlet_crusade_decision }
	}
	add_character_flag = has_formed_the_scarlet_crusade
	if = {
		limit = { has_faith = faith:holy_light }
		set_character_faith_with_conversion = faith:scarletism
	}
	trigger_event = scarlet.1005
}

# Any scope; reactive hooks plus the current holder's quarterly pulse.
wc_scarlet_update_crusade_effect = {
	wc_scarlet_migrate_crusade_effect = yes
	if = {
		limit = { exists = global_var:wc_scarlet_crusade_title }
		global_var:wc_scarlet_crusade_title = {
			# Retired requests release only their own pending window.
			if = {
				limit = { has_variable = wc_scarlet_darion_request_owner }
				if = { limit = { var:wc_scarlet_stratholme_open ?= var:wc_scarlet_darion_request_owner } remove_variable = wc_scarlet_stratholme_open }
				remove_variable = wc_scarlet_darion_pending
				remove_variable = wc_scarlet_darion_request_owner
				remove_variable = wc_scarlet_darion_agreed
			}
			# A reply cannot hold the conquest window forever or follow a new ruler.
			if = {
				limit = { has_variable = wc_scarlet_sponsor_body NOT = { var:wc_scarlet_sponsor_body = holder } }
				remove_variable = wc_scarlet_sponsor_body
			}
			if = {
				limit = { has_variable = wc_scarlet_darion_agreed NOT = { var:wc_scarlet_darion_agreed = holder } }
				remove_variable = wc_scarlet_darion_agreed
			}
			if = {
				limit = { character:60020 = { has_variable = wc_scarlet_civil_commander NOT = { var:wc_scarlet_civil_commander = prev.holder } } }
				character:60020 = { remove_variable = wc_scarlet_civil_commander remove_variable = wc_scarlet_civil_withdrawn }
			}
			if = {
				limit = {
					# Old saves may record initialization even though the title was destroyed.
					NOT = { exists = faith:scarletism.religious_head }
					exists = holder
					holder = { is_alive = yes has_faith = faith:scarletism }
				}
				if = {
					limit = { exists = var:wc_scarlet_prefer_isillien }
					save_scope_value_as = { name = wc_scarlet_prefer_isillien value = yes }
				}
				holder = { trigger_event = scarlet.1005 }
			}
			if = {
				limit = { exists = var:wc_scarlet_stratholme_open NOT = { var:wc_scarlet_stratholme_open = holder } }
				remove_variable = wc_scarlet_stratholme_open
			}
			if = {
				limit = { exists = var:wc_scarlet_purification_open NOT = { var:wc_scarlet_purification_open = holder } }
				remove_variable = wc_scarlet_purification_open
			}
			if = {
				limit = {
					exists = var:wc_scarlet_stratholme_resolved
					exists = holder
					holder = { is_alive = yes has_faith = faith:scarletism wc_scarlet_controls_campaign_target_trigger = yes }
				}
				wc_scarlet_grant_stratholme_effect = yes
			}
			else = { wc_scarlet_remove_stratholme_effect = yes }
			holder ?= {
				trigger_event = { id = wc_balnazzar_story.1008 }
			}
		}
	}
}
