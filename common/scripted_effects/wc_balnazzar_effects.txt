# Shared voluntary/fallback transition. The caller must pass the character who
# receives Balnazzar's realm; without a living recipient, the effect does nothing.
wc_balnazzar_relinquish_realm_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = wc_balnazzar_staged_death }
			$RECIPIENT$ = { is_alive = yes }
		}
		$RECIPIENT$ = { save_scope_as = wc_balnazzar_land_recipient }
		add_character_flag = wc_balnazzar_staged_death
		every_held_title = {
			title_tier > barony
			limit = {
				OR = {
					is_landless_type_title = no
					is_nomad_title = yes
				}
			}
			add_to_list = wc_balnazzar_relinquished_titles
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = wc_balnazzar_land_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = wc_balnazzar_relinquished_titles
			change_title_holder_include_vassals = {
				holder = scope:wc_balnazzar_land_recipient
				change = scope:wc_balnazzar_land_change
			}
		}
		resolve_title_and_vassal_change = scope:wc_balnazzar_land_change
		create_landless_adventurer_title_effect = {
			REASON = flag:voluntary
			FLAVOR_CHAR = root
		}
	}
}

# Story scope; convert once without cancelling committed exposure.
wc_balnazzar_migrate_suspicion_effect = {
	if = {
		limit = { NOT = { has_variable = wc_balnazzar_suspicion_v2 } }
		if = {
			limit = { exists = var:wc_balnazzar_suspicion }
			if = {
				limit = { var:wc_balnazzar_suspicion >= 9 }
				set_variable = { name = wc_balnazzar_suspicion value = 100 }
				set_variable = { name = wc_balnazzar_exposure_queued value = yes }
			}
			else = {
				set_variable = { name = wc_balnazzar_suspicion value = { value = var:wc_balnazzar_suspicion multiply = 10 } }
			}
		}
		else = { set_variable = { name = wc_balnazzar_suspicion value = 0 } }
		set_variable = { name = wc_balnazzar_suspicion_v2 value = yes }
	}
}

# Updates story suspicion and its visible state; call on the story owner.
wc_balnazzar_change_suspicion_effect = {
	save_scope_as = wc_balnazzar_suspicion_owner
	if = {
		limit = {
			has_character_flag = wc_balnazzar_disguise_active
			NOT = { has_character_flag = balnazzar_shed_the_mask }
			exists = global_var:balnazzar_dreadlord_story
			global_var:balnazzar_dreadlord_story = { story_owner = scope:wc_balnazzar_suspicion_owner }
		}
		global_var:balnazzar_dreadlord_story = {
			if = {
				limit = { NOT = { exists = var:wc_balnazzar_suspicion } }
				set_variable = { name = wc_balnazzar_suspicion value = 0 }
			}
			wc_balnazzar_migrate_suspicion_effect = yes
			change_variable = { name = wc_balnazzar_suspicion add = $CHANGE$ }
			clamp_variable = { name = wc_balnazzar_suspicion min = 0 max = 100 }
			wc_balnazzar_update_inquiry_effect = yes

			story_owner = {
				remove_character_modifier = wc_balnazzar_suspicion_whispers_modifier
				remove_character_modifier = wc_balnazzar_suspicion_inquiry_modifier
				remove_character_modifier = wc_balnazzar_suspicion_crisis_modifier
			}
			if = {
				limit = {
					exists = var:wc_balnazzar_suspicion
					var:wc_balnazzar_suspicion >= 60
				}
				story_owner = {
					add_character_modifier = { modifier = wc_balnazzar_suspicion_inquiry_modifier }
				}
			}
			else_if = {
				limit = {
					exists = var:wc_balnazzar_suspicion
					var:wc_balnazzar_suspicion >= 30
				}
				story_owner = {
					add_character_modifier = { modifier = wc_balnazzar_suspicion_whispers_modifier }
				}
			}
			if = {
				limit = { var:wc_balnazzar_suspicion >= 85 }
				story_owner = { add_character_modifier = { modifier = wc_balnazzar_suspicion_crisis_modifier } }
			}
		}
	}
}

# Story scope. Timed variables survive saves and owner transfers.
wc_balnazzar_update_inquiry_effect = {
	if = {
		limit = { NOT = { exists = var:wc_balnazzar_inquiry_migrated } }
		set_variable = { name = wc_balnazzar_inquiry_migrated value = yes }
		if = {
			limit = { exists = var:wc_balnazzar_suspicion_inquiry_started }
			set_variable = { name = wc_balnazzar_inquiry_cooldown value = yes years = 1 }
		}
		else = { set_variable = { name = wc_balnazzar_inquiry_ready value = yes } }
	}
	if = {
		limit = { NOT = { exists = var:wc_balnazzar_inquiry_cooldown } }
		set_variable = { name = wc_balnazzar_inquiry_ready value = yes }
	}
	# An old body's open window must not lock its successor's court.
	if = {
		limit = {
			exists = var:wc_balnazzar_inquiry_open
			NOT = { var:wc_balnazzar_inquiry_open = story_owner }
		}
		remove_variable = wc_balnazzar_inquiry_open
	}
	if = {
		limit = { exists = var:wc_balnazzar_exposure_queued }
		# A body change cannot cancel an exposure already committed.
		if = {
			limit = {
				OR = {
					NOT = { exists = var:wc_balnazzar_exposure_target }
					NOT = { var:wc_balnazzar_exposure_target = story_owner }
				}
			}
			set_variable = { name = wc_balnazzar_exposure_target value = story_owner }
			story_owner = { trigger_event = { id = wc_balnazzar_story.2000 days = 1 } }
		}
	}
	else_if = {
		limit = {
			OR = {
				var:wc_balnazzar_suspicion >= 100
				story_owner = { has_variable = wc_nathrezim_cover_crisis }
			}
		}
		story_owner = { wc_balnazzar_begin_cover_crisis_effect = yes }
	}
	else_if = {
		limit = {
			var:wc_balnazzar_suspicion >= 30
			exists = var:wc_balnazzar_inquiry_ready
			NOT = { exists = var:wc_balnazzar_inquiry_cooldown }
			NOT = { exists = var:wc_balnazzar_inquiry_open }
			story_owner = { any_courtier = { wc_balnazzar_valid_witness_trigger = yes } }
		}
		# No delay: delivery validates the court before consuming readiness.
		story_owner = { trigger_event = { id = wc_balnazzar_story.0101 } }
	}
	if = {
		limit = {
			var:wc_balnazzar_suspicion >= 85
			NOT = { exists = var:wc_balnazzar_exposure_queued }
			NOT = { exists = var:wc_balnazzar_inquiry_open }
			story_owner = {
				NOT = { has_character_flag = wc_balnazzar_crisis_seen }
				NOT = { has_variable = wc_nathrezim_cover_crisis }
				is_imprisoned = no
				is_incapable_trigger = no
			}
		}
		story_owner = { trigger_event = wc_balnazzar_story.0103 }
	}
}

# Story scope, polled by the existing story timer. Changing bodies or losing
# the last free worried witness resets the full-year pressure timer.
wc_balnazzar_witness_pressure_effect = {
	if = {
		limit = { NOT = { var:wc_balnazzar_pressure_body ?= story_owner } }
		set_variable = { name = wc_balnazzar_pressure_body value = story_owner }
		remove_variable = wc_balnazzar_pressure_waiting
		remove_variable = wc_balnazzar_pressure_cooldown
	}
	if = {
		limit = {
			var:wc_balnazzar_suspicion >= 30
			NOT = { exists = var:wc_balnazzar_exposure_queued }
			story_owner = { wc_balnazzar_free_witness_count > 0 }
			story_owner = { NOT = { has_variable = wc_nathrezim_cover_crisis } }
		}
		if = {
			limit = { NOT = { has_variable = wc_balnazzar_pressure_waiting } }
			set_variable = { name = wc_balnazzar_pressure_waiting value = yes }
			set_variable = { name = wc_balnazzar_pressure_cooldown value = yes years = 1 }
		}
		else_if = {
			limit = { NOT = { has_variable = wc_balnazzar_pressure_cooldown } }
			set_variable = { name = wc_balnazzar_pressure_cooldown value = yes years = 1 }
			story_owner = {
				if = {
					limit = { wc_balnazzar_free_witness_count >= 2 }
					wc_balnazzar_change_suspicion_effect = { CHANGE = 8 }
				}
				else = { wc_balnazzar_change_suspicion_effect = { CHANGE = 4 } }
			}
		}
	}
	else = {
		remove_variable = wc_balnazzar_pressure_waiting
		remove_variable = wc_balnazzar_pressure_cooldown
	}
}

		}

			limit = { can_openly_learn_magic_type_trigger = { MAGICTYPE = $MAGIC$ } }

				every_character_trait = {

					scope:wc_balnazzar_possessed_body = { add_trait = scope:wc_balnazzar_magic_trait }
				}
			}
		}

			if = {

				wc_add_magic_secret_effect = { MAGIC = $MAGIC$ }
			}
		}

# Leave Renault as the only filled assassin slot before grab_best_agent_murderer_effect,
# which otherwise chooses a random filled assassin.
wc_balnazzar_force_renault_as_assassin_effect = {
	scope:balnazzar_alexandros_murder_scheme = {
		random_scheme_agent_slot = {
			limit = {
				is_agent_slot_type = agent_assassin
				is_filled = no
				save_temporary_scope_as = balnazzar_slot_temp
				scope:renault = { is_valid_as_agent_in_slot = scope:balnazzar_slot_temp }
			}
			save_scope_as = balnazzar_renault_slot
		}
	}
	if = {
		limit = { exists = scope:balnazzar_renault_slot }
		scope:renault = {
			force_add_to_agent_slot = {
				agent_slot = scope:balnazzar_renault_slot
				years = 5
			}
		}
		# Remove the other default assassin slot so no other recruited agent can
		# out-compete Renault for the kill.
		scope:balnazzar_alexandros_murder_scheme = {
			random_scheme_agent_slot = {
				limit = {
					is_agent_slot_type = agent_assassin
					is_filled = no
				}
				save_scope_as = balnazzar_spare_assassin_slot
			}
		}
		if = {
			limit = { exists = scope:balnazzar_spare_assassin_slot }
			scope:balnazzar_alexandros_murder_scheme = {
				remove_agent_slot = scope:balnazzar_spare_assassin_slot
			}
		}
	}
}
