		limit = {
			has_trait = creature_human
			culture = { has_clothing_gfx = small_blood_vrykul_gfx } #a gfx is used and not culture for potential culture hybrids
		}
		random = {
			chance = 5
			set_blood_effect = { BLOOD = blood_vrykul }
		}
	}
	if = { #Big chance for the Mok'nathal to have Ogre blood
		limit = {
			has_trait = creature_orc
			culture = { has_clothing_gfx = blood_ogre_gfx }
		}
		set_blood_effect = { BLOOD = blood_ogre }
	}
	else_if = { # Small chance for naga to have Highborne or Night elf blood
		limit = {
			has_trait = creature_naga
		}
		random = {
			chance = 2
			random_list = {
				1 = { set_blood_effect = { BLOOD = blood_highborne } }
				1 = { set_blood_effect = { BLOOD = blood_night_elf } }
			}
		}
	}
	else_if = { # Chance for Man'ari, Broken and Lost Ones to have Draenei blood
		limit = {
			OR = {
				has_trait = creature_manari
				has_trait = creature_broken
				has_trait = creature_lost_one

	}

		limit = {
			has_trait = creature_faldorei
		}
		random = {
			chance = 15
			if = {
				limit = {
					OR = {
						real_father ?= { has_trait = blood_nightborne }
						mother ?= { has_trait = blood_nightborne }
						real_father ?= { has_trait = creature_nightborne }
						mother ?= { has_trait = creature_nightborne }
					}
				}
				set_blood_effect = { BLOOD = blood_nightborne }
			}
			else = {
				set_blood_effect = { BLOOD = blood_nightborne }
			}
		}
	}

remove_blood_traits_effect = {
	if = {
		limit = { any_character_trait = { has_trait_flag = blood_trait } }
		random_character_trait = {
			limit = { has_trait_flag = blood_trait }
			prev = { remove_trait = prev }
		}
	}
}

set_blood_effect = {
	remove_blood_traits_effect = yes
	add_trait = $BLOOD$

	add_character_flag = blood_giving_fired_flag
}

copy_blood_effect = {
	# Add new blood always at the end
	$CHARACTER$ = {
		random_character_trait = {
			limit = { has_trait_flag = blood_trait }
			save_temporary_scope_as = this_trait
		}
	}
	if = {
		limit = { exists = scope:this_trait }
		set_blood_effect = { BLOOD = scope:this_trait }
	}
}
