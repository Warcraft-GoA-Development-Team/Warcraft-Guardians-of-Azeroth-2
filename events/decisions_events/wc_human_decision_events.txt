wc_defias_brotherhood.0002 = { # By Ikarimonogatari
    type = character_event
    content_source = dlc_GOA
    title = wc_defias_brotherhood.0002.title
    desc = wc_defias_brotherhood.0002.desc
    theme = realm

    right_portrait = {
        character = root
        animation = thinking
    }

    option = { # Declare ourselves as the new Noble leaders of the Kingdom!
        name = wc_defias_brotherhood.0002.a
        get_title = title:k_stormwind
        if = {
            limit = { has_trait = gallowsbait }
            remove_trait = gallowsbait
        }
        if = {
            limit = { has_trait = peasant_leader }
            remove_trait = peasant_leader
        }
        add_legitimacy = 500
        add_prestige = 5000
        ai_chance = {
            base = 60
        }
    }
    option = { # Burn down the Noble's systems and continue our crusades
        name = wc_defias_brotherhood.0002.b
        destroy_title = title:k_stormwind
        spawn_army = { # Fellow peasents gather to liberate other lowborns!
            inheritable = yes
            uses_supply = no
            name = defias_league
            men_at_arms = {
                type = bombard
                stacks = 10
            }
            levies = 4990
            location = capital_province
			origin = capital_province
        }
        ai_chance = {
            base = 40
        }
    }
}

gilneas_decisions.1001 = {
	type = character_event
	title = gilneas_decisions.1001.t
	desc = gilneas_decisions.1001.desc
	theme = realm
	left_portrait = {
		character = scope:founder
		animation = personality_bold
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
		create_gilnean_empire_decision_scripted_effect = yes
		legend_seed_new_title_effect = yes
	}

	#Take the nickname.
	option = {
		name = gilneas_decisions.1001.a
		give_nickname = nick_the_great
		add_prestige_experience = massive_fame_gain
	}
	#No thanks on the nickname, more prestige instead please.
	option = {
		name = gilneas_decisions.1001.b
		add_prestige_experience = monumental_fame_gain
	}
}

#Inform any players.
gilneas_decisions.1002 = {
	type = character_event
	title = gilneas_decisions.1001.t
	desc = gilneas_decisions.1002.desc
	theme = realm
	left_portrait = {
		character = scope:founder
		animation = personality_bold
	}

	immediate = {
		#Gilneans, (who are not rival emperors), get some happy muzak.
		if = {
			limit = {
				culture = { has_cultural_pillar = heritage_gilnean }
				NOT = { highest_held_title_tier = tier_empire }
			}
			play_music_cue = "mx_cue_epic_sacral_moment"
		}
		#Empires get rivalry music.
		if = {
			limit = { highest_held_title_tier = tier_empire }
			play_music_cue = "mx_cue_combat_2"
		}
	}

	option = {
		name = { #Empires are less than impressed.
			trigger = {
				highest_held_title_tier = tier_empire
				NOT = {
					culture = { has_cultural_pillar = heritage_gilnean }
				}
			}
			text = gilneas_decisions.1002.a
		}
		name = { #Gilneans are pretty happy.
			trigger = {
				culture = { has_cultural_pillar = heritage_gilnean }
				NOT = { highest_held_title_tier = tier_empire }
			}
			text = gilneas_decisions.1002.b
		}
		name = { #No one else thinks it'll last much longer than the last one.
			trigger = {
				NOR = {
					culture = { has_cultural_pillar = heritage_gilnean }
					highest_held_title_tier = tier_empire
				}
			}
			text = gilneas_decisions.1002.c
		}
	}
}
