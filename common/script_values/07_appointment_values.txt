# For use when promoting a candidate via interaction.
appointment_score_interaction_multiplier_modifiers_value = {
	value = 1
	# Co-emperors are more efficient to promote.
	if = {
		limit = {
			exists = scope:target
			exists = scope:secondary_recipient
			appointment_interactions_recipient_is_suitable_co_emperor_trigger = yes
		}
		if = {
			limit = {
				scope:actor.top_liege = { has_diarchy_active_parameter = diarch_more_efficient_administrative_emperor_promotion_candidate_massive }
			}
			add = diarch_more_efficient_administrative_emperor_promotion_candidate_massive_value
		}
		else_if = {
			limit = {
				scope:actor.top_liege = { has_diarchy_active_parameter = diarch_more_efficient_administrative_emperor_promotion_candidate_major }
			}
			add = diarch_more_efficient_administrative_emperor_promotion_candidate_major_value
		}
		else_if = {
			limit = {
				scope:actor.top_liege = { has_diarchy_active_parameter = diarch_more_efficient_administrative_emperor_promotion_candidate_medium }
			}
			add = diarch_more_efficient_administrative_emperor_promotion_candidate_medium_value
		}
		else_if = {
			limit = {
				scope:actor.top_liege = { has_diarchy_active_parameter = diarch_more_efficient_administrative_emperor_promotion_candidate_mild }
			}
			add = diarch_more_efficient_administrative_emperor_promotion_candidate_mild_value
		}
	}
}

# Maximum investment to support a candidate, and root is the investor
appointment_investment_cap = {
	value = {
		add = {
			value = 50 # Base
			desc = BASE_VALUE
		}
		##### AGE
		if = { # ADULT - scaling bonus
			limit = { is_adult = yes }
			add = {
				value = {
					add = age
					subtract = 16
					divide = 2
					ceiling = yes
				}
				desc = appointment_investment_cap_age
			}
		}
		else = { # CHILD - scaling malus
			add = {
				value = {
					add = age
					subtract = 16
					divide = 2
					floor = yes
				}
				desc = appointment_investment_cap_age
			}
		}
		if = {
			limit = {
				government_has_mechanic = administrative
			}
			##### FAMILY RATING
			if = {
				limit = { house.house_head ?= this }
				add = {
					value = house.house_power_score
					desc = appointment_investment_cap_family_rating
				}
			}
			##### MERIT
			if = {
				limit = { merit_level >= 1 }
				add = {
					value = {
						add = merit_level
						multiply = 5
					}
					desc = "[merit_level|E]"
				}
			}
			##### CELESTIAL MINISTER
			if = {
				limit = {
					is_councillor = yes
					liege ?= {
						tgp_has_access_to_ministry_trigger = yes
					}
				}
				add = {
					value = 10
					desc = is_a_minister_desc
				}
			}
		}
		# admin case
		if = {
			limit = {
				government_has_flag = government_has_influence
			}
			##### INFLUENCE LEVEL
			if = { # NEGATIVE
				limit = { influence_level < 1 }
				add = {
					value = -20
					desc = appointment_investment_cap_influence_level
				}
			}
			else_if = { # POSITIVE
				limit = { influence_level > 1 }
				add = {
					value = {
						add = influence_level
						multiply = 10
					}
					desc = appointment_investment_cap_influence_level
				}
			}
		}
		# otherwise piety, we don't have target passed from code so use else for now
		else = {
			if = { # NEGATIVE
				limit = { piety_level < 1 }
				add = {
					value = -20
					desc = piety_level_holder_desc
				}
			}
			else_if = { # POSITIVE
				limit = { piety_level > 1 }
				add = {
					value = {
						add = piety_level
						multiply = 10
					}
					desc = piety_level_holder_desc
				}
			}
		}
		##### ESTATE BUILDINGS
		house.house_head.domicile ?= {
			add = {
				value = 0
				if = {
					limit = { has_domicile_parameter = estate_increase_house_member_investment_cap_1 }
					add = 10
				}
				else_if = {
					limit = { has_domicile_parameter = estate_increase_house_member_investment_cap_2 }
					add = 20
				}
				else_if = {
					limit = { has_domicile_parameter = estate_increase_house_member_investment_cap_3 }
					add = 30
				}
				else_if = {
					limit = { has_domicile_parameter = estate_increase_house_member_investment_cap_4 }
					add = 40
				}
				else_if = {
					limit = { has_domicile_parameter = estate_increase_house_member_investment_cap_5 }
					add = 50
				}
				if = {
					limit = { has_domicile_parameter = japanese_manor_increase_house_member_investment_cap_1 }
					add = 5
				}
				else_if = {
					limit = { has_domicile_parameter = japanese_manor_increase_house_member_investment_cap_2 }
					add = 10
				}
				else_if = {
					limit = { has_domicile_parameter = japanese_manor_increase_house_member_investment_cap_3 }
					add = 15
				}
				else_if = {
					limit = { has_domicile_parameter = japanese_manor_increase_house_member_investment_cap_4 }
					add = 20
				}
				else_if = {
					limit = { has_domicile_parameter = japanese_manor_increase_house_member_investment_cap_5 }
					add = 25
				}
				else_if = {
					limit = { has_domicile_parameter = japanese_manor_increase_house_member_investment_cap_6 }
					add = 30
				}
				desc = appointment_investment_cap_estate_buildings
			}
		}
		##### FAMILY ATTRIBUTES
		house ?= {
			add = {
				value = 0
				if = {
					limit = { has_house_aspiration_parameter = increased_investment_cap_bonus_3 }
					add = increased_investment_cap_bonus_3_value
				}
				else_if = {
					limit = { has_house_aspiration_parameter = increased_investment_cap_bonus_2 }
					add = increased_investment_cap_bonus_2_value
				}
				else_if = {
					limit = { has_house_aspiration_parameter = increased_investment_cap_bonus_1 }
					add = increased_investment_cap_bonus_1_value
				}
				desc = political_meddlers_attribute_name_desc
			}
		}
		##### EXPANDED POWER BASE
		add = appointment_expand_power_base_bonus
		min = 0
	}
}

appointment_expand_power_base_bonus = {
	### EXPANDED POWER BASE
	save_temporary_value_as = temp_total
	if = {
		limit = { has_character_modifier = ep3_expanded_power_base_minor_modifier }
		add = {
			value = scope:temp_total
			multiply = 0.10
			round = yes
			min = 25
			desc = ep3_expanded_power_base_minor_modifier
		}
	}
	if = {
		limit = { has_character_modifier = ep3_expanded_power_base_medium_modifier }

			value = scope:temp_total
			multiply = 0.20
			round = yes
			min = 50
			desc = ep3_expanded_power_base_medium_modifier
		}
	}
	if = {
		limit = { has_character_modifier = ep3_expanded_power_base_major_modifier }
		add = {
			value = scope:temp_total
			multiply = 0.30
			round = yes
			min = 75
			desc = ep3_expanded_power_base_major_modifier
		}
	}
}

# for all governor, regent, and admin emperor
appointment_score_base = {
	if = {
		limit = {
			top_liege ?= {
				has_realm_law = candidate_score_prestige_law
			}
		}
		# Base score gained from prestige rank
		add = {
			value = 10
			multiply = prestige_level
			desc = PRESTIGE_LEVEL_CONCEPT
		} 
		if = {
			limit = {
				exists = dynasty
			}
			# Base score gained from prestige rank
			add = {
				value = 10
				multiply = dynasty.dynasty_prestige_level
				desc = DYNASTY_PRESTIGE_CONCEPT
			}
		}
	}

	# Dynasty Legacy
	if = {
		limit = {
			dynasty ?= {
				has_dynasty_perk = tgp_chinese_legacy_4
			}
		}
		add = {
			value = legacy_candidate_score_bonus
			desc = "[GetDynastyPerk('tgp_chinese_legacy_4').GetName]"
		}
	}

	# Aspirations
	if = {
		limit = {
			exists = house
			has_trait = lifestyle_poet
		}
		if = {
			limit = {
				house = { has_house_aspiration_parameter = poet_trait_provides_candidate_score_1 }
			}
			add = {
				value = "has_trait_xp(lifestyle_poet)"
				multiply = poet_trait_provides_candidate_score_1_value
				desc = celestial_governor_poet_trait_desc
			}
		}
		else_if = {
			limit = {
				house = { has_house_aspiration_parameter = poet_trait_provides_candidate_score_2 }
			}
			add = {
				value = "has_trait_xp(lifestyle_poet)"
				multiply = poet_trait_provides_candidate_score_2_value
				desc = celestial_governor_poet_trait_desc
			}
		}
		else_if = {
			limit = {
				house = { has_house_aspiration_parameter = poet_trait_provides_candidate_score_3 }
			}
			add = {
				value = "has_trait_xp(lifestyle_poet)"
				multiply = poet_trait_provides_candidate_score_3_value
				desc = celestial_governor_poet_trait_desc
			}
		}
		else_if = {
			limit = {
				house = { has_house_aspiration_parameter = poet_trait_provides_candidate_score_4 }
			}
			add = {
				value = "has_trait_xp(lifestyle_poet)"
				desc = celestial_governor_poet_trait_desc
			}
		}
	}

	# Influence
	if = {
		limit = {
			influence_level != 1
		}
		add = {
			value = 10
			multiply = {
				value = influence_level
				subtract = 1 # Baseline is at 1, so we want the bonus to start at 2
				if = {
					limit = {
						scope:title.holder ?= { has_realm_law = acclamation_succession_law }
					}
					multiply = 2
				}
			}
			desc = "[influence_level|E]"
		}
	}

	# Traits
	if = {
		limit = {
			NOT = {
				scope:title.holder ?= {
					has_realm_law = acclamation_succession_law
				}
			}
			OR = {
				has_trait = inbred
				has_trait = blind
				is_incapable = yes
				has_trait_with_flag = age_related_ailment
			}
		}
		subtract = {
			value = 50
			desc = negative_trait_succession
		}
	}
	if = {
		limit = { has_trait = disinherited }
		subtract = {
			value = 200
			desc = trait_disinherited
		}
	}
	if = {
		limit = { has_trait = gallivanter }
		subtract = {
			value = 200
			desc = trait_gallivanter
		}
	}
	if = {
		limit = { has_trait = devoted }
		subtract = {
			value = 250
			desc = trait_devoted
		}
	}
	else_if = {
		limit = { is_clergy = yes }
		subtract = {
			value = 250
			desc = IS_CLERGY_TRIGGER
		}
	}
	# RITE BASED
	if = {
		limit = {
			exists = top_liege.primary_title.state_rite
		}
		top_liege.primary_title.state_rite = {
			save_temporary_scope_as = target_rite
		}
	}
	else = {
		top_liege.rite = {
			save_temporary_scope_as = target_rite
		}
	}
	# CRIMINAL
	if = {
		limit = {
			has_trait = deviant
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:deviant RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "deviant_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = incestuous
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:incestuous RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "incestuous_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = sodomite
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:sodomite RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "sodomite_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = cannibal
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:cannibal RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "cannibal_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = kinslayer_1
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:kinslayer_1 RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "kinslayer_1_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = kinslayer_2
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:kinslayer_2 RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "kinslayer_2_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = kinslayer_3
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:kinslayer_3 RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "kinslayer_3_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = witch
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:witch RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "witch_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = adulterer
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:adulterer RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "adulterer_and_criminal_desc"
		}
	}
	if = {
		limit = {
			has_trait = fornicator
			trait_is_criminal_in_rite_trigger = { TRAIT = trait:fornicator RITE = scope:target_rite GENDER_CHARACTER = root }
		}
		subtract = {
			value = appointment_score_crime_penalty
			desc = "fornicator_and_criminal_desc"
		}
	}
	if = {
		limit = { exists = scope:title }
		# CLAIMS
		if = {
			limit = { has_strong_claim_on = scope:title }
			add = {
				value = 25
				if = {
					limit = { scope:title.holder = top_liege }
					multiply = 2
				}
				desc = title_claim_desc
			}
		}
		else_if = {
			limit = { has_weak_claim_on = scope:title }
			add = {
				value = 15
				if = {
					limit = { scope:title.holder = top_liege }
					multiply = 2
				}
				desc = title_weak_claim_desc
			}
		}
	}

	# LOWBORN? EW no
	if = {
		limit = {
			is_lowborn = yes
		}
		subtract = {
			value = 25
			if = {
				limit = { scope:title.holder ?= top_liege }
				multiply = 4
			}
			desc = lowborn_desc_succession
		}
	}

	# personal relationship
	if = {
		limit = {
			scope:title.holder ?= {
				NOR = {
					government_has_flag = government_is_celestial  # should not happen
					has_realm_law = acclamation_succession_law  # has different calc
				}
			}
		}
		# RELATIONS
		if = { # Is child of holder
			limit = {
				scope:title.holder ?= { is_parent_of = root }
			}
			add = {
				value = 30
				desc = "parent_holder_desc"
			}
		}
		else_if = { # Is family of holder
			limit = {
				scope:title.holder ?= { is_close_family_of = root }
			}

				value = 20
				desc = "close_family_holder_desc"
			}
		}
		else_if = { # Is family of holder's spouse
			limit = {
				scope:title.holder ?= {
					any_spouse = { is_close_family_of = root }
					NOT = { is_close_family_of = root }
				}
			}
			add = {
				value = 10
				desc = "close_family_married_to_holder_desc"
			}
		}
		if = { # Is a friend
			limit = {
				scope:title.holder ?= { has_relation_friend = root }
			}

				value = 15
				desc = governor_friend_desc
			}
		}
		else_if = {
			limit = {
				scope:title.holder ?= { has_relation_best_friend = root }
			}
			add = {
				value = 30
				desc = governor_best_friend_desc
			}
		}

		# COURT POSITIONS
		if = { # Has a court position with current holder
			limit = {
				scope:title.holder ?= {
					any_court_position_holder ?= { this = root }
				}
			}
			add = {
				value = 5
				desc = court_position_succession_governor_desc
			}
		}
	}
	if = { # Is a elder
		limit = {
			scope:title.holder ?= { has_relation_elder = root }
		}
		add = {
			value = 15
			desc = governor_elder_desc
		}
	}
	if = { # Is a disciple
		limit = {
			scope:title.holder ?= { has_relation_disciple = root }
		}
		add = {
			value = 10
			desc = governor_disciple_desc
		}
	}

	if = {
		limit = { scope:title.holder.diarch ?= root }
		if = {
			limit = { scope:title.holder = { has_realm_law = acclamation_succession_law } }
			if = {
				limit = {
					scope:title.holder = { character_has_co_emperorship_trigger = yes }
				}
				# Once for their raw status.
				add = {
					value = 50
					desc = imperial_co_emperor_desc
				}
			}
			add = {
				value = scope:title.holder.diarchy_swing

			NOT = { government_has_mechanic = administrative }
		}
		add = -2500
	}
}

# avoid being governor in where you grow up
appointment_score_celestial = {
	# Domicile
	if = {
		limit = {
			exists = scope:title
			root.dynasty.dynast ?= {
				has_domicile = yes
			}
			scope:title = {
				is_de_jure_liege_or_above_target = root.dynasty.dynast.domicile.domicile_location.county
			}
		}
		subtract = {
			value = 20
			desc = estate_location_in_area
		}
	}
	# Same dynasty as holder penalty
	if = {
		limit = {
			exists = dynasty
			OR = {
				scope:title.holder.dynasty ?= root.dynasty
				liege.dynasty ?= root.dynasty
			}
		}
		subtract = {
			value = 50
			desc = too_closely_related
		}
	}
}
