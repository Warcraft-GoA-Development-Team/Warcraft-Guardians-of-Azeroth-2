# Your arcane experiments have manifested a strange change in the qualities of visual light in your area
arcane_magic_lifestyle_events.1001 = { 
	title = arcane_magic_lifestyle_events.1001.t
	desc = arcane_magic_lifestyle_events.1001.desc
	content_source = dlc_GOA
	theme = order

	trigger = {
		is_landed = yes
        is_magic_class_user_trigger = { MAGIC = order }
    }

	left_portrait = {
		character = root
		animation = stress
	}

	# WHAT - Study the changes and let them proceed
	option = { 
		name = arcane_magic_lifestyle_events.1001.opt.a

		if = {
			limit = {
				order_magic_lifestyle_perks >= 36 # Completion of all trees + 1 elemental tree
			}
			wc_change_amount_mp_effect = {
                CHANGE = add
                VALUE = 3
            }
		}
		else = {
			add_order_magic_lifestyle_xp = 100
		}

		if = {
			limit = {
				capital_county = {
					MAGIC_is_accepted_or_approved_trigger = { MAGIC = order }
				}
			}
			capital_county = {
				change_county_control = -20
			}
		}

		add_character_flag = arcane_magic_lifestyle_events_1041_large_boost

		stress_impact = {
			just = minor_stress_impact_gain
			diligent = medium_stress_impact_gain
			cynical = minor_stress_impact_loss
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_honor = -1
			}
			stress_impact_ai_chance_modifier = { TRAIT = just STRESS = minor_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = diligent STRESS = medium_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = cynical STRESS = minor_stress_impact_loss }
		}
	}

	# WHAT - Completely halt your investigations for a time
	option = { 
		name = arcane_magic_lifestyle_events.1001.opt.b

		add_prestige = medium_prestige_gain

		stress_impact = {
			arrogant = medium_stress_impact_gain
			diligent = minor_stress_impact_gain
			lazy = minor_stress_impact_loss
			fickle = minor_stress_impact_loss
		}

		add_character_modifier = {
			modifier = wc_stopping_arcane_studies_modifier
			years = 5
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_compassion = 1
				ai_rationality = -0.5
			}
			modifier = {
				factor = 2
				order_magic_lifestyle_perks >= 20
			}
			stress_impact_ai_chance_modifier = { TRAIT = arrogant STRESS = medium_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = diligent STRESS = minor_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = lazy STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = fickle STRESS = minor_stress_impact_loss }
		}
	}

	# WHAT - Find a way to reverse the changes but keep working
	option = { 
		name = arcane_magic_lifestyle_events.1001.opt.c
		trigger = { 
			has_magic_class_trait_level_or_secret_trigger = { CLASS = order LEVEL = 3 }
		}
		show_as_unavailable = { 
			NOT = { has_magic_class_trait_level_or_secret_trigger = { CLASS = order LEVEL = 3 } }
		}
		reason = order_magic_mastery

		wc_change_mana_by_value_effect = {
        	CHANGE = subtract	
    	    VALUE = medium_mana_value
	    }

		stress_impact = {
			just = minor_stress_impact_loss
			diligent = minor_stress_impact_gain
			cynical = minor_stress_impact_loss
		}

		add_character_flag = arcane_magic_lifestyle_events_1041_medium_boost

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_rationality = 1
				ai_greed = 0.5
			}
			stress_impact_ai_chance_modifier = { TRAIT = just STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = diligent STRESS = minor_stress_impact_gain }
			stress_impact_ai_chance_modifier = { TRAIT = cynical STRESS = minor_stress_impact_loss }
		}
	}
}

#You happen across an old book filled with arcane diagrams in a language you cannot understand. None of the courtiers can identify the script used in the book. After a while of trying to make sense of the book, you realize that the letters seem to change every time you attempt to observe them.
arcane_magic_lifestyle_events.1011 = { 
	title = arcane_magic_lifestyle_events.1011.t
	desc = arcane_magic_lifestyle_events.1011.desc
	content_source = dlc_GOA
	theme = order

	trigger = {
		has_character_flag = wc_recently_travelled
        is_magic_class_user_trigger = { MAGIC = order }
    }

	left_portrait = {
		character = root
		animation = page_flipping
	}

	# WHAT - Check for arcane enchantments obscuring the book's meaning.
	option = {
		name = arcane_magic_lifestyle_events.1011.opt.a
		trigger = { 
			has_magic_class_trait_level_or_secret_trigger = { CLASS = order LEVEL = 3 }
		}
		show_as_unavailable = { 
			NOT = { has_magic_class_trait_level_or_secret_trigger = { CLASS = order LEVEL = 3 } }
		}
		reason = order_magic_mastery

		random = {
			chance = 45
			modifier = {
				is_alive = yes
				add = order_magic_lifestyle_perks
			}

			send_interface_toast = {
				title = arcane_magic_lifestyle_events.1011.a_tt
				type = event_toast_effect_good
				left_icon = root
				wc_change_amount_mp_effect = {
					CHANGE = add
					VALUE = 3
				}
			}
		}

		stress_impact = {
			scholar = minor_stress_impact_loss
			arrogant = minor_stress_impact_loss
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_rationality = 1
			}
			stress_impact_ai_chance_modifier = { TRAIT = scholar STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = arrogant STRESS = minor_stress_impact_loss }
		}
	}

	# WHAT - Try to glean what you can from the diagrams
	option = {
		name = arcane_magic_lifestyle_events.1011.opt.b

		stress_impact = {
			scholar = minor_stress_impact_loss
			patient = minor_stress_impact_loss
		}

		duel = {
			skill = learning
			value = decent_skill_rating
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				min = 5
				send_interface_toast = {
					title = arcane_magic_lifestyle_events.1011.b_good
					type = event_toast_effect_good
					left_icon = root
					add_order_magic_lifestyle_perk_points = 1
					add_character_modifier = {
						modifier = wc_enchanted_diagrams_modifier
						years = 3	
					}
				}
			}
			50 =  {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				min = 5
				send_interface_toast = {
					title = arcane_magic_lifestyle_events.1011.b_bad
					type = event_toast_effect_bad
					left_icon = root
					add_stress = minor_stress_impact_gain
				}
			}
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_rationality = 0.5
			}
			modifier = {
				highest_skill = learning
				factor = 2
			}
			modifier = {
				learning >= decent_skill_rating
				factor = 1.5
			}
			modifier = {
				factor = 0
				order_magic_lifestyle_perks >= 27
			}
			stress_impact_ai_chance_modifier = { TRAIT = scholar STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = patient STRESS = minor_stress_impact_loss }
		}
	}

	# WHAT - Sell the book to a collector
	option = { 
		name = arcane_magic_lifestyle_events.1011.opt.c
		custom_tooltip = arcane_magic_lifestyle_events.1011.opt.c_tt
		add_gold = medium_gold_value

		stress_impact = {
			lazy = minor_stress_impact_loss
			greedy = minor_stress_impact_loss
			content = minor_stress_impact_gain
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_greed = 1.5
			}

			stress_impact_ai_chance_modifier = { TRAIT = lazy STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = greedy STRESS = minor_stress_impact_loss }
			stress_impact_ai_chance_modifier = { TRAIT = content STRESS = minor_stress_impact_gain }
		}
	}
}

arcane_magic_lifestyle_events.1031 = { 
	title = arcane_magic_lifestyle_events.1031.t
	desc = arcane_magic_lifestyle_events.1031.desc
	content_source = dlc_GOA
	theme = order
	cooldown = { years = 15 }

	trigger = {
        is_magic_class_user_trigger = { MAGIC = order }
    }

	# Get some light reading done
	option = { 
		name = arcane_magic_lifestyle_events.1031.opt.a
		# Gain 200 XP in military, stewardship, or learning
		random_list = {
			33 = { 
				modifier = { 
					martial_lifestyle_perks > 26
					factor = 0
				}
				add_martial_lifestyle_xp = 200
			}
			33 = { 
				modifier = { 
					stewardship_lifestyle_perks > 26
					factor = 0
				}
				add_stewardship_lifestyle_xp = 200
			}
			33 = { 
				modifier = { 
					learning_lifestyle_perks > 26
					factor = 0
				}
				add_learning_lifestyle_xp = 200
			}
			33 = {
				trigger = {
					AND = { 
						martial_lifestyle_perks > 26
						stewardship_lifestyle_perks > 26
						learning_lifestyle_perks > 26
					}
				}
				add_order_magic_lifestyle_xp = 200
			}
		}

		ai_chance = {
			ai_value_modifier = {
				ai_rationality = 1
			}
		}
	}

	# WHAT - Finally, a chance to rest, feast, and decompress
	option = { 
		name = arcane_magic_lifestyle_events.1031.opt.b
		trigger = { NOT = { has_trait = diligent } }
		# HOW -  Lose a huge amount of stress
		add_stress = massive_stress_loss

		ai_chance = {
			ai_value_modifier = {
				ai_energy = -1
			}
			modifier = {
				add = 50
				stress_level >= 3
			}
			modifier = {
				add = 50
				stress_level >= 2
			}
			modifier = {
				add = 50
				stress_level >= 1
			}
		}
	}

	# WHAT - Snoop on your court members
	option = { 
		name = arcane_magic_lifestyle_events.1031.opt.c
		# HOW -  Random effect: 
		random_list = {
			# 25% - learn 2 secrets amongst your courtiers and vassals
			25 = { 
					custom_tooltip = wc_reveal_two_vassal_courtier_secrets_description
					wc_reveal_two_vassal_courtier_secrets_effect = yes
			}
			# 25% - 20% bonus hostile scheme power for 4 years
			25 = { 
				add_character_modifier = {
					modifier = wc_arcane_studied_hostile_scheming_modifier
					years = 4
				}
			}
			# 25% - 20% bonus personal scheme power for 4 years
			25 = { 
				add_character_modifier = {
					modifier = wc_arcane_studied_personal_scheming_modifier
					years = 4
				}
			}
			# 25% - uncover a plot against you if available (check ep1.flavor.2010)
			25 = {
				trigger = { 
					any_courtier_or_guest = { 
						any_scheme = {
							is_scheme_category = hostile
							scheme_target_character = root
						}
					}
				}
				custom_tooltip = wc_expose_a_courtier_guest_scheme_description
				every_courtier_or_guest = {
					limit = { 
						any_scheme = {
							is_scheme_category = hostile
							scheme_target_character = root
						}
					}
					add_to_list = scheme_havers
				}
				random_in_list = {
					list = scheme_havers
					random_scheme = { 
						limit = { scheme_target_character = root }  
						expose_scheme = yes
					}
				}
			}
		}

		ai_chance = {
			ai_value_modifier = {
				ai_vengefulness = 1
			}
		}
	}
}

# A tear in spacetime occurs as a result of your arcane experiments. Strange magic is flooding the countryside. What do you do?
arcane_magic_lifestyle_events.1041 = { 
	title = arcane_magic_lifestyle_events.1041.t
	desc = {
		desc = arcane_magic_lifestyle_events.1041.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:dragonflight_leader
				}
				desc = arcane_magic_lifestyle_events.1041.blue_dragonflight
			}
			desc = arcane_magic_lifestyle_events.1041.desc_none
		}
	}
	content_source = dlc_GOA
	theme = order

	left_portrait = { 
		character = root
		animation = personality_rational
	}

	lower_center_portrait = {
		trigger = {
			exists = scope:dragonflight_leader
		}
		character = scope:dragonflight_leader
	}

	cooldown = { years = 10 }

	trigger = {
        is_magic_class_user_trigger = { MAGIC = order }
		NOT = { has_character_modifier = wc_stopping_arcane_studies_modifier }
    }

	weight_multiplier = {
		base = 1
		modifier = {
			has_character_flag = arcane_magic_lifestyle_events_1041_large_boost
			add = 1
		}
		modifier = {
			has_character_flag = arcane_magic_lifestyle_events_1041_medium_boost
			add = 0.5
		}
	}

	immediate = {
		if = {
			limit = {
				exists = title:e_blue_dragonflight.holder
			}
			title:e_blue_dragonflight.holder = {
				save_scope_as = dragonflight_leader
			}
		}

		remove_character_flag = arcane_magic_lifestyle_events_1041_large_boost
		remove_character_flag = arcane_magic_lifestyle_events_1041_medium_boost
	}

	# WHAT - Harvest the power of the rift
	option = { 
		name = arcane_magic_lifestyle_events.1041.opt.a
		# WHEN - Character has a high learning or more than 9 perks in Arcane.
		trigger = { 
			OR = {
				has_magic_class_trait_level_or_secret_trigger = { CLASS = order LEVEL = 3 }
				learning >= high_learning_skill
			}
		}
		show_as_unavailable = { 
			NOT = { has_magic_class_trait_level_or_secret_trigger = { CLASS = order LEVEL = 3 } }
		}

		wc_change_amount_mp_effect = {
        	CHANGE = add
    	    VALUE = 10
	    }

		# HOW - The damage due to the rift remains, but the character's arcane spellcasting ability is increased.
		capital_county = {
			change_development_level = -4
		}

		stress_impact = {
			compassionate = medium_stress_gain
			just = medium_stress_gain
			generous = medium_stress_gain
			arrogant = minor_stress_loss
			sadistic = minor_stress_loss
			arbitrary = minor_stress_loss
			ambitious = minor_stress_loss
		}

		if = {
			limit = {
				exists = scope:dragonflight_leader
			}
			random = {
				chance = 50 
				reverse_add_opinion = {
					target = scope:dragonflight_leader
					modifier = disappointed_opinion
					opinion = -25
				}
			}
		}

		ai_chance = {
			ai_value_modifier = {
				ai_greed = 1
			}
			stress_impact_ai_chance_modifier = { TRAIT = compassionate STRESS = medium_stress_gain }
			stress_impact_ai_chance_modifier = { TRAIT = just STRESS = medium_stress_gain }
			stress_impact_ai_chance_modifier = { TRAIT = generous STRESS = medium_stress_gain }
			stress_impact_ai_chance_modifier = { TRAIT = arrogant STRESS = minor_stress_loss }
			stress_impact_ai_chance_modifier = { TRAIT = sadistic STRESS = minor_stress_loss }
			stress_impact_ai_chance_modifier = { TRAIT = arbitrary STRESS = minor_stress_loss }
			stress_impact_ai_chance_modifier = { TRAIT = ambitious STRESS = minor_stress_loss }
		}
	}

	# WHAT - Close the rift
	option = { 
		name = arcane_magic_lifestyle_events.1041.opt.b
		# WHEN - Always
		# HOW - The character loses a small amount of mana, but the damage is repaired and the player is able to transmute some of the residual mana into saleable materials for gold.
		wc_change_mana_by_value_effect = { # Remove mana
			CHANGE = subtract
			VALUE = 10 # standardized or scaled amount? - should be a small amount
		}
		add_gold = tiny_gold_value

		stress_impact = { 
			greedy = minor_stress_loss
			humble = minor_stress_loss
			just = minor_stress_loss
			paranoid = minor_stress_loss
			ambitious = minor_stress_gain
			arbitrary = minor_stress_gain
			arrogant = minor_stress_gain
		}

		if = {
			limit = {
				exists = scope:dragonflight_leader
			}
			random = {
				chance = 50 
				reverse_add_opinion = {
					target = scope:dragonflight_leader
					modifier = grateful_opinion
					opinion = 15
				}
			}
		}

		ai_chance = {
			ai_value_modifier = {
				ai_honor = 1
			}
			stress_impact_ai_chance_modifier = { TRAIT = paranoid STRESS = minor_stress_loss }
			stress_impact_ai_chance_modifier = { TRAIT = greedy STRESS = minor_stress_loss }
			stress_impact_ai_chance_modifier = { TRAIT = humble STRESS = minor_stress_loss }
