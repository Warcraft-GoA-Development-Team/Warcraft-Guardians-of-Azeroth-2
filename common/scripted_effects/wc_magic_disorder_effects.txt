init_disorder_spells_effect = {
    add_to_global_variable_list = { name = disorder_spells target = flag:curse }
    add_to_global_variable_list = { name = disorder_spells target = flag:subjugate }
    add_to_global_variable_list = { name = disorder_spells target = flag:summon_demon }
}

cast_curse_effect = {
    get_recipient_effect = { SPELL = curse }
    save_scope_as = real_caster
	scope:this_recipient ?= {
		custom_tooltip = curse_tt
		if = {
			limit = {
				scope:real_caster = {
					has_variable = curse_type
				}
			}
			scope:real_caster = {
				var:curse_type = {
					save_scope_as = curse
				}
				remove_variable = curse_type
			}

			switch = {
				trigger = scope:curse

				flag:curse_of_exhaustion = { contract_disease_notify_effect  = { DISEASE = curse_of_exhaustion } }
				flag:curse_of_tongues = { contract_disease_notify_effect  = { DISEASE = curse_of_tongues } }
				flag:curse_of_weakness = { contract_disease_notify_effect  = { DISEASE = curse_of_weakness } }
			}

			add_opinion = {
				modifier = wc_cursed_me
				target = scope:real_caster
			}
        }
		else = {
			hidden_effect = {
				scope:real_caster = {
					wc_change_mana_by_value_effect = {
						CHANGE = add
						VALUE = wc_spell_curse_cost_mana_value
					}
				}
			}
		}
	}
}

cast_subjugate_effect = {
    save_scope_as = real_caster
	if = {
		limit = {
            NOT = { exists = scope:cancel }
        }
        get_recipient_effect = { SPELL = subjugate }

        scope:this_recipient ?= {
            set_relation_master = scope:real_caster
        }
    }
    else = {
        var:subjugate_recipient = {
            if = {
                limit = {
                    has_relation_master = scope:real_caster
                }
                remove_relation_master = scope:real_caster
            }
        }
    }
}

cast_summon_demon_effect = {
    if = {
        limit = {
            NOT = { exists = scope:cancel }
        }
        get_recipient_effect = { SPELL = summon_demon }

        custom_tooltip = cast_summon_demon_effect_tt

        if = {
            limit = {
                has_variable = summoning_demon
            }
            switch = {
                trigger = var:summoning_demon

                flag:imp = {
                    add_character_modifier = wc_imp_summon_modifier

                    if = {
                        limit = {
                            has_variable = imp_summon
                        }

                        var:imp_summon = {
                            save_scope_as = imp_summon

                        wc_resurrect_CHARACTER_with_STATUS_effect = { CHARACTER = scope:imp_summon STATUS = alive }
                        scope:resurrected_character = {
                            save_scope_as = demon
                        }
                        scope:imp_summon = {
                            make_unprunable = no
                        }

                        remove_variable = imp_summon
                    }
                    else = {
                        create_character = {
							template = summoned_imp_character
							employer = root
							dynasty = none
							save_scope_as = demon
                        }
                    }

                set_variable = {
                    name = observer_summon
                    value = scope:summon
                }
            }

        remove_modifier_if_present = { TYPE = character MODIFIER = wc_observer_summon_modifier }
    }

shivarra_secret_effect = {
    random_$TYPE$ = {
        limit = {
            any_secret = {
                is_magic_secret_trigger = yes
                NOT = {
                    is_known_by = root
                }
            }
        }
        random_secret = {
            limit = {
                is_magic_secret_trigger = yes
                NOT = {
                    is_known_by = root
                }
            }
            expose_secret = root
        }
    }
}

observer_observation_effect = {
    var:casted_target = {
        if = {
            limit = {
                0 < {
                    add = $TYPE$_magic_lifestyle_perk_points
                    add = $TYPE$_magic_lifestyle_perks
                }
            }

            add_$TYPE$_magic_lifestyle_xp = 50
        }
        else = {
            switch = {
                trigger = has_lifestyle

                light_magic_lifestyle = { add_light_magic_lifestyle_xp = 25 }
