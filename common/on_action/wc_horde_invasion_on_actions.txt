wc_invasion_on_actions = {
	effect = {
		if = {
			limit = {
				scope:title = title:c_northshire
				scope:new_holder = { is_from_the_horde_trigger = yes }
			}
			scope:new_holder = {
				trigger_event = {
					id = wc_horde_invasion.9001
					days = 7
				}
			}
		}
		if = { 
			limit = { 
				scope:title = title:k_stormwind
				scope:new_holder = { 
					has_title = title:e_horde
					owns_story_of_type = story_horde_invasion
				}
				has_ep3_dlc_trigger = yes # TODO: Remove when they make this vanilla
			}
			if = { 
				limit = { is_ai = yes }
				add_character_flag = { flag = force_legitimist_adventurer days = 10 }
			}
			create_landless_adventurer_title_effect = {
				REASON = flag:conquest
				FLAVOR_CHAR = root
			}
		}
	}
}

medivh_confrontation_travel_on_action = { 
	effect = { 
		debug_log = "WCHI medivh_confrontation_travel_on_action proc'd"
		if = { 
			limit = { location = scope:story.var:first_war_target.capital_province }
			delay_existing_travel_plan_effect = { DAYS = 31 }
			scope:story.var:first_war_target = { 
				trigger_event = wc_horde_invasion.2011
			}
		}
		else_if = { 
			limit = { location = title:c_deadmans_pass.title_province }
			scope:guldan_character = { 
				trigger_event = wc_horde_invasion.2101
			}
		}
		else_if = { 
			limit = { location = title:c_karazhan.title_province  }
			scope:medivh_character = { 
				trigger_event = wc_horde_invasion.2201
			}
		}
		else = { } # silence error
	}
}

blackhand_assassinate_durotan_on_action = { # only happens if durotan is AI and doomhammer is AI
	effect = { 
		if = { 
			limit = { 
				scope:durotan_character = { 
					is_alive = yes
					is_ai = yes
				}
				scope:blackhand_character = { 
					is_alive = yes
					has_title = title:e_horde
					is_ai = yes
				}
				character:durotan_assassin = { 
					is_alive = yes
					employer = scope:durotan_character
				}
			}
			scope:durotan_character = {
				override_death_killer_effect = {
					death_reason = death_murder
					killer = character:durotan_assassin
				}
			}
		}
	}
}

blackhand_death_notification_on_action = { 
	effect = { 
		
	# send 2902 to interested rulers who aren't vassals
    scope:blackhand_character = {
        # add his kids
        every_child = {
            add_to_list = 2902_recipients
        }
    }

	scope:orgrim_character = {
        # add his vassals and their vassals etc
        every_vassal_or_below = {
            add_to_list = 2902_recipients
        }
	}

    every_independent_ruler = {
        limit = {
			OR = { 
				# add all top level rulers with title above baron who have land in the "azeroth" EK region
				AND = { 
					highest_held_title_tier > tier_barony
            		any_sub_realm_county = {
						title_province = { 
							geographical_region = world_eastern_kingdoms_azeroth
						}
            		}
				}
				# add top level rulers with title above duke who have land in the EKs
				AND = { 
					highest_held_title_tier > tier_duchy
            		any_sub_realm_county = {
						title_province = { 
							geographical_region = world_eastern_kingdoms
						}
            		}
				}
				# OR = { 
				# 	culture = { has_cultural_pillar = heritage_orcish }
				# 	any_held_title = { is_orcish_clan_title_trigger = yes }
				# }
			}
        }	
        add_to_list = 2902_recipients
    }
	
    every_in_list = {
        list = 2902_recipients 
		limit = {
			NOT = { 
				this = scope:blackhand_character
				this = scope:orgrim_character
			}
		}
        trigger_event = wc_horde_invasion.2902
    }
	}
}
