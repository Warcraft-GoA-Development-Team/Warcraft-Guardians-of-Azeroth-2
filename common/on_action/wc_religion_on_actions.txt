# Vanilla 1.20 removed faith_fervor_events_pulse and its fervor_events.txt; GoA keeps the 1.19 fervor events (events/religion_events/fervor_events.txt)
# Called from on_faith_monthly in religion_on_actions.txt
# Root is the faith
wc_faith_fervor_events_pulse = {
	trigger = {
		NOR = {
			has_doctrine_parameter = unreformed
			has_variable = recent_fervor_event
		}
	}

	random_events = {
		chance_to_happen = 6 # ~53% chance of occurring within a given year. 90% chance within 3 years, 99.8% chance within 5.
		100 = fervor.1001
		100 = fervor.2001
	}
}

# Invader faiths arrive already in the Age of Crusades (issue #1973)
# Their temporal heads declare directed_great_holy_war. AI attackers wait out variable_first_ghw_cooldown.
# Doombringer holds both burning_legion_religion and orcish_fel.
on_game_start = {
	on_actions = { wc_ghw_invader_unlock_on_action }
}

wc_ghw_invader_unlock_on_action = {
	effect = {
		religion:deathbringer = {
			set_variable = {
				name = variable_ghw_unlocked
				value = yes
			}
			set_variable = {
				name = variable_first_ghw_cooldown
				value = yes
				days = 365
			}
		}
		religion:doombringer = {
			set_variable = {
				name = variable_ghw_unlocked
				value = yes
			}
			set_variable = {
				name = variable_first_ghw_cooldown
				value = yes
				days = 365
			}
		}
	}
}
