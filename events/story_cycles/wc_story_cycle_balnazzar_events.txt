# Light test scenario - Balnazzar's canon WC3 betrayal: does he rule Lordaeron
# for the Burning Legion, or carve out a throne for himself?
wc_balnazzar_story.1000 = {
	type = character_event
	title = wc_balnazzar_story.1000.title
	desc = wc_balnazzar_story.1000.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}

	immediate = {
		scope:story = {
			set_variable = { name = story_stage value = 1 }
		}
	}

	option = { # A throne of his own
		name = wc_balnazzar_story.1000.a

		scope:story = {
			set_variable = { name = balnazzar_ambition value = 1 }
		}
		add_character_flag = balnazzar_secret_king
		add_stress = minor_stress_loss # Ambition, briefly, feels like relief

		trigger_event = {
			id = wc_balnazzar_story.1001
			days = { 5 20 }
		}
	}

	option = { # A faithful servant, for now
		name = wc_balnazzar_story.1000.b

		scope:story = {
			set_variable = { name = balnazzar_ambition value = 0 }
		}
		add_character_flag = balnazzar_loyal_servant
		add_stress = minor_stress_gain # Obedience chafes

		trigger_event = {
			id = wc_balnazzar_story.1001
			days = { 5 20 }
		}
	}
}

# A Face Not His Own - Balnazzar fakes his death and takes Saidan Dathrohan's
# identity (character:60021), seizing the Scarlet Crusade / Order of the
# Silver Hand from within. Player transfer, mirrors the ck3-docs recipe-transfer pattern.
wc_balnazzar_story.1002 = {
	type = character_event
	title = wc_balnazzar_story.1002.title
	desc = wc_balnazzar_story.1002.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60021 # Saidan Dathrohan
		animation = worry
	}

	immediate = {
		character:60021 = { save_scope_as = dathrohan }
	}

	# ponytail: single option gated on Dathrohan being alive/playable - if he's dead
	# by this point the event has zero valid options. Add a fallback branch (delay
	# retrigger, or pick another host) once that's a real risk in playtesting.
	option = {
		name = wc_balnazzar_story.1002.a

		trigger = {
			scope:dathrohan = {
				is_alive = yes
				is_playable_character = yes
			}
		}

		show_as_tooltip = {
			death = { death_reason = death_vanished }
		}
		hidden_effect = {
			scope:dathrohan = {
				add_character_flag = balnazzar_wearing_dathrohan
				add_trait = balnazzar_disguise

				# The borrowed vitality burns off Dathrohan's own age-related
				# decline at the moment of possession. Ongoing protection while
				# he wears the trait lives in story_balnazzar_dreadlord's
				# effect_group (common/story_cycles/wc_story_cycle_balnazzar.txt).
				remove_trait = clouded_eyes
				remove_trait = faltering_heart
				remove_trait = fragile_bones
				remove_trait = infirm
				remove_trait = withering_mind
			}

			# Keep the story attached to whoever is actually playing Balnazzar's
			# arc. Without this, killing root below fires on_owner_death and
			# ends the story right here, even though the narrative continues.
			scope:story = {
				make_story_owner = scope:dathrohan
			}

			# Move the player OFF root before killing it. Killing the still-played
			# root triggers the engine's death/succession screen immediately and
			# aborts the rest of this effect - set_player_character must go first.
			# Must be called unscoped (root = current player), NOT from inside
			# scope:dathrohan - the engine requires the CURRENT scope to already
			# be the player ("Scoped character is not a player" otherwise).
			set_player_character = scope:dathrohan

			death = { death_reason = death_vanished } # Balnazzar's old face "dies" - root, now an NPC

			scope:dathrohan = {
				trigger_event = {
					id = wc_balnazzar_story.1003
					days = { 10 30 }
				}
			}
		}
	}
}

# Sylvanas Claims Her Due - with "Balnazzar" dead and the new Grand Crusader
# distracted, Sylvanas moves on Lordaeron's ruined heartland and Brill.
wc_balnazzar_story.1003 = {
	type = character_event
	title = wc_balnazzar_story.1003.title
	desc = wc_balnazzar_story.1003.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = {
		character = character:42076 # Sylvanas Windrunner
		animation = scheme
	}

	immediate = {
		character:42076 = { save_scope_as = sylvanas }

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = balnazzar_title_loss_change
			add_claim_on_loss = yes
		}

		# The duchy title alone is just the crown - transfer the actual county
		# (the land shown on the map) too, or nothing visibly changes hands.
		title:c_lordaeron = {
			change_title_holder_include_vassals = {
				holder = scope:sylvanas
				change = scope:balnazzar_title_loss_change
			}
		}
		title:d_lordaeron = {
			change_title_holder_include_vassals = {
				holder = scope:sylvanas
				change = scope:balnazzar_title_loss_change
			}
		}
		title:c_brill = {
			change_title_holder_include_vassals = {
				holder = scope:sylvanas
				change = scope:balnazzar_title_loss_change
			}
		}

		resolve_title_and_vassal_change = scope:balnazzar_title_loss_change
	}

	option = {
		name = wc_balnazzar_story.1003.a

		trigger_event = {
			id = wc_balnazzar_story.1004
			days = { 20 45 }
		}
	}
}

# Blood on Silver - the Ashbringer is corrupted and passed to Renault Mograine,
# and Dathrohan-Balnazzar takes the Order of the Silver Hand for his own,
# soon to be reforged as the Scarlet Crusade.
wc_balnazzar_story.1005 = {
	type = character_event
	title = wc_balnazzar_story.1005.title
	desc = wc_balnazzar_story.1005.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60019 # Renault Mograine
		animation = disbelief
	}

	immediate = {
		character:60019 = { save_scope_as = renault }
	}

	# ponytail: same single-option caveat as .1004 - no fallback if Renault is dead/missing.
	option = {
		name = wc_balnazzar_story.1005.a

		trigger = {
			scope:renault = { is_alive = yes }
		}

		hidden_effect = {
			scope:renault = {
				# Alexandros' original (uncorrupted) Ashbringer was inherited
				# on his death - destroy it before minting the corrupted one,
				# or Renault ends up carrying both swords.
				every_character_artifact = {
					limit = { has_variable = ashbringer }
					destroy_artifact = this
				}

				create_artifact_corrupted_ashbringer_effect = { OWNER = this }
			}
			scope:newly_created_artifact = {
				equip_artifact_to_owner_replace = yes
			}

			create_title_and_vassal_change = {
				type = granted
				save_scope_as = balnazzar_silver_hand_change
				add_claim_on_loss = yes
			}
			title:k_order_of_the_silver_hand = {
				change_title_holder_include_vassals = {
					holder = root
					change = scope:balnazzar_silver_hand_change
				}
			}
			resolve_title_and_vassal_change = scope:balnazzar_silver_hand_change

			# Alexandros' death lets normal succession run first, which can
			# briefly make Renault OUR liege (he inherits the Order before we
			# seize it back above). Force him back into vassalage now, or his
			# own duchy stays outside the realm we just took.
			if = {
				limit = {
					scope:renault = { NOR = { top_liege = root this = root } }
				}
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = balnazzar_renault_vassalize_change
				}
				scope:renault = {
					change_liege = {
						liege = root
						change = scope:balnazzar_renault_vassalize_change
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_renault_vassalize_change
			}

			# The Order becomes the Scarlet Crusade - crown, color and creed all at once.
			title:k_order_of_the_silver_hand = {
				set_title_name = scarlet_crusade_custom_name
				generate_coa = yes
				set_color_from_title = title:k_scarlet_monastery # The faith's own head title - authoritative Scarlet red
			}

			# Scoped to the Silver Hand realm only (root + vassals), not every
			# Holy Light believer on the map - a whole-world instant conversion
			# would be a much bigger swing than "the Crusade converts its own".
			if = {
				limit = { has_faith = faith:holy_light }
				set_character_faith_with_conversion = faith:scarletism
			}
			every_vassal_or_below = {
				limit = { has_faith = faith:holy_light }
				set_character_faith_with_conversion = faith:scarletism
			}
		}
	}
}
