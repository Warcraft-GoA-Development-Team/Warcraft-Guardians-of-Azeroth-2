has_misc_trait_trigger = { #For the traits that aren't races or infusions
	#OR = {
		has_trait = irradiated
	#}
}
has_culture_misc_trait_gfx_trigger = {
	culture = {
		#OR = {
			has_clothing_gfx = irradiated_gfx
		#}
	}
}

is_misc_trait_gene_trigger = {
	trigger_if = {
		limit = { has_misc_trait_trigger = yes }
		has_trait = $MISC$
	}
	trigger_else = {
		trigger_if = {
			limit = {
				exists = culture
				NOT = { exists = scope:culture }
			}
			culture = { save_temporary_scope_as = culture }
		}
		
		trigger_if = {
			limit = { exists = scope:culture }
			scope:culture = { has_clothing_gfx = $MISC$_gfx }
		}
		trigger_else = {
			always = no
		}
	}
}
