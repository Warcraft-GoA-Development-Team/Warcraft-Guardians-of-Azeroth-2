wc_story_titanforged_forge_network = {
	on_end = {
		var:wc_titanforged_network_county = {
			if = {
				limit = { var:wc_titanforged_maintenance = scope:story }
				remove_variable = wc_titanforged_maintenance
			}
		}
	}
	on_owner_death = {
		# The owner is only a scheduler; county ownership controls every effect.
		random_living_character = { limit = { is_landed = yes } save_scope_as = network_scheduler }
		if = { limit = { exists = scope:network_scheduler } make_story_owner = scope:network_scheduler }
	}
	effect_group = {
		days = 30
		triggered_effect = {
			effect = {
				var:wc_titanforged_network_county = { wc_titanforged_network_refresh_effect = yes }
				if = {
					limit = { var:wc_titanforged_network_county = { NOT = { has_variable = wc_titanforged_settlement_result } NOT = { has_variable = wc_titanforged_arcane_project } } }
					end_story = yes
				}
			}
		}
	}
}
