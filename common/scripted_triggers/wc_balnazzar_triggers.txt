# Event owner scope. Native scheme validity includes unlocks and available slots.
# Mirror the additional interaction restrictions before using its native starter.

	wc_balnazzar_inquiry_choice_valid_trigger = yes
	can_start_scheme = { type = $SCHEME$ target_character = scope:balnazzar_witness }
	NOT = { is_isolated_from_trigger = { target = scope:balnazzar_witness } }
	trigger_if = {
		limit = { has_government = landless_adventurer_government }
		ep3_laamp_diplo_range_trigger = { TARGET = scope:balnazzar_witness LAAMP = root }
	}
	trigger_if = {
		limit = { NOT = { flag:$SCHEME$ = flag:sway } }
		scope:balnazzar_witness = { NOT = { has_strong_hook = root } }
		trigger_if = {
			limit = { has_variable = no_scheming_allowed_var }
			var:no_scheming_allowed_var != scope:balnazzar_witness
		}
		NAND = {
			has_game_rule = no_players_multiplayer_murder_schemes
			is_ai = no
			scope:balnazzar_witness = { is_ai = no }
		}
		NAND = {
			has_game_rule = no_player_families_multiplayer_murder_schemes
			is_ai = no
			scope:balnazzar_witness = {
				OR = { is_ai = no any_close_family_member = { is_ai = no this != root } }
			}
		}
	}
}

		limit = { global_var:balnazzar_dreadlord_story = { var:wc_balnazzar_renault_cause = flag:prison } }
		character:60019 = { is_imprisoned_by = root }
	}
	trigger_else_if = {
		limit = { global_var:balnazzar_dreadlord_story = { var:wc_balnazzar_renault_cause = flag:darion } }
		wc_balnazzar_renault_present_trigger = yes
		wc_balnazzar_renault_darion_trigger = yes
	}
	trigger_else = {
		wc_balnazzar_renault_present_trigger = yes
		wc_balnazzar_renault_darion_trigger = no
		character:60019 = { opinion = { target = root value <= -25 } }
		has_hook_of_type = { target = character:60019 type = favor_hook }
	}
}
