#Interactions relating to the head of the dynasty

disinherit_interaction = {
	icon = icon_dynasty
	category = interaction_category_hostile
	interface_priority = 60

	desc = disinherit_interaction_desc
	use_diplomatic_range = no
	
	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			is_dynast = yes
			NOT = {
				has_trait = forgiving
			}
			is_at_war = no
		}
	}

	is_shown = {
		NOR = { 
			scope:recipient = scope:puppet_or_actor 

			AND = { # Explicit AND to ensure no funny business
				scope:recipient = { is_child_of = scope:puppet_or_actor }
				scope:puppet_or_actor.dynasty ?= {
					this = scope:recipient.dynasty
					has_dynasty_perk = fp2_coterie_legacy_5
				}
			}
		}

		scope:puppet_or_actor ?= {
			is_dynast = yes
			dynasty ?= scope:recipient.dynasty

		}
		scope:recipient = {
			NOT = {
				has_trait = disinherited
			}
		}
	}

	is_valid = {
		custom_description = {
			scope:puppet_or_actor.culture = {
				NOT = { has_cultural_parameter = cannot_disherit }
			} 
			text = "mystical_ancestors_disinherit"
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = {
				NOT = { # in such cases, it will be free so no need to check for Renown
					scope:recipient = {
						OR = {
							has_trait = disputed_heritage
							any_secret = {
								type = secret_disputed_heritage
								is_known_by = scope:puppet_or_actor
							}
							has_trait = bastard
						}
					}
				}
			}
			scope:puppet_or_actor.dynasty = {
				dynasty_prestige >= medium_dynasty_prestige_value
			}
		}
		scope:puppet_or_actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }	# Warcraft
		scope:recipient = { NOT = { has_strong_hook = scope:puppet_or_actor } }
		trigger_if = {
			limit = {
				scope:recipient = {
					NOR = {
						is_child_of = scope:puppet_or_actor
						is_grandchild_of = scope:puppet_or_actor
						is_great_grandchild_of = scope:puppet_or_actor
					}
				}
			}
			custom_description = {
				scope:recipient.top_liege = scope:puppet_or_actor.top_liege
				subject = scope:puppet_or_actor
				text = "same_realm_as"
			}
		}
		custom_tooltip = {
			text = cannot_take_overt_hostile_actions_against_diarch.tt
			NOT = { scope:recipient ?= scope:puppet_or_actor.diarch }
		}
	}

	on_auto_accept = {
		scope:recipient = {
			trigger_event = char_interaction.0020
		}
	}
	
	on_accept = {
		scope:puppet_or_actor = {
			# Only get stressed/SP penalties if you don't have the parameter
			if = {
				limit = {
					NOT = { sp_hostile_schemes_except_murder_no_stress_or_fulfillment_trigger = yes }
				}
				stress_and_fulfillment_impact = {
					forgiving = minor_stress_impact_gain
					compassionate = minor_stress_impact_gain
				}
			}
			
			#Legitimacy loss
			if = {
				limit = { has_legitimacy = yes }
				if = {
					limit = { scope:recipient = { is_primary_heir_of = scope:puppet_or_actor } }
					add_legitimacy = {
						value = medium_legitimacy_loss
						multiply = scope:puppet_or_actor.primary_title.tier
					}
				}
				else_if = {
					limit = { scope:recipient = { is_child_of = scope:puppet_or_actor } }
					add_legitimacy = {
						value = minor_legitimacy_loss
						multiply = scope:puppet_or_actor.primary_title.tier
					}
				}
				else = {
					add_legitimacy = {
						value = miniscule_legitimacy_loss
						multiply = scope:puppet_or_actor.primary_title.tier
					}
				}
			}
			hidden_effect = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = disinherit_interaction_notification

					left_icon = scope:puppet_or_actor					
					right_icon = scope:recipient

					scope:recipient = {
						show_as_tooltip = {
							disinherit_effect = { DISINHERITOR = scope:puppet_or_actor }
						}
					}
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					scope:is_puppet_action ?= yes
				}
				scope:actor = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = disinherit_interaction_notification

						left_icon = scope:puppet_or_actor
						right_icon = scope:recipient

						scope:recipient = {
							show_as_tooltip = {
								disinherit_effect = { DISINHERITOR = scope:puppet_or_actor }
							}
						}
					}
				}
			}
		}

		scope:recipient = {
			disinherit_effect = { DISINHERITOR = scope:puppet_or_actor }
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = medium_unity_loss
			DESC = clan_unity_disinheritance.desc
			REVERSE_NON_HOUSE_TARGET = no
		}

		if = {
			limit = {
				scope:recipient = {
					any_secret = {
						type = secret_disputed_heritage
						is_known_by = scope:puppet_or_actor
						save_temporary_scope_as = secret_to_reveal
					}
				}
			}

			scope:secret_to_reveal = {
				expose_secret = scope:recipient.mother
			}

			scope:recipient = {
				add_trait = disputed_heritage
			}
		}
	}
	
	cost = {
		renown = {
			value = {
				add = scope:puppet_or_actor.dynasty.medium_dynasty_prestige_value
				if = {
					limit = {
						scope:recipient = {
							is_landed_or_landless_administrative_or_theocratic = yes
						}
					}
					add = scope:puppet_or_actor.dynasty.major_dynasty_prestige_value
				}
				if = {
					limit = {
						scope:recipient = {
							is_landed_or_landless_administrative_or_theocratic = no
							any_heir_title = {
								exists = holder
							}
						}
					}
					add = scope:puppet_or_actor.dynasty.medium_dynasty_prestige_value
				}
				if = {
					limit = {
						scope:recipient = {
							OR = {
								has_trait = disputed_heritage
								any_secret = {
									type = secret_disputed_heritage
									is_known_by = scope:puppet_or_actor
								}
								has_trait = bastard
							}
						}
					}
					multiply = 0
				}
			}
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
		prestige = {
			value = {
				add = medium_prestige_value
				if = {
					limit = {
						scope:recipient = {
							is_landed_or_landless_administrative_or_theocratic = yes
						}
					}
					add = massive_prestige_value
				}
				if = {
					limit = {
						scope:recipient = {
							is_landed_or_landless_administrative_or_theocratic = no
							any_heir_title = {
								exists = holder
							}
						}
					}
					add = medium_prestige_value
				}
				if = {
					limit = {
						scope:recipient = {
							OR = {
								has_trait = disputed_heritage
								any_secret = {
									type = secret_disputed_heritage
									is_known_by = scope:puppet_or_actor
								}
								has_trait = bastard
							}
						}
					}
					multiply = 0
				}
			}
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}
	
	auto_accept = yes

	ai_targets = {
		ai_recipients = dynasty
		max = 20
	}
	
	ai_frequency_by_tier = {
		barony = 0
		county = 96
		duchy = 48
		kingdom = 12
		empire = 12
		hegemony = 12
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = -100
			scope:puppet_or_actor = {
				OR = {
					legitimacy_level = 0
					legitimacy_level = 1
				}
			}
			scope:recipient = { is_primary_heir_of = scope:puppet_or_actor }
		}
		
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					has_relation_rival = scope:puppet_or_actor
					has_relation_nemesis = scope:puppet_or_actor
				}
			}
		}
		
		modifier = {
			add = 100
			scope:recipient = {
				is_primary_heir_of = scope:puppet_or_actor
				OR = {
					has_trait = inbred
					has_trait = intellect_bad_3
					is_incapable_trigger = yes
				}
			}
		}
		
		modifier = {
			add = 5
			ai_vengefulness >= medium_positive_ai_value
			scope:recipient = {
				is_heir_of = scope:puppet_or_actor
			}
			OR = {
				AND = {
					scope:recipient = {
						has_trait = incestuous
					}
					NOT = { is_incestuous_trigger = yes }
					trait_is_criminal_in_rite_trigger = { TRAIT = trait:incestuous RITE = scope:puppet_or_actor.rite GENDER_CHARACTER = scope:recipient }
				}
				AND = {
					scope:recipient = {
						has_trait = kinslayer
					}
					NOT = { has_trait = kinslayer }
				}
				AND = {
					scope:recipient = {
						has_trait = adulterer
					}
					NOR = {
						has_trait = adulterer
						has_trait = fornicator
					}
					trait_is_criminal_in_rite_trigger = { TRAIT = trait:adulterer RITE = scope:puppet_or_actor.rite GENDER_CHARACTER = scope:recipient }
				}
				AND = {
					scope:recipient = {
						has_trait = fornicator
					}
					NOR = {
						has_trait = adulterer
						has_trait = fornicator
					}
					trait_is_criminal_in_rite_trigger = { TRAIT = trait:fornicator RITE = scope:puppet_or_actor.rite GENDER_CHARACTER = scope:recipient }
				}
				AND = {
					scope:recipient = {
						has_trait = deviant
					}
					is_deviant_trigger = no
					trait_is_criminal_in_rite_trigger = { TRAIT = trait:deviant RITE = scope:puppet_or_actor.rite GENDER_CHARACTER = scope:recipient }
				}
				AND = {
					scope:recipient = {
						has_trait = witch
					}
					is_witch_trigger = no
					trait_is_criminal_in_rite_trigger = { TRAIT = trait:witch RITE = scope:puppet_or_actor.rite GENDER_CHARACTER = scope:recipient }
				}
				AND = {
					scope:recipient = {
						has_trait = sodomite
					}
					NOR = {
						has_trait = sodomite
						any_secret = {
							type = secret_homosexual
						}
					}
					trait_is_criminal_in_rite_trigger = { TRAIT = trait:sodomite RITE = scope:puppet_or_actor.rite GENDER_CHARACTER = scope:recipient }
				}
				AND = {
					scope:recipient = {
						has_trait = cannibal
					}
					is_cannibal_trigger = no
					trait_is_criminal_in_rite_trigger = { TRAIT = trait:cannibal RITE = scope:puppet_or_actor.rite GENDER_CHARACTER = scope:recipient }
				}
			}
		}

		# Unity modifiers
		evaluate_action_decreasing_house_unity = {
			VALUE = 100
		}
		
		modifier = {
			factor = 0
			scope:recipient = {
				OR = {
					has_relation_friend = scope:puppet_or_actor
					has_relation_best_friend = scope:puppet_or_actor
				}
			}
		}
		
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	
		# Warcraft
		modifier = {
			add = 100
			scope:recipient = { is_criminal_being_trigger = { FAITH = scope:puppet_or_actor.faith } }
		}
	}
}

	category = interaction_category_friendly
	desc = restore_inheritance_interaction_desc

	interface_priority = 60

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			is_dynast = yes
			is_at_war = no
		}
	}

	is_shown = {
		scope:puppet_or_actor ?= {
			is_dynast = yes
			dynasty ?= scope:recipient.dynasty

		}
		scope:recipient = {
			has_trait = disinherited
		}
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor.dynasty ={
			dynasty_prestige >= miniscule_dynasty_prestige_value
		}
		scope:puppet_or_actor = { NOT = { is_isolated_from_trigger = { target = scope:recipient } } }	# Warcraft
	}

	on_auto_accept = {
		scope:recipient = {
			trigger_event = char_interaction.0030
		}
	}
	
	on_accept = {
		scope:puppet_or_actor = {
			stress_and_fulfillment_impact = {

dynast_claim_title_interaction = {
	icon = icon_dynasty
	category = interaction_category_diplomacy
	interface_priority = 34 #right after HoF Request Claim
	desc = dynast_claim_title_interaction_desc
	target_type = title
	target_filter = recipient_domain_titles

	ignores_pending_interaction_block = yes
	cost = {
		renown = {
			value = dynast_claim_title_interaction_cost_value
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}
	is_shown = {
		scope:puppet_or_actor ?= {
			this != scope:recipient
			is_dynast = yes
			religion ?= scope:recipient.religion
			dynasty ?= scope:recipient.dynasty
		}

		scope:recipient = {
			is_playable_character = yes
			any_held_title = {
				is_landless_type_title = no
			}
		}
	}
	
	can_be_picked_title = {
		custom_tooltip = {
			NOT = { any_claimant = { this = scope:puppet_or_actor } }
			text = "ALREADY_HAVE_CLAIM_ON_THIS_TITLE"
		}
		is_landless_type_title = no
	}

	is_valid_showing_failures_only = {
		custom_description = {
			scope:recipient = {
				any_held_title = {
					title_tier >= county
					NOT = { any_claimant = { this = scope:puppet_or_actor } }
					is_landless_type_title = no
				} 
			}

			object = scope:recipient
			text = "already_have_claims_on_all_titles"
		}
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				dynasty ?= {
					dynasty_prestige >= dynast_claim_title_interaction_cost_value
				}
			}
		}
	}

	on_auto_accept = {
		scope:recipient = {
			trigger_event = char_interaction.0060
		}
	}
	
	on_accept = {
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				if = {
					limit = { dynast_claim_title_interaction_cost_value > 0 }
					dynasty ?= { add_dynasty_prestige = { value = { add = dynast_claim_title_interaction_cost_value multiply = -1 } } }
				}
			}
		}
		dynast_claim_title_effect = yes
		scope:puppet_or_actor = {
			hidden_effect = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = dynast_claim_title_interaction_notification

					left_icon = scope:puppet_or_actor					
					right_icon = scope:recipient

					show_as_tooltip = {
						dynast_claim_title_effect = yes
					}
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					scope:is_puppet_action ?= yes
				}
				scope:actor = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = dynast_claim_title_interaction_notification

						left_icon = scope:puppet_or_actor
						right_icon = scope:recipient

						show_as_tooltip = {
							dynast_claim_title_effect = yes
						}
					}
				}
			}
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = medium_unity_loss
			DESC = clan_unity_title_claim.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}

	auto_accept = yes
}

dynast_legitimize_bastard_interaction = {
	icon = icon_dynasty
	category = interaction_category_diplomacy
	desc = dynast_legitimize_bastard_interaction_desc

	is_shown = {
		scope:puppet_or_actor ?= {
			this = scope:recipient.dynasty.dynast
			religion ?= scope:recipient.religion
			house != scope:recipient.house
			rite ?= { rite_has_parameter = bastards_legitimize }
		}
		scope:recipient = {
			has_trait = bastard
			this != scope:puppet_or_actor
			rite ?= { rite_has_parameter = bastards_legitimize }
		}
	}

	is_valid = {
		scope:puppet_or_actor.dynasty = {
			dynasty_prestige >= medium_dynasty_prestige_value
		}
	}

	auto_accept = yes

	cost = {
		renown = {
			value = scope:puppet_or_actor.dynasty.medium_dynasty_prestige_value
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}
	
	on_accept = {
		scope:puppet_or_actor = {
			stress_and_fulfillment_impact = {
				arrogant = minor_stress_impact_gain
			}
			send_interface_message = {
				type = event_dynasty_neutral
				title = dynast_legitimize_bastard_interaction_notification
				right_icon = scope:recipient
				show_as_tooltip = {
					scope:recipient = {
						add_trait_force_tooltip = legitimized_bastard
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = legitimized_me_opinion
						}
					}
				}
			}
		}
		
		hidden_effect = {
			scope:recipient = {
				trigger_event = bastard_interaction.0009
				every_parent = {
					limit = {
						is_playable_character = yes
						this != scope:puppet_or_actor
					}
					hidden_effect = { #Nudge towards friendship
						if = {
							limit = {
								NOR = {
									has_relation_friend = scope:puppet_or_actor
									has_relation_potential_friend = scope:puppet_or_actor
								}
							}
							set_relation_potential_friend = scope:puppet_or_actor
						}
					}
					trigger_event = bastard_interaction.0010
				}
				every_sibling = {
					limit = {
						is_playable_character = yes
						this != scope:puppet_or_actor
					}
					trigger_event = bastard_interaction.0010
				}
			}

			scope:puppet_or_actor = {
				if = {
					limit = {
						has_character_modifier = oath_of_children
					}
					hidden_effect = {
						remove_character_modifier = oath_of_children
						add_character_modifier = oath_of_children
					}
				}
			}
		}
	}
