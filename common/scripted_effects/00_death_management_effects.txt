silent_disappearance_effect = {
	hidden_effect = {
		if = {
			limit = {
				is_alive = yes
				is_ruler = no
			}

			override_death_effect = {
		 		death_reason = death_disappearance
		 	}
		}
	}
}

silent_disappearance_ai_effect = {
	hidden_effect = {
		if = {
			limit = {
				is_alive = yes
				is_ruler = no
				root = { is_ai = yes }
			}

			override_death_effect = {
		 		death_reason = death_disappearance
		 	}
		}
	}
}

silent_disappearance_ai_if_created_effect = {
	hidden_effect = {
		if = {
			limit = {
				is_alive = yes
				has_character_flag = created
			}
			if = {
				limit = {
					root = { is_ai = yes }
					is_ruler = no	# prevent kill created characters if they become ruler during event chain
				}
				override_death_effect = { #Warcraft
					death_reason = death_disappearance
				}
			}
			else = {
				remove_character_flag = created
			}
		}
	}
}

silent_disappear_ai_if_not_hired = {
	hidden_effect = {
		if = {
			limit = {
				is_alive = yes
				root = { is_ai = yes }
				is_ruler = no
				employer != root
			}

			override_death_effect = {
		 		death_reason = death_disappearance
		 	}
		}
		else_if = {
			limit = {
				is_alive = yes
				employer != root
			}
			move_to_pool = yes
		}
	}
}

silent_disappear_ai_if_not_hired_with_flag = {
	hidden_effect = {
		if = {
			limit = {
				is_alive = yes
				is_ruler = no
				employer != root
				has_character_flag = created
			}
			if = {
				limit = {
					root = { is_ai = yes }
				}

				override_death_effect = { #Warcraft
					death_reason = death_disappearance
				}
			}
			else = {
				move_to_pool = yes
				remove_character_flag = created
			}
		}
	}
}

		name = battle_death_enemy
		value = $ENEMY$
	}
	set_variable = {
		name = battle_death_leader
		value = $LEADER$
	}
	set_variable = {
		name = battle_death_location
		value = $LOCATION$
	}
}
