# An artifact has changed ownership
# root = the artifact
# scope:owner = the person who now holds the artifact
# scope:old_owner = the person who used to hold the artifact
wc_on_artifact_changed_owner = {
	effect = {
		# Warcraft - Path of the Damned
		if = {
			limit = {
				exists = global_var:path_of_the_damned_story
				global_var:path_of_the_damned_story = { exists = var:enchanted_urn }
				this = global_var:path_of_the_damned_story.var:enchanted_urn
			}

			if = {
				limit = { scope:owner = global_var:path_of_the_damned_story.story_owner }

				scope:owner = {
					trigger_event = {
						id = wc_path_of_the_damned.1900
						days = 1 # Delay
					}
				}
			}
			else = {
				global_var:path_of_the_damned_story = { set_variable = { name = urn_holder value = scope:owner } }
			}
		}
	}
}
