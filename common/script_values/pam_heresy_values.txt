# Faith-scope unless otherwise noted.
# Fervor reset targets, pulse spike curve, Component 2 main-rite gate, and per-component pluralism-scaled spread budgets.

# --- Fervor reset targets (faith scope, change_fervor delta to apply post-outbreak) ---
pam_heresy_fervor_reset_component_1_value = 60
pam_heresy_fervor_reset_component_2_value = 75
pam_heresy_fervor_reset_component_3_value = 90

# --- Pulse cooldown (faith variable lifetime in days) ---
# Shorter than vanilla because the fervor spike is the real cooldown gate.
pam_heresy_pulse_cooldown_days = {
	value = 5 # Years
	multiply = 365
}

# --- Component 2 grace period (rite variable lifetime in days) ---
# Shorter than rite_growth_grace_period_days so a dangerous rite is not shielded for a decade
pam_heresy_c2_grace_period_days = {
	value = 5 # Years
	multiply = 365
}

# --- Faith size threshold ---
# Below this many main-rite counties the faith is too small to spawn the more severe heresy components
# the per-month outbreak chance is significantly reduced
pam_heresy_main_rite_county_threshold_value = 100

# --- Spike curve: chance_of_no_event weight ---
# Used in faith_heresy_events_pulse alongside event weight 1,
# so the per-month event probability is approximately 1/(no_event + 1):
# fervor 31-40: ~1%, fervor 21-30: ~3%, fervor 11-20: ~8%, fervor 0-10: ~30%.
# Christian Situation phase modifies the curve (Crisis = more outbreaks, Stability = fewer)
# Small-faith multiplier dampens the chance further when main rite < 100 counties.
# Non-Christian faiths use the base curve.
pam_heresy_pulse_no_event_value = {
	if = {
		limit = { fervor > 30 }
		value = 99
	}
	else_if = {
		limit = { fervor > 20 }
		value = 32
	}
	else_if = {
		limit = { fervor > 10 }
		value = 11
	}
	else = { value = 2 }
	# In the more turbulent phases we lower this, so outbreaks happen more often
	if = {
		limit = {
			like_christianity_religion_trigger = yes
			situation:the_christian_church ?= {
				OR = {
					situation_current_phase = saeculum_obscurum
					situation_current_phase = investiture_crisis
				}
			}
		}
		divide = 1.5
	}
	else_if = {
		limit = {
			like_christianity_religion_trigger = yes
			situation:the_christian_church ?= { situation_current_phase = concord }
		}
		multiply = 2
	}
	else_if = {
		limit = {
			like_christianity_religion_trigger = yes
			situation:the_christian_church ?= {
				OR = {
					situation_current_phase = conversion_phase
					situation_current_phase = crusade_phase
				}
			}
		}
		multiply = 1.4
	}
	# For a small faith we make outbreaks much rarer
	if = {
		limit = { NOT = { pam_heresy_main_rite_substantial_trigger = yes } }
		multiply = 4
	}
	if = {
		limit = { has_game_rule = much_less_heresy_outbreak_frequency }
		multiply = 4
	}
	else_if = {
		limit = { has_game_rule = less_heresy_outbreak_frequency }
		multiply = 2
	}
	else_if = {
		limit = { has_game_rule = more_heresy_outbreak_frequency }
		multiply = 0.5
	}
	else_if = {
		limit = { has_game_rule = much_more_heresy_outbreak_frequency }
		multiply = 0.25
	}
	min = 1
}

# --- Component 1 light-spread budget (5-10 by default, scaled by pluralism severity) ---
pam_heresy_c1_spread_budget_base_value = {
	value = 12
	if = {
		limit = { scope:origin_faith.main_rite = { rite_has_parameter = pluralism_fundamentalist_vulnerable_to_heresy } }
		multiply = 1.5
	}
	else_if = {
		limit = { scope:origin_faith.main_rite = { rite_has_parameter = pluralism_pluralistic_resistant_to_heresy } }
		multiply = 0.7
	}
	min = 1
}
