# Scarlet institutional mechanics. State belongs to the founding title.

wc_scarlet_remove_stratholme_effect = {
	title:c_stratholme = {
		if = {
			limit = { has_county_modifier = wc_balnazzar_scarlet_command_modifier }
			remove_county_modifier = wc_balnazzar_scarlet_command_modifier
		}
	}
}

wc_scarlet_grant_stratholme_effect = {
	title:c_stratholme = {
		if = {
			limit = { NOT = { has_county_modifier = wc_balnazzar_scarlet_command_modifier } }
			add_county_modifier = { modifier = wc_balnazzar_scarlet_command_modifier }
		}
	}
}

# Character scope. Both foundation routes use this, without generic costs/armies.
wc_scarlet_register_crusade_effect = {
	if = {
		limit = { NOT = { exists = global_var:wc_scarlet_crusade_title } }
		set_global_variable = { name = wc_scarlet_crusade_title value = $TITLE$ }
		$TITLE$ = { set_variable = wc_scarlet_legacy_migrated }
	}
	if = {
		limit = {
			NOT = { is_target_in_global_variable_list = { name = unavailable_unique_decisions target = flag:form_the_scarlet_crusade_decision } }
		}
		add_to_global_variable_list = { name = unavailable_unique_decisions target = flag:form_the_scarlet_crusade_decision }
	}
	add_character_flag = has_formed_the_scarlet_crusade
	if = {
		limit = { has_faith = faith:holy_light }
		set_character_faith_with_conversion = faith:scarletism
	}
	global_var:wc_scarlet_crusade_title = { set_variable = wc_scarlet_head_initialized }
	trigger_event = scarlet.1005
}

# Any scope; reactive hooks plus the current holder's quarterly pulse.
wc_scarlet_update_crusade_effect = {
	wc_scarlet_migrate_crusade_effect = yes
	if = {
		limit = { exists = global_var:wc_scarlet_crusade_title }
		global_var:wc_scarlet_crusade_title = {
			if = {
				limit = {
					# Old saves may record initialization even though the title was destroyed.
					NOT = { exists = faith:scarletism.religious_head }
					exists = holder
					holder = { is_alive = yes has_faith = faith:scarletism }
				}
				set_variable = wc_scarlet_head_initialized
				if = {
					limit = { exists = var:wc_scarlet_prefer_isillien }
					save_scope_value_as = { name = wc_scarlet_prefer_isillien value = yes }
				}
				holder = { trigger_event = scarlet.1005 }
			}
			if = {
				limit = { exists = var:wc_scarlet_stratholme_open NOT = { var:wc_scarlet_stratholme_open = holder } }
				remove_variable = wc_scarlet_stratholme_open
			}
			if = {
				limit = { exists = var:wc_scarlet_purification_open NOT = { var:wc_scarlet_purification_open = holder } }
				remove_variable = wc_scarlet_purification_open
			}
			if = {
				limit = {
					exists = var:wc_scarlet_stratholme_resolved
					exists = holder
					holder = { is_alive = yes has_faith = faith:scarletism wc_scarlet_controls_stratholme_trigger = yes }
				}
				wc_scarlet_grant_stratholme_effect = yes
			}
			else = { wc_scarlet_remove_stratholme_effect = yes }
			holder ?= {
				trigger_event = { id = wc_balnazzar_story.1008 }
				trigger_event = { id = wc_balnazzar_story.1009 }
			}
		}
	}
}
