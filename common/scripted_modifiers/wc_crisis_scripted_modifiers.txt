religion_obey_crisis_modifier = {
    modifier = {
        scope:war.casus_belli = {
            $WINNER$ = { save_temporary_scope_as = temp_winner }
            $LOSER$ = { save_temporary_scope_as = temp_loser }

            scope:temp_winner = {
                has_title = title:e_scourge
                OR = {
                    scope:temp_loser = { has_character_flag = lich_king_champion_flag }

                    has_dark_authority_level_trigger = { LEVEL = 2 }
                    has_dark_authority_level_trigger = { LEVEL = 3 }
                }
            }
            scope:temp_loser = {
                religion = religion:deathbringer
                faith = scope:temp_winner.faith
            }
        }
        add = 100

        desc = "RELIGION_OBEY_SCOURGE"
    }
}
join_coalition_scripted_modifier = {
    modifier = {
        add = -400
        always = yes
    }
	#Faith compatibility
    modifier = { 
        $CRISIS_HOLDER$.faith = {
            faith_hostility_level = {
                target = root.faith
                value > faith_fully_accepted_level
            }
        }
        NOT = {
            $CRISIS_HOLDER$.ai_zeal <= -50
        }
        add = {
            value = 10
            if = {
                limit = {
                    $CRISIS_HOLDER$.faith = {
                        faith_hostility_level = {
                            target = root.faith
                            value >= faith_astray_level
                        }
                    }
                }
                multiply = 2
            }
            if = {
                limit = {
                    $CRISIS_HOLDER$.faith = {
                        faith_hostility_level = {
                            target = root.faith
                            value >= faith_hostile_level
                        }
                    }
                }
                multiply = 3
            }
            if = {
                limit = {
                    $CRISIS_HOLDER$.faith = {
                        faith_hostility_level = {
                            target = root.faith
                            value >= faith_evil_level
                        }
                    }
                }
                multiply = 4
            }
            if = {
                limit = {
                    $CRISIS_HOLDER$.ai_zeal <= -25
                }
                multiply = 2
            }
        }
		
    }
	# Personality
	modifier = {
    	trigger = { always = yes }
     	add = {
         	value = ai_energy
        	multiply = -0.5
     	}
 	}
	modifier = {
    	trigger = { always = yes }
     	add = {
         	value = ai_honor
        	multiply = 0.5
     	}
 	}
	modifier = {
    	trigger = { always = yes }
     	add = ai_rationality
 	}
	modifier = {
    	trigger = { always = yes }
     	add = {
         	value = ai_compassion
        	multiply = 0.5
     	}
 	}
	# 30k gives 200
	modifier = {
		trigger = { always = yes }
		add = {
			value = $CRISIS_HOLDER$.max_military_strength
			multiply = {
				value = 200
				divide = 30000
			}
		}
	}
	# 10k gives 300
	modifier = {
		trigger = { always = yes }
		add = {
			value = $CRISIS_HOLDER$.max_military_strength
			subtract = max_military_strength
			multiply = {
				value = 300
				divide = 10000
			}
		}
	}
	# Opinion
	opinion_modifier = { 
		who = root
		opinion_target = $CRISIS_HOLDER$
		multiplier = -2
	}
	# Crisis specific 
	modifier = {
		trigger = {
			title:e_scourge.holder ?= $CRISIS_HOLDER$
			OR = {
				location = { geographical_region = world_eastern_kingdoms_lordaeron }
				location = { geographical_region = world_eastern_kingdoms_quelthalas }
			}
		}
		add = 75
	}
	modifier = {
		trigger = {
			title:e_horde.holder ?= $CRISIS_HOLDER$
			OR = {
				location = { geographical_region = world_eastern_kingdoms_azeroth }
				location = { geographical_region = world_eastern_kingdoms_khaz_modan_dun_morogh }
			}
		}
		add = 75
	}
	# Less likely to join against the horde / minor crisis
	modifier = {
		trigger = {
			OR = {
				title:e_horde.holder ?= $CRISIS_HOLDER$
				title:e_pandaria_empire.holder ?= $CRISIS_HOLDER$
			}
		}
		add = -50
	}
	modifier = {
		trigger = {
			is_at_war_with = $CRISIS_HOLDER$
		}
		add = 100
	}
}
