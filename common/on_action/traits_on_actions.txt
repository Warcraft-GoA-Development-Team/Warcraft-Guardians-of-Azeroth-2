# Trait on actions.

# Happens whenever a character gains a trait for any reason
# Root is the character gaining a trait
# scope:trait = trait that was gained
on_trait_gained = {
	trigger = {
		is_alive = yes
	}
	effect = {
		# Taking vows is what makes a character clergy, so it is also when they are schooled
		if = {
			limit = { scope:trait = trait:devoted }
			christian_clergy_learn_liturgical_language_effect = yes
		}
		if = {
			limit = {
				always = no # Warcraft - no GoA tenet has this flag
				scope:trait = {
					has_trait_flag = gnostic_learning_piety_trait
				}
				NOT = {
					has_character_flag = gnostic_trait_gain_cooldown
				}
			}
			#To stop the trait adds here creating a loop
			add_character_flag = {
				flag = gnostic_trait_gain_cooldown
				days = 3
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = pam_gnostic_learning_piety.t
				
				if = {
					limit = {
						has_trait = lifestyle_scholar
					}
					add_trait_xp = {
						trait = lifestyle_scholar
						value = major_trait_xp
					}
				}
				else = {
					add_trait = lifestyle_scholar
				}
				if = {
					limit = {
						has_trait = lifestyle_mystic
					}
					add_trait_xp = {
						trait = lifestyle_mystic
						value = major_trait_xp
					}
				}
				else = {
					add_trait = lifestyle_mystic
				}
			}
		}
	}
}

# Happens whenever a character loses a trait for any reason
# Root is the character losing a trait
# scope:trait = trait that was lost
on_trait_lost = {
	effect = {
		if = {
			limit = {
				scope:trait = trait:excommunicated
				has_character_flag = pope_excom
			}
			remove_character_flag = pope_excom
		}
	}
}
