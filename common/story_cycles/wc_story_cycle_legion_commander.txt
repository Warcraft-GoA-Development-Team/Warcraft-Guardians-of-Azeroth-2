story_legion_commander = {
	
	on_setup = {
		if = {
			limit = {
				NOT = {
					exists = global_var:active_legion_commanders
				}
			}
			set_global_variable = {
				name = active_legion_commanders
				value = 1
			}
		}
		else = {
			change_global_variable = {
				name = active_legion_commanders
				add = 1
			}
		}
		story_owner = {
            add_trait = legion_commander
			set_variable = legion_commander
		}
	}
	
	on_end = {
		if = {
			limit = {
				exists = global_var:active_legion_commanders
			}
			change_global_variable = {
				name = active_legion_commanders
				add = -1
			}
		}
		story_owner = {
			if = {
				limit = {
					is_alive = yes
				}
				remove_trait = legion_commander
				remove_variable = legion_commander
			}
		}
	}
	
	on_owner_death = {
		end_story = yes
	}
	
	# Something happens - random events
	effect_group = {
		days = { 365 600 }
	
		trigger = {
			story_owner = {
				top_liege = this
			}
		}
		
		triggered_effect = { # Recover slowly over time
			trigger = {
				story_owner = {
					has_variable = lost_wars
					var:lost_wars >= 2
					is_at_war = no
				}
			}
			effect = {
				story_owner = {
					random = {
						chance = 25
						change_variable = {
							name = lost_wars
							add = -1
						}
					}
				}
			}
		}
	}

	effect_group = { # Handles the AI behavior
		days = { 30 60 }

		triggered_effect = {
			trigger = {
				story_owner = {
					is_ai = yes
				}
			}
			effect = {
				story_owner = {
					save_scope_as = root
					ai_legion_commander_yearly_effect = yes
					if = { # Extra, to prevent the pacing mechanic from not working just because someone has partition
						limit = {
							OR = {
								top_liege != this
								is_tributary = yes
                                is_landed = no
							}
							is_ai = yes
						}
						trigger_event = legion_commander.0002
					}
				}
			}
		}
	}

	effect_group = { # Special bonuses for AI's who have the overpowered modifier
		days = { 90 180 }

		triggered_effect = {
			trigger = {
				story_owner = {
					is_ai = yes
				}
			}
			effect = {
				story_owner = {
					add_stress = major_stress_loss
					every_maa_regiment = { # Armies grow
						limit = {
							is_maa_in_combat = no
							can_upgrade_maa = yes
						}
						change_maa_regiment_size = {
							size = 1
							reinforce = yes
						}
					}
					random_directly_owned_province = { # Capital Duchy grows
						limit = {
							is_occupied = no
							is_raided = no
							OR = {
								this = root.story_owner.capital_province
								this.duchy = root.story_owner.capital_province.county.duchy
							}
						}
						generate_building = yes
					}
				}
			}
		}
	}

	effect_group = { # Various ways to end the cycle, for players (AI also ends it by being cautious)
		days = { 900 1800 }
	
		triggered_effect = {
			trigger = {
				story_owner = {
					OR = {
						top_liege != this
						is_landed = no
						days_of_continuous_peace >= 3650 # 10 years
						prestige_level <= -1
						debt_level >= 3
						treasury_debt_level >= 3
						AND = {
							exists = var:lost_wars
							var:lost_wars >= 5
						}
						has_trait = incapable
					}
				}
			}
			effect = {
				story_owner = {
					trigger_event = legion_commander.0002
				}
			}
		}
	}
}
