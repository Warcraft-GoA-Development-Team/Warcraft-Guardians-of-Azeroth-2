namespace = WCADMIN

# 3000 - 3999: Yearly Events

# Dryad Courtier Candidate Setup
WCADMIN.3001 = { 
	type = character_event
	content_source = dlc_GOA
	hidden = yes
	cooldown = { years = 5 }

	immediate = { 
		random_list = { 
			30 = { add_character_flag = dryad_stewardship } # Stewardship specialist
			30 = { add_character_flag = dryad_learning } # Learning specialist
			30 = { add_character_flag = dryad_diplomacy  } # Diplomacy specialist
			10 = { add_character_flag = dryad_intrigue  } # Intrigue specialist
		}
		trigger_event = WCADMIN.3002
	}
}

scripted_effect dryad_setup_effect = { 
	add_trait = loyal
	add_opinion = {
		target = root
		modifier = loyalty_opinion
	}
	root = { add_visiting_courtier = scope:dryad }
}

# Dryad Courtier Candidate Arrival
WCADMIN.3002 = { 
	type = character_event
	content_source = dlc_GOA
	theme = administrative
	title = WCADMIN.3002.title
	desc = WCADMIN.3002.desc
	
	left_portrait = {
 		character = root
		animation = chaplain
	}
	right_portrait = {
		character = scope:dryad
		animation = idle
	}
	
	immediate = { 
		if = { limit = { has_character_flag = dryad_stewardship }
			create_character = {
				location = root.location
				template = bp1_yearly_2021_stewardship_person_template
				random_culture = { 
					culture:dryad = { weight = { base = 1 } }
					culture:frostnymph = {
						trigger = { root.location = { wc_is_cold_climate_trigger = yes } }
						weight = { base = 10 }
					}
				}
				gender = female
				after_creation = {
					save_scope_as = dryad
					dryad_setup_effect = yes
				}
			}
		}
		else_if = { limit = { has_character_flag = dryad_learning }
			create_character = {
				location = root.location
				template = bp1_yearly_2021_learning_person_template
				random_culture = { 
					culture:dryad = { weight = { base = 1 } }
					culture:frostnymph = {
						trigger = { root.location = { wc_is_cold_climate_trigger = yes } }
						weight = { base = 10 }
					}
				}
				gender = female
				after_creation = {
					save_scope_as = dryad
					dryad_setup_effect = yes
				}
			}
		}
		else_if = { limit = { has_character_flag = dryad_diplomacy }
			create_character = {
				location = root.location
				template = bp1_yearly_2021_diplomacy_person_template
				random_culture = { 
					culture:dryad = { weight = { base = 1 } }
					culture:frostnymph = {
						trigger = { root.location = { wc_is_cold_climate_trigger = yes } }
						weight = { base = 10 }
					}
				}
				gender = female
				after_creation = {
					save_scope_as = dryad
					dryad_setup_effect = yes
				}
			}
		}
		else = {
			create_character = {
				location = root.location
				template = bp1_yearly_2021_intrigue_person_template
				random_culture = { 
					culture:dryad = { weight = { base = 1 } }
					culture:frostnymph = {
						trigger = { root.location = { wc_is_cold_climate_trigger = yes } }
						weight = { base = 10 }
					}
				}
				gender = female
				after_creation = {
					save_scope_as = dryad
					dryad_setup_effect = yes
				}
			}
		}
	}

	option = { 
		name = WCADMIN.3002.opt.a
		pay_short_term_gold = {
			target = scope:dryad
			gold = {
				value = minor_gold_value
				max = 25
			}
		}
		scope:dryad = { 
			add_opinion = {
				target = root
				modifier = recruited_me_opinion
			}
		}
		add_courtier = scope:dryad
		ai_chance = { 
			base = 0
			modifier = {
				add = 10
				NOT = { short_term_gold < 25 }
			}
			modifier = { 
				add = 20
				AND = { 
					OR = { has_trait = generous has_trait = zealous has_trait = improvident has_trait = profligate }
					court_positions_currently_available > 0
				}
			}
		}
	}

	option = { 
		name = WCADMIN.3002.opt.b
		add_courtier = scope:dryad
		ai_chance = { base = 10 }
	}

	option = { 
		name = WCADMIN.3002.opt.c
		ai_chance = { 
			base = 0
			modifier = {
				court_positions_currently_available = 0
				add = 100
			}
		}
	}

	after = { 
		remove_character_flag = dryad_stewardship
		remove_character_flag = dryad_leaning
		remove_character_flag = dryad_diplomacy
		remove_character_flag = dryad_intrigue
	}
}
