init_elemental_earth_spells_effect = { 
    add_to_global_variable_list = { name = elemental_earth_spells target = flag:summon_earth_elemental }
    add_to_global_variable_list = { name = elemental_earth_spells target = flag:rockbiter_weapon }
    add_to_global_variable_list = { name = elemental_earth_spells target = flag:earth_shield }
    add_to_global_variable_list = { name = elemental_earth_spells target = flag:earth_spike }
    add_to_global_variable_list = { name = elemental_earth_spells target = flag:earthquake }
    add_to_global_variable_list = { name = elemental_earth_spells target = flag:stonemend }
    add_to_global_variable_list = { name = elemental_earth_spells target = flag:volcano }
}


cast_summon_earth_elemental_effect = {
    if = {
        limit = { exists = var:summon_earth_elemental_recipient }

        if = {
            limit = {
                exists = var:summon_earth_elemental_original_recipient
            }

            var:summon_earth_elemental_original_recipient = {
                save_scope_as = character_recipient
            }
            var:summon_earth_elemental_recipient = {
                save_scope_as = title_recipient
            }
        }
        else = {
            var:summon_earth_elemental_recipient = {
                save_temporary_scope_as = character_recipient
            }
        }
    }
    else_if = {
        limit = {
            exists = var:current_spell_name
            var:current_spell_name = flag:summon_earth_elemental
            exists = var:spellbook_open
            exists = var:spell_recipient
        }

        if = {
            limit = {
                exists = var:original_recipient
            }

            var:original_recipient = {
                save_scope_as = character_recipient
            }
            var:spell_recipient = {
                save_scope_as = title_recipient
            }
        }
        else = {
            var:spell_recipient = {
                save_temporary_scope_as = character_recipient
            }
        }
    }

    if = {
        limit = {
            exists = scope:character_recipient
            OR = { # current_spell_rank is for tooltips
                AND = {
                    exists = var:summon_earth_elemental_rank
                    var:summon_earth_elemental_rank >= 1
                }
                AND = {
                    exists = var:current_spell_rank
                    var:current_spell_rank >= 1
                }
            }
        }

        root = {
            save_temporary_scope_value_as = {
                name = duration
                value = wc_spell_summon_earth_elemental_duration_days_value
            }
        }
        scope:character_recipient = {
            add_character_modifier = {
                modifier = wc_summon_earth_elemental_modifier
                days = wc_current_spell_duration
            }
        }

        if = {
            limit = {
                OR = {
                    AND = {
                        exists = var:summon_earth_elemental_rank
                        var:summon_earth_elemental_rank >= 2
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank >= 2
                    }
                }
            }
            scope:character_recipient = {
                add_character_flag = {
                    flag = has_earth_elemental_maa
                    days = 735
                }

                spawn_army = {
                    uses_supply = no
                    inheritable = no
                    name = summoned_earth_elemental
                    men_at_arms = {
                        type = earth_elemental
                        stacks = 1
                    }
                    location = scope:title_recipient.title_province
                }
            }
            if = {
                limit = {
                    OR = {
                        AND = {
                            exists = var:summon_earth_elemental_rank
                            var:summon_earth_elemental_rank = 3
                        }
                        AND = {
                            exists = var:current_spell_rank
                            var:current_spell_rank = 3
                        }
                    }
                }

                scope:character_recipient = {
                    add_character_modifier = {
                        modifier = wc_empowered_earth_elemental_modifier
                        days = wc_current_spell_duration
                    }
                }
            }
        }
    }
}

cast_rockbiter_weapon_effect = {
    get_elemental_disposition_effect = { SPELL = rockbiter_weapon ELEMENT = elemental_earth }
    get_recipient_effect = { SPELL = rockbiter_weapon }

    scope:this_recipient ?= {
        root = {
            save_temporary_scope_value_as = {
                name = duration
                value = wc_spell_rockbiter_weapon_duration_days_value
            }
        }
        set_variable = {
            name = rockbiter_weapon_rank
            value = root.var:rockbiter_weapon_rank
            days =  wc_current_spell_duration
        }
        switch = {
            trigger = scope:elemental_disposition_elemental_earth
            flag:decay = {
                add_character_modifier = {
                    modifier = wc_rockbiter_weapon_decay_modifier
                    days = wc_current_spell_duration
                }
            } 
            flag:spirit = {
                add_character_modifier = {
                    modifier = wc_rockbiter_weapon_modifier
                    days = wc_current_spell_duration
                }
                add_character_modifier = {
                    modifier = wc_earth_shield_modifier
                    days = wc_current_spell_duration 
                } 
            } 
            flag:none  = {
                add_character_modifier = {
                    modifier = wc_rockbiter_weapon_modifier
                    days = wc_current_spell_duration
                }
            } 
        }
    }
}

cast_earthquake_effect = {

    scope:this_recipient ?= {

                    AND = {

                        exists = var:current_spell_rank
                        var:current_spell_rank = 2
                    }
                }
            }
            every_title_province = {
                add_province_modifier = wc_minor_earthquake
            }
        }
        else_if = {
            limit = {

                    AND = {
                        exists = var:earth_spike_rank
                        var:summon_earth_spike_rank = 3
                    }

cast_earth_spike_effect = {
    get_elemental_disposition_effect = { SPELL = earth_spike ELEMENT = elemental_earth }
    get_recipient_effect = { SPELL = earth_spike }

    scope:this_recipient ?= {
        if = {
            limit = {
                is_character_trigger = yes
            }
            increase_wounds_effect = { REASON = magic }
        }
        else_if = {
            limit = {
                is_army_trigger = yes
            }
            deplete_army_by_percentage = root.earth_spike_depletion_val
            if = {
                limit = {
                    root = {
                        OR = { # current_spell_rank is for tooltips
                            AND = {
                                exists = var:earth_spike_rank
                                var:earth_spike_rank = 3
                            }
                            AND = {
                                exists = var:current_spell_rank
                                var:current_spell_rank = 3
                            }
                        }
                    }
                }
                random = {
                    chance = {
                        value = { 30 50 }
                        if = {
                            limit = {
                                scope:elemental_disposition_elemental_earth = flag:decay
                            }
                            add = { 10 25 }
                        }
                    }
                    army_commander = {
                        increase_wounds_effect = { REASON = magic }
                    }
                    random_knight = {
                        limit = {
                            knight_army = prev
                        }
                        increase_wounds_effect = { REASON = magic }
                    }
                    random_knight = {
                        limit = {
                            knight_army = prev
                        }
                        increase_wounds_effect = { REASON = magic }
                    }
                }
            }
        }
    }
}

cast_stonemend_effect = {
    get_elemental_disposition_effect = { SPELL = stonemend ELEMENT = elemental_earth }
    get_recipient_effect = { SPELL = stonemend }
    
    scope:this_recipient ?= {
        if = {
            limit = {
                scope:elemental_disposition_elemental_earth = flag:decay
            }
            random = {
                chance = 1
                add_trait = scaly
            }
        }
        else_if = {
            limit = {
                scope:elemental_disposition_elemental_earth = flag:spirit
                root = {
                    can_add_hook = {
                        target = prev
                        type = favor_hook 
                    }
                }
            }
            random = {
                chance = 33
                root = {
                    add_hook = {
                        target = prev
                        type = favor_hook
                    }
                }
            }
        }



        if = {
            limit = {
                OR = { # current_spell_rank is for tooltips
                    AND = {
                        exists = var:stonemend_rank
                        var:stonemend_rank = 3
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank = 3
                    }
                }
            }
            scope:this_recipient = {
                if = {
                    limit = {
                        target_has_major_injury_trigger = yes
                    }
                    remove_major_injury_effect = yes
                }
                else = {
                    custom_tooltip = wc_dispel_no_effect_tooltip
                }
            }
        }
        else_if = {
            limit = {
                OR = {
                    AND = {
                        exists = var:stonemend_rank
                        var:stonemend_rank = 2
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank = 2
                    }
                }
            }
            scope:this_recipient = {
                if = {
                    limit = {
                        target_has_poison_trigger = yes
                    }
                    remove_poison_effect = yes
                }
                else = {
                    custom_tooltip = wc_dispel_no_effect_tooltip
                }
            }
        }
        else_if = {
            limit = {
                OR = {
                    AND = {
                        exists = var:stonemend_rank
                        var:stonemend_rank = 1
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank = 1
                    }
                }
            }
            scope:this_recipient = {
                if = {
                    limit = {
                        target_has_injury_trigger = yes
                    }
                    remove_injury_effect = { REDUCE_SEVERITY = yes }
                }
                else = {
                    custom_tooltip = wc_dispel_no_effect_tooltip
                }
            }
        }
    }

}

cast_volcano_effect = {
    get_recipient_effect = { SPELL = volcano }

    scope:this_recipient.title_province = {
        create_epidemic_outbreak = { type = wc_volcanic_erruption intensity = major }
    }
}
