has_magic_education_focus_trigger = {
    has_focus = education_light_magic
}
#Affinity/disaffinity
#For children
has_magic_education_affinity_childhood_trait_trigger = {
    OR = {
        has_trait = magic_good_noninheritable
        has_trait = magic_good_1
        has_trait = magic_good_2
        has_trait = magic_good_3
    }
}

has_magic_education_disaffinity_childhood_trait_trigger = {
    OR = {
        has_trait = magic_bad_noninheritable
        has_trait = magic_bad_1
        has_trait = magic_bad_2
        has_trait = magic_bad_3
    }
}

has_light_magic_education_affinity_childhood_trait_trigger = {
    has_magic_education_affinity_childhood_trait_trigger = yes
    any_parent = {
        even_if_dead = yes
        is_magic_class_user_trigger_no_secret = { MAGIC = light }
    }
}

has_light_magic_education_disaffinity_childhood_trait_trigger = {
    has_magic_education_disaffinity_childhood_trait_trigger = yes
    any_parent = {
        even_if_dead = yes
        is_magic_class_user_trigger_no_secret = { MAGIC = shadow }
    }
}
