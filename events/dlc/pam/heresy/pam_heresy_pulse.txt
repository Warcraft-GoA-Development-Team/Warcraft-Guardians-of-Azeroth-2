namespace = heresy_pulse

####################################################################
# Heresy outbreak system - Router                                  #
####################################################################
# Faith pulse: faith_heresy_events_pulse runs once a month per faith and
# fires heresy_pulse.0001 when fervor is critically low.
# The router picks Component 1 (light divergence), 2 (dangerous new rite), or 3 (historical heresy, Christianity-only)
# using pluralism + Christian Situation phase weights, and sets the cooldown variable that gates re-runs.

#####################################
# 0001 - Component selection router #
#####################################

heresy_pulse.0001 = {
	hidden = yes
	scope = faith

	immediate = {
		save_scope_as = origin_faith

		random_list = {
			# Component 1 - light rite divergence (most common)
			50 = {
				trigger = {
					any_faith_rite = {
						this != root.main_rite
						rite_counties >= 1
						head_of_rite ?= { is_ai = yes }
					}
				}
				# Pluralism severity scaling
				modifier = {
					factor = 0.2
					main_rite = { rite_has_parameter = pluralism_fundamentalist_vulnerable_to_heresy }
				}
				modifier = {
					factor = 1.6
					main_rite = { rite_has_parameter = pluralism_pluralistic_resistant_to_heresy }
				}
				# Christian Situation phase: Concord/Crusade favour C1, Crisis dampens
				# Warcraft
				# modifier = {
				# 	factor = 0.7
				# 	religion = religion:christianity_religion
				# 	situation:the_christian_church ?= {
				# 		OR = {
				# 			situation_current_phase = saeculum_obscurum
				# 			situation_current_phase = investiture_crisis
				# 		}
				# 	}
				# }
				# modifier = {
				# 	factor = 2
				# 	religion = religion:christianity_religion
				# 	situation:the_christian_church ?= { situation_current_phase = concord }
				# }
				# modifier = {
				# 	factor = 1.5
				# 	religion = religion:christianity_religion
				# 	situation:the_christian_church ?= {
				# 		OR = {
				# 			situation_current_phase = conversion_phase
				# 			situation_current_phase = crusade_phase
				# 		}
				# 	}
				# }
				scope:origin_faith = {
					trigger_event = heresy_light.0001
					pam_heresy_set_pulse_cooldown_effect = yes
				}
			}

			# Component 2 - new dangerously divergent rite (rare)
			20 = {
				# Hard 60% main-rite share gate.
				trigger = {
					pam_heresy_c2_main_rite_dominant_trigger = yes
				}
				# Pluralism severity scaling
				modifier = {
					factor = 1
					main_rite = { rite_has_parameter = pluralism_fundamentalist_vulnerable_to_heresy }
				}
				modifier = {
					factor = 0.5
					main_rite = { rite_has_parameter = pluralism_pluralistic_resistant_to_heresy }
				}
				# Phase: Crisis ramps C2, Concord dampens
				# Warcraft
				# modifier = {
				# 	factor = 1.5
				# 	religion = religion:christianity_religion
				# 	situation:the_christian_church ?= {
				# 		OR = {
				# 			situation_current_phase = saeculum_obscurum
				# 			situation_current_phase = investiture_crisis
				# 		}
				# 	}
				# }
				# modifier = {
				# 	factor = 0.7
				# 	religion = religion:christianity_religion
				# 	situation:the_christian_church ?= { situation_current_phase = concord }
				# }
				# modifier = {
				# 	factor = 1.3
				# 	religion = religion:christianity_religion
				# 	situation:the_christian_church ?= {
				# 		OR = {
				# 			situation_current_phase = fragile_unity
				# 			situation_current_phase = reform_phase
				# 		}
				# 	}
				# }
				modifier = {
					factor = 0.1
					rite_growth_rites_needed <= 0
				}
				scope:origin_faith = {
					trigger_event = heresy_dangerous_rite.0001
					pam_heresy_set_pulse_cooldown_effect = yes
				}
			}

			# Component 3 - historical heresy outbreak (Christianity-only)
			10 = {
				trigger = {
					pam_heresy_c3_eligible_trigger = yes
				}
				# Pluralism severity scaling (C3 is the most pluralism-sensitive)
				modifier = {
					factor = 3
					main_rite = { rite_has_parameter = pluralism_fundamentalist_vulnerable_to_heresy }
				}
				modifier = {
					factor = 0.5
					main_rite = { rite_has_parameter = pluralism_pluralistic_resistant_to_heresy }
				}
				# Phase: Crisis is prime time for historical heresies; Concord/Crusade dampen
				modifier = {
					factor = 2
					situation:the_christian_church ?= {
						OR = {
							situation_current_phase = saeculum_obscurum
							situation_current_phase = investiture_crisis
						}
					}
				}
				modifier = {
					factor = 0.5
					situation:the_christian_church ?= { situation_current_phase = concord }
				}
				modifier = {
					factor = 0.7
					situation:the_christian_church ?= {
						OR = {
							situation_current_phase = conversion_phase
							situation_current_phase = crusade_phase
						}
					}
				}
				modifier = {
					factor = 1.3
					situation:the_christian_church ?= {
						OR = {
							situation_current_phase = fragile_unity
							situation_current_phase = reform_phase
						}
					}
				}
				scope:origin_faith = {
					trigger_event = heresy_historical.0001
					pam_heresy_set_pulse_cooldown_effect = yes
				}
			}
		}
	}
}
