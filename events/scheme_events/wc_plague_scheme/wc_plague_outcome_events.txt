plague_outcome.0001 = {
	type = character_event
	content_source = dlc_GOA
	hidden = yes

	immediate = {
		scope:target.capital_county = { save_scope_as = plague_center }

		scope:scheme = {
			every_scheme_agent_character = {
				trigger_event_plague_outcome_0002_effect = { OWNER = scope:owner TARGET = scope:target PLAGUE_CENTER = scope:plague_center }
			}
		}
		trigger_event_plague_outcome_0002_effect = { OWNER = scope:owner TARGET = scope:target PLAGUE_CENTER = scope:plague_center }
	}
}

# Owner POV — Success
plague_outcome.0002 = {
	type = character_event
	content_source = dlc_GOA
	window = scheme_successful_event
	title = plague_outcome.0002.title
	desc = {
		desc = plague_outcome.0002.desc
		
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:scheme_discovered
				}
				desc = plague_outcome.0002.desc.exposed_scheme
			}
			desc = plague_outcome.0002.desc.not_exposed_scheme
		}
	}
	theme = plague
	right_portrait = scope:target

	immediate = {
		# General owner success effects.
		scheme_owner_plague_success_effect = yes

		# Launch the plague
		plague_success_effect = yes
	}

	option = {
		name = plague_outcome.0002.a
	}

	# Fire the rest of the outcome.
	after = {
		scope:target = { trigger_event = plague_outcome.0003 }
	}
}

# Target POV — Success
plague_outcome.0003 = {
	type = character_event
	content_source = dlc_GOA
	window = scheme_target_event
	title = plague_outcome.0003.title
	desc = {
		# Opening.
		## Plan.
		desc = plague_outcome.0003.desc
		# Secrecy.
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:scheme_discovered
				}
				desc = plague_outcome.0003.desc.exposed_scheme
			}
			desc = plague_outcome.0003.desc.not_exposed_scheme
		}
	}
	theme = plague
	right_portrait = root

	immediate = {
		plague_success_victim_immediate_effect = yes
		add_character_flag = { flag = plague_notification_timer years = 2 }

		force_owner_to_reveal_use_stealth_clothes_effect = yes
		if = {
			limit = {
				NOT = { is_target_in_global_variable_list = { name = unavailable_unique_events target = flag:arthas_expedition } }
				frostmourne_can_be_found_trigger = yes
				game_start_date <= 603.1.1

				is_independent_ruler = yes
				is_vulnerable_to_plague_character_trigger = yes
				capital_province = { geographical_region = world_eastern_kingdoms_lordaeron }
			}
			add_to_global_variable_list = {
				name = unavailable_unique_events
				target = flag:arthas_expedition
			}
			add_to_global_variable_list = {
				name = unavailable_unique_events
				target = flag:frostmourne_ongoing
			}

			save_scope_value_as = { name = arthas_exists value = yes }
			find_arthas_effect = yes
			primary_title = { save_scope_as = lordaeron }
		}
		scope:plague_center.title_province = {
			set_global_variable = { name = andorhal_province value = this }
		}
		find_hearthglen_effect = yes
		scope:hearthglen = { set_global_variable = { name = hearthglen value = this } }
		find_stratholme_effect = yes
		scope:stratholme = { set_global_variable = { name = stratholme value = this } }
		add_to_global_variable_list = {
			name = unavailable_unique_events
			target = flag:arthas_activity_good_to_go
		}

		global_var:arthas_character = {
			if = { limit = { is_ai = yes } add_character_flag = { flag = wc_cannot_serve_in_army years = 1 } }
			else = { add_character_flag = { flag = wc_cannot_serve_in_army weeks = 10 } }
			trigger_event = wc_arthas_story.9
		}
	}

	option = {
		trigger = {
			exists = scope:arthas_exists
			exists = global_var:terenas_character
			root = global_var:terenas_character
		}
		name = wc_arthas_story_9_1_OPT_A

		ai_chance = {
			base = 100
		}
	}

	option = {
		name = wc_arthas_story_9_1_OPT_B
		trigger = {
			can_be_in_coalition_trigger = yes
			NOT = { is_in_coalition_against_trigger = { CRISIS_HOLDER = title:e_scourge.holder } }
		}
		join_crisis_coalition_effect = { CRISIS_HOLDER = title:e_scourge.holder }
		ai_chance = {
			base = 0
			join_coalition_scripted_modifier = { CRISIS_HOLDER = title:e_scourge.holder  }
		}
	}
}

# Owner POV — Failure
plague_outcome.0004 = {
	type = character_event
	content_source = dlc_GOA
	window = scheme_failed_event
	title = plague_outcome.0004.title
	desc = {
		# Opening
		random_valid = {
			desc = plague_outcome.0004.desc.failure_declaration_1
			desc = plague_outcome.0004.desc.failure_declaration_2
			desc = plague_outcome.0004.desc.failure_declaration_3
		}
		# What was our plan?
		desc = plague_outcome.0004.desc
		# Secrecy.
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:scheme_discovered
				}
				desc = plague_outcome.0004.desc.exposed_scheme
			}
			desc = plague_outcome.0004.desc.not_exposed_scheme
		}
	}
	theme = plague
	left_portrait = {
		character = scope:owner
		animation = rage
	}
	right_portrait = {
		character = scope:target
		animation = shock
	}

	immediate = {
		show_as_tooltip = { plague_failure_effect = yes }
	}

	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { exists = scope:scheme_discovered }
						desc = plague_outcome.0004.a.discovered
					}
					desc = plague_outcome.0004.a
				}
			}
		}
		scope:scheme = { end_scheme = yes }
	}

	option = {
		name = plague_outcome.0004.b
		trigger = {
			NOT = { exists = scope:scheme_discovered }
		}
		restart_plague_scheme_effect = yes
	}

	# Fire the rest of the outcome.
	after = {
		scope:target = { trigger_event = plague_outcome.0005 }
	}
}
