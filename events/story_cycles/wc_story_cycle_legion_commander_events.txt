namespace = legion_commander

legion_commander.0001 = {
    type = character_event
    content_source = dlc_GOA
    title = legion_commander_0001_title
    desc = legion_commander_0001_desc
    theme = war

    right_portrait = {
		character = root
		animation = manic
	}

    trigger = {
		top_liege = this
		is_landed = yes
		NOR = {
			has_trait = legion_commander
			any_owned_story = {
				story_type = story_legion_commander
			}
            has_character_flag = used_to_be_legion_commander
		}
	}

    immediate = {
		save_scope_as = root_scope
		if = { # Starter package!
			limit = {
				is_ai = yes
			}
			add_treasury_or_gold = 500
			add_prestige = 800
		}
	}

    option = {
		name = legion_commander_0001_a
		add_trait = legion_commander
		create_story = story_legion_commander
		if = {
			limit = {
				any_vassal = { vassal_stance = belligerent }
			}
			every_vassal = {
				vassal_stance = belligerent
				custom = every_belligerent_vassal
				add_opinion = {
					modifier = approving_opinion
					target = root
					opinion = 15
				}
			}
		}
		if = {
			limit = {
				any_vassal = { vassal_stance = glory_hound }
			}
			every_vassal = {
				vassal_stance = glory_hound
				custom = every_glory_hound_vassal
				add_opinion = {
					modifier = approving_opinion
					target = root
					opinion = 15
				}
			}
		}
		if = {
			limit = {
				any_vassal = { vassal_stance = parochial }
			}
			every_vassal = {
				custom = every_parochial_vassal
				add_opinion = {
					modifier = disapproving_opinion
					opinion = -20
					target = root
				}
			}
		}
		ai_chance = {
			base = 100 
		}
	}

	option = { 
		name = legion_commander_0001_b
		add_prestige = minor_prestige_loss
		if = {
			limit = {
				any_vassal = { vassal_stance = belligerent }
			}
			every_vassal = {
				vassal_stance = belligerent
				custom = every_belligerent_vassal
				add_opinion = {
					modifier = approving_opinion
					target = root
					opinion = -15
				}
			}
		}
		if = {
			limit = {
				any_vassal = { vassal_stance = glory_hound }
			}
			every_vassal = {
				vassal_stance = glory_hound
				custom = every_glory_hound_vassal
				add_opinion = {
					modifier = disapproving_opinion
					target = root
					opinion = -15
				}
			}
		}
		if = {
			limit = {
				any_vassal = { vassal_stance = parochial }
			}
			every_vassal = {
				custom = every_parochial_vassal
				add_opinion = {
					modifier = approving_opinion
					opinion = 20
					target = root
				}
			}
		}
		
		ai_chance = {
			base = 0 
		}
	}
}

legion_commander.0002 = {
    type = character_event
    content_source = dlc_GOA
    title = legion_commander_0002_title
    desc = legion_commander_0002_desc
    theme = war

    right_portrait = {
		character = root
		animation = fear
	}

    immediate = {
		save_scope_as = root_scope
		every_owned_story = {
			type = story_legion_commander
			end_story = yes
		}
		add_character_flag = used_to_be_legion_commander
	}

    option = {
		name = legion_commander_0002_a
		remove_trait = legion_commander
		add_prestige_level = -2
        show_as_tooltip = {
			random_list = {
				50 = {
                    death = { death_reason = death_torture }
                    custom_tooltip = legion_commander_0002_a_die
				}
				50 = {
                    custom_tooltip = legion_commander_0002_a_live
				}
			}
		}
		if = {
			limit = {
				any_vassal = { vassal_stance = glory_hound }
			}
			every_vassal = {
				vassal_stance = glory_hound
				custom = every_glory_hound_vassal
				add_opinion = {
					modifier = disappointed_opinion
					target = root
					opinion = -50
				}
			}
		}
		if = {
			limit = {
				any_vassal = { has_vassal_stance = belligerent }
			}
			every_vassal = {
				limit = { has_vassal_stance = belligerent }
				custom = every_belligerent_vassal
				add_opinion = {
					modifier = disappointed_opinion
					target = root
					opinion = -50
				}
			}
		}
	}
}
