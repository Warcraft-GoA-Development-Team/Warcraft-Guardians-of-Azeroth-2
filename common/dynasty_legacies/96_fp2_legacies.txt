fp2_urbanism_legacy_track = {
	is_shown = {
		has_dlc_feature = the_fate_of_iberia
		OR = {
			has_game_rule = unrestricted_dynasty_legacies_all
			AND = {
				has_game_rule = unrestricted_dynasty_legacies_player_only
				is_ai = no
			}
			dynasty = {
				OR = {
					dynast = { # Warcraft - We don't use struggles
						culture = {
							OR = {
								has_cultural_pillar = heritage_arathi
								has_cultural_pillar = heritage_goblin
								has_cultural_pillar = heritage_gnomish
							}
						}
					}
					has_dynasty_perk = fp2_urbanism_legacy_1
				}
			}
		}
	}
}
