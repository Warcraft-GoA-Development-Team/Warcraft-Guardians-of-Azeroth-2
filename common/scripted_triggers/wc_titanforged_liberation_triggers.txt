	culture = culture:titanforged

	owns_building_titan_forge = yes

# County scope; conformity is independent of corruption, development and control.

	OR = {
		NOT = { culture = culture:titanforged }
		NOT = { faith = faith:mystery_of_the_makers }
	}
}

wc_titanforged_historical_county_trigger = {
	wc_titanforged_has_restoration_target_trigger = yes
	faith = { is_chaos_aligned_faith_trigger = yes }
	scope:councillor_liege = { faith = { is_titanforged_liberator_faith_trigger = yes } }
}

# County scope; shared by assignment, monthly tick and completion.
wc_titanforged_reorder_target_trigger = {
	trigger_if = {
		limit = {
			exists = scope:councillor_liege.var:wc_titanforged_restoration
			scope:councillor_liege.var:wc_titanforged_restoration = {
				NOT = { var:wc_titanforged_restoration_method = flag:pending }
				# A different assignment is free to choose another method.
				OR = {
					AND = {
						has_variable = wc_titanforged_restoration_county
						var:wc_titanforged_restoration_county = scope:county
					}
					AND = {
						NOT = { has_variable = wc_titanforged_restoration_county }
						exists = scope:councillor
						scope:councillor = {
							is_performing_council_task = task_reorder_county
							councillor_task_target = scope:county.title_province
						}
					}
				}
			}
		}
		trigger_if = {
			limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:implantation } }
			scope:councillor_liege = { wc_titanforged_implantation_ruler_trigger = yes }
			wc_titanforged_implantation_county_trigger = yes
			trigger_if = {
				limit = { scope:councillor_liege.var:wc_titanforged_restoration = { has_variable = wc_titanforged_project_version } }
				wc_titanforged_forge_served_trigger = yes
			}
		}
		trigger_else_if = {
			limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:arcane } }
			wc_titanforged_arcane_target_trigger = yes
		}
		trigger_else = {
			scope:councillor_liege.var:wc_titanforged_restoration = {
				OR = {
					var:wc_titanforged_restoration_method = flag:pact
					var:wc_titanforged_restoration_method = flag:mandate
					var:wc_titanforged_restoration_method = flag:legacy
				}
			}
			wc_titanforged_historical_county_trigger = yes
		}
	}
	trigger_else = {
		OR = {
			wc_titanforged_arcane_target_trigger = yes
			wc_titanforged_historical_county_trigger = yes
			AND = {
				scope:councillor_liege = { wc_titanforged_implantation_ruler_trigger = yes }
				wc_titanforged_implantation_county_trigger = yes
				wc_titanforged_forge_served_trigger = yes
			}
		}
	}
	scope:councillor_liege = {
		completely_controls = scope:county
		trigger_if = {
			limit = { exists = var:wc_titanforged_restoration }
			var:wc_titanforged_restoration = { story_owner = scope:councillor_liege }
		}
	}
	NOT = {
		this = title:c_lordaeron
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:liberate_lordaeron_from_the_scourge_decision
		}
	}
}

# County scope. Assignment callbacks must not borrow another county's method.

			}
		}
	}
}

# Event recipient is the ruler. Saved story must still be this ruler's active project.
wc_titanforged_restoration_audience_valid_trigger = {
	exists = scope:restoration
	exists = var:wc_titanforged_restoration
	var:wc_titanforged_restoration = scope:restoration
	scope:restoration = {
		story_owner = root
		var:wc_titanforged_restoration_method = flag:pending
	}
	wc_titanforged_reorder_active_trigger = yes
}

# Side effects must still belong to this marshal's uninterrupted assignment.
wc_titanforged_reorder_active_trigger = {
	exists = scope:councillor
	exists = scope:councillor_liege
	exists = scope:county
	scope:councillor_liege = { cp:councillor_marshal = scope:councillor }
	scope:councillor = {
		is_performing_council_task = task_reorder_county
		councillor_task_target = scope:county.title_province
	}
	scope:county = {
		wc_titanforged_reorder_target_trigger = yes
		wc_titanforged_reorder_assignment_trigger = yes
		has_variable = wc_titanforged_reorder_current_progress
		has_county_modifier = wc_titanforged_ongoing_reordering_county_modifier
		has_variable = wc_titanforged_reorder_started
		exists = scope:reorder_started
		var:wc_titanforged_reorder_started = scope:reorder_started
	}
}

# Duchy scope; root is the ruler making the decision.
wc_titanforged_duchy_ready_trigger = {
	tier = tier_duchy
	# An active settlement is not a political restoration, even in a marked duchy.
	NOT = {
		AND = {
			exists = root.var:wc_titanforged_restoration
			root.var:wc_titanforged_restoration = {
				var:wc_titanforged_restoration_method = flag:implantation
				var:wc_titanforged_restoration_duchy = prev
			}
		}
	}
	root = {
		faith = { is_titanforged_liberator_faith_trigger = yes }
		is_capable_adult = yes
		is_imprisoned = no
		completely_controls = prev
	}
	any_de_jure_county = { has_county_modifier = wc_titanforged_reordered_county_modifier }
	NOT = { any_de_jure_county = { faith = { is_chaos_aligned_faith_trigger = yes } } }
	NOT = { any_de_jure_county = { development_level < 10 } }
	NOT = { any_de_jure_county = { county_control < 80 } }
}

# Character scope. Include baronies and vassal territory, not only personal counties.
wc_titanforged_local_ruler_trigger = {
	is_alive = yes
	this != root
	is_vassal_or_below_of = root
	highest_held_title_tier < tier_kingdom
	culture = scope:ceded_vassal_county.culture
	faith = scope:ceded_vassal_county.faith
	scope:ceded_vassal_county = {
		holder = prev
		duchy = scope:ceded_duchy
		has_county_modifier = wc_titanforged_reordered_county_modifier
	}
	NOT = {
		any_held_title = {
			NOT = { this = scope:ceded_duchy }
			NOT = { target_is_de_jure_liege_or_above = scope:ceded_duchy }
		}
	}
	NOT = {
		any_realm_province = {
			NOT = { county.duchy = scope:ceded_duchy }
		}
	}
}

wc_titanforged_can_resolve_duchy_trigger = {
	exists = scope:ceded_duchy
	scope:ceded_duchy = { wc_titanforged_duchy_ready_trigger = yes }
}

# Check the engine result before consuming the cycle or granting rewards.
wc_titanforged_released_duchy_trigger = {
	scope:local_ruler = { is_independent_ruler = yes }
	scope:ceded_duchy = {
		holder = scope:local_ruler
		NOT = { any_de_jure_county = { holder != scope:local_ruler } }
	}
}

wc_titanforged_can_release_duchy_trigger = {
	wc_titanforged_can_resolve_duchy_trigger = yes
	exists = scope:local_ruler
	exists = scope:ceded_vassal_county
	scope:local_ruler = { wc_titanforged_local_ruler_trigger = yes }
	# County transfers may include baron vassals, never outlying territory.
	scope:ceded_duchy = {
		NOT = {
			any_de_jure_county = {
				any_in_de_facto_hierarchy = {
					NOT = { target_is_de_jure_liege_or_above = scope:ceded_duchy }
				}
			}
		}
	}
}
