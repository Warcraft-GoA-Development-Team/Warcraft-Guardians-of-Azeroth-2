# Canonizer is root, and this expects scope:saint_candidate and consecration_province already saved
pam_commit_canonization_effect = {
	if = {
		limit = { pam_needs_hof_approval_for_canonization_trigger = yes }
		faith.religious_head = { save_scope_as = petition_recipient }
		# Pick the wording here so saint_candidate / consecration_province / petition_recipient stay reachable in the loc key (SelectLocalization would drop them).
		if = {
			limit = { faith.religion = religion:tyrism_religion } #Warcraft
			custom_tooltip = pam_canonize_saint_petition_sent_tt_saint
		}
		else = {
			custom_tooltip = pam_canonize_saint_petition_sent_tt_va
		}
		# Players acknowledge via event 0001, whose option starts the HoF's clock (0002 -> 0003/0004). The AI never sees that event, so schedule the HoF decision directly for it.
		if = {
			limit = { is_ai = no }
			trigger_event = pam_saint_events.0001
		}
		else = {
			hidden_effect = {
				scope:petition_recipient = {
					trigger_event = {
						id = pam_saint_events.0002
						days = 30
					}
				}
			}
		}
	}
	else = {
		custom_tooltip = pam_canonize_saint_decision_canonize_tt
		pam_canonize_saint_effect = {
			CANONIZER = root
			CANDIDATE = scope:saint_candidate
			CONSECRATION_PROVINCE = scope:consecration_province
			PRESET = no
		}
		# No HoF (or self-HoF): canonization is done, so show the finalization event now (players only).
		if = {
			limit = { is_ai = no }
			trigger_event = pam_saint_events.0003
		}
	}
}

# The actual canonization.
# CANONIZER - the ruler putting the saint forward (not root; also run from the HoF approval event).
# CANDIDATE - the dead character being canonized.
# CONSECRATION_PROVINCE - where they are consecrated / buried.
# PRESET - yes for history burials, so the tomb gives the lighter travel homage.
pam_canonize_saint_effect = {
	$CANDIDATE$ = {
		add_trait = saint
	}
	# Register on the canonizer's faith for the Saints-list UI and faith-split; add_saint needs the saint trait (added above) and a burial province with a holding.
	$CANONIZER$.faith ?= {
		hidden_effect = {
			add_saint = {
				character = $CANDIDATE$
				burial_province = $CONSECRATION_PROVINCE$
			}
		}
	}

	if = {
		limit = {
			$CANDIDATE$ = {
				is_alive = no
				has_dead_character_variable = no_corpse_left
			}
		}
		$CONSECRATION_PROVINCE$ = {
			set_variable = no_corpse_left
		}
	}

	# The engine's saint registry (add_saint, above) owns the saint->province link (saint_burial_location) and the relic charges (remaining_saint_relics). All we keep is the buried-saint travel PoI, plus a thin province list the PoI builder iterates.
	$CONSECRATION_PROVINCE$ = {
		# Only readable through the tomb PoI, so skip provinces that will never get one
		if = {
			limit = {
				$PRESET$ = yes
				NOT = { has_variable = hide_location_for_saint }
			}
			set_variable = pam_preset_buried_saint
		}
		if = {
			limit = {
				NOR = {
					has_variable = hide_location_for_saint
					has_travel_point_of_interest = poi_buried_saint
				}
			}
			add_to_global_variable_list = {
				name = pam_buried_saint_poi_list
				target = $CONSECRATION_PROVINCE$
			}
			add_travel_point_of_interest = poi_buried_saint
		}
	}

	if = {
		limit = {
			$CANONIZER$ = { is_alive = yes }
			$CANONIZER$.rite ?= { rite_has_parameter = tenet_dulia_canonization_faith_bonus }
		}
		pam_grant_dulia_canonization_faith_bonus_effect = {
			CANONIZER = $CANONIZER$
			CANDIDATE = $CANDIDATE$
		}
	}

	# Cost of the next canonization scales with how many saints the dynasty has.
	$CANDIDATE$.dynasty ?= {
		if = {
			limit = { has_variable = pam_dynasty_saint_count }
			change_variable = {
				name = pam_dynasty_saint_count
				add = 1
			}
		}
		else = {
			set_variable = {
				name = pam_dynasty_saint_count
				value = 1
			}
		}
	}
}

# Character scope. Saves consecration_province from a free cathedral, holy site, or adjacent county capital.
pam_pick_actor_consecration_province_effect = {
	save_scope_as = pam_burial_site_actor
	if = {
		limit = { pam_has_free_primary_site_trigger = yes }
		if = {
			limit = { pam_faith_uses_cathedrals_trigger = yes }
			if = {
				limit = {
					pam_character_has_cathedral_trigger = yes
					domicile ?= { domicile_location = { pam_province_is_free_burial_site_trigger = yes } }
				}
				domicile = { domicile_location = { save_scope_as = consecration_province } }
			}
			else = {
				random_realm_province = {
					limit = {
						pam_province_is_free_burial_site_trigger = yes
						any_province_domicile = { owner = { pam_character_has_cathedral_trigger = yes } }
					}
					save_scope_as = consecration_province
				}
			}
		}
		else = {
			random_realm_province = {
				limit = {
					barony ?= { is_holy_site_of = scope:pam_burial_site_actor.faith }
					pam_province_is_free_burial_site_trigger = yes
				}
				save_scope_as = consecration_province
			}
		}
	}
	else_if = {
		limit = { pam_has_free_adjacent_site_trigger = yes }
		random_realm_province = {
			limit = {
				is_county_capital = yes
				pam_province_is_free_burial_site_trigger = yes
				county = {
					any_neighboring_county = {
						pam_county_has_actor_primary_site_trigger = { ACTOR = scope:pam_burial_site_actor }
					}
				}
			}
			save_scope_as = consecration_province
		}
	}
}

# Root = the dying monastic founder. Saves consecration_province if a free burial site exists.
pam_pick_founder_burial_province_effect = {
	if = {
		limit = {
			exists = capital_province
			capital_province = { pam_province_is_free_burial_site_trigger = yes }
		}
		capital_province = { save_scope_as = consecration_province }
	}
	if = {
		limit = {
			NOT = { exists = scope:consecration_province }
			exists = location
			location = { pam_province_is_free_burial_site_trigger = yes }
		}
		location = { save_scope_as = consecration_province }
	}
	if = {
		limit = {
			NOT = { exists = scope:consecration_province }
			faith = {
				any_faith_holy_order = {
					leader = root
					holy_order_patron ?= { is_alive = yes }
				}
			}
		}
		faith = {
			random_faith_holy_order = {
				limit = {
					leader = root
					holy_order_patron ?= { is_alive = yes }
				}
				holy_order_patron = { save_scope_as = pam_founder_burial_patron }
			}
		}
		scope:pam_founder_burial_patron ?= {
			pam_pick_actor_consecration_province_effect = yes
		}
	}
	if = {
		limit = {
			NOT = { exists = scope:consecration_province }
			faith = {
				any_holy_site_barony = {
					title_province ?= { pam_province_is_free_burial_site_trigger = yes }
				}
			}
		}
		faith = {
			random_holy_site_barony = {
				limit = {
					title_province ?= { pam_province_is_free_burial_site_trigger = yes }
				}
				title_province = { save_scope_as = consecration_province }
			}
		}
	}
}

# CANONIZER - the ruler whose faith gains the saint.
# CANDIDATE - the newly canonized saint, shown as the message's portrait.
pam_grant_dulia_canonization_faith_bonus_effect = {
	$CANDIDATE$ = { save_scope_as = pam_new_saint }
	custom_tooltip = {
		text = pam_dulia_canonization_faith_bonus_tt
		$CANONIZER$.faith ?= {
			every_faith_character = {
				limit = { is_ruler = yes }
				add_character_modifier = {
					modifier = pam_dulia_new_saint_modifier
					years = 5
				}
				if = {
					limit = { this != $CANONIZER$ }
					send_interface_message = {
						type = msg_pam_new_saint_canonized
						title = pam_dulia_new_saint_message_t
						desc = pam_dulia_new_saint_message_desc
						left_icon = $CANDIDATE$
						show_as_tooltip = {
							add_character_modifier = {
								modifier = pam_dulia_new_saint_modifier
								years = 5
							}
						}
					}
				}
			}
		}
	}
}

}

							rite_has_doctrine = unreformed_faith_doctrine
						}
					}
					create_relic_artifact_effect = { NAME = effigy TYPE = pedestal VISUAL = statue }
				}
				else = {
					create_relic_artifact_effect = { NAME = effigy TYPE = pedestal VISUAL = statue_roman_woman }
				}
			}
			10 = {
				trigger = {
					like_zoroastrianism_religion_trigger = yes
					NOT = { scope:relic_site = { has_variable = pam_relic_drawn_afarganyu } }
				}
				create_relic_artifact_effect = { NAME = afarganyu TYPE = pedestal VISUAL = urn }
			}
			10 = {
				trigger = {
					OR = {
						like_zoroastrianism_religion_trigger = yes # zorostrian seal
						like_yazidi_religion_trigger = yes
						rite = { rite_has_parameter = sanctify_artifacts_luxurious }
					}
					NOT = { scope:relic_site = { has_variable = pam_relic_drawn_ring } }
				}
				create_relic_artifact_effect = { NAME = ring TYPE = pedestal VISUAL = ring_court }
			}
			10 = {
				trigger = {
					OR = {
						like_zoroastrianism_religion_trigger = yes
						like_yazidi_religion_trigger = yes
						rite = { rite_has_parameter = sanctify_artifacts_luxurious }
					}
					NOT = { scope:relic_site = { has_variable = pam_relic_drawn_bracelet } }
				}
				create_relic_artifact_effect = { NAME = bracelet TYPE = pedestal VISUAL = oxus_bracelet }
			}
			10 = {
				trigger = {
					OR = {
						like_judaism_religion_trigger = yes
						rite = { rite_has_tenet = tenet_scholasticism }
					}
					NOT = { scope:relic_site = { has_variable = pam_relic_drawn_commentary } }
				}
				create_relic_artifact_effect = { NAME = commentary TYPE = pedestal VISUAL = book }
			}
			10 = {
				trigger = {
					like_judaism_religion_trigger = yes
					NOT = { scope:relic_site = { has_variable = pam_relic_drawn_tzitzit } }
				}
				create_relic_artifact_effect = { NAME = tzitzit TYPE = pedestal VISUAL = reliquary }
			}
			# the final fallback. should never pick this at least in vanilla game
			0 = {
				create_relic_artifact_effect = { NAME = relic TYPE = pedestal VISUAL = reliquary }
			}
		}
		hidden_effect_new_object = {
			if = {
				limit = { has_royal_court = no }
				scope:newly_created_artifact = { reforge_artifact = { type = miscellaneous quality = scope:quality wealth = scope:wealth } }
			}
			scope:newly_created_artifact = {
				set_variable = { name = relic_religion value = scope:relic_creator_saint.religion }
				add_scaled_artifact_modifier_piety_effect = yes
				add_scaled_artifact_modifier_devotion_effect = yes
				add_scaled_artifact_modifier_grandeur_small_effect = yes
				remove_artifact_modifier = artifact_placeholder_modifier
			}
		}
	}
}
# Run from the extracting ruler (root)
# $SITE$ is the burial province to extract from
pam_extract_relic_from_saint_effect = {
	$SITE$ = {
		save_scope_as = relic_site
		saint_in_province = { save_scope_as = relic_creator_saint }
	}
	save_scope_as = extractor
	# Set the relic's tier from the saint before the charge is spent.
	scope:relic_creator_saint = {
		save_scope_value_as = {
			name = quality
			value = pam_saint_relic_quality_value
		}
	}
	# Spend one of the saint's relic charges; the engine checks faith, realm and remaining charges.
	hidden_effect = { consume_saint_relic_charge = scope:relic_creator_saint }
	# Religion-appropriate relic creation (hidden_effect_new_object; saves scope:newly_created_artifact).
	create_appropriate_relic_artifact_from_character_effect = {
		OWNER = scope:extractor
		CREATOR = scope:relic_creator_saint
	}
	# Surface + rename the drawn relic (no repeated type per saint).
	if = {
		limit = { exists = scope:newly_created_artifact }
		hidden_effect_new_object = {
			scope:newly_created_artifact = { save_scope_as = relic_artifact }
			scope:relic_artifact = {
				# Stamped onto the artifact because relic_creator_saint is only in scope during extraction.
				set_variable = {
					name = pam_saint_relic_saint
					value = scope:relic_creator_saint
				}
			}
		}

		# Dulia: relics carry one extra blessing. Hidden so the on-site option tooltip doesn't pre-spoil the random roll.
		if = {
			limit = { rite = { rite_has_parameter = tenet_dulia_extra_relic_modifier } }
			hidden_effect = { pam_add_dulia_extra_relic_modifier_effect = yes }
		}
		# Dulia: a relic from one's own dynast glorifies the house.
		if = {
			limit = {
				rite = { rite_has_parameter = tenet_dulia_dynasty_prestige_from_relics }
				exists = dynasty
				exists = scope:relic_creator_saint.dynasty
				scope:relic_creator_saint.dynasty = dynasty
			}
			pam_grant_dulia_relic_dynasty_prestige_effect = yes
		}
	}
}

# Dulia: grants dynasty prestige for a kin relic so events can echo it via show_as_tooltip
pam_grant_dulia_relic_dynasty_prestige_effect = {
	custom_tooltip = pam_dulia_relic_dynasty_prestige_tt
	dynasty = { add_dynasty_prestige = medium_dynasty_prestige_gain }
}
