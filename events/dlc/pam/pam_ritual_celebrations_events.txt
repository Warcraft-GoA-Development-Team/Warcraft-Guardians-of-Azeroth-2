namespace = pam_ritual_celebrations_events

##################################################
# Praise the Gods feast intent event
#
# 0001	A Feast of Revelation
##################################################

# Root = the celebrant. Checks whether at least one unknown tenet or learnable doctrine remains.
scripted_trigger pam_praise_the_gods_unknown_belief_trigger = {
	OR = {
		any_character_tenet = {
			knowledge_filter = unknown
			count >= 1
		}
		any_character_doctrine = {
			knowledge_filter = unknown
			passive_rite_learning_doctrine_is_learnable_trigger = yes
			count >= 1
		}
	}
}

# Root = the celebrant. Selects one unknown tenet or learnable doctrine.
scripted_effect pam_praise_the_gods_select_belief_effect = {
	random_list = {
		50 = {
			trigger = {
				any_character_tenet = {
					knowledge_filter = unknown
					count >= 1
				}
			}
			random_character_tenet = {
				knowledge_filter = unknown
				save_scope_as = pam_praise_the_gods_tenet
			}
		}
		50 = {
			trigger = {
				any_character_doctrine = {
					knowledge_filter = unknown
					passive_rite_learning_doctrine_is_learnable_trigger = yes
					count >= 1
				}
			}
			random_character_doctrine = {
				knowledge_filter = unknown
				limit = { passive_rite_learning_doctrine_is_learnable_trigger = yes }
				save_scope_as = pam_praise_the_gods_doctrine
			}
		}
	}
}

pam_ritual_celebrations_events.0001 = {
	type = activity_event
	title = pam_ritual_celebrations_events.0001.t
	desc = pam_ritual_celebrations_events.0001.desc
	theme = feast_activity

	left_portrait = {
		character = root
		animation = toast
	}

	trigger = {
		has_activity_intent = praise_the_gods_intent
	}

	immediate = {
		if = {
			limit = { pam_praise_the_gods_unknown_belief_trigger = yes }
			pam_praise_the_gods_select_belief_effect = yes
		}
		involved_activity = {
			random_attending_character = {
				limit = {
					this != root
					is_alive = yes
					is_adult = yes
					is_clergy = yes
					rite = root.rite
				}
				save_scope_as = pam_praise_the_gods_clergy
			}
		}
	}

	option = {
		name = pam_ritual_celebrations_events.0001.a

		random_list = {
			100 = {
				desc = pam_praise_the_gods_revelation_outcome_tt
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_praise_the_gods_revelation_toast
					left_icon = root
					add_learning_lifestyle_xp = medium_lifestyle_xp
					change_spiritual_fulfillment = medium_spiritual_fulfillment_value
				}
			}
			100 = {
				trigger = {
					OR = {
						exists = scope:pam_praise_the_gods_tenet
						exists = scope:pam_praise_the_gods_doctrine
					}
				}
				desc = pam_praise_the_gods_belief_outcome_tt
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_praise_the_gods_belief_toast
					left_icon = root
					if = {
						limit = { exists = scope:pam_praise_the_gods_tenet }
						add_known_tenet = scope:pam_praise_the_gods_tenet
					}
					else_if = {
						limit = { exists = scope:pam_praise_the_gods_doctrine }
						add_known_doctrine = scope:pam_praise_the_gods_doctrine
					}
					change_spiritual_fulfillment = minor_spiritual_fulfillment_value
				}
			}
			100 = {
				trigger = {
					has_dlc_feature = legends
					NOR = {
						always = no # Warcraft - game rule commented out in GoA
						has_personal_legend_seed = holy
					}
				}
				desc = pam_praise_the_gods_legend_outcome_tt
				send_interface_toast = {
					type = msg_legend_seed_created
					title = pam_praise_the_gods_legend_toast
					left_icon = root
					create_legend_seed = {
						type = holy
						quality = famed
						chronicle = saintly_life
						properties = {
							religion = root.religion
						}
					}
					change_spiritual_fulfillment = minor_spiritual_fulfillment_value
				}
			}
			100 = {
				trigger = {
					scope:pam_praise_the_gods_clergy ?= {
						is_alive = yes
						is_adult = yes
						is_clergy = yes
						rite = root.rite
					}
					involved_activity ?= {
						any_attending_character = {
							this = scope:pam_praise_the_gods_clergy
						}
					}
				}
				desc = pam_praise_the_gods_clergy_outcome_tt
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_praise_the_gods_clergy_toast
					left_icon = scope:pam_praise_the_gods_clergy
					right_icon = root
					scope:pam_praise_the_gods_clergy = {
						add_opinion = {
							target = root
							modifier = pam_praise_the_gods_clergy_bond_opinion
						}
					}
					if = {
						limit = {
							can_set_relation_friend_trigger = {
								CHARACTER = scope:pam_praise_the_gods_clergy
							}
						}
						set_relation_friend = {
							reason = friend_feast_bonding
							target = scope:pam_praise_the_gods_clergy
						}
					}
					else_if = {
						limit = {
							can_set_relation_potential_friend_trigger = {
								CHARACTER = scope:pam_praise_the_gods_clergy
							}
						}
						set_relation_potential_friend = scope:pam_praise_the_gods_clergy
					}
					change_spiritual_fulfillment = medium_spiritual_fulfillment_value
				}
			}
			100 = {
				trigger = { NOT = { has_trait = lifestyle_scholar } }
				desc = pam_praise_the_gods_scholar_outcome_tt
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_praise_the_gods_scholar_toast
					left_icon = root
					add_trait = lifestyle_scholar
					change_spiritual_fulfillment = minor_spiritual_fulfillment_value
				}
			}
			100 = {
				trigger = {
					has_trait = lifestyle_scholar
					has_trait_xp = {
						trait = lifestyle_scholar
						value < 100
					}
				}
				desc = pam_praise_the_gods_scholar_xp_outcome_tt
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_praise_the_gods_scholar_xp_toast
					left_icon = root
					add_trait_xp = {
						trait = lifestyle_scholar
						value = 20
					}
					change_spiritual_fulfillment = minor_spiritual_fulfillment_value
				}
			}
		}

		if = {
			limit = { exists = involved_activity }
			involved_activity = {
				add_activity_log_entry = {
					key = pam_praise_the_gods_log
					tags = { good }
					character = root
				}
			}
		}
		complete_activity_intent = yes

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_sociability = 0.5
				ai_rationality = 0.25
			}
		}
	}

	option = {
		name = pam_ritual_celebrations_events.0001.b

		add_stress = major_stress_loss
		change_spiritual_fulfillment = minor_spiritual_fulfillment_value
		if = {
			limit = { exists = involved_activity }
			involved_activity = {
				add_activity_log_entry = {
					key = pam_praise_the_gods_quiet_log
					tags = { good }
					character = root
				}
			}
		}
		complete_activity_intent = yes

		ai_chance = {
			base = 75
			ai_value_modifier = {
				ai_sociability = -0.5
				ai_zeal = 0.5
			}
			modifier = {
				factor = 2
				OR = {
					has_trait = humble
					has_trait = shy
					has_trait = temperate
				}
			}
		}
	}
}
