cast_spell_ai_decision = {
    decision_group_type = wc_magic

	ai_check_interval_by_tier = {
		barony = 2
		county = 1
		duchy = 1
		kingdom = 1
		empire = 1
		hegemony = 1
	}

    is_shown = {
		is_ai = yes 
        is_ruler = yes
        is_magic_user_trigger = yes
        is_any_spell_castable_trigger = yes
	}

    effect = {
        random_list = {
            10 = { # Cast once
                select_spell_class_ai_effect = yes
            }
            1 = { # Cast twice
                select_spell_class_ai_effect = yes
                trigger_event = {
                    on_action = wc_ai_subsequent_cast_on_action
                    days = { 5 30 }
                }
            }
        }
        if = {
            limit = {
                debug_only = yes
            }
            change_global_variable = {
                name = ai_spells_pre_cast_stage_1
                add = 1
            }
        }
        every_courtier = {
            limit = {
                is_ai = yes
                is_magic_user_trigger = yes
                is_any_spell_castable_trigger = yes
                employer ?= root
            }
            select_spell_class_ai_effect = yes
        }
        every_close_or_extended_family_member = {
            limit = {
                is_ai = yes
                is_ruler = no
                is_magic_user_trigger = yes
                is_any_spell_castable_trigger = yes
            }
            select_spell_class_ai_effect = yes
        }
    }

    ai_will_do = {
		base = 25
        ai_will_cast_modifier = yes
	}
}
