become_a_demon_hunter_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/becoming_a_demon_hunter.dds" }
	decision_group_type = major

	ai_check_interval = 60

	desc = become_a_demon_hunter_decision_desc

	is_shown = {
		OR = {
			has_trait = creature_night_elf
			has_trait = creature_high_elf
			has_trait = creature_draenei
		}
		NOR = {
			has_trait = demon_hunter
			has_trait = being_void # Because of event: void_elf.0130
			has_trait = being_undead # Darkfallen Elves never became Demon Hunters
			has_trait = fallen_demon_hunter

			# Historically, these characters shouln't be allowed to become demon hunters
			is_tyrande_trigger = yes # Tyrande
			is_shandris_trigger = yes # Shandris
			is_malfurion_trigger = yes # Malfurion
			is_maiev_trigger = yes # Maiev Shadowsong
		}
	}

	is_valid = {
		custom_tooltip = {
			text = a_demon_is_imprisoned_by_you
			any_prisoner = {
				has_trait = being_demon
				NOT = { has_trait = demon_hunter }
				is_imprisoned_by = root
			}
		}
		OR = {
			has_trait = creature_night_elf
			has_trait = creature_high_elf
			has_trait = creature_draenei
		}
		OR = { # Only certain personalities can take such a drastic decision
			has_trait = vengeful
			has_trait = callous
			has_trait = cynical # Since you don't trust your old religion to defeat the demons
			has_trait = wrathful
		}
		NOR = {
			has_trait = forgiving
			has_trait = compassionate
			has_trait = loyal
			has_trait = zealous # Why would you abandon your old religion so easily?
			has_trait = content
			has_trait = craven
			has_trait = calm
		}
		custom_tooltip = {
			text = not_a_servent_of_burning_legion
			NOT = {
				has_trait = being_demon
				has_faith = faith:burning_legion_religion
			}
		}
	}

	cost = {
		piety = 2000
        prestige = 2000
    }

	is_valid_showing_failures_only = {
		is_at_war = no
	}

	effect = {
		show_as_tooltip = { set_character_faith = faith:illidari }
		hidden_effect = {
			set_character_faith_with_conversion = faith:illidari
			trigger_event = {
				id = demon_hunter_events.0001
				days = 1 # To trigger this one after the faith conversion event trigger
			}
		}
		add_character_flag = should_be_topless # Most of them seem to be topless after they become Demon Hunters
	}

	ai_potential = {
		is_at_war = no
	}

	ai_will_do = {
		base = 5
		modifier = {
			add = 95
			has_faith = faith:illidari # If you are already a Vindicatory faith elf but haven't became a Demon Hunter yet
		}
		modifier = {
			add = 10
			current_date >= 618.1.1 # Lore-wise, this is when Demon Hunters canonically start appearing
		}
	}
}
