# Act of Devotion
pam_decision_events.0002 = {
	type = character_event
	title = pam_decision_events.0002.t
	desc = {
		desc = pam_decision_events.0002.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						has_realm_law = crowned_king
						has_realm_law = crowned_emperor
					}
				}
				desc = pam_decision_events.0002.desc.crowned
			}
			desc = pam_decision_events.0002.desc.uncrowned
		}
		desc = pam_decision_events.0002.desc.mid
		first_valid = {
			triggered_desc = {
				trigger = {
					spiritual_fulfillment <= -50
				}
				desc = pam_decision_events.0002.desc.bad
			}
			triggered_desc = {
				trigger = {
					spiritual_fulfillment >= 50
				}
				desc = pam_decision_events.0002.desc.good
			}
			desc = pam_decision_events.0002.desc.neutral
		}
		desc = pam_decision_events.0002.desc.outro
	}
	theme = christian

	left_portrait = { 
		character = root
		animation = personality_zealous
	}
	right_portrait = { 
		character = scope:my_chaplain
		animation = preaching_no_book
		camera = camera_event_center_close_to_right
	}
	override_effect_2d = {
		reference = legend_glow
	}
	
	immediate = {
		save_scope_as = devotional_guy
		if = {
			limit = { exists = cp:councillor_court_chaplain }
			cp:councillor_court_chaplain = { save_scope_as = my_chaplain }
		}
		else_if = {
			limit = {
				any_vassal = {
					government_has_flag = government_is_theocracy
				}
			}
			ordered_vassal = {
				order_by = primary_title.tier
				limit = { government_has_flag = government_is_theocracy }
				save_scope_as = my_chaplain
			}
		}
		else = {
			create_character = {
				template = priest_character_template
				location = root.location
				save_scope_as = my_chaplain
			}
		}
	}
	
	option = { # Push harder
		show_as_unavailable = { always = yes }
		name = pam_decision_events.0002.a
		trigger = {
			custom_tooltip = {
				text = pam_decision_act_of_devotion_fame_tooltip
				prestige_level < 3
			}
		}
		add_piety_level = -1
		add_prestige_level = 1
		add_prestige = medium_prestige_gain

		ai_chance = {
			base = 0
		}
	}
	
	option = { # Collect your prestige
		name = pam_decision_events.0002.b
		add_prestige = major_prestige_gain

		ai_chance = {
			base = 100
		}
	}
}

		}
		triggered_animation = {
			trigger = {
				has_personality_submissive_trigger = yes
			}
			animation = worry
		}
		triggered_animation = {
			trigger = {
				OR = {
					learning >= high_skill_rating
					has_trait = theologian

				}
			}
			animation = interested
		}
		animation = prayer_standing
	}

		}

				type = favorite_child
				save_scope_as = target_child
			}
		}
		scope:target_child = { set_variable = pam_mentor_sp_taken_var }
	}
	
	option = { # Push them _more_
		name = pam_decision_events.0003.a

			limit = {
				NOT = { has_relation_grudge = scope:target_child }
			}
			scope:target_child = {
				set_relation_grudge = {
					target = root
					reason = grudge_pressured_me_sp
				}
			}
		}
		scope:target_child = {
			if = {
				limit = {
					is_available_quick = {
						advanced_ruler = yes
					}
				}
				change_spiritual_fulfillment = 50
			}
		}

		ai_chance = {
			base = 0
			modifier = {
				has_trait = callous
				add = 25
			}
		}
	}
	
	option = { # Listen, child
		name = pam_decision_events.0003.b
		scope:target_child = {
			if = {

				#learn_language = language_greek # Warcraft
			}
			else_if = {
				limit = { NOT = { has_trait = lifestyle_scholar } }
				add_trait = lifestyle_scholar
			}
			else = {
				add_trait_xp = {
					trait = lifestyle_scholar
					value = pam_clerical_court_trait_xp_gain_value
				}
			}
		}
		add_piety = pam_clerical_court_piety_reward_value
		ai_chance = { base = 100 }
	}

	# Opportunity: the scriptorium flourishes
	option = {
		name = {
			trigger = { religion = { like_christianity_religion_trigger = yes } } # Warcraft
			text = pam_decision_events.0032.scriptorium.christian
		}
		name = {
			trigger = { NOT = { religion = { like_christianity_religion_trigger = yes } } # Warcraft
			text = pam_decision_events.0032.scriptorium
		}
		trigger = {
			exists = scope:pam_clerical_opportunity
			scope:pam_clerical_opportunity = flag:scriptorium
		}
		add_character_modifier = {
			modifier = pam_scriptorium_modifier
			years = 10
		}
		add_piety = pam_clerical_court_piety_reward_value
		ai_chance = { base = 100 }
	}

	# Ordain lay courtiers into the clergy
	option = {
		name = {
			trigger = { religion = { like_christianity_religion_trigger = yes } } # Warcraft
			text = pam_decision_events.0032.a.christian
		}
		name = {
			trigger = { NOT = { religion = { like_christianity_religion_trigger = yes } } } # Warcraft }
			text = pam_decision_events.0032.a
		}
		trigger = {
			any_courtier = { pam_clerical_court_ordain_candidate_trigger = yes }
		}
		custom_tooltip = pam_decision_events.0032.a.tt
		add_piety = pam_clerical_court_piety_reward_value
		hidden_effect = {
			ordered_courtier = {
				limit = { pam_clerical_court_ordain_candidate_trigger = yes }
				order_by = learning
				max = pam_clerical_court_batch_value
				check_range_bounds = no
				pam_clerical_court_ordain_effect = yes
			}
		}
		ai_chance = {
			base = 100
			# Always bank a rolled opportunity
			modifier = {
				factor = 0
				exists = scope:pam_clerical_opportunity
			}
		}
	}

	# Instruct the clergy already at court
	option = {
		name = pam_decision_events.0032.b
		trigger = {
			any_courtier = { pam_clerical_court_pupil_trigger = yes }
		}
		custom_tooltip = pam_decision_events.0032.b.tt
		add_piety = pam_clerical_court_piety_reward_value
		hidden_effect = {
			ordered_courtier = {
				limit = { pam_clerical_court_pupil_trigger = yes }
				order_by = learning
				max = pam_clerical_court_batch_value
				check_range_bounds = no
				pam_clerical_court_instruct_effect = yes
			}
		}
		ai_chance = {
			base = 50
			# Always bank a rolled opportunity
			modifier = {
				factor = 0
				exists = scope:pam_clerical_opportunity
			}
		}
	}

	# We call for nearby novices and hide this if none exist, never spawning one
	# A named supplicant is recruited first
	option = {
		name = {
			trigger = { exists = scope:pam_clerical_novice }
			text = pam_decision_events.0032.c.named
		}
		name = {
			trigger = { NOT = { exists = scope:pam_clerical_novice } }
			text = pam_decision_events.0032.c
		}
		trigger = {
			any_pool_character = {
				province = root.location
				pam_clerical_court_novice_trigger = { RULER = root }
			}
		}
		custom_tooltip = pam_decision_events.0032.c.tt
		add_piety = pam_clerical_court_piety_reward_value
		hidden_effect = {
			set_variable = {
				name = pam_clerical_recruits_left
				value = pam_clerical_court_batch_value
			}
			while = {
				count = 4 # Hard bound matching the highest batch; the limits below are what actually end the loop
				limit = {
					var:pam_clerical_recruits_left > 0
					any_pool_character = {
						province = root.location
						pam_clerical_court_novice_trigger = { RULER = root }
					}
				}
				pam_clerical_court_recruit_effect = yes
				change_variable = {
					name = pam_clerical_recruits_left
					subtract = 1
				}
			}
			remove_variable = pam_clerical_recruits_left
		}
		ai_chance = {
			base = 1
			modifier = {
				add = 99
				any_courtier = {
					count < pam_clerical_court_size_cap_value
					is_clergy = yes
				}
			}
			# Always bank a rolled opportunity
			modifier = {
				factor = 0
				exists = scope:pam_clerical_opportunity
			}
		}
	}
}
