scripted_trigger luxury_contract_potential_rich_trigger = {
	exists = house
	is_available_quick = {
		ai = yes
		adult = yes
		imprisoned = no
		incapable = no
		in_army = no
	}

scripted_trigger luxury_contract_any_rich_local_trigger = {
	location.county ?= {
		OR = {
			holder = {
				OR = {
					luxury_contract_potential_rich_trigger = yes
					any_courtier = { luxury_contract_potential_rich_trigger = yes }
					any_vassal = {
						capital_province.duchy = root.location.duchy
						OR = {
							luxury_contract_potential_rich_trigger = yes
							any_courtier = { luxury_contract_potential_rich_trigger = yes }
						}
					}
				}
			}
			title_province = {
				any_province_domicile = {
					OR = {
						is_domicile_type = merchant_company
						is_domicile_type = east_asian_estate
						is_domicile_type = japanese_manor
					}
					owner = {
						OR = {
							luxury_contract_potential_rich_trigger = yes
							any_courtier = { luxury_contract_potential_rich_trigger = yes }
						}
					}
				}
			}
		}
	}
}

scripted_effect luxury_contract_find_rich_local_effect = {
	if = {
		limit = { luxury_contract_potential_rich_trigger = yes }
		add_to_list = potential_rich_local
	}
	every_courtier = {
		limit = { luxury_contract_potential_rich_trigger = yes }
		add_to_list = potential_rich_local
	}
}

scripted_effect luxury_contract_select_rich_local_effect = {
	location = {
		county.holder = {
			luxury_contract_find_rich_local_effect = yes
			every_vassal = {
				limit = { capital_province.duchy = root.location.duchy }
				luxury_contract_find_rich_local_effect = yes
			}
		}
		duchy = {
			every_de_jure_county = {
				title_province = {
					every_province_domicile = {
						limit = {
							OR = {
								is_domicile_type = merchant_company
								is_domicile_type = east_asian_estate
								is_domicile_type = japanese_manor
							}
						}
						owner = { luxury_contract_find_rich_local_effect = yes }
					}
				}
				every_pool_character = {
					province = title_province
					limit = { luxury_contract_potential_rich_trigger = yes }
					add_to_list = potential_rich_local
				}
			}
		}
	}
	# SELECT
	random_in_list = {
		list = potential_rich_local
		weight = {
			base = 1
			modifier = { add = gold }
			modifier = {
				government_has_flag = government_has_influence
				add = influence
			}
			modifier = {
				add = {
					value = prestige_level
					multiply = 10
				}
			}
			modifier = {
				add = {
					value = dynasty.dynasty_prestige_level
					multiply = 10
				}
			}
			modifier = {
				is_ruler = yes
				factor = 10
			}
			modifier = {
				primary_spouse ?= { is_ruler = yes }
				factor = 5
			}
			modifier = {
				any_parent = { is_ruler = yes }
				factor = 5
			}
			modifier = {
				is_landed = yes
				factor = 10
			}
			modifier = {
				primary_spouse ?= { is_landed = yes }
				factor = 5
			}
			modifier = {
				any_parent = { is_landed = yes }
				factor = 5
			}
			modifier = {
				is_ai = no
				factor = 0.01
			}
		}

	}

			create_character = {
				template = merchant_template

			}

	}
	scope:dealer ?= { assign_quirk_effect = yes }

scripted_effect luxury_contract_find_or_create_priest_effect = {
	random_pool_character = {
		province = scope:merchant.location
		limit = {
			is_available_quick = {
				adult = yes
				imprisoned = no
				incapable = no
				in_army = no
			}
			is_clergy = yes
			is_lowborn = yes
			faith = scope:religious_building_faith
			NOT = { faith.religious_head ?= this }
		}
		weight = {
			base = 1
			modifier = { add = learning }
			modifier = {
				has_trait = devoted
				factor = 10
			}
		}
		save_scope_as = priest
	}
	scope:local_county.holder = {
		if = {
			limit = { faith = scope:religious_building_faith }
			if = {
				limit = {
					cp:councillor_court_chaplain ?= {
						NOT = { faith.religious_head ?= this }
					}
				}
				cp:councillor_court_chaplain = { save_scope_as = priest }
			}
			else_if = {
				limit = {
					is_clergy = yes
					NOT = { faith.religious_head ?= this }
				}
				save_scope_as = priest
			}
		}
	}
	if = {
		limit = { NOT = { exists = scope:priest } }
		luxury_contract_save_spawn_culture_and_faith_effect = yes
		create_character = {
			template = priest_character_template
			dynasty = none
			location = location
			culture = scope:lc_spawn_culture
			faith = scope:religious_building_faith
			save_scope_as = priest
		}
		scope:priest = { add_character_flag = destroy_me trigger_race_giving_no_gene_effect = yes } # Warcraft
	}
	scope:priest ?= { assign_quirk_effect = yes }
}
