form_the_scarlet_crusade_decision = { # By Ikarimonogatari
    picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
    decision_group_type = major

    ai_check_interval = 12

    desc = form_the_scarlet_crusade_decision_desc

    is_shown = {
        has_faith = faith:scarletism
        has_trait = creature_human
        NOT = {
            is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:form_the_scarlet_crusade_decision
			}
        }
    }

    is_valid = {
        has_faith = faith:scarletism
        has_trait = creature_human
        NOT = { exists = faith.religious_head }
        is_independent_ruler = yes
        is_landed = yes
        prestige_level >= 3
        piety_level >= 2
    }

    is_valid_showing_failures_only = {
        is_capable_adult = yes
		is_imprisoned = no
    }

    cost = {
        gold = 200
        prestige = 2000
        piety = 1000
    }

    effect = {
        hidden_effect = { # Can only form this insanity once a playthrough
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:form_the_scarlet_crusade_decision
            }
            if = {
                limit = {
                    has_faith = faith:scarletism
                    has_trait = creature_human
                    this.capital_province = {
                        geographical_region = world_eastern_kingdoms_lordaeron
                    }
                }
                trigger_event = {
                    id = scarlet.1006
                    days = 3
                }
            }
        }
        add_character_modifier = wc_scarlet_crusader
        spawn_army = {
			uses_supply = no
			inheritable = yes
			name = scarlet_crusade
			men_at_arms = {
				type = trebuchet
				stacks = 5
			}
			men_at_arms = {
				type = archmage
				stacks = 5
			}
			men_at_arms = {
				type = armored_horsemen
				stacks = 5
			}
			men_at_arms = {
				type = armored_footmen
				stacks = 5
			}
			men_at_arms = {
				type = light_footmen
				stacks = 5
			}
			men_at_arms = {
				type = bowmen
				stacks = 5
			}
			men_at_arms = {
				type = pikemen_unit
				stacks = 5
			}
			location = capital_province
			origin = capital_province
		}
        custom_description_no_bullet = {
            text = create_temporal_head_of_faith

            # Spiritual HoF
            if = {
                limit = { faith = { has_doctrine_parameter = spiritual_head_of_faith } }
                trigger_event = scarlet.1005 # Re-uses existing title.
            }
        }
        add_character_flag = has_formed_the_scarlet_crusade
    }

    ai_potential = {
        always = yes
    }

    ai_will_do = {
        base = 100
    }
}

liberate_lordaeron_from_the_scourge_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/liberate_lordaeron.dds" }
    decision_group_type = major

    ai_check_interval = 12

    desc = liberate_lordaeron_from_the_scourge_decision_desc

    is_shown = {
        has_trait = creature_human
        OR = {
            has_faith = faith:holy_light
            has_faith = faith:argent
            has_faith = faith:scarletism
        }
        OR = { # This decision will be available once either the Scarlet Crusade/Argent Dawn have been formed
            is_target_in_global_variable_list = {
			    name = unavailable_unique_decisions
			    target = flag:form_the_argent_dawn_decision
			}
            is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:form_the_scarlet_crusade_decision
			}
        }
        NOT = { # Cannot take this decision more then once
            is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:liberate_lordaeron_from_the_scourge_decision
			}
        }
    }

    is_valid = {
        has_trait = creature_human
        OR = {
            has_faith = faith:holy_light
            has_faith = faith:argent
            has_faith = faith:scarletism
        }
        highest_held_title_tier >= tier_duchy
        completely_controls = title:d_lordaeron
        prestige_level >= 3
        piety_level >= 3
    }

    is_valid_showing_failures_only = {
        is_capable_adult = yes
		is_imprisoned = no
    }

    cost = {
        gold = 200
    }

    effect = {
        hidden_effect = {
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:liberate_lordaeron_from_the_scourge_decision
            }
            if = {
                limit = {
                    NOT = { has_character_flag = liberator_of_lordaeron }
                    has_trait = creature_human
                    this.capital_province = {
                        geographical_region = world_eastern_kingdoms_lordaeron
                    }
                }
                trigger_event = { # Notification event
                    id = liberate_lordaeron.1001
                    days = 2
                }
            }
            else = { # event for the one who takes the decision
                trigger_event = liberate_lordaeron.1000
            }
        }
    }

    ai_potential = {
        always = yes
    }

    ai_will_do = {
        base = 100
    }
}
