wc_titanforged_reorder_county_threshold = 12

wc_titanforged_pact_speed = 0.75
wc_titanforged_mandate_speed = 1.5
wc_titanforged_implantation_speed = 0.75
wc_titanforged_pact_development_loss = 2
wc_titanforged_mandate_development_fraction = 0.5

wc_titanforged_restoration_speed = {
	value = 1
	if = {
		limit = { exists = scope:councillor_liege.var:wc_titanforged_restoration }
		value = 0
		if = {
			limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:pact } }
			value = wc_titanforged_pact_speed
		}
		else_if = {
			limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:mandate } }
			value = wc_titanforged_mandate_speed
		}
		else_if = {
			limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:implantation } }
			value = wc_titanforged_implantation_speed
			if = {
				limit = { scope:councillor_liege.var:wc_titanforged_restoration = { has_variable = wc_titanforged_project_version } }
				multiply = wc_titanforged_forge_speed
				if = {
					limit = { scope:councillor_liege.var:wc_titanforged_restoration = { has_variable = wc_titanforged_waiting } }
					value = 0
				}
			}
		}
		else_if = {
			limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:arcane } }
			value = 1
		}
		else_if = {
			limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:legacy } }
			value = 1
		}
	}
}

wc_titanforged_implantation_development_change = {
	value = development_level
	multiply = 0.5
	ceiling = yes
	multiply = -1
}

# County scope. Negate only after capping/rounding the positive loss.
wc_titanforged_restoration_development_change = {
	value = development_level
	if = {
		limit = { scope:councillor_liege.var:wc_titanforged_restoration = { var:wc_titanforged_restoration_method = flag:pact } }
		max = wc_titanforged_pact_development_loss
	}
	else = {
		multiply = wc_titanforged_mandate_development_fraction
		ceiling = yes
	}
	multiply = -1
}

wc_titanforged_reorder_county_current_progress = {
	value = 0
	if = {
		limit = {
			scope:county ?= {
				has_variable = wc_titanforged_reorder_current_progress
			}
		}
		add = scope:county.var:wc_titanforged_reorder_current_progress
	}
}

wc_titanforged_reorder_county_friend_bonus = {
	value = 0
	if = {
		limit = {
			scope:councillor_liege = {
				has_relation_friend = scope:councillor
				NOT = { has_relation_best_friend = scope:councillor }
			}
		}
		add = {
			value = wc_titanforged_reorder_county_base
			multiply = {
				value = council_friend_impact_percentage
				divide = 100
			}
		}
