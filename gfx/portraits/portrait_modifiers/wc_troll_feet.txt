# An accessory's entity blocks are alternatives, not simultaneously attached
# pieces. Each normal accessory now uses one mesh containing both feet.
# Do not claim legwear genes here: clothing modifiers may bundle legwear and
# clothes. GoA's existing no_legwear accessory tag handles boot compatibility.
wc_troll_feet = {
    usage = game
    priority = 10
    selection_behavior = max
    wc_keep_native_feet = { dna_modifiers = {} weight = { base = 1 } }
    wc_show_troll_feet = {
        ignore_outfit_tags = yes
        dna_modifiers = {
            accessory = { mode = replace gene = hooves template = wc_troll_feet value = 1 }
            morph = { mode = add gene = gene_wc_native_feet_mask template = wc_native_feet_hidden value = 1 }
        }
        weight = { base = 0 modifier = { add = 100 wc_uses_troll_feet = yes NOT = { has_trait = one_legged } } }
    }
    wc_show_troll_feet_right_only = {
        ignore_outfit_tags = yes
        dna_modifiers = {
            accessory = { mode = replace gene = hooves template = wc_troll_feet_right_only value = 1 }
            morph = { mode = add gene = gene_wc_native_feet_mask template = wc_native_feet_hidden value = 1 }
        }
        weight = { base = 0 modifier = { add = 100 wc_uses_troll_feet = yes has_trait = one_legged } }
    }
}
