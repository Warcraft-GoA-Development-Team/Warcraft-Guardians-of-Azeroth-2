# Two mages have traveled to your court to settle a debate between them as to which is better; the antipodean magics (Ice and Fire) or the pure arcane. You can side with either one,  give a diplomatic answer, or dismiss them as a waste of your time.
arcane_magic_lifestyle_events.1021 = { 
	title = arcane_magic_lifestyle_events.1021.t
	desc = arcane_magic_lifestyle_events.1021.desc
	content_source = dlc_GOA
	theme = order

	right_portrait = { 
		character = antipodean_mage
		animation = anger
	}

	center_portrait = { 
		character = arcane_mage
		animation = dismissal
	}
	
	left_portrait = { 
		character = root
		animation = stress
	}

	trigger = { 
		NOT = { root.faith = { has_doctrine_parameter = order_magic_illegal } }
	}

	immediate = { 
		create_character = { 
			location = root.capital_province
			template = arcane_mage_character
			save_scope_as = antipodean_mage
			after_creation = {
				set_variable = { name = order_focus value = flag:fire }
				set_variable = { name = order_focus value = flag:ice }
			}
		}
		create_character = { 
			location = root.capital_province
			template = arcane_mage_character
			save_scope_as = arcane_mage
			after_creation = {
				set_variable = { name = order_focus value = flag:arcane }
			}
		}
	}
	
	# WHAT - Antipodean magics are better
	option = { 
		name = arcane_magic_lifestyle_events.1021.opt.a
		#HOW - (Duel: your diplomacy + sum of ice, fire, and arcane magic perks vs purist's learning + arcane perks) 
		duel = { 
			skill = learning
			target = scope:arcane_mage
			#success = you convince the purist that antipodean magics are better and gain bonuses to ice and fire lifestyle experience. 
			50 = { 
				modifier = { 
					add = total_perks_in_elemental_fire_lifestyle
					add = total_perks_in_elemental_ice_lifestyle
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.convince_success
				}
			} 
			# failure = nothing happens
			50 = { 
				modifier = { 
					add = arcane_mage.total_perks_in_order_magic_lifestyle
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.convince_failure
				}
			}
		}
	}

	# WHAT - Pure arcane magics are better
	option = { 
		name = arcane_magic_lifestyle_events.1021.opt.b
		# HOW - (Duel: your learning + arcane magic perks vs antipodean's diplomacy and sum of ice, fire, and learning perks ) 
			# success = you convince the antipodean that arcane magics are better and gain bonuses to arcane lifestyle experience. 
			50 = { 
				modifier = { 
					add = total_perks_in_order_magic_lifestyle
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.convince_success
				}
			} 
			# failure = nothing happens
			50 = { 
				modifier = { 
					add = antipodean_mage.total_perks_in_elemental_fire_lifestyle
					add = antipodean_mage.total_perks_in_elemental_ice_lifestyle
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.convince_failure
				}
			}
	}

	# WHAT - Make a case for a holistic study of the Arcane.
	option = { 
		name = arcane_magic_lifestyle_events.1021.opt.c
		# HOW - (Duel: your learning + diplomacy vs purist's learning + antipodean's diplomacy ) 
		random_list = { 
			# success = you convince them to stop fighting.  
			45 = { 
				modifier = { 
					add = diplomacy
				}
				modifier = { 
					add = learning
				}
				modifier = { 
					add = { 
						value = prestige_level
						multiply = 2
					}
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.convince_both_success
				}
			}
			# failure = you make yourself look like a dumbass and get a penalty to diplomacy and learning.
			55 = { 
				modifier = { 
					add = scope:arcane_mage.learning
				}
				modifier = { 
					add = scope:antipodean_mage.diplomacy
				}
				send_interface_toast = {
					title = elemental_magic_lifestyle.2032.toast.convince_both_failure
				}
			}
		}
	}

	# WHAT - Dismiss them to focus on your tasks as a ruler
	option = {
		name = arcane_magic_lifestyle_events.1021.opt.d
		# HOW - You get a bonus to development and income in your capital and a small amount of prestige
	}
}

				}
			}

			25 = { 
				add_character_modifier = {
					modifier = wc_arcane_studied_personal_scheming_modifier
					years = 4
				}
			}
			# 25% - uncover a plot against you if available (check ep1.flavor.2010)
			0 = {
				modifier = { 
					any_courtier_or_guest = { is_scheming_against = { target = root type = hostile } }
					add = 25
				}
				custom_tooltip = wc_expose_a_courtier_guest_scheme_description
				every_courtier_or_guest = {
					limit = { is_scheming_against = { target = root type = hostile } }
					add_to_list = scheme_havers
				}
				random_in_list = {
					list = scheme_havers
					random_scheme = { 
						limit = { target = root }  
						expose_scheme = yes
					}
				}
			}
		}
	}

# A young pyromaniac pt 1 - child
arcane_magic_lifestyle_events.7001 = { 
	title = arcane_magic_lifestyle_events.7001.t
	desc = arcane_magic_lifestyle_events.7001.desc
	content_source = dlc_GOA
	theme = order

	right_portrait = { 
		character = scope:pyromaniac
		animation = shame
	}

	left_portrait = { 
		character = root
		animation = thinking
	}

	weight_multiplier = {
		base = 0.1
		modifier = { 
			factor = { 
				value = 1
				add = total_perks_in_elemental_fire_magic_lifestyle
				add = total_perks_in_arcane_magic_lifestyle
			}
		}
		
		modifier = { 
			has_character_flag = had_arcane_magic_lifestyle_events_7001
			factor = 0
		}
	}

	trigger = { 
		NOT = { has_character_flag = had_arcane_magic_lifestyle_events_7001 }
	}

	immediate = { 
		create_MAGICTYPE_potential_child_effect = { MAGICTYPE = arcane }
		scope:magic_potential_child = { save_scope_as = pyromaniac }
	}

	# cool, let's burn stuff together
	option = { 
		name = arcane_magic_lifestyle_events.7001.opt_a
		# harm public opinion in province
		if = { 
			limit = {
				OR = { 
					location.county.holder != root
					is_landed = no
				}
			} 
			location.county.holder = {
				add_opinion = {
					modifier = outraged_opinion
					opinion = -30
					target = root
				}
			}
		}
		else = {
			random_held_county = {
				add_modifier = { 
					name = burned_county_for_fun
					years = 5
				}
				change_county_control = -25
			}
		}
		scope:pyromaniac = { 
			add_character_flag = burned_stuff_together
		}
		# add them to your court and teach them magic
		add_pyro_to_court_effect = yes
	}

	# compensate for damage and take them under my wing
	option = { 
		name = arcane_magic_lifestyle_events.7001.opt_b
		# pay some gold
		if = { 
			limit = { 
				OR = { 
					location.county.holder != root
					is_landed = no
				}
			}
			pay_short_term_gold = {
				target = location.county.holder
				gold = medium_gold_value
			}
		}
		else = { 
			remove_short_term_gold = minor_gold_value
		}
		# add them to your court and teach them magic
		add_pyro_to_court_effect = yes
	}

	# punish them for burining stuff
	option = {
		name = arcane_magic_lifestyle_events.7001.opt_c
		rightfully_imprison_character_effect = { 
			TARGET = scope:pyromaniac
			IMPRISONER = root
		}
	}
}
