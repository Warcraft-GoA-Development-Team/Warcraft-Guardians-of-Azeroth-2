cast_curse_effect = {
    get_recipient_effect = { SPELL = curse }
    save_scope_as = real_caster
	scope:this_recipient ?= {
		custom_tooltip = curse_tt
		if = {
			limit = {
				scope:real_caster = {
					has_variable = give_disease_type
				}
			}
			scope:real_caster = {
				var:give_disease_type = {
					save_scope_as = curse
				}
				remove_variable = give_disease_type
			}

			switch = {
				trigger = scope:curse

				flag:curse_of_exhaustion = { contract_disease_notify_effect  = { DISEASE = curse_of_exhaustion } }
				flag:curse_of_tongues = { contract_disease_notify_effect  = { DISEASE = curse_of_tongues } }
				flag:curse_of_weakness = { contract_disease_notify_effect  = { DISEASE = curse_of_weakness } }
                flag:curse_of_mutation = { contract_disease_notify_effect  = { DISEASE = curse_of_mutation } }
                flag:curse_of_the_elements = { contract_disease_notify_effect  = { DISEASE = curse_of_the_elements } }
                flag:curse_of_recklessness = { contract_disease_notify_effect  = { DISEASE = curse_of_recklessness } }
			}

			add_opinion = {
				modifier = wc_cursed_me
				target = scope:real_caster
			}
        }
		else = {
			hidden_effect = {
				scope:real_caster = {
					wc_change_mana_by_value_effect = {
						CHANGE = add
						VALUE = wc_spell_curse_cost_mana_value
					}
				}
			}
		}
	}
}

cast_subjugate_effect = {
    get_recipient_effect = { SPELL = subjugate }
    save_scope_as = real_caster
	if = {
		limit = {
            NOT = { exists = scope:cancel }
        }
        get_recipient_effect = { SPELL = subjugate }

        scope:this_recipient ?= {
            set_relation_master = scope:real_caster
        }
    }
    else = {
        var:subjugate_recipient = {
            if = {
                limit = {
                    has_relation_master = scope:real_caster
                }
                remove_relation_master = scope:real_caster
            }
        }
    }
}

clone_magic_add_modifier_to_list_effect = {
    if = {
        limit = {
            has_character_modifier = $MODIFIER$_modifier
            flag:$TYPE$ = flag:positive
            NOT = {
                flag:$SPELL$ = flag:none
            }
        }
        add_to_variable_list = {
            name = clone_magic_modifiers
            target = flag:$MODIFIER$
        }
    }
}

copy_spell_modifier_effect = {
    if = {
        limit = {
            exists = scope:clone_magic_modifier
            flag:$MODIFIER$ = scope:clone_magic_modifier
        }
        scope:real_caster = {
            add_character_modifier = {
                modifier = $MODIFIER$_modifier
                days = wc_current_spell_duration
            }
            every_relation = {
                type = summoner
                save_scope_as = master
            }
            scope:master = {
                add_character_modifier = {
                    modifier = $MODIFIER$_modifier
                    days = wc_current_spell_duration
                }
            }
        }
    }

    if = { # Error prevention
        limit = { flag:$TYPE$ = flag:error }
    }
    if = { # Error prevention
        limit = { flag:$SPELL$ = flag:clone_magic }
    } 
}

cast_immolate_effect = {
    get_recipient_effect = { SPELL = immolate }
    save_scope_as = caster
    if = {
        limit = {
            OR = { # current_spell_rank is for tooltips
                AND = {
                    exists = var:immolate_rank
                    var:immolate_rank = 1
                }
                AND = {
                    NOT = {
                        exists = var:immolate_rank
                    }
                    exists = var:current_spell_rank
                    var:current_spell_rank = 1
                }
            }
        }
        scope:this_recipient ?= { 
            add_or_increase_magic_injury_effect = { INJURY = burned CASTER = scope:caster }
        }
    }
    else = {
        scope:this_recipient = { deplete_army_by_percentage = 0.3 }
    }
}

        }
