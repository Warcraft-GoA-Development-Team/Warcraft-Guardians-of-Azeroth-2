namespace = nordrassilsdestruction

nordrassilsdestruction.0001 = {
    type = character_event
	content_source = dlc_GOA
	title = nordrassilsdestruction_0001_title
	desc = nordrassilsdestruction_0001_desc
	theme = emperor
	override_background = {
		reference = wc_background_destroyed_world_tree
	}

    immediate = {
        hidden_effect = {
            title:c_nordrassil.title_province = {
                remove_building = nordrassil_sacrificed
                add_special_building = nordrassil_destroyed
            }

            add_to_global_variable_list = {
                name = elven_lifespan
                target = flag:nordrassil_was_destroyed_by_a_demon
            }
        }
    }

    right_portrait = {
        character = scope:destroyer
        animation = manic
    }

    option = {
        name = nordrassilsdestruction_0001_a
        add_character_modifier = {
            modifier = weaker_world_tree_destroyed_by_demon
        }

        get_title = title:h_burning_legion

        title:h_burning_legion = {
            set_capital_county = title:c_nordrassil
        }

        hidden_effect = {
            every_living_character = {
                limit = {
                    OR = {
                        has_trait = creature_night_elf
                        has_trait = creature_dryad
                    }
                }
                trigger_event = { id = nordrassilsdestruction.0003 }
            }
        }
    }
}

nordrassilsdestruction.0003 = {
    type = character_event
	content_source = dlc_GOA
	title = nordrassilsdestruction_0003_title
	desc = nordrassilsdestruction_0003_desc
	theme = emperor
	override_background = {
		reference = wc_background_destroyed_world_tree
	}

    immediate = {
        if = {
            limit = {
                OR = {
                    has_character_modifier = world_tree_sacrificed_for_night_elf
                    has_character_modifier = world_tree_sacrificed_for_dryad_and_keeper
                }
            }
            remove_character_modifier = world_tree_sacrificed_for_night_elf
            remove_character_modifier = world_tree_sacrificed_for_dryad_and_keeper
        }
    }

    left_portrait = {
        character = root
        animation = fear
    }
    right_portrait = {
        character = scope:destroyer
        animation = manic
    }

    option = {
        name = nordrassilsdestruction_0003_a
        if = {
            limit = { has_trait = creature_night_elf }
            add_character_modifier = {
                modifier = world_tree_destroyed_for_night_elf
            }
        }
        else = {
            limit = { has_trait = creature_dryad }
            add_character_modifier = {
                modifier = world_tree_destroyed_for_dryad_and_keeper
            }
        }
    }
}
