# Light Magic Doctrines
doctrine_light_magic_crime = {
	doctrine_group_type = doctrine_light_magic
	index = 0
	divergence = 0

	can_pick = { can_idolise_light_magic_trigger = no }
	piety_cost = {
		value = faith_doctrine_cost_mid
		if = {
			limit = { rite_has_doctrine = doctrine_light_magic_crime }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_light_magic_ignored }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_light_magic_accepted }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_light_magic_approved }
			multiply = faith_changed_doctrine_cost_mult_four_step
		}
	}
	parameters = {
		light_magic_illegal
		light_magic_not_openly_learnable
		being_light_illegal
