namespace = pam_monastic_funeral

# Monastic funeral - fired to the host during the funeral ceremony phase when the activity is held at a temple leased to a monastic order.

##########################################################################################
# 0001 - Prayers Without End (the order offers perpetual care after the burial)
##########################################################################################
pam_monastic_funeral.0001 = {
	type = activity_event
	title = pam_monastic_funeral.0001.t
	desc = {
		desc = pam_monastic_funeral.0001.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = { scope:abbot = { is_female = yes } }
				desc = pam_monastic_funeral.0001.desc.sisters
			}
			desc = pam_monastic_funeral.0001.desc.brothers
		}
		desc = pam_monastic_funeral.0001.desc.outro
	}
	theme = funeral_activity
	override_background = { reference = temple_scope }
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		animation = grief
	}
	center_portrait = {
		character = scope:deceased
		animation = dead
		camera = camera_event_center
	}
	right_portrait = {
		character = scope:abbot
		animation = thurible
	}

	cooldown = { years = 1 }

	trigger = {
		scope:activity.activity_location.barony.lessee ?= {
			is_physically_able = yes
		}
		scope:activity = {
			has_activity_type = activity_funeral
			exists = var:body_to_bury
			activity_location ?= { pam_province_has_monastic_lease_trigger = yes }
		}
	}

	immediate = {
		play_music_cue = mx_cue_touching_moment
		scope:activity.var:body_to_bury = { save_scope_as = deceased }
		scope:activity.activity_location.barony.lessee = {
			save_scope_as = abbot
		}
		scope:activity.activity_location = {
			save_scope_as = background_temple_scope
		}
	}

	option = { # The order enters your house in its necrology and prays in perpetuity
		name = pam_monastic_funeral.0001.a
		add_character_modifier = {
			modifier = pam_monastic_commemoration_modifier
			years = 15
		}
		stress_and_fulfillment_impact = {
			cynical = minor_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = { has_trait = cynical factor = 0 }
		}
	}

	option = { # The scriptorium chronicles the deceased
		name = pam_monastic_funeral.0001.b
		trigger = {
			has_dlc_feature = legends
			NOT = { always = no } # Warcraft - game rule commented out in GoA
		}
		scope:background_temple_scope = {
			add_province_modifier = {
				modifier = pam_monastic_chronicle_modifier
				years = 15
			}
		}
		stress_and_fulfillment_impact = {
			humble = minor_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = { has_trait = humble factor = 0 }
		}
	}

	option = { # Opt-out - leave the dead in the order's care and go in peace
		name = pam_monastic_funeral.0001.c
		change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		scope:abbot = {
			add_opinion = {
				target = root
				modifier = pious_opinion
				opinion = 20
			}
		}
		stress_and_fulfillment_impact = {
			zealous = minor_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = { has_trait = zealous factor = 0 }
		}
	}
}
