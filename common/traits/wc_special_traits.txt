# A nathrezim wearing a stolen mortal identity - generic, any dreadlord can wear this.
nathrezim_disguise = {
	icon = deceitful.dds
	name = trait_nathrezim_disguise
	desc = trait_nathrezim_disguise_desc
	shown_in_ruler_designer = no

	intrigue = 6 # A demon's cunning bleeding through the mask - double creature_nathrezim's own +3
	dread_baseline_add = 15
	enemy_scheme_secrecy_add = -5 # Same paranoia as the race trait - assumes the worst
	life_expectancy = 80 # A stolen mortal shell strains under a demon's essence - some vitality, not a nathrezim's true 200
	health = 2

	# A borrowed body isn't a comfortable fit - the strain of holding a false
	# identity and a demon's essence in flesh not built for it.
	stress_loss_mult = -0.25

	# Blocks mundane disease and old-age "infirm" onset (same flag
	# being_undead uses) without full immortality - still bound by
	# life_expectancy = 80 above.
	flag = can_not_get_sick

	# Fully sterile like a true nathrezim (creature_nathrezim also has
	# can_have_children = no) - the shell doesn't conceive. wc_adoption lets
	# the player still name a successor without a blood heir.
	can_have_children = no
	flag = wc_adoption
}
