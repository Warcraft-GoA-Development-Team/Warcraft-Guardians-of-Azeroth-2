# Mantid Swarming cycle triggers. Events: events/wc_events/wc_mantid_swarm_events.txt

# Kingdom scope. The 7 kingdoms the Swarm may target: the 5 wall kingdoms plus the de jure mantid kingdoms held by others.
wc_swarm_target_set_trigger = {
	OR = {
		this = title:k_halfhill
		this = title:k_kun_lai
		this = title:k_kota
		this = title:k_krasarang
		this = title:k_mogushan
		this = title:k_niuzao
		this = title:k_townlong
	}
}

# Kingdom scope. The 5 kingdoms that contain a Serpent's Spine wall county.
wc_swarm_wall_kingdom_trigger = {
	OR = {
		this = title:k_halfhill
		this = title:k_kun_lai
		this = title:k_kota
		this = title:k_krasarang
		this = title:k_mogushan
	}
}

# County scope. The 8 counties with a Serpent's Spine wall province.
wc_swarm_wall_county_trigger = {
	OR = {
		this = title:c_stoneplow
		this = title:c_dustback
		this = title:c_paoquan
		this = title:c_serpents_kun_lai
		this = title:c_ox_gate
		this = title:c_gorai_doan
		this = title:c_north_setting_sun
		this = title:c_south_setting_sun
	}
}

# Character scope.
wc_swarm_is_aqir_trigger = {
	culture = { has_cultural_pillar = heritage_aqir }
}

# County scope. Both the holder and the holder's top liege follow a rite Yshaarjism may holy war against.
# Holder: what pam_determine_ghw_target_character_from_kingdom_effect tests. Top liege: what valid_ghw_target_trigger tests.
wc_swarm_hostile_county_trigger = {
	exists = holder
	faith:yshaarjism.main_rite = {
		is_hostile_enough_for_holy_war_trigger = { RITE = prev.holder.rite }
		is_hostile_enough_for_holy_war_trigger = { RITE = prev.holder.top_liege.rite }
	}
}

# Kingdom scope.
wc_swarm_target_kingdom_trigger = {
	wc_swarm_target_set_trigger = yes
	any_in_de_jure_hierarchy = {
		continue = { tier > tier_county }
		tier = tier_county
		wc_swarm_hostile_county_trigger = yes
	}
}

# Kingdom scope. Holds a Yshaarjism holy site (c_xuen is de jure k_binan, never a target).
wc_swarm_target_has_holy_site_trigger = {
	any_in_de_jure_hierarchy = {
		continue = { tier > tier_county }
		tier = tier_county
		OR = {
			this = title:c_niuzao
			this = title:c_lulkail
			this = title:c_chi_ji
		}
	}
}

# Kingdom scope. Needs scope:empress.
wc_swarm_target_defender_weaker_trigger = {
	exists = scope:empress
	exists = title_capital_county.holder
	title_capital_county.holder.top_liege.current_military_strength < scope:empress.current_military_strength
}

# Any scope.
wc_swarm_any_target_trigger = {
	OR = {
		title:k_halfhill = { wc_swarm_target_kingdom_trigger = yes }
		title:k_kun_lai = { wc_swarm_target_kingdom_trigger = yes }
		title:k_kota = { wc_swarm_target_kingdom_trigger = yes }
		title:k_krasarang = { wc_swarm_target_kingdom_trigger = yes }
		title:k_mogushan = { wc_swarm_target_kingdom_trigger = yes }
		title:k_niuzao = { wc_swarm_target_kingdom_trigger = yes }
		title:k_townlong = { wc_swarm_target_kingdom_trigger = yes }
	}
}

# Any scope.
wc_swarm_can_launch_trigger = {
	exists = title:d_seven_prophets.holder
	title:d_seven_prophets.holder = { is_imprisoned = no }

	NOT = { has_global_variable = wc_swarm_active }

	faith:yshaarjism = {
		NOT = { exists = great_holy_war }
	}

	wc_swarm_any_target_trigger = yes
}

# Only valid inside the war-end on_action effects, where code provides scope:war and scope:attacker.
wc_swarm_is_swarm_war_trigger = {
	has_global_variable = wc_swarm_active
	scope:war = { using_cb = undirected_great_holy_war }
	scope:attacker.faith = faith:yshaarjism
}

# Character scope, inside any_courtier of the event's root.
wc_swarm_cull_weak_trigger = {
	is_adult = yes
	is_ai = yes

	is_landed = no
	is_councillor = no
	NOT = { is_close_family_of = root }
	NOT = { has_trait = mantid_paragon }

	culture = culture:mantid

	prowess <= 6
}

# Character scope, inside any_courtier of the event's root.
wc_swarm_cull_strong_trigger = {
	is_adult = yes
	is_ai = yes

	is_landed = no

	culture = culture:mantid

	prowess >= 12
}

# Character scope, inside every_independent_ruler.
wc_swarm_shado_pan_letter_recipient_trigger = {
	exists = title:k_shado_pan.holder
	NOT = { this = title:k_shado_pan.holder }

	culture = { has_cultural_pillar = heritage_pandaren }

	exists = capital_province
	capital_province = { geographical_region = world_pandaria }
}
