######################################################################################################
# Investiture Council events
#
# Dedicated debate/guest/tally events for activity_investiture_council (the shipped ecumenical council
# events are phase-keyed to that activity, so these are bespoke; they reuse only the helper effects).
#
# 0001 - Arrival flavor (guest)
# 0003 - Pre-debate flavour (guest)
# 0004 - Player interrupt: busy/travelling voter summoned to the council
# 0002 - The council convenes (host, summons phase)
# 0010 - The debate: host champions Spiritual (host, debate phase)
# 0031 - The debate: guest picks a side (players without a locked stance)
# 0032 - Post-choice persuasion (clergy voter): Piety + Diplomacy/Learning duel to sway the assembly
# 0033 - Post-choice persuasion (secular attendee): bribe/convince either side
# 0060 - Outcome letter to a non-attending player HoF (from the host)
# 0061 - Outcome letter to a non-attending secular player (from their court chaplain)
# 0050 - Inform a player HoF an archbishop will host in their stead
# 0091 - Hidden retry nudge (re-attempt hosting until the council starts; hands off if the nominee is blocked)
# 0092 - Hidden delayed force-gather (a month after start, once invitations are processed)
# 0090 - Hidden backstop check
#
# The vote tally runs from the debate phase on_end (pam_investiture_council_tally_effect).
######################################################################################################

# Post-choice persuasion (clergy voter): a Diplomacy + Learning duel paid in Piety flips opposite-side AI voters
# (a poor result costs a vote). Chains up to the cap (+50% each); only offered while an opposite-side voter exists.
pam_investiture_council_events.0032 = {
	type = activity_event
	title = pam_investiture_council_events.0032.t
	desc = pam_investiture_council_events.0032.desc
	theme = pam_investiture_council
	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:investiture_persuade_face
		animation = contemplating
	}

	trigger = {
		exists = var:investiture_swaying_for
		scope:activity ?= {
			any_attending_character = {
				is_ai = yes
				this != root
				pam_valid_ecumenical_council_voter_trigger = yes
				trigger_if = {
					limit = { root.var:investiture_swaying_for = flag:innovator }
					is_in_guest_subset = { name = conservative }
				}
				trigger_else = {
					is_in_guest_subset = { name = innovator }
				}
			}
		}
	}

	immediate = {
		# First attempt of the chain: start the counter the scaled costs read (each attempt +50% of the base).
		if = {
			limit = { NOT = { has_variable = investiture_persuade_attempts } }
			set_variable = { name = investiture_persuade_attempts value = 1 days = 365 }
		}
		# Put a face to the opposing camp the player is trying to win over.
		scope:activity ?= {
			random_attending_character = {
				limit = {
					this != root
					pam_valid_ecumenical_council_voter_trigger = yes
					trigger_if = {
						limit = { root.var:investiture_swaying_for = flag:innovator }
						is_in_guest_subset = { name = conservative }
					}
					trigger_else = {
						is_in_guest_subset = { name = innovator }
					}
				}
				save_scope_as = investiture_persuade_face
			}
		}
	}

	# Sway the assembly - pay Piety and roll the Diplomacy + Learning duel.
	option = {
		name = pam_investiture_council_events.0032.a
		trigger = { piety >= pam_investiture_persuade_piety_cost_scaled }
		add_piety = {
			value = pam_investiture_persuade_piety_cost_scaled
			multiply = -1
		}
		# The clergy voter's side was chosen in 0031; pass it (and its opposite) to the shared duel.
		if = {
			limit = { var:investiture_swaying_for = flag:innovator }
			pam_investiture_persuade_duel_effect = { SKILL_A = diplomacy SKILL_B = learning SIDE = innovator OPPOSITE = conservative }
		}
		else = {
			pam_investiture_persuade_duel_effect = { SKILL_A = diplomacy SKILL_B = learning SIDE = conservative OPPOSITE = innovator }
		}
		# Chain the next, dearer attempt until the cap (the event's own trigger re-checks targets remain).
		hidden_effect = {
			change_variable = { name = investiture_persuade_attempts add = 1 }
			if = {
				limit = { var:investiture_persuade_attempts <= pam_investiture_persuade_max_attempts }
				trigger_event = { id = pam_investiture_council_events.0032 days = 1 }
			}
		}
	}
	# Leave it be.
	option = {
		name = pam_investiture_council_events.0032.b
	}
}

# Hidden retry nudge: re-attempts hosting until the nominee starts the council or the flag/pending expires.
# Self-healing - a blocked nominee hands the nomination back to the nudge if any substitute host exists.
pam_investiture_council_events.0091 = {
	type = character_event
	hidden = yes

	trigger = {
		has_character_flag = investiture_council_nominated
		faith = {
			has_variable = investiture_council_pending
			has_doctrine = doctrine_clerical_succession_conditional
		}
	}

	immediate = {
		if = {
			# The Head of Faith uses the prompt availability check, an archbishop the shared host trigger
			limit = {
				trigger_if = {
					limit = { faith.religious_head = this }
					pam_available_investiture_hof_ai_trigger = yes
				}
				trigger_else = {
					pam_valid_investiture_council_host_trigger = yes
				}
			}
			ai_attempt_to_host_activity = activity_investiture_council
			trigger_event = { id = pam_investiture_council_events.0091 days = pam_investiture_council_host_retry_days }
		}
		else_if = {
			# Blocked, and a substitute exists: hand the nomination back to the nudge to pick the next host.
			limit = {
				faith = {
					OR = {
						religious_head ?= { pam_available_investiture_hof_ai_trigger = yes }
						any_faith_character = { pam_valid_investiture_council_host_trigger = yes }
					}
				}
			}
			remove_character_flag = investiture_council_nominated
			faith = { pam_nudge_investiture_council_host_effect = yes }
		}
		else = {
			# With no substitute we keep waiting, and the in-absentia fallback still ends the crisis
			ai_attempt_to_host_activity = activity_investiture_council
			trigger_event = { id = pam_investiture_council_events.0091 days = pam_investiture_council_host_retry_days }
		}
	}
}

# Hidden delayed force-gather, a month after start: by then every invitation is processed, so force-adding
# can't collide with an in-flight acceptance (engine assert). Fired at the host from the activity on_start.
pam_investiture_council_events.0092 = {
	type = character_event
	hidden = yes

	# Skip if the council is gone or the host changed; the normal invitations carry on without the sweep.
	trigger = {
		involved_activity ?= {
			has_activity_type = activity_investiture_council
			activity_host = root
		}
	}

	immediate = {
		save_scope_as = host
		involved_activity = { save_scope_as = activity }
		pam_investiture_council_force_gather_effect = yes
	}
}

# Hidden backstop check, ~2y after the nudge: if the crisis is still unresolved (faith still pending), settle
