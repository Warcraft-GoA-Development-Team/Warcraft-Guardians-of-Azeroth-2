# CB context. Snapshot the defeated titles before on_title_gain can alter realms.
wc_scarlet_stratholme_victory_effect = {
	title:d_stratholme = {
		if = {
			limit = { wc_scarlet_defeated_holder_trigger = yes }
			add_to_list = wc_stratholme_titles
		}
		every_in_de_jure_hierarchy = {
			continue = { always = yes }
			limit = {
				NOT = { this = title:d_stratholme }
				wc_scarlet_defeated_holder_trigger = yes
			}
			add_to_list = wc_stratholme_titles
		}
	}
	if = {
		limit = {
			character:59100 = {
				is_alive = yes
				wc_scarlet_defeated_character_trigger = yes
				NOT = { has_character_flag = wc_scarlet_naxxramas_retreated }
				any_held_title = { is_in_list = wc_stratholme_titles }
			}
		}
		character:59100 = {
			save_scope_as = wc_retreating_kelthuzad
			# Losing his last county may detach these vassals before the retreat.
			every_vassal = { add_to_list = wc_kelthuzad_vassals }
		}
	}
	create_title_and_vassal_change = {
		type = conquest_claim
		save_scope_as = wc_stratholme_change
		add_claim_on_loss = yes
	}
	every_in_list = {
		list = wc_stratholme_titles
		change_title_holder = {
			holder = scope:claimant
			change = scope:wc_stratholme_change
			take_baronies = no
		}
	}
	resolve_title_and_vassal_change = scope:wc_stratholme_change
	wc_scarlet_kelthuzad_retreat_effect = yes
}

wc_scarlet_kelthuzad_retreat_effect = {
	if = {
		limit = {
			exists = scope:wc_retreating_kelthuzad
			scope:wc_retreating_kelthuzad = {
				is_alive = yes
				NOT = { any_held_title = { is_in_list = wc_stratholme_titles } }
			}
			title:d_stratholme.holder = scope:claimant
			wc_scarlet_northrend_refuge_trigger = yes
		}
		title:h_scourge.holder = { save_scope_as = wc_scourge_refuge }
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = wc_kelthuzad_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = wc_kelthuzad_vassals
			limit = { is_alive = yes is_landed = yes NOT = { this = scope:wc_scourge_refuge } }
			change_liege = { liege = scope:wc_scourge_refuge change = scope:wc_kelthuzad_change }
		}
		scope:wc_retreating_kelthuzad = {
			# Strip remaining holdings individually; exterior subrealms stay Scourge.
			every_held_title = {
				change_title_holder = {
					holder = scope:wc_scourge_refuge
					change = scope:wc_kelthuzad_change
					take_baronies = no
				}
			}
		}
		resolve_title_and_vassal_change = scope:wc_kelthuzad_change
		scope:wc_retreating_kelthuzad = {
			if = { limit = { is_imprisoned = yes } release_from_prison = yes }
		}
		scope:wc_scourge_refuge = { add_courtier = scope:wc_retreating_kelthuzad }
		if = {
			limit = {
				scope:wc_retreating_kelthuzad = {
					is_alive = yes
					is_landed = no
					is_imprisoned = no
					is_courtier_of = scope:wc_scourge_refuge
				}
			}
			scope:wc_retreating_kelthuzad = {
				set_location = scope:wc_scourge_refuge.capital_province
				add_character_flag = wc_scarlet_naxxramas_retreated
			}
			scope:attacker = { trigger_event = wc_scarlet_stratholme.0001 }
		}
	}
}
