namespace = invasionofkalimdor

#### invasionofkalimdor.0001 -> scope out frostwolf clan and warsong clan
#### invasionofkalimdor.0002 -> thrall has a bad dream
#### invasionofkalimdor.0003 -> thrall and the prophet have a nice chit chat

invasionofkalimdor.0001 = {
    type = character_event
	content_source = dlc_GOA
    hidden = yes

    immediate = {
        if = {
            limit = {
                exists = title:k_frostwolf_clan.holder
            }
            title:k_frostwolf_clan.holder = { save_scope_as = thrall_or_frostwolf_leader }
        }
        else_if = {
            limit = {
                any_ruler = {
                    has_culture = culture:frostwolf
                    culture.culture_head = this
                }
            }
            save_scope_as = thrall_or_frostwolf_leader
        }
        else = {
            limit = { any_living_character = { is_thrall_trigger = yes } }
            save_scope_as = thrall_or_frostwolf_leader
        }

        if = {
            limit = {
                exists = title:k_warsong_clan.holder
            }
            title:k_warsong_clan.holder = { save_scope_as = grommash_or_warsong_leader }
        }
        else_if = {
            limit = {
                any_ruler = {
                    has_culture = culture:warsong
                    culture.culture_head = this
                }
            }
            save_scope_as = grommash_or_warsong_leader
        }
        else_if = {
            limit = { any_living_character = { is_grommash_trigger = yes } }
            save_scope_as = grommash_or_warsong_leader
        }
        else = { # Grommash doesn't even spawn most of the times in 583
            create_character = {
                template = grommash_character_template
                name = "Grommash"
                location = title:c_alterac_valley.title_province
                father = character:golmash
                save_scope_as = grommash_or_warsong_leader
            }
        }

        trigger_event = {
            id = invasionofkalimdor.0002
            days = 2
        }
    }
}

invasionofkalimdor.0002 = {
    type = character_event
    content_source = dlc_GOA
    title = invasionofkalimdor_0002_title
    desc = invasionofkalimdor_0002_desc
    theme = horde
    override_background = { reference = bedchamber }

    right_portrait = {
        character = scope:thrall_or_frostwolf_leader
        animation = fear
    }

    option = {
        name = invasionofkalimdor_0002_a
        trigger_event = invasionofkalimdor.0003
    }
}

invasionofkalimdor.0003 = {
    type = character_event
    content_source = dlc_GOA
    title = invasionofkalimdor_0003_title
    desc = invasionofkalimdor_0003_desc
    theme = horde

    left_portrait = {
