wc_scourge_plague.0009 = {
	type = character_event
	title = wc_scourge_plague.0009.t
	desc = wc_scourge_plague.0009.desc
	theme = plague
	override_background = { reference = dungeon }
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:prisoner
		animation = prisondungeon
	}
	cooldown = { years = 10 }

	trigger = {
		is_available_healthy_adult = yes
		is_travelling = no
		is_landed = yes
	}

	immediate = {
		random_owned_story = {
			limit = { story_type = story_cycle_scourge_plague }
			save_scope_as = plague_story
		}

		create_character = {
			template = merchant_template
			dynasty = none
			location = root.location
			culture = scope:plague_story.var:previous_culture
			faith = scope:plague_story.var:previous_faith
			gender_female_chance = 50
			save_scope_as = prisoner

			after_creation = {
				trigger_race_giving_no_gene_effect = yes
				random = { # He can really be a cultist
					chance = 50
					make_character_crypto_religionist_effect = { CRYPTO_RELIGION = scope:plague_story.var:plaguer.faith }
				}
			}
		}

		# Imprison it
		imprison = {
			target = scope:prisoner
			type = dungeon
		}

		save_court_physician_as_effect = { SCOPE_NAME = physician }
	}

	option = { # Study me this
		name = wc_scourge_plague.0009.a

		trigger = {
			court_physician_available_trigger = yes
		}

		random_list = {
			25 = {
				add_character_modifier = {
					modifier = plague_insights_gained
					years = 5
				}
				hidden_effect = {
					send_interface_toast = {
						title = wc_scourge_plague.0009.a.tt
						left_icon = root
					}
				}
			}
			75 = {}
		}

		ai_chance = {
			base = 100
		}
	}

	option = { # You're going to tell me everything you know
		name = wc_scourge_plague.0009.b

		root = { save_scope_as = actor }
		scope:prisoner = { save_scope_as = recipient }

		# Torture effects
		scope:recipient = {
			add_character_flag = {
				flag = is_being_tortured
				months = 6
			}
		}
		scope:actor = { save_scope_as = bg_override_char }
		###EFFECT TOOLTIP###
		torture_interaction_actor_effect = yes
		show_as_tooltip = {
			scope:recipient = {
				add_character_modifier = {
					modifier = recently_tortured
					years = 5
				}
			}
		}
		if = {
			limit = {
				scope:actor = {
					domicile ?= { has_domicile_parameter = estate_torture_grants_influence }
				}
			}
			scope:actor = { custom_tooltip = estate_torture_grants_influence_desc }
		}

		###EVENTS###
		hidden_effect = {
			#Torture method for descs
			random_list = {
				3 = { save_scope_value_as = { name = torture_method value = flag:rack } }
				3 = { save_scope_value_as = { name = torture_method value = flag:whip } }
				1 = { save_scope_value_as = { name = torture_method value = flag:rats } }
				8 = {
					trigger = {
						scope:actor = { has_trait = poet }
					}
					save_scope_value_as = {
						name = torture_method
						value = flag:poetry
					}
				}
			}

			#They have a secret I want to know about
			if = {
				limit = {
					scope:recipient = {
						OR = {
							any_secret = {
								torture_secret_trigger = { PARTICIPANT = scope:recipient }
							}
							any_known_secret = {
								secret_owner = { save_temporary_scope_as = torture_secret_owner }
								torture_secret_trigger = { PARTICIPANT = scope:torture_secret_owner }
							}
						}
					}
				}
				scope:recipient = {
					trigger_event = {
						id = prison.1001
						days = 1
					}
				}
			}
			else = {
				scope:actor = {
					trigger_event = {
						id = prison.1010
						days = 1
					}
				}
			}
		}

		# Dark Insights Perk:
		if = {
			limit = {
				scope:actor = { has_perk = dark_insights_perk }
			}
			scope:actor = {
				dark_insights_skill_gain_roll_effect = yes
			}
		}

		ai_chance = {
			base = 100

			dark_frenzy_torture_chance_modifiers = yes
			modifier = {
				has_trait = sadistic
				add = 30
			}
			ai_value_modifier = {
				ai_compassion = tiny_chance_impact_negative_ai_value #Adds +50 for highly uncompassionate characters, -50 for highly compassionate characters
			}
			opinion_modifier = {
				opinion_target = scope:recipient
				multiplier = -0.25
			}
		}
	}

	option = { # Stay at our court, we will protect you
		name = wc_scourge_plague.0009.c

		scope:prisoner = {
			release_from_prison = yes
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 25
			}
		}
		add_courtier = scope:prisoner

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_compassion = 1
				ai_energy = 1
			}
		}
	}
}

### They are coming ###
wc_scourge_plague.1000 = { #Scourge plague story cycle pacing control
	hidden = yes

	immediate = {
		if = {
			limit = {
				any_neighboring_top_liege_realm_owner = {
					has_character_flag = scourge_plague_splash
				}
			}
			if = {
				limit = {
					has_character_flag = scourge_plague_closer
				}
				remove_character_flag = scourge_plague_closer
			}
			add_character_flag = scourge_plague_nearby
		}
		else_if = {
			limit = {
				any_neighboring_top_liege_realm_owner = {
					any_neighboring_top_liege_realm_owner = {
						has_character_flag = scourge_plague_splash
					}
				}
			}	
			add_character_flag = scourge_plague_closer
		}
		if = {
			limit = {
				OR = {
					has_character_flag = scourge_plague_closer
					has_character_flag = scourge_plague_nearby
				}
				NOT = {
					has_character_flag = scourge_plague_info_event
				}
			}
			trigger_event = {
				id = wc_scourge_plague.1001
				days = { 2 4 }
			}
		}
	}
}
