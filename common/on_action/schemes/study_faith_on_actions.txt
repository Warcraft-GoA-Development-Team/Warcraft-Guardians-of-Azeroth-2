study_faith_success = {
	effect = {
		notify_puppeteer_of_scheme_outcome_effect = { OWNER = scope:owner TARGET = scope:target TYPE = event_toast_effect_good TITLE = scheme_outcome_study_faith_success_puppeteer }
		if = {
			limit = {
				has_pam_dlc_trigger = yes
				is_ruler = yes
				any_character_situation = {
					always = no # Warcraft - the_christian_church situation does not exist in GoA
					situation_top_has_catalyst = catalyst_the_christian_church_study_faith_scheme_completed
				}
			}
			situation:the_christian_church = {
				trigger_situation_catalyst = catalyst_the_christian_church_study_faith_scheme_completed
			}
		}
		if = {
			limit = { has_variable = pending_conversion_to_liege }
			# You were asked to convert and promised to do so after studying the faith.
			# Fired with a delay so the breakthrough event's add_known_rite lands first.
			trigger_event = {
				id = study_faith_outcome.0100
				days = 2
			}
		}
	}
	random_events = {
		100 = study_faith_outcome.1000 # Breakthrough - learn the full rite at once
		100 = study_faith_outcome.1500 # Nothing left to learn - end scheme gracefully
	}
}

# A phase below the success chance is still a minor success, just a slower one
study_faith_failure = {
	effect = {
		notify_puppeteer_of_scheme_outcome_effect = { OWNER = scope:owner TARGET = scope:target TYPE = event_toast_effect_neutral TITLE = scheme_outcome_study_faith_failure_puppeteer }
		if = {
			# We resolve a pending conversion here as we do on success, since 1500 would just end the scheme
			limit = {
				has_variable = pending_conversion_to_liege
				NOT = {
					study_faith_has_anything_to_learn_trigger = { RITE = scope:target.rite }
				}
			}
			trigger_event = {
				id = study_faith_outcome.0100
				days = 2
			}
		}
	}
	random_events = {
		100 = study_faith_outcome.1200 # Learn one Tenet
		100 = study_faith_outcome.1400 # Learn one Doctrine
		100 = study_faith_outcome.1500 # Nothing left to learn - end scheme gracefully
		100 = study_faith_outcome.0110 #You were asked to convert, you asked to study faith to know it better before converting
	}
}
