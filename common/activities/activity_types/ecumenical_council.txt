activity_ecumenical_council = {

	notify_player_can_join_activity = yes
	
	activity_group_type = grand
	
	is_shown = {
		has_pam_dlc_trigger = yes
		can_hold_ecumenical_council_trigger = yes
		# Don't offer a regular ecumenical council while this faith has an Investiture Council pending (it would preempt it).
		NOT = { faith = { has_variable = investiture_council_pending } }
	}
	
	can_start_showing_failures_only = {
		is_available_adult = yes
		is_landless_adventurer = no
		NOT = { is_activity_type_on_cooldown = activity_ecumenical_council }
		# Same guard as is_shown: block a routine council while the special one is pending.
		NOT = { faith = { has_variable = investiture_council_pending } }
		# Hard block, unlike the AI-only window below: two councils at once would fight over the same rite.
		custom_tooltip = {
			text = ecumenical_council_already_ongoing_tt
			NOT = { faith = { has_variable = ecumenical_council_ongoing } }
		}
		custom_tooltip = {
			text = can_host_ecumenical_council_tt
			OR = {
				any_held_title = { is_head_of_faith = yes }
				top_participant_group:the_christian_church ?= { participant_group_type = christian_clerical_main_power_rulers }
				government_has_flag = government_is_ecclesiastical
				has_theological_agent_puppet_trigger = yes
			}
		}
		trigger_if = {
			limit = { faith ?= { has_doctrine_parameter = has_clerical_regions } }
			custom_tooltip = {
				text = ecumenical_council_faith_needs_clerical_region_holders_tt
				faith = {
					any_faith_ruler = {
						count >= ecumenical_council_min_clerical_region_holders
						pam_is_clerical_region_holder_trigger = yes
					}
				}
			}
		}
	}

	is_valid = {
		scope:host = {
			is_physically_able = yes
			custom_tooltip = {
				text = can_host_ecumenical_council_tt
				OR = {
					any_held_title = { is_head_of_faith = yes }
					top_participant_group:the_christian_church ?= { participant_group_type = christian_clerical_main_power_rulers }
					government_has_flag = government_is_ecclesiastical
					has_theological_agent_puppet_trigger = yes
				}
			}
		}
		# If no one shows up
		trigger_if = {
			limit = { is_current_phase_active = yes }
			has_attending_activity_guests = yes
		}
		trigger_if = {
			limit = { is_current_phase_active = yes }
			custom_tooltip = {
				text = ecumenical_council_needs_clerical_region_quorum_tt
				pam_ecumenical_council_has_quorum_trigger = yes
			}
		}
	}

	on_invalidated = {
		# Read off the activity because the host may already be dead by the time we get here.
		if = {
			limit = { exists = var:ec_throttled_faith }
			var:ec_throttled_faith = { remove_variable = ecumenical_council_ongoing }
			# A council that collapsed before ruling on anything gives the faith its window back.
			if = {
				limit = { NOT = { exists = var:first_debate_resolution } }
				var:ec_throttled_faith = { remove_variable = recent_ecumenical_council }
			}
		}
		# A council cut short after ruling on something still owes the faith an account of it.
		if = {
			limit = {
				OR = {
					var:first_debate_resolution ?= flag:passed
					var:second_debate_resolution ?= flag:passed
					var:third_debate_resolution ?= flag:passed
				}
			}
			pam_ecumenical_council_send_ruling_letter_effect = yes
		}
		if = {
			limit = {
				scope:host = { is_landed = no }
			}
			every_attending_character = { trigger_event = activity_system.0321 }
		}
		else = {
			invalidation_incapable_effect = yes
			invalidation_imprisoned_effect = yes
			invalidation_nobody_present_effect = yes
			invalidation_host_isolated_effect = yes
			if = {
				limit = {
					NOT = { has_variable = invalidation_effect_set }
					NOT = { pam_ecumenical_council_has_quorum_trigger = yes }
				}
				every_attending_character = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = ecumenical_council_no_quorum_toast
						left_icon = scope:host
						custom_tooltip = ecumenical_council_no_quorum_toast_tt
					}
				}
			}
		}
	}

	on_host_death = {
		invalidation_host_death_effect = yes
	}
	
	is_location_valid = {
		custom_tooltip = {
			text = ecumenical_council_not_valid_province
			OR = {
				AND = { # Holy Sites held by someone of our faith
					barony ?= { is_holy_site_of = scope:host.faith }
					county.holder.faith ?= scope:host.faith
				}
				AND = { # Grand Cathedral of our faith
					county.holder.faith ?= scope:host.faith
					has_cathedral_great_project_building_trigger = yes
				}
				AND = { # Large Religious centers of our faith
					county.faith ?= scope:host.faith
					is_religious_studies_location = yes
				}
			}
		}
	}

	province_description = {
		# Is it a Holy Site?
		triggered_desc = {
			trigger = {
				barony = { is_holy_site_of = scope:host.faith }
			}
			desc = activity_ecumenical_council_holy_site_province
		}
		# Is it a Cathedral?
		triggered_desc = {
			trigger = {
				OR = {
					is_religious_studies_location = yes
					has_cathedral_great_project_building_trigger = yes
				}
			}
			desc = activity_ecumenical_council_cathedral_province
		}
		# Is it your capital?
		triggered_desc = {
			trigger = { this = scope:host.capital_province }
			desc = activity_ecumenical_capital_province
		}
	}
	
	province_score = {
		add = ecumenical_council_province_reference_value
	}

	province_filter = all
	ai_province_filter = holy_sites

	# desc is only shown in debug AI watch window
	ai_will_do = {
		value = 0
		# Base
		add = {
			value = 50
			desc = "base"
		}
		# We add their Learning
		add = {
			value = learning
			max = 30
			desc = "learning"
		}
		# More likely during a Crisis phase
		if = {
			limit = {
				situation:the_christian_church ?= {
					has_situation_top_phase_parameter = the_christian_church_cheaper_ecumenical_councils
				}
			}
			add = {
				value = 30
				desc = CHRISTIAN_CHURCH_PHASE_REASON
			}
		}
		# Personality
		if = { # Avoids responsibilities
			limit = { has_trait = lazy }
			add = {
				value = -20
				desc = lazy
			}
		}
		if = { # Likes showing off
			limit = { has_trait = arrogant }
			add = {
				value = 20
				desc = arrogant
			}
		}
		else_if = { # Or not
			limit = { has_trait = humble }
			add = {
				value = -10
				desc = humble
			}
		}
		if = { # Likes this kind of thing
			limit = {
				has_trait = erudite
			}
			add = {
				value = 20
				desc = erudite
			}
		}
		if = { # Likes this kind of thing
			limit = {
				has_trait = theologian
			}
			add = {
				value = 20
				desc = theologian
			}
		}
		if = { # Doesn't like this kind of thing
			limit = { has_trait = reclusive }
			add = {
				value = -25
				desc = reclusive
			}
		}

		add = activity_general_ai_will_do_value

		# Only holders of clerical titles call councils, and every such title is duchy tier or above.
		if = {
			limit = {
				NOR = {
					any_held_title = { has_clerical_region = yes }
					any_held_title = { is_head_of_faith = yes }
					this = rite.head_of_rite
					exists = clerical_elector_title
				}
			}
			multiply = {
				value = 0
				desc = "holds no clerical title"
			}
		}
		# One council per faith per generation.
		if = {
			limit = {
				faith = { has_variable = recent_ecumenical_council }
			}
			multiply = {
				value = 0
				desc = "faith already held a council recently"
			}
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60 # 5y
		kingdom = 60 # 5y
		empire = 60 # 5y
		hegemony = 60 # 5y
	}

	ai_will_select_province = {
		# Base
		add = {
			value = 5
			desc = "base"
		}
		if = {
			limit = {
				barony = { is_holy_site_of = scope:host.faith }
			}
			add = {
				value = 50
				desc = activity_ecumenical_council_holy_site_province
			}
		}
		if = {
			limit = {
				OR = {
					is_religious_studies_location = yes
					has_cathedral_great_project_building_trigger = yes
				}
			}
			add = {
				value = 25
				desc = activity_ecumenical_council_cathedral_province
			}
		}
		if = {
			limit = {
				this = scope:host.capital_province
			}
			add = {
				value = 15 # Even if the first Vatican EC was in the 19th c, the Lateran ECs (the medieval residence of the Pope) were very important
				desc = activity_ecumenical_capital_province
			}
		}
	}

	###################
	# PARAMETERS
	###################

	is_single_location = yes

	cooldown = { years = 25 }

	max_province_icons = 3

	max_route_deviation_mult = 3.0
	
	cost = {
		treasury = {
			if = {
				limit = {
					has_treasury = yes
				}
				value = ecumenical_council_scaled_cost
			}
		}
		gold = {
			if = {
				limit = {
					has_treasury = no
				}
				value = ecumenical_council_scaled_cost
			}
		}
	}

	# Used to show a rough minimum or expected cost at all to plan this
	ui_predicted_cost = {
		treasury_or_gold = ecumenical_council_scaled_cost
	}

	conclusion_description = {
		desc = ecumenical_council_conclusion.intro
	}

	###################
	# OPTIONS
	###################

	special_option_category = special_type

	options = {
		special_type = {
			# Tenet/Doctrinal council
			ecumenical_council_type_doctrine = {
				default = yes
				tenet_doctrine_based_activity = yes
				tenet_1 = known_character
				tenet_2 = known_character
				# Players can only pick doctrines their faith holds, so the AI follows the same rule
				doctrine_1 = active_faith
				ai_will_do = {
					value = 100
				}
			}
		}
	}

	###################
	# PHASES
	###################

	phases = {
		# Summons Phase
		ecumenical_council_phase_summons = {
			order = 1
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }
					# Add all the clergy characters to the clergy group
					ecumenical_council_gather_clergy_effect = yes
					# Add all non-clergy characters as non-clergy
					ecumenical_council_gather_non_clergy_effect = yes
				}
				# Trigger starting event
				trigger_event = ecumenical_council_events.0002
				# Progress phase
				if = {
					limit = {
						this = scope:host
					}
					scope:activity = { progress_activity_phase_after = { months = 1 } }
				}
			}

			on_weekly_pulse = {
				trigger_event = { on_action = ecumenical_council_summons_events }
			}
		}

		# First Debate
		ecumenical_council_phase_first_debate = {
			order = 2
			is_predefined = yes

			on_enter_phase = {
				scope:activity = {
					# Clear leftover current_topic or this debate's ruling lands on it
					if = {
						limit = { exists = activity_tenet:1 }
						activity_tenet:1 = {
							save_scope_as = ec_debate_topic_1
						}
						set_variable = {
							name = current_topic
							value = scope:ec_debate_topic_1
						}
					}
					else = {
						remove_variable = current_topic
					}
				}
				# First Debate starting event
				if = {
					limit = { this = scope:host }
					trigger_event = ecumenical_council_events.0010
					scope:activity = { progress_activity_phase_after = { days = 3 } }
				}
			}

			on_phase_active = {
				# Progress phase
				if = {
					limit = {
						this = scope:host
					}
					scope:activity = { progress_activity_phase_after = { months = 1 } }
				}
			}

			on_weekly_pulse = {
				trigger_event = { on_action = pam_ecumenical_council_debate_tenets_events }
			}

			# Resolve the debate vote (tally innovators vs conservatives, apply outcome)
			on_end = {
				if = {
					limit = { this = scope:host }
					trigger_event = ecumenical_council_events.0080
				}
			}
		}

		# Second Debate
		ecumenical_council_phase_second_debate = {
			order = 3
			is_predefined = yes

			on_enter_phase = {
				scope:activity = { progress_activity_phase_after = { days = 3 } }
			}

			on_phase_active = {
				scope:activity = {
					if = {
						limit = { exists = activity_tenet:2 }
						activity_tenet:2 = {
							save_scope_as = ec_debate_topic_2
						}
						set_variable = {
							name = current_topic
							value = scope:ec_debate_topic_2
						}
					}
					else = {
						remove_variable = current_topic
					}
				}
				# Second Debate starting event, all other events (including the weekly pulse) go through here
				if = {
					limit = { this = scope:host }
					trigger_event = ecumenical_council_events.0020
					scope:activity = { progress_activity_phase_after = { days = 3 } }
				}
				# Progress phase
				if = {
					limit = {
						this = scope:host
					}
					scope:activity = { progress_activity_phase_after = { months = 1 } }
				}
			}

			on_weekly_pulse = {
				trigger_event = { on_action = pam_ecumenical_council_debate_tenets_events }
			}

			# Resolve the debate vote (tally innovators vs conservatives, apply outcome)
			on_end = {
				if = {
					limit = { this = scope:host }
					trigger_event = ecumenical_council_events.0080
				}
			}
		}

		# Third Debate

					else = {

					}
				}

							OR = {
