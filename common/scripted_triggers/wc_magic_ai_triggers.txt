
wc_salvation_trigger = {
    custom_tooltip = {
        text = wc_salvation_trigger
        $TARGET$ = {
            is_alive = yes
            NOR = {
                this = root
                has_trait = being_demon
            }
        }
    }
}

has_good_salvation_target_trigger = {
    OR = {
        any_relation = { type = soulmate wc_salvation_trigger = { TARGET = this } }
        any_relation = { type = best_friend wc_salvation_trigger = { TARGET = this } }
        any_relation = { type = soulmate wc_salvation_trigger = { TARGET = this } }
        any_spouse = { wc_salvation_trigger = { TARGET = this } }
        any_child = { wc_salvation_trigger = { TARGET = this } }
        any_sibling = { wc_salvation_trigger = { TARGET = this } }
        any_parent = { wc_salvation_trigger = { TARGET = this } }
    }
}

is_good_salvation_target_trigger = {
    OR = {
        any_relation = { type = soulmate wc_salvation_trigger = { TARGET = $TARGET$ } }
        any_relation = { type = best_friend wc_salvation_trigger = { TARGET = $TARGET$ } }
        any_relation = { type = soulmate wc_salvation_trigger = { TARGET = $TARGET$ } }
        any_spouse = { wc_salvation_trigger = { TARGET = $TARGET$ } }
        any_child = { wc_salvation_trigger = { TARGET = $TARGET$ } }
        any_sibling = { wc_salvation_trigger = { TARGET = $TARGET$ } }
        any_parent = { wc_salvation_trigger = { TARGET = $TARGET$ } }
    }
}
