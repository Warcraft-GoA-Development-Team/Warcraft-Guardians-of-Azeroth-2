ai_will_cast_modifier = {
    modifier = { # Has some buff promoting casting
		any_spell_modifier_trigger = { OPERATOR = OR TRIGGER = has_positive_spell_modifier }
		add = 25
	}
    modifier = { # The more powerful will cast often
        wc_mp_value >= 5
		add = {
			value = wc_mp_value
			multiply = 1.5
		}
	}
    modifier = { # I have opps
		AND = {
            has_any_negative_relation_trigger = yes
            any_known_spell_trigger = { TRIGGER = spell_is_harmful_trigger } 
        }
		add = 25
	}
    modifier = { # I have homies <3
		AND = {
            has_any_positive_relation_trigger = yes
            any_known_spell_trigger = { TRIGGER = spell_is_helpful_trigger } 
        }
		add = 25
	}
    modifier = { 
        add = health 
        multiply = 2
    }
    modifier = {
		has_trait = brave
		add = 10
	}
	modifier = {
		has_trait = diligent
		add = 10
	}
    ai_value_modifier = {
		ai_energy = 1
	}
}
ai_wont_cast_modifier = {
    modifier = { # They're injured and cant buff themselves
		AND = {
            OR = {
                is_healthy = no
                target_has_magic_illness_trigger = yes 
                target_has_illness_trigger = yes
                target_has_magic_injury_trigger = yes
                target_has_injury_trigger = yes
                target_has_poison_trigger = yes 
            }
            NOR = { 
                any_known_spell_trigger = { TRIGGER = spell_is_dispel_type_trigger } 
                any_known_spell_trigger = { TRIGGER = spell_is_helpful_trigger } 
            }
        }
		add = 50
	}
    modifier = { # Has some debuff against casting
		any_spell_modifier_trigger = { OPERATOR = OR TRIGGER = has_negative_spell_modifier }
		add = 25
	}
    modifier = { # The lower mana percent you have, the less likely you are to cast a spell
		var:wc_current_mana_variable <= wc_thirty_percent_of_max_mana_value
		add = {
			value = 0
			add = var:wc_current_mana_percent_missing_value
			multiply = 100
		}
	}
	modifier = {
		has_trait = craven
		add = 10
	}
	modifier = {
		has_trait = lazy
		add = 15
	}
    ai_value_modifier = {
		ai_energy = -1
	}
}
