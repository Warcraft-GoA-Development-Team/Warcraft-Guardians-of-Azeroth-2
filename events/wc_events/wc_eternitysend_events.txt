namespace = eternitysend

#### eternitysend.0001 -> See if Cenarius is alive, if he is, then try different ways to kill him or if he is already dead start the invasion of Kaldorei

eternitysend.0001 = {
    scope = none
    hidden = yes

    immediate = {
        title:h_burning_legion.holder = { save_scope_as = archimonde }
        if = {
            limit = { character:410002 = { is_alive = no } }
            scope:archimonde = {
                start_war = {
                    casus_belli = mongol_invasion_war
                    target = title:e_kaldorei.holder
                    target_title = title:k_hyjal
                }
            }
        }
        else = {
            limit = { character:410002 = { is_alive = yes } }
            trigger_event = { id = eternitysend.0100 }
        }
    }
}

eternitysend.0100 = {
    scope = none
    hidden = yes

    immediate = {
        hidden_effect = {
            if = {
                limit = { any_living_character = { is_cenarius_trigger = yes } }
                save_scope_as = cenarius
            }

            if = {
                limit = {
                    any_independent_ruler = {
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:warsong
                        this = culture.culture_head
                    }
                }
                save_scope_as = warsong_or_alternative

                scope:warsong_or_alternative = {
                    trigger_event = { id = eternitysend.0200 }
                }
            }
            else_if = {
                limit = {
                    NOT = { exists = scope:warsong_or_alternative }
                    any_independent_ruler = {
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:durotaric
                        this = culture.culture_head
                    }
                }
                save_scope_as = frostwolf_or_alternative

                scope:frostwolf_or_alternative = {
                    trigger_event = { id = eternitysend.0300 }
                }
            }
            else = {
                limit = {
                    NAND = {
                        exists = scope:warsong_or_alternative
                        exists = scope:frostwolf_or_alternative
                    }
                }

                scope:archimonde = {
                    ordered_courtier = {
                        order_by = prowess
                        max = 1
                        limit = {
                            OR = {
                                has_trait = being_demon
                                has_trait = creature_fel_orc
                            }
                        }
                        save_scope_as = challenger_of_cenarius
                    }
                }

                scope:cenarius = {
                    trigger_event = { id = eternitysend.0400 }
                }
            }
        }
    }
}
