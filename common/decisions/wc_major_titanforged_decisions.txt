# Warcraft
use_titan_forge_to_create_pops = {
	ai_check_interval_by_tier = {
		barony = 0
		county = 12
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}
	picture = { reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds" }
	desc = use_titan_forge_to_create_pops_desc
	selection_tooltip = use_titan_forge_to_create_pops_tooltip
	decision_group_type = major

	is_shown = {
		is_landed = yes
		has_trait_with_flag = titanforged_class
	}

	is_valid = {
		owns_building_titan_forge = yes
		#To limit players
		any_courtier = { has_trait_with_flag = titanforged_class count < 6 }
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "TITAN_FORGE_CHOOSE_CREATION"
		show_from_start = yes

		item = {
			value = forge_titanforged_earthen
			is_shown = { culture = culture:titanforged }
			is_valid = { always = yes }
			current_description = forge_titanforged_earthen_desc
			localization = forge_titanforged_earthen
			icon = "gfx/interface/icons/traits/creature_dwarf.dds"
			ai_chance = {
				value = 100
				if = {
					limit = {
						OR = {
							NOT = { exists = cp:councillor_marshal }
							NOT = { exists = cp:councillor_steward }
							NOT = { any_knight = { is_alive = yes } }
						}
					}
					add = 400
				}
			}
		}

		item = {
			value = forge_titanforged_watcher
			is_shown = { culture = culture:titanforged }
			is_valid = { always = yes }
			current_description = forge_titanforged_watcher_desc
			localization = forge_titanforged_watcher
			icon = "gfx/interface/icons/traits/creature_watcher.dds"
			ai_chance = {
				value = 25
				if = {
					limit = { NOT = { any_courtier = { has_trait = creature_watcher } } }
					add = 200
				}
			}
		}

		item = {
			value = forge_existing_titanforged
			is_shown = { NOT = { culture = culture:titanforged } }
			is_valid = { always = yes }
			current_description = forge_existing_titanforged_desc
			localization = forge_existing_titanforged
			icon = "gfx/interface/icons/traits/creature_mechagnome.dds"
			ai_chance = { value = 100 }
		}
	}

	effect = {
		hidden_effect = {
			if = {
				limit = {
					is_landed = yes
					owns_building_titan_forge = yes
					any_courtier = { has_trait_with_flag = titanforged_class count < 6 }
				}
				if = {
					limit = {
						culture = culture:titanforged
						scope:forge_titanforged_earthen = yes
					}
					create_new_titanforged_earthen_courtier = yes
				}
				else_if = {
					limit = {
						culture = culture:titanforged
						scope:forge_titanforged_watcher = yes
					}
					create_new_titanforged_watcher_courtier = yes
				}
				else_if = {
					limit = {
						NOT = { culture = culture:titanforged }
						scope:forge_existing_titanforged = yes
					}
					create_new_titanforged_courtier = yes
				}
			}
		}
		custom_tooltip = use_titan_forge_to_create_pops_effect_tooltip
	}

	cooldown = { days = titan_forge_cooldown_time_of_forge }
	
	ai_will_do = {
		base = 40
		
		modifier = {
			factor = 2.25
			NOT = { any_courtier = { has_trait_with_flag = titanforged_class } }
		}
		
		modifier = {
			factor = 1.5
			any_courtier = { has_trait_with_flag = titanforged_class count < 5 }
		}
	}
}
