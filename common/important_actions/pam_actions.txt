# PAM player alerts for excommunication.
#
# action_excommunicated_recovery - shown while excommunicated. Click opens the right absolution path local bishop (Declaration of Repentance) OR travel-to-Rome petition (Lift Excommunication).
# action_excommunication_at_risk - warns the player they may be excommunicated soon.
# action_losing_realm_to_clerical_region - warns the player they may lose their current land if inheriting a region
#
action_excommunicated_recovery = {
	type = alert
	icon = action_take_decision
	priority = 900

	check_create_action = {
		if = {
			limit = {
				has_pam_dlc_trigger = yes
				has_trait = excommunicated
			}
			# Find a local bishop for the Declaration of Repentance path (skipped if player must petition Rome).
			excommunicated_recovery_find_cleric_effect = yes
			if = {
				limit = {
					exists = scope:repentance_cleric
				}
				try_create_important_action = {
					important_action_type = action_excommunicated_recovery
					actor = root
					recipient = scope:repentance_cleric
				}
			}
			# Still show the alert even with no bishop found; description explains other options.
			else = {
				try_create_important_action = {
					important_action_type = action_excommunicated_recovery
					actor = root
				}
			}
		}
	}

	effect = {
		# Year of penance: no absolution yet; tell the player to wait.
		if = {
			limit = {
				has_character_modifier = recent_excommunication
			}
			send_interface_toast = {
				type = event_toast_text_bad
				title = action_excommunicated_recovery_penance_toast
				desc = action_excommunicated_recovery_penance_toast_desc
				left_icon = root
			}
		}
		# Must petition the Head of Faith (kings+, or anyone the Head of Faith excommunicated personally)
		# We save religious_head_or_challenger, not religious_head
		# Antipope followers opened an empty window otherwise
		else_if = {
			limit = {
				petition_head_of_faith_repentance_requires_petition_trigger = yes
			}
			petition_head_of_faith_save_religious_authority_effect = { SCOPE = petition_religious_authority }
			open_interaction_window = {
				interaction = petition_head_of_faith_repentance_setup_interaction
				actor = root
				recipient = scope:petition_religious_authority
			}
		}
		# Local excommunication: open Declaration of Repentance with the bishop we found earlier.
		else_if = {
			limit = {
				exists = scope:recipient
			}
			open_interaction_window = {
				interaction = declaration_of_repentance_interaction
				actor = scope:actor
				recipient = scope:recipient
			}
		}
		# Fallback if no bishop was stored on the alert when it was created.
		else = {
			excommunicated_recovery_find_cleric_effect = yes
			if = {
				limit = {
					exists = scope:repentance_cleric
				}
				open_interaction_window = {
					interaction = declaration_of_repentance_interaction
					actor = root
					recipient = scope:repentance_cleric
				}
			}
			else = {
				send_interface_toast = {
					type = event_toast_text_bad
					title = action_excommunicated_recovery_no_cleric_toast
					desc = action_excommunicated_recovery_no_cleric_toast_desc
					left_icon = root
				}
			}
		}
	}
}

# Warns the player before excommunication lands: hostile Pope, or a neighbor who could request it.
action_excommunication_at_risk = {
	priority = 400
	combine_into_one = yes
	is_dangerous = yes
	type = action
	icon = action_take_decision

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				has_pam_dlc_trigger = yes
				exists = faith.religious_head
				NOR = {
					has_trait = excommunicated
					faith.religious_head = root
				}
				faith = { faith_has_sacraments_central_trigger = yes }
				faith = faith.religious_head.faith
				# No warning for an antipope's obedience, which the Pope they rejected cannot excommunicate
				NOT = {
					pam_antipope_protected_from_excommunication_trigger = { EXCOMMUNICATOR = faith.religious_head }
				}
				OR = {
					has_relation_rival = faith.religious_head
					has_relation_nemesis = faith.religious_head
					is_at_war_with = faith.religious_head
					AND = {
						opinion = {
							target = faith.religious_head
							value < 0
						}
						any_neighboring_top_liege_realm_owner = {
							faith = root.faith
							NOT = { this = root }
							OR = {
								has_relation_rival = root
								has_relation_nemesis = root
								is_at_war_with = root
							}
							root = { save_temporary_scope_as = secondary_recipient }
							save_temporary_scope_as = puppet_or_actor
							piety >= request_excommunication_piety_cost_value
						}
					}
				}
				NOR = {
					AND = {
						opinion = {
							target = faith.religious_head
							value >= 50
						}
						NOR = {
							has_relation_rival = faith.religious_head
							has_relation_nemesis = faith.religious_head
							is_at_war_with = faith.religious_head
						}
					}
				}
			}
			# Pope may excommunicate you directly.
			if = {
				limit = {
					OR = {
						has_relation_rival = faith.religious_head
						has_relation_nemesis = faith.religious_head
						is_at_war_with = faith.religious_head
					}
				}
				try_create_important_action = {
					important_action_type = action_excommunication_at_risk
					actor = root
					recipient = faith.religious_head
					secondary_recipient = faith.religious_head
				}
			}
			# A hostile neighbor with enough piety could request your excommunication.
			if = {
				limit = {
					opinion = {
						target = faith.religious_head
						value < 0
					}
				}
				every_neighboring_top_liege_realm_owner = {
					limit = {
						faith = root.faith
						NOT = { this = root }
						OR = {
							has_relation_rival = root
							has_relation_nemesis = root
							is_at_war_with = root
						}
						root = { save_temporary_scope_as = secondary_recipient }
						save_temporary_scope_as = puppet_or_actor
						piety >= request_excommunication_piety_cost_value
					}
					try_create_important_action = {
						important_action_type = action_excommunication_at_risk
						actor = root
						recipient = this
						secondary_recipient = root.faith.religious_head
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:secondary_recipient }
			scope:secondary_recipient = {
				open_view_data = {
					view = character
					player = scope:actor
				}
			}
		}
	}
}

# warns the player they may lose their current land if inheriting a region
action_losing_realm_to_clerical_region = {
	priority = 450
	combine_into_one = no
	is_dangerous = yes
	type = alert
	icon = action_take_decision

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				has_pam_dlc_trigger = yes
				is_in_line_for_external_clerical_region = yes
			}
			try_create_important_action = {
				important_action_type = action_losing_realm_to_clerical_region
				actor = root
			}
		}
	}

	effect = {
		faith = {
			random_clerical_region_title = {
				limit = {
					not = {
						root = {
							has_clerical_region_in_realm = prev
						}
					}
					any_title_heir = {
						this = root
					}
					"place_in_line_of_succession(root)" = 1
				}
				open_view_data = { view = title_view_window }
			}
		}
	}
}

action_can_request_new_clerical_region = {
	type = action
	icon = action_take_decision
	priority = 400

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				has_pam_dlc_trigger = yes
				government_has_flag = government_is_ecclesiastical
				is_landed = yes
				faith = { has_doctrine_parameter = has_clerical_regions }
				NOT = {
					any_held_title = {
						has_clerical_region = yes
					}
				}
				capital_county ?= {
					NOT = { exists = clerical_region_title }
				}
			}
			if = {
				limit = { exists = religious_head_or_challenger }
				religious_head_or_challenger = {
					save_temporary_scope_as = clerical_region_authority
				}
			}
			else_if = {
				limit = { exists = faith.main_rite.head_of_rite }
				faith.main_rite.head_of_rite = {
					save_temporary_scope_as = clerical_region_authority
				}
			}
			if = {
				limit = { exists = scope:clerical_region_authority }
				ordered_held_title = {
					order_by = {
						value = tier
						if = {
							limit = { title_capital_county = root.capital_county }
							add = 100
						}
					}
					max = 1
					check_range_bounds = no
					limit = {
						pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = root }
						save_temporary_scope_as = clerical_region_anchor_candidate
						root = {
							is_character_interaction_potentially_accepted = {
								interaction = request_ecclesiastical_title_creation_interaction
								recipient = scope:clerical_region_authority
								target_title = scope:clerical_region_anchor_candidate
							}
						}
					}
					save_temporary_scope_as = clerical_region_anchor
					root = {
						try_create_important_action = {
							important_action_type = action_can_request_new_clerical_region
							actor = root
							recipient = scope:clerical_region_authority
							target_title = scope:clerical_region_anchor
						}
					}
				}
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = request_ecclesiastical_title_creation_interaction
			actor = scope:actor
			recipient = scope:recipient
			target_title = scope:target_title
		}
	}
}

action_can_request_clerical_region_expansion = {
	type = action
	icon = action_take_decision
	priority = 400

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				has_pam_dlc_trigger = yes
				is_landed = yes
				pam_government_is_religious_playable_trigger = yes
				faith = { has_doctrine_parameter = has_clerical_regions }
				NOT = { pam_is_clerical_title_authority_trigger = yes }
				capital_county ?= {
					clerical_region_title ?= {
						has_clerical_region = yes
						is_head_of_faith = no
						holder ?= root
					}
				}
			}
			if = {
				limit = { exists = religious_head_or_challenger }
				religious_head_or_challenger = {
					save_temporary_scope_as = clerical_region_authority
				}
			}
			else_if = {
				limit = { exists = faith.main_rite.head_of_rite }
				faith.main_rite.head_of_rite = {
					save_temporary_scope_as = clerical_region_authority
				}
			}
			if = {
				limit = {
					exists = scope:clerical_region_authority
					scope:clerical_region_authority != root
				}
				capital_county.clerical_region_title = {
					save_temporary_scope_as = acting_clerical_region
					every_county_in_clerical_region = {
						every_neighboring_county = {
							limit = {
								exists = duchy
								NOT = { clerical_region_title ?= scope:acting_clerical_region }
							}
							duchy = {
								if = {
									limit = {
										NOR = {
											is_in_list = preferred_clerical_region_expansion_targets
											is_in_list = clerical_region_expansion_targets
										}
										save_temporary_scope_as = clerical_region_expansion_candidate
										root = {
											has_variable_list = clergy_expands
											any_in_list = {
												variable = clergy_expands
												this = scope:clerical_region_expansion_candidate
											}
										}
									}
									add_to_temporary_list = preferred_clerical_region_expansion_targets
								}
								else_if = {
									limit = {
										NOR = {
											is_in_list = preferred_clerical_region_expansion_targets
											is_in_list = clerical_region_expansion_targets
										}
									}
									add_to_temporary_list = clerical_region_expansion_targets
								}
							}
						}
					}
				}
				ordered_in_list = {
					list = preferred_clerical_region_expansion_targets
					max = 1
					check_range_bounds = no
					limit = {
						save_temporary_scope_as = clerical_region_expansion_candidate
						root = {
							is_character_interaction_potentially_accepted = {
								interaction = request_ecclesiastical_title_expansion_interaction
								recipient = scope:clerical_region_authority
								target_title = scope:clerical_region_expansion_candidate
							}
						}
					}
					save_temporary_scope_as = clerical_region_expansion_target
				}
				if = {
					limit = { NOT = { exists = scope:clerical_region_expansion_target } }
					ordered_in_list = {
						list = clerical_region_expansion_targets
						max = 1
						check_range_bounds = no
						limit = {
							save_temporary_scope_as = clerical_region_expansion_candidate
							root = {
								is_character_interaction_potentially_accepted = {
									interaction = request_ecclesiastical_title_expansion_interaction
									recipient = scope:clerical_region_authority
									target_title = scope:clerical_region_expansion_candidate
								}
							}
						}
						save_temporary_scope_as = clerical_region_expansion_target
					}
				}
				if = {
					limit = { exists = scope:clerical_region_expansion_target }
					try_create_important_action = {
						important_action_type = action_can_request_clerical_region_expansion
						actor = root
						recipient = scope:clerical_region_authority
						target_title = scope:clerical_region_expansion_target
					}
				}
			}
		}
	}

	effect = {

			interaction = request_ecclesiastical_title_expansion_interaction
			actor = scope:actor
			recipient = scope:recipient
			target_title = scope:target_title
		}
