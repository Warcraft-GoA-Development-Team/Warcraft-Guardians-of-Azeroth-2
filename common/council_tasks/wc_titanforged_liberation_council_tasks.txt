task_reorder_county = {
	position = councillor_marshal

	is_shown = {
		scope:councillor_liege = { has_trait_with_flag = titanforged_class }
	}

	task_type = task_type_county
	county_target = realm
	ai_county_target = realm
	task_progress = task_progress_value
	task_current_value = scope:councillor_liege.wc_titanforged_reorder_county_current_progress
	task_max_value = wc_titanforged_reorder_county_threshold
	highlight_own_realm = yes
	restart_on_finish = no

	asset = {
		icon = "gfx/interface/icons/council_task_types/task_purge_population.dds"
		background = "gfx/interface/buttons/button_round_bg.dds"
		frame = "gfx/interface/buttons/button_round_frame.dds"
		glow = "gfx/particles/halo.dds"
	}

	effect_desc = { desc = task_reorder_county_desc }

	potential_county = {
		custom_tooltip = {
			text = wc_titanforged_reorder_target_tt
			scope:county = { wc_titanforged_reorder_target_trigger = yes }
		}
	}

	progress = {
		value = 0
		add = {
			value = wc_titanforged_reorder_county_base
			desc = WC_TITANFORGED_REORDER_BASE
		}
		add = {
			value = wc_titanforged_reorder_county_skill_bonus
			desc = WC_TITANFORGED_REORDER_SKILL_BONUS
		}
		add = {
			value = wc_titanforged_reorder_county_friend_bonus
			desc = WC_TITANFORGED_REORDER_FRIEND_BONUS
		}
		add = {
			value = wc_titanforged_reorder_county_best_friend_bonus
			desc = WC_TITANFORGED_REORDER_BEST_FRIEND_BONUS
		}
		add = {
			value = wc_titanforged_reorder_county_rival_bonus
			desc = WC_TITANFORGED_REORDER_RIVAL_BONUS
		}
		add = {
			value = wc_titanforged_reorder_county_nemesis_bonus
			desc = WC_TITANFORGED_REORDER_NEMESIS_BONUS
		}
		add = {
			value = wc_titanforged_reorder_county_same_culture_bonus
			desc = WC_TITANFORGED_REORDER_SAME_CULTURE_BONUS
		}
		add = {
			value = wc_titanforged_reorder_county_same_faith_bonus
			desc = WC_TITANFORGED_REORDER_SAME_FAITH_BONUS
		}
		add = {
			value = wc_titanforged_reorder_county_development_penalty
			desc = WC_TITANFORGED_REORDER_DEVELOPMENT_PENALTY
		}
		add = {
			value = wc_titanforged_reorder_county_unraised_army_bonus
			desc = WC_TITANFORGED_REORDER_UNRAISED_ARMY_BONUS
		}
		min = 0
		multiply = wc_titanforged_restoration_speed
	}

	on_start_task_county = {
		scope:councillor_liege = { wc_titanforged_restoration_detach_effect = yes }
		scope:county = {
			add_county_modifier = wc_titanforged_ongoing_reordering_county_modifier
			# Retain assignment identity even if its story ends before an engine callback.
			remove_variable = wc_titanforged_reorder_implantation
			set_variable = { name = wc_titanforged_reorder_county_project value = yes }
			set_variable = { name = wc_titanforged_reorder_current_progress value = 0 }
			if = {
				limit = { NOT = { has_variable = wc_titanforged_reorder_started } }
				set_variable = { name = wc_titanforged_reorder_started value = 0 }
			}
			change_variable = { name = wc_titanforged_reorder_started add = 1 }
		}
		scope:councillor_liege = { wc_titanforged_restoration_start_effect = yes }
	}

	on_cancel_task_county = {
		scope:councillor_liege = { wc_titanforged_restoration_detach_effect = yes }
		scope:county = {
			remove_county_modifier = wc_titanforged_ongoing_reordering_county_modifier
			remove_variable = wc_titanforged_reorder_current_progress
		}
	}

	on_finish_task_county = {
		if = {
			limit = {
				scope:county = { wc_titanforged_reorder_target_trigger = yes }
				scope:county = { wc_titanforged_reorder_assignment_trigger = yes }
				wc_titanforged_settlement_finish_ready_trigger = yes
				OR = {
					AND = {
						NOT = { exists = scope:councillor_liege.var:wc_titanforged_restoration }
						scope:county = { NOT = { has_variable = wc_titanforged_reorder_implantation } }
						scope:county = { NOT = { has_variable = wc_titanforged_reorder_county_project } }
						scope:county = { wc_titanforged_historical_county_trigger = yes }
					}
					AND = {
						exists = scope:councillor_liege.var:wc_titanforged_restoration
						scope:councillor_liege.var:wc_titanforged_restoration = {
							NOT = { var:wc_titanforged_restoration_method = flag:pending }
						}
					}
				}
			}
			scope:county = {
				wc_titanforged_restoration_complete_county_effect = yes
			}
			scope:councillor_liege = { trigger_event = wc_titanforged_reorder_task.0001 }
		}
		scope:county = {
			remove_county_modifier = wc_titanforged_ongoing_reordering_county_modifier
			remove_variable = wc_titanforged_reorder_current_progress
		}
		scope:councillor_liege = { wc_titanforged_restoration_detach_effect = yes }
	}

	on_monthly_county = {
		if = {
			limit = {
				scope:county = { wc_titanforged_reorder_target_trigger = yes }
				scope:county = { wc_titanforged_reorder_assignment_trigger = yes }
				OR = {
					exists = scope:councillor_liege.var:wc_titanforged_restoration
					AND = {
						scope:county = { NOT = { has_variable = wc_titanforged_reorder_implantation } }
						scope:county = { NOT = { has_variable = wc_titanforged_reorder_county_project } }
					}
				}
			}
			scope:county = {
				if = {
					limit = { NOT = { has_variable = wc_titanforged_reorder_current_progress } }
					set_variable = { name = wc_titanforged_reorder_current_progress value = 0 }
				}
				change_variable = {
					name = wc_titanforged_reorder_current_progress
					add = scope:councillor.council_task_monthly_progress
				}
				wc_titanforged_settlement_tick_effect = yes
			}
		}
		else = {
			scope:councillor = { start_default_task = yes }
		}
		scope:councillor_liege.var:wc_titanforged_restoration ?= { wc_titanforged_implantation_check_end_effect = yes }
	}

	ai_target_score = {
		value = 20
	}

	ai_will_do = {
		value = 50
	}
}
