pam_interactions_events.0001 = {
	type = letter_event
	opening = {
		desc = pam_interactions_events.0001.opening
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:education_hook = yes
				}
				desc = pam_interactions_events.0001.hook_intro
			}
			desc = pam_interactions_events.0001.intro
		}
		desc = pam_interactions_events.0001
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:offer_hook = yes
				}
				desc = pam_interactions_events.0001.hook
			}
			triggered_desc = {
				trigger = {
					scope:tithe = yes
				}
				desc = pam_interactions_events.0001.tithe
			}
			triggered_desc = {
				trigger = {
					scope:donate_faith = yes
				}
				desc = pam_interactions_events.0001.donate
			}
			triggered_desc = {
				trigger = {
					scope:monastic_support = yes
				}
				desc = pam_interactions_events.0001.monks
			}
			desc = pam_interactions_events.0001.normal
		}
		desc = pam_interactions_events.0001.mid
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						character_considered_wicked_trigger = { RITE = scope:recipient.rite }
						scope:recipient = {
							has_relation_rival = scope:puppet_or_actor
						}
						tyranny > medium_tyranny
						has_trait = excommunicated
					}
					scope:recipient = {
						opinion = {
							target = scope:puppet_or_actor
							value < 50
						}
						NOT = {
							has_dread_level_towards = {
								target = scope:puppet_or_actor
								level >= 2
							}
						}
					}
				}
				desc = pam_interactions_events.0001_disgust
			}
			triggered_desc = {
				trigger = {
					OR = {
						character_considered_morally_dubious_trigger = { RITE = scope:recipient.rite }
						scope:recipient = {
							is_ai = yes
							opinion = {
								target = scope:puppet_or_actor
								value <= -50
							}
						}
						tyranny >= low_tyranny
					}
					scope:recipient = {
						opinion = {
							target = scope:puppet_or_actor
							value < 50
						}
					}
				}
				desc = pam_interactions_events.0001_disapprove
			}
			triggered_desc = {
				trigger = {
					OR = {
						piety_level >= 4
						has_trait = saint
						has_trait = herald
						#has_trait = saoshyant_descendant
						has_trait = order_member
						has_trait = faith_warrior
						#has_trait = saoshyant
						has_trait = paragon
						has_trait = savior
						has_trait = crusader_king
						has_trait = zealous
						has_trait = theologian
						has_trait = divine_blood
						has_trait = consecrated_blood
						has_trait = sayyid
					}
				}
				desc = pam_interactions_events.0001_approve
			}
			triggered_desc = {
				trigger = {
					scope:offer_hook = yes
				}
				desc = pam_interactions_events.0001_offer_hook_end
			}
		}
	}
	sender = scope:recipient

	immediate = {
		request_legitimacy_interaction_effect = yes
		notify_puppeteer_of_outcome_effect = {
			TYPE = event_toast_effect_good
			TITLE = pam_interactions_events.0001.toast_puppeteer
		}
	}
	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = {
							ai_zeal < low_negative_ai_value
						}
						desc = pam_interactions_events.0001.a.low_zeal
					}
					triggered_desc = {
						trigger = {
							ai_zeal > low_positive_ai_value
						}
						desc = pam_interactions_events.0001.a.high_zeal
					}
					desc = pam_interactions_events.0001.a
				}
			}
		}
	}
}

#Letter event for player clergy to grant claim
pam_interactions_events.0023 = {
	orphan = yes # Warcraft - caller commented out in GoA
	type = letter_event
	opening = {
		desc = pam_interactions_events.0023.opening
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor.highest_held_title_tier < scope:target.tier
					
				}
				desc = pam_interactions_events.0023_shrimp
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor.highest_held_title_tier_minus_one > scope:target.tier
				}
				desc = pam_interactions_events.0023_shark
			}
			desc = pam_interactions_events.0023
		}
		desc = pam_interactions_events.0023_end
	}
	sender = scope:puppet_or_actor

	immediate = {
	}
	#Yes
	option = {
		name = pam_interactions_events.0023.a
		scope:puppet_or_actor = {
			ecclesiastic_grant_claim_interaction_effect = yes
			trigger_event = pam_interactions_events.0020
			scope:recipient = {
				#Event where they're informed
				trigger_event = pam_interactions_events.0022
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = { clerical_hesitation_claim_interaction_value >= -2 }
			}
			stress_impact = {
				zealous = medium_stress_impact_gain
				just = medium_stress_impact_gain
				honest = medium_stress_impact_gain
			}
		}
		else = {
			stress_impact = {
				just = medium_stress_impact_gain
				honest = medium_stress_impact_gain
			}
		}
	}
	#No
	option = {
		name = pam_interactions_events.0023.b
		scope:puppet_or_actor = {
			trigger_event = pam_interactions_events.0021
			#Ex. a king trying to claim a county
			if = {
				limit = {
					highest_held_title_tier_minus_one > scope:target.tier
				}
				add_opinion = {
					modifier = refusal_opinion
					target = scope:secondary_recipient
					opinion = -15
				}
			}
			#Ex. a king trying to claim a duchy
			if = {
				limit = {
					highest_held_title_tier > scope:target.tier
				}
				add_opinion = {
					modifier = refusal_opinion
					target = scope:secondary_recipient
					opinion = -25
				}
			}
			#Ex. a king trying to claim a kingdom
			if = {
				limit = {
					highest_held_title_tier = scope:target.tier
				}
				add_opinion = {
					modifier = refusal_opinion
					target = scope:secondary_recipient
					opinion = -30
				}
			}
			#Ex. a king trying to claim an empire
			if = {
				limit = {
					highest_held_title_tier < scope:target.tier
				}
				add_opinion = {
					modifier = refusal_opinion
					target = scope:secondary_recipient
					opinion = -35
				}
			}
			root = {
				if = {
					limit = {
						scope:target.tier > root.highest_held_title_tier
					}
					add_piety = medium_piety_gain
				}
				else_if = {
					limit = {
						scope:target.tier < root.highest_held_title_tier_minus_one
					}
					add_piety = miniscule_piety_gain
				}
				else = {
					add_piety = minor_piety_gain
				}
			}
		}
		scope:recipient = {
			add_opinion = {
				opinion = {
					value = 5
					if = {
						limit = {
							highest_held_title_tier = scope:target.tier
						}
						multiply = 6
					}
					else_if = {
						limit = {
							highest_held_title_tier_minus_one = scope:target.tier
						}
						multiply = 4
					}
					else_if = {
						limit = {
							highest_held_title_tier_minus_two = scope:target.tier
						}
						multiply = 3
					}
				}
				modifier = grateful_opinion
				target = scope:secondary_recipient
			}
		}
		if = {
			limit = {
				OR = {
					scope:puppet_or_actor.highest_held_title_tier_minus_one > scope:target.tier
					scope:threaten_priest = yes
				}
			}
			stress_impact = {
				craven = minor_stress_impact_gain
			}
		}
		else = {
			stress_impact = {
				generous = minor_stress_impact_gain
			}
		}
	}
}

#Events around transferring land from interaction
#Actor chooses negotiation method to use after recipient doesn't instantly accept
pam_interactions_events.0030 = {
	type = character_event
	title = pam_interactions_events.0030.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = {
						scope:representative = root
					}
				}
				desc = pam_interactions_events.0030.desc_intro_envoy
			}
			desc = pam_interactions_events.0030.desc_intro
		}
		desc = pam_interactions_events.0030.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = {
						scope:recipient = scope:target.holder
					}
				}
				desc = pam_interactions_events.0030.desc_vassal
			}
			desc = pam_interactions_events.0030.desc_target
		}
	}
	theme = vassal
	override_background = {
		reference = holy_site_generic
	}
	left_portrait = {
		character = scope:representative
		triggered_animation = {
			trigger = {
				ai_compassion <= 0
			}
			animation = villain
		}
		animation = zealous_superior
		camera = camera_event_very_left
	}	
	right_portrait = {
		character = scope:recipient
		triggered_animation = {
			trigger = {
				has_any_good_relationship_with_character_trigger = { CHARACTER = scope:actor }
				ai_compassion >= low_positive_ai_value
			}
			animation = sadness
		}
		triggered_animation = {
			trigger = {
				scope:recipient = {
					opinion = {
						target = scope:actor
						value >= 60
					}
				}
			}
			animation = contemplating
		}
		animation = war_defender
		camera = camera_event_very_right
	}
	lower_center_portrait = scope:actor_non_present
	lower_right_portrait = scope:vassal
	trigger = {
		is_alive = yes
		is_imprisoned = no
		is_ruler = yes
		scope:recipient = {
			is_ruler = yes
			is_alive = yes
			trigger_if = {
				limit = {
					is_imprisoned = yes
				}
				imprisoner = {
					OR = {
						this = root
						any_liege_or_above = {
							this = root
						}
					}
				}
			}
		}
		scope:target.holder ?= {
			OR = {
				this = scope:recipient
				is_vassal_or_below_of = scope:recipient
			}
		}
	}
	immediate = {
		if = {
			limit = {
				scope:target.holder = {
					NOT = {
						this = scope:recipient
					}
				}
			}
			scope:target.holder = { save_scope_as = vassal }
		}
		#when root is high tier negotiating with low tier, flavor feels better with priest serving as envoy
		if = {
			limit = {
				highest_held_title_tier >= tier_kingdom
				scope:recipient.highest_held_title_tier <= tier_duchy
				scope:secondary_recipient = {
					is_available_quick = {
						alive = yes
						incapable = no
					}
					trigger_if = {
						limit = {
							is_imprisoned = yes
						}
						imprisoner = {
							OR = {
								this = root
								any_liege_or_above = {
									this = root
								}
							}
						}
					}
					has_contagious_deadly_disease_trigger = no
				}
			}
			scope:secondary_recipient = {
				save_scope_as = representative
			}
		}
		else = {
			save_scope_as = representative
			#We want to show clergy at bottom only in this circumstance
			scope:secondary_recipient = {
				save_scope_as = actor_non_present
			}
		}
		#save forceable relation scope so that changing relations doesn't mess with outcome later
		if = {
			limit = {
				pam_negotiate_land_forced_relation_trigger = yes
			}
			save_scope_as = forceable_relation
		}
	}
	#use hook
	option = {
		trigger = {
			OR = {
				pam_negotiate_land_forced_hook_trigger = yes
				pam_negotiate_land_possible_hook_trigger = yes
			}
		}
		name = pam_interactions_events.0030.a
		flavor = pam_interactions_events.0030.a.flavor
		save_scope_as = method_hook
		show_as_tooltip = {
			if = {
				limit = {
					pam_negotiate_land_forced_hook_trigger = yes
				}
				add_internal_flag = special
				custom_tooltip = pam_forced_negotiate_land_hook_tt
				custom_tooltip = clergy_land_interaction_recipient_forced_tt
			}
			else = {
				if = {
					limit = {
						scope:recipient = {
							is_ai = yes
							ai_honor > medium_positive_ai_value
						}
					}
					custom_tooltip = clergy_land_increased_possibility_hook_tt
				}
				custom_tooltip = clergy_land_interaction_recipient_possible_tt
				pam_negotiate_land_hook_effect = yes
				ecclesiastic_transfer_land_interaction_transfer_effect = yes
				custom_tooltip = clergy_land_interaction_recipient_refuse_tt
				clergy_land_refusal_legitimacy_piety_effect = yes
			}
		}
		if = {
			limit = {
				pam_negotiate_land_forced_hook_trigger = yes
			}
			save_scope_as = forced_acceptance
		}
		ai_chance = {
			base = 25
			ai_value_modifier = {
				ai_energy = 0.5
				ai_rationality = -0.5
			}
			modifier = {
				add = 25
				NOT = { is_vassal_of = scope:recipient }
			}
			modifier = {
				add = 25
				scope:recipient = {
					ai_honor > medium_positive_ai_value
				}
			}
			modifier = {
				add = 25
				scope:recipient = {
					ai_honor > 0
				}
			}
			modifier = {
				factor = 10
				pam_negotiate_land_forced_hook_trigger = yes
			}
		}
	}
	#Destroy relations
	option = {
		trigger = {
			OR = {
				pam_negotiate_land_forced_relation_trigger = yes
				pam_negotiate_land_possible_relation_trigger = yes
			}
		}
		name = pam_interactions_events.0030.b
		flavor = pam_interactions_events.0030.b.flavor
		save_scope_as = method_relation
		show_as_tooltip = {
			if = {
				limit = {
					scope:recipient = { is_ai = yes }
					pam_negotiate_land_forced_relation_trigger = yes
				}
				add_internal_flag = special
				custom_tooltip = clergy_land_interaction_recipient_forced_tt
				show_as_tooltip = {
					pam_negotiate_land_relation_effect = yes
					ecclesiastic_transfer_land_interaction_transfer_effect = yes
				}
			}
			else = {
				if = {
					limit = {
						scope:recipient = {
							is_ai = yes
							ai_compassion > medium_positive_ai_value
						}
					}
					custom_tooltip = clergy_land_increased_possibility_relation_tt
				}
				custom_tooltip = clergy_land_interaction_recipient_possible_tt
				pam_negotiate_land_relation_accept_effect = yes
				ecclesiastic_transfer_land_interaction_transfer_effect = yes
				custom_tooltip = clergy_land_interaction_recipient_refuse_tt
				pam_negotiate_land_relation_effect = yes
				clergy_land_refusal_legitimacy_piety_effect = yes
			}
		}
		if = {
			limit = {
				pam_negotiate_land_forced_relation_trigger = yes
			}
			save_scope_as = forced_acceptance
		}
		stress_impact = {
			deceitful = minor_stress_impact_loss
			fickle = minor_stress_impact_loss
			compassionate = medium_stress_impact_gain
			trusting = medium_stress_impact_gain
			generous = minor_stress_impact_gain
			forgiving = medium_stress_impact_gain
		}
		ai_chance = {
			base = 25
			ai_value_modifier = {
				ai_compassion = -1
				ai_honor = -0.5
			}
			modifier = {
				add = 50
				NOT = { is_allied_to = scope:recipient }
				half_current_military_strength >= scope:recipient.twice_current_military_strength
				spiritual_fulfillment >= 0
			}
			modifier = {
				add = 25
				scope:recipient = {
					ai_compassion > medium_positive_ai_value
				}
			}
			modifier = {
				add = 25
				scope:recipient = {
					ai_compassion > 0
				}
			}
			modifier = {
				factor = 10
				pam_negotiate_land_forced_relation_trigger = yes
			}
			modifier = {
				factor = 0
				OR = {
					has_trait = compassionate
					has_trait = trusting
					has_trait = generous
					has_trait = forgiving
				}
			}
		}
