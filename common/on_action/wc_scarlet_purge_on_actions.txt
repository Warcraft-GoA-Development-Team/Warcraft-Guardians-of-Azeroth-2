quarterly_playable_pulse = { on_actions = { wc_scarlet_purge_pulse } }
on_imprison = { on_actions = { wc_scarlet_purge_captured } }

wc_scarlet_purge_pulse = {
	trigger = { wc_scarlet_purge_authority_trigger = yes }
	effect = { wc_scarlet_purge_scan_effect = yes }
}

wc_scarlet_purge_captured = {
	trigger = {
		has_trait = being_undead
		var:wc_scarlet_purge_summons ?= scope:imprisoner
		scope:imprisoner = { wc_scarlet_purge_authority_trigger = yes }
	}
	effect = {
		scope:imprisoner = { save_scope_as = wc_purge_authority }
		wc_scarlet_purge_open_trial_effect = yes
	}
}
