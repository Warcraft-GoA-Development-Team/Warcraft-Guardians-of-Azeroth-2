wc_balnazzar_contingency = {
	# Basic Setup
	skill = intrigue
	desc = wc_balnazzar_contingency_desc
	success_desc = "WC_BALNAZZAR_CONTINGENCY_SUCCESS_DESC"
	icon = icon_scheme_personal
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	category = personal
	target_type = character

	is_secret = no
	is_basic = yes
	uses_resistance = no
	cooldown = { years = 5 }

	# Parameters
	speed_per_skill_point = -15
	spymaster_speed_per_skill_point = 0
	base_progress_goal = 730
	base_maximum_success = 95
	minimum_success = 5

	# Core Triggers
	allow = {
		is_imprisoned = no
		is_ruler = yes
	}
	valid = {
		# Balnazzar's own story hinges on this specific body (Grand Crusader,
		# the whole Scarlet Crusade chain downstream) - restricted here too,
		# not just at the interaction level, so no other path (AI, debug) can
		# start this against anyone else.
		scope:target = character:60021 # Saidan Dathrohan
		scope:target = {
			is_adult = yes
			is_imprisoned = no
			exists = location
		}
		NOT = { is_at_war_with = scope:target }
	}

	odds_prediction = {
		add = base_odds_prediction_target_is_char_value
		min = 0
	}

	# Base Chances
	base_success_chance = {
		base = 30

		modifier = {
			add = 2
			scope:owner = { has_trait = education_intrigue_1 }
			desc = "SCHEME_WC_BALNAZZAR_CONTINGENCY_MY_EDUCATION"
		}
		modifier = {
			add = 4
			scope:owner = { has_trait = education_intrigue_2 }
			desc = "SCHEME_WC_BALNAZZAR_CONTINGENCY_MY_EDUCATION"
		}
		modifier = {
			add = 6
			scope:owner = { has_trait = education_intrigue_3 }
			desc = "SCHEME_WC_BALNAZZAR_CONTINGENCY_MY_EDUCATION"
		}
		modifier = {
			add = 8
			scope:owner = { has_trait = education_intrigue_4 }
			desc = "SCHEME_WC_BALNAZZAR_CONTINGENCY_MY_EDUCATION"
		}
		modifier = {
			add = 10
			scope:owner = { has_trait = education_intrigue_5 }
			desc = "SCHEME_WC_BALNAZZAR_CONTINGENCY_MY_EDUCATION"
		}
		modifier = {
			add = -10
			scope:target = { has_trait = paranoid }
			desc = "SCHEME_WC_BALNAZZAR_CONTINGENCY_THEIR_PARANOIA"
		}
		modifier = {
			add = 5
			scope:target = { has_trait = trusting }
			desc = "SCHEME_WC_BALNAZZAR_CONTINGENCY_THEIR_TRUST"
		}
	}

	# On Actions
	on_start = {
		scheme_owner = {
			trigger_event = wc_balnazzar_contingency_outcome.1001
		}
	}
	on_phase_completed = {
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }
		scope:scheme.scheme_owner = {
			trigger_event = wc_balnazzar_contingency_outcome.0001
		}
	}
	on_invalidated = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:owner = {
			send_interface_toast = {
				type = event_generic_neutral
				title = wc_balnazzar_contingency_invalidated_title
				custom_tooltip = wc_balnazzar_contingency_invalidated_desc
			}
		}
	}
	on_monthly = {
		scheme_owner = {
			trigger_event = { on_action = wc_balnazzar_contingency_ongoing }
		}
	}
}
