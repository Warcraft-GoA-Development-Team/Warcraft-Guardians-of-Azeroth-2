# Used by wc_heresy.0001 / 0010 / 0011 (events/religion_events/wc_heresy_events.txt)
# and by Component 4 of the heresy_pulse.0001 router (events/dlc/pam/heresy/pam_heresy_pulse.txt).
# Ported from the pre-1.20 vanilla heresy triggers (is_valid_heresiarch, invalid_for_heresy_events,
# is_valid_heresy, is_preferred_heresy, no_heretical_hof_faith_trigger) and the inline triggers of the
# old events/religion_events/heresy_events.txt, which vanilla 1.20 removed.
#
# wc_is_not_head_of_FAITH_trigger			- Character is not the religious head of FAITH
# wc_is_valid_heresiarch_trigger			- Character can lead a heresy of root (root = origin faith)
# wc_invalid_for_heresy_events_trigger		- Faith can never be picked as a same-religion heresy
# wc_is_valid_heresy_trigger				- HERETICAL_FAITH is a valid heresy of ORIGIN_FAITH
# wc_is_preferred_heresy_trigger			- HERETICAL_FAITH is a preferred heresy of ORIGIN_FAITH
# wc_nearby_county_of_faith_trigger		- A county near the heresiarch follows this faith (needs the event's list)
# wc_is_pickable_heresy_trigger			- Faith in the candidate list can be picked by wc_heresy.0001 (root = origin faith)
# wc_faith_has_heresy_candidate_trigger	- wc_heresy.0001 would find a heresy for this faith (root = this faith)
# wc_faith_can_start_heresy_trigger		- wc_heresy.0001 would find a heresiarch and a heresy (root = this faith)

# Character scope.
# Was no_heretical_hof_faith_trigger (FAITH = root) and the inline no_heretical_hof_faith_0011_trigger (FAITH = scope:origin_faith).
wc_is_not_head_of_FAITH_trigger = {
	NOT = { $FAITH$.religious_head ?= this }
}

# Character scope. Root must be the origin faith.
wc_is_valid_heresiarch_trigger = {
	is_capable_adult_ai = yes
	is_imprisoned = no

	is_landed = yes
	is_playable_character = yes
	is_landless_adventurer = no
	OR = {
		highest_held_title_tier = tier_duchy
		highest_held_title_tier = tier_county
	}
	primary_title ?= {
		is_landless_type_title = no
		is_noble_family_title = no
	}
	exists = capital_county

	wc_is_not_head_of_FAITH_trigger = { FAITH = root } # Head of Faith shouldn't convert.

	recently_converted_faith_trigger = no # As amusing as it may be, rulers shouldn't 'flip-flop' between Faiths, so we need to enforce a cooldown on heresy conversions.

	faith = root
	NOT = { has_trait = heresiarch }
}

# Faith scope (the candidate heresy).
# Vanilla listed real-world faiths here; GoA's 1.12 compatch replaced the last one with always = yes,
# which makes every faith invalid, so no same-religion faith is ever a heresy candidate.
# Kept as-is to preserve current GoA behaviour (see notes: NEEDS HUMAN REVIEW).
wc_invalid_for_heresy_events_trigger = {
	OR = {
		always = yes # Warcraft
	}
}

# Any scope.

	NOT = { $ORIGIN_FAITH$ = $HERETICAL_FAITH$ }

wc_is_preferred_heresy_trigger = {
	wc_is_valid_heresy_trigger = {

		HERETICAL_FAITH = $HERETICAL_FAITH$

# Faith scope. Needs the neaby_same_religion_counties list built by wc_heresy.0001.
wc_nearby_county_of_faith_trigger = {
	any_in_list = {
		list = neaby_same_religion_counties
		faith = prev
	}
}

# Faith scope (a faith in wc_heresy.0001's potential_heresies list). Root must be the origin faith.
# True when wc_heresy.0001's random_in_list can pick it with a weight above 0:
# never the origin faith itself, and under the strict regional heresy rule only preferred heresies.
wc_is_pickable_heresy_trigger = {
	NOT = { this = root }
	trigger_if = {
		limit = { has_game_rule = strict_regional_heresy }
		wc_is_preferred_heresy_trigger = {
			ORIGIN_FAITH = root
			HERETICAL_FAITH = this
		}
	}
}

# Faith scope (the origin faith). Root must be this faith.
# Mirrors how wc_heresy.0001 builds its potential_heresies list; keep the two in sync.
wc_faith_has_heresy_candidate_trigger = {
	OR = {
		religion = {
			any_faith = {
				wc_invalid_for_heresy_events_trigger = no
				wc_is_pickable_heresy_trigger = yes
			}
		}
		# Deathbringer get bonus ebon blade heresy
		AND = {
			religion = religion:deathbringer
			faith:ebon = { wc_is_pickable_heresy_trigger = yes }
		}
		# Any faith can convert to evil
		AND = {
			NOT = { main_rite = { rite_has_doctrine = doctrine_pluralism_dark } }
			OR = {
				religion:deathbringer = {
					any_faith = { wc_is_pickable_heresy_trigger = yes }
				}
				religion:doombringer = {
					any_faith = { wc_is_pickable_heresy_trigger = yes }
				}
				religion:shathgral_religion = {
					any_faith = { wc_is_pickable_heresy_trigger = yes }
				}
			}
		}
	}
}

# Faith scope (the origin faith). Root must be this faith.
# Cheap pre-check for the heresy_pulse.0001 router: wc_heresy.0001 will find both a heresiarch and a heresy.
wc_faith_can_start_heresy_trigger = {
	any_faith_ruler = {
		wc_is_valid_heresiarch_trigger = yes
	}
	wc_faith_has_heresy_candidate_trigger = yes
}
