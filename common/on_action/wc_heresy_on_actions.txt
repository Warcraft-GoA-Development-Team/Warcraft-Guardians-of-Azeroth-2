# Historical heresy outbreaks: GoA's own faith pulse. It replaces BGA's component 3 entry point
# (heresy_pulse.0001 stays stubbed). Unlike vanilla faith_heresy_events_pulse it also runs for unreformed
# and small faiths: each heresy names its origin faith in wc_heresy_historical_eligible_trigger.

on_faith_monthly = {
	on_actions = { wc_faith_historical_heresy_pulse_delayed }
}

# The delay lives in this wrapper so it never applies to vanilla's on_faith_monthly entries, whatever the merge order.
# 31-35 days: after vanilla's faith_heresy_events_pulse (25-30), whose current_heresy lock we share.
# Depending on merge order this may stack on vanilla's trailing delay (up to ~65 days). That is harmless:
# it still fires once a month, and the trigger is re-checked after the delay.
wc_faith_historical_heresy_pulse_delayed = {
	on_actions = {
		delay = { days = { 31 35 } }
		wc_faith_historical_heresy_pulse
	}
}

wc_faith_historical_heresy_pulse = {
	trigger = {
		fervor <= 40
		NOT = { has_variable = current_heresy }
		wc_heresy_historical_eligible_trigger = yes
	}

	random_events = {
		chance_of_no_event = wc_heresy_pulse_no_event_value
		1 = wc_heresy_historical.0001
	}
}
