scourge_quelthalas_frostmourne_check_trigger = {
    custom_tooltip = {
        text = has_frostmourne
        any_character_artifact = {
            OR = {
                has_variable = frostmourne
            }
        }
    }
}

scourge_quelthalas_phylactery_check_trigger = {
    custom_tooltip = {
        text = has_the_phylactery_of_kelthuzad
        any_character_artifact = {
            OR = {
                has_variable = urn_of_king_of_lordaeron
            }
        }
    }
}

frostmourne_holder_is_loyal_to_lich_king_trigger = {
    trigger_if = {
        limit = { exists = secret_faith }
        OR = {
            secret_faith.religion = title:h_scourge.holder.faith.religion
            secret_faith.religion = religion:doombringer
        }
    }
    trigger_else = {
        OR = {
            faith.religion = title:h_scourge.holder.faith.religion
            faith.religion = religion:doombringer
        }
    }
}
