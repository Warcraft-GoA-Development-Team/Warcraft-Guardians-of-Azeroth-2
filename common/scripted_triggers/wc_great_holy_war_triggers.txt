# Great Holy War triggers for GoA (issue #1973)
# Used by wc_proclaim_holy_war_decision, wc_proclaim_holy_war_swap_effect and wc_holy_war_events.
# Roster: spiritual-head faiths that gain a crusade tenet when their head proclaims.

# Faith scope: roster faiths that gain Armed Pilgrimages
wc_holy_war_armed_pilgrimages_faith_trigger = {
	OR = {
		this = faith:holy_light
		this = faith:cult_of_sunwell
	}
}

# Faith scope: roster faiths that gain Struggle and Submission
wc_holy_war_struggle_faith_trigger = {
	OR = {
		this = faith:zandism
		this = faith:gurubism
		this = faith:drakkism
		this = faith:amism
		this = faith:farrism
		this = faith:rushism
		this = faith:cult_of_sethraliss
		this = faith:spider_philosophy
		this = faith:forgotten_shadow
	}
}

# Faith scope: a roster faith whose crusade tenet is not core yet (Aldor: still holds Pacifism)
wc_faith_can_proclaim_holy_war_trigger = {
	OR = {
		AND = {
			wc_holy_war_armed_pilgrimages_faith_trigger = yes
			main_rite = {
				NOT = {
					has_tenet_status = {
						tenet = tenet_armed_pilgrimages
						status = core
					}
				}
			}
		}
		AND = {
			wc_holy_war_struggle_faith_trigger = yes
			main_rite = {
				NOT = {
					has_tenet_status = {
						tenet = tenet_struggle_submission
						status = core
					}
				}
			}
		}
		AND = {
			this = faith:aldor
			main_rite = {
				has_tenet_status = {
					tenet = tenet_pacifism
					status = core
				}
				has_tenet_status = { # Holy Struggle is what gives the Aldor their holy wars
					tenet = tenet_holy_struggle
					status = core
				}
				NOT = {
					has_tenet_status = {
						tenet = tenet_communion
						status = core
					}
				}
			}
		}
	}
}

# Character scope: the head of a faith that can still proclaim
wc_can_proclaim_holy_war_trigger = {
	this = faith.religious_head

	faith = { wc_faith_can_proclaim_holy_war_trigger = yes }
}

# Tenet scope: tenets that cannot stay core beside Armed Pilgrimages
wc_tenet_is_incompatible_with_armed_pilgrimages_trigger = {
	OR = {
		pam_tenet_is_incompatible_with_armed_pilgrimages_trigger = yes
		this = tenet:tenet_holy_struggle
		this = tenet:tenet_united_against_darkness
	}
}

# Tenet scope: tenets that cannot stay core beside Struggle and Submission
wc_tenet_is_incompatible_with_struggle_submission_trigger = {
	OR = {
		pam_tenet_is_incompatible_with_armed_pilgrimages_trigger = yes # Peace of God and both pacifisms forbid holy war
		this = tenet:tenet_megaliths
	}
}

# Tenet scope: the core tenet may be given up by FAITH's proclamation
# FAITH = the proclaiming faith (scope:actor.faith inside the decision widget)
wc_tenet_can_be_replaced_for_holy_war_trigger = {
	trigger_if = {
		limit = { $FAITH$ = faith:aldor }
		custom_tooltip = {
			text = wc_proclaim_holy_war_aldor_tt
			this = tenet:tenet_pacifism
		}
	}
	trigger_else = {
		trigger_if = {
			limit = { $FAITH$ = faith:holy_light }
			custom_tooltip = {
				text = wc_proclaim_holy_war_communion_tt
				NOT = { this = tenet:tenet_communion }
			}
		}
		trigger_if = {
			limit = {
				$FAITH$ = { wc_holy_war_armed_pilgrimages_faith_trigger = yes }
				$FAITH$.main_rite = {
					any_rite_tenet = {
						status = core
						wc_tenet_is_incompatible_with_armed_pilgrimages_trigger = yes
					}
				}
			}
			custom_tooltip = {
				text = wc_proclaim_holy_war_incompatible_tt
				wc_tenet_is_incompatible_with_armed_pilgrimages_trigger = yes
			}
		}
		trigger_else_if = {
			limit = {
				$FAITH$ = { wc_holy_war_struggle_faith_trigger = yes }
				$FAITH$.main_rite = {
					any_rite_tenet = {
						status = core
						wc_tenet_is_incompatible_with_struggle_submission_trigger = yes
					}
				}
			}
			custom_tooltip = {
				text = wc_proclaim_holy_war_incompatible_struggle_tt
				wc_tenet_is_incompatible_with_struggle_submission_trigger = yes
			}
		}
		trigger_else = { # Engine requires an else after trigger_else_if; no conflict means any tenet may go
			always = yes
		}
	}
}

# Tenet scope: the tenet FAITH's AI head gives up when it can
wc_tenet_is_holy_war_ai_drop_trigger = {
	OR = {
		AND = {
			$FAITH$ = faith:holy_light
			this = tenet:tenet_dulia
		}
		AND = {
			$FAITH$ = {
				OR = {
					this = faith:zandism
					this = faith:gurubism
					this = faith:drakkism
					this = faith:amism
					this = faith:farrism
					this = faith:rushism
				}
			}
			OR = {
				this = tenet:tenet_bloody_feast
				this = tenet:tenet_adaptive
			}
		}
		AND = {
			$FAITH$ = faith:cult_of_sethraliss
			this = tenet:tenet_adaptive
		}
		AND = {
			$FAITH$ = faith:spider_philosophy
			this = tenet:tenet_literalism
		}
		AND = {
			$FAITH$ = faith:forgotten_shadow
			this = tenet:tenet_communal_identity
		}
		AND = {
			$FAITH$ = faith:cult_of_sunwell
			this = tenet:tenet_astrology
		}
		AND = {
			$FAITH$ = faith:aldor
			this = tenet:tenet_pacifism
		}
	}
}
