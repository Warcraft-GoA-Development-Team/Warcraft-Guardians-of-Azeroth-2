#### To Keep DNAs intact
on_game_start = { on_actions = { unprune_characters } }

unprune_characters = {
	effect = {
		character:60572 = { make_unprunable = yes } ##Whitemane
	}
}

#### Triggers Scarlet Setup - survivors of lordaeron plague get an option to link up
scarlet_setup = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = scarlet_event_chain_started } }
				every_ruler = { limit = {
				geographical_region = world_eastern_kingdoms_lordaeron
				NOT = { has_trait = being_undead }
				has_trait = creature_human
				has_religion = religion:tyrism_religion
				}
			}
		}
		trigger_event = scarlet.1000 
		set_global_variable = scarlet_event_chain_started
	}
}
