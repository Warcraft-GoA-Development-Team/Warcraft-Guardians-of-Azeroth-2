		character = root

	}

		is_vulnerable_to_plague_character_trigger = yes

			has_character_modifier = important_lore_character

		# For graphics
		hidden_effect = { remove_trait = ill }
	}
}

# World Weary - Vanish without a trace
WCHEA.9000 = { 
	type = character_event
	content_source = dlc_GOA
	title = WCHEA.9000.TITLE
	theme = death

	left_portrait = { 
		character = root
		animation = depressed
	}

	desc = { 
		first_valid = { 
			triggered_desc = { # TODO: MAGIC -> add a way for these beings to transcend the physical plane
				trigger = { 
					has_trait = being_order 	has_trait = being_light
					has_trait = being_life		has_trait = being_void
				}
				desc = WCHEA.9000.DESC.other_being
			}
			triggered_desc = { 
				trigger = { has_trait = being_undead }
				desc = WCHEA.9000.DESC.undead
			}
			desc = WCHEA.9000.DESC
		}
	}

	trigger = { 
		is_incapable = no
		AND = { 
			elder_culture_undead_suicide_trigger = yes
			demon_suicide_trigger = no
		}
	}

	immediate = { 
		player_heir = {
			save_scope_as = heir
		}
	}

	# Do it
	option = { 
		name = WCHEA.9000.OPT.A
		random_list = {  
			40 = { 
				modifier = { 
					add = learning
				}
				modifier = { 
					has_trait = reclusive
					add = 10
				}
				max = 95
				elder_culture_undead_suicide_effect = yes
			}
			60 = { 
				modifier = { 
					add = { 
						value = intrigue
						multiply = -1
					}
				}
				min = 5
				send_interface_toast = { 
					type = event_toast_effect_neutral
					text = world_weary_fail_elder_undead_tt
					left_icon = scope:root
					add_prestige = medium_prestige_loss
				}
			}
		}
	}

	# Back out
	option = { 
		name = WCHEA.9000.OPT.B
	}
}

# World Weary - Retreat into the Twisting Nether 
WCHEA.9001 = { 
	type = character_event
	content_source = dlc_GOA
	title = WCHEA.9001.TITLE
	desc = WCHEA.9001.DESC
	theme = battle

	left_portrait = { 
		character = root
		animation = celebrate_axe
	}

	trigger = { 
		is_incapable = no
		demon_suicide_trigger = yes
	}

	immediate = { 
		player_heir = {
			save_scope_as = heir
		}
	}

	# Do it
	option = { 
		name = WCHEA.9000.OPT.A
		random = { 
			chance = { 
				value = learning
				add = { 
					value = intrigue
					multiply = 0.5
				}
			}
			save_scope_value_as = {
				name = self_initiated
				value = yes
			}
			if = {
				limit = {
					any_vassal = {
						count >= 1
					}
				}
				every_vassal = {
					custom = all_vassals
					add_opinion = {
						modifier = suicide_opinion
						target = root
						opinion = -15
					}
				}
			}
			trigger_event = WCBEI.100
		}
	}

	# Back out
	option = { 
		name = WCHEA.9000.OPT.B
	}

}
