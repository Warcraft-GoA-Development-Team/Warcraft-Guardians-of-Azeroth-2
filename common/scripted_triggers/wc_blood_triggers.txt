is_blood_no_gene_trigger = {
	trigger_if = {
		limit = { has_blood_trait_trigger = yes }
		has_trait = $BLOOD$
	}
	trigger_else = {
		trigger_if = {
			limit = {
				NOT = { exists = scope:culture }
			}
			trigger_if = {
				limit = { exists = culture }
				culture = { save_temporary_scope_as = culture }
			}
			trigger_else = {
				save_temporary_scope_as = culture
			}
		}
		
		trigger_if = {
			limit = { exists = scope:culture }
			scope:culture = { has_clothing_gfx = $BLOOD$_gfx }
		}
		trigger_else = {
			always = no
		}
	}
}

has_blood_trait_trigger = {
	has_trait = blood
}
