namespace = wc_titanforged_restoration

wc_titanforged_restoration.0001 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_titanforged_restoration.0001.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					wc_titanforged_implantation_audience_trigger = yes
					scope:county = { var:wc_titanforged_settlement_result = flag:coexistence }
				}
				desc = wc_titanforged_restoration.0001.upgrade
			}
			triggered_desc = {
				trigger = {
					wc_titanforged_implantation_audience_trigger = yes
					scope:county = { has_variable = wc_titanforged_installations }
				}
				desc = wc_titanforged_restoration.0001.resume
			}
			desc = wc_titanforged_restoration.0001.desc
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:restoration
					wc_titanforged_historical_audience_trigger = yes
					scope:restoration.var:wc_titanforged_restoration_marshal ?= { is_alive = yes has_trait = compassionate }
				}
				desc = wc_titanforged_restoration.0001.marshal_pact
			}
			triggered_desc = {
				trigger = {
					exists = scope:restoration
					wc_titanforged_historical_audience_trigger = yes
					scope:restoration.var:wc_titanforged_restoration_marshal ?= { is_alive = yes has_trait = zealous }
				}
				desc = wc_titanforged_restoration.0001.marshal_mandate
			}
		}
	}
	theme = realm
	left_portrait = { character = root animation = emotion_thinking_scepter }

	option = {
		name = wc_titanforged_restoration.0001.pact
		trigger = { wc_titanforged_historical_audience_trigger = yes }
		custom_tooltip = wc_titanforged_restoration.pact_tt
		ai_chance = {
			base = 50
			modifier = { factor = 3 has_trait = compassionate }
			modifier = { factor = 2 has_trait = just }
		}
		if = {
			limit = { wc_titanforged_historical_audience_trigger = yes }
			scope:restoration = { set_variable = { name = wc_titanforged_restoration_method value = flag:pact } }
			if = {
				limit = { scope:councillor = { has_trait = compassionate } }
				scope:councillor = { add_opinion = { target = root modifier = wc_titanforged_method_approved_opinion } }
			}
			else_if = {
				limit = { scope:councillor = { has_trait = zealous } }
				scope:councillor = { add_opinion = { target = root modifier = wc_titanforged_method_rejected_opinion } }
			}
		}
		else = { wc_titanforged_restoration_cancel_audience_effect = yes }
	}
	option = {
		name = wc_titanforged_restoration.0001.mandate
		trigger = { wc_titanforged_historical_audience_trigger = yes }
		custom_tooltip = wc_titanforged_restoration.mandate_tt
		ai_chance = {
			base = 50
			modifier = { factor = 3 has_trait = zealous }
			modifier = { factor = 2 has_trait = callous }
		}
		if = {
			limit = { wc_titanforged_historical_audience_trigger = yes }
			scope:restoration = { set_variable = { name = wc_titanforged_restoration_method value = flag:mandate } }
			if = {
				limit = { scope:councillor = { has_trait = compassionate } }
				scope:councillor = { add_opinion = { target = root modifier = wc_titanforged_method_rejected_opinion } }
			}
			else_if = {
				limit = { scope:councillor = { has_trait = zealous } }
				scope:councillor = { add_opinion = { target = root modifier = wc_titanforged_method_approved_opinion } }
			}
		}
		else = { wc_titanforged_restoration_cancel_audience_effect = yes }
	}
	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { scope:county = { var:wc_titanforged_settlement_result = flag:coexistence } }
						desc = wc_titanforged_restoration.0001.implantation_upgrade
					}
					triggered_desc = {
						trigger = { scope:county = { has_variable = wc_titanforged_installations } }
						desc = wc_titanforged_restoration.0001.implantation_resume
					}
					desc = wc_titanforged_restoration.0001.implantation
				}
			}
		}
		trigger = { wc_titanforged_implantation_audience_trigger = yes }
		custom_tooltip = wc_titanforged_restoration.implantation_tt
		if = {
			limit = { scope:county = { NOT = { has_variable = wc_titanforged_installations } } }
			custom_tooltip = wc_titanforged_restoration.installation_cost_tt
		}
		if = {
			limit = { scope:county = { var:wc_titanforged_settlement_result = flag:coexistence } }
			custom_tooltip = wc_titanforged_restoration.upgrade_tt
			custom_tooltip = wc_titanforged_forge.dominant_tt
		}
		else_if = {
			limit = { scope:county = { has_variable = wc_titanforged_settlement_cycle_active var:wc_titanforged_institutions = flag:coexistence } }
			custom_tooltip = wc_titanforged_restoration.institutions_fixed_tt
			custom_tooltip = wc_titanforged_forge.coexist_tt
		}
		else_if = {
			limit = { scope:county = { has_variable = wc_titanforged_settlement_cycle_active var:wc_titanforged_institutions = flag:dominant } }
			custom_tooltip = wc_titanforged_restoration.institutions_fixed_tt
			custom_tooltip = wc_titanforged_forge.dominant_tt
		}
		else = {
			custom_tooltip = wc_titanforged_restoration.institution_choice_tt
		}
		ai_chance = {
			base = 50
			modifier = { factor = 3 has_trait = callous }
			modifier = { factor = 2 has_trait = zealous }
			modifier = { factor = 0.25 has_trait = compassionate }
		}
		if = {
			limit = { wc_titanforged_implantation_audience_trigger = yes }
			scope:restoration = { set_variable = { name = wc_titanforged_restoration_method value = flag:implantation } }
			scope:county = { set_variable = { name = wc_titanforged_reorder_implantation value = yes } wc_titanforged_settlement_resume_effect = yes }
		}
		else = { wc_titanforged_restoration_cancel_audience_effect = yes }
	}
	option = {
		name = wc_titanforged_forge.arcane
		trigger = { wc_titanforged_arcane_audience_trigger = yes }
		custom_tooltip = wc_titanforged_forge.arcane_tt
		ai_chance = { base = 25 }
		if = {
			limit = { wc_titanforged_arcane_audience_trigger = yes }
			remove_short_term_gold = 100
			scope:restoration = { set_variable = { name = wc_titanforged_restoration_method value = flag:arcane } }
		}
		else = { wc_titanforged_restoration_cancel_audience_effect = yes }
	}
	option = {
		name = wc_titanforged_restoration.0001.legacy
		trigger = { wc_titanforged_legacy_audience_trigger = yes }
		custom_tooltip = wc_titanforged_restoration.legacy_task_tt
		ai_chance = { base = 50 }
		if = {
			limit = { wc_titanforged_legacy_audience_trigger = yes }
			scope:restoration = { set_variable = { name = wc_titanforged_restoration_method value = flag:legacy } }
		}
		else = { wc_titanforged_restoration_cancel_audience_effect = yes }
	}
	option = {
		name = wc_titanforged_restoration.0001.cancel
		trigger = { wc_titanforged_restoration_audience_valid_trigger = yes }
		custom_tooltip = wc_titanforged_restoration.cancel_tt
		ai_chance = { base = 0 }
		wc_titanforged_restoration_cancel_audience_effect = yes
	}
	option = {
		name = wc_titanforged_restoration.wait
		trigger = { wc_titanforged_restoration_audience_valid_trigger = no }
		wc_titanforged_restoration_cancel_audience_effect = yes
	}
}
