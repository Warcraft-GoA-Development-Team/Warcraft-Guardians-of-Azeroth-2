# Setup - 0001-0099

# Hidden setup - creates the story on root (root = player-controlled Balnazzar).
# create_story needs the block form (save_scope_as inside it) to actually capture the story -
# a bare create_story followed by a separate save_scope_as just re-saves whatever scope was
# already current, so global_var:balnazzar_dreadlord_story never got set at all before this.
wc_balnazzar_story.0001 = {
	hidden = yes

	immediate = {
		create_story = {
			type = story_balnazzar_dreadlord
			save_scope_as = story_owner
		}
		set_global_variable = {
			name = balnazzar_dreadlord_story
			value = scope:story_owner
		}

		trigger_event = {
			id = wc_balnazzar_story.1000
			days = { 2 14 }
		}
	}
}

# The Old Kingdom Slips - loses Lordaeron, goes to ground in the Plaguelands, no target yet,
# just an exit plan. Fires from on_title_lost only when c_lordaeron is actually lost. First of
# the .1001 -> .1010 -> .1009 -> .1002 expedition chain. Used to be a real activity_type - worked,
# but needed a manual click for a one-shot beat that didn't warrant the overhead, so back to a
# plain event chain.
wc_balnazzar_story.1001 = {
	type = character_event
	title = wc_balnazzar_story.1001.title
	desc = wc_balnazzar_story.1001.desc
	theme = intrigue

	trigger = {
		NOT = { has_trait = nathrezim_disguise }
	}

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}

	option = {
		name = wc_balnazzar_story.1001.a

		custom_tooltip = wc_balnazzar_story.1001.a_tooltip
		hidden_effect = {
			# change_government to landless_adventurer_government alone silently fails - that
			# government needs a landless-type duchy title, which create_landless_adventurer_title_effect
			# mints but doesn't give up any titles still held. Hand everything to his liege first
			# (same title-transfer pattern as vanilla's become_landless_adventurer_decision), so he
			# genuinely walks out with nothing, matching .1010's framing.
			if = {
				limit = { top_liege != this }
				liege = { save_scope_as = balnazzar_land_recipient }

				every_held_title = {
					title_tier > barony
					limit = {
						OR = {
							is_landless_type_title = no
							is_nomad_title = yes
						}
					}
					add_to_list = balnazzar_relinquished_titles
				}

				create_title_and_vassal_change = {
					type = conquest
					save_scope_as = balnazzar_land_change
					add_claim_on_loss = no
				}
				every_in_list = {
					list = balnazzar_relinquished_titles
					change_title_holder_include_vassals = {
						holder = scope:balnazzar_land_recipient
						change = scope:balnazzar_land_change
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_land_change
			}

			create_landless_adventurer_title_effect = {
				REASON = flag:voluntary
				FLAVOR_CHAR = root
			}

			trigger_event = {
				id = wc_balnazzar_story.1010
				days = { 5 15 }
			}
		}
	}
}

	title = wc_balnazzar_story.1002.title

	left_portrait = {
		character = root

	}
	right_portrait = {
		character = character:60021 # Saidan Dathrohan
		animation = paranoia
	}

	immediate = {
		character:60021 = { save_scope_as = dathrohan }
	}

	# .1002.b: fallback if Dathrohan is dead/unplayable - avoids a soft-lock.

		name = wc_balnazzar_story.1002.a

				is_alive = yes
				is_playable_character = yes

			death = { death_reason = death_vanished }

			if = {
				limit = { has_character_flag = balnazzar_dathrohan_alerted }
				scope:dathrohan = { add_trait = wounded_1 }
				remove_character_flag = balnazzar_dathrohan_alerted

				add_trait = nathrezim_disguise

# The Mask Comes Off - shared resolution for all "reject the mask" triggers (old age, exposure, paranoia).
# Disguise ends, Crusade burns, the Risen rise at Stratholme.
wc_balnazzar_story.2000 = {
	type = character_event
	title = wc_balnazzar_story.2000.title
	desc = wc_balnazzar_story.2000.desc
	theme = death

	# Self-guard against double-firing - the flag is only set once .2000
	# actually runs, not when it's queued, so two callers could race.
	trigger = {
		NOT = { has_character_flag = balnazzar_shed_the_mask }
	}

	override_background = wc_stratholme

	left_portrait = {
		character = root
		animation = rage
	}

	immediate = {
		add_character_flag = balnazzar_shed_the_mask
		remove_trait = nathrezim_disguise

		if = {
			limit = { has_variable_list = balnazzar_known_rivals }
			custom_tooltip = wc_balnazzar_story.2000.rivals_react_tooltip
			every_in_list = {
				variable = balnazzar_known_rivals
				limit = { is_alive = yes }
				add_opinion = { modifier = wc_suspects_scheming_against_me target = root }
				add_prestige = minor_prestige_value
			}
		}
	}

	option = {
		name = wc_balnazzar_story.2000.a

		custom_tooltip = wc_balnazzar_story.2000.a_tooltip
		hidden_effect = {
			# Courtiers convert if physically in Hasic - excludes guests/
			# hostages who happen to be present but aren't his household.
			every_courtier = {
				limit = {
					is_alive = yes
					NOT = { has_trait = being_undead }
					location.county = root.capital_county
				}
				raise_undead_faith_only_effect = { MASTER = root }
				add_opinion = { modifier = loyal_undead target = root }
			}

			# Resurrect Balnazzar's nathrezim body (character:430400, "died" at .1002) - one-shot, purges source.
			# Falls through to the shell surviving if unavailable, instead of an unrecoverable Game Over.
			if = {
				limit = {
					exists = character:430400
					is_landed = yes
					exists = capital_county
				}

				wc_resurrect_CHARACTER_with_STATUS_effect = {
					CHARACTER = character:430400
					STATUS = alive
				}

				# Capital county + every held duchy break away with Balnazzar.
				# type = granted, not "independence" (invalid, fails "No type set" at read-time).
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = balnazzar_reveal_title_change
					add_claim_on_loss = no
				}
				root.capital_county = {
					change_title_holder_include_vassals = {
						holder = scope:resurrected_character
						change = scope:balnazzar_reveal_title_change
					}
				}
				every_held_title = {
					limit = { tier = tier_duchy }
					save_scope_as = balnazzar_duchy_grant
					change_title_holder_include_vassals = {
						holder = scope:resurrected_character
						change = scope:balnazzar_reveal_title_change
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_reveal_title_change

				# swear_fealty's tier check compares primary_title, not highest held title.
				# Without this, Balnazzar stays count-tier and vassalizing fails "same or higher tier".
				if = {
					limit = { exists = scope:balnazzar_duchy_grant }
					scope:resurrected_character = {
						set_primary_title_to = scope:balnazzar_duchy_grant
					}
				}

				# Grant leaves him a vassal of the de jure liege - sever explicitly.
				# type = independency, not "independence" (invalid value).
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = balnazzar_independence_change
				}
				scope:resurrected_character = {
					change_liege_or_become_independent = {
						CHANGE = scope:balnazzar_independence_change
						VASSAL = this
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_independence_change

				# include_vassals above already re-vassalized everyone in the duchy - just convert them.
				# Army composition mirrors this mod's Scourge pattern, not .1006's living Crusade.
				# _no_notification variant - this can be a whole duchy's worth of vassals at
				# once, and become_undead_effect's notification leaks memory per call.
				scope:resurrected_character = {
					every_vassal_or_below = {
						limit = {
							is_alive = yes
							NOT = { has_trait = being_undead }
						}
						raise_undead_faith_only_no_notification_effect = { MASTER = liege }
						add_opinion = { modifier = loyal_undead target = liege }
					}
					spawn_army = {
						uses_supply = no
						inheritable = yes
						name = the_risen
						men_at_arms = { type = ghoul_maa stacks = 8 }
						men_at_arms = { type = deathcaller stacks = 4 }
						men_at_arms = { type = banshee_maa stacks = 4 }
						men_at_arms = { type = frost_wyrm stacks = 1 }
						location = capital_province
						origin = capital_province
					}
				}

				# Move the player off root before killing it - killing the
				# still-played root aborts the rest of the effect otherwise.
				set_player_character = scope:resurrected_character

				# Fires on_owner_death (story_balnazzar_dreadlord), ending
				# the story - no separate end_story call needed.
				death = { death_reason = death_vanished }
			}
			else = {
				# No valid handoff body - root survives, exposed but alive,
				# rather than risk an unrecoverable Game Over.
				global_var:balnazzar_dreadlord_story = {
					end_story = yes
				}
			}
		}
	}
}
