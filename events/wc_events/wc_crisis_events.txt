namespace = wc_crisis_events

# Join Wars in a coalition
wc_crisis_events.0010 = {
    type = character_event
	content_source = dlc_GOA
	hidden = yes

    trigger = {
        is_in_any_coaltion_trigger = yes
        is_ai = yes 
    }

    immediate = {
        save_scope_as = member
        
        every_in_list = {
            variable = coalition_targets
            save_scope_as = war_target
            random_character_war = {
                limit = {
                    OR = {
                        primary_attacker = scope:war_target
                        primary_attacker = scope:war_target
                    }
                    NOT = { is_participant = scope:member }
                }
                save_scope_as = war
                if = {
                    limit = {
                        primary_attacker = scope:war_target
                    }
                    primary_defender = { save_scope_as = war_partner }
                }
                else = {
                    primary_attacker = { save_scope_as = war_partner }
                }
                random_list = {
                    1000 = {}
                    250 = {
                        modifier = {
                            always = yes
                            scope:member = { join_war_against_coalition_scripted_modifier = yes }
                        }

                        if = {
                            limit = {
                                primary_attacker = scope:war_target
                            }
                            set_called_to = scope:member
                            add_defender = scope:member
                            debug_log = "A coalition member has joined a def war"
                            debug_log_date = yes
                        }
                        else = {
                            set_called_to = scope:member
                            add_attacker = scope:member
                            debug_log = "A coalition member has joined a atk war"
                            debug_log_date = yes
                        }
                    }
                }
            }
        }
    }
}

# The coalition against you is large
wc_crisis_events.0200 = {
    type = character_event
	content_source = dlc_GOA
    theme = alliance
    title = wc_crisis_events.0200.title
    desc = wc_crisis_events.0200.desc

    trigger = {
        is_crisis_trigger = yes 
        any_in_list = {
            variable = coalition_members
            count >= 10
        }
        NOT = { has_character_flag = wc_crisis_events_200 }
    }

    left_portrait = {
		character = root
		animation = personality_honorable
	}
    
    option = {
        name = wc_crisis_events.0200.a
        add_dread = minor_dread_gain
        add_character_modifier = {
            modifier = crisis_bold_modifier
            years = 5
        }

        if = {
            limit = {
                NOT = { has_trait = aggressive_attacker }
            }
            random = {
                chance = 10
                add_trait = aggressive_attacker
            }
        }

        stress_impact = {
            craven = minor_stress_impact_gain
            brave = minor_stress_impact_loss
        }

        ai_chance = {
            base = 100
            ai_value_modifier = {
                ai_boldness = 1
                ai_energy = 1
            }
            modifier = {
                factor = 0.5
                has_trait = craven
            }
        }
    }

    option = {
        name = wc_crisis_events.0200.b

        custom_tooltip = wc_crisis_events.0200.b_tooltip

        add_character_modifier = {
            modifier = crisis_scheme_modifier
            years = 5
        }

        ai_chance = {
            base = 100
            ai_value_modifier = {
                ai_honor = -1
            }
            modifier = {
                factor = 0
                is_incapable_trigger = yes #Cant use schemes
            }
        }
    }

    option = {
        name = wc_crisis_events.0200.c

        add_character_modifier = {
            modifier = crisis_defense_modifier
            years = 5
        }

        stress_impact = {
            craven = minor_stress_impact_loss
        }

        if = {
            limit = {
                NOT = { has_trait = unyielding_defender }
            }
            random = {
                chance = 10
                add_trait = unyielding_defender
            }
        }

        ai_chance = {
            base = 100
            modifier = {
                factor = 1.5
                has_trait = craven
            }
            modifier = {
                factor = 1.5
                coalition_strength_value > max_military_strength
            }
        }
    }
    option = {
        name = wc_crisis_events.0200.d
        trigger = {
            any_character_war = {
                count >= 1
            }
        }
        ordered_character_war = {
            if = {
                limit = {
                    primary_attacker = root
                }
                end_war = defender
            }
            else = {
                end_war = attacker
            }
        }
        add_character_modifier = {
            modifier = crisis_surrender_modifier
            years = 10
        }
        stress_impact = {
            base = massive_stress_impact_loss
            arrogant = medium_stress_impact_gain
            wrathful = medium_stress_impact_gain
        }
        every_vassal_or_below = {
            limit = { has_vassal_stance = glory_hound }
            custom = every_glory_hound_vassal
            add_opinion = {
                modifier = disapproving_opinion
                target = root
                opinion = -10
            }
        }
        ai_chance = {
            base = 0
        }
    }
    after = {
        add_character_flag = wc_crisis_events_200
    }
}
