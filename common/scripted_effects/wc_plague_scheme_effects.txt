# Plague Scheme Effects

start_plague_scheme_effect = {
	start_scheme = {
		type = plague
		target_character = $TARGET$
	}

	random_scheme = {
		limit = {
			scheme_type = plague
			scheme_target_character = $TARGET$
		}
		set_variable = {
			name = plague_center
			value = $PLAGUE_CENTER$
		}
	}
}

plague_scheme_prep_effect = {
	save_scope_as = scheme
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:wc_scheme_critical_moments.0001
	}
	if = {
		limit = {
			NOT = { exists = scope:suppress_next_event }
		}
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

trigger_event_plague_outcome_0002_effect = {
	$OWNER$ = {
		save_scope_as = owner
	}
	
	$TARGET$ = {
		save_scope_as = target
	}

	$PLAGUE_CENTER$ = {
		save_scope_as = plague_center
	}
	
	trigger_event = plague_outcome.0002
}

# Restart's the current plague scheme, setting all its variables and such back to zero.
restart_plague_scheme_effect = {
	show_as_tooltip = {
		start_scheme = {
			type = plague
			target_character = scope:target
		}
	}
	hidden_effect = {
		scope:scheme = {
			# Track how many times we've restarted, so we can reset progress.
			increment_variable_effect = {
				VAR = scheme_reset_to_zero_value
				VAL = scheme_success_chance
			}
			# Reset our scheme.
			reset_failed_scheme_effect = yes
		}
	}
}

# Regular plague effects
known_or_unknown_plague_effect = {
	scope:owner = {
		if = {
			limit = { scope:plagued_county = title:c_wintergarde }
			set_variable = { name = wintergarde_plague_fired value = yes }
		}
		else_if = {
			limit = { 
				scope:plagued_county = title:c_andorhal
				NOT = { has_global_variable = scourge_of_lordaeron_has_happened }
			}
			set_global_variable = scourge_of_lordaeron
			
			# Arthas can start his activity
			add_to_global_variable_list = {
				name = unavailable_unique_events
				target = flag:arthas_activity_good_to_go
			}
		}
	}
	
	if = {
		limit = { exists = scope:scheme_discovered }
		known_plague_effect = {
			PLACE = scope:plagued_county
			PLAGUER = scope:owner
			EXPOSER = scope:exposer
			TARGET = scope:target
		}
	}
	else = {
		unknown_plague_effect = {
			PLACE = scope:plagued_county
			PLAGUER = scope:owner
			TARGET = scope:target
		}
	}
}

# Apply scope:owner's general ending & consequences.
scheme_owner_plague_success_effect = {
	if = {
		limit = { this = scope:owner }
		scope:owner = {
			if = {
				limit = { faith = { has_doctrine_parameter = piety_gain_from_successful_intrigue_schemes } }
				add_piety = minor_piety_gain
			}
		}
	}

	# Stress
	stress_impact = {
		base = minor_stress_impact_loss
		arbitrary = minor_stress_impact_loss
		compassionate = medium_stress_impact_gain
		gregarious = minor_stress_impact_gain
	}

	# Show other consequences.
	scope:target = {
		show_as_tooltip = { plague_success_effect = yes }
	}
}

# Actually plague scope:target and end the scheme
plague_success_effect = {
	scope:plague_center = { save_scope_as = plagued_county }
	
	if = {
		limit = { exists = scope:scheme_discovered }
		scope:owner = { save_scope_as = owner_to_reveal }
	}	
	if = {
		limit = { this = scope:owner }
		scope:owner = {
			known_or_unknown_plague_effect = yes
			
			#Stress reduction
			if = {
				limit = { has_trait = sadistic }
				stress_impact = {
					sadistic = major_stress_impact_loss
				}
			}
			
			if = {
				limit = { exists = scope:scheme }
				scope:scheme = { end_scheme = yes }
			}
		}
	}
}

plague_success_victim_immediate_effect = {
	if = {
		limit = {
			exists = scope:target
			this = scope:target
		}
		add_character_modifier = { modifier = plague_watchful_modifier days = watchful_modifier_duration }
	}
	if = {
		limit = {
			exists = scope:scheme_discovered
			NOT = { scope:owner.faith = this.faith }
		}
		plague_consequences_known_or_exposed_effect = {
			PLAGUER = scope:owner
			VICTIM = this
		}
	}
	if = {
		limit = { this = root }
		show_as_tooltip = {
			known_or_unknown_plague_effect = yes
		}
	}
}

plague_failure_effect = {

		}

	# If the attempt was discovered, process the consequences of that.
	if = {
		limit = { exists = scope:scheme_discovered }
