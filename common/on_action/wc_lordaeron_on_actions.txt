on_death = {
	on_actions = {
		wc_death_on_actions
		wc_bandarion_keep_death_on_actions
	}
}

wc_death_on_actions = {
	effect = {
		if = {
			limit = { exists = scope:killer }
			hurt_scourge_jailor_consequences_effect = { HURT_JAILOR = root SCOURGE = scope:killer }
			if = {
				limit = {
					scope:killer = { is_child_of = root }
					this = { is_terenas_trigger = yes }
					OR = {
						# making sure arthas is actually evil arthas and didnt kill his father for no reason
						scope:killer = {
							is_ai = yes
							current_date >= 603.5.1
						}
						scope:killer = { 
							is_ai = no
							faith = faith:death_god
							culture = culture:scourge
						}
					}
				}
				save_scope_as = terenas
				# Uther Stress Event
				title:k_order_of_the_silver_hand.holder = {
					save_scope_as = uther
					scope:killer = { save_scope_as = arthas }
					trigger_event = { 
						id = order_of_the_silver_hand_events.001
						days = 3
					}
				}
				# Sally Whitemane Conversion
				character:60572 = {
					set_character_faith_with_conversion = faith:scarletism
				}

				# Terenas Funeral
				# Compile list of people who should come.
				every_close_or_extended_family_member = { add_to_temporary_list = funeral_attendees }
				every_vassal_or_below = { add_to_temporary_list = funeral_attendees }
				every_courtier = { add_to_temporary_list = funeral_attendees }
				character:60036 = { add_to_temporary_list = funeral_attendees } # Uther
				every_relation = { type = lover add_to_temporary_list = funeral_attendees }
				every_relation = { type = friend add_to_temporary_list = funeral_attendees }

				every_in_list = {
					list = funeral_attendees
					# have to save these now because it wont work after he dies
					if = {
						limit = { # Terenas had a good opinion of them or vv
							is_alive = yes 
							OR = {
								opinion = {
									value > 0
									target = scope:terenas 
								}
								scope:terenas = {
									opinion = {
										value > 0
										target = prev
									}
								}
							}
						}
						add_character_flag = {
							flag = terenas_good
							days = 30
						}
					}

					trigger_event = {
						id = lordaeron.2001
						days = 6
					}
				}
			}
		}
	}
}

# Garek Bandarion dies holding his barony: name it Bandarion Keep and,
# if Danitha Morr can lead it, found Tyr's Watch there (forsaken_kingdom.0001, shown
# to the liege with Garek's portrait via scope:garek). Any death counts, including
# the scripted Naxxramas sacrifice in forsaken_kingdom.0003.
# on_death runs before titles pass on, so has_title and liege still describe Garek.
# The flag is set at game start only from 605.5.1, so a 603-era human Garek triggers nothing.
wc_bandarion_keep_death_on_actions = {
	trigger = {
		has_title = title:b_autumnacre

		has_character_flag = wc_captain_of_the_deathguard
		NOT = { has_global_variable = wc_bandarion_keep_named }
	}
	effect = {
		save_scope_as = garek

		set_global_variable = wc_bandarion_keep_named

		# One day lets the barony finish passing before the event reads it
		liege ?= {
			if = {
				limit = { this != root }
				trigger_event = {
					id = forsaken_kingdom.0001
					days = 1
				}
			}
		}
	}
}

on_game_start = {
	on_actions = {
		wc_forsaken_kingdom_game_start_on_actions
	}
}

# Marks the Forsaken Kingdom characters at game start: Garek Bandarion for
# wc_bandarion_keep_death_on_actions and wc_naxxramas_sacrifice_on_action; Danitha Morr
# (future leader of Tyr's Watch) and Evin Schmidtt (future keeper of Bandarion Keep) for
# forsaken_kingdom.0001. Set here rather than in their 605.5.1 history so CK3's
# load-time check sees a setter (it otherwise logs "used but is never set").
# Only on a start on or after 605.5.1, when all three have joined the Forsaken.
wc_forsaken_kingdom_game_start_on_actions = {
	trigger = { current_date >= 605.5.1 }
	effect = {
		character:garek_bandarion ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = wc_captain_of_the_deathguard
			}
		}
		character:danitha_morr ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = wc_awaits_the_light
			}
		}
		character:evin_schmidtt ?= {
			if = {
				limit = { is_alive = yes }
				add_character_flag = wc_awaits_the_light
			}
		}
	}
}

# Garek Bandarion's Naxxramas sacrifice (forsaken_kingdom.0003). Listed in
# wc_yearly_global_pulse, so checked every 1 January from 607; fires on 1 April, the date
# in his history. Needs Garek alive and still holding Autumnacre, so the keep story can
# follow. Otherwise nothing fires and he dies whenever the engine decides.
wc_naxxramas_sacrifice_on_action = {
	trigger = {
		current_date >= 607.1.1
		character:garek_bandarion ?= {
			is_alive = yes

			has_title = title:b_autumnacre
			exists = liege

			has_character_flag = wc_captain_of_the_deathguard
		}
	}
	effect = {
		character:garek_bandarion = {
			save_scope_as = garek
			liege = {
				trigger_event = {
					id = forsaken_kingdom.0003
					days = 90
				}
			}
		}
	}
}
