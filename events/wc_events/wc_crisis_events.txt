# Join Wars in a coalition
wc_crisis_events.0010 = {
    type = character_event
	content_source = dlc_GOA
	hidden = yes

    trigger = {
        is_in_any_coalition_trigger = yes
        is_ai = yes 
    }

    immediate = {
        save_scope_as = member
        
        every_in_list = {
            variable = coalition_targets
            save_scope_as = war_target
            random_character_war = {
                limit = {
                    OR = {
                        primary_attacker = scope:war_target
                        primary_attacker = scope:war_target
                    }
                    NOT = { is_participant = scope:member }
                }
                save_scope_as = war
                if = {
                    limit = {
                        primary_attacker = scope:war_target
                    }
                    primary_defender = { save_scope_as = war_partner }
                }
                else = {
                    primary_attacker = { save_scope_as = war_partner }
                }
                random_list = {
                    1000 = {}
                    250 = {
                        join_war_against_coalition_scripted_modifier = { CRISIS_HOLDER = scope:war_target }

                        if = {
                            limit = {
                                primary_attacker = scope:war_target
                            }
                            set_called_to = scope:member
                            add_defender = scope:member
                            debug_log = "A coalition member has joined a def war"
                            debug_log_date = yes
                        }
                        else = {
                            set_called_to = scope:member
                            add_attacker = scope:member
                            debug_log = "A coalition member has joined a atk war"
                            debug_log_date = yes
                        }
                    }
                }
            }
        }
    }
}
