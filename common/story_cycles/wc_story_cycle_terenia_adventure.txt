story_cycle_pa_childhood = {
	on_setup = {
		set_variable = { name = my_guardian value = story_owner.var:my_guardian }
	}

	on_end = {
	}

	on_owner_death = {
		end_story = yes
	}

	effect_group = {
		years = 1
		
		triggered_effect = {
			trigger = {
				story_owner.age >= 16
			}
			effect = {
				story_owner = {
					trigger_event = { id = personal_adventure.050 days = { 5 10 } }
				}
				end_story = yes
			}
		}
		triggered_effect = {
			trigger = {
				var:my_guardian ?= { is_alive = yes }
			}
			effect = {
				story_owner = {
					if = {
						limit = {
							NOT = {
								any_relation = {
									type = ward
									this = prev.var:my_guardian
								}
							}
						}
						if = {
							limit = {
								any_relation = {
									type = guardian
								}
							}
							random_relation = {
								type = guardian
								remove_relation_ward = prev
							}
						}
						var:my_guardian = { set_relation_ward = prev }
					}
				}
			}
		}
		triggered_effect = {
			trigger = {
				story_owner.age = 8
			}
			effect = {
				story_owner = { trigger_event = { id = personal_adventure.005 days = { 10 20 } } }
			}
		}
		triggered_effect = {
			trigger = {
				story_owner.age = 11
			}
			effect = {
				story_owner = { trigger_event = { id = personal_adventure.006 days = { 10 20 } } }
			}
		}
		triggered_effect = {
			trigger = {
				story_owner.age = 14
			}
			effect = {
				story_owner = { trigger_event = { id = personal_adventure.007 days = { 10 20 } } }
			}
		}
	}
}
story_cycle_terenia_invasion = {
	on_setup = {
		story_owner = {
			#if someone died already
			if = {
				limit = {
					has_variable_list = supporters_of_terenia_menethil
				}
				every_in_list = {
					variable = supporters_of_terenia_menethil
					limit = { is_alive = no }
					prev = { remove_list_variable = { name = supporters_of_terenia_menethil target = prev } }
				}
			}
			#end own wars, leave wars as allies
			if = {
				limit = { is_at_war = yes }
				every_character_war = {
					limit = { is_war_leader = prev }
					end_war = white_peace
				}
				every_character_war = {
					limit = { NOT = { is_war_leader = prev } }
					remove_participant = prev
				}
			}
			if = {
				limit = { NOT = { has_variable = kill_lich_king_first } }
				every_held_title = {
					limit = { tier >= tier_county }
					add_to_temporary_list = my_previous_titles
				}
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = title_change
				add_claim_on_loss = yes
			}
			title:c_solliden = {
				change_title_holder = {
					holder = prev
					change = scope:title_change
				}
				set_county_culture = culture:lordaeronian
				set_county_faith = prev.faith
			}
			title:c_glistening_shore = {
				change_title_holder = {
					holder = prev
					change = scope:title_change
				}
				set_county_culture = culture:lordaeronian
				set_county_faith = prev.faith
				title_province = {
					if = {
						limit = {
							has_holding_type = tribal_holding
						}
						set_holding_type = castle_holding
						generate_building = yes
					}
				}
			}
			title:d_deathknell = {
				change_title_holder = {
					holder = prev
					change = scope:title_change
				}
			}
			resolve_title_and_vassal_change = scope:title_change
			if = {
				limit = { is_independent_ruler = no is_ruler = yes }
				top_liege = { save_scope_as = my_top_liege }
				create_title_and_vassal_change = {
					type = created
					save_scope_as = change
					add_claim_on_loss = no
				}
				every_in_list = {
					list = my_previous_titles
					change_title_holder = {
						holder = scope:my_top_liege
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = change
					add_claim_on_loss = no
				}
				becomes_independent = { change = scope:change }
				resolve_title_and_vassal_change = scope:change
			}
			set_realm_capital = title:c_solliden
			if = {
				limit = { highest_held_title_tier < tier_kingdom }
				set_primary_title_to = title:d_deathknell
			}
			every_player = {
				limit = { NOT = { this = prev } capital_province = { geographical_region = world_eastern_kingdoms } }
				trigger_event = {
					id = personal_adventure.211
					days = 4
				}
			}
			#1000
			if = {
				limit = { NOT = { has_variable = kill_lich_king_first } }
				spawn_army = {
					uses_supply = no
					inheritable = yes
					name = remnants_of_alliance_troops
					men_at_arms = {
						type = house_guard
						stacks = 8
					}
					men_at_arms = {
						type = armored_horsemen
						stacks = 3
					}
					men_at_arms = {
						type = trebuchet
						stacks = 5
					}
					location = capital_province
					origin = capital_province
				}
			}
			save_temporary_scope_as = terenia_menethil_scope
			if = {
				limit = {
					any_character_to_title_neighboring_county = {
						kingdom = title:k_lordaeron
						holder.top_liege = { NOR = { is_allied_to = scope:terenia_menethil_scope faith = scope:terenia_menethil_scope.faith } }
					}
				}
				every_character_to_title_neighboring_county = {
					limit = {
						kingdom = title:k_lordaeron
						holder.top_liege = { NOT = { is_allied_to = scope:terenia_menethil_scope faith = scope:terenia_menethil_scope.faith } }
					}
					holder.top_liege = {
						add_to_temporary_list = war_targets
					}
				}
				ordered_in_list = {
					list = war_targets
					max = 3
					check_range_bounds = no
					order_by = current_military_strength
					if = {
						limit = { NOT = { exists = scope:war_target_1 } }
						save_scope_as = war_target_1
					}
					else_if = {
						limit = { NOT = { exists = scope:war_target_2 } }
						save_scope_as = war_target_2
					}
					else_if = {
						limit = { NOT = { exists = scope:war_target_3 } }
						save_scope_as = war_target_3
					}
				}
				trigger_event = { id = personal_adventure.212 days = 2 }
			}
		}
	}

	on_end = {
		story_owner = { if = { limit = { is_alive = yes } clear_variable_list = supporters_of_terenia_menethil } }
	}

	on_owner_death = {
		end_story = yes
	}

	effect_group = {
		years = 1
		
		triggered_effect = {
			trigger = {
				story_owner = { is_at_war = no NOT = { exists = involved_activity } }
			}
			effect = {
				story_owner = {
					save_temporary_scope_as = terenia_menethil_scope
					if = {
						limit = {
							is_independent_ruler = yes
							any_character_to_title_neighboring_county = {
								kingdom = title:k_lordaeron
								holder.top_liege = {
									NOR = {
										is_allied_to = scope:terenia_menethil_scope
										has_truce = scope:terenia_menethil_scope
									}
									trigger_if = {
										limit = { has_faith = faith:scarletism }
										NOT = { scope:terenia_menethil_scope.faith = faith:scarletism }
									}
								}
							}
						}
						every_character_to_title_neighboring_county = {
							limit = {
								kingdom = title:k_lordaeron
								holder.top_liege = {
									NOR = {
										is_allied_to = scope:terenia_menethil_scope
										has_truce = scope:terenia_menethil_scope
									}
									trigger_if = {
										limit = { has_faith = faith:scarletism }
										NOT = { scope:terenia_menethil_scope.faith = faith:scarletism }
									}
								}
							}
							holder.top_liege = {
								add_to_temporary_list = war_targets
							}
						}
						ordered_in_list = {
							list = war_targets
							max = 3
							check_range_bounds = no
							order_by = current_military_strength
							if = {
								limit = { NOT = { exists = scope:war_target_1 } }
								save_scope_as = war_target_1
							}
							else_if = {
								limit = { NOT = { exists = scope:war_target_2 } }
								save_scope_as = war_target_2
							}
							else_if = {
								limit = { NOT = { exists = scope:war_target_3 } }
								save_scope_as = war_target_3
							}
						}
						trigger_event = personal_adventure.212
					}
					else_if = {
						limit = {
							is_independent_ruler = yes
							any_character_to_title_neighboring_county = {
								kingdom = title:k_lordaeron
								holder.top_liege = { NOR = { is_allied_to = scope:terenia_menethil_scope faith = scope:terenia_menethil_scope.faith } }
							}
						}
					}
					else = {
						root = { end_story = yes }
					}
				}
			}
		}
	}
}
