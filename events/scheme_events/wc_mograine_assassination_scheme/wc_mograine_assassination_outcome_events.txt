
# Outcome events for the wc_mograine_assassination scheme
# root here is the scheme owner, Balnazzar (same convention as
# wc_balnazzar_contingency) - he's pulling the strings and the one who sees
# the scheme UI. Renault, the actual instrument, is referenced explicitly by
# character:60019 wherever he needs to appear.

namespace = wc_mograine_assassination_outcome

# MAINTENANCE EVENTS 0000 - 0099
##################################

# Rolls success/failure when the scheme completes (on_phase_completed pulse)
wc_mograine_assassination_outcome.0001 = {
	hidden = yes

	immediate = {
		random = {
			chance = scope:scheme.scheme_success_chance

			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}

		hidden_effect = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = wc_mograine_assassination_outcome.2001
			}
			else = {
				trigger_event = wc_mograine_assassination_outcome.4001
			}
		}
	}
}

# Failure retry: the flag set by .1004 only lets the decision fire once, so a
# failed scheme re-attempts itself here instead of waiting on the decision to
# reappear - scheduled from .4001's hidden_effect, well past the scheme's own
# cooldown so start_scheme doesn't reject it.
wc_mograine_assassination_outcome.0002 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				root = { is_alive = yes } # Balnazzar
				character:60017 = { is_alive = yes } # Alexandros Mograine
				character:60019 = { is_alive = yes } # Renault Mograine
			}
			start_scheme = {
				type = wc_mograine_assassination
				target_character = character:60017
			}
		}
	}
}

# The Ambush at Stratholme - Renault kills Alexandros with the dropped Ashbringer, corrupting
# it. Kel'Thuzad claims it almost right away (.2003), not years later.
wc_mograine_assassination_outcome.2001 = {
	type = character_event
	window = scheme_successful_event
	title = wc_mograine_assassination_outcome.2001.title
	desc = wc_mograine_assassination_outcome.2001.desc
	theme = death

	override_background = wc_stratholme

	left_portrait = {
		character = character:60017 # Alexandros Mograine
		animation = personality_honorable
	}
	right_portrait = {
		character = character:60019 # Renault Mograine
		animation = disbelief
	}

	immediate = {
		character:60017 = { save_scope_as = alexandros }
	}

	option = {
		name = wc_mograine_assassination_outcome.2001.a

		show_as_tooltip = {
			scope:alexandros = { death = { death_reason = death_battle killer = character:60019 } }
		}
		hidden_effect = {
			scope:alexandros = {
				death = { death_reason = death_battle killer = character:60019 }
			}

			character:60019 = {
				# Destroy inherited uncorrupted Ashbringer before minting corrupted one,
				# or Renault ends up carrying both.
				every_character_artifact = {
					limit = { has_variable = ashbringer }
					destroy_artifact = this
				}

				# Renault leaves the corrupted blade behind unequipped; Kel'Thuzad claims it in .2003.
				create_artifact_corrupted_ashbringer_effect = { OWNER = this }

				# strong_blackmail_hook needs a real secret behind it, not just a target.
				add_secret = {
					type = secret_murder
					target = scope:alexandros
				}
				random_secret = {
					limit = {
						secret_type = secret_murder
						secret_target = scope:alexandros
					}
					save_scope_as = renault_murder_secret
				}
			}

			# Balnazzar ordered it too - his own secret, so he carries risk too, not just Renault.
			add_secret = {
				type = secret_murder
				target = scope:alexandros
			}

			# Real leverage over Renault now, not just narration.
			add_hook = {
				type = strong_blackmail_hook
				target = character:60019 # Renault Mograine
				secret = scope:renault_murder_secret
			}

			if = {
				limit = { exists = scope:scheme }
				scope:scheme = {
					# The least discreet agent is the easiest to lean on afterward.
					ordered_scheme_agent_character = {
						order_by = {
							value = 100
							subtract = intrigue
						}
						save_scope_as = mc_agent
					}
				}
				if = {
					limit = { exists = scope:mc_agent }
					root = {
						add_hook_no_toast = {
							type = mindcontrol_hook
							target = scope:mc_agent
						}
					}
					scope:mc_agent = {
						add_character_flag = wc_balnazzar_mc_agent_flag
						# Stored directly instead of relying on liege/employer, which isn't
						# necessarily Balnazzar for every courtier who could fill this slot.
						set_variable = { name = wc_balnazzar_mc_owner value = root }
						trigger_event = {
							id = wc_mograine_assassination_outcome.2006
							days = { 200 340 }
						}
					}
				}

				# Everyone else who helped gets bound with an ordinary loyalty hook.
				scope:scheme = {
					every_scheme_agent_character = {
						limit = { NOT = { this = scope:mc_agent } }
						root = {
							add_hook = {
								type = loyalty_hook
								target = prev
							}
						}
					}
				}

				scope:scheme = { end_scheme = yes }
			}

			# root = Balnazzar throughout this file (scheme owner) - .1006 assumes
			# the same, so a plain trigger_event is enough, no rerouting needed.
			trigger_event = {
				id = wc_balnazzar_story.1006
				days = { 10 25 }
			}

			trigger_event = {
				id = wc_mograine_assassination_outcome.2002
				days = { 5 15 }
			}

			trigger_event = {
				id = wc_mograine_assassination_outcome.2003
				days = { 3 10 }
			}

			trigger_event = {
				id = wc_mograine_assassination_outcome.2004
				days = { 20 40 }
			}

			trigger_event = {
				id = wc_mograine_assassination_outcome.2005
				days = { 30 60 }
			}
		}
	}
}

# A Brother's Grief - Darion, Renault's favored brother, learns Alexandros is
# dead and starts to wonder. Fires only if Darion is still alive - if he isn't,
# there's no one left to suspect.
wc_mograine_assassination_outcome.2002 = {
	type = character_event
	title = wc_mograine_assassination_outcome.2002.title
	desc = wc_mograine_assassination_outcome.2002.desc
	theme = intrigue

	trigger = {
		exists = character:60020 # Darion Mograine
		character:60020 = { is_alive = yes }
	}

	override_background = sm_pa_scarlet_monastery_sm

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60020 # Darion Mograine
		animation = worry
	}

	immediate = {
		character:60020 = {
			add_stress = medium_stress_gain
		}
	}

	option = {
		name = wc_mograine_assassination_outcome.2002.a

		random_list = {
			60 = { # Grief wins out over doubt
				custom_tooltip = wc_mograine_assassination_outcome.2002.a.accepts
			}
			40 = { # Doubt wins out
				custom_tooltip = wc_mograine_assassination_outcome.2002.a.suspects
				character:60020 = {
					set_relation_rival = {
						target = root
						reason = rival_darion_suspects_balnazzar
					}
					add_opinion = { modifier = wc_suspects_scheming_against_me target = root }
				}
				add_to_variable_list = {
					name = balnazzar_known_rivals
					target = character:60020
				}
				add_character_modifier = { modifier = watched_by_a_witness_modifier }
			}
		}
	}
}

# What the Blood Bought - reads on Balnazzar's own network of agents from the ambush: did
# anyone come out of it actually bound to him, or did they all just walk away once paid.
wc_mograine_assassination_outcome.2005 = {
	type = character_event
	title = wc_mograine_assassination_outcome.2005.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					any_hooked_character = {
						count > 0
						root = { has_hook_of_type = { target = prev type = loyalty_hook } }
					}
				}
				desc = wc_mograine_assassination_outcome.2005.desc_bound
			}
			desc = wc_mograine_assassination_outcome.2005.desc_alone
		}
	}
	theme = intrigue

	override_background = wc_stratholme

	left_portrait = {
		character = root
		animation = thinking
	}

	option = {
		name = wc_mograine_assassination_outcome.2005.a
		trigger = {
			any_hooked_character = {
				count > 0
				root = { has_hook_of_type = { target = prev type = loyalty_hook } }
			}
		}

		add_character_modifier = {
			modifier = bound_by_blood_and_secrets
			years = 5
		}
	}

	option = {
		name = wc_mograine_assassination_outcome.2005.b
		trigger = {
			NOT = {
				any_hooked_character = {
					count > 0
					root = { has_hook_of_type = { target = prev type = loyalty_hook } }
				}
			}
		}

		add_character_modifier = {
			modifier = alone_in_the_dark
			years = 5
		}
	}
}

# A Puppet Among the Faithful - the agent Balnazzar quietly took hold of during the ambush
# either stays his, or finally shakes him off.
wc_mograine_assassination_outcome.2006 = {
	type = character_event
	title = wc_mograine_assassination_outcome.2006.title
	desc = wc_mograine_assassination_outcome.2006.desc
	theme = intrigue

	trigger = {
		has_character_flag = wc_balnazzar_mc_agent_flag
		var:wc_balnazzar_mc_owner ?= { has_hook_of_type = { target = root type = mindcontrol_hook } }
	}

	left_portrait = {
		character = root
		animation = paranoia
	}
	right_portrait = {
		character = var:wc_balnazzar_mc_owner
		animation = scheme
	}

	option = { # Control holds - stays flagged, checks back in later
		name = wc_mograine_assassination_outcome.2006.a

		trigger_event = {
			id = wc_mograine_assassination_outcome.2006
			days = { 200 340 }
		}
	}

	option = { # Breaks free
		name = wc_mograine_assassination_outcome.2006.b
		trigger = { is_ai = no }

		var:wc_balnazzar_mc_owner = {
			remove_hook = {
				target = root
				type = mindcontrol_hook
			}
			add_stress = medium_stress_gain
			add_opinion = {
				modifier = wc_suspects_scheming_against_me
				target = root
			}
		}
		remove_character_flag = wc_balnazzar_mc_agent_flag
	}
}
