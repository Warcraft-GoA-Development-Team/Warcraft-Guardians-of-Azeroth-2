scripted_effect last_stand_of_defender_effect = {
	duel = {
		skill = $SKILL$
		value = scope:duel_diff
		# Victory!
		50 = {
			compare_modifier = {
				value = -40
				multiplier = 3.5
				min = -49
			}
			min = 1
			desc = last_stand_of_defender_effect.$SKILL$.tt.success

			scope:defender = { add_trait = wounded_2 }
		}
		# Defeat!
		50 = {
			compare_modifier = {
				value = -40
				multiplier = -3.5
				min = -49
			}
			min = 5
			desc = last_stand_of_defender_effect.$SKILL$.tt.failure

			trigger_death_of_defender_effect = yes
		}
	}
}

scripted_effect trigger_death_of_defender_effect = {
	if = {
		limit = { scope:defender = { is_ai = no } }
		show_as_tooltip = { death_of_defender_effect = yes }
		scope:defender = { trigger_event = wc_path_of_the_damned.2005 }
	}
	else = {
		show_as_tooltip = { death_of_defender_effect = yes }
		scope:attacker = { trigger_event = wc_path_of_the_damned.2005 }
	}
}

scripted_effect death_of_defender_effect = {
	scope:defender = {
		if = {
			limit = { is_at_war_with = scope:attacker }
			every_character_war = {
				limit = {
					primary_defender = scope:defender
					is_attacker = scope:attacker
				}
				end_war = attacker
			}
		}

		if = {
			limit = { scope:attacker = { any_character_artifact = { this = scope:frostmourne } } }
			override_death_killer_artifact_effect = {
				death_reason = death_battle
				killer = scope:attacker
				artifact = scope:frostmourne
			}
		}
		else = {
			override_death_killer_effect = {
				death_reason = death_battle
				killer = scope:attacker
			}
		}
	}
}
