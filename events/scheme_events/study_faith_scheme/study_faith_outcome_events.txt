study_faith_outcome.0100 = {
    hidden = yes
    
    trigger = {
        has_variable = pending_conversion_to_liege
    }
    
    immediate = {
        # Save scopes
        var:pending_conversion_to_liege = { save_scope_as = actor }
        save_scope_as = recipient
        
        if = {
            limit = {
                scope:actor = { is_alive = yes }
                is_vassal_of = scope:actor
            }
            # Now trigger the actual conversion flow
            # Player gets choice, AI converts
            trigger_event = study_faith_outcome.0101
        }
        else = {
            # Liege died or relationship changed - obligation void
            remove_variable = pending_conversion_to_liege
        }
    }
}

# Player/AI conversion event after successful study
study_faith_outcome.0101 = {
    type = character_event
    title = study_faith_outcome.0101.t
    desc = study_faith_outcome.0101.desc
    theme = study_faith

	left_portrait = {
		character = root
		animation = prayer_standing_eyes
	}
	right_portrait = {
		character = scope:actor
		animation = page_flipping
		camera = camera_event_left_forward
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
    # Option A: Convert now (should have good knowledge, minimal penalties)
    option = {
        name = study_faith_outcome.0101.a

		trigger = {
			scope:recipient.faith.main_rite = {
				is_rite_enabled = yes
				can_convert_to_rite = yes
			}
		}

        # End the ongoing study before converting, or the faith change invalidates it noisily
        hidden_effect = {
            random_scheme = {
                type = study_faith
                end_scheme = yes
            }
        }
        demand_conversion_interaction_effect = yes
        remove_variable = pending_conversion_to_liege
        # Notify liege
        scope:actor = {
            trigger_event = {
                id = study_faith_outcome.0102
                days = 1
            }
        }
        ai_chance = {
            base = 100
        }
    }
    
    # Option B: Refuse anyway (break promise - consequences)
    option = {
        name = study_faith_outcome.0101.b
        # Breaking promise has consequences
        add_prestige = medium_prestige_loss
        scope:actor = {
            add_opinion = {
                target = root
                modifier = broke_promise
            }
            # Notify liege of betrayal
            trigger_event = {
                id = study_faith_outcome.0103
                days = 1
            }
        }
        remove_variable = pending_conversion_to_liege
        stress_and_fulfillment_impact = {
            zealous = minor_stress_impact_loss
        }
        ai_chance = {
            base = 0
            # Only zealous AI would break promise
            modifier = {
                add = 50
                has_trait = zealous
            }
        }
    }
}
