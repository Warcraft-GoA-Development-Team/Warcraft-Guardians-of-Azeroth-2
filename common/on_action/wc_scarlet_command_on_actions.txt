# Independent institutions; shared engine hooks accumulate through redirects.
on_title_gain = { on_actions = { wc_scarlet_command_update } }
on_title_lost = { on_actions = { wc_scarlet_command_update } }
on_vassal_change = { on_actions = { wc_scarlet_command_update } }
on_county_faith_change = { on_actions = { wc_scarlet_command_update } }
on_character_faith_change = { on_actions = { wc_scarlet_command_update } }
yearly_global_pulse = { on_actions = { wc_scarlet_command_update } }
quarterly_playable_pulse = { on_actions = { wc_scarlet_command_pulse } }

wc_scarlet_command_update = {
	effect = { wc_scarlet_update_crusade_effect = yes }
}

wc_scarlet_command_pulse = {
	trigger = {
		OR = {
			has_character_flag = has_formed_the_scarlet_crusade
			AND = {
				exists = global_var:wc_scarlet_crusade_title
				global_var:wc_scarlet_crusade_title = { holder = root }
			}
		}
	}
	effect = {
		# Identify the legacy Order before considering an ordinary founder's title.
		wc_scarlet_migrate_crusade_effect = yes
		# A living generic founder supplies its own title on old saves.
		if = {
			limit = {
				NOT = { exists = global_var:wc_scarlet_crusade_title }
				has_character_flag = has_formed_the_scarlet_crusade
				exists = primary_title
			}
			set_global_variable = { name = wc_scarlet_crusade_title value = primary_title }
		}
		wc_scarlet_update_crusade_effect = yes
	}
}
