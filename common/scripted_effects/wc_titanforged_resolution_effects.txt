# Root is the deciding ruler; scopes are saved by the cession event.
wc_titanforged_release_duchy_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	scope:ceded_duchy = {
		# Never include the duchy's de facto vassals: some may hold land elsewhere.
		change_title_holder = { holder = scope:local_ruler change = scope:title_change }
		every_de_jure_county = {
			change_title_holder_include_vassals = {
				holder = scope:local_ruler
				change = scope:title_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:title_change
	scope:local_ruler = {
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
			add_claim_on_loss = no
		}
		becomes_independent = { change = scope:change }
		resolve_title_and_vassal_change = scope:change
		add_opinion = {
			modifier = wc_titanforged_grateful_for_liberation_opinion
			target = root
		}
	}
}

wc_titanforged_resolve_duchy_effect = {
	wc_titanforged_restoration_resolve_effect = yes
	scope:ceded_duchy = {
		every_de_jure_county = {
			remove_county_modifier = wc_titanforged_reordered_county_modifier
		}
	}
	add_prestige = major_prestige_value
	add_piety = major_piety_value
}
