			id = wc_balnazzar_story.1004
			days = 3
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 50
	}

# Active lever for wc_mograine_assassination - the scheme itself has no agents
# to recruit, so this is the only way to directly push it besides waiting.
# Yearly cooldown via the flag below, no built-in decision cooldown field exists.
balnazzar_press_renault_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = decisions

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	desc = balnazzar_press_renault_decision_desc

	is_shown = {
		any_scheme = {
			type = wc_mograine_assassination
			count > 0
		}
	}

	is_valid_showing_failures_only = {
		NOT = { has_character_flag = balnazzar_pressed_renault_cooldown }
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		every_scheme = {
			limit = { type = wc_mograine_assassination }
			add_scheme_progress = 15
		}
		add_stress = medium_stress_gain
		add_character_flag = { flag = balnazzar_pressed_renault_cooldown years = 1 }
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 30
	}
}

# Trigger 3 of 3 for wc_balnazzar_story.2000 - lets player strike first rather than wait to be exposed.
# Gated on high stress since nathrezim_disguise's stress_loss_mult already makes it hard to shed.
balnazzar_shed_the_mask_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = decisions

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	desc = balnazzar_shed_the_mask_decision_desc

	is_shown = {
		has_trait = nathrezim_disguise
		stress_level >= 2
		NOT = { has_character_flag = balnazzar_shed_the_mask }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	# ck3 quirk: bare trigger_event with no other effect = blank Effects panel tooltip.
	# Wrap in custom_tooltip, same fix as purify_reclaimed above.
	effect = {
		custom_tooltip = balnazzar_shed_the_mask_decision_effect_tooltip

		trigger_event = {
			id = wc_balnazzar_story.2000
			days = 3
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 10

		modifier = {
			add = 40
			has_trait = wrathful
		}
		modifier = {
			add = 40
			has_trait = paranoid
		}
	}
}
