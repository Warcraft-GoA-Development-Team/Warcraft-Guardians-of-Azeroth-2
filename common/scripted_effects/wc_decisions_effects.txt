create_arathor_empire_scripted_effect = {
	primary_title = { save_scope_as = old_primary_title }
	add_trait = augustus
	add_prestige = 2500
	dynasty ?= { add_dynasty_prestige = 5000 }
	#Create Empire of Arathor, shift dejure/history/laws.

	if = {
		limit = {
			NOT = { has_realm_law = single_heir_succession_law }
		}
		add_realm_law_skip_effects = single_heir_succession_law
	}

	if = {
		limit = { capital_county = { this = title:c_stromgarde } }
		title:h_arathor = {
			set_capital_county = title:c_stromgarde
		}
	}

	hidden_effect = {
		every_held_title = { #Should shift all dejure of all Empires owned at the time.
			title_tier = empire
			set_de_jure_liege_title = title:h_arathor
		}
		every_empire = {
			limit = {
				NOT = { exists = hegemony }
				any_de_jure_county = {
					percent >= 0.6
					holder = {
						OR = {
							this = root
							any_liege_or_above = { this = root }
						}
					}
				}
			}
			set_de_jure_liege_title = title:h_arathor
		}
		title:k_hinterlands = { set_de_jure_liege_title = title:e_arathor } #Hinterlands becomes dejure part from Zul'aman to Arathor
		set_primary_title_to = title:h_arathor
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:h_arathor = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change

	if = { # Move your Title MaAs to your new title if you have any
		limit = {
			government_allows = administrative
			primary_title = {
				any_title_maa_regiment = { }
			}
		}
		hidden_effect = {
			primary_title = { transfer_title_maa_ownership = title:h_arathor }
		}
	}

	every_held_title = { #Should destroy all other Empires owned at the time.
		title_tier = empire

		root = { destroy_title = prev }
	}
	
	if = { #Automatically move capital to Stromgarde, unless it's been set to Lordaeron.
		limit = { # If you already hold Lordaeron, it becomes your capital
			title:c_lordaeron.holder = root
			NOT = { capital_county = { this = title:c_stromgarde } }
			culture = { has_cultural_pillar = heritage_arathi }
		}
		hidden_effect = {
			if = { #Usurp if not held personally.
				limit = {
					NOT = { this = title:c_lordaeron.holder }
				}
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:c_lordaeron = {
					change_title_holder = {
						holder = root
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		root = { set_realm_capital = title:c_lordaeron }
	}
	else_if = {
		limit = { # If you already hold Stromgarde, it becomes your capital
			NOT = { capital_county = { this = title:c_lordaeron } }
			title:c_stromgarde.holder = root
			culture = { has_cultural_pillar = heritage_arathi }
		}
		hidden_effect = {
			if = { #Usurp if not held personally.
				limit = {
					NOT = { this = title:c_stromgarde.holder }
				}
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:c_stromgarde = {
					change_title_holder = {
						holder = root
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		root = { set_realm_capital = title:c_stromgarde }
	}
	else = { # You get Stromgarde as your capital if you hold neither Lordaeron nor Stromgarde
		hidden_effect = {
			if = { #Usurp if not held personally.
				limit = {
					NOT = { this = title:c_stromgarde.holder }
				}
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:c_stromgarde = {
					change_title_holder = {
						holder = root
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		root = { set_realm_capital = title:c_stromgarde }
	}
}
