namespace = wc_magic_flavor_events_pulse

wc_magic_flavor_events_pulse.0001 = {
	hidden = yes

	trigger = {
		employs_court_position = court_caster_court_position
		is_magic_user_trigger = yes
	}

	cooldown = { years = 10 }

	weight_multiplier = {
		base = 1
		add = {
			value = scope:court_caster.wc_mp_value
		}
		add = {
			value = 2
			has_variable = court_caster_task_manage_mana
		}
	}

	immediate = {
		court_position:court_caster_court_position = {
			save_scope_as = court_caster
		}

		send_interface_message = {
			type = msg_court_caster_pulse_action_good
			left_icon = scope:court_caster
			title = wc_magic_flavor_events_pulse.0001.message
			desc = {
				triggered_desc = {
					trigger = {
						scope:court_caster.aptitude:court_caster_court_position <= 2
					}
					desc = court_caster_good_unskilled_notification_tooltip
				}
				triggered_desc = {
					trigger = {
						scope:court_caster.aptitude:court_caster_court_position > 2
					}
					desc = court_caster_good_skilled_notification_tooltip
				}
			}
			wc_change_mana_by_value_effect = { # add mana
                CHANGE = add
                VALUE = minor_mana_value
            }
		}
	}
}
