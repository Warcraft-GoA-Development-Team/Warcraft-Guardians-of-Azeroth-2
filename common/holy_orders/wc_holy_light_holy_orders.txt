# Brotherhood of Northshire
# Only created through enforce_the_brotherhood_scripted_effect
ho_holy_light_brotherhood_of_northshire = {

	type = military
	government = holy_order_government
	title = d_brotherhood_of_northshire
	name = "holy_order_brotherhood_of_northshire"
	coat_of_arms = "ho_brotherhood_of_northshire"

	trigger = {
		scope:founder.faith = { wc_valid_light_faith_for_northshire_trigger = yes }
		brotherhood_have_not_existed_trigger = yes
		
		exists = scope:province
		exists = title:c_northshire.holder
		scope:province.county.holder = title:c_northshire.holder
	}

	weight = { value = 10000 }

	men_at_arms = { order_serjeants }
}

# Champions of Peace
ho_holy_light_champions_of_peace = {

	type = military
	government = holy_order_government
	name = "holy_order_champions_of_peace"

	trigger = {
		scope:founder.faith = faith:holy_light
		NOT = {
			scope:founder.faith = {
				any_faith_holy_order = {
					title = { has_variable = ho_champions_of_peace }
				}
			}
		}
	}

	weight = {
		value = 10
		#More likely if founder is compassionate
		if = {
			limit = {
				scope:founder = {
					ai_compassion >= high_positive_ai_value
				}
			}
			add = 50
		}
	}

	men_at_arms = { order_knights }

	effect = {
		set_variable = ho_champions_of_peace
	}
}

# Champions of Light
ho_holy_light_champions_of_light = {

	type = military
	government = holy_order_government
	name = "holy_order_champions_of_light"

	trigger = {
		scope:founder.faith = faith:holy_light
		NOT = {
			scope:founder.faith = {
				any_faith_holy_order = {
					title = { has_variable = ho_champions_of_light }
				}
			}
		}
	}

	weight = {
		value = 10
		#More likely if first holy order
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							leader ?= {
								government_has_flag = government_is_military_holy_order
							}
						}
					}
				}
			}
			add = 50
		}
	}

	men_at_arms = { order_knights }

	effect = {
		set_variable = ho_champions_of_light
	}
}

			}
			add = 50
		}
	}

	men_at_arms = { order_knights }

	effect = {
		set_variable = ho_champions_of_truth
	}
}
