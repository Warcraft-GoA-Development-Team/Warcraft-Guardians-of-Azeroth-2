close_dark_portal = {
	ai_check_interval = 8
	picture = { reference = "gfx/interface/illustrations/decisions/dark_portal_in_draenor.dds" }
	desc = close_dark_portal_desc
	selection_tooltip = close_dark_portal_tooltip
	decision_group_type = major

	is_shown = {
		OR = {
			has_title = title:c_dark_portal
			AND = {
				is_landless_adventurer = yes
				is_ai = yes 
				current_date >= 602.1.1
				location = { geographical_region = world_eastern_kingdoms_azeroth }
				NOT = {
					title:c_dark_portal.holder ?= {
						can_destroy_dark_portal_trigger = yes 
					}
				}
				NOT = { is_target_in_global_variable_list = { name = unavailable_unique_events target = flag:beyond_dark_portal } }
			} 
		}
		
		can_destroy_dark_portal_trigger = yes
		is_incapable_trigger = no 
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { is_independent_ruler = no }
			any_liege_or_above = {
				can_destroy_dark_portal_trigger = yes
			}
		}
	}

	effect = {
		destroy_dark_portal_effect = yes
	}
	
	ai_will_do = {
		base = 100
		modifier = {
			factor = 0.1
			is_landless_adventurer = yes
		}
	}
}
