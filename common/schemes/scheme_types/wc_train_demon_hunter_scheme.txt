train_demon_hunter = {
    # Basic Setup
    skill = prowess
    category = personal
    desc = train_demon_hunter_desc_general
    success_desc = "TRAIN_DEMON_HUNTER_SUCCESS_DESC"
    icon = icon_scheme_convert_to_witchcraft
    illustration = "gfx/interface/illustrations/character_view/tomb_of_sargeras.dds"
    target_type = character
    is_basic = yes
    is_secret = no
    cooldown = { years = 10 }

    # Parameters
    base_maximum_success = 95
    minimum_success = 5
    base_progress_goal = 730 # 2 Years of training atleast
    speed_per_skill_point = -2
    spymaster_speed_per_skill_point = 0

    allow = {
        trigger_if = {
			limit = {
				NOT = { government_has_flag = government_is_landless_adventurer }
			}
			is_travelling = no
		}
        age >= 18 # Not using is_adult = yes since for some elder warcraft races the "adult" age is calculated as 200 years or more
		is_imprisoned = no
        is_incapable = no
        OR = {
            has_trait = creature_night_elf
            has_trait = creature_high_elf
            has_trait = creature_draenei
            has_trait = creature_broken
        }

        scope:target = {
            has_trait = demon_hunter
            has_character_flag = became_demon_hunter_through_a_ritual
            NOT = { has_character_flag = became_demon_hunter_by_being_a_novice } # Cannot train other Demon Hunters if you didn't become one in a Ritualistic way
        }
    }

    valid = {
        NOT = { is_isolated_from_trigger = { target = scope:target } }

        is_incapable = no
        scope:owner = {
            age >= 18
            has_faith = faith:illidari
        }

        NOT = {
            scope:owner = {
                has_trait = demon_hunter
            }
        }

        scope:target = { is_incapable_trigger = no }

        # One of you is imprisoned
		is_imprisoned = no
		scope:target = {
			is_imprisoned = no
		}

        # You're too far away
		scope:target = {
			OR = {
				exists = location
				in_diplomatic_range = scope:owner
			}
		}
    }

    base_success_chance = {
        base = 0
        scheme_type_skill_success_chance_modifier = { SKILL = PROWESS }

        #Physique good
		modifier = {
			desc = "scheme_physique_good_1"
			scope:owner = { has_trait = physique_good_1 }
			add = 10
		}
		modifier = {
			desc = "scheme_physique_good_2"
			scope:owner = { has_trait = physique_good_2 }
			add = 20
		}
		modifier = {
			desc = "scheme_physique_good_3"
			scope:owner = { has_trait = physique_good_3 }
			add = 30
		}

        ##### If your trainer is an experienced Demon Hunter or not
        ### Spectral Assassin Track modifiers
        modifier = {
            desc = spectral_assassin_modifier_desc_decent
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = spectral_assassin
                    value > 33
                }
            }
            add = 5
        }
        modifier = {
            desc = spectral_assassin_modifier_desc_good
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = spectral_assassin
                    value > 66
                }
            }
            add = 5
        }
        modifier = {
            desc = spectral_assassin_modifier_desc_best
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = spectral_assassin
                    value >= 100
                }
            }
            add = 5
        }
        ### Slayer Track modifiers
        modifier = {
            desc = slayer_modifier_desc_decent
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = slayer
                    value > 33
                }
            }
            add = 5
        }
        modifier = {
            desc = slayer_modifier_desc_good
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = slayer
                    value > 66
                }
            }
            add = 5
        }
        modifier = {
            desc = slayer_modifier_desc_best
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = slayer
                    value >= 100
                }
            }
            add = 5
        }
        ### Fury Track modifiers
        modifier = {
            desc = fury_modifier_desc_decent
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = fury
                    value > 33
                }
            }
            add = 5
        }
        modifier = {
            desc = fury_modifier_desc_good
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = fury
                    value > 66
                }
            }
            add = 5
        }
        modifier = {
            desc = fury_modifier_desc_best
            scope:target = {
                has_trait_xp = {
                    trait = demon_hunter
                    track = fury
                    value >= 100
                }
            }
            add = 5
        }
    }

    on_phase_completed = {
		# Grab our scopes.
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }
		# Roll our result.
		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		# How'd we do?
		scope:owner = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = train_demon_hunter.0001
			}
			else = {
				trigger_event = train_demon_hunter.0002
			}
		}
	}

    on_invalidated = {
        scheme_owner = {
            send_interface_message = {
                type = event_generic_neutral
                title = train_demon_hunter
                left_icon = scope:target
                custom_tooltip = train_demon_hunter_invalid
            }
        }
    }
}
