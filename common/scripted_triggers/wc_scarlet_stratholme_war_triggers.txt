# CB context: only the institution's ruler pressing their own campaign claim.
wc_scarlet_stratholme_claimant_trigger = {
	scope:attacker = scope:claimant
	exists = global_var:wc_scarlet_crusade_title
	global_var:wc_scarlet_crusade_title = {
		holder = scope:attacker
		var:wc_scarlet_campaign_target ?= title:c_stratholme
	}
}

wc_scarlet_stratholme_selected_trigger = {
	wc_scarlet_stratholme_claimant_trigger = yes
	any_in_list = { list = target_titles this = title:d_stratholme }
}

wc_scarlet_stratholme_claim_trigger = {
	wc_scarlet_stratholme_selected_trigger = yes
	any_in_list = { list = target_titles count = 1 always = yes }
}

# Character/title scopes, evaluated before any transfer changes allegiance.
wc_scarlet_defeated_character_trigger = {
	OR = { this = scope:defender is_vassal_or_below_of = scope:defender }
}

wc_scarlet_defeated_holder_trigger = {
	exists = holder
	holder = { wc_scarlet_defeated_character_trigger = yes }
}

wc_scarlet_northrend_refuge_trigger = {
	exists = title:h_scourge.holder
	title:h_scourge.holder = {
		is_alive = yes
		is_landed = yes
		is_imprisoned = no
		NOT = { this = scope:wc_retreating_kelthuzad }
		NOT = { this = scope:attacker }
		NOT = { is_vassal_or_below_of = scope:attacker }
		exists = capital_province
		capital_province = { geographical_region = world_northrend }
	}
}
