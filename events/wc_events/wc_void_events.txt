namespace = wc_void

wc_void.100 = { # Magister Umbric proposes a solution for mana addiction
	type = character_event
	content_source = dlc_GOA
	#window = fullscreen_event
	title = wc_void.100_TITLE
	desc = wc_void.100_DESC
	theme = plague
	left_portrait = { 
		character = root
		animation = shocked
	}
	right_portrait = { 
		character = character:magister_umbric
		animation = personality_zealous
	}
	trigger = { 
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_scourge_quelthalas_decision
			}
		}
	}
	
	# Accept the solution
	option = { 
		name = wc_void.100.a
		custom_tooltip = wc_void.100.a_tt
	}
	
	# Outlaw Umbric's research
	option = { 
		name = wc_void.100.b
		custom_tooltip = wc_void.100.b_tt
	}
}

wc_void.101 = { # Creation of the Darkwell
	type = character_event
	content_source = dlc_GOA
	#window = fullscreen_event
	title = wc_void.101_TITLE
	desc = wc_void.101_DESC
	theme = plague
	left_portrait = { 
		character = root
		animation = personality_zealous
	}
	option = { 
		name = wc_void.101.a
		# go ahead with the corruption and summon Kel'thuzad
		trigger_event = wc_void.102
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:resurrect_kelthuzad_decision
			}
		}
	}
}

wc_void.102 = { # Creation of the Darkwell
	type = character_event
	content_source = dlc_GOA
	#window = fullscreen_event
	title = wc_void.102_TITLE
	desc = wc_void.102_DESC
	theme = plague
	left_portrait = { 
		character = root
		animation = personality_vengeful
	}
	immediate = { 
		title:c_sunwell_plateau.title_province = {
			remove_building = sunwell_01
			add_special_building = sunwell_corrupted_void
		}
	}
	option = { 
		name = wc_void.102.a

		hidden_effect = { # Turns all High Elves and Blood Elves into Void Elves
			every_living_character = {
                limit = {
                    location = { geographical_region = world_eastern_kingdoms_quelthalas }
                    OR = {
						has_culture = culture:high_elf
						has_culture = culture:blood_elf
					}
                }
                set_culture = culture:void_elf
				set_faith = faith:voidfarer
            }
		}
		every_county_in_region = {
			region = world_eastern_kingdoms_quelthalas
			hidden_effect = {
				if = {
					limit = {
						culture = { has_cultural_pillar = heritage_thalassian }
					}
					set_county_culture = culture:void_elf
					set_country_faith = faith:voidfarer
				}
			}
		}
	}
}
