namespace = invasionofkalimdor

#### invasionofkalimdor.0001 -> scope out frostwolf clan
#### invasionofkalimdor.0002 -> thrall has a bad dream
#### invasionofkalimdor.0003 -> scope out warsong clan, thrall and the prophet have a nice chit chat
#### invasionofkalimdor.0004 -> grommash consider's the prospect of going to kalimdor
#### invasionofkalimdor.0005 -> Warsong and Frostwolf go together
#### invasionofkalimdor.0006 -> Frostwolf goes all alone
#### invasionofkalimdor.0007 -> Transfer Grommash and Thrall to Kalimdor
#### invasionofkalimdor.0008 -> Transfer Thrall to Kalimdor
#### invasionofkalimdor.0009 -> Landing Notification event for Grommash and Thrall
#### invasionofkalimdor.0010 -> Landing Notification event for Thrall
#### invasionofkalimdor.0100 -> Thrall will get k_frostwolf_clan again and convert culture to Durotan
#### invasionofkalimdor.0101 -> Grommmash will get k_warsong_clan again

invasionofkalimdor.0001 = {
    scope = none
    hidden = yes

    immediate = {
        if = {
            limit = {
                exists = title:k_frostwolf_clan
                title:k_frostwolf_clan.holder = { has_culture = culture:frostwolf }
            }
            title:k_frostwolf_clan.holder = { save_scope_as = thrall_or_frostwolf_leader }
        }
        else_if = {
            limit = {
                any_ruler = {
                    has_culture = culture:frostwolf
                    this = culture.culture_head
                }
            }
            random_ruler = {
                limit = {
                    has_culture = culture:frostwolf
                    this = culture.culture_head
                }
                save_scope_as = thrall_or_frostwolf_leader
            }
        }
        else_if = {
            limit = {
                any_ruler = { has_culture = culture:frostwolf }
            }
            random_ruler = {
                limit = { has_culture = culture:frostwolf }
                save_scope_as = thrall_or_frostwolf_leader
            }
        }
        else = {
            limit = { any_living_character = { is_thrall_trigger = yes } }
            character:10021 = { save_scope_as = thrall_or_frostwolf_leader }
        }

        scope:thrall_or_frostwolf_leader = {
            trigger_event = { id = invasionofkalimdor.0002 days = 1 }
        }

        hidden_effect = {
			add_to_global_variable_list = {
				name = warcraft_invaders_story
				target = flag:orcish_migration_story_started
			}
		}
    }
}

invasionofkalimdor.0003 = {
    type = character_event
    content_source = dlc_GOA
    title = invasionofkalimdor_0003_title
    desc = invasionofkalimdor_0003_desc
    theme = horde

    immediate = {
        if = {
            limit = {
                exists = title:k_warsong_clan
                title:k_warsong_clan.holder = { has_culture = culture:warsong }
            }
            title:k_warsong_clan.holder = { save_scope_as = grommash_or_warsong_leader }
        }
        else_if = {
            limit = {
                any_ruler = {
                    has_culture = culture:warsong
                    this = culture.culture_head
                }
            }
            random_ruler = {
                limit = {
                    has_culture = culture:warsong
                    this = culture.culture_head
                }
                save_scope_as = grommash_or_warsong_leader
            }
        }
        else_if = {
            limit = {
                any_ruler = { has_culture = culture:warsong }
            }
            random_ruler = {
                limit = { has_culture = culture:warsong }
                save_scope_as = grommash_or_warsong_leader
            }
        }
        else_if = {
            limit = { any_living_character = { is_grommash_trigger = yes } }
            if = {
                limit = { character:10200 = { is_alive = yes } }
                character:10200 = { save_scope_as = grommash_or_warsong_leader }
            }
            else = {
                limit = { global_var:grommash = { is_alive = yes } }
                global_var:grommash = { save_scope_as = grommash_or_warsong_leader }
            }
        }
        else = { # Grommash doesn't even spawn most of the times in 583
            create_character = {
                template = grommash_character_template
                name = "Grommash"
                location = title:c_alterac_valley.title_province
                father = character:golmash
                save_scope_as = grommash_or_warsong_leader
            }
        }
    }

    left_portrait = {
        character = scope:thrall_or_frostwolf_leader
        animation = worry
    }
    right_portrait = {
        character = character:5000
        animation = personality_zealous
    }

    option = { # You try to contact the leader of the warsong clan
        trigger = { exists = scope:grommash_or_warsong_leader }
        name = invasionofkalimdor_0003_a
        custom_tooltip = {
            text = you_will_talk_to_the_warsong_leader
            scope:grommash_or_warsong_leader = {
                create_artifact_gorehowl_effect = { OWNER = this }
                trigger_event = {
                    id = invasionofkalimdor.0004
                    days = 3
                }
            }
        }
        ai_chance = {
            base = 100
        }

        following_lore_path_effect = yes
    }
    option = {
        trigger = { NOT = { exists = scope:grommash_or_warsong_leader } }
        name = invasionofkalimdor_0003_a
        scope:thrall_or_frostwolf_leader = {
            trigger_event = {
                id = invasionofkalimdor.0006
                days = 7
            }
        }
        ai_chance = {
            base = 100
        }

        following_lore_path_effect = yes
    }
    option = { # You stay in Azeroth. Your home shall be here and now!
        name = invasionofkalimdor_0003_b
        custom_tooltip = {
            text = this_end_the_story_content_for_orcs
        }
        ai_chance = {
            base = 0
        }

        following_alternate_path_effect = yes
    }
}

invasionofkalimdor.0007 = {
    type = character_event
	content_source = dlc_GOA
    hidden = yes

    immediate = {
        give_land_to_settler = { TO_WHO = scope:thrall_or_frostwolf_leader COUNTY_NAME = title:c_bladefist GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = yes }
        give_land_to_settler = { TO_WHO = scope:grommash_or_warsong_leader COUNTY_NAME = title:c_drygulch GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = yes }

        hidden_effect = {
			get_rid_of_all_titles_in_region_effect = { REGION = world_eastern_kingdoms WHO_GIVES = scope:thrall_or_frostwolf_leader }
            get_rid_of_all_titles_in_region_effect = { REGION = world_eastern_kingdoms WHO_GIVES = scope:grommash_or_warsong_leader }

            scope:thrall_or_frostwolf_leader = {
                if = {
                    #end own wars, leave wars as allies
                    limit = { is_at_war = yes }
                    every_character_war = {
                        limit = { is_war_leader = prev }
                        end_war = white_peace
                    }
                    every_character_war = {
                        limit = { NOT = { is_war_leader = prev } }
                        remove_participant = prev
                    }
                }
            }
            scope:grommash_or_warsong_leader = {
                if = {
                    #end own wars, leave wars as allies
                    limit = { is_at_war = yes }
                    every_character_war = {
                        limit = { is_war_leader = prev }
                        end_war = white_peace
                    }
                    every_character_war = {
                        limit = { NOT = { is_war_leader = prev } }
                        remove_participant = prev
                    }
                }
            }
		}

        scope:thrall_or_frostwolf_leader = {
            trigger_event = { id = invasionofkalimdor.0009 days = 1 }
        }
        scope:grommash_or_warsong_leader = {
            trigger_event = { id = invasionofkalimdor.0009 days = 1 }
        }
    }
}

invasionofkalimdor.0009 = {
    type = character_event
    content_source = dlc_GOA
    title = invasionofkalimdor_0009_title
    desc = invasionofkalimdor_0009_desc
    theme = horde

    override_background = { reference = terrain }

    left_portrait = {
        character = scope:grommash_or_warsong_leader
        animation = personality_bold
    }
    right_portrait = {
        character = scope:thrall_or_frostwolf_leader
        animation = survey
    }

    option = {
        name = invasionofkalimdor_0009_a
        add_prestige = 1000
		add_gold = 300

        title:c_bladefist.duchy = {
			every_de_jure_county_holder = {
				limit = { this != scope:thrall_or_frostwolf_leader }
				top_liege = { add_to_list = potential_war_targets_for_thrall }
			}
		}
        every_in_list = {
			list = potential_war_targets_for_thrall
			scope:thrall_or_frostwolf_leader = {
				start_war = {
					cb = migration_orcs_war
					target = prev.top_liege
					target_title = title:c_bladefist.duchy 
				}
			}
		}

        title:c_drygulch.duchy = {
			every_de_jure_county_holder = {
				limit = { this != scope:grommash_or_warsong_leader }
				top_liege = { add_to_list = potential_war_targets_for_grommash }
			}
		}
        every_in_list = {
			list = potential_war_targets_for_grommash
			scope:grommash_or_warsong_leader = {
				start_war = {
					cb = migration_orcs_war
					target = prev.top_liege
					target_title = title:c_drygulch.duchy 
				}
			}
		}

        if = {
			limit = { is_ai = no } # For Player
			spawn_army = { # Total: 2,500
				uses_supply = no
				inheritable = yes
				name = survivors_of_horde
				men_at_arms = {
					type = savage
					stacks = 10
				}
				men_at_arms = {
					type = blackrock_legionnaire
					stacks = 5
				}
				men_at_arms = {
					type = wolfrider
					stacks = 5
				}
				men_at_arms = {
					type = caster_unit
					stacks = 6
				}
				men_at_arms = {
					type = trebuchet
					stacks = 5
				}
				location = capital_province
				origin = capital_province
			}
		}
		else = {
			limit = { is_ai = yes } # For AI since AI is soo bad at survival
			spawn_army = { # Total: 5,000
				uses_supply = no
				inheritable = yes
				name = survivors_of_horde
				men_at_arms = {
					type = savage
					stacks = 20
				}
				men_at_arms = {
					type = blackrock_legionnaire
					stacks = 10
				}
				men_at_arms = {
					type = wolfrider
					stacks = 10
				}
				men_at_arms = {
					type = caster_unit
					stacks = 12
				}
				men_at_arms = {
					type = trebuchet
					stacks = 10
				}
				location = capital_province
				origin = capital_province
			}
		}

        following_lore_path_effect = yes
    }
}
