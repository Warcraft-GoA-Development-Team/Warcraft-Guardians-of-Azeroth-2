pam_legacy_1 = { # Private Devotions
	legacy = pam_legacy_track

	can_be_picked = { eligible_for_pam_legacy_trigger = yes }

	effect = {
		hidden_effect = {
			dynasty = {
				every_dynasty_member = {
					limit = {
						is_available_quick = {
							advanced_ruler = yes
						}
					}
					change_spiritual_fulfillment = 10
				}
			}
		}
	}

	character_modifier = {
		character_starting_spiritual_fulfillment_add = 10
		character_spiritual_fulfillment_loss_mult = -0.1
	}

	ai_chance = {
		value = 11
		if = {
			limit = {
				like_christianity_religion_trigger = yes
				OR = {
					has_trait = zealous
					has_trait = devoted
				}
			}
			multiply = 10
		}
		if = {
			limit = {
				can_start_new_legacy_track_trigger = no
			}
			multiply = 0
		}
	}
}
