# Scope: actor (character) unless noted. Triggers with "recipient" in the name expect root = recipient.

clerical_chastity_actor_can_demand_chastity_trigger = {
	OR = {
		rite = {
			#rite_has_parameter = gnosticism_cleric_interactions  # Warcraft
			NOT = {
				trait_is_virtue_rite = lustful
			}
		}
		AND = {
			is_clergy = yes
			rite = {
				NOR = {
					trait_is_sin_rite = chaste
					trait_is_virtue_rite = lustful
				}
			}
		}
	}
	is_physically_able_adult = yes
}

clerical_chastity_recipient_realm_borders_actor_trigger = {
	scope:puppet_or_actor = {
		any_sub_realm_county = {
			any_neighboring_county = {
				holder = {
					this = scope:recipient
				}
			}
		}
	}
}

clerical_chastity_recipient_valid_for_actor_scope_trigger = {
	OR = {
		religious_head_or_challenger ?= scope:puppet_or_actor
		scope:puppet_or_actor = {
			highest_held_title_tier >= tier_duchy
		}
		clerical_chastity_recipient_realm_borders_actor_trigger = yes
	}
}

recipient_in_actor_clerical_region_trigger = {
	scope:recipient = {
		OR = {
			AND = {
				is_landed = yes
				capital_county ?= {
					clerical_region_title.holder ?= scope:puppet_or_actor
				}
			}
			AND = {
				is_landed = no
				liege ?= {
					capital_county ?= {
						clerical_region_title.holder ?= scope:puppet_or_actor
					}
				}
			}
		}
	}
}
