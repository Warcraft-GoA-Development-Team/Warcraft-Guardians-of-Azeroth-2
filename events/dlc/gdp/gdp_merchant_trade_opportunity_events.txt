namespace = gdp_merchant_trade_opportunity

##################################################
# GENERAL SCRIPTED TRIGGERS & EFFECTS

# Syncs stress lost from successfully completing a trade deal scheme.
scripted_effect merchant_base_contract_schemes_acknowledge_success_stress_effect = {
	stress_and_fulfillment_impact = { arrogant = minor_stress_impact_loss }
}

# Syncs stress gained from failing a trade deal scheme.
scripted_effect merchant_base_contract_schemes_acknowledge_failure_stress_effect = {
	if = {
		limit = { scope:task_contract_tier >= 2 }
		stress_and_fulfillment_impact = {
			base = miniscule_stress_impact_gain
			humble = minor_stress_impact_loss
			arrogant = minor_stress_impact_gain
		}
	}
}

scripted_effect save_trade_opportunity_contract_scopes_effect = {
	scope:task_contract = {
		task_contract_target ?= { save_scope_as = task_contract_target }
		task_contract_employer ?= { save_scope_as = associate }
		var:task_contract_trade_good ?= { save_scope_as = trade_good_type }
		var:task_contract_trade_category ?= { save_scope_as = trade_good_category }
		task_contract_location = {
			save_scope_as = task_contract_location
			save_scope_as = destination
			barony ?= {
				save_scope_as = task_contract_title
				trade_market = { save_scope_as = local_market }
			}
		}
	}
}

gdp_merchant_trade_opportunity.0001 = { # intro
	type = character_event
	title = gdp_merchant_trade_opportunity.0001.t
	desc = gdp_merchant_trade_opportunity.0001.desc
	theme = trade
	left_portrait = {
		character = root
		animation = pondering
	}
	lower_right_portrait = scope:task_contract_target

	trigger = {
		scope:task_contract = {
			has_task_contract_type = challenge_competitor
			exists = task_contract_target
		}
		scope:task_contract.task_contract_target = { is_alive = yes }
	}

	immediate = {
		save_trade_opportunity_contract_scopes_effect = yes
		start_scheme = {
			type = merchant_challenge_competitor_trade_opportunity_scheme
			contract = scope:task_contract
			target_title = scope:task_contract_target.primary_title
		}
		scope:task_contract.scheme = { save_scope_as = task_contract_scheme } # save scheme
	}

	# Go for a balanced agent mix.
	option = {
		name = gdp_merchant_trade_opportunity.0001.d
		reason = give_me_balanced_agents
		flavor = EVENT_WINDOW_OPTION_AGENTS_BALANCED
		scope:task_contract_scheme ?= {
			# Success.
			add_agent_slot = agent_bailiff
			add_agent_slot = agent_muscle
			add_agent_slot = agent_bookkeeper
			# Speed.
			add_agent_slot = agent_overseer
			add_agent_slot = agent_messenger
		}
		# No stress, we're just selecting our starter batch.
		# AI always selects balanced agents.
		ai_chance = { base = 1000 }
	}

	# Focus on success chance agents.
	option = {
		name = gdp_merchant_trade_opportunity.0001.a
		reason = give_me_success_agents
		flavor = EVENT_WINDOW_OPTION_AGENTS_SUCCESS
		scope:task_contract_scheme ?= {
			# Success.
			add_agent_slot = agent_bailiff
			add_agent_slot = agent_muscle
			add_agent_slot = agent_bookkeeper
			add_agent_slot = agent_scribe
			# Speed.
			add_agent_slot = agent_overseer
		}
		# No stress, we're just selecting our starter batch.
		# AI always picks the balanced config.
	}
	
	# Focus on speedy agents.
	option = {
		name = gdp_merchant_trade_opportunity.0001.b
		reason = give_me_speed_agents
		flavor = EVENT_WINDOW_OPTION_AGENTS_SPEED
		scope:task_contract_scheme ?= {
			# Success.
			add_agent_slot = agent_bailiff
			add_agent_slot = agent_bookkeeper
			# Speed.
			add_agent_slot = agent_overseer
			add_agent_slot = agent_messenger
			add_agent_slot = agent_gabbler
		}
		# No stress, we're just selecting our starter batch.
		# AI always picks the balanced config.
	}
	
	# Now show us the panel.
	after = {
		open_view = {
			view = intrigue_window
			player = root
		}
	}
}

	title = gdp_merchant_trade_opportunity.0002.t
	desc = gdp_merchant_trade_opportunity.0002.desc
	theme = trade
	center_portrait = {
		character = root
		animation = war_over_tie
	}
	right_portrait = {
		character = scope:target_title.holder
		animation = dismissal
	}
	override_background = { reference = sitting_room }

	immediate = {
		merchant_base_trade_opportunity_schemes_set_up_outcome_immediate_effect = yes
	}

	# Diplo
	option = {
		name = gdp_merchant_trade_opportunity.0002.b
		flavor = merchant_base_trade_opportunity_schemes.0000.f.contract_concluded
		trigger = {
			custom_tooltip = {
				text = gdp_merchant_trade_opportunity.0002.b.tt
				OR = {
					has_relation_friend = scope:target_title.holder
					is_spouse_of = scope:target_title.holder
					diplomacy >= high_skill_rating
				}
				NOR = {
					has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:target_title.holder }
					house_has_feud_relation_with_trigger = { TARGET = scope:target_title.holder }
				}
			}
		}
		show_as_unavailable = { always = yes }
		scope:task_contract = { complete_task_contract = success_critical }
		# And wrap things up.
		scope:scheme ?= { end_scheme = yes }
	}

	# Antag
	option = {
		name = gdp_merchant_trade_opportunity.0002.a
		flavor = merchant_base_trade_opportunity_schemes.0000.f.contract_concluded
		scope:task_contract = { complete_task_contract = success_standard }
		# And wrap things up.
		scope:scheme ?= { end_scheme = yes }
	}
}

scripted_effect gdp_seek_debtors_find_debtor_effect = {

		holder ?= {
			gdp_seek_debtors_list_effect = yes
			primary_heir ?= { gdp_seek_debtors_list_effect = yes }
		}
		every_in_de_jure_hierarchy = {
			holder ?= {
				gdp_seek_debtors_list_effect = yes
				primary_heir ?= { gdp_seek_debtors_list_effect = yes }
			}
		}
	}
	random_in_list = {
		list = potential_debtors
		limit = {
			OR = {
				highest_held_title_tier >= tier_county
				AND = {
					is_ruler = no
					primary_heir ?= { highest_held_title_tier >= tier_county }
				}
			}
		}
		alternative_limit = { always = yes }
		weight = {
			base = 1
			modifier = {
				add = {
					value = 1
					add = highest_held_title_tier
					multiply = 100
				}
			}
			modifier = {
				is_ruler = no
				add = {
					value = 1
					ordered_heir_title = {
						order_by = tier
						add = tier
					}
					multiply = 75
				}
			}
			modifier = {
				exists = capital_county
				add = capital_county.prosperity_level
			}
			modifier = {
				gold < 0
				factor = 4
			}
			modifier = {
				has_trait = improvident
				add = 50
			}
			modifier = {
				has_trait = profligate
				add = 25
			}
			modifier = {
				is_at_war = yes
				add = 50
			}
		}
		save_scope_as = debtor
		save_scope_as = bg_override_char
	}
}
