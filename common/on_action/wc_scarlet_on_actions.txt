
#### Triggers Scarlet Setup - survivors of lordaeron plague get an option to link up
scarlet_setup = {
    effect = {
        if = {
            limit = {
                has_global_variable = scourge_lordaeron_complete
                exists = title:k_order_of_the_silver_hand.holder
                NOT = { has_global_variable = scarlet_event_chain_started }
            }
            title:k_order_of_the_silver_hand.holder = {
                trigger_event = {
                    id = scarlet.1000
                    days = 3 
                }
            }
            set_global_variable = scarlet_event_chain_started
        }
    }
}
