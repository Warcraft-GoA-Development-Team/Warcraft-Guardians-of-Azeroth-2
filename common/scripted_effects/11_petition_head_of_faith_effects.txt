# Setup on_accept runs this and we need actor and recipient
# secondary_recipient is only set for excommunication
# $TYPE$ = petition flag, $HOOK$ = hook send-option, $PIETY_VALUE$ = tier
petition_head_of_faith_prepare_petition_effect = {
	scope:puppet_or_actor = {
		petition_head_of_faith_clear_all_effect = yes
		set_variable = {
			name = pending_petition_head_of_faith_type
			value = flag:$TYPE$
		}
		set_variable = {
			name = pending_petition_head_of_faith_recipient
			value = scope:recipient
		}
		if = {
			limit = { exists = scope:secondary_recipient }
			set_variable = {
				name = pending_petition_head_of_faith_target_character
				value = scope:secondary_recipient
			}
		}
		if = {
			limit = { scope:$HOOK$ = yes }
			add_character_flag = pending_petition_head_of_faith_use_hook
		}
		else = {
			#peace of god makes excommunicating free against aggressors
			if = {
				limit = {
					NAND = {
						exists = scope:excom_request
						rite = {
							rite_has_parameter = tenet_peace_of_god_excommunication
						}
						scope:puppet_or_actor = {
							has_opinion_modifier = {
								modifier = declared_war
								target = scope:secondary_recipient
							}
						}
					}
				}
				set_variable = {
					name = pending_petition_head_of_faith_reserved_piety_value
					value = $PIETY_VALUE$
				}
				add_piety = {
					value = 0
					subtract = $PIETY_VALUE$
				}
			}
		}
		petition_head_of_faith_block_recipient_effect = yes
		petition_head_of_faith_begin_travel_effect = yes
	}
}

# We save scope:petition_gravest_sin, because repentance uses scope:actor instead of petition_target
petition_head_of_faith_save_gravest_sin_effect = {
	if = {
		limit = { petition_head_of_faith_target_net_sinful_trigger = yes }
		scope:petition_target = {
			ordered_character_trait = {
				limit = { petition_head_of_faith_trait_sin_value < 0 }
				order_by = petition_head_of_faith_trait_sin_magnitude_value
				save_scope_as = petition_gravest_sin
			}
		}
	}
	else_if = {
		limit = { petition_head_of_faith_target_has_grave_sins_trigger = yes }
		scope:petition_target = {
			ordered_character_trait = {
				limit = {
					OR = {
						trait_is_criminal_in_rite_trigger = {
							TRAIT = this
							RITE = scope:recipient.rite
							GENDER_CHARACTER = scope:petition_target
						}
						AND = {
							scope:petition_target = { has_trait = murderer }
							NOT = {
								scope:recipient.rite = {
									trait_is_virtue_rite = murderer
								}
							}
						}
						#scope:petition_target = { has_trait = despoiler_of_byzantium } # Warcraft - trait removed
					}
				}
				order_by = petition_head_of_faith_trait_sin_magnitude_value
				save_scope_as = petition_gravest_sin
			}
		}
	}
	else_if = {
		limit = { petition_head_of_faith_actor_net_sinful_trigger = yes }
		scope:actor = {
			ordered_character_trait = {
				limit = { petition_head_of_faith_trait_sin_value < 0 }
				order_by = petition_head_of_faith_trait_sin_magnitude_value
				save_scope_as = petition_gravest_sin
			}
		}
	}
}

# scholastic_debate argument (last resort). Learning duel vs Head of Faith. Reserved piety forfeited on failure (refunded on withdraw only). Petitioner scope.

	add_character_flag = petition_head_of_faith_argument_learning
	duel = {
		skill = learning
		target = var:petition_head_of_faith_recipient
		# Base 10/10 = 50% at equal skill, neutral disposition
		10 = {
			desc = petition_head_of_faith_duel_success
			compare_modifier = {
				value = scope:duel_value
				multiplier = 1
			}
			modifier = {
				top_participant_group:the_christian_church ?= {
					has_participant_group_parameter = the_christian_church_more_likely_to_accept_request_excommunications
				}
				add = 5
			}
			# Virtuous/sinful traits modify duel chance.
			modifier = {
				num_virtuous_traits >= 1
				add = 4
			}
			modifier = {
				num_virtuous_traits >= 2
				add = 4
			}
			modifier = {
				num_sinful_traits >= 1
				add = -4
			}
			modifier = {
				num_sinful_traits >= 2
				add = -4
			}
			modifier = {
				petition_head_of_faith_disposition_favorable_trigger = yes
				add = 5
			}
			modifier = {
				petition_head_of_faith_disposition_skeptical_trigger = yes
				add = -3
			}
			modifier = {
				petition_head_of_faith_disposition_hostile_trigger = yes
				add = -8
			}
			min = 1
			petition_head_of_faith_excommunication_scholastic_debate_success_effect = yes
		}
		10 = {
			desc = petition_head_of_faith_argument_dismissed
			compare_modifier = {
				value = scope:duel_value
				multiplier = -1
			}
			petition_head_of_faith_excommunication_affront_failure_effect = yes
		}
	}
}

# We copy declaration_of_repentance_interaction_effect, because those scopes are absent here

	# Petitioner scope, not root, since root is the Head of Faith
	# Whoever calls this consumes the hook, whether by invoking it or by a human Head of Faith granting it
	save_temporary_scope_as = petition_hof_petitioner
	if = {
		limit = {
			NOR = {
				has_character_flag = petition_head_of_faith_use_hook
				rite ?= { rite_has_parameter = tenet_purgatory_absolution_costs_less_prestige }
			}
		}
		# Without hook (Walk to Canossa): massive prestige loss and prestige level drop.
		add_prestige = massive_prestige_loss
		add_prestige_level = -1
	}
	else_if = {
		limit = { NOT = { has_character_flag = petition_head_of_faith_use_hook } }
		add_prestige = major_prestige_loss
	}
	remove_trait = excommunicated
	clear_excommunication_tracking_effect = yes
	add_character_modifier = {
		modifier = excommunication_recently_lifted
		years = 10
	}
	stress_and_fulfillment_impact = {
		cynical = minor_stress_impact_gain
	}
	var:petition_head_of_faith_recipient = {
		add_prestige = major_prestige_gain
		# repentant_opinion only for AI Head of Faith; human players choose their own response.
		if = {
			limit = { is_ai = yes }
			add_opinion = {
				modifier = repentant_opinion
				target = prev
				opinion = 10
			}
		}
	}
	add_clan_unity_interaction_effect = {
		CHARACTER = scope:petition_hof_petitioner
		TARGET = var:petition_head_of_faith_recipient
		VALUE = medium_unity_gain
		DESC = clan_unity_repentance.desc
		REVERSE_NON_HOUSE_TARGET = no
	}
}

# Decision confirm only, before a target is chosen. Matches disposition triggers when no target exists.
petition_head_of_faith_display_recipient_disposition_effect = {
	if = {
		limit = {
			scope:petition_recipient = {
				opinion = {
					target = scope:petitioner
					value >= petition_head_of_faith_favorable_opinion_threshold_value
				}
			}
		}
		custom_tooltip = petition_head_of_faith_decision_receptive_favorable
	}
	else_if = {
		limit = {
			scope:petition_recipient = {
				OR = {
					has_relation_rival = scope:petitioner
					opinion = {
						target = scope:petitioner
						value <= petition_head_of_faith_hostile_opinion_threshold_value
					}
				}
			}
		}
		custom_tooltip = petition_head_of_faith_decision_receptive_hostile
	}
	else_if = {
		limit = {
			scope:petition_recipient = {
				opinion = {
					target = scope:petitioner
					value <= petition_head_of_faith_skeptical_opinion_threshold_value
				}
			}
		}
		custom_tooltip = petition_head_of_faith_decision_receptive_skeptical
	}
	else = {
		custom_tooltip = petition_head_of_faith_decision_receptive_neutral
	}
}

	}
