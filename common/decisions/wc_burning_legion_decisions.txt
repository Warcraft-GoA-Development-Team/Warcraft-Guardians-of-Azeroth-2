summon_burning_legion_decision = { # By Ikarimonogatari
	picture = { reference = "gfx/interface/illustrations/decisions/wc_summon_burning_legion.dds" }
	decision_group_type = major

	ai_check_interval = 1

	desc = summon_burning_legion_decision_desc

	is_shown = {
		OR = {
			AND = { # For arthas
				scourge_quelthalas_frostmourne_check_trigger = yes
				is_landed = yes
			}
			trigger_if = {
				limit = { exists = title:h_scourge }
				is_primary_heir_of = title:h_scourge.holder # If arthas expedition fails to trigger
			}
		}
		game_start_date < 605.6.6

		is_target_in_global_variable_list = { # Only make this available before kelthuzad is spawned
			name = unavailable_unique_decisions
			target = flag:resurrect_kelthuzad_decision
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:summon_burning_legion_decision
			}
		}
	}

	is_valid = {
		OR = {
			AND = { # For arthas
				scourge_quelthalas_frostmourne_check_trigger = yes
				is_landed = yes
			}
			trigger_if = {
				limit = { exists = title:h_scourge }
				custom_tooltip = {
					text = primary_heir_of_lich_king
					is_primary_heir_of = title:h_scourge.holder # If arthas expedition fails to trigger
				}
			}
		}
		any_courtier = {
			has_trait = creature_lich
		}
	}
	

	cost = {
		prestige = 100
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = {
			text = summon_burning_legion_decision.tt
			if = { # If the Lich King or a fellow vassal holds it, give the title to Arthas
				limit = {
					OR = {
						title:c_dalaran.holder = { has_title = title:h_scourge }
						title:c_dalaran.holder = { is_vassal_of = title:h_scourge.holder }
					}
				}
				trigger_event = burninglegion.0001
			}
			else_if = { # Invade Dalaran
				limit = { NOT = { title:c_dalaran.holder = root } }
				trigger_event = burninglegion.0002
			}
			else = { # Start searching for the Book of Medivh and Summon the Legion
				limit = { title:c_dalaran.holder = root }
				trigger_event = burninglegion.0003
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}

embrace_felblood_decision = { # By Ikarimonogatari
	picture = { reference = "gfx/interface/illustrations/decisions/embrace_felblood.dds" }
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 12
		duchy = 24
		kingdom = 36
		empire = 0 # Will only do it if you are weak and desperate
		hegemony = 0 # Will only do it if you are weak and desperate
	}

	desc = embrace_felblood_decision_desc

	is_shown = {
		has_trait = creature_orc
		OR = {
			has_faith = faith:burning_legion_religion
			has_faith = faith:orcish_fel
			has_faith = faith:black_harvest
		}
		NOT = { has_trait = creature_fel_orc }
	}

	is_valid = {
		has_trait = creature_orc
		custom_tooltip = {
			text = you_are_a_fel_religion_fangirl
			OR = {
				has_faith = faith:burning_legion_religion
				has_faith = faith:orcish_fel
				has_faith = faith:black_harvest
			}
		}
		OR = {
			custom_tooltip = {
				text = a_non_demon_hunter_demon_is_your_prisoner
				any_prisoner = {
					NOT = { has_trait = demon_hunter } # Demon Hunters are not REAL demons
					has_trait = being_demon
					is_imprisoned_by = root
				}
			}
			custom_tooltip = {
				text = a_non_demon_hunter_demon_is_your_liege
				is_independent_ruler = no
				OR = {
					liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
					top_liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
				}
			}
		}
	}

	cost = {
		prestige = 500
		piety = 1000
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
	}

	effect = {
		hidden_effect = {
			remove_trait = creature_orc
		}
		if = {
			limit = {
				is_independent_ruler = no
				OR = {
					liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
					top_liege = {
						has_trait = being_demon
						NOT = { has_trait = demon_hunter }
					}
				}
			}
			custom_tooltip = {
				text = demon_liege_tt
				trigger_event = felblood.0001
			}
		}
		else = {
			limit = {
				any_prisoner = {
					has_trait = being_demon
					NOT = { has_trait = demon_hunter }
					is_imprisoned_by = root
				}
			}
			custom_tooltip = {
				text = demon_prisoner_tt
				trigger_event = felblood.0002
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
	}

	ai_will_do = {
		base = 25
		modifier = {
			add = 50
			top_liege = { has_trait = being_demon }
		}
		modifier = {
			add = 25
			liege = { has_trait = being_demon }
		}
		modifier = {
			add = 25
			any_prisoner = { has_trait = being_demon }
		}
		modifier = {
			add = 25
			has_trait = ambitious
		}
	}
}
