####################################################
# Faith Rite Growth System                          #
####################################################

# Faiths run a monthly pulse via faith_rite_growth_pulse
# see game/common/on_action/religion_on_actions.txt
# When a faith is below its target rite count and not in cooldown, the pulse rolls into rite_growth.0001.

############################
# 0001 - Picker            #
############################
rite_growth.0001 = {
	hidden = yes
	scope = faith

	immediate = {
		save_scope_as = origin_faith

		# Find the largest-by-counties rite of this faith.
		rite_growth_resolve_source_rite_effect = yes

		if = {
			limit = { exists = scope:source_rite }
			rite_growth_pick_founder_effect = yes

			if = {
				limit = { exists = scope:founder }
				# If the founder is secular and can't create the rite by themselves, save the realm priest.
				scope:founder = {
					if = {
						limit = {
							rite = {
								rite_has_doctrine = doctrine_theocracy_temporal
							}
							is_clergy = no
							cp:councillor_court_chaplain ?= {
								is_ai = yes
								is_clergy = yes
							}
						}
						cp:councillor_court_chaplain = {
							save_scope_as = founder_realm_priest
						}
					}
				}
				
				# Cooldown days scale with how many rites the faith still needs (more needed -> shorter cooldown)
				set_variable = {
					name = rite_growth_recent_spawn
					value = yes
					days = rite_growth_cooldown_days
				}
				# The fallback pulse only waits out this shorter cooldown
				set_variable = {
					name = rite_growth_recent_spawn_fallback
					value = yes
					days = rite_growth_fallback_cooldown_days
				}

				scope:founder = {
					trigger_event = rite_growth.0010
				}
			}
		}
	}
}

# 0011 - Recursive Sweep   #
############################
rite_growth.0011 = {
	hidden = yes
	scope = faith

	# random_faith_ruler and change_variable need an explicit faith scope even when root is the faith
	trigger = {
		exists = scope:origin_faith
		exists = scope:source_rite
		exists = scope:new_rite
		exists = scope:founder
		scope:new_rite = {
			var:rite_growth_remaining_converts > 0
		}
	}

	immediate = {
		clear_saved_scope = convert_ruler

		scope:origin_faith = {
			random_faith_ruler = {
				limit = {
					is_eligible_rite_growth_target_trigger = yes
					rite_growth_in_founder_spread_area_trigger = yes
				}

				weight = {
					base = 100

					# Further capitals are less likely.
					modifier = {
						add = -50
						exists = scope:founder.capital_county
						capital_county = {
							squared_distance = {
								target = scope:founder.capital_county
								value > squared_distance_small
							}
						}
					}
					modifier = {
						add = -25
						exists = scope:founder.capital_county
						capital_county = {
							squared_distance = {
								target = scope:founder.capital_county
								value > squared_distance_medium
							}
						}
					}

					modifier = {
						factor = 10
						any_held_title = { has_clerical_region = yes }
					}

					modifier = {
						factor = 10
						highest_held_title_tier = tier_kingdom
					}

					modifier = {
						factor = 10
						liege.rite ?= scope:new_rite
					}

					modifier = {
						factor = 10
						faith = { always = no } # Warcraft - no GoA doctrine has this parameter
						exists = capital_county.clerical_region_title
						scope:founder.capital_county.clerical_region_title ?= capital_county.clerical_region_title
					}

					# Cultural affinity helps.
					modifier = {
						factor = 2
						culture = { has_same_culture_heritage = scope:founder.culture }
					}
					modifier = {
						factor = 3
						has_same_culture_as = scope:founder
					}

					# Theological fitness traits
					modifier = {
						factor = 1.5
						OR = {
							has_trait = theologian
							has_trait = lifestyle_scholar
							has_trait = lifestyle_mystic
							has_trait = zealous
						}
					}

					# Targets that bring several source-rite counties along stretch the budget further.
					modifier = {
						factor = 2
						any_held_title = {
							tier = tier_county
							rite = scope:source_rite
							count >= 2
						}
					}

					# Counts spread the rite less than dukes/kings.
					modifier = {
						factor = 0.25
						highest_held_title_tier = tier_county
					}

					# Avoid having multiple large kings converting
					modifier = {
						factor = 0.01
						scope:origin_faith = {
							any_faith_ruler = {
								highest_held_title_tier >= tier_kingdom
								sub_realm_size >= 25
								rite = scope:new_rite
							}
						}
					}

					# Nomads really aren't a good target for this
					modifier = {
						factor = 0.001
						OR = {
							government_has_flag = government_is_nomadic
							government_has_flag = government_is_herder
						}
					}
				}

				save_scope_as = convert_ruler
			}
		}

		if = {
			limit = {
				scope:convert_ruler ?= {
					is_alive = yes
					rite != scope:new_rite
				}
			}

			scope:convert_ruler = {
				rite_growth_convert_ruler_to_new_rite_effect = yes
			}

			# Higher-tier converts cost more so a kingdom isn't drowned by county pickups
			scope:new_rite = {
				# A successful pick resets the miss allowance below.
				remove_variable = rite_growth_sweep_retries_left
				if = {
					limit = {
						scope:convert_ruler = {
							highest_held_title_tier >= tier_kingdom
						}
					}
					change_variable = {
						name = rite_growth_remaining_converts
						subtract = 3
					}
				}
				else_if = {
					limit = {
						scope:convert_ruler = {
							highest_held_title_tier = tier_duchy
						}
					}
					change_variable = {
						name = rite_growth_remaining_converts
						subtract = 2
					}
				}
				else = {
					change_variable = {
						name = rite_growth_remaining_converts
						subtract = 1
					}
				}

				if = {
					limit = { var:rite_growth_remaining_converts > 0 }
					scope:origin_faith = {
						trigger_event = {
							id = rite_growth.0011
							days = { 7 14 }
						}
					}
				}
				else = {
					remove_variable = rite_growth_remaining_converts
				}
			}

			# Discreet notification to faith-mates living near the converted ruler.
			if = {
				limit = { scope:convert_ruler ?= { rite = scope:new_rite } }
				every_player = {
					limit = {
						faith = scope:origin_faith
						NOR = {
							this = scope:convert_ruler
							this = scope:founder
						}
						capital_county ?= {
							squared_distance = {
								target = scope:convert_ruler.capital_county
								value <= squared_distance_medium
							}
						}
					}
					send_interface_message = {
						type = msg_pam_rite_growth_spread
						title = rite_growth.0011.notification.title
						desc = rite_growth.0011.notification.desc
						left_icon = scope:convert_ruler
						right_icon = scope:new_rite
						show_as_tooltip = {
							scope:convert_ruler = {
								set_character_rite = scope:new_rite
							}
						}
					}
				}
			}
		}
		else = {
			# Misses retry a couple of times while budget remains rather than ending the spread
			scope:new_rite = {
				if = {
					limit = {
						NOT = { has_variable = rite_growth_sweep_retries_left }
					}
					set_variable = {
						name = rite_growth_sweep_retries_left
						value = 2
					}
				}
				if = {
					limit = { var:rite_growth_sweep_retries_left > 0 }
					change_variable = {
						name = rite_growth_sweep_retries_left
						subtract = 1
					}
					scope:origin_faith = {
						trigger_event = {
							id = rite_growth.0011
							days = { 30 60 }
						}
					}
				}
				else = {
					remove_variable = rite_growth_sweep_retries_left
					remove_variable = rite_growth_remaining_converts
				}
			}
		}
	}
