# Great Holy War effects for GoA (issue #1973)

# Starts the Age of Crusades for the head's religion if needed, then swaps scope:removed_tenet
# for the faith's crusade tenet (Aldor: Pacifism for Communion).
# Root: the proclaiming head of faith. Needs scope:removed_tenet.
# Called from wc_proclaim_holy_war_decision (tooltip) and wc_holy_war.0001 (applies it).
wc_proclaim_holy_war_swap_effect = {
	religion = {
		if = {
			limit = {
				NOT = { exists = var:variable_ghw_unlocked }
			}
			set_variable = {
				name = variable_ghw_unlocked
				value = yes
			}
			set_variable = { # Same wait vanilla great_holy_war.0010 gives before the first GHW
				name = variable_first_ghw_cooldown
				value = yes
				days = 120
			}
		}
	}

	if = {
		limit = {
			faith = { wc_holy_war_armed_pilgrimages_faith_trigger = yes }
		}
		faith.main_rite = {
			remove_rite_tenet = scope:removed_tenet
			add_rite_tenet = tenet_armed_pilgrimages
		}
		faith = {
			if = {
				limit = { has_tenet_popularity_trigger = yes }
				change_tenet_popularity = {
					target = scope:removed_tenet
					change = major_tenet_popularity_loss
				}
				change_tenet_popularity = {
					target = tenet:tenet_armed_pilgrimages
					change = major_tenet_popularity_gain
				}
			}
		}
	}
	else_if = {
		limit = {
			faith = { wc_holy_war_struggle_faith_trigger = yes }
		}
		faith.main_rite = {
			remove_rite_tenet = scope:removed_tenet
			add_rite_tenet = tenet_struggle_submission
		}
	}
	else_if = {
		limit = { faith = faith:aldor }
		faith.main_rite = {
			remove_rite_tenet = scope:removed_tenet
			add_rite_tenet = tenet_communion
		}
		faith = {
			if = {
				limit = { has_tenet_popularity_trigger = yes }
				change_tenet_popularity = {
					target = scope:removed_tenet
					change = major_tenet_popularity_loss
				}
				change_tenet_popularity = {
					target = tenet:tenet_communion
					change = major_tenet_popularity_gain
				}
			}
		}
	}
}
