defeat_of_the_champion_effect = {	
	make_story_owner = scope:lich_king

	scope:lich_king = { trigger_event = wc_path_of_the_damned.0100 }
}

find_tichondrius_effect = {
	find_nerzhul_effect = yes

	scope:nerzhul = {
		every_relation = {
			type = scourge_jailor
			add_to_temporary_list = valid_tichondrius
		}
	}
	ordered_in_list = {
		limit = { is_ai = yes }
		list = valid_tichondrius
		order_by = find_tichondrius_value
		max = 1
			
		save_scope_as = tichondrius
	}
	if = {
		limit = { NOT = { exists = scope:tichondrius } }

		create_character = {
			template = tichondrius_character_template
			employer = scope:nerzhul
				
			after_creation = { save_scope_as = tichondrius }
		}
	}
}

find_nerzhul_effect = {
	title:h_scourge.holder = { save_scope_as = nerzhul }
}

turn_urn_into_phylactery_effect = {
	$PREV_INSIDE$ = { save_scope_as = prev_inside }
	$INSIDE$ = { save_scope_as = inside }

	reforge_artifact = {
		modifier = phylactery_modifier
		visuals = phylactery_of_lich
	}
	set_artifact_name = phylactery_name
	set_artifact_description = phylactery_urn_of_king_of_lordaeron_desc
}
