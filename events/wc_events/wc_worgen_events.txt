worgen.0002 = {
	type = character_event
	content_source = dlc_GOA
	title = worgen.0002.t
	desc = {
		desc = worgen.0002.desc
	}
	theme = skull

	immediate = {
		ROOT = {
			random_realm_county = {
				limit = {
					any_realm_province = {
						any_province_epidemic = {
							epidemic_type.epidemic_trait = trait:worgen_bite
						}
					}
				}

				save_scope_as = infected_county_scope

				add_county_modifier = { 
					modifier = wc_mysterious_disappearances_modifier 
					years = 2
				}
			}
		}
	}

	override_background = {
		reference = wc_gilneas
	}

	left_portrait = {
		character = scope:infected_county_scope.holder
		animation = fear
	}


	option = {
		name = {
			text = worgen.0002.option
		}
		add_stress = 10
		scope:infected_county_scope.holder = {
			add_stress = 15
		}
	}

}
