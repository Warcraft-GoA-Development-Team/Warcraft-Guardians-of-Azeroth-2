story_scourge_champion_search = {
	on_setup = {
		set_global_variable = { # Save the story globally
			name = scourge_champion_search_story
			value = this 
		}

		# Set base variables
		set_variable = { # Terenas
			name = plagued_king
			value = global_var:plagued_king
		}
		remove_global_variable = plagued_king

		set_variable = { # Andorhal
			name = plagued_province
			value = global_var:plagued_province
		}
		remove_global_variable = plagued_province

		set_variable = { # Hearthglen
			name = big_city_1
			value = global_var:big_city_1
		}
		remove_global_variable = big_city_1

		set_variable = { # Stratholme
			name = big_city_2
			value = global_var:big_city_2
		}
		remove_global_variable = big_city_2

		var:plagued_king = { find_arthas_effect = yes }

		# Find knights for the future champion
		var:future_champion = {
			if = {
				limit = { is_ai = yes }
				add_character_flag = { flag = wc_cannot_serve_in_army years = 1 }
			}
			else = { add_character_flag = { flag = wc_cannot_serve_in_army weeks = 10 } }
			trigger_event = wc_arthas_story.9
		}
	}

	on_end = {
		# Start Path of the Damned here
		# Then remove the variable
	}

	on_owner_death = {
		end_story = yes
	}
}
