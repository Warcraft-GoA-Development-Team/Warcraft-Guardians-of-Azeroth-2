on_faction_demand_accepted = { on_actions = { wc_nathrezim_cover_concessions } }

quarterly_playable_pulse = { on_actions = { wc_nathrezim_public_pulse } }

wc_nathrezim_public_pulse = {
	trigger = { is_alive = yes has_variable = wc_nathrezim_public_nature }
	effect = { wc_nathrezim_refresh_public_opposition_effect = yes }
}

wc_nathrezim_cover_concessions = {
	trigger = {
		wc_nathrezim_cover_active_trigger = yes
		has_variable = wc_nathrezim_cover_crisis
		exists = scope:faction
		scope:faction = { has_variable = wc_nathrezim_ultimatum_sent }
	}
	effect = {
		save_scope_as = wc_cover_body
		scope:faction = {
			every_faction_member = {
				scope:wc_cover_body = { add_to_variable_list = { name = wc_nathrezim_settled_witnesses target = prev } }
			}
		}
		remove_variable = wc_nathrezim_cover_investigator
	}
}
