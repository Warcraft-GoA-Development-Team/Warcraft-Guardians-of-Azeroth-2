# Warcraft
on_game_start_after_lobby = {
	on_actions = {
		on_game_start_quelthalas
	}
}

yearly_global_pulse = { 
	on_actions = { inert_sunwell_check } 
}

on_game_start_quelthalas = {
    effect = {
        title:e_quelthalas.holder ?= {
            trigger_event = { id = wc_quelthalas_events.1 }
        }
    }
}

inert_sunwell_check = {
    trigger = {
        title:b_sunwell_plateau.title_province = { has_building = sunwell_inert }
	}
    effect = {
        title:e_quelthalas.holder ?= {
            trigger_event = { id = wc_quelthalas_events.101 }
        }
		set_global_variable = sunwell_chain_started
    }
}
