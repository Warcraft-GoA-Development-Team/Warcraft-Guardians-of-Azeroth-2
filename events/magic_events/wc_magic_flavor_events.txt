# 0010 - Someone is selling a mana potion
# 0020 - You find mana crystals (leylines locations only)
# 0030 - You find literature on a forbidden magic
# 0040 - Your religion offers you insight on magic

scripted_effect get_potion_effect_0010 = {
    random_list = {
        50 = {
            desc = wc_magic_flavor_events.0010.a_good
            show_chance = no
            modifier = {
                add = 25
                scope:merchant = { has_trait = honest }
            }
            send_interface_toast = {
                title = wc_magic_flavor_events.0010.a_good
                type = event_toast_effect_good
                left_icon = root
                right_icon = scope:merchant
                wc_change_mana_by_value_effect = {
                    CHANGE = add
                    VALUE = medium_mana_value
                }
            }
        }
        30 = {
            desc = wc_magic_flavor_events.0010.a_bad
            show_chance = no
            modifier = {
                add = 25
                scope:merchant = { has_trait = deceitful }
            }
            modifier = {
                add = 10
                scope:merchant = {
                    OR = {
                        has_trait = greedy
                        has_trait = avaricious
                    }
                }
            }
            send_interface_toast = {
                title = wc_magic_flavor_events.0010.a_bad
                type = event_toast_effect_neutral
                left_icon = root
                right_icon = scope:merchant
                wc_change_mana_by_value_effect = {
                    CHANGE = add
                    VALUE = tiny_mana_value
                }
            }
        }
        20 = {
            desc = wc_magic_flavor_events.0010.a_terrible
            show_chance = no
            modifier = {
                add = 50
                scope:merchant = { has_trait = deceitful }
            }
            modifier = {
                add = 10
                scope:merchant = {
                    OR = {
                        has_trait = sadistic
                        has_trait = arbitrary
                    }
                }
            }
            send_interface_toast = {
                title = wc_magic_flavor_events.0010.a_terrible
                type = event_toast_effect_bad
                left_icon = root
                right_icon = scope:merchant
                add_stress = minor_stress_gain
                add_character_modifier = {
                    modifier = wc_bad_mana_potion_modifier
                    days = 120
                }
            }
        }
    }
}
# Someone offers to sell you a mana potion
wc_magic_flavor_events.0010 = {
    type = character_event
	content_source = dlc_GOA
    theme = magic
    title = wc_magic_flavor_events.0010.t
    desc = wc_magic_flavor_events.0010.d

    cooldown = { years = 10 }

    trigger = {
        has_variable = wc_current_mana_variable
        var:wc_current_mana_variable <= wc_fourty_percent_of_max_mana_value
    }

    weight_multiplier = {
		base = 25
		modifier = { 
			add = {
				if = {
					limit = { 
						var:wc_current_mana_variable <=  wc_twenty_percent_of_max_mana_value
					}
					add = 50
				}
			}
        }
    }

    immediate = {
        create_character = {
			template = merchant_template
			location = root.location
			random_traits_list = {
                count = 1
                honest = {}
                deceitful = {}
            }
			save_scope_as = merchant
			after_creation = {
				add_character_flag = created
			}
		}
    }

    left_portrait = {
 		character = root
        animation = interested
	}
	right_portrait = {
 		character = scope:merchant
		animation = steward
	}

    # Ok how much is it 
    option = {
        name = wc_magic_flavor_events.0010.a
        trigger = {
            short_term_gold >= medium_gold_value
        }
        show_as_unavailable = { always = yes }
        
        pay_short_term_gold = { gold = medium_gold_value target = scope:merchant }

        get_potion_effect_0010 = yes

        stress_impact = {
            trusting = minor_stress_impact_loss   
            greedy = minor_stress_impact_gain
        }  

        ai_chance = {
            base = 75
            ai_value_modifier = {
                ai_greed = -1
            }
        }
    }

    # Let me examine the potion
    option = {
        name = wc_magic_flavor_events.0010.b

        duel = {
            skill = learning
            value = high_skill_rating
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = 3.5
                }
                custom_tooltip = wc_magic_flavor_events.0010.b_tt
                send_interface_toast = {
                    title = wc_magic_flavor_events.0010.b_tt
                    type = event_toast_effect_good
                    left_icon = root
                    right_icon = scope:merchant
                    wc_change_mana_by_value_effect = {
                        CHANGE = add
                        VALUE = minor_mana_value
                    }
                }
            }
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = -3.5
                }
                send_interface_toast = {
                    title = wc_magic_flavor_events.0010.b_bad
                    type = event_toast_effect_neutral
                    left_icon = root
                    right_icon = scope:merchant
                    add_stress = minor_stress_gain
                }
            }
        }

        stress_impact = {
            paranoid = minor_stress_impact_loss  
            stubborn = minor_stress_impact_loss 
        }  

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_rationality = 1
            }
        }
    }

    # Just give it to me or die
    option = {
        name = wc_magic_flavor_events.0010.c 
        custom_tooltip = wc_magic_flavor_events.0010.c_tt
        add_dread = minor_dread_gain
        add_tyranny = minor_tyranny_gain

        get_potion_effect_0010 = yes

        stress_impact = {
            compassionate = medium_stress_impact_gain
            just = medium_stress_impact_gain
            callous = minor_stress_impact_loss   
            arbitrary = minor_stress_impact_loss   
            sadistic = minor_stress_impact_loss  
        }  

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_sociability = -1
                ai_boldness = 0.5
            }
        } 
    }

    after = {
        hidden_effect = {
            scope:merchant = { silent_disappearance_effect = yes }
        }
    }
}
