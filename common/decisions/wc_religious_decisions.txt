# Loa Decisions

### Separate From Zandalari Religious Control ###
separate_from_zul_head_of_faith_decision = { # Separate From Zandalari Religious Control
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 48
		kingdom = 24
		empire = 24
		hegemony = 24
	}

	title = separate_from_zul_head_of_faith_decision

	desc = separate_from_zul_head_of_faith_decision_desc

	selection_tooltip = separate_from_zul_head_of_faith_decision_tooltip


	is_shown = {
		# Standard filter checks.
		is_landed_or_landless_administrative = yes
		exists = dynasty
		# Check some religious stuff; namely that you're the correct religion and your existing-HoF doesn't share your faith.
		religion = religion:loa_group # Warcraft
		exists = faith.religious_head
		faith != faith.religious_head.faith
		# And filter out anyone who'd make this unreliable.
		OR = {
			government_has_flag = government_is_tribal
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_mechanic = administrative
		}
	}

	is_valid = {
		# Must be independent.
		top_liege = this
		# Should have a decent piety level.
		piety_level >= high_piety_level
		# Can't be the current Zul Prophet
		custom_description = {
			text = separate_from_zul_head_of_faith_decision.cant_be_zul_prophet
			NOT = {
				faith.religious_head = {
					any_liege_or_above = { this = root }
				}
			}
		}
		# Plus, you shouldn't be _too_ sinful, even with your piety level.
		custom_description = {
			text = separate_from_zul_head_of_faith_decision.no_more_than_x_sinful_traits
			num_sinful_traits <= 2
		}
		# Now, pick any three of the valid reasons.
		calc_true_if = {
			amount >= 3
			
			# Being outside of your head's diplo range helps.
			custom_description = {
				text = separate_from_zul_head_of_faith_decision.zul_prophet_not_in_diplo_range
				NOT = { in_diplomatic_range = faith.religious_head }
			}
			# Having a holy site.
			custom_description = {
				text = separate_from_zul_head_of_faith_decision.control_one_holy_site
				faith = {
					any_holy_site = {
						count >= 1
						holder.top_liege = root
					}
				}
			}
			# Having *2* holy sites.
			custom_description = {
				text = separate_from_zul_head_of_faith_decision.control_two_holy_sites
				faith = {
					any_holy_site = {
						count >= 2
						holder.top_liege = root
					}
				}
			}
			# Having a higher-than-minimum piety level.
			piety_level >= very_high_piety_level
			# And just having a Zul Prophet who doesn't live up to the root's faith's ideals.
			custom_description = {
				text = separate_from_zul_head_of_faith_decision.zul_prophet_has_x_or_more_traits_sinful_to_your_faith
				faith.religious_head = {
					num_sinful_traits = {
						value >= 2
						target = root.rite
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		# Has to be around for the ceremony.
		not = { exists = involved_activity }
		has_contagious_deadly_disease_trigger = no
		# Has to be a valid priestly gender for the current faith.
		faith = { has_allowed_gender_for_clergy = root }
	}

	effect = {
		random_vassal_or_below = {
			limit = { is_available_ai_adult = yes }
			save_scope_as = founder
		}
		faith.religious_head = { save_scope_as = zul_prophet }
		show_as_tooltip = { appoint_a_righteous_zul_prophet_scripted_effect = yes }
		# Notify co-faithists; we do this first so we can sort the feed message for how many did/didn't convert.
		faith = {
			every_faith_ruler = {
				limit = {
					highest_held_title_tier >= tier_county
					this != root
				}
				trigger_event = religious_decision.0602
			}
		}
		# Now notify other players.
		every_player = {
			limit = {
				OR = {
					religion = religion:loa_group # Warcraft
					any_neighboring_and_across_water_top_liege_realm_owner = { this = root }
					any_liege_or_above = { religion = religion:loa_group } # Warcraft
				}
				# Exempt same-faith characters, who already got an event.
				faith != root.faith
			}
			# Players who share the faith get
			trigger_event = religious_decision.0603
		}
		# Finally, we trigger the main conversion.
		trigger_event = religious_decision.0601
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					has_treasury = no
				}
				add = 500
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
				}
				add = 500
			}
		}
		piety = 1000
	}

	ai_potential = {
		# Must be independent.
		top_liege = this
		# Should have a decent piety level.
		piety_level >= high_piety_level
		OR = {
			# Zealous people think they know better.
			has_trait = zealous
			# And cynics want that power for themselves.
			has_trait = cynical
		}
	}

	ai_will_do = {
		base = 100
	}
}

# Holy War Decisions

### Proclaim the Age of Holy War ###
# Issue #1973. The head of a roster faith (wc_great_holy_war_triggers.txt) makes its crusade tenet core.
# By God Alone gives the Holy Light Dulia instead of Armed Pilgrimages, so without this it never crusades.
wc_proclaim_holy_war_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	decision_group_type = major
	sort_order = 480

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	desc = wc_proclaim_holy_war_decision_desc
	selection_tooltip = wc_proclaim_holy_war_decision_tooltip
	confirm_text = wc_proclaim_holy_war_decision_confirm

	is_shown = {
		wc_can_proclaim_holy_war_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_imprisoned = no

		custom_tooltip = {
			text = wc_proclaim_holy_war_decision_select_tt
			exists = scope:selected_item
		}
	}

	cost = {
		piety = {
			value = faith_tenet_cost_mid
		}
		treasury_or_gold = { # treasury for treasury governments, gold for everyone else
			value = medium_treasury_or_gold_value
		}
	}

	effect = {
		scope:selected_item = { save_scope_as = removed_tenet }

		if = {
			limit = {
				religion = {
					NOT = { exists = var:variable_ghw_unlocked }
				}
			}
			custom_tooltip = wc_proclaim_holy_war_decision_unlock_tt
		}
		show_as_tooltip = {
			wc_proclaim_holy_war_swap_effect = yes
		}

		trigger_event = wc_holy_war.0001
	}

	widget = {
		gui = "decision_view_widget_select_tenet"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "REPLACE_TENET"

		setup_items = {
			scope:actor = {
				faith.main_rite = {
					ordered_rite_tenet = {
						status = core
						order_by = pam_raw_tenet_popularity_value
						max = 200
						check_range_bounds = no
						add_to_list = item_list
					}
				}
			}
		}

		# Player preselection: the roster's preferred tenet, else any tenet that may go
		default_item = {
			scope:actor = {
				faith.main_rite = {
					if = {
						limit = {
							any_rite_tenet = {
								status = core
								wc_tenet_is_holy_war_ai_drop_trigger = { FAITH = scope:actor.faith }
								wc_tenet_can_be_replaced_for_holy_war_trigger = { FAITH = scope:actor.faith }
							}
						}
						random_rite_tenet = {
							status = core
							limit = {
								wc_tenet_is_holy_war_ai_drop_trigger = { FAITH = scope:actor.faith }
								wc_tenet_can_be_replaced_for_holy_war_trigger = { FAITH = scope:actor.faith }
							}
							save_scope_as = selected_item
						}
					}
					else = {
						random_rite_tenet = {
							status = core
							limit = {
								wc_tenet_can_be_replaced_for_holy_war_trigger = { FAITH = scope:actor.faith }
							}
							save_scope_as = selected_item
						}
					}
				}
			}
		}

		# AI pick (default_item is player-only, see _decisions.info); same rule as above
		ai_select_item = {
			scope:actor = {
				faith.main_rite = {
					if = {
						limit = {
							any_rite_tenet = {
								status = core
								wc_tenet_is_holy_war_ai_drop_trigger = { FAITH = scope:actor.faith }
								wc_tenet_can_be_replaced_for_holy_war_trigger = { FAITH = scope:actor.faith }
							}
						}
						random_rite_tenet = {
							status = core
							limit = {
								wc_tenet_is_holy_war_ai_drop_trigger = { FAITH = scope:actor.faith }
								wc_tenet_can_be_replaced_for_holy_war_trigger = { FAITH = scope:actor.faith }
							}
							save_scope_as = selected_item
						}
					}
					else = {
						random_rite_tenet = {
							status = core
							limit = {
								wc_tenet_can_be_replaced_for_holy_war_trigger = { FAITH = scope:actor.faith }
							}
							save_scope_as = selected_item
						}
					}
				}
			}
		}

		is_item_valid = {
			wc_tenet_can_be_replaced_for_holy_war_trigger = { FAITH = scope:actor.faith }
		}
	}

	ai_potential = {
		years_from_game_start >= 10
	}

	ai_will_do = {
		base = 25

		modifier = {
			add = 25
			ai_has_warlike_personality = yes
		}
		modifier = {
			add = -25
			ai_has_cautious_personality = yes
		}
		modifier = { # A holy site of the faith is held by another religion
			add = 50
			faith = {
				any_holy_site_barony = {
					exists = holder
					holder.top_liege.faith.religion != root.religion
				}
			}
		}
	}
}
