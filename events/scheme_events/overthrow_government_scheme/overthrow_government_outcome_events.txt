scripted_effect government_overthrow_title_change_effect = {
	# First we build all the lists ahead of time
	scope:target = {
		primary_title = { save_scope_as = liege_primary }
		capital_county = { save_scope_as = old_realm_capital }
		every_held_title = {
			limit = {
				tier >= tier_county
				is_head_of_faith = no
				is_noble_family_title = no
			}
			add_to_list = doge_titles
		}
		every_vassal = { add_to_list = direct_vassals_list }
		add_character_flag = { 
			flag = government_republic_conversion_protection
			days = 1
		}
	}

	# Then we make sure you get the landed titles before the kingdom, so it doesn't get destroyed
	create_title_and_vassal_change = {
		type = usurped
		save_scope_as = change
		add_claim_on_loss = no
		suppress_law_messages = yes # The overthrow events already explain the loss of the republic
	}
	every_in_list = {
		list = doge_titles
		change_title_holder_include_vassals = {
			holder = scope:owner
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change

	hidden_effect = {
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = liege_change
			suppress_law_messages = yes
		}
		every_in_list = {
			list = direct_vassals_list
			limit = {
				this != scope:owner
			}
			change_liege = {
				liege = scope:owner
				change = scope:liege_change
			}
		}
		scope:target = {
			change_liege = {
				liege = scope:owner
				change = scope:liege_change
			}
		}
		resolve_title_and_vassal_change = scope:liege_change
		scope:target = {
			if = {
				limit = { NOT = { has_government = republic_government } }
				change_government = republic_government
			}
		}
	}

	if = {
		limit = {
			exists = scope:old_realm_capital
			capital_county != scope:old_realm_capital
			scope:old_realm_capital.holder = scope:owner
		}
		scope:owner = {
			set_realm_capital = scope:old_realm_capital
		}
	}
}

scripted_effect government_overthrow_law_quick_effect = {
	if = {
		limit = {
			NOT = { has_realm_law = $LAW$ }
		}
		add_realm_law_skip_effects = $LAW$
	}
}

scripted_effect government_overthrow_law_changes_effect = {
	scope:owner = {
		if = {
			limit = {
				house = {
					has_house_aspiration_parameter = republic_commerce 
				}
			}
			government_overthrow_law_quick_effect = { LAW = republic_assembly_authority_constitutional }
			government_overthrow_law_quick_effect = { LAW = republic_assembly_composition_parliament  }
		}
		
		if = {
