#Event adding relevant characters to your travel entourage
gdp_travel_events.0001 = {
	hidden = yes
	trigger = {
		government_has_flag = government_is_merchant
	}
	immediate = {
		every_courtier = {
			limit = {
				is_adult = yes
				is_available_ai = yes
				NOT = { has_court_position = partner_court_position }
				save_temporary_scope_as = gdp_entourage_candidate
				NOT = {
					root = {
						any_scheme = {
							scheme_is_character_agent = scope:gdp_entourage_candidate
						}
					}
				}
			}
			add_to_list = gdp_entourage_members
		}

		every_child = {
			limit = {
				is_available_ai_child = yes
				is_courtier_of = root
			}
			add_to_list = gdp_entourage_members
		}

		ordered_in_list = {
			list = gdp_entourage_members
			order_by = {
				value = 0
				if = {
					limit = {
						OR = {
							has_court_position = fixer_court_position
							has_court_position = translator_court_position
							has_court_position = market_agent_court_position
							has_court_position = bailiff_court_position
						}
					}
					add = 200
				}
				if = {
					limit = { has_trait = merchant_associate }
					add = 100
				}
				if = {
					limit = {
						OR = {
							is_spouse_of = root
							is_close_family_of = root
						}
					}
					add = 50
				}
			}
			max = 10
			check_range_bounds = no
			root.current_travel_plan ?= { add_companion = prev }
		}
	}
}

		current_travel_plan = {

#What xp you gain for listening to your associate
scripted_effect gdp_associate_lifestyle_xp_gain_effect = {
	if = {
		limit = {
			scope:associate = { has_trait = education_stewardship }
		}
		add_trait_xp = {
			trait = lifestyle_merchant
			track = caravaneer
			value = lifestyle_merchant_xp_gain_medium_value
		}
	}
	else_if = {
		limit = {
			scope:associate = { has_trait = education_learning }
		}
		add_trait_xp = {
			trait = lifestyle_merchant
			track = artisan
			value = lifestyle_merchant_xp_gain_medium_value
		}
	}
	else_if = {
		limit = {
			scope:associate = { has_trait = education_diplomacy }
		}
		add_trait_xp = {
			trait = lifestyle_merchant
			track = caravaneer
			value = lifestyle_merchant_xp_gain_medium_value
		}
	}
	else_if = {
		limit = {
			scope:associate = { has_trait = education_intrigue }
		}
		add_trait_xp = {
			trait = lifestyle_merchant
			track = creditor
			value = lifestyle_merchant_xp_gain_medium_value
		}
	}
	else_if = {
		limit = {
			scope:associate = { has_trait = education_martial }
		}
		add_trait_xp = {
			trait = lifestyle_merchant
			track = creditor
			value = lifestyle_merchant_xp_gain_medium_value
		}
	}
}

	if = {
		limit = { location = scope:tower_of_pisa_province }
		trigger_event = gdp_travel_events.2000
	}
	else = {
		start_travel_plan = {
			destination = scope:tower_of_pisa_province
			on_arrival_event = gdp_travel_events.2000
			on_travel_planner_cancel_event = gdp_travel_events.2004
		}
	}
}
