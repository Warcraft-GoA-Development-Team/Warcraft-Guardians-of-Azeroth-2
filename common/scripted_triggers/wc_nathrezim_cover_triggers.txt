# Character scope. Keep cover state on the body, never on a global story.
wc_nathrezim_cover_active_trigger = {
	is_alive = yes
	has_character_flag = wc_balnazzar_disguise_active
	NOT = { has_character_flag = balnazzar_shed_the_mask }
	NOT = { has_variable = wc_nathrezim_public_nature }
}

wc_nathrezim_can_prepare_cover_trigger = {
	wc_nathrezim_cover_active_trigger = yes
	is_imprisoned = no
	is_incapable_trigger = no
	NOT = { has_variable = wc_nathrezim_cover_crisis }
}

wc_nathrezim_valid_prepared_body_trigger = {
	is_alive = yes
	is_ai = yes
	is_adult = yes
	is_landed = yes
	is_playable_character = yes
	is_imprisoned = no
	has_trait = creature_human
	has_being_trait_trigger = no
	NOT = { has_character_flag = wc_balnazzar_disguise_active }
	NOT = { has_variable = wc_nathrezim_cover_crisis }
	NOT = { has_variable = wc_nathrezim_public_nature }
}

wc_nathrezim_body_reserved_trigger = {
	exists = var:wc_nathrezim_prepared_for
	var:wc_nathrezim_prepared_for = {
		wc_nathrezim_cover_active_trigger = yes
		var:wc_nathrezim_prepared_body ?= prev
	}
}

wc_nathrezim_can_escape_cover_trigger = {
	wc_nathrezim_cover_active_trigger = yes
	is_imprisoned = no
	is_incapable_trigger = no
	exists = var:wc_nathrezim_prepared_body
	var:wc_nathrezim_prepared_body = {
		wc_nathrezim_valid_prepared_body_trigger = yes
		var:wc_nathrezim_prepared_for ?= prev
	}
}

# Witness scope. A settlement belongs to one body and never erases knowledge.
wc_nathrezim_actionable_witness_trigger = {
	is_alive = yes
	is_imprisoned = no
	is_incapable_trigger = no
	NOT = { this = scope:wc_cover_body }
	NOT = { has_truce = scope:wc_cover_body }
	scope:wc_cover_body = {
		NOT = { is_target_in_variable_list = { name = wc_nathrezim_settled_witnesses target = prev } }
		OR = {
			liege ?= prev
			is_courtier_of = prev
			prev = { liege ?= scope:wc_cover_body }
			prev = { is_courtier_of = scope:wc_cover_body }
		}
	}
}

# Both scopes are captured by the dispatcher and rechecked at every choice.
wc_nathrezim_cover_case_valid_trigger = {
	exists = scope:wc_cover_body
	exists = scope:wc_cover_witness
	scope:wc_cover_body = {
		wc_nathrezim_cover_active_trigger = yes
		has_variable = wc_nathrezim_cover_crisis
		NOT = { has_variable = wc_nathrezim_cover_resolved }
		NOT = { has_variable = wc_nathrezim_cover_negotiation_cooldown }
		is_at_war = no
		NOT = { any_targeting_faction = { has_variable = wc_nathrezim_ultimatum_sent } }
		var:wc_nathrezim_cover_investigator ?= scope:wc_cover_witness
	}
	scope:wc_cover_witness = {
		NOT = { this = scope:wc_cover_body }
		wc_nathrezim_actionable_witness_trigger = yes
	}
}

# Legal authority over this body, not a universal crime of being a demon.
wc_nathrezim_cover_authority_trigger = {
	wc_nathrezim_cover_case_valid_trigger = yes
	scope:wc_cover_body = {
		OR = {
			liege ?= scope:wc_cover_witness
			is_courtier_of = scope:wc_cover_witness
		}
	}
}
