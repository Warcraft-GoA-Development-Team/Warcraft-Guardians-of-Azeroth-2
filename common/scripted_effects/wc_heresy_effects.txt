# Historical heresy outbreaks (By God Alone component 3) for GoA.
# All effects expect the router's saved scopes: origin_faith, heresy_faith, heresy_key, heresy_strict_capital,
# and (after the pick) epicenter_ruler, birth_location, founder.
# Per-heresy state lives on scope:heresy_strict_capital, the seat barony: wc_heresy_first_fired,
# wc_heresy_cooldown (timed) and wc_heresy_spread_remaining.
# Ports of vanilla pam_heresy_shared_effects.txt blocks are marked "port of".

##### Selection (faith scope: the origin faith) #####

# KEY: heresy key (seline). TARGET: heresy faith key (forgotten_shadow). SEAT: seat barony key (b_northshire).
wc_heresy_select_effect = {
	faith:$TARGET$ = { save_scope_as = heresy_faith }
	title:$SEAT$ = { save_scope_as = heresy_strict_capital }
	save_scope_value_as = {
		name = heresy_key
		value = flag:$KEY$
	}
	if = {
		limit = { wc_heresy_is_resurgence_trigger = { SEAT = $SEAT$ } }
		save_scope_value_as = {
			name = is_resurgence
			value = yes
		}
	}
}

##### Epicenter and founder (faith scope: the origin faith) #####

wc_heresy_pick_epicenter_ruler_effect = {
	clear_saved_scope = epicenter_ruler

	# Ignores vanilla's strict/relaxed regional heresy game rules on purpose: strict seat first, then the home region
	if = {
		limit = {
			scope:heresy_strict_capital.holder ?= {
				faith = root
				is_eligible_heresy_epicenter_ruler_trigger = yes
			}
		}
		scope:heresy_strict_capital.holder = { save_scope_as = epicenter_ruler }
	}
	if = {
		limit = { NOT = { exists = scope:epicenter_ruler } }
		random_faith_ruler = {
			limit = {
				is_eligible_heresy_epicenter_ruler_trigger = yes
				exists = capital_county
				capital_county.title_province = { wc_heresy_in_home_region_trigger = yes }
			}
			weight = {
				base = 1
				modifier = {
					factor = 2
					highest_held_title_tier >= tier_kingdom
				}
			}
			save_scope_as = epicenter_ruler
		}
	}
}

# One case per heresy key. Lore founders lead first outbreaks only; resurgences and unusable lore
# founders get an anonymous one.
wc_heresy_dispatch_founder_effect = {
	clear_saved_scope = founder

	switch = {
		trigger = scope:heresy_key
		# Seline's Shadow
		flag:seline = {
			if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					character:1620 ?= {
						wc_heresy_founder_usable_trigger = { TARGET = faith:forgotten_shadow }
						NOT = { faith = faith:forgotten_shadow }
					}
				}
				character:1620 = { save_scope_as = founder }
			}
			# Spawn a new Seline only when the real one is dead or missing; never a second living Natalie
			else_if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					OR = {
						NOT = { exists = character:1620 }
						character:1620 = { is_alive = no }
					}
				}
				wc_heresy_spawn_seline_effect = yes
			}
			else = {
				wc_heresy_spawn_anonymous_preacher_effect = yes
			}
		}
		# Cult of the Damned: Kel'Thuzad leads in secret
		flag:damned = {
			if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					character:59100 ?= {
						wc_heresy_founder_usable_trigger = { TARGET = faith:death_god }
						NOT = { faith = faith:death_god }
						NOT = { exists = secret_faith }
					}
				}
				save_scope_value_as = {
					name = founder_is_crypto
					value = yes
				}
				character:59100 = { save_scope_as = founder }
			}
			else = {
				wc_heresy_spawn_anonymous_preacher_effect = yes
			}
		}
		# Scarlet zeal
		flag:scarlet = {
			if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					character:60550 ?= {
						wc_heresy_founder_usable_trigger = { TARGET = faith:scarletism }
						NOT = { faith = faith:scarletism }
					}
				}
				character:60550 = { save_scope_as = founder }
			}
			else = {
				wc_heresy_spawn_anonymous_preacher_effect = yes
			}
		}
		# Death's Head: Charlga Razorflank, openly. She often rules the epicenter as well.
		flag:deaths_head = {
			if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					character:charlga_razorflank ?= {
						wc_heresy_founder_usable_trigger = { TARGET = faith:death_god }
						NOT = { faith = faith:death_god }
					}
				}
				character:charlga_razorflank = { save_scope_as = founder }
			}
			else = {
				wc_heresy_spawn_anonymous_preacher_effect = yes
			}
		}
		# Atal'ai revival: Jin'do is already the Hakkari head and leads without converting
		flag:atalai = {
			if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					character:jindo ?= {
						wc_heresy_founder_usable_trigger = { TARGET = faith:cult_of_hakkar }
					}
				}
				character:jindo = { save_scope_as = founder }
			}
			else = {
				wc_heresy_spawn_anonymous_preacher_effect = yes
			}
		}
		# Burning Blade: Neeru leads in secret. From 596 his history already gives him the secret faith.
		flag:burning_blade = {
			if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					character:10654 ?= {
						wc_heresy_founder_usable_trigger = { TARGET = faith:orcish_fel }
						NOT = { faith = faith:orcish_fel }
						OR = {
							secret_faith ?= faith:orcish_fel
							NOT = { exists = secret_faith }
						}
					}
				}
				save_scope_value_as = {
					name = founder_is_crypto
					value = yes
				}
				character:10654 = { save_scope_as = founder }
			}
			# Before his 596 history (a 583 start) he is still openly fel, and leads in the open
			else_if = {
				limit = {
					NOT = { exists = scope:is_resurgence }
					character:10654 ?= {
						wc_heresy_founder_usable_trigger = { TARGET = faith:orcish_fel }
						faith = faith:orcish_fel
					}
				}
				character:10654 = { save_scope_as = founder }
			}
			else = {
				wc_heresy_spawn_anonymous_preacher_effect = yes
			}
		}
	}
}

# Fallback when character 1620 is dead or missing
wc_heresy_spawn_seline_effect = {
	create_character = {
		name = "Natalie"
		age = 25
		gender_female_chance = 100
		dynasty = none
		random_traits = no
		location = scope:birth_location.title_province
		faith = scope:heresy_faith
		culture = scope:birth_location.culture
		trait = creature_human
		trait = education_learning_3
		trait = zealous
		trait = theologian
		diplomacy = { min_template_decent_skill max_template_decent_skill }
		martial = { min_template_low_skill max_template_low_skill }
		stewardship = { min_template_low_skill max_template_low_skill }
		intrigue = { min_template_decent_skill max_template_decent_skill }
		learning = { min_template_high_skill max_template_high_skill }
		prowess = { min_template_low_skill max_template_low_skill }
		save_scope_as = founder
	}
	scope:founder ?= {
		set_variable = {
			name = wc_shadow_magic_lifestyle_additional_perks_variable
			value = wc_perks_needed_for_level_2_magic_trait_value
		}
		# The variable alone only takes effect at game start (WCRAC.30); grant the perks now
		add_perks_in_nonelemental_MAGIC_lifestyle_effect = { MAGIC = shadow }
		historical_character_finalization_effect = yes
	}
}

wc_heresy_spawn_anonymous_preacher_effect = {
	create_character = {
		age = { 25 45 }
		gender_female_chance = wc_heresy_preacher_female_chance_value
		dynasty = none
		random_traits = no
		location = scope:birth_location.title_province
		faith = scope:heresy_faith
		culture = scope:birth_location.culture
		trait = education_learning_3
		trait = zealous
		learning = { min_template_decent_skill max_template_high_skill }
		save_scope_as = founder
	}
	# The epicenter ruler's race (a quilboar preacher in quilboar lands). set_race_effect also sets
	# race_giving_fired_flag, so WCRAC.10 does not re-roll the race a day later.
	scope:founder ?= {
		copy_race_effect = { CHARACTER = scope:epicenter_ruler }
		if = {
			limit = { has_race_trait_trigger = no }
			set_race_effect = { RACE = creature_human }
		}
	}
}

##### Conversion #####

# Crypto founders take the secret faith (once); open founders convert and become heresiarchs.
# A founder who also rules the epicenter (Charlga) converts like a ruler, so their counties and court follow.
wc_heresy_convert_founder_effect = {
	scope:founder = {
		if = {
			limit = { exists = scope:founder_is_crypto }
			if = {
				limit = { NOT = { secret_faith ?= scope:heresy_faith } }
				make_character_crypto_religionist_effect = { CRYPTO_RELIGION = scope:heresy_faith }
			}
		}
		else = {
			if = {
				limit = { this = scope:epicenter_ruler }
				wc_heresy_convert_ruler_effect = yes
			}
			else_if = {
				limit = { NOT = { faith = scope:heresy_faith } }
				set_character_faith_with_conversion = scope:heresy_faith
				# Same fallback as wc_heresy_convert_ruler_effect, in case the conversion rule refuses
				if = {
					limit = { NOT = { faith = scope:heresy_faith } }
					set_character_faith = scope:heresy_faith
				}
			}
			pam_heresy_finalise_founder_effect = yes
		}
	}
}

# Port of vanilla pam_heresy_c3_already_declared_setup_effect (county part). The epicenter ruler keeps their faith.
wc_heresy_convert_epicenter_effect = {
	scope:epicenter_ruler.capital_county ?= {
		# Mark only if this outbreak changes it, so reclaim never reverts a county that already had the faith
		if = {
			limit = { NOT = { faith = scope:heresy_faith } }
			wc_heresy_mark_converted_county_effect = yes
		}
		set_county_faith = scope:heresy_faith
		every_neighboring_county = {
			limit = {
				faith = scope:origin_faith
				holder.faith ?= scope:origin_faith
			}
			set_county_faith = scope:heresy_faith
			wc_heresy_mark_converted_county_effect = yes
		}
	}
}

# Character scope. Port of vanilla pam_heresy_convert_ruler_to_heresy_faith_effect (minus clerical regions).
wc_heresy_convert_ruler_effect = {
	add_character_flag = {
		flag = recently_converted_by_heresy
		years = 10
	}

	if = {
		limit = { NOT = { faith = scope:heresy_faith } }
		set_character_faith_with_conversion = scope:heresy_faith
		# The faith conversion rule can refuse a new heresy, so vanilla converts directly then
		if = {
			limit = { NOT = { faith = scope:heresy_faith } }
			set_character_faith = scope:heresy_faith
		}
	}

	every_held_title = {
		limit = {
			tier = tier_county
			faith = scope:origin_faith
		}
		set_county_faith = scope:heresy_faith
		wc_heresy_mark_converted_county_effect = yes
	}
	every_courtier_or_guest = {
		limit = {
			is_physically_able_ai = yes
			faith = scope:origin_faith
			NOT = { has_character_flag = ai_will_not_convert }
		}
		set_character_faith = scope:heresy_faith
	}
	every_courtier_away = {
		limit = {
			is_physically_able_ai = yes
			faith = scope:origin_faith
			NOT = { has_character_flag = ai_will_not_convert }
		}
		set_character_faith = scope:heresy_faith
	}
	every_pool_character = {
		province = capital_province
		limit = {
			is_physically_able_ai = yes
			faith = scope:origin_faith
			NOT = { has_character_flag = ai_will_not_convert }
		}
		set_character_faith = scope:heresy_faith
	}
}

# County scope. Records which outbreak converted this county, so reclaim only reverts those.
wc_heresy_mark_converted_county_effect = {
	set_variable = {
		name = wc_heresy_outbreak
		value = scope:heresy_strict_capital
		years = 20
	}
}

# Port of the heresy_faith branch of vanilla pam_heresy_send_spread_message_effect. Expects scope:convert_ruler.
wc_heresy_send_spread_message_effect = {
	every_player = {
		limit = {
			faith = scope:origin_faith
			NOR = {
				this = scope:convert_ruler
				scope:founder ?= this
			}
			OR = {
				scope:convert_ruler.liege ?= this
				capital_county ?= {
					squared_distance = {
						target = scope:convert_ruler.capital_county
						value <= squared_distance_medium
					}
				}
			}
		}
		if = {
			limit = { exists = scope:founder_is_crypto }
			send_interface_message = {
				type = msg_pam_heresy_spread
				title = wc_heresy_historical.spread.title
				left_icon = scope:convert_ruler
				right_icon = scope:epicenter_ruler
				show_as_tooltip = {
					scope:convert_ruler = { set_character_faith = scope:heresy_faith }
				}
			}
		}
		else = {
			send_interface_message = {
				type = msg_pam_heresy_spread
				title = wc_heresy_historical.spread.title
				left_icon = scope:convert_ruler
				right_icon = scope:founder
				show_as_tooltip = {
					scope:convert_ruler = { set_character_faith = scope:heresy_faith }
				}
			}
		}
	}
}

##### Budget, reclaim, cooldown #####

# AMOUNT is added (negative shrinks the spread). No-op once the sweep has removed the variable.
wc_heresy_change_spread_budget_effect = {
	scope:heresy_strict_capital ?= {
		if = {
			limit = { has_variable = wc_heresy_spread_remaining }
			change_variable = {
				name = wc_heresy_spread_remaining
				add = $AMOUNT$
			}
			if = {
				limit = { var:wc_heresy_spread_remaining <= 0 }
				remove_variable = wc_heresy_spread_remaining
			}
		}
	}
}

# Character scope (a player). Reverts the counties this outbreak converted in their own domain, to their own faith.
# Counties that were already the heresy faith before the outbreak are left alone.
wc_heresy_reclaim_domain_effect = {
	save_temporary_scope_as = reclaimer
	every_held_title = {
		limit = {
			tier = tier_county
			faith = scope:heresy_faith
			var:wc_heresy_outbreak ?= scope:heresy_strict_capital
		}
		remove_variable = wc_heresy_outbreak
		set_county_faith = scope:reclaimer.faith
	}
}

wc_heresy_set_cooldown_effect = {
	scope:heresy_strict_capital = {
		set_variable = {
			name = wc_heresy_first_fired
			value = yes
		}
		set_variable = {
			name = wc_heresy_cooldown
			value = yes
			years = wc_heresy_cooldown_years_value
		}
	}
}
