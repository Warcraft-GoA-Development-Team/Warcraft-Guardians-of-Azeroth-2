# GoA faith heresy values (wc_heresy.0001 / 0010 / 0011)
# Vanilla 1.20 removed both values below from 02_religion_values.txt, and GoA's 1.20 copy of that file no longer has them.

# How much fervor the old faith gains when a ruler splits off to a heresy (was heretical_defector_fervor_gain = 15)
wc_heretical_defector_fervor_gain_value = {
	value = 15
}

# Weight factor for regional heresies, read by regional_heresy_modifiers (common/scripted_modifiers/wc_religious_modifiers.txt). Was vanilla's regional_heresy_factor, which 1.20 removed.
wc_regional_heresy_factor_value = {
	if = {
		limit = { has_game_rule = default_regional_heresy }
		value = 5
	}
	else_if = {
		limit = { has_game_rule = strict_regional_heresy }
		value = 200
	}
	else = {
		value = 1
	}
}
