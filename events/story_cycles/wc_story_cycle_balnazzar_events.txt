# A Face Not His Own - the kill itself. Balnazzar kills Saidan Dathrohan and takes his body,
# finishing the identity swap .1010 already set in motion.
wc_balnazzar_story.1001 = {
	type = character_event
	title = wc_balnazzar_story.1001.title
	desc = wc_balnazzar_story.1001.desc
	theme = death

	# Arc's climax, needs portraits visible - fullscreen_event never renders them, so
	# big_event_window instead (same big background art, portraits actually show).
	window = big_event_window

	trigger = {
		NOT = { has_character_flag = wc_balnazzar_disguise_active }
	}

	# The kill happens in Stratholme now, not Lordaeron's undercity.
	override_background = wc_stratholme

	left_portrait = {
		character = root
		animation = spymaster
	}
	right_portrait = {
		character = character:60021 # Saidan Dathrohan
		animation = paranoia
	}

	immediate = {
		character:60021 = { save_scope_as = dathrohan }
	}

	# .1001.b: fallback if Dathrohan is dead/unplayable - avoids a soft-lock.
	option = {
		name = wc_balnazzar_story.1001.a

		trigger = {
			scope:dathrohan = {
				is_alive = yes
				is_playable_character = yes
			}
		}

		# Console tests and old saves can reach this event before the setup event.
		# Self-heal the story reference before transferring its owner.
		if = {
			limit = { NOT = { exists = global_var:balnazzar_dreadlord_story } }
			create_story = {
				type = story_balnazzar_dreadlord
				save_scope_as = wc_balnazzar_recovered_story
			}
			set_global_variable = {
				name = balnazzar_dreadlord_story
				value = scope:wc_balnazzar_recovered_story
			}
		}

		# Trait/secret/perk changes render natively in the confirm panel.
			scope:dathrohan = {
				# Possession burns off Dathrohan's age-related decline.
				# Ongoing protection while trait held: wc_story_cycle_balnazzar.txt effect_group.
				remove_trait = clouded_eyes
				remove_trait = faltering_heart
				remove_trait = fragile_bones
				remove_trait = infirm
				remove_trait = withering_mind

				# The truth itself, tracked as a real CK3 secret - exposure
				# (however it happens) resolves the story via .2000.
				add_dreadlord_disguise_secret_effect = yes

				# Balnazzar's magic talents (Disorder, Death) carry over with the disguise.
				# Wrapped in hidden_effect to avoid listing every perk in the confirm panel.
				custom_tooltip = wc_balnazzar_story.1001.a_magic_tooltip
				hidden_effect = {
					wc_balnazzar_copy_magic_talents_effect = yes
				}
			}

			# Keep story attached to whoever plays Balnazzar's arc.
			# Without this, killing root fires on_owner_death and ends story early.
			global_var:balnazzar_dreadlord_story = {
				make_story_owner = scope:dathrohan
			}

			# Move player off root before killing it, or engine aborts effect at death screen.
			# Call unscoped (root=player), not inside scope:dathrohan - engine requires current scope = player.
			set_player_character = scope:dathrohan

		add_trait = in_twisting_nether # Balnazzar's true body goes into standby, not death - see wc_being_events.txt WCBEI.100/110 for the reference pattern this reuses.
		set_focus = no # A dormant body must not keep earning lifestyle XP.

			scope:dathrohan = {
				trigger_event = {
					id = wc_balnazzar_story.1002
					days = { 1 3 }
				}
			}
	}

	option = {
		name = wc_balnazzar_story.1001.b

		trigger = {
			NOT = {
				scope:dathrohan = {
					is_alive = yes
					is_playable_character = yes
				}
			}
		}

		custom_tooltip = wc_balnazzar_story.1001.b_tooltip
			global_var:balnazzar_dreadlord_story = {
				end_story = yes
			}
	}
}

# The Ambush That Wasn't - pays off .1004's Kel'Thuzad pact (Stratholme's dead cover the
# kill) and shows the moment itself. Fires from murder_on_actions.txt in place of the
# vanilla "Assassin!" popup when the forced-Renault murder scheme against Alexandros
# succeeds - see wc_balnazzar_force_renault_as_assassin_effect.
wc_balnazzar_story.1006 = {
	type = character_event
	title = wc_balnazzar_story.1006.title
	desc = wc_balnazzar_story.1006.desc
	theme = intrigue

	# Another major beat, same big_event_window as .1001/.1007.
	window = big_event_window

	# Same Stratholme background every other event in this arc uses (and what
	# the old, now-deleted mograine_assassination outcome events used too).
	override_background = wc_stratholme

	left_portrait = {
		character = character:60019 # Renault Mograine
		animation = assassin
	}
	right_portrait = {
		character = character:60017 # Alexandros Mograine
		animation = map_fear
	}

	# Re-derive scope:owner/scope:target fresh rather than trust that they survive
	# the 1-day delay on the trigger_event that fires this. scheme_owner_pov_murder_success_effect
	# applies prestige/piety/dread/stress for real, but murder_success_kill_effect
	# (nested inside it) wraps the actual death in show_as_tooltip when called with
	# root = scope:owner - a UI preview only, it never runs. Vanilla only gets a real
	# kill because murder_outcome_reworked.0001 (owner POV) hands off to .0002 (target
	# POV) via an `after` block, and murder_success_kill_effect checks `this = scope:target`
	# to pick its real, unwrapped hidden_effect branch. Replicate that by calling it a
	# second time scoped into scope:target instead of adding a second event.
	immediate = {
		save_scope_as = owner
		character:60017 = { save_scope_as = target } # Alexandros Mograine
		# Guard against the 1-day gap between the scheme resolving and this event firing -
		# an unrelated death in that window (war, another scheme) would otherwise hand an
		# already-dead character to murder_success_kill_effect, which expects a live target.
		if = {
			limit = { scope:target = { is_alive = yes } }
			scheme_owner_pov_murder_success_effect = { REASON = death_mysterious }
			scope:target = {
				murder_success_kill_effect = { REASON = death_mysterious }
			}
		}
	}

	option = {
		name = wc_balnazzar_story.1006.a
	}
}

	trigger = {

		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = { story_owner = root }
	}

	# Another major beat, same big_event_window as .1001.
	window = big_event_window

	override_background = sm_pa_scarlet_monastery_sm

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60019 # Renault Mograine
		animation = shame # Hollow, guilt-ridden - days removed from the kill itself
	}

	immediate = {
		character:60019 = { save_scope_as = renault }
		character:60020 = { save_scope_as = darion }
	}

	# .1007.b is the fallback if Renault is dead/missing by this point.
	option = {
		name = wc_balnazzar_story.1007.a

		trigger = {
			scope:renault = { is_alive = yes }
		}

		# Renault leaves the blade behind after murdering Alexandros. Darion
		# recovers it instead of letting the killer inherit it permanently.
		hidden_effect = {
			if = {
				limit = { scope:darion = { is_alive = yes } }
				scope:renault = {
					every_character_artifact = {
						limit = {
							OR = {
								has_variable = ashbringer
								has_variable = corrupted_ashbringer
							}
						}
						set_owner = {
							target = scope:darion
							history = {
								type = inherited
								recipient = scope:darion
							}
						}
						equip_artifact_to_owner_replace = yes
					}
				}
			}

				target = flag:form_the_scarlet_crusade_decision

			# TODO: Varimathras arc (fake loyalty to Sylvanas, secretly still
			# Legion-bound) deliberately deferred - see conversation notes.
	}

# The Mask Comes Off - shared resolution for all "reject the mask" triggers (old age, exposure, paranoia).
# Disguise ends and the Risen rise from the current host's capital.
wc_balnazzar_story.2000 = {
	type = character_event
	title = wc_balnazzar_story.2000.title
	desc = wc_balnazzar_story.2000.desc
	theme = death
	window = big_event_window

	# Self-guard against double-firing - the flag is only set once .2000
	# actually runs, not when it's queued, so two callers could race.
	trigger = {
		NOT = { has_character_flag = balnazzar_shed_the_mask }
	}

	override_background = wc_stratholme
	widget = {
		gui = "event_window_widget_vfx_conclusion_smoke"
		container = "background_shader_vfx_container"
	}

	left_portrait = {
		character = character:430400
		animation = rage
	}

	immediate = {
		add_character_flag = balnazzar_shed_the_mask
		remove_dreadlord_disguise_secret_effect = yes
	}

	right_portrait = {
		character = root
		animation = dead
		hide_info = yes
	}

	option = {
		name = wc_balnazzar_story.2000.a

		# Changes render natively in the confirm panel.
			# Resurrect Balnazzar's nathrezim body (character:430400, "in limbo" at .1001) - one-shot.
			# Falls through to the shell surviving if unavailable, instead of an unrecoverable Game Over.
			if = {
				limit = {
					exists = character:430400
					character:430400 = { has_trait = in_twisting_nether }
					is_landed = yes
					exists = capital_county
				}

				character:430400 = {
					# Compatibility repair for bodies already left earning Fel XP in limbo.
					set_focus = no
					while = {
						limit = { disorder_magic_lifestyle_perk_points > 0 }
						add_disorder_magic_lifestyle_perk_points = -1
					}
					remove_trait = in_twisting_nether
					save_scope_as = resurrected_character
				}

				# Tyr's Hand is the fixed core. Prefer the host's current capital duchy
				# as the dynamic second duchy; if the capital is already in Tyr's Hand,
				# use another ordinary duchy the host personally holds.
				# Light's Hope remains a living enclave, and foreign-held land is untouched.
				root.capital_county = {
					save_scope_as = wc_balnazzar_risen_capital
				}
				root = {
					if = {
						limit = {
							capital_county.duchy = {
								NOT = { this = title:d_tyrs_hand }
							}
						}
						capital_county.duchy = {
							save_scope_as = wc_balnazzar_risen_dynamic_duchy
						}
					}
					else = {
						random_held_title = {
							title_tier = duchy
							limit = {
								NOT = { this = title:d_tyrs_hand }
								is_landless_type_title = no
								is_nomad_title = no
							}
							save_scope_as = wc_balnazzar_risen_dynamic_duchy
						}
					}
				}
				title:d_tyrs_hand = {
					every_de_jure_county = {
						limit = {
							NOT = { this = title:c_lights_hope }
							exists = holder
							holder = {
								OR = {
									this = root
									is_vassal_or_below_of = root
								}
							}
						}
						add_to_list = wc_balnazzar_risen_counties
					}
				}
				if = {
					limit = {
						exists = scope:wc_balnazzar_risen_dynamic_duchy
						scope:wc_balnazzar_risen_dynamic_duchy = {
							NOT = { this = title:d_tyrs_hand }
						}
					}
					scope:wc_balnazzar_risen_dynamic_duchy = {
						every_de_jure_county = {
							limit = {
								exists = holder
								holder = {
									OR = {
										this = root
										is_vassal_or_below_of = root
									}
								}
							}
							add_to_list = wc_balnazzar_risen_counties
						}
					}
				}
				# Seed the list so the duplicate guard is valid for the first holder.
				root = { add_to_list = wc_balnazzar_risen_vassals }
				every_in_list = {
					list = wc_balnazzar_risen_counties
					holder = {
						if = {
							limit = {
								NOT = { this = root }
								is_vassal_or_below_of = root
								NOT = { is_in_list = wc_balnazzar_risen_vassals }
							}
							add_to_list = wc_balnazzar_risen_vassals
						}
					}
				}
				root = { remove_from_list = wc_balnazzar_risen_vassals }
				title:c_lights_hope = {
					if = {
						limit = {
							exists = holder
							holder = {
								is_alive = yes
								NOT = { this = root }
								is_vassal_or_below_of = root
							}
						}
						holder = { save_scope_as = wc_balnazzar_argent_enclave }
					}
				}

				# Create the titular realm before moving vassals so on_vassal_gained
				# recognizes Balnazzar as the revealed ruler of the Risen.
				if = {
					limit = { title:d_wc_the_risen = { NOT = { exists = holder } } }
					create_title_and_vassal_change = {
						type = created
						save_scope_as = wc_balnazzar_risen_title_change
						add_claim_on_loss = no
					}
					title:d_wc_the_risen = {
						change_title_holder = {
							holder = scope:resurrected_character
							change = scope:wc_balnazzar_risen_title_change
						}
					}
					resolve_title_and_vassal_change = scope:wc_balnazzar_risen_title_change
				}
				if = {
					limit = { title:d_wc_the_risen = { holder = scope:resurrected_character } }
					scope:resurrected_character = {
						set_primary_title_to = title:d_wc_the_risen
					}
				}

				# Keep the two target duchies, personal counties, and county-holder
				# vassal trees together. Vassal county holders keep their own titles.
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = balnazzar_reveal_title_change
					add_claim_on_loss = no
				}
				title:d_tyrs_hand = {
					if = {
						limit = { holder = root }
						change_title_holder = {
							holder = scope:resurrected_character
							change = scope:balnazzar_reveal_title_change
							take_baronies = no
						}
					}
				}
				if = {
					limit = {
						exists = scope:wc_balnazzar_risen_dynamic_duchy
						scope:wc_balnazzar_risen_dynamic_duchy = {
							NOT = { this = title:d_tyrs_hand }
							holder = root
						}
					}
					scope:wc_balnazzar_risen_dynamic_duchy = {
						change_title_holder = {
							holder = scope:resurrected_character
							change = scope:balnazzar_reveal_title_change
							take_baronies = no
						}
					}
				}
				every_in_list = {
					list = wc_balnazzar_risen_counties
					if = {
						limit = { holder = root }
						change_title_holder = {
							holder = scope:resurrected_character
							change = scope:balnazzar_reveal_title_change
							take_baronies = yes
						}
					}
				}
				every_in_list = {
					list = wc_balnazzar_risen_vassals
					change_liege = {
						liege = scope:resurrected_character
						change = scope:balnazzar_reveal_title_change
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_reveal_title_change

				# Existing undead still need Balnazzar's faith; living vassals and their
				# descendants are raised without flooding the player with notifications.
				every_in_list = {
					list = wc_balnazzar_risen_vassals
					if = {
						limit = {
							is_alive = yes
							NOT = { has_trait = being_undead }
						}
						raise_undead_faith_only_no_notification_effect = { MASTER = scope:resurrected_character }
					}
					else_if = {
						limit = { is_alive = yes }
						set_character_faith_with_conversion = scope:resurrected_character.faith
					}
					if = {
						limit = { is_alive = yes }
						add_opinion = { modifier = loyal_undead target = scope:resurrected_character }
					}
					every_vassal_or_below = {
						limit = { is_alive = yes }
						if = {
							limit = { NOT = { has_trait = being_undead } }
							raise_undead_faith_only_no_notification_effect = { MASTER = scope:resurrected_character }
						}
						else = {
							set_character_faith_with_conversion = scope:resurrected_character.faith
						}
						add_opinion = { modifier = loyal_undead target = scope:resurrected_character }
					}
				}

				# The Risen are undead locals, not a new Nathrezim people.
				every_in_list = {
					list = wc_balnazzar_risen_counties
					set_county_culture = culture:scourge
					set_county_faith = scope:resurrected_character.faith
				}

				# Raise only household courtiers physically present in the current capital.
				every_courtier = {

						is_alive = yes
						NOT = { has_trait = being_undead }
						location.county = scope:wc_balnazzar_risen_capital
					}
					raise_undead_faith_only_effect = { MASTER = scope:resurrected_character }
					if = {

						add_opinion = { modifier = loyal_undead target = scope:resurrected_character }

					check_available_government_effect = yes

					title:c_lights_hope = { set_county_faith = faith:argent }

					limit = {
						character:60019 = {

							has_trait = creature_human
							NOT = { has_trait = being_undead }

							is_alive = yes
							has_trait = creature_human
							NOT = { has_trait = being_undead }
							has_faith = faith:scarletism
						}
						save_scope_as = wc_balnazzar_scarlet_successor
					}
				}
				if = {
					limit = {
						exists = scope:wc_balnazzar_scarlet_successor
						title:k_order_of_the_silver_hand = { holder = root }
					}
					create_title_and_vassal_change = {
						type = granted
						save_scope_as = wc_balnazzar_scarlet_title_change
						add_claim_on_loss = no
					}
					every_held_title = {
						limit = {
							tier = tier_duchy
							de_jure_liege = title:k_order_of_the_silver_hand

				every_character_artifact = {

						target = scope:resurrected_character
						history = {
							type = inherited
							recipient = scope:resurrected_character

						id = wc_balnazzar_story.2001
						days = 1
					}

				# the story - no separate end_story call needed.
				death = { death_reason = death_vanished }
			}
			else = {
				# No valid handoff body - root survives, exposed but alive,
				# rather than risk an unrecoverable Game Over.
				global_var:balnazzar_dreadlord_story = {
					end_story = yes
				}
			}
	}
