namespace = wc_scarlet_inquisition

# A short judicial procedure, deliberately adapted for GoA2.
wc_scarlet_inquisition.1 = {
	type = character_event
	hidden = yes
	trigger = { wc_inquisition_case_matches_trigger = yes }
	immediate = {
		if = {
			limit = {
				wc_inquisition_case_valid_trigger = yes
				scope:wc_accused = { var:wc_inquisition_stage = 1 }
			}
			scope:wc_accused = { set_variable = { name = wc_inquisition_stage value = 2 } }
			scope:wc_inquisitor = { trigger_event = wc_scarlet_inquisition.2 }
		}
		else = { wc_inquisition_discard_effect = yes }
	}
}

wc_scarlet_inquisition.2 = {
	type = character_event
	title = wc_scarlet_inquisition.2.t
	desc = {
		desc = wc_scarlet_inquisition.2.desc
		first_valid = {
			triggered_desc = {
				trigger = { wc_inquisition_known_crime_trigger = yes }
				desc = wc_scarlet_inquisition.known_crime
			}
			desc = wc_scarlet_inquisition.accusation
		}
	}
	theme = faith
	left_portrait = { character = root animation = personality_zealous }
	right_portrait = { character = scope:wc_accused animation = worry }
	trigger = { wc_inquisition_judge_valid_trigger = { STAGE = 2 } }
	option = {
		name = wc_scarlet_inquisition.acquit
		trigger = { wc_inquisition_judge_valid_trigger = { STAGE = 2 } }
		wc_inquisition_propose_effect = { SENTENCE = acquittal }
		ai_chance = {
			base = 30
			modifier = { add = 50 has_trait = compassionate }
			modifier = { add = 40 has_trait = forgiving }
			modifier = { factor = 0.2 wc_inquisition_known_crime_trigger = yes }
			modifier = { add = 40 has_relation_friend = scope:wc_accused }
			modifier = { factor = 0.2 has_relation_rival = scope:wc_accused }
		}
	}
	option = {
		name = wc_scarlet_inquisition.penance
		trigger = { wc_inquisition_judge_valid_trigger = { STAGE = 2 } }
		wc_inquisition_propose_effect = { SENTENCE = penance }
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = forgiving }
			modifier = { add = 30 has_trait = compassionate }
		}
	}
	option = {
		name = wc_scarlet_inquisition.execute
		trigger = { wc_inquisition_judge_valid_trigger = { STAGE = 2 } }
		wc_inquisition_propose_effect = { SENTENCE = execution }
		ai_chance = {
			base = 15
			modifier = { add = 50 has_trait = zealous }
			modifier = { add = 60 has_trait = sadistic }
			modifier = { add = 60 has_relation_rival = scope:wc_accused }
			modifier = { add = 80 wc_inquisition_known_crime_trigger = yes }
			modifier = { factor = 0.2 has_trait = compassionate }
			modifier = { factor = 0.4 has_trait = forgiving }
		}
	}
	option = {
		name = wc_scarlet_inquisition.lapsed
		trigger = { NOT = { wc_inquisition_judge_valid_trigger = { STAGE = 2 } } }
		hidden_effect = { wc_inquisition_discard_effect = yes }
	}
}

wc_scarlet_inquisition.3 = {
	type = character_event
	title = wc_scarlet_inquisition.3.t
	desc = {
		desc = wc_scarlet_inquisition.3.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:wc_accused = { var:wc_inquisition_sentence = flag:execution } }
				desc = wc_scarlet_inquisition.3.execution
			}
			triggered_desc = {
				trigger = { scope:wc_accused = { var:wc_inquisition_sentence = flag:penance } }
				desc = wc_scarlet_inquisition.3.penance
			}
			desc = wc_scarlet_inquisition.3.acquittal
		}
	}
	theme = faith
	left_portrait = { character = scope:wc_inquisitor animation = personality_zealous }
	right_portrait = { character = scope:wc_accused animation = worry }
	trigger = { wc_inquisition_command_valid_trigger = yes }
	option = {
		name = wc_scarlet_inquisition.accept
		trigger = { wc_inquisition_command_valid_trigger = yes }
		if = {
			limit = { wc_inquisition_command_valid_trigger = yes }
			wc_inquisition_resolve_effect = yes
		}
		else = { wc_inquisition_discard_effect = yes }
		ai_chance = {
			base = 60
			modifier = { add = 40 has_relation_rival = scope:wc_accused }
			modifier = { add = 30 has_trait = zealous }
			modifier = { add = 40 wc_inquisition_known_crime_trigger = yes }
		}
	}
	option = {
		name = wc_scarlet_inquisition.plead
		trigger = {
			wc_inquisition_command_valid_trigger = yes
			scope:wc_accused = { NOT = { var:wc_inquisition_sentence = flag:acquittal } }
		}
		if = {
			limit = { wc_inquisition_command_valid_trigger = yes }
			scope:wc_accused = { set_variable = { name = wc_inquisition_stage value = 4 } }
			scope:wc_inquisitor = { trigger_event = wc_scarlet_inquisition.4 }
		}
		else = { wc_inquisition_discard_effect = yes }
		ai_chance = {
			base = 20
			modifier = { add = 60 has_trait = compassionate }
			modifier = { add = 40 has_trait = forgiving }
			modifier = { add = 60 is_close_family_of = scope:wc_accused }
			modifier = { add = 40 has_relation_friend = scope:wc_accused }
			modifier = { factor = 0.2 has_relation_rival = scope:wc_accused }
		}
	}
	option = {
		name = wc_scarlet_inquisition.hook
		trigger = {
			wc_inquisition_command_valid_trigger = yes
			has_usable_hook = scope:wc_inquisitor
			scope:wc_accused = { var:wc_inquisition_sentence = flag:execution }
		}
		wc_inquisition_hook_effect = yes
		ai_chance = {
			base = 10
			modifier = { add = 80 is_close_family_of = scope:wc_accused }
			modifier = { add = 60 has_relation_friend = scope:wc_accused }
			modifier = { add = 40 has_trait = compassionate }
			modifier = { factor = 0.1 has_relation_rival = scope:wc_accused }
		}
	}
	option = {
		name = wc_scarlet_inquisition.denounce
		trigger = {
			wc_inquisition_command_valid_trigger = yes
			scope:wc_accused = { NOT = { var:wc_inquisition_sentence = flag:acquittal } }
		}
		if = {
			limit = { wc_inquisition_command_valid_trigger = yes }
			add_opinion = { modifier = angry_opinion target = scope:wc_inquisitor opinion = -30 }
			scope:wc_inquisitor = { add_opinion = { modifier = angry_opinion target = scope:wc_commander opinion = -30 } }
			scope:wc_accused = {
				add_opinion = { modifier = grateful_opinion target = scope:wc_commander opinion = 20 }
				every_close_family_member = {
					limit = { is_alive = yes NOT = { this = scope:wc_commander } }
					add_opinion = { modifier = grateful_opinion target = scope:wc_commander opinion = 20 }
				}
			}
			wc_inquisition_resolve_effect = yes
		}
		else = { wc_inquisition_discard_effect = yes }
		ai_chance = {
			base = 10
			modifier = { add = 40 has_relation_rival = scope:wc_inquisitor }
			modifier = { add = 30 has_trait = compassionate }
			modifier = { add = 40 is_close_family_of = scope:wc_accused }
			modifier = { factor = 0.2 wc_inquisition_known_crime_trigger = yes }
		}
	}
	option = {
		name = wc_scarlet_inquisition.lapsed
		trigger = { NOT = { wc_inquisition_command_valid_trigger = yes } }
		hidden_effect = { wc_inquisition_discard_effect = yes }
	}
}

wc_scarlet_inquisition.4 = {
	type = character_event
	title = wc_scarlet_inquisition.4.t
	desc = wc_scarlet_inquisition.4.desc
	theme = faith
	left_portrait = { character = scope:wc_commander animation = disapproval }
	right_portrait = { character = scope:wc_accused animation = worry }
	trigger = { wc_inquisition_judge_valid_trigger = { STAGE = 4 } }
	option = {
		name = wc_scarlet_inquisition.uphold
		trigger = { wc_inquisition_judge_valid_trigger = { STAGE = 4 } }
		if = {
			limit = { wc_inquisition_judge_valid_trigger = { STAGE = 4 } }
			wc_inquisition_resolve_effect = yes
		}
		else = { wc_inquisition_discard_effect = yes }
		ai_chance = {
			base = 40
			modifier = { add = 40 has_trait = zealous }
			modifier = { add = 50 has_trait = sadistic }
			modifier = { add = 60 has_relation_rival = scope:wc_accused }
			modifier = { add = 60 wc_inquisition_known_crime_trigger = yes }
			modifier = { add = 30 has_relation_rival = scope:wc_commander }
		}
	}
	option = {
		name = wc_scarlet_inquisition.mercy
		trigger = { wc_inquisition_judge_valid_trigger = { STAGE = 4 } }
		if = {
			limit = { wc_inquisition_judge_valid_trigger = { STAGE = 4 } }
			scope:wc_accused = {
				if = {
					limit = { var:wc_inquisition_sentence = flag:execution }
					set_variable = { name = wc_inquisition_sentence value = flag:penance }
				}
				else = { set_variable = { name = wc_inquisition_sentence value = flag:acquittal } }
			}
			wc_inquisition_resolve_effect = yes
		}
		else = { wc_inquisition_discard_effect = yes }
		ai_chance = {
			base = 25
			modifier = { add = 50 has_trait = compassionate }
			modifier = { add = 50 has_trait = forgiving }
			modifier = { add = 40 has_relation_friend = scope:wc_accused }
			modifier = { add = 40 opinion = { target = scope:wc_commander value >= 50 } }
			modifier = { factor = 0.3 wc_inquisition_known_crime_trigger = yes }
		}
	}
	option = {
		name = wc_scarlet_inquisition.lapsed
		trigger = { NOT = { wc_inquisition_judge_valid_trigger = { STAGE = 4 } } }
		hidden_effect = { wc_inquisition_discard_effect = yes }
	}
}

# The prisoner owns this callback, so a dead inquisitor cannot strand the case.
wc_scarlet_inquisition.9 = {
	type = character_event
	hidden = yes
	trigger = { wc_inquisition_case_matches_trigger = yes }
	immediate = {
		if = {
			limit = { wc_inquisition_case_valid_trigger = yes }
			trigger_event = { id = wc_scarlet_inquisition.9 days = 7 }
		}
		else = { wc_inquisition_discard_effect = yes }
	}
}
