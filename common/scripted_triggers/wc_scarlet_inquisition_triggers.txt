# Prisoner scope. Reward locks follow the person, including transfers outside the Crusade.
wc_inquisition_transfer_unlocked_trigger = {
	NOT = { has_variable = wc_inquisition_stage }
	NOT = { has_character_flag = wc_inquisition_cooldown }
}

# Named scopes are carried by the queued events and saved with the game.
wc_inquisition_case_matches_trigger = {
	exists = scope:wc_accused
	exists = scope:wc_case_generation
	scope:wc_accused = {
		has_variable = wc_inquisition_stage
		exists = var:wc_inquisition_generation
		var:wc_inquisition_generation = scope:wc_case_generation
	}
}

wc_inquisition_case_valid_trigger = {
	wc_inquisition_case_matches_trigger = yes
	exists = scope:wc_inquisitor
	exists = scope:wc_court_title
	exists = global_var:wc_scarlet_crusade_title
	scope:wc_court_title = global_var:wc_scarlet_crusade_title
	scope:wc_court_title = {
		exists = holder
		holder = { is_alive = yes is_adult = yes has_faith = faith:scarletism }
	}
	scope:wc_inquisitor = {
		is_alive = yes
		is_imprisoned = no
		has_faith = faith:scarletism
	}
	scope:wc_accused = {
		is_alive = yes
		is_adult = yes
		has_faith = faith:scarletism
		faith.religious_head = scope:wc_inquisitor
		faith = { has_doctrine = tenet_purge_of_heresy has_doctrine = doctrine_spiritual_head }
		is_imprisoned_by = scope:wc_inquisitor
		trigger_if = {
			limit = { has_variable = wc_inquisition_commander }
			var:wc_inquisition_commander = scope:wc_court_title.holder
		}
	}
}

wc_inquisition_judge_valid_trigger = {
	wc_inquisition_case_valid_trigger = yes
	this = scope:wc_inquisitor
	scope:wc_accused = { var:wc_inquisition_stage = $STAGE$ }
}

wc_inquisition_command_valid_trigger = {
	wc_inquisition_case_valid_trigger = yes
	this = scope:wc_court_title.holder
	scope:wc_accused = { var:wc_inquisition_stage = 3 }
}

# Public traits only. The mark is an accusation, never proof of a separate crime.
# Pass the accused explicitly: the existing broad helper uses the event root's gender.
wc_inquisition_known_crime_trigger = {
	scope:wc_accused = {
		any_character_trait = {
			trait_is_criminal_in_faith_trigger = {
				TRAIT = this
				FAITH = faith:scarletism
				GENDER_CHARACTER = scope:wc_accused
			}
		}
	}
}
