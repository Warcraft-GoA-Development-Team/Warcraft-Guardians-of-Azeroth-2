# Gain Diplomacy Skill
study_scripture_outcome.140 = {
	type = character_event
	window = scheme_successful_event
	title = study_scripture_outcome.140.t
	desc = study_scripture_outcome.140.desc
	theme = faith
	left_portrait = {
		character = root
		animation = crying
		camera = camera_body_right
	}
	lower_left_portrait = scope:preacher

	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}

	cooldown = { years = 5 }

	trigger = {
		diplomacy < 8
		# Under lay clergy the realm's preachers are the chaplains; otherwise look to clergy subordinates and vassals
		trigger_if = {
			limit = {
				rite ?= { rite_has_doctrine = doctrine_theocracy_lay_clergy }
			}
			cp:councillor_court_chaplain ?= {
				pam_study_scripture_preacher_trigger = yes
			}
		}
		trigger_else = {
			OR = {
				any_subordinate = {
					pam_study_scripture_preacher_trigger = yes
				}
				any_vassal = {
					pam_study_scripture_preacher_trigger = yes
				}
			}
		}
	}

	immediate = {
		scope:owner = {
			save_scope_as = owner
		}
		if = {
			limit = {
				rite ?= { rite_has_doctrine = doctrine_theocracy_lay_clergy }
			}
			cp:councillor_court_chaplain = {
				save_scope_as = preacher
			}
		}
		else = {
			every_subordinate = {
				limit = {
					pam_study_scripture_preacher_trigger = yes
				}
				add_to_list = pam_ss_preacher_candidates
			}
			every_vassal = {
				limit = {
					pam_study_scripture_preacher_trigger = yes
					NOT = { is_in_list = pam_ss_preacher_candidates }
				}
				add_to_list = pam_ss_preacher_candidates
			}
			random_in_list = {
				list = pam_ss_preacher_candidates
				weight = {
					base = 1
					modifier = {
						add = diplomacy
					}
					modifier = {
						add = 200
						diplomacy >= 20
					}
					modifier = {
						add = 50
						has_trait = diplomat
					}
				}
				save_scope_as = preacher
			}
		}
	}

	option = {
		name = study_scripture_outcome.140.a

		add_diplomacy_skill = 1

		stress_and_fulfillment_impact = {
			base = minor_stress_impact_gain
			humble = minor_stress_impact_loss
			gregarious = minor_stress_impact_loss
			arrogant = medium_stress_impact_gain
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_trait = arrogant
			}
		}
	}

	option = {
		name = study_scripture_outcome.140.b

		add_piety = pam_study_scripture_piety_minor_value
		stress_and_fulfillment_impact = {
			gregarious = medium_stress_impact_gain
			trusting = minor_stress_impact_gain
		}
		ai_chance = {
			base = 0
		}
	}
}
