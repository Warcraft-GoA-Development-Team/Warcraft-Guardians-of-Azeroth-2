						any_de_jure_county = {
							holder = scope:actor
						}
					}
				}
			}
		}
		flag = land_send_option
		localization = TRADE_LAND_FOR_BETTER_AI_ACCEPTANCE
	}

	send_option = { # Promise a pilgrimage
		is_shown = {
			has_pam_dlc_trigger = yes
			pam_offer_pilgrimage_option_shown_trigger = { ACTOR = scope:actor }
		}
		is_valid = {
			pam_offer_pilgrimage_option_valid_trigger = { ACTOR = scope:actor }
		}
		flag = offer_pilgrimage
		localization = OFFER_PILGRIMAGE_INTERACTION_OPTION
	}

	send_options_exclusive = no

	on_accept = {
		if = {
			limit = { scope:hook = yes }
			scope:actor = {
				use_hook = scope:recipient
			}
		}
		scope:recipient = {
			custom_tooltip = anoinment_request_accepted_tt
			add_to_variable_list = {
				name = supported_anointments
				target = scope:actor
			}
		}
		if = {
			limit = {
				scope:gold_send_option = yes
			}
			scope:recipient = {
				add_short_term_gold = scope:actor.major_gold_value
			}
		}
		if = {
			limit = {
				scope:land_send_option = yes
			}
			create_title_and_vassal_change = {
				type = returned
				save_scope_as = change
			}
			scope:recipient = {
				every_held_title = {
					limit = {
						is_head_of_faith = no
					}
					every_de_jure_county = {
						limit = {
							holder = scope:actor
						}
						change_title_holder_include_vassals = {
							holder = scope:recipient
							change = scope:change
						}
					}
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		pam_offer_pilgrimage_effect = {
			ACTOR = scope:actor
			CLERIC = scope:recipient
		}
		scope:actor = {
			trigger_event = coronation_events.0400
		}
	}

	on_decline = {
		scope:actor = {
			add_opinion = {
				target = scope:recipient
				modifier = refused_my_anointment_opinion
			}
			if = {
				limit = {
					can_set_relation_rival_trigger = { CHARACTER = scope:recipient }
				}
				set_relation_rival = {
					target = scope:recipient
					reason = rival_refused_my_anointment_reason
				}
			}
			trigger_event = religious_interaction.32
		}
	}

	ai_will_do = {
		base = 50
		modifier = {
			has_game_rule = coronation_laws_off
			add = -25
		}
	}
