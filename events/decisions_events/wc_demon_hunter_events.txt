namespace = demon_hunter_events

scripted_effect become_landless_fallen_demon_hunter = {
    save_scope_as = adventurer
    if = {
        limit = {
            player_heir ?= {
                NOT = { this = root }
                is_adult = yes
            }
        }
        player_heir = {
            save_scope_as = laamp_heir
            save_scope_as = laamp_inheritor
            custom_tooltip = become_landless_adventurer_decision_title_scope_tt
        }
    }
    else_if = {
        limit = { is_independent_ruler = no }
        liege = {
            save_scope_as = laamp_liege
            save_scope_as = laamp_inheritor
            custom_tooltip = become_landless_adventurer_decision_title_scope_tt
        }
    }
    else = {
        custom_tooltip = become_landless_adventurer_decision_title_created_tt
        if = {
            limit = {
                NOT = { exists = scope:laamp_inheritor }
            }
            hidden_effect_new_object = {
                create_character = {
                    template = laamp_inheritor_template
                    culture = root.capital_county.culture
                    faith = root.capital_county.faith
                    dynasty = generate
                    location = root.capital_province
                    after_creation = {
                        save_scope_as = laamp_inheritor
                        add_character_flag = new_laamp_inheritor
                    }
                }
            }
        }
    }
    create_landless_adventurer_title_tooltip_effect = yes
    show_as_tooltip = {
        add_prestige = medium_prestige_gain
        add_character_modifier = {
            modifier = ep3_voluntary_laamp_character_modifier
            years = 5
        }
    }
    trigger_event = { id = ep3_laamps.0030 }
    # Debug tracking
    if = {
        limit = {
            is_ai = yes
            debug_only = yes
        }
        if = {
            limit = { exists = global_var:ai_voluntary_laamp_count }
            change_global_variable = {
                name = ai_voluntary_laamp_count
                add = 1
            }
        }
        else = {
            set_global_variable = {
                name = ai_voluntary_laamp_count
                value = 1
            }
        }
        debug_log = "AI became adventurer voluntarily"
        debug_log_scopes = yes
    }
    stress_impact = {
        craven = minor_stress_impact_gain
        content = minor_stress_impact_gain
        lazy = minor_stress_impact_gain
    }
}

demon_hunter_events.0001 = { # First, Kill the demon
    type = character_event
    content_source = dlc_GOA
    title = demon_hunter_0001_title
    desc = demon_hunter_0001_desc
    theme = prison
    window = visit_settlement_window

    immediate = {
        random_prisoner = {
            limit = {
                has_trait = being_demon
			    NOT = { has_trait = demon_hunter }
			    is_imprisoned_by = root
            }
			save_scope_as = first_demon_prey
		}
        root = { save_scope_as = newborn_demon_hunter }
        scope:first_demon_prey = {
            override_death_killer_effect = {
                death_reason = death_merge
                killer = scope:newborn_demon_hunter
            }
        }
    }

    left_portrait = {
        character = scope:newborn_demon_hunter
        animation = assassin
    }
    right_portrait = {
        character = scope:first_demon_prey
        animation = severelywounded
    }

    option = {
        name = demon_hunter_0001_A
        trigger_event = demon_hunter_events.0002
    }
}

demon_hunter_events.0010 = {
    type = character_event
    content_source = dlc_GOA
    title = demon_hunter_0010_title
    desc = demon_hunter_0010_desc
    theme = skull
    window = visit_settlement_window

    override_background = {
        reference = wc_background_tomb_of_sargeras
    }

    immediate = {
        exists = player_heir
        player_heir = {
            save_scope_as = child_of_hunter
        }
        root = {
            save_scope_as = new_fallen_hunter
        }
    }

    left_portrait = {
        character = scope:child_of_hunter
        animation = war_defender
    }
    right_portrait = {
        character = scope:new_fallen_hunter
        animation = sword_coup_degrace
    }

    option = { # Continue playing as the Demon Hunter turned Demon (AI will always pick this)
        add_internal_flag = dangerous
        name = demon_hunter_0010_A
        scope:new_fallen_hunter = {
            trigger_event = {
               id = demon_hunter_events.0011
               days = 1
            }
        }
        ai_chance = {
			base = 100
		}
        scope:child_of_hunter = {
            trigger_event = {
                id = demon_hunter_events.0012
                days = 2
            }
        }
    }
    option = { # Play as the Demon Hunter's Heir, if there is one (AI will never pick this)
        name = demon_hunter_0010_B
        trigger = {
            exists = player_heir
        }
        scope:new_fallen_hunter = {
            trigger_event = {
               id = demon_hunter_events.0011
               days = 1
            }
        }
        ai_chance = {
			base = 0
		}
        set_player_character = scope:child_of_hunter
        scope:child_of_hunter = {
            trigger_event = {
                id = demon_hunter_events.0012
                days = 2
            }
        }
    }
}
