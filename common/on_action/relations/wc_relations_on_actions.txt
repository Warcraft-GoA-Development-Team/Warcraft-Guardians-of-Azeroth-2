        add_relation_flag = {

}

on_death_relation_summoner = {
    effect = {
        scope:target = {
            if = {
                limit = {
                    root = {
                        NOT = {
                            has_character_flag = dispelled
                        }
                    }
                }
                root = {
                    add_character_flag = dispelled
                }
                execute_scoped_spell_effect = {
                    SPELL = flag:summon
                    EFFECT = cancel_channeling_spell_effect
                }
            }
        }
    }
}
