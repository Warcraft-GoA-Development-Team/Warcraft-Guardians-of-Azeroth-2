# Teacher scope. OWNER = the studier. Same rows, same order, as wc_study_faith_has_hidden_cult_trigger.
wc_study_faith_pick_hidden_cult_effect = {
	if = {
		limit = {
			wc_study_faith_cult_damned_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:death_god OWNER = $OWNER$ }
		}
		faith:death_god = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_burning_blade_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:orcish_fel OWNER = $OWNER$ }
		}
		faith:orcish_fel = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_atalai_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:cult_of_hakkar OWNER = $OWNER$ }
		}
		faith:cult_of_hakkar = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_dark_iron_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:ragnarism OWNER = $OWNER$ }
		}
		faith:ragnarism = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_usable_trigger = { CULT = faith:twilights_hammer_cult OWNER = $OWNER$ }
		}
		faith:twilights_hammer_cult = { save_scope_as = actual_faith }
	}
}

# Studier scope. Needs scope:actual_faith and scope:scheme. Becomes a secret cultist and learns one core tenet of the cult.
wc_study_faith_initiation_effect = {
	random_character_tenet = {
		knowledge_filter = unknown
		limit = {
			passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:actual_faith.main_rite }
		}
		save_scope_as = initiation_tenet
	}

	make_character_crypto_religionist_effect = { CRYPTO_RELIGION = scope:actual_faith }

	if = {
		limit = { exists = scope:initiation_tenet }
		add_known_tenet = scope:initiation_tenet
	}
	scope:scheme = { end_scheme = yes }
}

# Studier scope. Needs scope:scheme. Vanilla's study_faith_find_new_teacher_effect is local to
# study_faith_ongoing_events.txt, so wc_ files cannot call it. This copy stores progress before ending the scheme.
wc_study_faith_find_new_teacher_effect = {
	set_variable = {
		name = study_faith_progress
		value = scope:scheme.scheme_progress
	}
	set_variable = study_faith_dissatisfied_with_current_teacher

	scope:scheme = { end_scheme = yes }

	trigger_event = {
		id = study_faith_outcome.3000
		days = 3
	}
}

# Peasant scope (scope:chosen_peasant), called from study_faith_ongoing.1030's immediate inside hidden_effect_new_object.
# Needs scope:scheme and scope:dummy_gender. Always saves scope:newly_created_artifact.
wc_study_faith_create_peddler_artifact_effect = {
	set_variable = {
		name = religious_text
		value = scope:scheme.scheme_target_character.faith
	}
	create_artifact_book_effect = {
		OWNER = scope:chosen_peasant
		CREATOR = scope:dummy_gender
		SET_SUBJECT = flag:religious_text
		SET_TOPIC = flag:religious_text
	}
	scope:newly_created_artifact = {
		set_variable = {
			name = faith
			value = scope:scheme.scheme_target_character.faith
		}
	}
}
