# Balnazzar's hidden nature (added via add_dreadlord_disguise_secret_effect).

secret_dreadlord_disguise = {
	category = religious

	is_valid = {
		scope:secret_owner = { has_character_flag = wc_balnazzar_disguise_active }
	}

	is_criminal = {
		always = no
	}

	# A dreadlord unmasked as a fraud plausibly loses standing, same as any other secret.
	is_shunned = {
		always = no
	}

	on_discover = {
		scope:secret_owner = {
			add_to_variable_list = { name = wc_nathrezim_imposture_witnesses target = scope:discoverer }
			add_to_variable_list = { name = wc_nathrezim_demon_witnesses target = scope:discoverer }
		}
	}

	on_expose = {
		# Commit before event delivery: knowledge cannot be undone by a body swap.
		save_scope_as = secret
		scope:secret_owner = {
			set_variable = { name = wc_nathrezim_public_nature value = yes }
			if = {
				limit = {
					exists = global_var:balnazzar_dreadlord_story
					global_var:balnazzar_dreadlord_story = { story_owner = scope:secret_owner }
				}
				global_var:balnazzar_dreadlord_story = {
					set_variable = { name = wc_balnazzar_exposure_queued value = yes }
				}
				wc_balnazzar_change_suspicion_effect = { CHANGE = 0 }
			}
		}
		secret_exposed_notification_effect = yes
	}
}
