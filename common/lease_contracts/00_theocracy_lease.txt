# Definition of lease used for organized churches. Required for the game to run.
theocracy_lease = {
	hierarchy = { # hierarchy definition is required
		type = vassal
		ruler_valid = {
			rite = { rite_has_doctrine = doctrine_theocracy_temporal }
			NOT = { government_has_flag = government_is_theocracy }
		}
		liege_or_vassal_valid = {
			faith = scope:target.faith
		}
		lessee = cp:councillor_court_chaplain	# should be a councillor

		barony_valid = {
			title_province = { has_holding_type = church_holding }
		}
	}

	government = theocracy_government
	valid_holdings = { church_holding }

	ruler_share_min_opinion_from_lessee = 1
	hook_strength_max_opinion = strong

	tax_split = {
		lease_liege = 0.25

		ruler_max = 0.25	# Ceiling on the ruler's share, also quoted by the 'up to' tooltips.
		ruler = {
			if = {
				limit = {
					scope:lessee = {
						court_chaplain_endorse_override_trigger = yes
					}
				}
				add = {
					desc = THEOCRATIC_LEASE_RATIO_STRONG_HOOK
					value = 0.25
				}
			}
			else_if = {
				limit = {
					is_ai = yes
				}
				add = {
					desc = THEOCRATIC_LEASE_RATIO_FROM_OPINION
					value = {
						scope:lessee = { value = "opinion(scope:ruler)" }
					}
					divide = 400 # Divide opinion range -100/+100 with 200 gives range -0.25/+0.25
				}
			}

			save_temporary_value_as = current_total
			if = {
				limit = {
					scope:current_total > 0.25
				}
				max = {
					desc = THEOCRATIC_LEASE_RATIO_MAX_25
					value = 0.25
				}
			}
		}
	}

	levy_split = {
		lease_liege = 0.15

		ruler_max = 0.50	# Ceiling on the ruler's share, also quoted by the 'up to' tooltips.
		ruler = {
			if = {
				limit = {
					scope:lessee = { court_chaplain_endorse_override_trigger = yes }
				}
				add = {
					desc = THEOCRATIC_LEASE_RATIO_STRONG_HOOK
					value = 0.5
				}
			}
			else = {
				add = {
					desc = THEOCRATIC_LEASE_RATIO_FROM_OPINION
					value = {
						scope:lessee = { value = "opinion(scope:ruler)" }
					}
					divide = 200 # Divide opinion range -100/+100 with 200 gives range -0.5/+0.5
				}
			}
			max = 0.5
		}
	}
}
