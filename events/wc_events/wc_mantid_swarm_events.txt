# Mantid Swarming cycle: roughly every 100 years the Yshaarjism head sends the Swarm against the Pandaren behind the Serpent's Spine.
# Clock and war-end hooks: common/on_action/wc_mantid_swarm_on_actions.txt. Shared logic: common/scripted_effects/wc_mantid_swarm_effects.txt
#
# 0001 The Swarm Stirs            - mantid, one year before launch
# 0010 Swarm launch (AI)          - hidden, on the head of faith
# 0011 The Swarm Is Due           - player head picks a target or holds the Swarm back
# 0012 Swarm launch               - hidden, faith scope, starts the Great Holy War
# 0020 The Klaxxi Wake a Paragon  - mantid
# 0030 Culling at the Wall        - mantid, up to 2 per Swarm
# 0040 The Swarm Triumphant       - mantid, attacker won
# 0041 Tempered by Defeat         - mantid, defender won
# 0050 Signal Fires on the Wall   - Pandaren, at launch
# 0060 Shado-Pan Call to the Wall - Pandaren letter, at launch
# 0070 The Wall Falls             - Pandaren, attacker won
# 0071 The Wall Holds             - Pandaren, defender won
# 0080 Shado-Pan Veterans         - Shado-Pan leader, defender won
# 0090 War-end fan-out            - hidden, on the war's primary attacker

# Swarm launch. Root is faith:yshaarjism; scope:target_kingdom comes from 0010/0011.
# Mirrors great_holy_war.0001 without its Age of Crusades (variable_ghw_unlocked) gate.
wc_mantid_swarm.0012 = {
	scope = faith
	hidden = yes

	trigger = {
		exists = scope:target_kingdom
		exists = religious_head
		NOT = { exists = great_holy_war }
	}

	immediate = {
		save_scope_as = the_faith
		religious_head = { save_scope_as = ghw_sponsor }
		pam_determine_ghw_target_character_from_kingdom_effect = { FAITH = scope:the_faith }

		if = {
			limit = {
				NOT = { exists = scope:target_character }
			}
			debug_log = "wc_mantid_swarm: no target character in the chosen kingdom, retrying next year"
			remove_global_variable = wc_swarm_decision_pending
		}
		else = {
			# Launch the preparation phase
			start_great_holy_war = {
				target_character = scope:target_character
				target_title = scope:target_kingdom
				delay = { 365 545 }
			}
			scope:target_character.faith = { save_scope_as = target_faith } #Used for counterpledges.

			great_holy_war = {
				set_war_declarer = scope:ghw_sponsor
				if = {
					limit = { is_directed_ghw = no }
					undirected_ghw_recipient_selection_effect = yes
				}
			}
			if = {
				limit = {
					scope:ghw_sponsor.gold > 0
				}
				great_holy_war = {
					change_war_chest_gold = {
						value = scope:ghw_sponsor.gold
						divide = 2
						max = 5000
					}
				}
				scope:ghw_sponsor = {
					remove_short_term_gold = {
						value = gold
						divide = 2
						max = 5000
					}
				}
			}
			great_holy_war = {
				pledge_attacker = scope:ghw_sponsor
				set_variable = {
					name = redirect_cost
					value = 500
				}
			}
			#If no Claimant or Title Holder is available to be the Papal choice and the Crusade has not proceeded far enough to have a favorite beneficiary, select a fallback Recipient so that the CB's tooltip does not appear incomplete.
			if = { #Random Courtier in the Pope's court.
				limit = {
					NOT = { exists = scope:the_faith.great_holy_war.ghw_title_recipient }
					exists = scope:ghw_sponsor
					exists = scope:the_faith.great_holy_war
					scope:ghw_sponsor = {
						any_courtier = {
							is_adult = yes
							is_ai = yes
							NOR = {
								has_trait = excommunicated
								is_incapable_trigger = yes  # Warcraft
								has_trait = devoted
							}
							faith = scope:the_faith
						}
					}
				}
				scope:ghw_sponsor = {
					random_courtier = {
						limit = {
							scope:the_faith.main_rite = { rite_has_dominant_ruling_gender = prev }
							is_adult = yes
							is_ai = yes
							NOR = {
								has_trait = excommunicated
								is_incapable_trigger = yes  # Warcraft
								has_trait = devoted
							}
							faith = scope:the_faith
						}
						alternative_limit = {
							is_adult = yes
							is_ai = yes
							NOR = {
								has_trait = excommunicated
								is_incapable_trigger = yes  # Warcraft
								has_trait = devoted
							}
							faith = scope:the_faith
						}
						save_scope_as = fallback_recipient
						add_character_flag = flag_is_ghw_fallback_recipient
					}
				}
			}
			else_if = { #Else Generate a fallback recipient.
				limit = {
					NOT = { exists = scope:the_faith.great_holy_war.ghw_title_recipient }
				}
				create_character = {
					location = scope:ghw_sponsor.capital_province
					template = new_commander_character
					faith = scope:the_faith
					culture = scope:ghw_sponsor.culture
					save_scope_as = fallback_recipient
					gender_female_chance = {
						if = {
							limit = {
								scope:the_faith = { main_rite = { rite_has_doctrine = doctrine_gender_male_dominated } }
							}
							add = 0
						}
						else_if = {
							limit = {
								scope:the_faith = { main_rite = { rite_has_doctrine = doctrine_gender_female_dominated } }
							}
							add = 100
						}
						else = {
							add = 50
						}
					}
				}
				scope:fallback_recipient = { add_character_flag = flag_is_ghw_fallback_recipient }
			}
			great_holy_war = {
				set_variable = {
					name = var_fallback_recipient
					value = scope:fallback_recipient
				}
			}

			every_player = { #Inform players
				limit = {
					faith = scope:the_faith
					highest_held_title_tier >= tier_county
					this != scope:ghw_sponsor
				}
				trigger_event = great_holy_war.0003 #Send event immediately for player, then delay it for all the AIs.
			}
			every_player = { #Inform defending players
				limit = {
					OR = {
						faith = scope:target_faith #All infidels.
						any_liege_or_above = {
							scope:the_faith.great_holy_war = { has_pledged_defender = prev }
						}
						AND = {
							faith != scope:the_faith
							scope:the_faith.great_holy_war = { has_pledged_defender = prev }
						}
					}
				}
				trigger_event = great_holy_war.0050
			}

			#Attackers Pledge to the Crusade. (AI only)
			every_faith_ruler = {
				limit = {
					is_ai = yes
					is_landed = yes
					highest_held_title_tier >= tier_county
					NOR = {
						this = scope:ghw_sponsor
						root.great_holy_war = { has_pledged_attacker = prev }
						target_is_liege_or_above = scope:target_character
					}
				}
				trigger_event = {
					#AI pledging. There is a chance the AI might pledge after the Crusade starts, but it's a lot less likely.
					id = great_holy_war.0020
					days = { 50 450 }
				}
			}
			every_faith_holy_order = {
				trigger_event = {
					#Military Orders always join
					id = great_holy_war.0090
					days = { 30 60 }
				}
			}

			#Defenders Pledge to the Crusade. (AI only)
			scope:target_character = {
				religion = {
					every_faith = {
						every_faith_ruler = {
							limit = {
								is_ai = yes
								is_landed = yes
								highest_held_title_tier >= tier_county
								NOT = {
									root.great_holy_war = { has_pledged_defender = prev }
								}
							}
							trigger_event = {
								#AI counterpledging. Unlike attackers, defenders should be selected much faster during the preparation phase and then stop.
								id = great_holy_war.0021
								days = { 25 150 }
							}
						}
					}
				}
				faith = {
					every_faith_holy_order = {
						trigger_event = {
							# Military Orders always join
							id = great_holy_war.0091
							days = { 30 60 }
						}
					}
				}
			}

			set_global_variable = {
				name = wc_swarm_active
				value = yes
			}
			set_global_variable = {
				name = wc_swarm_target
				value = scope:target_kingdom
			}
			set_global_variable = {
				name = wc_swarm_culls
				value = 0
			}
			wc_swarm_reset_clock_effect = yes
			wc_swarm_save_scopes_effect = yes

			# Mantid flavor
			scope:empress ?= {
				trigger_event = {
					id = wc_mantid_swarm.0020
					days = { 30 90 }
				}
				trigger_event = {
					id = wc_mantid_swarm.0030
					days = { 180 365 }
				}
			}

			# Pandaren flavor, only when the Swarm hits the wall
			if = {
				limit = {
					scope:target_kingdom = { wc_swarm_wall_kingdom_trigger = yes }
				}
				scope:target_kingdom = {
					every_in_de_jure_hierarchy = {
						continue = { tier > tier_county }
						limit = {
							tier = tier_county
							exists = holder
						}
						holder.top_liege = {
							if = {
								limit = {
									NOT = { wc_swarm_is_aqir_trigger = yes }
									NOT = { is_in_list = swarm_signal_recipients }
								}
								add_to_temporary_list = swarm_signal_recipients
							}
						}
						if = {
							limit = { wc_swarm_wall_county_trigger = yes }
							holder = {
								if = {
									limit = {
										NOT = { wc_swarm_is_aqir_trigger = yes }
										NOT = { is_in_list = swarm_signal_recipients }
									}
									add_to_temporary_list = swarm_signal_recipients
								}
							}
						}
					}
				}
				every_in_list = {
					list = swarm_signal_recipients
					trigger_event = {
						id = wc_mantid_swarm.0050
						days = { 0 5 }
					}
				}
				if = {
					limit = { exists = title:k_shado_pan.holder }
					every_independent_ruler = {
						limit = { wc_swarm_shado_pan_letter_recipient_trigger = yes }
						trigger_event = {
							id = wc_mantid_swarm.0060
							days = { 15 45 }
						}
					}
				}
			}
		}
	}
}

# War-end fan-out. Root is the war's primary attacker; scope:swarm_defender comes from wc_swarm_record_war_end_effect.
wc_mantid_swarm.0090 = {
	hidden = yes

	immediate = {
		wc_swarm_save_scopes_effect = yes
		global_var:wc_swarm_final_target = { save_scope_as = final_target }

		if = {
			limit = { global_var:wc_swarm_outcome = flag:attacker_won }
			# Mantid triumph
			scope:swarm_head ?= {
				add_character_flag = {
					flag = wc_swarm_got_aftermath
					days = 1
				}
				trigger_event = wc_mantid_swarm.0040
			}
			if = {
				limit = {
					exists = scope:empress
					NOT = { scope:empress = scope:swarm_head }
				}
				scope:empress = {
					add_character_flag = {
						flag = wc_swarm_got_aftermath
						days = 1
					}
					trigger_event = wc_mantid_swarm.0040
				}
			}
			every_player = {
				limit = {
					NOT = { has_character_flag = wc_swarm_got_aftermath }
					faith = faith:yshaarjism
					wc_swarm_is_aqir_trigger = yes
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = wc_mantid_swarm.0090.triumph_toast
					left_icon = scope:final_target
				}
			}
			# The wall falls
			if = {
				limit = {
					scope:final_target = { wc_swarm_wall_kingdom_trigger = yes }
				}
				scope:final_target = {
					every_in_de_jure_hierarchy = {
						continue = { tier > tier_county }
						limit = {
							tier = tier_county
							wc_swarm_wall_county_trigger = yes
							exists = holder
						}
						holder = {
							if = {
								limit = {
									NOT = { wc_swarm_is_aqir_trigger = yes }
									NOT = { is_in_list = swarm_wall_holders }
								}
								add_to_temporary_list = swarm_wall_holders
							}
						}
					}
				}
				every_in_list = {
					list = swarm_wall_holders
					trigger_event = wc_mantid_swarm.0070
				}
			}
		}
		else = {
			# Mantid defeat
			every_ruler = {
				limit = {
					faith = faith:yshaarjism
					wc_swarm_is_aqir_trigger = yes
				}
				add_character_modifier = {
					modifier = wc_tempered_by_defeat
					years = 10
				}
			}
			scope:swarm_head ?= { trigger_event = wc_mantid_swarm.0041 }
			if = {
				limit = {
					exists = scope:empress
					NOT = { scope:empress = scope:swarm_head }
				}
				scope:empress = { trigger_event = wc_mantid_swarm.0041 }
			}
			# The wall holds
			if = {
				limit = {
					scope:final_target = { wc_swarm_wall_kingdom_trigger = yes }
				}
				scope:swarm_defender ?= {
					if = {
						limit = {
							NOT = { wc_swarm_is_aqir_trigger = yes }
						}
						add_to_temporary_list = swarm_wall_holders
					}
				}
				scope:final_target = {
					every_in_de_jure_hierarchy = {
						continue = { tier > tier_county }
						limit = {
							tier = tier_county
							wc_swarm_wall_county_trigger = yes
							exists = holder
						}
						holder = {
							if = {
								limit = {
									NOT = { wc_swarm_is_aqir_trigger = yes }
									NOT = { is_in_list = swarm_wall_holders }
								}
								add_to_temporary_list = swarm_wall_holders
							}
						}
					}
				}
				every_in_list = {
					list = swarm_wall_holders
					trigger_event = wc_mantid_swarm.0071
				}
				scope:shado_pan_leader ?= { trigger_event = wc_mantid_swarm.0080 }
			}
		}

		wc_swarm_cleanup_effect = yes
	}
}
