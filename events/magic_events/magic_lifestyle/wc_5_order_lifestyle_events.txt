# Time Stops
arcane_magic_lifestyle_events.1031 = { 
	title = arcane_magic_lifestyle_events.1031.t
	desc = arcane_magic_lifestyle_events.1031.desc
	content_source = dlc_GOA
	theme = order
	cooldown = { years = 10 }

	# Get some light reading done
	option = { 
		name = arcane_magic_lifestyle_events.1031.opt.a
		# Gain 200 XP in military, stewardship, or learning
		random_list = {
			33 = { 
				modifier = { 
					martial_lifestyle_perks > 26
					factor = 0
				}
				add_martial_lifestyle_xp = 200
			}
			33 = { 
				modifier = { 
					stewardship_lifestyle_perks > 26
					factor = 0
				}
				add_stewardship_lifestyle_xp = 200
			}
			33 = { 
				modifier = { 
					learning_lifestyle_perks > 26
					factor = 0
				}
				add_learning_lifestyle_xp = 200
			}
			33 = {
				trigger = {
					AND = { 
						martial_lifestyle_perks > 26
						stewardship_lifestyle_perks > 26
						learning_lifestyle_perks > 26
					}
				}
				add_order_magic_lifestyle_xp = 200
			}
		}

		ai_chance = {
			ai_value_modifier = {
				ai_rationality = 1
			}
		}
	}

	# WHAT - Finally, a chance to rest, feast, and decompress
	option = { 
		name = arcane_magic_lifestyle_events.1031.opt.b
		trigger = { NOT = { has_trait = diligent } }
		# HOW -  Lose a huge amount of stress
		add_stress = massive_stress_loss

		ai_chance = {
			ai_value_modifier = {
				ai_energy = -1
			}
			modifier = {
				add = 50
				stress_level >= 3
			}
			modifier = {
				add = 50
				stress_level >= 2
			}
			modifier = {
				add = 50
				stress_level >= 1
			}
		}
	}

	# WHAT - Snoop on your court members
	option = { 
		name = arcane_magic_lifestyle_events.1031.opt.c
		# HOW -  Random effect: 
		random_list = {
			# 25% - learn 2 secrets amongst your courtiers and vassals
			25 = { 
					custom_tooltip = wc_reveal_two_vassal_courtier_secrets_description
					wc_reveal_two_vassal_courtier_secrets_effect = yes
			}
			# 25% - 20% bonus hostile scheme power for 4 years
			25 = { 
				add_character_modifier = {
					modifier = wc_arcane_studied_hostile_scheming_modifier
					years = 4
				}
			}
			# 25% - 20% bonus personal scheme power for 4 years
			25 = { 
				add_character_modifier = {
					modifier = wc_arcane_studied_personal_scheming_modifier
					years = 4
				}
			}
			# 25% - uncover a plot against you if available (check ep1.flavor.2010)
			25 = {
				trigger = { 
					any_courtier_or_guest = { 
						any_scheme = {
							is_scheme_category = hostile
							scheme_target_character = root
						}
					}
				}
				custom_tooltip = wc_expose_a_courtier_guest_scheme_description
				every_courtier_or_guest = {
					limit = { 
						any_scheme = {
							is_scheme_category = hostile
							scheme_target_character = root
						}
					}
					add_to_list = scheme_havers
				}
				random_in_list = {
					list = scheme_havers
					random_scheme = { 
						limit = { scheme_target_character = root }  
						expose_scheme = yes
					}
				}
			}
		}

		ai_chance = {
			ai_value_modifier = {
				ai_vengefulness = 1
			}
		}
	}
}

# A young pyromaniac pt 1 - child
arcane_magic_lifestyle_events.7001 = { 
	title = arcane_magic_lifestyle_events.7001.t
	desc = arcane_magic_lifestyle_events.7001.desc
	content_source = dlc_GOA
	theme = order

	right_portrait = { 
		character = scope:pyromaniac
		animation = shame
	}

	left_portrait = { 
		character = root
		animation = thinking
	}

	weight_multiplier = {
		base = 0.1
		modifier = { 
			factor = { 
				value = 1
				add = perks_in_elemental_fire_magic_lifestyle
				add = perks_in_order_magic_lifestyle
			}
		}
	}

	immediate = { 
		create_MAGICTYPE_potential_child_effect = { MAGICTYPE = order }
		scope:magic_potential_child = { save_scope_as = pyromaniac }
	}

	# cool, let's burn stuff together
	option = { 
		name = arcane_magic_lifestyle_events.7001.opt_a
		# harm public opinion in province
		if = { 
			limit = {
				OR = { 
					location.county.holder != root
					is_landed = no
				}
			} 
			location.county.holder = {
				add_opinion = {
					modifier = outraged_opinion
					opinion = -30
					target = root
				}
			}
		}
		else = {
			random_held_title = {
				limit = {
					tier = tier_county
				}
				add_county_modifier = { 
					modifier = burned_county_for_fun
					years = 5
				}
				change_county_control = -25
			}
		}
		scope:pyromaniac = { 
			add_character_flag = burned_stuff_together
		}
		# add them to your court and teach them magic
		add_pyro_to_court_effect = yes
	}

	# compensate for damage and take them under my wing
	option = { 
		name = arcane_magic_lifestyle_events.7001.opt_b
		# pay some gold
		if = { 
			limit = { 
				OR = { 
					location.county.holder != root
					is_landed = no
				}
			}
			pay_short_term_gold = {
				target = location.county.holder
				gold = medium_gold_value
			}
		}
		else = { 
			remove_short_term_gold = minor_gold_value
		}
		# add them to your court and teach them magic
		add_pyro_to_court_effect = yes
	}

	# punish them for burining stuff
	option = {
		name = arcane_magic_lifestyle_events.7001.opt_c
		rightfully_imprison_character_effect = { 
			TARGET = scope:pyromaniac
			IMPRISONER = root
		}
	}
}

scripted_effect find_scholarly_location_effect = {
	random_realm_county = {
		limit = { 
			OR = { 
				is_higher_studies_location = yes
				has_county_modifier = pandidakterion_university_modifier
			}
		}
		save_scope_as = scholarly_location
	}
}

arcane_magic_lifestyle_events.7002 = { 
	title = arcane_magic_lifestyle_events.7002.t
	content_source = dlc_GOA
	theme = order

	
	desc = { 
		desc = arcane_magic_lifestyle_events.7002.desc.intro # the pyomancer has come of age
		first_valid = {
			triggered_desc = { # A fire mage just like me!
				trigger = { 
					has_focus = elemental_fire_magic_focus_3
				}
				desc = arcane_magic_lifestyle_events.7002.desc.pyro
			}
			desc = arcane_magic_lifestyle_events.7002.desc.ele # a dumbass shammy
		}
	}

	trigger = { 
		scope:pyromaniac = { 
			this = exists
			is_alive = yes
		}
	}

	# - expected outcome: he becomes an arcane magic user
	# - alternate outcome: he decides to become a shaman instead
	immediate = { 
		if = { 
			limit = {
				scope:pyromaniac = {
					NOR = {
						has_focus = elemental_fire_magic_focus_1
						has_focus = elemental_fire_magic_focus_2
						has_focus = elemental_fire_magic_focus_3
						has_focus = order_magic_focus_1
						has_focus = order_magic_focus_2
						has_focus = order_magic_focus_3
					}
				}
			}
		}
		scope:pyromaniac = {
			random_list = {
				75 = {
					modifier = { 
						trigger = { has_character_flag = burned_stuff_together }
						add = 100
					}
					set_focus = elemental_fire_magic_focus_3
					# setup magical abilities
					set_variable = { name = order_focus value = flag:fire }
					set_variable = { name = wc_order_magic_lifestyle_additional_perks_variable value = 4 }
					add_weighted_perks_in_order_magic_lifestyle_effect = yes
				}
				25 = {
					random_list = { 
						50 = { 
							set_focus = elemental_fire_magic_focus_1
						}
						50 = { 
							set_focus = elemental_fire_magic_focus_2
						}
					}
					set_variable = { name = wc_elemental_fire_magic_lifestyle_additional_perks_variable value = 4 }
					add_perks_in_elemental_MAGIC_lifestyle_effect = { MAGIC = elemental_fire }
				}
			}
		}
		# find a potential "scholarly location" to send them to
		if = { 
			limit = { 
				exists = top_liege 
			}
			top_liege = { 
				find_scholarly_location_effect = yes
			}
		}
		else_if = {
			limit = { 
				is_landed = yes
			}
			find_scholarly_location_effect = yes
		}
		else = { 
			# TODO - if landless adventurer, get a "scholarly location" from random patron who has one in their realm
		}
	}


	# keep them in your court -> 7003 & 7004
	option = { 
		name = {
			trigger = { 
				has_focus = elemental_fire_magic_focus_3
			}
			text = arcane_magic_lifestyle_events.7002.opt.a.pyro
		}
		name = {
			trigger = { 
				NOT = { has_focus = elemental_fire_magic_focus_3 }
