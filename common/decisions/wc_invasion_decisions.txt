join_invasion_war_decision = {
	picture = { reference = "gfx/interface/illustrations/army_status/moving.dds" }
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 1
		duchy = 1
		kingdom = 1
		empire = 1
		hegemony = 0
	}

	desc = join_invasion_war_decision_desc
	selection_tooltip = join_invasion_war_decision_tooltip

	is_shown = {
		is_landed = yes
		exists = liege

		liege = {
			any_character_war = {
				using_cb = great_invasion_cb
			}
		}
	}

	is_valid = {
		# You cannot join if you are already in a non-invasion war
		custom_tooltip = {
			text = join_invasion_war_decision_not_at_other_war_tt
			NOT = {
				any_character_war = {
					NOT = { using_cb = great_invasion_cb }
				}
			}
		}

		# You cannot join the same war twice
		custom_tooltip = {
			text = join_invasion_war_decision_not_already_joined_tt
			liege = {
				any_character_war = {
					using_cb = great_invasion_cb
					root = {
						NOR = {
							is_attacker_in_war = prev
							is_defender_in_war = prev
						}
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
	}

	effect = {
		liege = {
			random_character_war = {
				limit = {
					using_cb = great_invasion_cb
					root = {
						NOR = {
							is_attacker_in_war = prev
							is_defender_in_war = prev
						}
					}
				}
				save_scope_as = invasion_war
			}
		}

		# Join the war
		custom_tooltip = {
			text = join_invasion_war_decision_effect_tt
			if = {
				limit = { liege = { is_attacker_in_war = scope:invasion_war } }
				scope:invasion_war = { add_attacker = root }
			}
			else = {
				scope:invasion_war = { add_defender = root }
			}
		}
	}

	ai_will_do = {
		base = 100

		# Traits
		modifier = {
			add = 15
			has_trait = brave
		}
		modifier = {
			add = -25
			has_trait = craven
		}

		# Hooks
		modifier = {
			add = 25
			has_strong_hook = liege
		}
		modifier = {
			add = -35
			liege = {
				any_character_war = {
					using_cb = great_invasion_cb
					root = {
						NOR = {
							is_attacker_in_war = prev
							is_defender_in_war = prev
						}
					}
					primary_attacker = { save_temporary_scope_as = joinable_invasion_attacker }
				}
			}
			OR = {
				root = { has_strong_hook = scope:joinable_invasion_attacker }
				scope:joinable_invasion_attacker = {
					any_vassal = { root = { has_strong_hook = prev } }
				}
			}
		}

		# Council
		modifier = {
			add = 45
			is_councillor_of = top_liege
		}

		# Religion
		modifier = {
			add = -200
			liege = {
				any_character_war = {
					using_cb = great_invasion_cb
					root = {
						NOR = {
							is_attacker_in_war = prev
							is_defender_in_war = prev
						}
					}
					primary_attacker = { save_temporary_scope_as = joinable_invasion_attacker }
				}
			}
			OR = {
				religion = scope:joinable_invasion_attacker.religion
				AND = {
					exists = secret_faith
					secret_faith.religion = scope:joinable_invasion_attacker.religion
				}
			}
		}

		# Magic
		modifier = {
			add = -90
			liege = {
				any_character_war = {
					using_cb = great_invasion_cb
					root = {
						NOR = {
							is_attacker_in_war = prev
							is_defender_in_war = prev
						}
					}
					primary_attacker = { save_temporary_scope_as = joinable_invasion_attacker }
				}
			}
			OR = {
				AND = {
					OR = {
						death_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_death_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = death }
						MAGIC_is_ignored_trigger = { MAGIC = death }
					}
					scope:joinable_invasion_attacker = { MAGIC_is_approved_trigger = { MAGIC = death } }
				}
				AND = {
					OR = {
						life_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_life_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = life }
						MAGIC_is_ignored_trigger = { MAGIC = life }
					}
					scope:joinable_invasion_attacker = { MAGIC_is_approved_trigger = { MAGIC = life } }
				}
				AND = {
					OR = {
						order_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_order_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = order }
						MAGIC_is_ignored_trigger = { MAGIC = order }
					}
					scope:joinable_invasion_attacker = { MAGIC_is_approved_trigger = { MAGIC = order } }
				}
				AND = {
					OR = {
						disorder_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_disorder_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = disorder }
						MAGIC_is_ignored_trigger = { MAGIC = disorder }
					}
					scope:joinable_invasion_attacker = { MAGIC_is_approved_trigger = { MAGIC = disorder } }
				}
				AND = {
					OR = {
						light_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_light_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = light }
						MAGIC_is_ignored_trigger = { MAGIC = light }
					}
					scope:joinable_invasion_attacker = { MAGIC_is_approved_trigger = { MAGIC = light } }
				}
				AND = {
					OR = {
						shadow_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_shadow_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = shadow }
						MAGIC_is_ignored_trigger = { MAGIC = shadow }
					}
					scope:joinable_invasion_attacker = { MAGIC_is_approved_trigger = { MAGIC = shadow } }
				}
				AND = {
					OR = {
						elemental_fire_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_elemental_fire_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = elemental_fire }
						MAGIC_is_ignored_trigger = { MAGIC = elemental_fire }
					}
					scope:joinable_invasion_attacker.faith = {
						OR = {
							has_doctrine_parameter = elemental_fire_spirit_magic_approved
							has_doctrine_parameter = elemental_fire_decay_magic_approved
						}
					}
				}
				AND = {
					OR = {
						elemental_water_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_elemental_water_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = elemental_water }
						MAGIC_is_ignored_trigger = { MAGIC = elemental_water }
					}
					scope:joinable_invasion_attacker.faith = {
						OR = {
							has_doctrine_parameter = elemental_water_spirit_magic_approved
							has_doctrine_parameter = elemental_water_decay_magic_approved
						}
					}
				}
				AND = {
					OR = {
						elemental_earth_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_elemental_earth_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = elemental_earth }
						MAGIC_is_ignored_trigger = { MAGIC = elemental_earth }
					}
					scope:joinable_invasion_attacker.faith = {
						OR = {
							has_doctrine_parameter = elemental_earth_spirit_magic_approved
							has_doctrine_parameter = elemental_earth_decay_magic_approved
						}
					}
				}
				AND = {
					OR = {
						elemental_air_magic_lifestyle_perks >= 1
						any_secret = { secret_type = secret_elemental_air_magic_user }
					}
					OR = {
						MAGIC_is_illegal_trigger = { MAGIC = elemental_air }
						MAGIC_is_ignored_trigger = { MAGIC = elemental_air }
					}
					scope:joinable_invasion_attacker.faith = {
						OR = {
							has_doctrine_parameter = elemental_air_spirit_magic_approved
							has_doctrine_parameter = elemental_air_decay_magic_approved
						}
					}
				}
			}
		}
	}
}
