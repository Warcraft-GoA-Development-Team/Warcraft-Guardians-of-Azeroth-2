
# Outcome events for the wc_balnazzar_contingency scheme

##################################
# MAINTENANCE EVENTS 0000 - 0099
##################################

# Rolls success/failure on scheme completion
wc_balnazzar_contingency_outcome.0001 = {
	hidden = yes

	immediate = {
		scope:scheme.scheme_owner = { save_scope_as = owner }

		if = {
			limit = { scope:owner = { exists = var:wc_balnazzar_contingency_phases_elapsed } }
			save_scope_value_as = {
				name = phases_elapsed
				value = scope:owner.var:wc_balnazzar_contingency_phases_elapsed
			}
		}

		if = {
			limit = {
				exists = scope:phases_elapsed
				scope:phases_elapsed = 1
			}
			trigger_event = wc_balnazzar_contingency_outcome.1002
		}
		else_if = {
			limit = {
				exists = scope:phases_elapsed
				scope:phases_elapsed = 2
			}
			trigger_event = wc_balnazzar_contingency_outcome.1003
		}
		else = {
			random = {
				chance = scope:scheme.scheme_success_chance

				save_scope_value_as = {
					name = scheme_successful
					value = yes
				}
			}

			hidden_effect = {
				if = {
					limit = { exists = scope:scheme_successful }
					trigger_event = wc_balnazzar_contingency_outcome.2001
				}
				else = {
					trigger_event = wc_balnazzar_contingency_outcome.4001
				}
			}
		}
	}
}

		add_character_flag = balnazzar_contingency_ready
	}

		name = wc_balnazzar_contingency_outcome.2001.a

# FAILURE - 4000 - 4999
#######################################

# An Unfinished Thought - the preparation comes apart.
wc_balnazzar_contingency_outcome.4001 = {
	type = character_event
	window = scheme_failed_event
	title = wc_balnazzar_contingency_outcome.4001.title
	desc = wc_balnazzar_contingency_outcome.4001.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}

	option = {
		name = wc_balnazzar_contingency_outcome.4001.a

		add_stress = medium_stress_gain

		hidden_effect = {
			random = {
				chance = 40
				random_courtier = {
					limit = {
						is_alive = yes
						NOT = { has_trait = being_undead }
						NOT = { has_character_flag = balnazzar_noticed_something_wrong }
					}
					add_character_flag = balnazzar_noticed_something_wrong
					save_temporary_scope_as = balnazzar_witness

					trigger_event = {
						id = wc_balnazzar_story.0101
						days = { 10 30 }
					}
				}
			}

			random = {
				chance = 15
				random_secret = {
					type = secret_dreadlord_disguise
					expose_secret = root
				}
			}

			if = {
				limit = { exists = scope:scheme }
				scope:scheme = { end_scheme = yes }
			}
		}
	}
}
