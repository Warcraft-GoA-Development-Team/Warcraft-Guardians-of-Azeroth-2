is_weak_claimant_due_to_gender_trigger = {
	OR = {
		AND = {
			$RITE$ = { rite_has_parameter = female_claims_are_weak }
			$CHARACTER$ = { is_female = yes }
			NOT = {
				$RITE$ = { rite_has_parameter = tenet_hyperdulia_female_claims_are_strong }
			}
		}
		AND = {
			$RITE$ = { rite_has_parameter = male_claims_are_weak }
			$CHARACTER$ = { is_male = yes }
		}
	}
}

can_be_combatant_based_on_gender_trigger = {
	custom_description = {
		text = "can_be_combatant_based_on_gender"
		object = this
		subject = $ARMY_OWNER$
		OR = {
			trigger_if = { # Has the Royal Court and thus can modify pillars
				limit = {
					has_dlc_feature = diverge_culture # Can modify pillars
				}
				trigger_if = {
					limit = {
						$ARMY_OWNER$.culture = { has_cultural_parameter = martial_custom_male_only_combatant }
					}
					is_male = yes
				}
				trigger_else_if = {
					limit = {
						$ARMY_OWNER$.culture = { has_cultural_parameter = martial_custom_female_only_combatant }
					}
					is_female = yes
				}
				trigger_else = {}
			}
			trigger_else = { # Does not have the Royal Court and thus combatant is governed by faith
				OR = {
					$ARMY_OWNER$.rite = { rite_has_parameter = combatant_can_be_either_gender_if_no_roco }
					AND = {
						$ARMY_OWNER$.rite = { rite_has_parameter = combatant_must_be_male_if_no_roco }
						is_male = yes
					}
					AND = {
						$ARMY_OWNER$.rite = { rite_has_parameter = combatant_must_be_female_if_no_roco }
						is_female = yes
					}
				}
			}
			AND = {
				# No need to gender diversify this: if they're a right-gender shieldperson of the culture, then they've already got permission to fight from the previous AND conditions.
				has_trait = shieldmaiden
				$ARMY_OWNER$ = {
					culture = { has_cultural_parameter = has_access_to_shieldmaidens }
				}
			}
			# Event-based special exceptions
			OR = {
				has_character_modifier = ignores_gender_army_rules
				has_character_modifier = mpo_destined_leader_modifier
			}
			AND = {
				prowess >= 10
				$ARMY_OWNER$.culture = { has_cultural_parameter = high_prowess_ignores_knight_restrictions }
				NOT = { $ARMY_OWNER$.culture = { has_cultural_parameter = minimum_prowess_for_knights } }
			}
			AND = {
				prowess >= 12
				$ARMY_OWNER$.culture = {
					has_cultural_parameter = high_prowess_ignores_knight_restrictions
					has_cultural_parameter = minimum_prowess_for_knights
				}
			}
		}
	}
	# Separate custom_description so barred clergy get a clergy-specific reason, not the gender tooltip
	trigger_if = {
		limit = {
			is_clergy = yes
		}
		custom_description = {
			text = "clergy_can_be_combatant"
			object = this
			subject = $ARMY_OWNER$
			OR = {
				rite = { rite_has_parameter = clergy_can_fight }
				AND = {
					has_personal_tenet_flag = clergy_can_fight_personal
					NOT = {
						rite = { rite_has_parameter = pacifist }
					}
				}
				culture = { has_cultural_parameter = culture_clergy_can_fight }
				AND = {
					$ARMY_OWNER$.culture = {
						has_cultural_parameter = high_prowess_ignores_knight_restrictions
					}
					prowess >= 10
				}
			}
		}
	}
}

#Needs CHARACTER and RITE
relation_with_character_is_incestuous_in_rite_trigger = {
	OR = {
		# doctrine_consanguinity_dynastic; not within the same descent group
		AND = {
			$RITE$ = { rite_has_parameter = consanguinity_restricted_dynastic_marriage }
			is_lowborn = no
			$CHARACTER$.dynasty ?= dynasty
		}	
		#doctrine_consanguinity_restricted; absolutely no family business
		AND = {
			$RITE$ = { rite_has_doctrine = doctrine_consanguinity_restricted }
			is_close_or_extended_family_of = $CHARACTER$ #[parents, children, siblings, grandparents, grandchildren, cousins, uncles, aunts, nephews, nieces]
		}
		#doctrine_consanguinity_cousins; the only acceptable incest is with your cousin
		AND = {
			$RITE$ = { rite_has_parameter = consanguinity_cousins_marriage }
			OR = {
	            is_close_family_of = $CHARACTER$ #[parents, children, siblings, grandparents, grandchildren]
	            is_uncle_or_aunt_of = $CHARACTER$
	            is_nibling_of = $CHARACTER$
	        }
		}
		#doctrine_consanguinity_aunt_nephew_and_uncle_niece; extended family is ok
		AND = {
			$RITE$ = { rite_has_parameter = consanguinity_aunt_nephew_and_uncle_niece_marriage }
			is_close_family_of = $CHARACTER$
		}
		#doctrine_consanguinity_unrestricted; all forms of incest is acceptable
	}
}

relation_with_character_is_incestuous_in_my_or_lieges_rite_trigger = {
	OR = {
		AND = {
			rite = { #My rite doesn't approve
				save_temporary_scope_as = check_rite
			}
			relation_with_character_is_incestuous_in_rite_trigger = { CHARACTER = $CHARACTER$ RITE = scope:check_rite }
		}
		AND = { #Liege's rite doesn't approve
			liege ?= {
				rite = {
					save_temporary_scope_as = check_rite
				}
			}
			relation_with_character_is_incestuous_in_rite_trigger = { CHARACTER = $CHARACTER$ RITE = scope:check_rite }
		}
	}
}

murdering_character_is_kinslaying_in_rite_trigger = {
	OR = {
		is_close_or_extended_family_of = $CHARACTER$
		dynasty = $CHARACTER$.dynasty
	}
	$RITE$ = { NOT = { rite_has_parameter = kinslaying_accepted } }
}

murdering_character_is_kinslaying_in_my_or_same_dynasty_lieges_rite_trigger = {
	save_temporary_scope_as = kinslayer_me
	rite = {
		save_temporary_scope_as = check_rite
	}
	trigger_if = {
		limit = {
			liege ?= { dynasty = scope:kinslayer_me.dynasty }
		}
		liege = {
			rite = {
				save_temporary_scope_as = check_rite_liege
			}
		}
	}
	OR = {
		murdering_character_is_kinslaying_in_rite_trigger = { CHARACTER = $CHARACTER$ RITE = scope:check_rite } #My rite doesn't approve
		AND = { #Liege's rite doesn't approve
			liege ?= { dynasty = scope:kinslayer_me.dynasty }
			murdering_character_is_kinslaying_in_rite_trigger = { CHARACTER = $CHARACTER$ RITE = scope:check_rite_liege }
		}
	}
}

					superior ?= scope:recipient
				}
				AND = {
					government_has_flag = government_is_theocracy
					superior ?= scope:recipient
				}
			}
		}
	}
}

faith_can_have_heresy_trigger = {
	OR = {
		NOR = {
			has_doctrine = eastern_hostility_doctrine
			has_doctrine = sinitic_hostility_doctrine
		}
		has_doctrine = heresy_hostility_doctrine #Zoroastrians also get heresies, despite being Eastern.
	}
}

faith_is_in_same_rf = {
	trigger_if = {
		limit = { is_in_family = rf_abrahamic }
		$FAITH$ = { is_in_family = rf_abrahamic }
	}
	trigger_else_if = {
		limit = { is_in_family = rf_pagan }
		$FAITH$ = { is_in_family = rf_pagan }
	}
	trigger_else_if = {
		limit = { is_in_family = rf_eastern }
		$FAITH$ = { is_in_family = rf_eastern }
	}
	trigger_else_if = {
		limit = { is_in_family = rf_sinitic }
		$FAITH$ = { is_in_family = rf_sinitic }
	}
	trigger_else = {
		always = no
	}
}

faith_is_syncretic_with = {
	$FAITH$ = {
		trigger_if = {
			limit = { has_doctrine_parameter = unreformed }
			prev = { has_doctrine_parameter = unreformed_syncretic_actor }
		}
		trigger_else_if = {
			limit = { religion_tag = christianity_religion }
			prev = { has_doctrine_parameter = christian_syncretic_actor }
		}
		trigger_else_if = {
			limit = { religion_tag = islam_religion }
			prev = { has_doctrine_parameter = islamic_syncretic_actor }
		}
		trigger_else_if = {
			limit = { religion_tag = judaism_religion }
			prev = { has_doctrine_parameter = jewish_syncretic_actor }
		}
		trigger_else_if = {
			limit = { is_in_family = rf_eastern }
			prev = { has_doctrine_parameter = eastern_syncretic_actor }
		}
		trigger_else_if = {
			limit = { is_in_family = rf_sinitic }
			prev = { has_doctrine_parameter = sinitic_syncretic_actor }
		}
		trigger_else = {
			always = no
		}
	}
}
