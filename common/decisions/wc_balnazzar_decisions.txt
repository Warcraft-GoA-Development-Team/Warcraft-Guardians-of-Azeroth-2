# Repeatable - fires when a de jure duchy under k_stratholme is fully held but not yet converted.
# Scoped to k_stratholme only - k_lordaeron is a later, larger objective.
balnazzar_purify_reclaimed_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = independent

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 24
		empire = 24
		hegemony = 24
	}

	desc = balnazzar_purify_reclaimed_decision_desc

	# any_de_jure_duchy/every_de_jure_duchy don't exist; use any_realm_de_jure_duchy/every_realm_de_jure_duchy instead.
	# holder check: OR = { top_liege=root this=root } - a directly-held county can fail bare top_liege=root.
	is_shown = {
		has_trait = balnazzar_disguise
		any_realm_de_jure_duchy = {
			kingdom = title:k_stratholme
			any_de_jure_county = {
				count = all
				holder = { OR = { top_liege = root this = root } }
			}
			any_de_jure_county = {
				OR = {
					NOT = { faith = faith:scarletism }
					NOT = { culture = culture:lordaeronian }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	cost = {
		gold = major_gold_value
	}

	# ck3 quirk: deeply nested every_x/limit produces no auto-generated tooltip (blank Effects panel).
	# Wrap in custom_tooltip, same fix as the is_valid bug above.
	effect = {
		custom_tooltip = balnazzar_purify_reclaimed_decision_effect_tooltip

		every_realm_de_jure_duchy = {
			limit = {
				kingdom = title:k_stratholme
				any_de_jure_county = {
					count = all
					holder = { OR = { top_liege = root this = root } }
				}
			}
			every_de_jure_county = {
				set_county_faith = faith:scarletism
				set_county_culture = culture:lordaeronian

				# Purification isn't free - burns the county's development
				# back to nothing and leaves a lasting opinion/control scar.
				change_development_level = { subtract = development_level }
				add_county_modifier = {
					modifier = balnazzar_purge_scarred_modifier
					years = 10
				}
			}
		}

		add_piety = medium_piety_value

		# ponytail: only checked right after a purge - counties converted another way never trigger the capstone. Acceptable, unlikely case.
		# trigger_event fires from root (character), not from inside a title scope (wrong-scope error otherwise).
		if = {
			limit = {
				any_realm_de_jure_duchy = {
					count = all
					kingdom = title:k_stratholme
					any_de_jure_county = {
						count = all
						holder = { OR = { top_liege = root this = root } }
						faith = faith:scarletism
					}
				}
			}
			trigger_event = {
				id = wc_balnazzar_story.1008
				days = { 3 10 }
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 20

		# Cost alone doesn't stop AI spam - only consider once gold is comfortable.
		modifier = {
			factor = 0
			NOT = { gold >= major_gold_value }
		}
	}
}
