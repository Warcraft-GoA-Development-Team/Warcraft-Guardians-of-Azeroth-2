# Raises $ANTIPOPE$ as an independent rival Head of Faith, sponsored by $SPONSOR$
pam_establish_antipope_effect = {
	$ANTIPOPE$ = { save_scope_as = antipope }
	$SPONSOR$ = { save_scope_as = sponsor }
	# A sponsored antipope takes up their sponsor's rite
	if = {
		limit = {
			scope:antipope != scope:sponsor
			scope:antipope = {
				NOT = { rite = scope:sponsor.rite }
			}
		}
		scope:antipope = {
			set_character_rite = scope:sponsor.rite
		}
	}
	if = {
		limit = { exists = scope:antipope.capital_county }
		scope:antipope.capital_county = { save_scope_as = capital_county }
	}
	else = {
		scope:sponsor.capital_county = { save_scope_as = capital_county }
	}
	scope:antipope = {
		faith = { save_scope_as = pam_antipope_faith }
		if = {
			limit = { exists = capital_province }
			capital_province = { save_scope_as = pam_antipope_capital }
		}
		hidden_effect_new_object = {
			if = {
				limit = {
					faith.religious_head_title.tier = tier_kingdom
				}
				create_dynamic_title = {
					tier = kingdom
					name = {
						first_valid = {
							triggered_desc = {
								trigger = { exists = scope:pam_antipope_capital }
								desc = PAM_ANTIPOPE_TITLE_NAME
							}
							desc = PAM_ANTIPOPE_TITLE_NAME_NO_CAPITAL
						}
					}
				}
			}
			else = {
				create_dynamic_title = {
					tier = duchy
					name = {
						first_valid = {
							triggered_desc = {
								trigger = { exists = scope:pam_antipope_capital }
								desc = PAM_ANTIPOPE_TITLE_NAME
							}
							desc = PAM_ANTIPOPE_TITLE_NAME_NO_CAPITAL
						}
					}
				}
			}
			scope:new_title = {
				save_scope_as = pam_antipope_title
				# Faith challenger list is dropped by code if the rite detaches into its own faith
				set_variable = { name = pam_antipope_office }
				set_destroy_if_invalid_heir = yes
				set_no_automatic_claims = yes
				set_definitive_form = yes
				set_always_follows_primary_heir = yes
				set_can_be_named_after_dynasty = no
				set_capital_county = scope:capital_county
				set_landless_title = yes
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = pam_antipope_title_change
				add_claim_on_loss = no
			}
			scope:pam_antipope_title = {
				change_title_holder = {
					holder = scope:antipope
					change = scope:pam_antipope_title_change
				}
			}
			resolve_title_and_vassal_change = scope:pam_antipope_title_change
			# Needs a holder before it can derive a coat of arms
			scope:pam_antipope_title = {
				generate_coa = religious_title
			}
			set_primary_title_to = scope:pam_antipope_title
		}
		# GOVERNMENT
		if = {
			limit = {
				NOT = { government_has_flag = government_is_theocracy }
				scope:sponsor != this
				theocratic_lay_clergy_trigger = yes
				trigger_if = {
					limit = {
						rite = { rite_has_doctrine = doctrine_theocracy_lay_clergy }
					}
					is_landed = no
				}
			}
			if = {
				limit = { like_christianity_religion_trigger = yes } #Warcraft
				change_government = ecclesiastical_government
			}
			else = { change_government = theocracy_government }
		}
		# A rival Head of Faith answers to no secular liege
		if = {
			limit = { NOT = { this = top_liege } }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = pam_antipope_independence_change
				add_claim_on_loss = no
			}
			becomes_independent = { change = scope:pam_antipope_independence_change }
			resolve_title_and_vassal_change = scope:pam_antipope_independence_change
		}
	}
	scope:antipope = {
		if = {
			limit = { exists = clerical_elector_title }
			pam_destroy_cardinal_title_effect = { CARDINALATE = clerical_elector_title }
		}
	}
	if = {
		limit = { scope:antipope = scope:sponsor }
		hidden_effect_new_object = {
			scope:pam_antipope_title = {
				sponsor_new_religious_head_challenger = scope:pam_antipope_title
			}
		}
		pam_antipope_declared_catalyst_effect = yes
	}
	else = {
		hidden_effect_new_object = {
			scope:sponsor.primary_title ?= {
				sponsor_new_religious_head_challenger = scope:pam_antipope_title
			}
		}
		if = {
			limit = { has_pam_dlc_trigger = yes }
			pam_make_antipope_clerical_puppet_effect = {
				ANTIPOPE = scope:antipope
				SPONSOR = scope:sponsor
			}
		}
		# The sponsor gains a favor hook over the antipope they installed
		scope:sponsor = {
			add_hook_if_possible_default_length_effect = {
				TARGET = scope:antipope
				TYPE = favor_hook
			}
		}
		scope:antipope = {
			add_opinion = {
				modifier = grateful_opinion
				target = scope:sponsor
				opinion = 25
			}
		}
		pam_antipope_sponsored_catalyst_effect = yes
	}
	scope:antipope = {
		if = {
			limit = { exists = clerical_elector_title }
			pam_destroy_cardinal_title_effect = { CARDINALATE = clerical_elector_title }
			situation:the_christian_church ?= {
				add_manual_participant = prev
			}
		}
	}
	# Flavourization flag
	scope:antipope = {
		add_character_flag = antipope_flag
		hidden_effect_new_object = {
			pam_family_becomes_prelate_effect = {
				TITLE = scope:pam_antipope_title
				KIND = flag:antipope
			}
		}
	}
	# CE3 achievement: Guido's Guiding Light
	if = {
		limit = {
			scope:antipope = { this = character:et_milano_1066_1 }
		}
		scope:antipope = {
			add_achievement_global_variable_effect = {
				VARIABLE = ce3_02_guido_declared_antipope
				VALUE = yes
			}
		}
	}
}

# Root is the antipope losing the office
pam_strip_antipope_office_effect = {
	save_temporary_scope_as = pam_removed_antipope
	faith = {
		save_temporary_scope_as = pam_removed_antipope_faith
		every_religious_head_challenger = {
			limit = { holder ?= scope:pam_removed_antipope }
			save_temporary_scope_as = pam_title_to_destroy
			# Release the sponsor's grip before the sponsorship record goes away
			hidden_effect = {
				scope:pam_removed_antipope_faith = { remove_religious_head_challenger = scope:pam_title_to_destroy }
			}
			scope:pam_removed_antipope = {
					clear_puppet_status = yes
					remove_puppet_modifiers_effect = yes
					destroy_title = scope:pam_title_to_destroy
			}
		}
	}
	# On rite-detach, code unregisters the challenger before any script runs
	# The marked office title is then all that is left of it
	every_held_title = {
		limit = { has_variable = pam_antipope_office }
		save_temporary_scope_as = pam_title_to_destroy
		scope:pam_removed_antipope = {
			clear_puppet_status = yes
			remove_puppet_modifiers_effect = yes
			destroy_title = scope:pam_title_to_destroy
		}
	}
	remove_character_flag = antipope_flag
}

pam_remove_antipope_effect = {
	if = {
		limit = {
			scope:new_hof ?= { pam_is_antipope_trigger = yes }
		}
		scope:new_hof = { pam_strip_antipope_office_effect = yes }
	}
	else_if = {
		limit = { pam_is_antipope_trigger = yes }
		pam_strip_antipope_office_effect = yes
	}
}

pam_remove_sponsored_antipope_effect = {
	$SPONSOR$ = { save_temporary_scope_as = pam_sponsor_lookup }
	$FAITH$ = {
		every_religious_head_challenger = {
			limit = {
				challenger_sponsor ?= {
					holder ?= scope:pam_sponsor_lookup
				}
			}
			holder ?= { pam_strip_antipope_office_effect = yes }
		}
	}
}

pam_antipope_become_hof_effect = {
	$WINNER$ = { save_scope_as = new_hof }
	scope:new_hof.faith = { save_scope_as = antipope_faith }
	scope:antipope_faith = {
		if = {
			limit = { exists = religious_head_title }
			religious_head_title = { save_scope_as = hof_title }
		}
	}
	scope:antipope_faith.religious_head ?= { save_scope_as = old_hof }
	if = {
		limit = { exists = scope:old_hof }
		if = {
			limit = {
				scope:antipope_faith = { has_doctrine_parameter = spiritual_head_of_faith }
				scope:old_hof = { exists = capital_county }
			}
			scope:old_hof = {
				capital_county = { save_scope_as = old_hof_capital }
			}
		}
		create_title_and_vassal_change = {
			type = usurped
			save_scope_as = hof_titles_change
			add_claim_on_loss = no
		}
		scope:old_hof = {
			every_held_title = {
				limit = {
					tier >= tier_county
				}
				change_title_holder = {
					holder = scope:new_hof
					change = scope:hof_titles_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:hof_titles_change
	}
	if = {
		limit = {
			scope:new_hof = {
				has_title = title:k_papal_state
				faith = { has_doctrine_parameter = has_clerical_electors }
				NOT = { has_realm_law = theocratic_elective_succession_law }
			}
		}
		scope:new_hof = {
			add_realm_law_skip_effects = theocratic_elective_succession_law
		}
	}
	if = {
		limit = { exists = scope:hof_title }
		scope:antipope_faith = {
			set_religious_head_title = scope:hof_title
		}
	}
	if = {
		limit = {
			scope:antipope_faith = { has_doctrine_parameter = spiritual_head_of_faith }
			exists = scope:old_hof_capital
			scope:new_hof = {
				is_landed = yes
				NOT = { capital_county = scope:old_hof_capital }
			}
		}
		scope:new_hof = {
			set_realm_capital = scope:old_hof_capital
		}
	}
	scope:new_hof = {
		pam_set_pope_rite_mainline_effect = yes
	}
	pam_remove_antipope_effect = yes
}

pam_antipope_sponsor_vassalize_hof_effect = {
	$SPONSOR$ = { save_scope_as = sponsor }
	if = {
		limit = {
			exists = scope:sponsor
			exists = scope:new_hof
			scope:sponsor != scope:new_hof
			scope:sponsor = {
				is_alive = yes
				is_landed = yes
			}
			scope:new_hof = {
				is_alive = yes
				is_landed = yes
			}
			scope:sponsor.highest_held_title_tier > scope:new_hof.highest_held_title_tier
		}
		scope:new_hof = {
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = antipope_hof_vassal_change
				add_claim_on_loss = no
			}
			change_liege = {
				liege = scope:sponsor
				change = scope:antipope_hof_vassal_change
			}
			resolve_title_and_vassal_change = scope:antipope_hof_vassal_change
		}
	}
}

pam_antipope_challenge_hof_victory_effect = {
	pam_challenge_hof_war_save_participants_effect = yes
	scope:defender = {
		pam_challenge_hof_war_persist_scopes_effect = {
			OUTCOME = flag:attacker_victory
		}
	}
	# Set before he takes office: that title change is what fires the succession reaction
	if = {
		limit = {
			exists = scope:pam_ch_hof_challenger
			exists = scope:pam_ch_hof_sponsor
		}
		scope:pam_ch_hof_challenger = {
			set_variable = {
				name = challenger_sponsor
				value = scope:pam_ch_hof_sponsor
				days = 7
			}
		}
	}
	if = {
		limit = {
			scope:attacker = { pam_is_antipope_trigger = yes }
		}
		scope:attacker.faith = {
			every_religious_head_challenger = {
				limit = { holder ?= scope:attacker }
				challenger_sponsor ?= {
					holder ?= { save_scope_as = challenger_sponsor }
				}
			}
		}
		scope:attacker = {
			pam_antipope_become_hof_effect = { WINNER = scope:attacker }
		}
	}
	else_if = {
		limit = {
			scope:attacker = { pam_is_antipope_sponsor_trigger = yes }
		}
		scope:attacker = { save_scope_as = challenger_sponsor }
		scope:attacker.faith = {
			every_religious_head_challenger = {
				limit = {
					challenger_sponsor ?= { holder ?= scope:attacker }
				}
				holder ?= { save_scope_as = antipope_winner }
			}
		}
		if = {
			limit = { exists = scope:antipope_winner }
			pam_antipope_become_hof_effect = { WINNER = scope:antipope_winner }
			pam_antipope_sponsor_vassalize_hof_effect = { SPONSOR = scope:challenger_sponsor }
		}
	}
	# Only hold a judgment if the old Head of Faith actually lost the seat.
	if = {
		limit = {
			NOT = { scope:attacker.faith.religious_head ?= scope:defender }
		}
		pam_challenge_hof_war_on_attacker_victory_events_effect = yes
	}
}

pam_challenge_hof_war_save_participants_effect = {
	scope:attacker = { save_scope_as = pam_ch_hof_attacker }
	scope:defender = { save_scope_as = pam_ch_hof_old_hof }
	if = {
		limit = {
			scope:attacker = { pam_is_antipope_trigger = yes }
		}
		scope:attacker = { save_scope_as = pam_ch_hof_challenger }
		scope:attacker.faith = {
			every_religious_head_challenger = {
				limit = {
					holder ?= scope:attacker
					NOT = {
						challenger_sponsor ?= { holder ?= scope:attacker }
					}
				}
				challenger_sponsor ?= {
					holder ?= { save_scope_as = pam_ch_hof_sponsor }
				}
			}
		}
	}
	else_if = {
		limit = {
			scope:attacker = { pam_is_antipope_sponsor_trigger = yes }
		}
		scope:attacker = { save_scope_as = pam_ch_hof_sponsor }
		scope:attacker.faith = {
			every_religious_head_challenger = {
				limit = {
					challenger_sponsor ?= { holder ?= scope:attacker }
				}
				holder ?= { save_scope_as = pam_ch_hof_challenger }
			}
		}
	}
}

pam_challenge_hof_war_try_execute_effect = {
	$VICTIM$ = { save_scope_as = pam_ch_hof_victim }
	$EXECUTIONER$ = { save_scope_as = pam_ch_hof_executioner }
	pam_challenge_hof_war_excommunicate_if_possible_effect = { TARGET = scope:pam_ch_hof_victim }
	scope:pam_ch_hof_victim = {
		set_variable = {
			name = pam_ch_hof_fate
			value = flag:executed
			days = 30
		}
		set_variable = {
			name = pam_ch_hof_executioner
			value = scope:pam_ch_hof_executioner
			days = 7
		}
		if = {
			limit = { is_ai = no }
			pam_challenge_hof_war_notify_victim_fate_effect = {
				VICTIM = scope:pam_ch_hof_victim
				JUDGE = scope:pam_ch_hof_executioner
			}
		}
		else = {
			death = {
				killer = scope:pam_ch_hof_executioner
				death_reason = death_execution
			}
		}
	}
}

pam_challenge_hof_war_on_defender_victory_events_effect = {
	scope:defender = {
		if = {
			limit = { is_ai = no }
			trigger_event = {
				id = pam_antipope_events.0010
				days = 1
			}
		}
		else = {
			pam_challenge_hof_war_ai_defender_victory_judgment_effect = yes
			pam_challenge_hof_war_fire_followup_events_effect = yes
		}
	}
}

pam_challenge_hof_war_on_attacker_victory_events_effect = {
	scope:attacker.faith.religious_head ?= {
		save_scope_as = pam_ch_hof_new_hof
		set_variable = {
			name = pam_ch_hof_var_carrier
			value = scope:defender
			days = 30
		}
		# Only now do we know the new Head of Faith
		scope:defender = {
			set_variable = {
				name = pam_ch_hof_new_hof
				value = scope:pam_ch_hof_new_hof
				days = 30
			}
		}
		# The event finds the war's people through the old Head of Faith
		if = {
			limit = {
				exists = scope:attacker
				scope:attacker = { is_ai = no }
			}
			scope:attacker = {
				set_variable = {
					name = pam_ch_hof_var_carrier
					value = scope:defender
					days = 30
				}
				trigger_event = {
					id = pam_antipope_events.0012
					days = 1
				}
			}
		}
		else_if = {
			limit = { is_ai = no }
			trigger_event = {
				id = pam_antipope_events.0012
				days = 1
			}
		}
		else = {
			pam_challenge_hof_war_ai_attacker_victory_judgment_effect = yes
			pam_challenge_hof_war_fire_followup_events_effect = yes
		}
	}
}

pam_challenge_hof_war_fire_followup_events_effect = {
	# The letter always comes from whoever now holds the seat, never from a lay sponsor.
	if = {
		limit = { has_variable = pam_ch_hof_var_carrier }
		var:pam_ch_hof_var_carrier = {
			pam_challenge_hof_war_restore_scopes_from_vars_effect = yes
			if = {
				limit = { var:pam_ch_hof_outcome ?= flag:attacker_victory }
				scope:pam_ch_hof_new_hof ?= { save_scope_as = pam_ch_hof_winner }
			}
			else = {
				save_scope_as = pam_ch_hof_winner
			}
		}
	}
	else = {
		pam_challenge_hof_war_restore_scopes_from_vars_effect = yes
		if = {
			limit = { var:pam_ch_hof_outcome ?= flag:attacker_victory }
			scope:pam_ch_hof_new_hof ?= { save_scope_as = pam_ch_hof_winner }
		}
		else = {
			save_scope_as = pam_ch_hof_winner
		}
	}
	if = {
		limit = { NOT = { exists = scope:pam_ch_hof_winner } }
		faith.religious_head ?= { save_scope_as = pam_ch_hof_winner }
	}
	pam_challenge_hof_war_fire_faith_letter_effect = yes
}

pam_challenge_hof_war_fire_faith_letter_effect = {
	every_player = {
		limit = {
			exists = scope:pam_ch_hof_winner
			faith = scope:pam_ch_hof_winner.faith
			NOT = { this = scope:pam_ch_hof_winner }
			NAND = {
				exists = scope:pam_ch_hof_attacker
				this = scope:pam_ch_hof_attacker
			}
			NOT = { has_variable = pam_ch_hof_got_fate_event }
		}
		trigger_event = {
			id = pam_antipope_events.0015
			days = 2
		}
	}
}

		SPONSOR = scope:defender
		FAITH = scope:defender.faith
	}

		add_piety = major_piety_gain
		add_piety_level = 1
		if = {
			limit = {
				faith = {
					exists = religious_head
					religious_head != scope:attacker
				}
			}
			faith.religious_head = {
				add_opinion = {
					modifier = grateful_opinion
					target = scope:attacker
					opinion = 30
				}
			}
			add_hook_if_possible_default_length_effect = {
				TARGET = faith.religious_head
				TYPE = deposed_antipope_hook
			}
		}
		if = {
			limit = {
				has_personal_tenet_flag = tenet_apostolic_succession_antipope_opposition
			}
			add_legitimacy_effect = {
				LEGITIMACY = medium_legitimacy_gain
			}
			change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		}
	}
	pam_antipope_removed_catalyst_effect = yes
}
