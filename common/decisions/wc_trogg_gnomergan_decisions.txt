invade_gnomeregan_decision = {
	title = {
		first_valid = {
			desc = invade_gnomeregan_decision
		}
	}
	desc = {
		first_valid = {
			desc = invade_gnomeregan_decision.desc
		}
	}
	
	ai_check_interval = 12
	decision_group_type = major
	picture = { reference = "gfx/interface/illustrations/decisions/gnomergan.dds" }
	selection_tooltip = invade_gnomeregan_decision_tt

	is_shown = {
		ROOT = {
			has_trait = creature_trogg
			is_landed = yes
		}

		current_year > 603

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_gnomergan_decision
			}
		}
	}
	

	cost = {
		gold = 10
		prestige = 100
	}
	
	is_valid = {
		title:k_gnomeregan.holder = {
			has_trait = creature_gnome
		}
	}

	effect = {
		custom_tooltip = invade_gnomergan_decision_effect.tt

		trigger_event = trogg_invasion.0001

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_gnomergan_decision
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100

		modifier = {
			add = -50
			title:k_gnomeregan.holder = {
				is_allied_to = title:k_ironforge.holder
			}
		}

		modifier = {
			add = 10
			title:k_gnomeregan.holder = {
				is_at_war = yes
			}
		}

		modifier = {
			add = 75
			title:k_gnomeregan.holder = {
				OR = {
					is_at_war_with = title:k_lordaeron.holder
					is_at_war_with = title:k_lordaeron.holder.liege
				}
			}
		}
	}
}

spread_radiation_decision = {
	title = {
		first_valid = {
			desc = spread_radiation_decision
		}
	}
	desc = {
		first_valid = {
			desc = spread_radiation_decision.desc
		}
	}
	
	ai_check_interval = 12
	decision_group_type = major
	picture = { reference = "gfx/interface/illustrations/decisions/gnomergan.dds" }
	selection_tooltip = spread_radiation_decision_tt

	is_shown = {
		has_trait = creature_gnome
		has_trait = irradiated
		is_landed = yes
	}
	
	effect = {
		custom_tooltip = spread_radiation_decision_effect.tt

		hidden_effect = {
			every_courtier_or_guest = {
				add_trait = irradiated
				set_culture = culture:leper_gnome
				set_character_faith = faith:glowbound
			}

			if = {
				limit = {
					any_realm_county = {
						OR = {
							has_county_modifier = wc_high_debris_damage_modifier
							has_county_modifier = wc_medium_debris_damage_modifier
							has_county_modifier = wc_small_debris_damage_modifier 
						}				
					}
				}

				every_realm_county = {
					remove_all_county_modifier_instances = wc_high_debris_damage_modifier
					remove_all_county_modifier_instances = wc_medium_debris_damage_modifier
					remove_all_county_modifier_instances = wc_small_debris_damage_modifier
				}
			}
		}
	}


	cost = {
		prestige = 30
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}

	cooldown = { 
		years = 3
	}
}

gnomeregan_rebuilding_clear_debris_decision = {
	title = {
		first_valid = {
			desc = gnomeregan_rebuilding_clear_debris_decision
		}
	}
	desc = {
		first_valid = {
			desc = gnomeregan_rebuilding_clear_debris_decision.desc
		}
	}
	
	ai_check_interval = 12
	decision_group_type = major
	picture = { reference = "gfx/interface/illustrations/decisions/gnomergan.dds" }
	selection_tooltip = gnomeregan_rebuilding_clear_debris_decision_tt

	is_shown = {
		NOT = {
			has_trait = irradiated
		}

		any_realm_county = {
			OR = {
				has_county_modifier = wc_high_debris_damage_modifier
				has_county_modifier = wc_medium_debris_damage_modifier
				has_county_modifier = wc_small_debris_damage_modifier 
			}				
		}
	}
	
	effect = {
		custom_tooltip = gnomeregan_rebuilding_clear_debris_decision_effect.tt
		hidden_effect = {
			random_realm_county = {
				limit = {
					OR = {
						has_county_modifier = wc_high_debris_damage_modifier
						has_county_modifier = wc_medium_debris_damage_modifier
						has_county_modifier = wc_small_debris_damage_modifier 					
					}
				}
				if = {
					limit = {
						has_county_modifier = wc_high_debris_damage_modifier
					}

					remove_county_modifier = wc_high_debris_damage_modifier
					add_county_modifier = wc_medium_debris_damage_modifier
				}
				else_if = {
					limit = {
						has_county_modifier = wc_medium_debris_damage_modifier
					}

					remove_county_modifier = wc_medium_debris_damage_modifier
					add_county_modifier = wc_small_debris_damage_modifier
				}
				else_if = {
					limit = {
						has_county_modifier = wc_small_debris_damage_modifier
					}

					remove_county_modifier = wc_small_debris_damage_modifier
				}
			}
		}

	}


	cost = {
		prestige = 100
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 70
	}

	cooldown = { 
		years = 1
	}
}
