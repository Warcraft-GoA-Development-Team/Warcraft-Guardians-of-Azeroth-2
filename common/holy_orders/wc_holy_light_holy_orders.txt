# Brotherhood of Northshire
# Only created through enforce_the_brotherhood_scripted_effect or set_up_historical_military_orders_effect
ho_holy_light_brotherhood_of_northshire = {

	type = military
	government = holy_order_government
	title = d_brotherhood_of_northshire
	name = "holy_order_brotherhood_of_northshire"
	coat_of_arms = "ho_brotherhood_of_northshire"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			scope:province.county ?= title:c_northshire
		}
		trigger_else = {
			# Never through the generic holy order decision
			scope:founder = { has_variable = enforce_the_brotherhood_founder }
			scope:founder.faith = { wc_valid_light_faith_for_northshire_trigger = yes }
			brotherhood_have_not_existed_trigger = yes

			exists = scope:province
			exists = title:c_northshire.holder
			scope:province.county.holder = title:c_northshire.holder
		}
	}

	weight = {
		value = 10000
		#Trying to put them here on game start
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			add = 1000000
		}
	}

	men_at_arms = { order_serjeants }
}

# Champions of Peace
ho_holy_light_champions_of_peace = {

	type = military
	government = holy_order_government
	title = d_champions_of_peace
	name = "holy_order_champions_of_peace"
	coat_of_arms = "ho_champions_of_peace"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			scope:province.county ?= title:c_windswept
		}
		trigger_else = {
			scope:founder.faith = faith:holy_light
		}
	}

	weight = {
		value = 10
		#Trying to put them here on game start
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			add = 1000000
		}
		#More likely if founder is compassionate
		if = {
			limit = {
				scope:founder = {
					ai_compassion >= high_positive_ai_value
				}
			}
			add = 50
		}
	}

	men_at_arms = { order_knights }
}

		value = 10
