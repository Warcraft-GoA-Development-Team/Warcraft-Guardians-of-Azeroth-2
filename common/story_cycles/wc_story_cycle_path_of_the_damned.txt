story_path_of_the_damned = {
	on_setup = {
		set_global_variable = { # Save the story globally
			name = path_of_the_damned_story
			value = this 
		}

		set_variable = {
			name = story_stage
			value = 1
		}
		# Save variables
		set_variable = { name = malganis value = story_owner.var:malganis }
		set_variable = { name = frostmourne value = story_owner.var:frostmourne_weapon }
		set_variable = { name = king_of_lordaeron value = story_owner.var:king_of_lordaeron }

		if = { # Funeral of Terenas
			limit = {
				exists = var:king_of_lordaeron
				story_owner = { any_killed_character = { this = root.var:king_of_lordaeron } }
			}

			story_owner = { save_scope_as = killer }
			funeral_of_terenas_effect = yes
		}

		if = { # Arthas things
			limit = { story_owner = character:60003 }

			story_owner = { # Arthas raise invincible
				add_character_flag = is_in_naming_event
				trigger_event = {
					id = wc_path_of_the_damned.0001
					days = 2
				}
			}
		}

		story_owner = { # Path of the Damned introduction
			trigger_event = {
				id = wc_path_of_the_damned.1001
				days = 14
			}
		}
	}

	on_end = {
		remove_global_variable = path_of_the_damned_story
	}

	on_owner_death = {
		# Save scopes
		story_owner = { save_scope_as = old_champion }
		title:h_scourge.holder = { save_scope_as = lich_king }

		if = {
			limit = { exists = scope:lich_king }
			defeat_of_the_champion_effect = yes
		}
		else = { end_story = yes }
	}

	effect_group = { # A new Champion
		days = { 30 60 }

		trigger = {
			story_owner = {
				# Is the Lich King
				any_held_title = { this = title:h_scourge }
				has_trait = lich_king

				# Frostmourne
				any_character_artifact = { has_variable = frostmourne }
			}
		}

		triggered_effect = {
			effect = {
				story_owner = {
					save_scope_as = lich_king # Save scope

					ordered_vassal_or_below = { # Find a new champion
						limit = {
							is_adult = yes
							has_religion = prev.religion
						}
						order_by = martial
						max = 1

						save_scope_as = new_champion
					}

					if = {
						limit = { exists = scope:new_champion }

						every_character_artifact = {
							limit = { has_variable = frostmourne }
							save_scope_as = frostmourne
						}

						# Notification and effect
						trigger_event = wc_path_of_the_damned.0101
						every_vassal_or_below = { trigger_event = wc_path_of_the_damned.0101 }
					}
				}

				if = {
					limit = { exists = scope:new_champion }

					make_story_owner = scope:new_champion
				}
			}
		}
	}
}
