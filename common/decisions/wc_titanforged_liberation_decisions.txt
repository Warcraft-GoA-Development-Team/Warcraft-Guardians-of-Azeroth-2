liberate_chaos_duchy_decision = {
	decision_group_type = wc_titanforged
	picture = { reference = "gfx/interface/illustrations/decisions/liberate_lordaeron.dds" }

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	desc = liberate_chaos_duchy_decision_desc
	selection_tooltip = liberate_chaos_duchy_decision.tt

	is_shown = {
		faith = { is_titanforged_liberator_faith_trigger = yes }
		highest_held_title_tier >= tier_duchy
	}

	is_valid = {
		faith = { is_titanforged_liberator_faith_trigger = yes }
		any_realm_de_jure_duchy = {
			completely_controls = this
			any_de_jure_county = {
				faith = { is_chaos_aligned_faith_trigger = yes }
			}
			NOT = {
				has_variable_list = wc_titanforged_liberated_duchies
			}
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
	}

	effect = {
		hidden_effect = {
			ordered_realm_de_jure_duchy = {
				limit = {
					completely_controls = this
					any_de_jure_county = {
						faith = { is_chaos_aligned_faith_trigger = yes }
					}
				}
				save_scope_as = liberated_duchy
			}
			root = {
				add_to_variable_list = {
					name = wc_titanforged_liberated_duchies
					target = scope:liberated_duchy
				}
			}
			scope:liberated_duchy = {
				every_de_jure_county = {
					limit = {
						faith = { is_chaos_aligned_faith_trigger = yes }
					}
					reforge_liberated_county_effect = yes
				}
			}
		}
		trigger_event = wc_titanforged_liberation.0001
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

reconstruct_liberated_duchy_decision = {
	decision_group_type = wc_titanforged
	picture = { reference = "gfx/interface/illustrations/decisions/decontamination_gnomeregan.dds" }

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 6
		kingdom = 6
		empire = 6
		hegemony = 6
	}

	desc = reconstruct_liberated_duchy_decision.desc
	selection_tooltip = reconstruct_liberated_duchy_decision.tt

	is_shown = {
		faith = { is_titanforged_liberator_faith_trigger = yes }
		has_variable_list = wc_titanforged_liberated_duchies
	}

	is_valid = {
		faith = { is_titanforged_liberator_faith_trigger = yes }
		if = {
			limit = { NOT = { exists = var:order_points } }
			custom_tooltip = reconstruct_liberated_duchy_decision_not_enough_order_points_tt
		}
		else = {
			var:order_points >= 20
		}
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
	}

	effect = {
		if = {
			limit = { NOT = { exists = var:order_points } }
			set_variable = { name = order_points value = 0 }
		}
		change_variable = { name = order_points subtract = 20 }
		random_in_list = {
			variable = wc_titanforged_liberated_duchies
			save_scope_as = target_duchy
		}
		scope:target_duchy = {
			random_de_jure_county = {
				change_development_level = 1
			}
		}
	}

	cooldown = {
		years = 1
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 70
	}
}
