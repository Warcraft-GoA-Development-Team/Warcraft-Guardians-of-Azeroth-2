can_have_blood_trait_trigger = {
    NOT = { has_trait = creature_$BLOOD$ }
    OR = { 
        real_father ?= { has_trait = creature_$BLOOD$ }
        mother ?= { has_trait = creature_$BLOOD$ }
        AND = {
            real_father ?= { has_trait = blood_$BLOOD$ }
            mother = { has_trait = blood_$BLOOD$ }
        }
    }
}

has_blood_trait_trigger = {
	has_trait = blood
