pam_decision_confession = {
	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					spiritual_fulfillment < 0
				}
				desc = pam_decision_confession_name_low
			}
			desc = pam_decision_confession_name
		}
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_confession.dds"
	}
	decision_group_type = religious
	
	desc = pam_decision_confession_desc
	selection_tooltip = pam_decision_confession_tooltip

	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	sort_order = 110
	
	is_shown = {
		#has_spiritual_fulfillment_type = christian_fulfillment Warcraft
		root.rite ?= {
			has_at_least_tenet_status = {
				tenet = tenet_confession
				status = permitted
			}
		}
		has_pam_dlc_trigger = yes
	}

	is_valid = {
		NOT = { has_trait = excommunicated }
		custom_tooltip = {
			text = pam_decision_confession_sp_court_chaplain
			exists = cp:councillor_court_chaplain
		}
	}
	
	is_valid_showing_failures_only = {
		is_alive = yes
		is_in_activity = no
		is_imprisoned = no
		cp:councillor_court_chaplain ?= {
			is_available = yes
		}
	}

	cooldown = { years = 5 }

	effect = {
		if = {
			limit = { spiritual_fulfillment < 0 }
			custom_tooltip = pam_decision_confession_tooltip_sp_effect
		}
		custom_tooltip = pam_decision_confession_tooltip_stress_effect
		custom_tooltip = pam_decision_events.0001.a.confession_random_outcome
		trigger_event = pam_decision_events.0001
	}

	ai_potential = {
		stress_level >= 1
	}

	ai_will_do = {
		base = 35
		modifier = {
			stress_level >= 2
			add = 30
		}
		modifier = {
			stress_level >= 3
			add = 30
		}
		modifier = {
			has_trait = humble
			add = 20
		}
		modifier = {
			has_trait = zealous
			add = 20
		}
		modifier = {
			has_trait = cynical
			add = -40
		}
	}
}

# Adopt Theocratic Rule
pam_convert_to_theocracy_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}
	decision_group_type = realm
	desc = pam_convert_to_theocracy_decision_desc
	selection_tooltip = pam_convert_to_theocracy_decision_tooltip

	ai_check_interval = 0

	is_shown = {
		highest_held_title_tier <= tier_duchy
		highest_held_title_tier >= tier_county
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_flag = government_is_japan_feudal
			government_has_flag = government_is_mandala
		}
		trigger_if = {
			limit = {
				exists = faith.religious_head
			}
			faith.religious_head = {
				NOT = { government_has_flag = government_is_ecclesiastical } #blocked for Christians as they should seek the Ecclesiastical path
			}
		}
		has_pam_dlc_trigger = yes
		is_ruler = yes
	}

	is_valid = {
		custom_tooltip = {
			text = HOF_INTERACTION_UNREFORMED_FAITH
			NOT = {
				faith = { has_doctrine_parameter = unreformed }
			}
		}
		piety_level >= medium_piety_level
		liege = {
			trigger_if = {
				limit = {
					this != root
					government_has_flag = government_is_theocracy
				}
				custom_tooltip = {
					text = same_faith_as_liege
					OR = {
						religion = liege.religion
						faith = {
							faith_hostility_level = {
								target = root.liege.faith
								value <= faith_fully_accepted_level
							}
						}
					}
				}
				custom_tooltip = {
					text = same_culture_as_liege
					OR = {
						culture = liege.culture
						culture = {
							cultural_acceptance = { target = root.liege.culture value >= 90 }
						}
					}
				}
			}
			trigger_else = {
				NOT = { rite = { rite_has_doctrine = doctrine_theocracy_lay_clergy } }
			}
		}
		trigger_if = {
			limit = {
				faith = {
					exists = religious_head
				}
			}
			OR = {
				custom_tooltip = {
					text = empire_faith_gate_hof_temporal_is_hof
					this = faith.religious_head
				}
				custom_tooltip = {
					text = different_rite_than_hof
					rite != faith.main_rite
					rite = { "divergence(faith.main_rite)" >= faith.heresy_threshold }
				}
				rite = { rite_has_parameter = no_head_of_faith }
			}
		}
		trigger_if = {
			limit = {
				top_liege = this
				is_ai = no
			}
			custom_tooltip = {
				text = all_powerful_vassals_required_tt
				any_powerful_vassal = {
					count = all
					opinion = {
						target = root
						value >= 50
					}
				}
			}
		}
		trigger_if = {
			limit = { is_ai = yes }
			calc_true_if = {
				amount >= 2
				has_trait = zealous
				rite = { rite_has_doctrine = doctrine_theocracy_temporal }
				rite = { rite_has_tenet = tenet_communal_identity }
				culture = { has_cultural_pillar = ethos_spiritual }
				culture = { has_cultural_pillar = ethos_communal }
				culture = { has_cultural_tradition = tradition_monastic_communities }
				dynasty = { has_dynasty_perk = erudition_legacy_2 }
				dynasty = { has_dynasty_perk = erudition_legacy_4 }
				dynasty = { has_dynasty_perk = glory_legacy_5 }
				dynasty = { has_dynasty_perk = law_legacy_2 }
			}
		}
	}

	is_valid_showing_failures_only = {
		is_at_war = no
	}
	cost = {
		piety = massive_piety_value
		prestige = massive_prestige_value
	}

	effect = {
		show_as_tooltip = {
			primary_title = {
				every_title_heir = {
					add_opinion = {
						modifier = angry_opinion
						target = root
						opinion = -50
					}
				}
			}
			change_government = theocracy_government
		}
		hidden_effect = {
			every_held_title = {
				title_tier >= county
				every_title_heir = {
					add_to_list = heirs
				}
			}
			every_in_list = {
				list = heirs
				add_opinion = {
					modifier = angry_opinion
					target = root
					opinion = -50
				}
			}
		}
		government_type = {
			save_scope_as = old_government
		}
		add_piety_level = 1
		trigger_event = pam_decision_events.0009
		hidden_effect = {
			save_scope_as = theocratic_ruler
			every_vassal_or_below = {
				trigger_event = {
					id = pam_decision_events.0010
					days = 1
				}
			}
		}
	}

	ai_potential = {
		prestige >= massive_prestige_value
		piety >= massive_piety_value
		piety_level >= high_piety_level
	}

	ai_will_do = {
		base = 0
	}
}

# Restore Temporal Rule
pam_convert_theocracy_to_feudal_decision = {
	# picture = {
	# 	trigger = {
	# 		has_tgp_dlc_trigger = yes
	# 		liege = {
	# 			culture = { has_cultural_pillar = heritage_japonic }
	# 		}
	# 	}
	# 	reference = "gfx/interface/illustrations/decisions/tgp_dominance.dds"
	# }
	picture = {
		trigger = {
			liege = {
				government_has_flag = government_is_clan
			}
		}
		reference = "gfx/interface/illustrations/event_scenes/bp1_courtyard_mena.dds"
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = realm
	desc = pam_convert_theocracy_to_feudal_decision_desc
	selection_tooltip = pam_convert_theocracy_to_feudal_decision_tooltip

	ai_check_interval = 0

	is_shown = {
		government_has_flag = government_is_theocracy
		is_landed = yes
		has_pam_dlc_trigger = yes
		NOT = { any_held_title = { is_head_of_faith = yes } }
		rite.head_of_rite != root
	}

	is_valid = {
		trigger_if = {
			limit = {
				top_liege = this
				is_ai = no
			}
			prestige_level >= high_prestige_level
		}
		trigger_if = {
			limit = {
				top_liege = this
				is_ai = no
			}
			custom_tooltip = {
				text = all_powerful_vassals_required_tt
				any_powerful_vassal = {
					count = all
					opinion = {
						target = root
						value >= 50
					}
				}
			}
		}
		trigger_if = {
			limit = { is_ai = yes }
			calc_true_if = {
				amount >= 2
				prestige_level >= high_prestige_level
				has_trait = cynical
				culture = { has_cultural_pillar = ethos_bureaucratic }
				culture = { has_cultural_pillar = ethos_courtly }
				culture = { has_cultural_pillar = ethos_stoic }
				culture = { has_cultural_tradition = tradition_legalistic }
				has_realm_law = church_authority_3
				capital_county = { development_level > medium_development_level }
				dynasty ?= { has_dynasty_perk = law_legacy_5 }
				dynasty ?= { has_dynasty_perk = erudition_legacy_5 }
				NOT = { has_trait = zealous }
				NOT = { rite = { rite_has_doctrine = doctrine_theocracy_temporal } }
			}
		}
		is_landed = yes
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		custom_tooltip = {
			text = is_antipope_blocked_tt
			pam_is_antipope_trigger = no
		}
	}
	cost = {
		prestige = major_prestige_value
	}

	effect = {
		if = {
			limit = {
				government_has_flag = government_is_ecclesiastical
			}
			if = {
				limit = {
					exists = faith.religious_head
					can_set_relation_rival_trigger = { CHARACTER = faith.religious_head }
				}
				progress_towards_rival_effect = {
					REASON = rival_refuted_faith
					CHARACTER = faith.religious_head
					OPINION = default_rival_opinion
				}
			}
			else_if = {
				limit = {
					exists = rite.head_of_rite
					can_set_relation_rival_trigger = { CHARACTER = rite.head_of_rite }
				}
				progress_towards_rival_effect = {
					REASON = rival_refuted_faith
					CHARACTER = rite.head_of_rite
					OPINION = default_rival_opinion
				}
			}
			if = {
				limit = { exists = clerical_elector_title }
				destroy_title = clerical_elector_title
				remove_character_flag = cardinal_flag
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			every_held_title = {
				limit = {
					has_clerical_region = yes
				}
				change_title_holder_include_vassals = {
					holder = current_heir
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		trigger_event = major_decisions.0001
		convert_whole_realm_to_feudalism_effect = yes
		add_piety_level = -1
		hidden_effect = {
			save_scope_as = actor
			every_vassal = {
				trigger_event = {
					id = major_decisions.0002
					days = 3
				}
			}
		}
	}

	ai_potential = {
		prestige >= major_prestige_value
		piety_level < 1
		NOT = { government_has_flag = government_is_ecclesiastical }
	}

	ai_will_do = {
		base = 0
	}
}

		hegemony = 12
	}

	is_shown = {
		OR = {
			is_landed = yes
			government_has_flag = government_is_theocracy
		}
		faith ?= {
			OR = {
				has_doctrine_parameter = spiritual_head_of_faith
				has_doctrine_parameter = temporal_head_of_faith
			}
			NOT = { exists = religious_head }
			NOT = { has_doctrine_parameter = block_from_create_hof_parameter }
		}
	}

	is_valid = {
		is_landed = yes
		can_create_head_of_faith_title_trigger = { FAITH = root.faith } # Includes Schism requirements
	}

	cost = {
		treasury_or_gold = create_hof_gold_cost
		piety = create_hof_piety_cost
	}

	effect = {
		faith = { save_scope_as = faith }
		# Tell the player who will actually receive the title
		if = {
			limit = { faith = { has_doctrine_parameter = temporal_head_of_faith } }
			custom_tooltip = create_hof_decision_temporal_self_tt
		}
		else_if = {
			limit = { is_clergy = yes }
			custom_tooltip = create_hof_decision_spiritual_self_tt
		}
		else_if = {
			limit = { puppet:theological_agent_puppet ?= { rite = root.rite } }
			custom_tooltip = create_hof_decision_spiritual_puppet_tt
		}
		else_if = {
			limit = { puppet:external_theological_agent_puppet ?= { rite = root.rite } }
			custom_tooltip = create_hof_decision_spiritual_external_puppet_tt
		}
		else_if = {
			limit = { cp:councillor_court_chaplain ?= { rite = root.rite } }
			custom_tooltip = create_hof_decision_spiritual_realm_priest_tt
		}
		else = {
			custom_tooltip = create_hof_decision_spiritual_other_tt
		}
		if = {
			limit = {
				faith = { has_doctrine_parameter = spiritual_head_of_faith }
				NOT = { rite = faith.main_rite }
			}
			custom_tooltip = create_hof_decision_mainline_rite_tt
		}
		if = {
			limit = {
				faith = { has_doctrine_parameter = spiritual_head_of_faith }
				is_clergy = no
			}
			custom_tooltip = create_hof_decision_landed_hof_warning_tt
		}
		force_create_head_of_faith_title_effect = yes
	}

	ai_potential = {
		is_adult = yes
		# Defer to a same-faith liege; they should be the one establishing the HoF
		OR = {
			top_liege = this
			NOT = {
				any_liege_or_above = {
					faith = root.faith
				}
			}
		}
	}

	ai_will_do = {
		base = 100
	}
}

# Proclaim yourself a living voice of the faith and gain the herald trait
pam_proclaim_living_word_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_act_of_devotion.dds"
	}
	desc = pam_proclaim_living_word_decision_desc
	selection_tooltip = pam_proclaim_living_word_decision_tooltip
	decision_group_type = major
	sort_order = 120

	ai_check_interval_by_tier = {
		barony = 0
		county = 60
		duchy = 48
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	is_shown = {
		OR = {
			government_has_flag = government_is_theocracy
			has_perk = prophet_perk
		}
		NOR = {
			faith.religious_head ?= this
			has_trait = herald
		}
		has_pam_dlc_trigger = yes
	}

	is_valid = {
		calc_true_if = {
			amount >= 6
			has_perk = prophet_perk
			custom_tooltip = {
				text = pam_proclaim_living_word_decision_sf_tt
				spiritual_fulfillment >= spiritual_fulfillment_3
			}
			piety_level >= max_piety_level
			has_trait_xp = {
				trait = pilgrim
				value >= 100
			}
			has_trait_xp = {
				trait = lifestyle_scholar
				value >= 100
			}
			has_trait_xp = {
				trait = lifestyle_mystic
				value >= 100
			}
			custom_tooltip = {
				text = pam_proclaim_living_word_decision_archbishop_tt
				any_held_title = { has_clerical_region = yes }
			}
			custom_tooltip = {
				text = pam_proclaim_living_word_decision_puppet_tt
				OR = {
					puppet:theological_agent_puppet ?= { pam_is_archbishop_or_higher_trigger = yes }
					puppet:external_theological_agent_puppet ?= { pam_is_archbishop_or_higher_trigger = yes }
				}
			}
			custom_tooltip = {
				text = pam_proclaim_living_word_decision_realm_rite_tt
				is_landed = yes
				# count = all is satisfied by a single county, so the floor is what makes this a real bar
				any_sub_realm_county = {
					count >= 5
					rite = root.rite
				}
				any_sub_realm_county = {
					count = all
					rite = root.rite
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		NOT = { has_trait = excommunicated }
	}

	cost = {
		piety = 3000
		prestige = 1500
	}

	effect = {
		add_trait = herald
		custom_tooltip = pam_proclaim_living_word_decision_effect_tt
		trigger_event = pam_decision_events.0021
	}

	ai_potential = {
		ai_zeal >= 40
	}

	ai_will_do = {
		base = 100 # The requirements are the hard part; a zealous AI that qualifies should take it
	}
}

	}
	decision_group_type = major

	desc = pam_declare_rite_heresy_decision_desc
	selection_tooltip = pam_declare_rite_heresy_decision_tooltip
	confirm_text = pam_declare_rite_heresy_decision_confirm

	sort_order = 108

	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		this = faith.religious_head
		faith = { has_papacy_trigger = no } # The pope uses bulls
		faith ?= {
			any_faith_rite = {
				pam_rite_can_be_declared_heresy_trigger = yes
			}
		}
	}

	is_valid = {
		trigger_if = {
			limit = { has_trait = zealous }
			custom_tooltip = {
				text = pam_declare_rite_heresy_decision_valid_zealous_tt
				faith = {
					any_faith_rite = {
						pam_rite_can_be_declared_heresy_trigger = yes
					}
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = pam_declare_rite_heresy_decision_valid_tt
				faith = {
					any_faith_rite = {
						pam_rite_can_be_declared_heresy_trigger = yes
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_imprisoned = no
		custom_tooltip = {
			text = pam_rite_heresy_select_tt
			exists = scope:selected_item
		}
	}

	cooldown = { years = 5 }

	cost = {
		piety = major_piety_value
	}

	effect = {
		save_scope_as = bull_issuer
		scope:selected_item = { save_scope_as = heresy_rite }
		send_interface_toast = {
			title = pam_declare_rite_heresy_effect_toast_tt
			right_icon = scope:heresy_rite
			pam_declare_rite_heresy_effect = yes
		}
		faith = {
			every_faith_character = {
				limit = {
					is_ruler = yes
					this != scope:bull_issuer
				}
				trigger_event = { id = pam_decision_events.0012 days = 1 }
			}
		}
		if = {
			limit = { exists = scope:heresy_rite }
			hidden_effect = {
				if = {
					limit = {
						scope:heresy_rite.faith.religious_head ?= {
							NOT = { has_trait = heresiarch }
						}
					}
					scope:heresy_rite.faith.religious_head = { add_trait = heresiarch }
				}
				else_if = {
					limit = {
						scope:heresy_rite.main_rite.head_of_rite ?= {
							NOT = { has_trait = heresiarch }
						}
					}
					scope:heresy_rite.main_rite.head_of_rite = { add_trait = heresiarch }
				}
				# Warcraft
			}
		}
	}

	widget = {
		gui = "decision_view_widget_select_rite"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT"

		setup_items = {
			scope:actor = {
				faith = {
					every_faith_rite = {
						limit = {
							pam_rite_can_be_declared_heresy_trigger = yes
						}
						add_to_list = item_list
					}
				}
			}
		}

		default_item = {
			scope:actor = {
				faith = {
					random_faith_rite = {
						limit = {
							pam_rite_can_be_declared_heresy_trigger = yes
						}
						save_scope_as = selected_item
					}
				}
			}
		}

		is_item_valid = {
			pam_rite_can_be_declared_heresy_trigger = yes
		}

		ai_select_item = {
			scope:actor = {
				faith = {
					ordered_faith_rite = {
						limit = {
							pam_rite_can_be_declared_heresy_trigger = yes
						}
						order_by = "divergence(faith.main_rite)"
						save_scope_as = selected_item
					}
				}
			}
		}
	}

	ai_potential = {
		this = faith.religious_head
		faith = {
			any_faith_rite = {
				pam_rite_can_be_declared_heresy_trigger = yes
			}
		}
	}

	ai_will_do = {
		base = 20
		modifier = {
			add = 25
			has_trait = zealous
		}
		modifier = {
			add = 15
			has_trait = theologian
		}
		modifier = {
			add = -25
			has_trait = cynical
		}
		modifier = {
			add = -50
			rite = { rite_has_doctrine = doctrine_pluralism_pluralistic }
		}
		modifier = {
			add = -100
			scope:selected_item = scope:actor.rite
		}
	}
}

# Head of Faith: formally take a stance on a tenet known to the faith or learned personally.
pam_permit_tenet_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_story/pam_the_christian_church_stability.dds"
	}
	decision_group_type = major

	desc = pam_permit_tenet_decision_desc
	selection_tooltip = pam_permit_tenet_decision_tooltip
	confirm_text = pam_permit_tenet_decision_confirm

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		this = faith.religious_head
		faith = { has_papacy_trigger = no } # The pope uses bulls
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		custom_tooltip = {
			text = pam_tenet_stance_select_tt
			exists = scope:selected_item
		}
		custom_tooltip = {
			text = ecumenical_council_already_ongoing_tt
			NOT = { always = no } # Warcraft - ecumenical council emptied in GoA, never set
		}
	}

	cooldown = { years = 3 }

	cost = {
		piety = medium_piety_value
	}

	effect = {
		save_scope_as = pam_tenet_stance_issuer
		save_scope_as = pam_tenet_stance_permit
		scope:selected_item = { save_scope_as = pam_tenet_stance_tenet }
		faith = {
			set_tenet_status = {
				tenet = scope:selected_item
				status = permitted
			}
			change_fervor = {
				value = significant_fervor_loss
				desc = fervor_loss_pam_tenet_permitted
			}
			if = {
				limit = { has_tenet_popularity_trigger = yes }
				change_tenet_popularity = {
					target = scope:selected_item
					change = medium_tenet_popularity_gain
				}
			}
			pam_faith_cleanup_forbidden_personal_tenet_secrets_effect = yes
			every_faith_character = {
				limit = {
					is_ruler = yes
					this != scope:pam_tenet_stance_issuer
				}
				trigger_event = { id = pam_decision_events.0011 days = 1 }
			}
		}
	}

	widget = {
		gui = "decision_view_widget_select_tenet"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT"

		# List is tenets the HoF personally knows + tenets the rite knows
		setup_items = {
			scope:actor = {
				ordered_character_tenet = {
					knowledge_filter = known
					order_by = pam_raw_tenet_popularity_value
					max = 200
					check_range_bounds = no
					add_to_list = item_list
				}
			}
		}

		default_item = {
			scope:actor = {
				random_character_tenet = {
					knowledge_filter = known
					limit = { pam_tenet_can_be_permitted_trigger = yes }
					save_scope_as = selected_item
				}
			}
		}

		is_item_valid = {
			pam_tenet_can_be_permitted_trigger = yes
		}

		ai_select_item = {
			scope:actor = {
				if = { # A tenet the faith adores gets sanctioned whether or not the Head of Faith cares for it
					limit = {
						any_character_tenet = {
							knowledge_filter = known
							pam_tenet_can_be_permitted_trigger = yes
							pam_tenet_is_overwhelmingly_popular_trigger = { CHARACTER = scope:actor }
						}
					}
					ordered_character_tenet = {
						knowledge_filter = known
						limit = {
							pam_tenet_can_be_permitted_trigger = yes
							pam_tenet_is_overwhelmingly_popular_trigger = { CHARACTER = scope:actor }
						}
						order_by = pam_raw_tenet_popularity_value
						position = 0
						save_scope_as = selected_item
					}
				}
				else = {
					random_desired_tenet = {
						limit = {
							pam_tenet_can_be_permitted_trigger = yes
							pam_tenet_ai_may_permit_trigger = { CHARACTER = scope:actor }
						}
						only_positive = yes
						threshold = 1
						selection_count = 97
						weight = {
							base = 10
							modifier = {
								add = pam_tenet_popularity_impact_value
							}
						}
						save_scope_as = selected_item
					}
				}
			}
		}
	}

	ai_potential = {
		this = faith.religious_head
		OR = {
			any_desired_tenet = {
				only_positive = yes
				threshold = 1
				selection_count = 97
				root.rite = {
					NOR = {
						has_tenet_status = { tenet = prev status = permitted }
						has_tenet_status = { tenet = prev status = core }
					}
				}
				pam_tenet_ai_may_permit_trigger = { CHARACTER = root }
			}
			any_character_tenet = {
				knowledge_filter = known
				pam_tenet_can_be_permitted_no_actor_trigger = yes
				pam_tenet_is_overwhelmingly_popular_trigger = { CHARACTER = root }
			}
		}
	}

	ai_will_do = {
		base = 50
		modifier = {
			add = 20
			has_trait = zealous
		}
		modifier = {
			add = 20
			has_trait = theologian
		}
		modifier = {
			add = -30
			has_trait = cynical
		}

		modifier = {
			add = 200
			any_personal_tenet = {
				this = scope:selected_item
			}
		}
	}
}

		}

		}

		}

	}

# formally take a stance on a tenet known to the faith or learned personally.
pam_evangelize_personal_tenet_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_kingdom_of_heaven.dds"
	}
	decision_group_type = religious

	desc = pam_evangelize_personal_tenet_decision_desc
	selection_tooltip = pam_evangelize_personal_tenet_decision_tooltip
	confirm_text = pam_evangelize_personal_tenet_decision_confirm

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		has_tenet_popularity_trigger = yes
		is_playable_character = yes
		any_personal_tenet = {}
		has_pam_dlc_trigger = yes
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = pam_evangelize_personal_tenet_religious_authority_tt
			OR = {
				is_clergy = yes
				AND = { # The puppet route is reserved for players;
					is_ai = no
					has_theological_agent_puppet_trigger = yes
				}
			}
		}
		is_available_adult = yes
		custom_tooltip = {
			text = pam_evangelize_tenet_select_tt
			exists = scope:selected_item
		}
		custom_tooltip = {
			text = pam_evangelize_personal_tenet_valid_targets_tt
			OR = {
				any_subject = {
					pam_characters_to_evangelize_personal_tenet_trigger = yes
				}
				capital_county.clerical_region_title ?= {
					any_ruler_in_clerical_region = {
						NOT = {
							is_vassal_of = root
						}
						pam_characters_to_evangelize_personal_tenet_trigger = yes
					}
				}
			}
		}
	}

	cooldown = { years = 10 }

	cost = {
		piety = {
			value = medium_piety_value
			add = {
				add = evangelized_subjects_count
				add = evangelized_clerical_region_rulers_count
				multiply = 10
			}
		}
	}

	effect = {
		scope:selected_item = { save_scope_as = pam_evangelize_tenet }
		pam_tenet_popularity_on_evangelized_tenet_effect = {
			TENET = scope:pam_evangelize_tenet
			CHANGE = medium_tenet_popularity_gain
		}
		every_subject = {
			limit = {
				pam_characters_to_evangelize_personal_tenet_trigger = yes
			}
			add_to_list = evangelized_subjects
		}
		capital_county.clerical_region_title ?= {
			save_scope_as = clerical_region
			every_ruler_in_clerical_region = {
				limit = {
					NOT = {
						is_vassal_of = root
					}
					pam_characters_to_evangelize_personal_tenet_trigger = yes
				}
				add_to_list = evangelized_clerical_region_rulers
			}
		}
		save_scope_value_as = {
			name = evangelized_subjects_count
			value = list_size:evangelized_subjects
		}
		save_scope_value_as = {
			name = evangelized_clerical_region_rulers_count
			value = list_size:evangelized_clerical_region_rulers
		}
		if = {
			limit = {
				scope:evangelized_subjects_count > 0
				scope:evangelized_clerical_region_rulers_count > 0
			}
			custom_tooltip = can_pick_one_option_tt
		}
		if = {
			limit = { scope:evangelized_subjects_count > 0 }
			custom_tooltip = decision_every_subject_evangelized_personal_tenet_tt
		}
		if = {
			limit = { scope:evangelized_clerical_region_rulers_count > 0 }
			custom_tooltip = decision_every_clerical_region_ruler_evangelized_personal_tenet_tt
		}
		trigger_event = pam_decision_events.0020
	}

	widget = {
		gui = "decision_view_widget_select_tenet"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT_TENET"

		# We leave this unfiltered on purpose, because ai_select_item and ai_potential do the AI's skipping
		setup_items = {
			scope:actor = {
				ordered_personal_tenet = {
					order_by = {
						value = pam_raw_tenet_popularity_value
						multiply = -1
					}
					max = 200
					check_range_bounds = no
					add_to_list = item_list
				}
			}
		}

		default_item = {
			scope:actor = {
				ordered_personal_tenet = {
					order_by = pam_tenet_unpopularity_impact_value
					save_scope_as = selected_item
				}
			}
		}

		ai_select_item = {
			scope:actor = {
				random_personal_tenet = {
					limit = {
						save_temporary_scope_as = pam_evangelize_tenet_check
						scope:actor = {
							pam_evangelize_tenet_below_very_popular_trigger = yes
						}
					}
					weight = {
						base = 10
						modifier = {
							add = pam_tenet_unpopularity_impact_value
						}
						modifier = {
							scope:actor.rite  = {
								NOT = {
									has_tenet_status = {
										tenet = prev
										status = core
									}
								}
							}
							add = 10
						}
					}
					save_scope_as = selected_item
				}
			}
		}
	}

	ai_potential = {
		is_clergy = yes
		piety >= medium_piety_value
		any_personal_tenet = {
			save_temporary_scope_as = pam_evangelize_tenet_check
			root = {
				pam_evangelize_tenet_below_very_popular_trigger = yes
			}
		}
	}

	ai_will_do = {
		base = 50
		modifier = {
			add = 20
			has_trait = zealous
		}
		modifier = {
			add = 20
			has_trait = theologian
		}
		modifier = {
			add = -30
			has_trait = cynical
		}

		modifier = {
			add = 200
			any_personal_tenet = {
				root.rite  = {
					NOT = {
						has_tenet_status = {
							tenet = prev
							status = core
						}
					}
				}
				this = scope:selected_item
			}
		}

		modifier = { # Nobody in reach clings to a Popular belief worth luring them away from
			add = -200
			NOR = {
				any_subject = {
					pam_characters_to_evangelize_personal_tenet_trigger = yes
					any_personal_tenet = {
						pam_tenet_is_at_least_popular_trigger = { CHARACTER = root }
					}
				}
				capital_county.clerical_region_title ?= {
					any_ruler_in_clerical_region = {
						NOT = {
							is_vassal_of = root
						}
						pam_characters_to_evangelize_personal_tenet_trigger = yes
						any_personal_tenet = {
							pam_tenet_is_at_least_popular_trigger = { CHARACTER = root }
						}
					}
				}
			}
		}
	}
}

# Spread slander against a tenet known to the faith, chipping away at its popularity.

	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_send_legatine_mission.dds"
	}
	decision_group_type = religious

	desc = pam_persecute_tenet_decision_desc
	selection_tooltip = pam_persecute_tenet_decision_tooltip
	confirm_text = pam_persecute_tenet_decision_confirm

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		has_tenet_popularity_trigger = yes
		is_playable_character = yes
		has_pam_dlc_trigger = yes
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = pam_evangelize_personal_tenet_religious_authority_tt
			OR = {
				is_clergy = yes
				AND = { # The puppet route is reserved for players; the AI preaches through its own clergy
					is_ai = no
					has_theological_agent_puppet_trigger = yes
				}
			}
		}
		is_available_adult = yes
		custom_tooltip = {
			text = pam_persecute_tenet_select_tt
			exists = scope:selected_item
		}
	}

	cooldown = { years = 10 }

	cost = {
		piety = medium_piety_value
	}

	effect = {
		scope:selected_item = { save_scope_as = pam_persecute_tenet }
		# ROOT resolves to nothing in the toast's loc context, so the persecutor travels as a scope
		save_scope_as = pam_tenet_persecutor
		faith = {
			if = {
				limit = { has_tenet_popularity_trigger = yes }
				change_tenet_popularity = {
					target = scope:pam_persecute_tenet
					change = medium_tenet_popularity_loss
				}
			}
		}
		# Hidden so the new popularity is reported after the change lands, not previewed in the decision tooltip
		hidden_effect = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = pam_persecute_tenet_decision_toast
				left_icon = root
				custom_tooltip = pam_persecute_tenet_decision_toast_tt
			}
		}
	}

	widget = {
		gui = "decision_view_widget_select_tenet"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT"

		# List is tenets the character personally knows + tenets the rite knows
		setup_items = {
			scope:actor = {
				ordered_character_tenet = {
					knowledge_filter = known
					order_by = pam_raw_tenet_popularity_value
					max = 200
					check_range_bounds = no
					add_to_list = item_list
				}
			}
		}

		default_item = {
			scope:actor = {
				ordered_character_tenet = {
					knowledge_filter = known
					limit = { pam_tenet_can_be_persecuted_trigger = yes }
					order_by = pam_tenet_popularity_impact_value
					position = 0
					save_scope_as = selected_item
				}
			}
		}

		is_item_valid = {
			pam_tenet_can_be_persecuted_trigger = yes
		}

		ai_select_item = {
			scope:actor = {
				random_undesired_tenet = {
					limit = {
						pam_tenet_can_be_persecuted_trigger = yes
						pam_tenet_is_at_least_popular_trigger = { CHARACTER = scope:actor }
					}
					known_tenets = yes
					only_negative = yes
					threshold = -1
					selection_count = 97
					weight = {
						base = 10
						modifier = {
							add = pam_tenet_popularity_impact_value
						}
						modifier = { # The louder the faithful cheer it, the more worth drowning out
							factor = 2
							pam_tenet_is_at_least_very_popular_trigger = { CHARACTER = scope:actor }
						}
					}
					save_scope_as = selected_item
				}
			}
		}
	}

	ai_potential = {
		is_clergy = yes
		piety >= medium_piety_value
		any_undesired_tenet = {
			known_tenets = yes
			only_negative = yes
			threshold = -1
			selection_count = 97
			pam_tenet_can_be_persecuted_no_actor_trigger = yes
			pam_tenet_is_at_least_popular_trigger = { CHARACTER = root }
		}
	}

	ai_will_do = {
		base = 50
		modifier = {
			add = 20
			has_trait = zealous
		}
		modifier = {
			add = 20
			has_trait = theologian
		}
		modifier = {
			add = -30
			has_trait = cynical
		}

		modifier = { # A belief the whole faith cheers is the one most urgent to tear down
			add = 50
			scope:selected_item ?= {
				pam_tenet_is_at_least_very_popular_trigger = { CHARACTER = root }
			}
		}

           	exists = rite.head_of_rite

        NOT = {
            has_game_rule = off_custom_kingdoms

                    highest_held_title_tier >= tier_empire

        trigger_if = {

                government_has_flag = government_is_nomadic

    					exists = faith.religious_head

    					limit = {

        if = {
        	limit = {
        		scope:religious_authority = root

					opinion = -20

        	}
        }

    ai_potential = {
        NOR = {

        }
    }

### Invite Syncretic Exiles ###
pam_invite_syncretic_exiles_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_kingdom_of_heaven.dds"
	}
	decision_group_type = religious
	sort_order = 100
	cooldown = { years = 15 }

	desc = pam_invite_syncretic_exiles_decision_desc
	selection_tooltip = pam_invite_syncretic_exiles_decision_tooltip
	confirm_text = pam_invite_syncretic_exiles_decision_confirm

	is_shown = {
		pam_personal_syncretism_invite_exiles_available_trigger = yes
		has_pam_dlc_trigger = yes
	}

	is_valid = {
		pam_personal_syncretism_invite_exiles_available_trigger = yes
		is_landed = yes
		exists = capital_province
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		exists = capital_province
		custom_tooltip = {
			text = invite_exiles_same_faith_county_req
			any_sub_realm_county = {
				faith = root.faith
			}
		}
		trigger_if = {
			limit = {
				any_sub_realm_county = {
					faith = root.faith
				}
			}
			custom_tooltip = {
				text = pam_invite_syncretic_exiles_no_pool
				any_religion_global = {
					any_faith = {
						any_faith_rite = {
							pam_personal_syncretism_exile_rite_probe_trigger = yes
						}
					}
				}
			}
		}
	}

	cost = {
		piety = major_piety_value
	}

	effect = {
		save_scope_as = root_scope
		save_scope_as = from_decision
		save_scope_as = pam_syncretic_exiles_decision
		hidden_effect = {
			random_religion_global = {
				limit = {
					any_faith = {
						any_faith_rite = {
							pam_personal_syncretism_exile_rite_trigger = yes
						}
					}
				}
				random_faith = {
					limit = {
						any_faith_rite = {
							pam_personal_syncretism_exile_rite_trigger = yes
						}
					}
					ordered_faith_rite = {
						limit = {
							pam_personal_syncretism_exile_rite_trigger = yes
						}
						order_by = var:converted_development
						save_scope_as = target_rite
					}
				}
			}
			if = {
				limit = { exists = scope:target_rite }
				custom_tooltip = pam_invite_syncretic_exiles_decision_effect
				if = {
					limit = {
						scope:root_scope.faith != scope:target_rite.faith
					}
					scope:root_scope = {
						save_scope_as = ruler_target
						save_scope_as = seeking_any_counties
						trigger_event = global_religion.1000
					}
				}
				else = {
					scope:root_scope = {
						save_scope_as = ruler_target
						save_scope_as = seeking_same_faith_counties
						trigger_event = global_religion.1000
					}
				}
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:target_rite }
			}
			add_piety = major_piety_value
			custom_tooltip = pam_invite_syncretic_exiles_no_pool
		}
		close_view = {
			view = decisions
			player = root
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}
}
