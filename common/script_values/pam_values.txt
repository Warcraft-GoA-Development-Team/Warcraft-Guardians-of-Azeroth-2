# AI cleric host score when the Head of Faith cannot host
pam_investiture_council_host_score_value = {
	value = 0
	# +300 outranks the tie-breaks, so a cardinal hosts only if no prelate is available
	if = {
		limit = {
			NOT = { has_character_flag = cardinal_flag }
			any_held_title = { has_clerical_region = yes }
		}
		add = 300
	}
	add = {
		value = piety_level
		multiply = 10
	}
	add = {
		value = highest_held_title_tier
		multiply = 5
	}
	if = {
		limit = {
			exists = scope:investiture_hof_capital
			capital_province ?= {
				squared_distance = { target = scope:investiture_hof_capital value < squared_distance_medium }
			}
		}
		add = 100
	}
	if = {
		limit = {
			top_participant_group:the_christian_church ?= { participant_group_type = christian_clerical_main_power_rulers }
		}
		add = 40
	}
}

# ai_will_pick for tenets in activities
# do not remove, used by code
# Scopes:
#	root: character planning the activity
#	activity: the activity type for which the tenet is being selected
#	tenet: the tenet being evaluated
#	index: the number of tenet being selected (either 1, 2 or 3 for tenet_1, tenet_2, tenet_3)
activity_tenet_pick_score = {
	value = 0

	if = {
		limit = {
			any_desired_tenet = {
				this = scope:tenet
			}
		}
		add = 100
	}
	# Favour a personal tenet unless it is already core in the faith's main rite
	if = {
		limit = {
			has_personal_tenet = scope:tenet
			NOT = {
				faith.main_rite = {
					has_tenet_status = {
						tenet = scope:tenet
						status = core
					}
				}
			}
		}
		add = 100
	}
	# As above, but against the character's own rite rather than the faith's main rite
	if = {
		limit = {
			has_personal_tenet = scope:tenet
			NOT = {
				rite = {
					has_tenet_status = {
						tenet = scope:tenet
						status = core
					}
				}
			}
		}
		add = 100
	}
	if = {
		limit = {
			faith.main_rite = {
				has_tenet_status = {
					tenet = scope:tenet
					status = core
				}
			}
		}
		add = -100000
	}
	if = {
		limit = {
			scope:tenet = tenet:tenet_armed_pilgrimages
		}
		add = -100000
	}
	if = {
		limit = {
			scope:tenet = tenet:tenet_polyamory
			NOT = { scope:index ?= 2 }
		}
		multiply = 0.25
	}
	# Debate topic 1: favor high-popularity tenets; debate 2: favor low-popularity tenets.
	if = {
		limit = {
			NOT = { exists = scope:actor }
		}
		save_temporary_scope_as = actor
	}
	if = {
		limit = { scope:index ?= 2 }
		# An already-prohibited tenet has nothing left to prohibit, rehabilitating a popular one runs through topic 1
		if = {
			limit = {
				faith.main_rite = {
					has_tenet_status = {
						tenet = scope:tenet
						status = prohibited
					}
				}
			}
			add = -1000
		}
		else = {
			add = scope:tenet.pam_tenet_unpopularity_impact_value
		}
	}
	else = {
		add = scope:tenet.pam_tenet_popularity_impact_value
		add = scope:tenet.pam_christian_vice_restraint_value
	}
}

# Per-attendee stance score for an Ecumenical Council tenet debate (first/second debate)
ecumenical_council_attendee_base_support_value = {
	value = 0
	add = {
		value = 0
		if = {
			limit = {
				has_personal_tenet = scope:tenet
			}
			add = 6
		}
		if = {
			limit = {
				any_desired_tenet = { this = scope:tenet }
			}
			add = 4
		}
		multiply = ecumenical_council_tenet_attachment_direction_value
	}
	if = {
		limit = {
			has_trait = zealous
		}
		add = -3
	}
	if = {
		limit = {
			has_trait = cynical
		}
		add = 3
	}
	# Learned churchmen are more comfortable revisiting doctrine
	if = {
		limit = {
			has_trait = theologian
		}
		add = 2
	}
	if = {
		limit = {
			has_trait = lifestyle_scholar
		}
		add = 1
	}
	# The more pious, the more attached to the faith as it currently stands
	add = {
		value = piety_level
		multiply = -0.5
	}

	# (Rivals of the host are already excluded from ever supporting in the sort effect.)
	add = {
		value = "opinion(scope:host)"
		divide = 25
	}

	# Resistance to how radical the status change is (same in either direction)
	add = {
		value = ecumenical_council_proposal_change_magnitude_value
		multiply = -1
	}
	# What the faith at large makes of the tenet, which the rest of this score never sees.
	add = ecumenical_council_tenet_popularity_support_value
	# And what the tenet would make of the attendee: a saint by the new rite, or a sinner.
	add = ecumenical_council_tenet_judgement_support_value
	# Blocked: make_core of a still-prohibited tenet
	if = {
		limit = {
			scope:activity.var:current_debate_proposal ?= flag:make_core
			scope:host.faith.main_rite = {
				has_tenet_status = {
					tenet = scope:tenet
					status = prohibited
				}
			}
		}
		add = -16
	}
	# Zealots are the most scandalized by loosening a ban
	if = {
		limit = {
			has_trait = zealous

				has_tenet_status = {
					tenet = scope:tenet
					status = prohibited
				}
			}

				scope:activity.var:current_debate_proposal ?= flag:permit
				scope:activity.var:current_debate_proposal ?= flag:make_core
			}
		}
		add = -3
	}
	if = {
		limit = {
			pam_ec_tenet_is_christian_abomination_trigger = yes
			ecumenical_council_proposal_direction_value > 0
		}
		add = -30
	}
}

# Keyed on ec_desire_target: activity_doctrine_pick_score scores alternatives in a context where

# Scopes:
#	root: the evaluating character
#	ec_desire_target: the doctrine being evaluated

	value = 0
	save_temporary_scope_as = voter
	# generally
	# good to me-> +100
	# very good to me-> +125
	# alternative option but still good to me -> +75
	# bad to opposite of me -> +50
	# bad to me -> -100
	# I don't like but not harmful to me -> -50

	switch = {
		trigger = scope:ec_desire_target
		doctrine:doctrine_monogamy = {
			# A chaste ruler favours monogamy over concubinage.
			if = {
				limit = {
					has_trait = chaste
					faith = {
						NOT = { has_doctrine = doctrine_monogamy }
					}
				}
				add = 50
			}
			else_if = {
				limit = {
					OR = {
						has_trait = lustful
						any_personal_tenet = { tenet_has_virtue = lustful }
						rite = { trait_is_virtue_rite = lustful }
						any_consort = { count > 1 }
					}
				}
				add = -100
			}
		}
		doctrine:doctrine_polygamy = {
			if = {
				limit = {
					the_christian_church_radical_change_allowed_trigger = yes
					OR = {
						has_trait = lustful
						any_personal_tenet = { tenet_has_virtue = lustful }
						rite = { trait_is_virtue_rite = lustful }
					}
					# don't switch between polygamy and concubines - it make no sense.
					faith = {
						has_doctrine = doctrine_monogamy
					}
				}
				add = 75
			}
			else_if = {
				limit = {
					OR = {
						has_trait = chaste
						any_consort = { count <= 1 }
					}
				}
				add = -50
			}
		}
		doctrine:doctrine_concubines = {
			# A lustful ruler favours being able to take concubines.
			if = {
				limit = {
					the_christian_church_radical_change_allowed_trigger = yes
					OR = {
						has_trait = lustful
						any_personal_tenet = { tenet_has_virtue = lustful }
						rite = { trait_is_virtue_rite = lustful }
					}
					faith = {
						has_doctrine = doctrine_monogamy
					}
				}
				add = 100
			}
			else_if = {
				limit = {

									NOT = {
										any_child = {
											is_child_of = scope:voter
										}
									}
								}
							}
						}
					}
				}
				add = -100
			}
		}

					is_clergy = yes

					}

						NOT = { has_doctrine = doctrine_divorce_approval }
					}
				}
				add = 100
			}

				character_considered_wicked_trigger = { RITE = scope:cleric.rite }

					has_relation_rival = scope:actor
				}
				tyranny > medium_tyranny
				has_trait = excommunicated
			}
		}
		add = 4
	}
	else_if = {
		limit = {
			OR = {
				character_considered_morally_dubious_trigger = { RITE = scope:cleric.rite }
				scope:cleric = {
					is_ai = yes
					opinion = {
						target = scope:actor
						value <= -50
					}
				}
				tyranny >= low_tyranny
			}
		}
		add = 2
	}
	if = {
		limit = {
			OR = {
				piety_level >= 5
				has_trait = saint
				has_trait = herald
				has_trait = saoshyant_descendant

				has_trait = faith_warrior
				has_trait = saoshyant
				has_trait = paragon
				has_trait = savior
				has_trait = crusader_king
			}
		}

		limit = {
			OR = {
				piety_level >= 4
				has_trait = zealous
				has_trait = theologian
				has_trait = divine_blood
				has_trait = consecrated_blood
				has_trait = sayyid
			}
		}
		add = -2
	}
	else_if = {
		limit = {
			piety_level >= 3
		}
		add = -1
	}
	#Your rite divergence
	if = {
		limit = {
			rite = {
				this != scope:cleric.rite
				"divergence(scope:cleric.rite)" >= monumental_rite_divergence_value
			}
		}
		add = 2
	}
	else_if = {
		limit = {
			rite = {
				this != scope:cleric.rite
				"divergence(scope:cleric.rite)" >= massive_rite_divergence_value
			}
		}
		add = 1
	}
	#Scope saved in interactions that do not need these values
	if = {
		limit = {
			NOT = { exists = scope:only_consideration }
		}
		add = clerical_hesitation_from_target_holder_value
		add = clerical_hesitation_from_potential_defender_value
	}

clerical_hesitation_from_target_holder_value = {
	value = 0
	#Target holder exists
	if = {
		limit = {
			exists = scope:owner
		}
		#Target holder is wicked
		if = {
			limit = {
				scope:owner = {
					OR = {
						character_considered_wicked_trigger = { RITE = scope:cleric.rite }
						has_relation_rival = scope:cleric
						tyranny > medium_tyranny
						has_trait = excommunicated
						faith = {
							faith_hostility_level = {
								target = scope:cleric.faith
								value >= faith_evil_level
							}
						}
					}
				}
			}
			add = -4
		}
		#Target holder is morally dubious
		else_if = {
			limit = {
				scope:owner = {
					OR = {
						character_considered_morally_dubious_trigger = { RITE = scope:cleric.rite }
						scope:cleric = {
							is_ai = yes
							opinion = {
								target = scope:owner
								value <= -50
							}
						}
						faith = {
							faith_hostility_level = {
								target = scope:cleric.faith
								value >= faith_hostile_level
							}
						}
					}
				}
			}
			add = -2
		}
		#Target holder is same-faith, very pious
		if = {
			limit = {
				scope:owner = {
					faith = scope:cleric.faith
					OR = {
						piety_level >= 4
						has_trait = saint
						has_trait = herald
						has_trait = saoshyant_descendant
						has_trait = order_member
						has_trait = faith_warrior
						has_trait = saoshyant
						has_trait = paragon
						has_trait = savior
						has_trait = crusader_king
						has_trait = zealous
						has_trait = theologian
						has_trait = divine_blood
						has_trait = consecrated_blood
						has_trait = sayyid
					}
					OR = {
						rite = scope:cleric.rite
						rite = {
							"divergence(scope:cleric.rite)" < massive_rite_divergence_value
						}
					}
				}
			}
			add = 4
		}
		#Target holder is same faith
		else_if = {
			limit = {
				scope:owner = {
					piety_level >= 1
					NOT = { has_trait = excommunicated }
					faith = scope:cleric.faith
					OR = {
						rite = scope:cleric.rite
						rite = {
							"divergence(scope:cleric.rite)" < monumental_rite_divergence_value
						}
					}
				}
			}
			add = 2
		}
		#Target holder is a friendly faith
		else_if = {
			limit = {
				scope:owner = {
					piety_level >= 1
					NOT = { has_trait = excommunicated }
					OR = {
						AND = {
							faith = scope:cleric.faith
							NOT = { rite = scope:cleric.rite }
						}
						faith = {
							faith_hostility_level = {
								target = scope:cleric.faith
								value < faith_astray_level
							}
						}
					}
				}
			}
			add = 1
		}
	}
}

clerical_hesitation_from_potential_defender_value = {
	value = 0
	if = {
		limit = {
			exists = scope:target
		}
		#Target holder exists
		if = {
			limit = {
				exists = scope:owner
			}
			#Save potential defender too
			#If same realm but not same sub-realm, their liege matters
			if = {
				limit = { 
					scope:owner = { is_independent_ruler = no }
					scope:actor = { is_independent_ruler = no }
					scope:owner.top_liege = scope:actor.top_liege
					NOR = {
						scope:actor.liege = scope:owner.liege
						scope:owner.liege = scope:actor
						scope:actor.top_liege = scope:owner
					}
				}
				scope:owner.liege = { add = clerical_hesitation_sub_value }
			}
			#If not in the same realm, their top liege matters
			else_if = {
				limit = {
					scope:owner = { is_independent_ruler = no }
					NOT = { scope:actor.top_liege = scope:owner.top_liege }
				}
				scope:owner.top_liege = { add = clerical_hesitation_sub_value }
			}
			else = {
				scope:owner = {
				}
			}
		}
		#No target holder - look at de jure defender only
		else = {
			#De jure liege who controls at least one county of it
			if = {
				limit = {
					scope:target = {
						de_jure_liege ?= {
							exists = holder
							NOR = {
								holder = scope:cleric
								holder = scope:actor
							}
						}
					}
				}
				#Save top liege if other realm
				scope:target = {
					if = {
						limit = {
							de_jure_liege.holder = {
								OR = {
									is_independent_ruler = yes
									top_liege = scope:actor.top_liege
								}
							}
						}
						de_jure_liege.holder = { add = clerical_hesitation_sub_value }
					}
					else = {
						de_jure_liege.holder.top_liege = { add = clerical_hesitation_sub_value }
					}
				}
			}
		}
	}
}
