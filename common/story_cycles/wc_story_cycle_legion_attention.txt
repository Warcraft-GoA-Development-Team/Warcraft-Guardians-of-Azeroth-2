# Legion attention - invisible driver for the wandering arcanist pack.
# Created by wc_legion_attention_start_story_effect on the first gain; ends at 0 attention,
# when the owner leaves permanently (e.g. accepted the Dreadlord's Offer), or on death.
story_cycle_wc_legion_attention = {
	on_setup = {}

	on_end = {
		story_owner = { remove_variable = wc_legion_attention }
	}

	on_owner_death = {
		end_story = yes
	}

	# Yearly decay, then end the story if nothing is left
	effect_group = {
		days = 365

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				story_owner = {
					wc_change_legion_attention_effect = { VALUE = wc_legion_attention_decay_value }
				}
				if = {
					limit = {
						story_owner = {
							OR = {
								wc_legion_attention_current_value <= 0
								wc_legion_attention_permanently_excluded_trigger = yes
							}
						}
					}
					end_story = yes
				}
			}
		}
	}

	# Something notices you - stage events (.0003-.0005 pick themselves by stage)
	effect_group = {
		days = { 90 270 }

		trigger = {
			story_owner = {
				wc_legion_attention_noticed_trigger = yes
				wc_legion_attention_stage_event_trigger = yes
			}
		}

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				story_owner = {
					trigger_event = { on_action = wc_legion_attention_events }
				}
			}
		}
	}
}
