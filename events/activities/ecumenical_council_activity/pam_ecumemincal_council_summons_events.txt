# Learn another guest's secret
scripted_trigger ecumenical_council_summons_events_0001_valid_guest = {
	this != root
	this != scope:host
	is_ai = yes
	is_landed = yes
	any_secret = {
		NOT = { is_known_by = root }
	}
	save_temporary_scope_as = guest_1_temp
	scope:activity = { # They have someone to confide on
		any_attending_character = {
			is_ai = yes
			this != root 
			this != scope:host
			this != scope:guest_1_temp
			opinion = {
				target = scope:guest_1_temp
				value >= 10
			}
		}
	}
}

ecumenical_council_summons_events.0001 = {
	type = activity_event
	title = ecumenical_council_summons_events.0001.t
	desc = ecumenical_council_summons_events.0001.desc
	theme = ecumenical_council
	override_effect_2d = {
		trigger = { exists = scope:rain_day }
		reference = rain
	}
	left_portrait = {
		character = root
		animation = eavesdrop
		camera = camera_event_very_left
	}
	center_portrait = {
		character = scope:guest_2
		animation = interested
	}
	right_portrait = {
		character = scope:guest_1
		animation = worry
		camera = camera_event_center_pointing_very_left
	}
	cooldown = { years = 5 }
	
	trigger = {
		involved_activity = {
			any_attending_character = {
				ecumenical_council_summons_events_0001_valid_guest = yes # Also accounts for guest_2
			}
		}
	}

	immediate = {
		random_rain_snow_chance_effect = yes
		involved_activity = {
			random_attending_character = {
				limit = { ecumenical_council_summons_events_0001_valid_guest = yes }
				weight = {
					base = 1
					is_of_major_interest_to_weight_up_modifier = { CHARACTER = root }
					is_of_minor_interest_to_weight_up_modifier = { CHARACTER = root }
				}
				save_scope_as = guest_1
				random_secret = {
					limit = { NOT = { is_known_by = root } }
					save_scope_as = guest_secret
				}
			}
			ordered_attending_character = {
				limit = {
					this != root
					this != scope:host
					this != scope:guest_1
					is_ai = yes
					opinion = {
						target = scope:guest_1
						value >= 10
					}
				}
				order_by = "opinion(scope:guest_1)" 
				save_scope_as = guest_2
			}
		}
	}
	
	# Eavesdrop
	option = {
		name = ecumenical_council_summons_events.0001.a
		custom_tooltip = ecumenical_council_summons_events.0001.a.learn_secret_tt
		hidden_effect = {
			send_interface_toast = {
				title = ecumenical_council_summons_events.0001.a.learn_secret_tt
				right_icon = scope:guest_1
				scope:guest_secret = { reveal_to = root }
			}
		}
		# Chance you get caught
		random = {
			chance = {
				value = scope:guest_1.intrigue
				multiply = 3.5
				min = 10
			}
			scope:guest_1 = {
				progress_towards_rival_effect = {
					REASON = rival_eavesdropped_on_secret_reason
					CHARACTER = root
					OPINION = -20
				}
			}
		}
		stress_and_fulfillment_impact = {
			deceitful = medium_stress_impact_loss
			compassionate = major_stress_impact_gain
			honest = major_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = {
				has_trait = deceitful
				factor = 2
			}
			modifier = {
				OR = {
					has_trait = compassionate
					has_trait = honest
				}
				factor = 0
			}
		}
	}
	
	# Offer comfort
	option = {
		name = ecumenical_council_summons_events.0001.b
		reason = opinion_positive
		trigger = {
			scope:guest_1 = {
				opinion = {
					target = root
					value >= 10
				}
			}
		}
		progress_towards_friend_effect = {
			REASON = friend_comforted_at_council_reason
			CHARACTER = scope:guest_1
			OPINION = default_friend_opinion
		}
		stress_and_fulfillment_impact = {
			gregarious = medium_stress_impact_loss
			compassionate = medium_stress_impact_loss
			shy = major_stress_impact_gain
			callous = major_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = {
				OR = {
					has_trait = gregarious
					has_trait = compassionate
				}
				factor = 2
			}
			modifier = {
				OR = {
					has_trait = shy
					has_trait = callous
				}
				factor = 0
			}
		}
	}
	
	# Leave it
	option = {
		name = ecumenical_council_summons_events.0001.c
		stress_and_fulfillment_impact = {
			base = medium_stress_impact_loss
		}
		ai_chance = {
			base = 100
		}
	}
}

ecumenical_council_summons_events.0010 = {
	type = activity_event
	title = ecumenical_council_summons_events.0010.t
	desc = ecumenical_council_summons_events.0010.desc
	theme = ecumenical_council
	override_effect_2d = {
		trigger = { exists = scope:rain_day }
		reference = rain
	}
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = {
				OR = {
					piety_level >= high_piety_level
					learning >= high_skill_rating
				}
			}
			animation = personality_zealous
		}
		animation = interested
		camera = camera_event_left_to_the_left
	}
	center_portrait = {
		character = scope:servant
		animation = worry
		camera = camera_event_left
	}
	right_portrait = {
		character = scope:host
		animation = mental_anguish
		camera = camera_event_crowd
	}
	cooldown = { years = 5 }
	
	trigger = {
		this != scope:host
		scope:host = {
			any_secret = {
				NOT = { is_known_by = root }
			}
		}
	}

	immediate = {
		random_rain_snow_chance_effect = yes
		scope:host = {
			random_secret = {
				limit = { NOT = { is_known_by = root } }
				save_scope_as = host_secret
			}
		}
		if = {
			limit = { 
				scope:host = {
					any_courtier = { ecumenical_council_summons_events_0010_valid_servant = yes }
				}
			}
			scope:host = {
				random_courtier = {
					limit = { ecumenical_council_summons_events_0010_valid_servant = yes }
					save_scope_as = servant
				}
			}
		}
		else = {
			hidden_effect = {
				create_character = {
					location = scope:host.location
					template = servant_character
					gender_female_chance = 50
					faith = scope:host.faith
					rite = scope:host.rite
					culture = scope:host.culture
					save_scope_as = servant
				}
			}
		}
		if = {
			limit = {
				involved_activity = {
					any_attending_character = { this != scope:servant }
				}
			}
			hidden_effect = { scope:servant = { add_to_activity_without_travel = scope:activity } }
		}
	}
	
	# Leverage your piety
	option = {
		name = ecumenical_council_summons_events.0010.a
		add_internal_flag = special
		show_as_unavailable = { always = yes }
		reason = piety_level
		trigger = {
			piety_level >= high_piety_level
		}
		custom_tooltip = ecumenical_council_summons_events.0010.a.learn_secret_tt
		send_interface_toast = {
			title = ecumenical_council_summons_events.0010.a.learn_secret_tt
			right_icon = scope:host
			scope:host_secret = { reveal_to = root }
		}
		add_piety = medium_piety_gain
		stress_and_fulfillment_impact = {
			zealous = medium_stress_impact_loss
			arrogant = medium_stress_impact_loss
		}
		ai_chance = {
			base = 50
			modifier = {
				OR = {
					has_trait = zealous
					has_trait = arrogant
				}
				factor = 2
			}
		}
	}
	
	# Quote the Bible spectacularly
	option = {
		name = ecumenical_council_summons_events.0010.b
		flavor = ecumenical_council_summons_events.0010.b.flavor
		add_internal_flag = special
		show_as_unavailable = { always = yes }
		trigger = {
			learning >= high_skill_rating
		}
		custom_tooltip = ecumenical_council_summons_events.0010.a.learn_secret_tt
		send_interface_toast = {
			title = ecumenical_council_summons_events.0010.a.learn_secret_tt
			right_icon = scope:host
			scope:host_secret = { reveal_to = root }
		}
		add_learning_lifestyle_xp = medium_lifestyle_xp
		add_scholar_trait_or_xp_effect = { VALUE = 10 }
		stress_and_fulfillment_impact = {
			theologian = medium_stress_impact_loss
			erudite = medium_stress_impact_loss
		}
		ai_chance = {
			base = 50
			modifier = {
				learning >= very_high_skill_rating
				factor = 2
			}
		}
	}
	
	# Let it be
	option = {
		name = ecumenical_council_summons_events.0010.c
		ai_chance = {
			base = 100
		}
	}
}

# Clergy: get hook on Host
scripted_effect ecumenical_council_summons_events_0020_get_appointment_score_effect = {
	if = {
		limit = {
			is_clerical_appointment_valid_trigger = {
				TITLE = scope:host.primary_title
				CANDIDATE = root
				TOP_LIEGE = scope:host
			}
		}
		custom_tooltip = {
			text = ecumenical_council_summons_events.0020.papabile_tt
			if = {
				limit = { exists = var:pam_papabile_score }
				change_variable = {
					name = pam_papabile_score
					add = 10
				}
			}
			else = {
				set_variable = {
					name = pam_papabile_score
					value = 10
				}
			}
		}
	}
	else = {
		add_hook = {
			target = scope:host
			type = favor_hook
		}
	}
}
