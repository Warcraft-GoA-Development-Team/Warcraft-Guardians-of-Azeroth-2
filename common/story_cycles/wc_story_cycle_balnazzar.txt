# Balnazzar - Dreadlord personal story
# Created for character:430400 in the Frozen Throne bookmark, whether human- or AI-controlled.
# Decision/event-driven chain, not periodic.

story_balnazzar_dreadlord = {
	visible = yes

	icon = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/icons/traits/creature_nathrezim.dds" 	}

	background = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/illustrations/event_scenes/wc_undercity_throne_room.dds" 	}

	visualization = {
		custom_string_key = "BALNAZZAR_DREADLORD_CUSTOM_STRING_KEY"
	}

	on_setup = {
		set_variable = { name = wc_balnazzar_suspicion_v2 value = yes }
		set_variable = { name = wc_balnazzar_suspicion value = 0 }
		set_variable = { name = wc_balnazzar_stratholme_migrated value = yes }
		set_global_variable = { # Anchor, so the world can reference this playthrough's story
			name = balnazzar_dreadlord_story
			value = this
		}
	}

	on_end = {
		wc_scarlet_migrate_crusade_effect = yes
		remove_global_variable = balnazzar_dreadlord_story
	}

	on_owner_death = {
		# Just end the story cleanly rather than try to hand it to an heir - the arc is
		# built around one specific character's choices, not a dynasty.
		end_story = yes
	}

	# One-time conversion, including idle old saves.
	effect_group = {
		days = { 1 1 }
		trigger = { NOT = { has_variable = wc_balnazzar_suspicion_v2 } }
		triggered_effect = { effect = { wc_balnazzar_migrate_suspicion_effect = yes } }
	}

	# Age traits trigger on chronological age alone; strip them from the disguise wearer
	# rather than patch the shared vanilla health system.
	effect_group = {
		days = { 30 60 }

		triggered_effect = {
			effect = {
				wc_balnazzar_update_stratholme_effect = yes
				wc_balnazzar_update_renault_effect = yes
				wc_balnazzar_update_fairbanks_effect = yes
				if = {
					limit = { story_owner = { has_character_flag = wc_balnazzar_disguise_active } }
					if = {
						limit = {
							OR = {
								NOT = { exists = var:wc_balnazzar_possession_body_migrated }
								NOT = { exists = var:wc_balnazzar_disguise_immunity_migrated }
							}
							exists = character:430400
							character:430400 = { is_alive = yes }
						}
						character:430400 = { trigger_event = { id = wc_balnazzar_story.0010 } }
					}
					story_owner = {
						# The AI was expected to accept the pact; when the scripted
						# 605.9.9 death beats the .1003/.1004 offer, seal the canonical
						# choice now so the death outcome still resolves. A human
						# refusal stays alt-history.
						if = {
							limit = {
								is_ai = yes
								this = character:60021
								NOT = { has_character_flag = balnazzar_kelthuzad_pact_sealed }
								NOT = { has_character_flag = wc_balnazzar_kelthuzad_pact_refused }
								character:60017 = { is_alive = no }
								character:59100 = { is_alive = yes }
							}
							add_character_flag = wc_balnazzar_kelthuzad_pact_accepted
							add_character_flag = balnazzar_kelthuzad_pact_sealed
						}
						# .1004 resolved after the scripted death forced the AI onto
						# the refuse option. Overturn that lapsed refusal (AI only;
						# a genuine refusal with Alexandros alive, or a human
						# refusal, carries no marker and stays alt-history).
						if = {
							limit = {
								is_ai = yes
								this = character:60021
								has_character_flag = wc_balnazzar_pact_refused_lapsed
								NOT = { has_character_flag = wc_balnazzar_kelthuzad_pact_accepted }
								character:60017 = { is_alive = no }
								character:59100 = { is_alive = yes }
							}
							remove_character_flag = wc_balnazzar_kelthuzad_pact_refused
							remove_character_flag = wc_balnazzar_pact_refused_lapsed
							add_character_flag = wc_balnazzar_kelthuzad_pact_accepted
							add_character_flag = balnazzar_kelthuzad_pact_sealed
						}
						wc_balnazzar_change_suspicion_effect = { CHANGE = 0 }
						# A pact accepted before the scripted Alexandros death still
						# resolves, whether or not the native Renault scheme opened.
						# Routed through a hidden character-rooted event: the shared
						# resurrection effect reads root.location, and root is still the
						# story object inside this effect_group.
						trigger_event = { id = wc_balnazzar_story.1029 }
						# Also covers ordinary/natural death and a supporter becoming available.
						# The event rechecks its own character-root eligibility at delivery.
						if = {
							limit = {
								this = character:60021
								NOT = { has_character_flag = has_formed_the_scarlet_crusade }
								OR = {
									character:60017 = { is_alive = no }
									wc_balnazzar_parricide_trigger = yes
								}
								OR = {
									NOT = { has_character_flag = wc_balnazzar_fairbanks_direct_witness }
									has_character_flag = wc_balnazzar_fairbanks_accusation_heard
									# A dead or detained witness cannot leave the foundation window locked.
									character:high_inquisitor_fairbanks = { is_alive = no }
									character:high_inquisitor_fairbanks = { is_imprisoned = yes }
								}
							}
							trigger_event = { id = wc_balnazzar_story.1007 }
						}
						remove_trait = clouded_eyes
						remove_trait = faltering_heart
						remove_trait = fragile_bones
						remove_trait = infirm
						remove_trait = withering_mind
					}
					wc_balnazzar_witness_pressure_effect = yes
					# AI only: the AI never plays the player-paced suspicion
					# minigame, so the mask degrades on a bounded timer. The
					# reveal pulse below only fires once suspicion maxes out.
					if = {
						limit = {
							story_owner = {
								is_ai = yes
								this = character:60021
								has_character_flag = wc_balnazzar_disguise_active
								NOT = { has_character_flag = balnazzar_shed_the_mask }
							}
							exists = var:wc_balnazzar_scarlet_founded
							NOT = { exists = var:wc_balnazzar_exposure_queued }
						}
						if = {
							limit = { NOT = { exists = var:wc_balnazzar_ai_suspicion_cooldown } }
							set_variable = { name = wc_balnazzar_ai_suspicion_cooldown value = yes months = 6 }
							story_owner = { wc_balnazzar_change_suspicion_effect = { CHANGE = 45 } }
						}
					}
				}
			}
		}
	}

	# Yearly check for aging with no heir - offers succession confirmation event once via balnazzar_shell_failing_notified flag.
	effect_group = {
		days = { 365 365 }

		trigger = {
			story_owner = {
				has_character_flag = wc_balnazzar_disguise_active
				age >= 70
				NOT = { exists = primary_heir }
				NOT = { has_character_flag = balnazzar_shed_the_mask }
				NOT = { has_character_flag = balnazzar_shell_failing_notified }
			}
		}

		triggered_effect = {
			effect = {
				story_owner = {
					add_character_flag = balnazzar_shell_failing_notified
					trigger_event = {
						id = wc_balnazzar_story.0100
						days = { 5 15 }
					}
				}
			}
		}
	}

	# Appended to preserve existing timer order in saves. Also catches console edits.
	# Argent liaison: report the historical foundation once; it does not create it.
	effect_group = {
		days = { 7 14 }
		trigger = {
			story_owner = {
				is_ai = no
				this = character:60021
				has_character_flag = wc_balnazzar_disguise_active
				has_character_flag = wc_balnazzar_fairbanks_fallout_resolved
				NOT = { has_character_flag = wc_balnazzar_argent_dawn_notified }
			}
			title:e_argent_dawn = { holder = character:60625 }
		}
		triggered_effect = {
			effect = { story_owner = { trigger_event = { id = wc_balnazzar_story.1018 } } }
		}
	}

	effect_group = {
		days = { 1 1 }
		trigger = {
			exists = var:wc_balnazzar_suspicion
			var:wc_balnazzar_suspicion >= 100
			NOT = { exists = var:wc_balnazzar_exposure_queued }
			story_owner = {
				wc_nathrezim_cover_active_trigger = yes
				NOT = { has_variable = wc_nathrezim_cover_resolved }
				NOT = { has_variable = wc_nathrezim_cover_opening_seen }
				NOT = { has_variable = wc_nathrezim_cover_opening_pending }
			}
		}
		triggered_effect = {
			effect = { story_owner = { wc_balnazzar_begin_cover_crisis_effect = yes } }
		}
	}

	# AI only: the AI never plays the player-paced suspicion minigame, so the
	# maintenance pulse (above) gives it a visible, bounded buildup. The reveal
	# runs on the same rule as the player: only a maxed suspicion ends the cover.
	# No free timer and no stratholme_resolved shortcut.
	effect_group = {
		days = { 7 14 }
		trigger = {
			exists = var:wc_balnazzar_scarlet_founded
			exists = var:wc_balnazzar_suspicion
			var:wc_balnazzar_suspicion >= 100
			NOT = { exists = var:wc_balnazzar_exposure_queued }
			story_owner = {
				is_ai = yes
				this = character:60021
				has_character_flag = wc_balnazzar_disguise_active
				NOT = { has_character_flag = balnazzar_shed_the_mask }
			}
		}
		triggered_effect = {
			effect = {
				# .2000 clears the disguise, so this pulse stops on delivery.
				story_owner = { trigger_event = { id = wc_balnazzar_story.2000 } }
			}
		}
	}
}
