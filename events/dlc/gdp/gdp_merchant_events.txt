# Become Merchant - Voluntary
gdp_merchant.0001 = {
	type = character_event
	window = fullscreen_event
	queue_icon = "gfx/interface/icons/splash_icons/splash_merchant.dds"
	title = gdp_merchant.0001.t
	desc = {
		desc = gdp_merchant.0001.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:laamp_heir
					scope:old_primary_title ?= { holder ?= scope:laamp_inheritor }
				}
				desc = gdp_merchant.0001.heir
			}
			triggered_desc = {
				trigger = {
					exists = scope:laamp_liege
					scope:old_primary_title ?= { holder ?= scope:laamp_inheritor }
				}
				desc = gdp_merchant.0001.liege
			}
			triggered_desc = {
				trigger = { scope:old_primary_title ?= { holder ?= scope:laamp_inheritor } }
				desc = gdp_merchant.0001.new
			}
		}
	}
	theme = realm
	left_portrait = {
		character = root
		animation = steward
	}
	lower_right_portrait = {
		character = scope:laamp_inheritor
		trigger = { exists = scope:laamp_inheritor }
	}
	override_background = { reference = gdp_fullscreen_merchant_positive }
	cooldown = { days = 5 }

	trigger = {
		OR = {
			exists = scope:laamp_inheritor
			government_has_flag = government_is_landless_adventurer
		}
	}

	immediate = {
		if = {
			limit = { exists = scope:laamp_inheritor }
			scope:laamp_inheritor = { save_scope_as = become_merchant_flavor_char }
		}
		else = { save_scope_as = become_merchant_flavor_char }
		# Create Merchant
		create_merchant_title_effect = {
			REASON = flag:voluntary
			FLAVOR_CHAR = scope:become_merchant_flavor_char
		}
		scope:laamp_inheritor ?= {
			# Tooltip for player
			custom_tooltip = become_landless_adventurer_decision_title_scope_tt
			# Create a loving family so inheritance isn't so borked
			if = {
				limit = { has_character_flag = new_laamp_inheritor }
				trigger_event = game_rule.1001
				remove_character_flag = new_laamp_inheritor
			}
		}
		save_scope_as = merchant
		scope:old_liege ?= { trigger_event = gdp_merchant.0002 }
		adventurer_prestige_reset_effect = yes
		trigger_event = gdp_merchant.1001
		stress_impact = {
			craven = minor_stress_impact_gain
			content = minor_stress_impact_gain
			lazy = minor_stress_impact_gain
		}
	}

	option = {
		name = gdp_merchant.0001.a
		add_character_modifier = {
			modifier = gdp_voluntary_merchant_modifier
			years = 5
		}
	}
}

# Associate asks to be adopted
gdp_merchant.0010 = {
	type = character_event
	title = gdp_merchant.0010.t
	desc = {
		desc = gdp_merchant.0010.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:associate = { has_character_flag = generated_starting_character }
				}
				desc = gdp_merchant.0010.desc.starting
			}
			desc = gdp_merchant.0010.desc.starting_fallback
		}
		desc = gdp_merchant.0010.desc.cont
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:heir }
				desc = gdp_merchant.0010.desc.children
			}
			desc = gdp_merchant.0010.desc.no_children
		}
		desc = gdp_merchant.0010.desc.outro
	}
	theme = merchant
	left_portrait = {
		character = root
		animation = interested
	}
	right_portrait = {
		character = scope:associate
		animation = nervous
	}

	cooldown = { years = 100 }

	trigger = {
		# Mechanical Player Guidance event
		is_ai = no
		# Only for merchants
		government_has_flag = government_is_merchant
		# Relevant conditions
		is_adult = yes
		NOT = {
			any_child = { is_adult = yes } # includes having no children at all
		}
		any_courtier = {
			NOR = {
				is_spouse_of = root
				is_close_family_of = root
				has_trait = adopted_no_inheritance
				house ?= root.house
			}
		}
		# Some time should have passed
		years_from_game_start >= 2
	}

	weight_multiplier = {
		base = 1
		modifier = { # Increase weight as time goes on, per ruler
			# If we're in China/Korea, there is Confucian content we want to expedite instead
			location = {
				NOR = {
					geographical_region = world_asia_china
					geographical_region = world_asia_korea
				}
			}
			add = {
				value = years_as_ruler
				multiply = 10
				min = 0 # avoid -1
			}
		}
	}

	immediate = {
		random_courtier = {
			limit = {
				can_be_child_of = root
				NOR = {
					is_spouse_of = root
					is_close_family_of = root
					has_trait = adopted_no_inheritance
					house ?= root.house
				}
			}
			alternative_limit = {
				age <= root.age
				NOR = {
					is_spouse_of = root
					is_close_family_of = root
					has_trait = adopted_no_inheritance
					house ?= root.house
				}
			}
			alternative_limit = {
				NOR = {
					is_spouse_of = root
					is_close_family_of = root
					has_trait = adopted_no_inheritance
					house ?= root.house
				}
			}
			weight = {
				base = 1
				modifier = {
					factor = 2
					has_character_flag = generated_starting_character
				}
			}
			save_scope_as = associate
		}
		# Saving any children for loc
		if = {
			limit = {
				any_child = { count >= 1 }
			}
			primary_heir = { save_scope_as = heir }
		}
	}

	option = { # MY DAUGHTER/SON
		name = gdp_merchant.0010.a
		custom_tooltip = merchant_adoption_event_tt
		merchant_adopt_effect = {
			CHILD = scope:associate
			ADOPTER = root
		}
		create_character_memory = {
			type = merchant_family_adopted
			participants = { adoptee = scope:associate }
		}
		scope:new_memory = {
			set_variable = {
				name = adopted_house
				value = root.house
			}
		}
		scope:associate = {
			create_character_memory = {
				type = was_merchant_family_adopted
				participants = { adopter = root }
			}
			scope:new_memory = {
				set_variable = {
					name = adopted_house
					value = root.house
				}
			}
		}
		reverse_add_opinion = {
			target = scope:associate
			modifier = grateful_opinion
			opinion = 30
		}
		house = {
			add_unity_value = {
				value = medium_unity_loss
				character = root
				desc = merchant_unity_adoption_loss
			}
		}
		stress_impact = {
			callous = minor_stress_impact_gain
			arrogant = medium_stress_impact_gain
			compassionate = medium_stress_impact_loss
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_sociability = 1
				ai_compassion = 1
				ai_zeal = -1
			}
			modifier = {
				has_trait = compassionate
				add = 20
			}
			modifier = {
				OR = {
					has_trait = callous
					has_trait = arrogant
				}
				factor = 0
			}
		}
	}

	option = { # Maybe you should do more.
		name = gdp_merchant.0010.b
		scope:associate = {
			add_character_modifier = {
				modifier = gdp_driven_by_motivation_modifier
				years = 10
			}
		}
		reverse_add_opinion = {
			modifier = irritated_opinion
			target = scope:associate
			opinion = -10
		}
		house = {
			add_unity_value = {
				value = medium_unity_gain
				character = root
				desc = merchant_unity_adoption_loss
			}
		}
		stress_impact = {
			ambitious = medium_stress_impact_loss
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_boldness = 1
				ai_energy = 1
			}
			modifier = {
				has_trait = ambitious
				add = 50
			}
		}
	}

	option = { # I'm good
		name = gdp_merchant.0010.c
		reverse_add_opinion = {
			modifier = insulted_opinion
			target = scope:associate
			opinion = -5
		}
		stress_impact = {
			content = minor_stress_impact_loss
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_energy = -1
			}
		}
	}
}

scripted_effect gdp_merchant_event_0015_appoint_effect = {
	$CHOSEN$ = { save_temporary_scope_as = new_trader }
	trade_route_trader_opinion_effect = { TRADER = scope:new_trader }
	scope:new_trader = {
		show_as_tooltip = {
			if = {
				limit = {
					NOT = { has_trait = lifestyle_merchant }
				}
				add_trait = lifestyle_merchant # actually added by appointment on_action
			}
			add_trait_xp = { # always rewarded by this event
				trait = lifestyle_merchant
				track = caravaneer
				value = lifestyle_merchant_xp_gain_medium_value
			}
		}
		hidden_effect = {
			if = {
				limit = { has_trait = lifestyle_merchant } # added by appointment on_action, but prevent errors
				add_trait_xp = {
					trait = lifestyle_merchant
					track = caravaneer
					value = lifestyle_merchant_xp_gain_medium_value
				}
			}
		}
	}
	hidden_effect = {
		while = {
			count = 3
			trigger_event = { on_action = spawn_trade_opportunity_contract_on_action }
		}
	}
}

 scripted_trigger gdp_merchant_event_0020_better_partners_trigger = {
	is_available_quick = {
		adult = yes
		imprisoned = no
		incapable = no
	}
	can_be_employed_as = partner_court_position
	"aptitude(partner_court_position)" >= 2
	NOR = {
		scope:candidate_1 ?= this
		scope:candidate_2 ?= this
		scope:candidate_3 ?= this
		has_any_moderate_bad_relationship_with_character_trigger = { CHARACTER = root }
	}
	opinion = {
		target = root
		value >= -25
	}
	# We keep disloyal and loyalty-hooked characters out, for tone
	NOR = {
		has_trait = disloyal
		root = { has_strong_hook = prev }
	}
}

scripted_effect gdp_merchant_0020_apply_appointment_effect = {
	reverse_add_opinion = {
		target = $CHOSEN$
		modifier = grateful_opinion
		opinion = 50
	}
	every_in_list = {
		list = candidates
		limit = {
			NOT = { this = $CHOSEN$ }
		}
		custom = gdp_merchant.0020.a.custom
		add_opinion = {
			target = $EMPLOYER$
			modifier = annoyed_opinion
			opinion = -10
		}
	}
	if = {
		limit = { NOT = { has_hook = $CHOSEN$ } }
		add_hook = {
			target = $CHOSEN$
			type = loyalty_hook
		}
	}
	# House Unity impact
	house ?= {
		if = {
			limit = { $CHOSEN$.house ?= this }
			add_unity_value = {
				value = medium_unity_gain
				character = $EMPLOYER$
				desc = merchant_unity_partner_gain
			}
		}
		else = {
			add_unity_value = {
				value = medium_unity_loss
				character = $EMPLOYER$
				desc = merchant_unity_partner_loss
			}
		}
	}
}

# Associates struggle for empty/ineffectual Partner position
gdp_merchant.0020 = {
	type = character_event
	title = gdp_merchant.0020.t
	desc = {
		desc = gdp_merchant.0020.desc
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:old_holder }
				desc = gdp_merchant.0020.desc.partner_exists
			}
			desc = gdp_merchant.0020.desc.fallback
		}
		desc = gdp_merchant.0020.desc.body
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:candidate_3 }
				desc = gdp_merchant.0020.desc.outro_three
			}
			triggered_desc = {
				trigger = { exists = scope:candidate_2 }
				desc = gdp_merchant.0020.desc.outro_two
			}
			desc = gdp_merchant.0020.desc.outro_one
		}
	}
	theme = merchant
	override_background = { reference = study }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		trigger = { exists = scope:old_holder }
		character = scope:old_holder
		animation = stressed_teacher
	}
	lower_left_portrait = scope:candidate_1
	lower_center_portrait = scope:candidate_2
	lower_right_portrait = scope:candidate_3

	cooldown = { years = 10 }

	trigger = {
		# Only for merchants
		government_has_flag = government_is_merchant
		# Either you don't have a Partner, or your Partner sucks.
		OR = {
			can_employ_court_position_type = partner_court_position
			AND = {
				employs_court_position = partner_court_position
				any_court_position_holder = {
					type = partner_court_position
					"aptitude(partner_court_position)" <= 1
				}
			}
		}
		# Standard checks.
		is_adult = yes
		is_available_allow_travelling = yes
		# Make sure we're on dry land.
		is_location_valid_for_travel_event_on_land = yes
		# Squabbles shouldn't happen in concerted households
		NOT = {
			house ?= { has_house_unity_stage = ennobled_cs }
		}
		# You have at least one characters who'd be better.
		any_courtier = {
			gdp_merchant_event_0020_better_partners_trigger = yes
		}
	}

	weight_multiplier = {
		modifier = { # Increase weight early on
			# If we're in China/Korea, there is Confucian content we want to expedite instead
			location = {
				NOR = {
					geographical_region = world_asia_china
					geographical_region = world_asia_korea
				}
			}
			add = {
				value = 100
				subtract = {
					value = years_as_ruler
					multiply = 10
				}
				min = 0 # avoid -1
			}
		}
		modifier = { # Increase weight if partner honks
			employs_court_position = partner_court_position
			any_court_position_holder = {
				type = partner_court_position
			}
			add = {
				value = 5
				subtract = "court_position:partner_court_position.aptitude(partner_court_position)"
				multiply = 10
			}
		}
	}

	immediate = {
		# Is the spot vacant, or merely incompetent?
		court_position_old_holder_effect = { POS = partner EMPLOYER = root }
		# Available options
		ordered_courtier = {
			limit = { gdp_merchant_event_0020_better_partners_trigger = yes }
			order_by = {
				value = "aptitude(partner_court_position)"
				# Stewardship only breaks ties, so we add a little of it after scaling aptitude
				multiply = 10
				add = {
					add = stewardship
					divide = 10
					ceiling = yes
				}
				# then ai_energy
				add = {
					value = ai_energy
					multiply = 0.1
				}
			}
			check_range_bounds = no
			max = 3
			if = {
				limit = { exists = scope:candidate_2 }
				save_scope_as = candidate_3
			}
			else_if = {
				limit = { exists = scope:candidate_1 }
				save_scope_as = candidate_2
			}
			else = { save_scope_as = candidate_1 }
			add_to_list = candidates
		}
	}

	# Choose scope:candidate_1
	option = {
		name = gdp_merchant.0020.a
		court_position_grant_effect = {
			POS = partner
			CANDIDATE = scope:candidate_1
			EMPLOYER = root
		}
		gdp_merchant_0020_apply_appointment_effect = {
			CHOSEN = scope:candidate_1
			EMPLOYER = root
		}
		ai_chance = {
			base = 1
			modifier = {
				add = {
					value = "scope:candidate_1.aptitude(partner_court_position)"
					multiply = 10
				}
			}
			opinion_modifier = { opinion_target = scope:candidate_1 multiplier = 0.25 }
			opinion_modifier = {
				trigger = { exists = scope:old_holder }
				opinion_target = scope:old_holder
				multiplier = -0.25
			}
		}
	}

	# Choose scope:candidate_2
	option = {

scripted_trigger fixer_secret_trigger = {
	OR = {
		is_shunned_for = secret_owner
		AND = {
			NOT = {
				is_blackmailable_secret_trigger = {
					PARTICIPANT = secret_owner
					BLACKMAILER = root
				}
			}
			secret_is_always_interesting_trigger = yes
		}
	}
	NOT = { is_known_by = root }
	# Spymaster should not reveal secrets targeting themselves unless it is a murder attempt
	trigger_if = {
		limit = {
			exists = secret_target
			NOT = { secret_type = secret_murder_attempt }
		}
		secret_target != $FIXER$
	}
	# If the spymaster is the father of a secret bastard, don't reveal that
	trigger_if = {
		limit = {
			OR = {
				secret_type = secret_disputed_heritage
				secret_type = secret_unmarried_illegitimate_child
			}
			exists = secret_target.real_father
		}
		secret_target.real_father != $FIXER$
	}
	NOT = {
		any_secret_participant = {
			OR = {
				this = $FIXER$
				this = root
			}
		}
	}
	save_temporary_scope_as = suitable_secret
	trigger_if = {
		limit = {
			secret_type = secret_disputed_heritage
			exists = secret_target
		}
		NOT = { secret_target = { has_trait = legitimized_bastard } }
	}
	OR = {
		secret_is_always_interesting_trigger = yes
		is_blackmailable_secret_trigger = { BLACKMAILER = root PARTICIPANT = scope:suitable_secret.secret_owner }
	}
}

		}
	}

# Accuser's rite treats this loan as a religious offense.
scripted_trigger gdp_usury_penance_shunned_doctrine_trigger = {
	OR = {
		rite = { rite_has_parameter = usury_forbidden }
		AND = {
			rite = { rite_has_parameter = usury_outsiders_only }
			religion = root.religion
		}
	}
}

scripted_trigger gdp_usury_penance_discouraged_doctrine_trigger = {
	rite = { rite_has_parameter = usury_discouraged }
}

scripted_trigger gdp_usury_penance_accepted_doctrine_trigger = {
	gdp_usury_penance_shunned_doctrine_trigger = no
	gdp_usury_penance_discouraged_doctrine_trigger = no
}

scripted_effect gdp_usury_penance_cast_fallback_clergy_effect = {
	if = {
		limit = {
			NOT = { exists = scope:accuser }
		}
		random_pool_character = {
			province = scope:debtor.location
			limit = {
				is_adult = yes
				faith = scope:debtor.faith
				rite = scope:debtor.rite
				OR = {
					has_trait = cleric
					has_trait = devoted
					has_trait = education_learning_1
					has_trait = education_learning_2
					has_trait = education_learning_3
					has_trait = education_learning_4
				}
			}
			save_scope_as = accuser
		}
	}
	if = {
		limit = { NOT = { exists = scope:accuser } }
		scope:debtor = { save_scope_as = monk_employer }
		create_character = {
			location = scope:debtor.location
			template = pam_monk_courtier_template
			save_scope_as = accuser
			after_creation = {
				add_character_flag = created
				# Warcraft
				trigger_race_giving_no_gene_effect = yes
			}
		}
	}

						add = 1
						# Prioritize food
						market_has_food_need_trigger = yes
					}
					modifier = {
						factor = 2
						scope:commissioner.capital_province.barony.trade_market ?= this
					}
				}
				save_scope_as = trade_market
				# 1. Establish all categories with need or good need
				every_trade_category_with_unfulfilled_need = { add_to_list = needed_categories }
				every_trade_good_with_unfulfilled_need = {
					trade_good_need_category = { add_to_list = needed_categories }
				}
				# 2. Find a suitable good need from one of those categories
				random_in_list = {
					list = needed_categories
					weight = {
						base = 2
						# Weight by need
						diversify_commission_route_category_need_modifier = yes
						# Prefer to diversify if possible
						diversify_commission_route_category_routes_modifier = yes
						# Prioritize food
						modifier = { 
							OR = {
								this = trade_good_category:trade_category_cereals
								this = trade_good_category:trade_category_meat
							}
							factor = 2
						}
					}
					save_scope_as = category
					scope:category ?= {
						random_trade_good_in_category = {
							weight = {
								base = 2
								# Weight by need
								diversify_commission_route_good_need_modifier = yes
								# Prefer to diversify if possible
								diversify_commission_route_good_routes_modifier = yes
							}
							save_scope_as = target
						}
					}
				}
			}
		}
		if = {
			limit = { exists = scope:target }
			random_tributary = {
				limit = {
					government_has_flag = government_is_merchant
					commission_trade_route_merchant_trigger = {
						COMMISSIONER = scope:commissioner
						MERCHANT = this
					}
					commission_trade_route_good_trigger = {
						COMMISSIONER = scope:commissioner
						MARKET = scope:trade_market
						GOOD = scope:target
						MERCHANT = this
					}
				}
				weight = {
					base = 1
					# REALM
					modifier = {
						capital_province.county.holder.top_liege = scope:commissioner.top_liege
						add = 1
					}
					# COMPANY SIZE
					modifier = { # Prefer cool companies
						add = company_size_value
					}
					# CULTURE
					modifier = {
						culture = scope:commissioner.culture
						add = 1
					}
					modifier = {
						culture = { has_same_culture_heritage = scope:commissioner.culture }
						add = 1
					}
					modifier = {
						culture = { has_same_culture_language = scope:commissioner.culture }
						add = 1
					}
					# FAITH
					modifier = {
						rite = scope:commissioner.rite
						add = 1
					}
					modifier = {
						faith = scope:commissioner.faith
						add = 1
					}
					modifier = {
						religion = scope:commissioner.religion
						add = 1
					}
					modifier = {
						faith = {
							faith_hostility_level = {
								target = scope:commissioner.faith
								value >= faith_astray_level
							}
						}
						add = -1
					}
					modifier = {
						faith = {
							faith_hostility_level = {
								target = scope:commissioner.faith
								value >= faith_hostile_level
							}
						}
						add = -1
					}
					modifier = {
						faith = {
							faith_hostility_level = {
								target = scope:commissioner.faith
								value >= faith_evil_level
							}
						}
						add = -1
					}
					# EXPERTISE
					first_valid = {
						modifier = {
							factor = 4
							exists = scope:target
							primary_title = { save_temporary_scope_as = company_title_temp }
							scope:target = {
								any_trade_category = {
									save_temporary_scope_as = category_expertise_temp
									scope:company_title_temp = {
										trade_good_category_level = {
											target = scope:category_expertise_temp
											value >= merchant_monopoly_trade_expertise_level_high_value
										}
									}
								}
							}
						}
						modifier = {
							factor = 3
							exists = scope:target
							primary_title = { save_temporary_scope_as = company_title_temp }
							scope:target = {
								any_trade_category = {
									save_temporary_scope_as = category_expertise_temp
									scope:company_title_temp = {
										trade_good_category_level = {
											target = scope:category_expertise_temp
											value >= merchant_monopoly_trade_expertise_level_medium_value
										}
									}
								}
							}
						}
						modifier = {
							factor = 2
							exists = scope:target
							primary_title = { save_temporary_scope_as = company_title_temp }
							scope:target = {
								any_trade_category = {
									save_temporary_scope_as = category_expertise_temp
									scope:company_title_temp = {
										trade_good_category_level = {
											target = scope:category_expertise_temp
											value >= merchant_monopoly_trade_expertise_level_low_value
										}
									}
								}
							}
						}
					}
					# Favor players for visibility
					modifier = {
						is_ai = no
						factor = 3
					}
				}
				save_scope_as = merchant
			}
		}
		if = {
			limit = { exists = scope:merchant }
			scope:merchant = {
				trigger_event = gdp_merchant.9021
			}
		}
		# ERROR SUPPRESSION
		if = {
			limit = {
				always = no
				any_in_list = { variable = commission_routes }
			}
			add_to_variable_list = {
				name = commission_routes
				target = scope:commissioner
			}
		}
	}
}

gdp_merchant.9021 = {
	type = letter_event
	opening = gdp_merchant.9021.opening
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {  
					scope:commissioner = {
						OR = {
							has_diarchy_type = regency
							has_diarchy_type = temporary_regency
						}	
					}
				}
				desc = gdp_merchant.9021.diarch.desc
			}
			desc = gdp_merchant.9021.no_diarch.desc
		}
		desc = gdp_merchant.9021.desc
	}
	sender = scope:letter_sender

	immediate = {
		## Making sure that child rulers are represented by their diarch
		if = {
			limit = {
				scope:commissioner = {
					OR = {
						has_diarchy_type = regency
						has_diarchy_type = temporary_regency
					}
				}
			}
			scope:commissioner.diarch ?= {
				save_scope_as = letter_sender
			}	
		}
		else = {
			scope:commissioner = {
				save_scope_as = letter_sender
			}
		}
	}

	option = {
		name = gdp_merchant.9021.a
		# The market can change while the letter waits for an answer
		trigger = {
			commission_trade_route_good_trigger = {
				COMMISSIONER = scope:commissioner
				MARKET = scope:trade_market
				GOOD = scope:target
				MERCHANT = scope:merchant
			}
		}
		show_as_unavailable = { always = yes }
		scope:commissioner = {
			# ...and again between choosing this option and it taking effect
			if = {
				limit = {
					commission_trade_route_good_trigger = {
						COMMISSIONER = scope:commissioner
						MARKET = scope:trade_market
						GOOD = scope:target
						MERCHANT = scope:merchant
					}
				}
				add_prestige = {
					subtract = gdp_commission_route_cost_value
				}
				scope:merchant = {
					add_prestige = {
						value = gdp_commission_route_cost_value
						divide = 2
					}
				}
				commission_merchant_effect = {
					MERCHANT = scope:merchant
					GOOD = scope:target
					DEST = scope:trade_market
					YEARS = 10
				}
			}
		}
		custom_tooltip = commission_merchant_cancel_warning_tt
		ai_chance = {
			base = 25
			opinion_modifier = {
				opinion_target = scope:commissioner
				multiplier = 0.25
			}
			ai_value_modifier = {
				ai_greed = 0.5
			}
		}
	}

	option = {
		name = gdp_merchant.9021.b
		# Turning down a commission that could not go ahead anyway is not held against the merchant
		if = {
			limit = {
				commission_trade_route_good_trigger = {
					COMMISSIONER = scope:commissioner
					MARKET = scope:trade_market
					GOOD = scope:target
					MERCHANT = scope:merchant
				}
			}
			custom_tooltip = {
				text = gdp_merchant.9021.b.cooldown_tt
				set_variable = {
					name = commission_trade_route_cooldown
					years = commission_trade_route_merchant_cooldown_years_value
				}
			}
			reverse_add_opinion = {
				modifier = disappointed_opinion
				target = scope:commissioner
				opinion = -15
			}
		}
		ai_chance = {
			base = 10
			opinion_modifier = {
				opinion_target = scope:commissioner
				multiplier = -0.25
			}
			ai_value_modifier = {
				ai_greed = -0.5
			}
		}
	}
	# Let's debug
	after = {
		if = {
			limit = {
				this = character:bookmark_wuhsha_al_dallala
			}
			if = {
				limit = {
					NOT = { exists = global_var:commissions_tally }
				}
				set_global_variable = {
					name = commissions_tally
					value = 1
				}
			}
			change_global_variable = {
				name = commissions_tally
				add = 1
			}
		}
	}
}
