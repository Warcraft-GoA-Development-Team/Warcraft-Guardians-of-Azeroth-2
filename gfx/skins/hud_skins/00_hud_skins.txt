norse = {
	is_valid = {
		has_dlc_feature = the_northern_lords
	}
	is_shown = {
		OR = {
			# Warcraft
			like_norse_culture_trigger = yes
			AND = {
				# Warcraft
				like_germanic_religion_trigger = yes
				culture = { like_heritage_north_germanic_trigger = yes }
			}
		}
	}
}

		culture = { like_heritage_berber_trigger = yes }
		# Why not?

iranian = {
	is_valid = {
		has_dlc_feature = legacy_of_persia
	}

		culture = { has_cultural_pillar = heritage_gilnean }
	}
}

byzantine = {
	is_valid = {
		has_dlc_feature = roads_to_power
	}
	is_shown = {
		culture = { like_heritage_byzantine_trigger = yes }
	}
}

nomad = {
	is_valid = {
		has_dlc_feature = khans_of_the_steppe
	}
	is_shown = {
		government_has_flag = government_is_nomadic
	}
}

asia = {
	is_shown = {
		has_tgp_dlc_trigger = yes
		OR = {
			culture = { like_heritage_chinese_trigger = yes }
			culture = { like_heritage_japonic_trigger = yes }
			culture = { like_heritage_indo_aryan_trigger = yes }
			culture = { like_heritage_tai_trigger = yes }
			culture = { like_heritage_mon_khmer_trigger = yes }
			culture = { like_heritage_austronesian_trigger = yes }
			culture = { like_heritage_korean_trigger = yes }
			culture = { like_heritage_tungusic_trigger = yes }
			culture = { like_heritage_ainuic_trigger = yes }
			culture = { like_heritage_qiangic_trigger = yes }
			culture = { like_heritage_buyeo_trigger = yes }
			culture = { like_heritage_tibetan_trigger = yes }
			culture = { like_heritage_viet_trigger = yes }
			culture = { like_heritage_hmongic_trigger = yes }
			culture = { like_heritage_nivkh_trigger = yes }
			culture = { like_heritage_burman_trigger = yes }
			culture = { like_heritage_dravidian_trigger = yes }
			AND = {
				culture = { like_heritage_mongolic_trigger = yes }
				NOT = {
					government_has_flag = government_is_nomadic
				}
			}
		}
	}
}

christian = {
	is_shown = {
		AND = {
			like_christianity_religion_trigger = yes

			NOT = {
				OR = {
					culture = { like_heritage_byzantine_trigger = yes }
					culture = { like_heritage_caucasian_trigger = yes }
					culture = { like_heritage_iberian_trigger = yes }
					culture = { like_heritage_north_germanic_trigger = yes }
				}
			}
		}
	}
}
