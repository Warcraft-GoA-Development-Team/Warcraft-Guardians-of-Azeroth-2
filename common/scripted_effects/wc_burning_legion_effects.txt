summon_burning_legion_effect = {
    create_character = {
        template = archimonde_template
        location = title:c_dalaran.title_province
        save_scope_as = lordarchimonde

		after_creation = {
			add_trait = legion_commander
		}
    }
    give_land_to_settler = { TO_WHO = scope:lordarchimonde COUNTY_NAME = title:c_dalaran GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
    give_land_to_settler = { TO_WHO = scope:lordarchimonde COUNTY_NAME = title:h_burning_legion GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
    give_land_to_settler = { TO_WHO = scope:lordarchimonde COUNTY_NAME = title:c_lordaeron GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }

    title:h_burning_legion = {
        set_capital_county = title:c_dalaran
    }

    hidden_effect = {
		scope:lordarchimonde = {
			every_courtier = {
				limit = { has_faith = faith:death_god }
				set_character_faith = faith:burning_legion_religion
			}
			every_vassal_or_below = {
				limit = {
                    has_faith = faith:death_god
                    NOT = { has_character_modifier = important_lore_character } # Stop Arthas from converting
                }
				set_character_faith = faith:burning_legion_religion
			}
		}
	}
	scope:lordarchimonde = {
		every_sub_realm_county = {
			limit = {
				holder.top_liege = scope:lordarchimonde
				culture = culture:scourge
			}
			set_county_faith = faith:burning_legion_religion
		}
		scope:lordarchimonde = { every_held_county = { change_county_control = 100 } }
	}

	hidden_effect = {
		add_to_global_variable_list = {
			name = warcraft_invaders_story
			target = flag:burning_legion_story_started
		}
	}

	stops_being_legions_puppet_effect = yes

	scope:lordarchimonde = {
		trigger_event = { # Delayed army spawn & give them HoF title
			id = burninglegion.0005
			days = 3
		}
	}

	scope:lordarchimonde.dynasty = {
		add_dynasty_prestige = 37000
	}

    # Creates or finds dreadlords
	# Tichondrius
	if = {
		limit = {
			exists = character:430000
			character:430000 = { is_lowborn = no } # Confirms it's not purged from history
		}
	}
	else = {
		create_character = {
            employer = scope:lordarchimonde
			template = tichondrius_character_template
			name = "Tichondrius"
		}
	}
    scope:lordarchimonde = {
        add_courtier = character:430000
    }
	# Varimathras
