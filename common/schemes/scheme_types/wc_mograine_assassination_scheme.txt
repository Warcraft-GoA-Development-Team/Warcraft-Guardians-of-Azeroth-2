wc_mograine_assassination = {
	# Basic Setup
	skill = intrigue
	desc = wc_mograine_assassination_desc
	success_desc = "WC_MOGRAINE_ASSASSINATION_SUCCESS_DESC"
	discovery_desc = "WC_MOGRAINE_ASSASSINATION_DISCOVERY_DESC"
	icon = icon_scheme_personal
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	category = hostile
	target_type = character

	# hostile + is_secret = yes is required for the Execute Scheme button to
	# work - category = personal left it dead (no personal-category scheme in
	# the game uses on_hud_click).
	is_secret = yes
	maximum_secrecy = 90
	is_basic = no
	uses_resistance = no
	cooldown = { years = 2 }
	phases_per_agent_charge = 1

	agent_groups_owner_perspective = { courtiers }
	agent_groups_target_character_perspective = { courtiers }
	valid_agent = {
		is_valid_agent_standard_trigger = yes
		# Darion Mograine - the favored son, loyal to Alexandros, would never
		# conspire against him.
		NOT = { this = character:60020 }
	}
	# Same dynamic recruitment as vanilla murder_scheme: courtiers can join on
	# their own over time in addition to whoever fills the on_start slots.
	agent_leave_threshold = -25
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_basic_suite_modifier = yes
	}

	# Parameters
	speed_per_skill_point = -15
	spymaster_speed_per_skill_point = 0
	success_chance_growth_per_skill_point = 0.1
	base_progress_goal = 365
	base_maximum_success = 90
	minimum_success = 5

	# Core Triggers
	allow = {
		is_imprisoned = no
	}
	valid = {
		scope:target = {
			is_alive = yes
			is_adult = yes
			is_imprisoned = no
			exists = location
		}
		# Owner is Balnazzar, not Renault, so his death wouldn't otherwise
		# invalidate the scheme - checked explicitly here instead.
		exists = character:60019 # Renault Mograine
		character:60019 = { is_alive = yes }
	}

	odds_prediction = {
		add = base_odds_prediction_target_is_char_value
		min = 0
	}

	# Base Chances
	base_success_chance = {
		base = 35

		modifier = {
			add = 2
			scope:owner = { has_trait = education_intrigue_1 }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_MY_EDUCATION"
		}
		modifier = {
			add = 4
			scope:owner = { has_trait = education_intrigue_2 }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_MY_EDUCATION"
		}
		modifier = {
			add = 6
			scope:owner = { has_trait = education_intrigue_3 }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_MY_EDUCATION"
		}
		modifier = {
			add = 8
			scope:owner = { has_trait = education_intrigue_4 }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_MY_EDUCATION"
		}
		modifier = {
			add = 10
			scope:owner = { has_trait = education_intrigue_5 }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_MY_EDUCATION"
		}
		# The paladin mask (education, personality) has no scheming instincts of
		# its own - this is the demon underneath actually doing the work.
		modifier = {
			add = 15
			scope:owner = { has_trait = nathrezim_disguise }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_TRUE_NATURE"
		}
		modifier = {
			add = -10
			scope:target = { has_trait = paranoid }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_THEIR_PARANOIA"
		}
		modifier = {
			add = 5
			scope:target = { has_trait = trusting }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_THEIR_TRUST"
		}

		# Renault leading his own father into the trap and Kel'Thuzad's counsel
		# on making it look like an accident push the flat baseline (35+15-30
		# Plot Armor = 20) up to 50 before per-phase accumulation.
		modifier = {
			add = 15
			exists = character:60019 # Renault Mograine
			character:60019 = { is_alive = yes }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_RENAULTS_ACCESS"
		}
		modifier = {
			add = 15
			scope:owner = { has_character_flag = balnazzar_kelthuzad_pact_sealed }
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_KELTHUZAD_COUNSEL"
		}
	}
	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value

		# Renault's blood, Kel'Thuzad's pact, and a complicit court push the
		# nominal base to 60 before target countermeasures apply. Note:
		# base_secrecy takes add = { value = X desc = Y }, not modifier {}
		# like base_success_chance above - ck3-tiger errors on the latter.
		if = {
			limit = {
				exists = character:60019 # Renault Mograine
				character:60019 = { is_alive = yes }
			}
			add = {
				value = 20
				desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_RENAULTS_HAND"
			}
		}
		if = {
			limit = {
				scope:owner = { has_character_flag = balnazzar_kelthuzad_pact_sealed }
			}
			add = {
				value = 15
				desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_KELTHUZAD_VEIL"
			}
		}
		add = {
			value = 5
			desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_WILLING_BLINDNESS"
		}

		# No vanilla field grants flat secrecy per phase, so it's tracked by
		# hand via a scheme variable incremented in on_phase_completed, capped
		# at +30 (6 phases). Guarded by exists = var:X since base_secrecy also
		# evaluates during the pre-start odds preview, before the variable is set.
		if = {
			limit = { exists = var:wc_mograine_secrecy_phases_elapsed }
			add = {
				value = var:wc_mograine_secrecy_phases_elapsed
				multiply = 5
				max = 30
				desc = "SCHEME_WC_MOGRAINE_ASSASSINATION_GROWING_ROUTINE"
			}
		}
	}

	# On Actions
	on_start = {
		scheme_owner = {
			trigger_event = wc_mograine_assassination_outcome.1001
		}
		add_agent_slot = wc_mograine_whitemane_agent
		add_agent_slot = wc_mograine_voss_agent
		add_agent_slot = wc_mograine_founders_agent
		add_agent_slot = wc_mograine_enforcer_agent
		add_agent_slot = wc_mograine_watcher_agent
		# Populates the real "Advantages" resource the Execute Scheme button
		# spends from - without this the button shows a phantom count and
		# does nothing (confirmed: murder_scheme.txt calls this identically).
		add_scheme_starting_opportunities_intrigue_effect = yes
		set_variable = { name = wc_mograine_secrecy_phases_elapsed value = 0 }
	}
	# Progress reaching 100% only readies the ambush - suppress_scheme_follow_up_event_till_input_given_effect
	# holds the roll until Execute Scheme is clicked (on_hud_click below), same as murder_scheme.txt.
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		wc_mograine_assassination_scheme_prep_effect = yes
		# Self-heals schemes started before this variable existed - on_start
		# only runs once, at creation, so it initializes to 0 here if missing.
		if = {
			limit = { NOT = { exists = var:wc_mograine_secrecy_phases_elapsed } }
			set_variable = { name = wc_mograine_secrecy_phases_elapsed value = 0 }
		}
		change_variable = { name = wc_mograine_secrecy_phases_elapsed add = 1 }
	}
	on_hud_click = {
		wc_mograine_assassination_scheme_prep_effect = yes
	}
	on_invalidated = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:owner = {
			send_interface_toast = {
				type = event_generic_neutral
				title = wc_mograine_assassination_invalidated_title
				custom_tooltip = wc_mograine_assassination_invalidated_desc
			}
		}
	}
	on_monthly = {
		hostile_scheme_monthly_discovery_chance_effect = yes
		if = {
			limit = { NOT = { exists = scope:discovery_event_happening } }
			scheme_owner = {
				trigger_event = { on_action = wc_mograine_assassination_ongoing }
			}
		}
	}
}
