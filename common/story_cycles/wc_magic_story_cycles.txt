# Whispers of the Old Ones
story_cycle_whispers = {
	on_setup = {
        story_owner = {
            add_character_flag = had_whispers
        }
	}

	on_end = {
	}

	on_owner_death = {
        # Send a message to the Tormentor
        end_story = yes
	}

    # Something happens - random events
	effect_group = {
		days = { 30 60 }
		chance = 50

		trigger = {
			scope:tormentor = { is_alive = yes }
		}

		first_valid = {
			triggered_effect = {
				trigger = { always = yes }
				effect = {
					story_owner = {
						trigger_event = {
							on_action = ongoing_whispers_void_events
						}
					}
				}
			}
		}
	}

    # The whispers end
    effect_group = {
		years = { 2 4 }
		
		first_valid = {
			triggered_effect = {
				trigger = { always = yes }
				effect = {
					story_owner = {
						trigger_event = {
							on_action = whispers_void_outcome_events
						}
					}
				}
			}
		}
	}
}
