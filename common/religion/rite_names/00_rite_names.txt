# Warcraft - vanilla rite names are tied to real-world religions/cultures (replace_path).
# Only the generic entries are kept; dynamic rites otherwise fall back to d_dynamic_rite_<n>.

# Generic mainline rite for faiths with no scripted rites - they will be created without a founder for faiths without any scripted rite
d_mono_rite_faith_name = {
	trigger = {
		NOT = { exists = scope:founder }
	}
}

# Generic Rite, Ruler
generic_rite = {
	name = d_fallback_rite
	trigger = {
		scope:founder ?= {
			is_ruler = yes
		}
	}
}

# Generic Rite, Unlanded w. Location
generic_rite_2 = {
	name = d_fallback_rite_2
	trigger = {
		exists = scope:founder.location.county
	}
}

# Generic Rite, Unlanded without Location
generic_rite_3 = {
	name = d_generic_rite
}
