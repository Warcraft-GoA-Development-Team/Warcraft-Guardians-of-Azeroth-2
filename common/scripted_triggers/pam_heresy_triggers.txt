# Per-heresy gating, geography helpers, and component eligibility checks.
# Resurgence model: each historical heresy can fire more than once per game.
#   first outbreak: pam_heresy_X_first_fired global var is unset
#   resurgence:    first_fired set, plus pam_heresy_X_cooldown timed flag absent
# A 50-year cooldown is set after every outbreak via timed global variable.

# --- Component eligibility (faith scope) ---

# The second and third heresy components are blocked below this
# The first still fires, but we dampen the pulse with pam_heresy_pulse_no_event_value
pam_heresy_main_rite_substantial_trigger = {
	main_rite.rite_counties >= pam_heresy_main_rite_county_threshold_value
}

# Component 2 gate: main rite must be substantial (size floor) AND hold at least 60% of faith counties (dominance share).
pam_heresy_c2_main_rite_dominant_trigger = {
	pam_heresy_main_rite_substantial_trigger = yes
	main_rite.rite_counties >= pam_heresy_c2_main_rite_share_required_value
}

pam_heresy_c3_eligible_trigger = {
	like_christianity_religion_trigger = yes
	pam_heresy_main_rite_substantial_trigger = yes
	OR = {
		pam_heresy_lollard_eligible_trigger = yes
		pam_heresy_hussite_eligible_trigger = yes
		pam_heresy_joachimite_eligible_trigger = yes
		pam_heresy_cathar_eligible_trigger = yes

	}
}

# True when the recipient is already on the heresy's rite.
# Used to gate "embrace" / reframe "reject" options and description text so the
# event does not address one of the heresy's own followers as an outside observer.
pam_heresy_player_is_heresy_insider_trigger = {
	trigger_if = {
		limit = { exists = scope:new_rite }
		rite = scope:new_rite
	}
	trigger_else_if = {
		limit = { exists = scope:heresy_faith }
		faith = scope:heresy_faith
	}
	trigger_else = {
		always = no
	}
}

# --- Component 2 founder eligibility (character scope) ---
# Mirrors is_eligible_rite_growth_founder_trigger but is permissive about existing heresiarch / herald carriers
# C2 is the "charismatic schismatic" path and those traits are an asset here.
# Archbishop-only candidates (no held county on the source rite) are accepted via the clerical_region branch.
is_eligible_pam_heresy_c2_founder_trigger = {
	is_available_ai_adult = yes
	rite = scope:source_rite
	learning >= medium_skill_rating
	NOR = {
		recently_converted_faith_trigger = yes
		has_character_flag = recently_converted_by_heresy
		has_character_flag = recently_converted_by_rite_growth
		faith.religious_head ?= this
		scope:source_rite.head_of_rite ?= this
		highest_held_title_tier >= tier_empire
		government_has_flag = government_is_nomadic
		government_has_flag = government_is_herder
	}
	OR = {
		AND = {
			is_landed = yes
			highest_held_title_tier >= tier_duchy
			any_held_title = {
				tier = tier_county
				rite = scope:source_rite
			}
		}
		any_held_title = { has_clerical_region = yes }
	}
}

# Test convenience trigger for narrative branching (e.g. event text).
pam_heresy_c2_founder_is_archbishop_trigger = {
	scope:founder = {
		any_held_title = { has_clerical_region = yes }
	}
}

# --- Epicenter ruler eligibility (character scope) ---
# A duke+ same-faith landed AI ruler.
# Capital-in-region check is applied separately by the picker (with title_province wrapping).
is_eligible_heresy_epicenter_ruler_trigger = {
	is_landed = yes
	is_ai = yes
	highest_held_title_tier >= tier_duchy
	NOR = {
		this = faith.religious_head
		has_trait = heresiarch
		has_trait = herald
		recently_converted_faith_trigger = yes
	}
}

# --- Component 3 game-rule gates ---

pam_heresy_relaxed_rule_trigger = {
	has_game_rule = relaxed_regional_heresy
}

pam_heresy_strict_rule_trigger = {
	has_game_rule = strict_regional_heresy
}

pam_heresy_default_rule_trigger = {
	OR = {
		has_game_rule = default_regional_heresy
		# Defensive: if no rule explicitly matches, treat as default.
		AND = {
			NOT = { has_game_rule = strict_regional_heresy }
			NOT = { has_game_rule = relaxed_regional_heresy }
		}
	}
}
