# A title is transferred to a new character
# root = the new holder
# scope:title = the title that changes hands
# scope:previous_holder = previous holder. Might be dead
wc_on_title_gain = {
	effect = {
		# Horde characters give Black Morass lands back to Black Tooth Grin
		if = {
			limit = {
				is_from_the_horde_trigger = yes
				exists = title:d_blacktoothgrin_clan.holder
				NOT = { this = title:d_blacktoothgrin_clan.holder }
				title:d_blacktoothgrin_clan.holder = { is_from_the_horde_trigger = yes }
				
				# The title we just gained isn't itself in Black Morass...
				NOT = { scope:title.title_capital_county.title_province = { geographical_region = world_eastern_kingdoms_azeroth_black_morass } }
				# ...but we're still sitting on land there
				any_held_title = {
					tier = tier_county
					title_province = { geographical_region = world_eastern_kingdoms_azeroth_black_morass }
				}
			}

			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
			}

			every_held_title = {
				limit = {
					tier = tier_county
					title_province = { geographical_region = world_eastern_kingdoms_azeroth_black_morass }
				}
				change_title_holder = {
					holder = title:d_blacktoothgrin_clan.holder
					change = scope:change
				}
			}

			resolve_title_and_vassal_change = scope:change

			title:d_blacktoothgrin_clan.holder = { save_temporary_scope_as = black_morass_new_holder }

			send_interface_message = {
				type = msg_black_morass_returned
				title = black_morass_returned.t
				desc = black_morass_returned.tooltip
				left_icon = root
				right_icon = scope:black_morass_new_holder
			}
			scope:black_morass_new_holder = {
				send_interface_message = {
					type = msg_black_morass_received
					title = black_morass_received.t
					desc = black_morass_received.tooltip
					left_icon = root
					right_icon = scope:black_morass_new_holder
				}
			}
		}

		# Horde - AI scripted land distribution
		if = {
			limit = {
				is_ai = yes
				is_horde_invader_trigger = yes
			}

			if = { # Stormwind: Horde -> Gul'dan or Stormreaver
				limit = {

				}
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
				}

				resolve_title_and_vassal_change = scope:change
			}
			
			if = { # Nortshire: Horde -> Twilight's Hammer
				limit = {

				}
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
				}

				resolve_title_and_vassal_change = scope:change
			}

			if = { # Ironforge: Horde -> Bleeding Hollow
				limit = {

				}
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
				}

				resolve_title_and_vassal_change = scope:change
			}

			if = { # Wetlands - Grim Batol: Horde -> Dragonmaw
				limit = {

				}
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
				}

				resolve_title_and_vassal_change = scope:change
			}
		}
	}
}
