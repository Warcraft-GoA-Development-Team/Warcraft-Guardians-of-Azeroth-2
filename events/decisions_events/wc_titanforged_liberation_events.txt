namespace = wc_titanforged_liberation

wc_titanforged_liberation.0002 = {
	type = character_event
	content_source = dlc_GOA
	title = wc_titanforged_liberation.0002.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:ceded_duchy }
				desc = wc_titanforged_liberation.0002.desc
			}
			desc = wc_titanforged_duchy_invalid_tt
		}
	}
	theme = realm

	immediate = {
		random_realm_de_jure_duchy = {
			limit = { wc_titanforged_duchy_ready_trigger = yes }
			save_scope_as = ceded_duchy
		}
		if = {
			limit = { exists = scope:ceded_duchy }
			scope:ceded_duchy = {
				random_de_jure_county = {
					limit = {
						save_temporary_scope_as = ceded_vassal_county
						holder = { wc_titanforged_local_ruler_trigger = yes }
					}
					save_scope_as = ceded_vassal_county
					holder = { save_scope_as = local_ruler }
				}
			}
		}
	}

	left_portrait = {
		character = root
		animation = emotion_thinking_scepter
	}

	option = {
		name = wc_titanforged_liberation.0002.a
		trigger = { wc_titanforged_can_release_duchy_trigger = yes }
		ai_chance = { base = 1 }
		custom_tooltip = wc_titanforged_release_duchy_tt
		if = {
			limit = { wc_titanforged_can_release_duchy_trigger = yes }
			wc_titanforged_release_duchy_effect = yes
			if = {
				limit = { wc_titanforged_released_duchy_trigger = yes }
				wc_titanforged_resolve_duchy_effect = yes
			}
		}
	}

	option = {
		name = wc_titanforged_liberation.0002.b
		trigger = { wc_titanforged_can_release_duchy_trigger = yes }
		ai_chance = { base = 1 }
		custom_tooltip = wc_titanforged_protectorate_tt
		if = {
			limit = { wc_titanforged_can_release_duchy_trigger = yes }
			wc_titanforged_release_duchy_effect = yes
			scope:local_ruler = {
				start_tributary = {
					contract_group = tributary_settled
					suzerain = root
				}
				if = {
					limit = { is_tributary = yes }
					tributary_contract_set_obligation_level = { type = default_tributary_taxes level = 0 }
					tributary_contract_set_obligation_level = { type = default_tributary_levies level = 0 }
					tributary_contract_set_obligation_level = { type = default_tributary_prestige level = 0 }
				}
			}
			if = {
				limit = {
					wc_titanforged_released_duchy_trigger = yes
					scope:local_ruler = { suzerain = root }
				}
				wc_titanforged_resolve_duchy_effect = yes
			}
		}
	}

	option = {
		name = wc_titanforged_liberation.0002.c
		trigger = { wc_titanforged_can_resolve_duchy_trigger = yes }
		ai_chance = { base = 1 }
		custom_tooltip = wc_titanforged_keep_duchy_tt
		if = {
			limit = {
				OR = {
					scope:ceded_duchy = { any_de_jure_county = { has_variable = wc_titanforged_county_pact } }
					AND = {
						exists = var:wc_titanforged_restoration
						var:wc_titanforged_restoration = {
							NOT = { has_variable = wc_titanforged_restoration_county }
							var:wc_titanforged_restoration_duchy = scope:ceded_duchy
							var:wc_titanforged_restoration_method = flag:pact
						}
					}
				}
			}
			custom_tooltip = wc_titanforged_restoration.break_pact_tt
		}
		if = {
			limit = { wc_titanforged_can_resolve_duchy_trigger = yes }
			wc_titanforged_resolve_duchy_effect = yes
		}
	}

	option = {
		name = wc_titanforged_liberation.0002.invalid
		trigger = { wc_titanforged_can_resolve_duchy_trigger = no }
		custom_tooltip = wc_titanforged_duchy_invalid_tt
	}
}
