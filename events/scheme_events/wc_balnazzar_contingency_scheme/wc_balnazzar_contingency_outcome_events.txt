
# Outcome events for the wc_balnazzar_contingency scheme

namespace = wc_balnazzar_contingency_outcome

##################################
# MAINTENANCE EVENTS 0000 - 0099
##################################

# Rolls success/failure when the scheme completes (on_phase_completed pulse)
wc_balnazzar_contingency_outcome.0001 = {
	hidden = yes

	immediate = {
		random = {
			chance = scope:scheme.scheme_success_chance

			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}

		hidden_effect = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = wc_balnazzar_contingency_outcome.2001
			}
			else = {
				trigger_event = wc_balnazzar_contingency_outcome.4001
			}
		}
	}
}

# SUCCESS - 2000 - 2999
#######################################

# Ready, If It Comes to That - the plan is complete.
wc_balnazzar_contingency_outcome.2001 = {
	type = character_event
	window = scheme_successful_event
	title = wc_balnazzar_contingency_outcome.2001.title
	desc = wc_balnazzar_contingency_outcome.2001.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}

	immediate = {
		add_character_flag = balnazzar_contingency_ready
	}

	option = {
		name = wc_balnazzar_contingency_outcome.2001.a

		hidden_effect = {
			if = {
				limit = { exists = scope:scheme }
				scope:scheme = { end_scheme = yes }
			}
		}
	}
}
