macro = {
	description = "Returns true if the DLC in this datacontext is GoA"
	definition = "DLCIsGOA"
	replace_with = "EqualTo_string(Dlc.GetName,Localize('dlc_GOA'))"
}

macro = {
	description = "Returns true if a DLC is enabled. Makes sure the GoA 'DLC' always shows up as enabled even though it technically isn't"
	definition = "GOADLCIsEnabled"
	replace_with = "Or(Dlc.IsEnabled,DLCIsGOA)"
}
