# The Master's Key - inventory artifact that opens Karazhan (read by the future expedition)
wc_wandering_arcanist_create_masters_key_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_illustrious = yes

	create_artifact = {
		name = wc_wandering_arcanist_masters_key_name
		description = wc_wandering_arcanist_masters_key_description
		type = miscellaneous
		visuals = pocket_basic_case
		wealth = scope:wealth
		quality = scope:quality
		template = wc_wandering_arcanist_masters_key_template
		history = {
			type = created
			date = 560.1.1
			recipient = character:5000 # Medivh
			location = province:46 # Karazhan
		}
		modifier = wc_wandering_arcanist_masters_key_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}

	scope:newly_created_artifact = {
		set_variable = { name = wc_wandering_arcanist_masters_key value = yes } # the key itself is the hook the Karazhan expedition reads; not unique, so no historical_unique_artifact
	}
}
