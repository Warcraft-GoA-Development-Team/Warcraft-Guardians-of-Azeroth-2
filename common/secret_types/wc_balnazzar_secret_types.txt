# never add_secret directly - matches the convention noted at the top of 00_secret_types.txt.

secret_dreadlord_disguise = {
	category = religious

	is_valid = {
		scope:secret_owner = { has_trait = balnazzar_disguise }
	}

	is_criminal = {
		always = yes
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			trigger_event = {
				id = wc_balnazzar_story.2000
				days = 2
			}
		}
		secret_exposed_notification_effect = yes
	}
}
