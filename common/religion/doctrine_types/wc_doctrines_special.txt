# Evil 
# note that this is a pluralism doctrine technically but this was the best file to keep it in [SS]
doctrine_pluralism_dark = {
	doctrine_group_type = doctrine_pluralism
	index = -1

	piety_cost = {
		value = faith_doctrine_cost_high
		if = {
			limit = { rite_has_doctrine = doctrine_pluralism_dark }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_pluralism_pluralistic }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
	}
	parameters = {
		pluralism_fundamentalist_proselytization_bonus
		pluralism_fundamentalist_faction_join_chance
		pluralism_fundamentalist_vulnerable_to_heresy
		pluralism_fundamentalist_revocation_tyranny_minimal

		human_sacrifice_active
		warmonger
		flower_war_cb_active

		conquest_cb_enabled
		invasion_cb_enabled
		great_holy_wars_active

		faith_can_raid
		clergy_can_fight
	}

	special_parameters = {
		hostility_towards_us_mult = 2
		warmonger = yes
	}
	character_modifier = {
		minority_different_faith_opinion = -25
		
		ai_war_chance = 0.5
		ai_war_cooldown = -0.25
		
		accolade_glory_gain_mult = 0.1
		title_creation_cost_mult = -0.5
		tyranny_gain_mult = -0.8
	}
}
