clear_crisis_gui = {
	scope = character
	is_shown = {
		always = yes
	}
	is_valid = {
		always = yes
	}
	effect = {
		clear_crisis_gui_effect = yes
	}
}
join_coalition_gui = {
	scope = character
	is_shown = {
		NOT = { is_in_coaltion_against_trigger = { CRISIS_HOLDER = var:crisis_character } }
	}
	is_valid = {
		is_character_interaction_valid = {
			recipient = var:crisis_character
			interaction = join_coalition_interaction
		}
	}
	effect = {
		open_interaction_window = {
			interaction = join_coalition_interaction
			actor = root
			recipient = var:crisis_character
		}
	}
}
leave_coalition_gui = {
	scope = character
	is_shown = {
		has_variable = crisis_character
		is_character_interaction_shown = {
			recipient = var:crisis_character
			interaction = leave_coalition_interaction
		}
	}
	is_valid = {
		is_character_interaction_valid = {
			recipient = var:crisis_character
			interaction = leave_coalition_interaction
		}
	}
	effect = {
		open_interaction_window = {
			interaction = leave_coalition_interaction
			actor = root
			recipient = var:crisis_character
		}
	}
}
