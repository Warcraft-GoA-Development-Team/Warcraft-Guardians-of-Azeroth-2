namespace = eternitysend

#### eternitysend.0001 -> See if Cenarius is alive, if he is, then try different ways to kill him or if he is already dead start the invasion of Kaldorei

eternitysend.0001 = {
    scope = none
    hidden = yes

    immediate = {
        title:h_burning_legion.holder = { save_scope_as = archimonde }
        if = {
            limit = { character:410002 = { is_alive = no } }
            scope:archimonde = {
                start_war = {
                    casus_belli = mongol_invasion_war
                    target = title:e_kaldorei.holder
                    target_title = title:k_hyjal
                }
            }
        }
        else = {
            limit = { character:410002 = { is_alive = yes } }
            trigger_event = eternitysend.0100
        }
    }
}

eternitysend.0100 = {
    scope = none
    hidden = yes

    immediate = {}
}
