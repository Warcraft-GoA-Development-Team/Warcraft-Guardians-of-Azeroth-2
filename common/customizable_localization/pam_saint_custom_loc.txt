
# Canonizing Saints - custom localization
# Swaps the Saint/canonize vocabulary per religion in one place: Christian faiths keep "Saint"/"canonize", religions with an attested term for the venerated dead use it (Wali, Divus, Siddha, Kami; gendered where the language genders it), and everything else falls back to "Venerated Ancestor"/"venerate" (matches the vanilla saint trait and 20_pam_flavorization.txt). Terms that denote a living holy person (Tzadik, Arhat, Sant, Zhenren, Sage, Ashavan) name the herald trait instead - see 00_traits.txt. These commands return single fragments only - the surrounding sentence stays a normal loc key so its saved scopes stay live, since a custom loc only carries the scope it is called on.
# Gendered branches (Wali/Waliyya, Divus/Diva) resolve from the called scope, so call the noun commands on the saint (visited_saint, saint_candidate) when one is saved; scopeless/hypothetical contexts call on GetPlayer and read the generic masculine form as generic.

# Verb: "canonize" (Christian) / "venerate" (other). Called on GetPlayer so it gates on the viewing player's religion, like the scopeless SelectLocalization keys.
PamCanonizeVenerateVerb = {
	type = character
	text = {
		trigger = {
			faith.religion = religion:christianity_religion
		}
		localization_key = pam_canonize_verb_christian
	}
	text = {
		trigger = {
			faith.religion = religion:hellenism_religion
		}
		localization_key = pam_canonize_verb_hellenism
	}
	text = {
		trigger = {
			faith.religion = religion:confucianism_religion
		}
		localization_key = pam_canonize_verb_confucianism
	}
	text = {
		trigger = {
			faith.religion = religion:taoism_religion
		}
		localization_key = pam_canonize_verb_taoism
	}
	text = {
		fallback = yes
		localization_key = pam_canonize_verb_other
	}
}
