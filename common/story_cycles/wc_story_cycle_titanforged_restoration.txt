wc_story_titanforged_restoration = {
	on_end = {
		wc_titanforged_restoration_migrate_pact_effect = yes
		story_owner = {
			if = {
				limit = {
					exists = var:wc_titanforged_restoration
					var:wc_titanforged_restoration = scope:story
				}
				remove_variable = wc_titanforged_restoration
				# Cancel only the active project's task, including interruption/death.
				cp:councillor_marshal ?= {
					if = {
						limit = {
							scope:story = {
								OR = {
									has_variable = wc_titanforged_restoration_county
									var:wc_titanforged_restoration_method = flag:implantation
									var:wc_titanforged_restoration_method = flag:pending
								}
							}
							is_performing_council_task = task_reorder_county
							OR = {
								AND = {
									scope:story = { has_variable = wc_titanforged_restoration_county }
									councillor_task_target.county = scope:story.var:wc_titanforged_restoration_county
								}
								AND = {
									scope:story = { NOT = { has_variable = wc_titanforged_restoration_county } }
									councillor_task_target.county.duchy = scope:story.var:wc_titanforged_restoration_duchy
								}
							}
						}
						start_default_task = yes
					}
				}
			}
		}
	}

	# Conservative first-lot behavior; eligible heir handover remains on the roadmap.
	on_owner_death = { end_story = yes }

	effect_group = {
		days = 30
		triggered_effect = {
			effect = {
				if = {
					limit = { has_variable = wc_titanforged_restoration_county }
					wc_titanforged_restoration_check_county_effect = yes
				}
				else_if = {
					limit = { var:wc_titanforged_restoration_method = flag:implantation }
					wc_titanforged_implantation_check_end_effect = yes
				}
				else_if = {
					limit = {
						OR = {
							story_owner = { faith = { is_titanforged_liberator_faith_trigger = no } }
							NOT = {
								var:wc_titanforged_restoration_duchy = {
									any_de_jure_county = { scope:story.story_owner = { completely_controls = prev } }
								}
							}
						}
					}
					story_owner = { trigger_event = wc_titanforged_restoration.0090 }
					end_story = yes
				}
				else_if = {
					limit = { exists = story_owner.cp:councillor_marshal }
					set_variable = { name = wc_titanforged_restoration_marshal value = story_owner.cp:councillor_marshal }
				}
			}
		}
	}
}
