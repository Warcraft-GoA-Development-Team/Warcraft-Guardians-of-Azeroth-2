# Balnazzar turns an ordinary murder scheme into preparation for possession.
wc_balnazzar_prepare_vessel_interaction = {
	category = interaction_category_hostile
	icon = creature_nathrezim
	interface_priority = 80
	common_interaction = yes
	use_diplomatic_range = yes
	ignores_pending_interaction_block = yes
	scheme = murder

	send_name = wc_balnazzar_prepare_vessel_interaction
	desc = wc_balnazzar_prepare_vessel_interaction_desc
	greeting = negative
	notification_text = wc_balnazzar_prepare_vessel_interaction_notification

	is_shown = {
		scope:actor = {
			is_ai = no
			has_character_flag = wc_balnazzar_disguise_active
			OR = {
				has_character_flag = has_formed_the_scarlet_crusade
				has_character_flag = wc_balnazzar_next_vessel_unlocked
			}
			NOT = { has_character_flag = balnazzar_shed_the_mask }
		}
		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = { story_owner = scope:actor }
		scope:recipient != scope:actor
		scope:recipient = {
			is_alive = yes
			is_ai = yes
			is_adult = yes
			is_landed = yes
			is_playable_character = yes
			has_trait = creature_human
			NOT = { has_character_flag = wc_balnazzar_disguise_active }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_incapable_trigger = no
			NOT = { is_isolated_from_trigger = { target = scope:recipient } }
			can_start_scheme = {
				type = murder
				target_character = scope:recipient
			}
		}
		scope:recipient = { NOT = { has_strong_hook = scope:actor } }
		trigger_if = {
			limit = { scope:actor = { has_variable = no_scheming_allowed_var } }
			scope:actor.var:no_scheming_allowed_var != scope:recipient
		}
	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = murder
					target_character = scope:recipient
				}
				random_scheme = {
					type = murder
					limit = {
						scheme_target_character = scope:recipient
					}
					set_variable = {
						name = wc_balnazzar_next_vessel_scheme
						value = yes
					}
				}
			}
		}
	}

	auto_accept = yes
}
