join_coalition_interaction = {
	interface_priority = 60
	category = interaction_category_diplomacy
	desc = action_join_coalition_combined_group_description

	is_shown = {
		scope:actor = {
			NOT = { is_in_coaltion_against_trigger = { CRISIS_HOLDER = scope:recipient } }
		}
		scope:recipient = {
			is_crisis_trigger = yes
		}
		NOT = { scope:recipient = scope:actor }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			has_crisis_coalition_cooldown_trigger = no
			can_be_in_coalition_trigger = yes 
			can_be_in_coalition_against_trigger = { CRISIS_HOLDER = scope:recipient } 
		}
	}
	
	on_accept = {
		scope:actor = {
			join_crisis_coalition_effect = { CRISIS_HOLDER = scope:recipient }
		}
	}
	auto_accept = yes

	ai_potential = {
		always = yes
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	ai_frequency = 12
	ai_will_do = {
		base = 0
		join_coalition_scripted_modifier = { CRISIS_HOLDER = scope:recipient }
	}
}
leave_coalition_interaction = {
	interface_priority = 60
	category = interaction_category_diplomacy
	desc = leave_coalition_interaction_desc

	is_shown = {
		scope:actor = {
			is_in_coaltion_against_trigger = { CRISIS_HOLDER = scope:recipient }
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			is_at_war = no
		}
	}
	
	on_accept = {
		scope:actor = {
			leave_crisis_coalition_effect = { CRISIS_HOLDER = scope:recipient }
		}
	}
	auto_accept = yes

	ai_potential = {
		always = yes
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			can_be_in_coalition_trigger = no
		}
	}
}
