# Named candidates for the Scarlet spiritual office; character scope.
wc_scarlet_head_candidate_trigger = {
	is_adult = yes
	wc_scarlet_support_trigger = yes
}

# Current eligibility, shared by the decision and its pending choices.
wc_lordaeron_liberation_valid_trigger = {
	is_alive = yes
	is_capable_adult = yes
	is_imprisoned = no
	has_trait = creature_human
	OR = {
		has_faith = faith:holy_light
		has_faith = faith:argent
		has_faith = faith:scarletism
	}
	highest_held_title_tier >= tier_duchy
	completely_controls = title:d_lordaeron
	# Recreate the crown, or redistribute our own; never annex a foreign kingdom.
	OR = {
		NOT = { exists = title:k_lordaeron.holder }
		title:k_lordaeron.holder = this
	}
	prestige_level >= 4
	piety_level >= 4
	NOT = {
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:liberate_lordaeron_from_the_scourge_decision
		}
	}
}

wc_lordaeron_beneficiary_valid_trigger = {
	dynasty = dynasty:30000
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	has_trait = creature_human
	has_being_trait_trigger = no
	OR = {
		has_faith = faith:holy_light
		has_faith = faith:argent
		has_faith = faith:scarletism
	}
}


wc_scarlet_support_trigger = {
	is_alive = yes
	is_imprisoned = no
	has_trait = creature_human
	has_being_trait_trigger = no
	OR = { has_faith = faith:holy_light has_faith = faith:scarletism }
	OR = {
		NOT = { exists = liege }
		top_liege = {
			OR = { has_faith = faith:holy_light has_faith = faith:scarletism }
		}
	}
}

wc_scarlet_controls_stratholme_trigger = {
	exists = title:c_stratholme.holder
	title:c_stratholme.holder = {
		OR = {
			this = prev
			is_vassal_or_below_of = prev
		}
	}
}

# The institution follows its title holder, including normal succession.
wc_scarlet_commander_valid_trigger = {
	is_alive = yes
	has_faith = faith:scarletism
	exists = global_var:wc_scarlet_crusade_title
	global_var:wc_scarlet_crusade_title = { holder = root }
	wc_scarlet_controls_stratholme_trigger = yes
}

wc_scarlet_stratholme_owner_valid_trigger = {
	wc_scarlet_commander_valid_trigger = yes
	global_var:wc_scarlet_crusade_title = {
		NOT = { exists = var:wc_scarlet_stratholme_resolved }
	}
}

wc_scarlet_stratholme_choice_valid_trigger = {
	wc_scarlet_stratholme_owner_valid_trigger = yes
	global_var:wc_scarlet_crusade_title = {
		exists = var:wc_scarlet_stratholme_open
		var:wc_scarlet_stratholme_open = root
	}
}

wc_scarlet_purification_valid_trigger = {
	wc_scarlet_commander_valid_trigger = yes
	global_var:wc_scarlet_crusade_title = {
		exists = var:wc_scarlet_stratholme_resolved
		NOT = { exists = var:wc_scarlet_purification_resolved }
	}

	title:k_stratholme = {
		any_in_de_jure_hierarchy = { tier = tier_county count > 0 }
		NOT = {
			any_in_de_jure_hierarchy = {
				tier = tier_county
				# The living Argent enclave and its subrealm need no Scarlet conversion.
				NOT = {
					exists = title:c_lights_hope.holder
					title:c_lights_hope.holder = {
						is_alive = yes
						has_faith = faith:argent
						NOT = { has_trait = being_undead }
						NOT = { this = root }
					}
					holder = {
						OR = {
							this = title:c_lights_hope.holder
							is_vassal_or_below_of = title:c_lights_hope.holder
						}
					}
				}
				OR = {
					NOT = { holder = { OR = { this = root is_vassal_or_below_of = root } } }
					NOT = { faith = faith:scarletism }
				}
			}
		}
	}
}
