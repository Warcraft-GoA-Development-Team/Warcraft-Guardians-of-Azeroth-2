namespace = liberate_lordaeron

liberate_lordaeron.1000 = { # Event for one who takes the decision
    type = character_event
	content_source = dlc_GOA
	title = lordaeron_1000_title
	desc = lordaeron_1000_desc
	theme = realm
    override_background = {
		reference = wc_lordaeron_throne_room
	}

	left_portrait = {
		character = root
		animation = emotion_thinking_scepter
	}

	lower_center_portrait = {
		character = scope:menethil_family_leader
	}

	right_portrait = {
		character = scope:my_head_of_faith
		animation = war_defender
	}

	immediate = {
		play_music_cue = "mx_cue_sacredrite"
		hidden_effect = {
			title:d_lordaeron = { save_scope_as = liberated_lordaeron }
			title:k_lordaeron = { save_scope_as = new_lordaeron_kingdom }

			if = {
                limit = {
                    exists = faith.religious_head
                }
                faith.religious_head = {
                    save_scope_as = my_head_of_faith
                }
            }

			if = {
				limit = {
					any_living_character = {
						has_dynasty = yes
						dynasty = 30000
						is_house_head = yes
					}
				}
				save_scope_as = menethil_family_leader
			}
		}
	}

	option = { # I Put myself on the Throne of Lordaeron
		name = lordaeron_1000_A
		add_character_modifier = saviour_of_lordaeron
		add_character_flag = liberator_of_lordaeron   # To prevent decision-maker from getting the notification event
		scope:liberated_lordaeron = {
			every_sub_realm_county = {
				set_county_faith = root.faith
				set_county_culture = culture.lordaeronian
			}
		}
		hidden_effect = {
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:new_lordaeron_kingdom = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}

			scope:new_lordaeron_kingdom.holder = { save_scope_as = new_ruler_of_lordaeron }
		}
		custom_tooltip = lordaeron_1000_A_tt

		ai_chance = {
			base = 0
			modifier = {
				add = 50
				has_trait = greedy
			}
			modifier = {
				add = 15
				has_trait = deligent
			}
			modifier = {
				add = 50
				has_trait = arrogant
			}
			modifier = {
				add = 25
				has_trait = deceitful
			}
			modifier = {
				add = 1000
				has_trait = ambitious
			}
			modifier = {
				add = 50
				has_trait = callous
			}
			modifier = {
				add = 15
				has_trait = stubborn
			}
		}
	}

	option = { # I Look for a living Menethil and put them on the Throne of Lordaeron
		trigger = { exists = scope:menethil_family_leader }
		name = lordaeron_1000_B
		add_character_modifier = servant_of_lordaeron
		add_character_flag = liberator_of_lordaeron   # To prevent decision-maker from getting the notification event
		scope:liberated_lordaeron = {
			every_sub_realm_county = {
				set_county_faith = root.faith
				set_county_culture = culture.lordaeronian
			}
		}
		hidden_effect = {
			create_title_and_vassal_change = {
				type = returned
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:new_lordaeron_kingdom = {
				change_title_holder = {
					holder = scope:menethil_family_leader
					change = scope:change
				}
			}

			scope:new_lordaeron_kingdom.holder = { save_scope_as = new_ruler_of_lordaeron }
		}
		custom_tooltip = lordaeron_1000_B_tt

		ai_chance = {
			base = 0
			modifier = {
				add = 50
				has_trait = temperate
			}
			modifier = {
				add = 25
				has_trait = generous
			}
			modifier = {
				add = 25
				has_trait = humble
			}
			modifier = {
				add = 50
				has_trait = content
			}
			modifier = {
				add = 25
				has_trait = just
			}
			modifier = {
				add = 100
				has_trait = trusting
			}
			modifier = {
				add = 100
				has_trait = compassionate
			}
		}
	}

	option = { # I Put the Spiritual Head of Faith on the Throne of Lordaeron
		trigger = {
			exists = faith.religious_head
			faith = { has_doctrine_parameter = spiritual_head_of_faith }
		}
		name = lordaeron_1000_C
		add_character_modifier = warrior_of_lordaeron
		add_character_flag = liberator_of_lordaeron   # To prevent decision-maker from getting the notification event
		scope:liberated_lordaeron = {
			every_sub_realm_county = {
				set_county_faith = root.faith
				set_county_culture = culture.lordaeronian
			}
		}
		hidden_effect = {
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:new_lordaeron_kingdom = {
				change_title_holder = {
					holder = scope:my_head_of_faith
					change = scope:change
				}
			}

			scope:new_lordaeron_kingdom.holder = { save_scope_as = new_ruler_of_lordaeron }
		}
		custom_tooltip = lordaeron_1000_C_tt

		ai_chance = {
			base = 0
			modifier = {
				add = 25
				has_trait = temperate
			}
			modifier = {
				add = 50
				has_trait = generous
			}
			modifier = {
				add = 25
				has_trait = content
			}
			modifier = {
				add = -100
				has_trait = cynical
			}
			modifier = {
				add = 100
				has_trait = zealous
			}
			modifier = {
				add = 50
				has_trait = compassionate
			}
		}
	}
}
