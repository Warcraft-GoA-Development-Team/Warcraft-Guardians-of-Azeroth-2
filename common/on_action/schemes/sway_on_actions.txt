sway_success = {
	random_events = {
		100 = sway_outcome.0001 #Standard success
		2 = sway_outcome.1001 #Diplomacy duel for more opinion
		2 = sway_outcome.1002 #Learning duel for more opinion
		2 = sway_outcome.1003 #They're warming up to me, can I take it further?
		#50 = sway_outcome.1004 #Only triggered from the Persian Struggle 
	}
	effect = {
		notify_puppeteer_of_scheme_outcome_effect = { OWNER = scope:owner TARGET = scope:target TYPE = event_toast_effect_good TITLE = scheme_outcome_sway_success_puppeteer }
		scope:owner = {
			if = {
				limit = {
					has_personal_tenet_flag = tenet_harmonious_society_sway_fulfillment
					OR = {
						liege ?= scope:target
						target_is_vassal_or_below = scope:target
					}
				}
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_generic_good
						title = harmonious_society_sway_fulfillment_toast
						left_icon = scope:target
						change_spiritual_fulfillment = minor_spiritual_fulfillment_value
					}
				}
				else = {
					change_spiritual_fulfillment = minor_spiritual_fulfillment_value
				}
			}
			# Warcraft
			ascension_checker_effect = { SOURCE = scheme }
		}
	}
}
