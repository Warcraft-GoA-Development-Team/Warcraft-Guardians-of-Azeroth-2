create_artifact_terenas_crown_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }

	set_artifact_rarity_famed = yes
	
	# Create the artifact
	create_artifact = {	
		name = terenas_crown_name
		description = terenas_crown_desc
		type = helmet
		visuals = terenas_crown
		template = terenas_crown_template
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created
			date = 364.08.08
			recipient = character:63146 #Lordain Menethil
			location = title:c_lordaeron.title_province # Lordaeron
		}
		modifier = terenas_crown_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}

	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = {
			name = terenas_crown
			value = yes
		}
		add_artifact_title_history = { # ties the crown to the title, for Coronations DLC
			target = title:k_lordaeron
			date = 364.08.08
		}
		save_scope_as = epic
	}

	if = { 
		limit = { NOT = { scope:owner.dynasty = dynasty:30000 } }
		dynasty:30000 = { 
			every_dynasty_member = { 
				limit = { this = house.house_head }
				house = { add_house_artifact_claim = scope:newly_created_artifact }
			}
		}
	}
}
