# Index:
#   wc_study_faith_pick_hidden_cult_effect      teacher scope, saves scope:actual_faith (1012)

# Teacher scope. OWNER = the studier. Same rows, same order, as wc_study_faith_has_hidden_cult_trigger.
wc_study_faith_pick_hidden_cult_effect = {
	if = {
		limit = {
			wc_study_faith_cult_damned_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:death_god OWNER = $OWNER$ }
		}
		faith:death_god = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_burning_blade_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:orcish_fel OWNER = $OWNER$ }
		}
		faith:orcish_fel = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_atalai_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:cult_of_hakkar OWNER = $OWNER$ }
		}
		faith:cult_of_hakkar = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_dark_iron_row_trigger = yes
			wc_study_faith_cult_usable_trigger = { CULT = faith:ragnarism OWNER = $OWNER$ }
		}
		faith:ragnarism = { save_scope_as = actual_faith }
	}
	else_if = {
		limit = {
			wc_study_faith_cult_usable_trigger = { CULT = faith:twilights_hammer_cult OWNER = $OWNER$ }
		}
		faith:twilights_hammer_cult = { save_scope_as = actual_faith }
	}
}

# Studier scope. Needs scope:actual_faith and scope:scheme. Becomes a secret cultist and learns one core tenet of the cult.
wc_study_faith_initiation_effect = {
	random_character_tenet = {
		knowledge_filter = unknown
		limit = {
			passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:actual_faith.main_rite }
		}
		save_scope_as = initiation_tenet
	}

	make_character_crypto_religionist_effect = { CRYPTO_RELIGION = scope:actual_faith }

	if = {
		limit = { exists = scope:initiation_tenet }
		add_known_tenet = scope:initiation_tenet
	}
	scope:scheme = { end_scheme = yes }
}

# Studier scope. Needs scope:scheme. Vanilla's study_faith_find_new_teacher_effect is local to
# study_faith_ongoing_events.txt, so wc_ files cannot call it. This copy stores progress before ending the scheme.
wc_study_faith_find_new_teacher_effect = {
	set_variable = {
		name = study_faith_progress
		value = scope:scheme.scheme_progress
	}
	set_variable = study_faith_dissatisfied_with_current_teacher

	scope:scheme = { end_scheme = yes }

	trigger_event = {
		id = study_faith_outcome.3000
		days = 3
	}
}

# Peasant scope (scope:chosen_peasant), called from study_faith_ongoing.1030's immediate inside hidden_effect_new_object.
# Needs scope:scheme and scope:dummy_gender. Always saves scope:newly_created_artifact.
wc_study_faith_create_peddler_artifact_effect = {
	set_variable = {
		name = religious_text
		value = scope:scheme.scheme_target_character.faith
	}
	scope:scheme.scheme_target_character.faith = { save_scope_as = peddler_faith }

	# Named relics are offered to players only, so AI schemers cannot use up the 10 relics
	if = {
		limit = {
			root = { is_ai = no }
			wc_study_faith_relic_offered_trigger = yes
		}
		wc_study_faith_create_matching_relic_effect = yes
	}
	else = {
		create_artifact_book_effect = {
			OWNER = scope:chosen_peasant
			CREATOR = scope:dummy_gender
			SET_SUBJECT = flag:religious_text
			SET_TOPIC = flag:religious_text
		}
		scope:newly_created_artifact = {
			set_variable = {
				name = faith
				value = scope:peddler_faith
			}
		}
	}
}

# Forbidden study (0100 / 0110 / 0120). Studier scope.

wc_study_faith_void_backlash_effect = {
	if = {
		limit = { has_trait = void_sickness }
		if = {
			limit = {
				NOR = {
					has_trait = lunatic_1
					has_trait = lunatic_genetic
				}
			}
			add_trait = lunatic_1
		}
	}
	else_if = {
		limit = { can_contract_disease_trigger = { DISEASE = void_sickness } }
		contract_disease_effect = {
			DISEASE = void_sickness
			TREATMENT_EVENT = no
		}
	}
	else = {
		add_stress = medium_stress_gain
	}
}

wc_study_faith_fel_backlash_effect = {
	remove_all_character_modifier_instances = wc_study_faith_fel_taint
	add_character_modifier = {
		modifier = wc_study_faith_fel_taint
		years = 5
	}
	add_stress = medium_stress_gain
}

# Your necromantic notes are found. If you could not openly study death magic, you now hold its secret.
wc_study_faith_death_backlash_effect = {
	if = {
		limit = {
			NOR = {
				can_openly_learn_magic_type_trigger = { MAGICTYPE = death }
				is_secret_nonelemental_magic_type_user = { MAGICTYPE = death }
			}
		}
		wc_add_magic_secret_effect = { MAGIC = death }
	}
	else = {
		add_prestige = major_prestige_loss
	}
}

# Studier scope. 15% chance of a curse from Karazhan's trapped books.
wc_study_faith_karazhan_curse_effect = {
	if = {
		limit = { scope:archive = title:c_karazhan }
		random = {
			chance = 15
			random_list = {
				50 = {
					trigger = { can_contract_disease_trigger = { DISEASE = curse_of_tongues } }
					contract_disease_effect = {
						DISEASE = curse_of_tongues
						TREATMENT_EVENT = no
					}
				}
				50 = {
					trigger = { can_contract_disease_trigger = { DISEASE = curse_of_exhaustion } }
					contract_disease_effect = {
						DISEASE = curse_of_exhaustion
						TREATMENT_EVENT = no
					}
				}
			}
		}
	}
}

# Teacher scope. TEACHER = scope:target. Saves scope:zealot. Same order as wc_study_faith_has_zealot_trigger.
wc_study_faith_pick_zealot_effect = {
	if = {
		limit = {
			cp:councillor_court_chaplain ?= { wc_study_faith_cleric_candidate_trigger = { TEACHER = $TEACHER$ } }
		}
		cp:councillor_court_chaplain = { save_scope_as = zealot }
	}
	else_if = {
		limit = {
			liege ?= {
				cp:councillor_court_chaplain ?= { wc_study_faith_cleric_candidate_trigger = { TEACHER = $TEACHER$ } }
			}
		}
		liege = {
			cp:councillor_court_chaplain = { save_scope_as = zealot }
		}
	}
	else = {
		random_courtier = {
			limit = { wc_study_faith_zealot_candidate_trigger = { TEACHER = $TEACHER$ } }
			save_scope_as = zealot
		}
	}
}

# Relics (study_faith_ongoing.1030)

# Peasant scope (the owner-to-be). RELIC = key, VISUALS = a vanilla visual. Saves scope:newly_created_artifact
# and reserves the relic with a global flag.
wc_study_faith_create_relic_effect = {
	set_artifact_rarity_famed = yes
	create_artifact = {
		name = wc_study_faith_relic_$RELIC$_name
		description = wc_study_faith_relic_$RELIC$_desc
		type = miscellaneous
		visuals = $VISUALS$
		wealth = scope:wealth
		quality = scope:quality
		template = wc_study_faith_relic_template
		history = {
			type = created
			recipient = scope:chosen_peasant
			location = root.location
		}
		modifier = wc_study_faith_relic_$RELIC$_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}
	scope:newly_created_artifact = {
		set_variable = {
			name = wc_study_faith_relic
			value = flag:$RELIC$
		}
		set_variable = {
			name = faith
			value = scope:peddler_faith
		}
	}
	set_global_variable = {
		name = wc_relic_$RELIC$
		value = yes
	}
}

# Any scope. 1030 "Not interested" destroys the item; free the relic so a later peddler can offer it again.
wc_study_faith_release_relic_effect = {
	if = {
		limit = {
			scope:newly_created_artifact ?= { has_variable = wc_study_faith_relic }
		}
		switch = {
			trigger = scope:newly_created_artifact.var:wc_study_faith_relic
			flag:libram_ancient_kings = { remove_global_variable = wc_relic_libram_ancient_kings }
			flag:arrival_of_the_naaru = { remove_global_variable = wc_relic_arrival_of_the_naaru }
			flag:headdress_first_shaman = { remove_global_variable = wc_relic_headdress_first_shaman }
			flag:puzzle_box = { remove_global_variable = wc_relic_puzzle_box }
			flag:doomsday_prophecy = { remove_global_variable = wc_relic_doomsday_prophecy }
			flag:praying_mantid = { remove_global_variable = wc_relic_praying_mantid }
			flag:edicts_thunder_king = { remove_global_variable = wc_relic_edicts_thunder_king }
			flag:talonpriest_mask = { remove_global_variable = wc_relic_talonpriest_mask }
			flag:atalai_scepter = { remove_global_variable = wc_relic_atalai_scepter }
			flag:druid_priest_statues = { remove_global_variable = wc_relic_druid_priest_statues }
		}
	}
}
