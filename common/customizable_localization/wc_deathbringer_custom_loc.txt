GetDeathGodHighGodName = {
	type = character

	# Death
	text = {
		localization_key = deathbringer_high_god_name
		trigger = {
			has_title = title:h_scourge
		}
	}

	# Lich Queen
	text = {
		localization_key = death_god_lich_queen_name
		trigger = {
			title:h_scourge.holder ?= { is_female = yes }
		}
	}

	# Lich King
	text = {
		localization_key = death_god_lich_king_name
		trigger = {
			exists = title:h_scourge.holder
		}
	}

	# Death
	text = {
		localization_key = deathbringer_high_god_name
		fallback = yes
	}
}
