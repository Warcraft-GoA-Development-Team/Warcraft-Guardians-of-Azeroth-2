scripted_effect gdp_1200_create_fallback_negotiator_effect = {
	if = {
		limit = {
			scope:patron ?= {
				faith_dominant_gender_female_or_equal = yes
				faith_dominant_gender_male_or_equal = no
			}
		}
		create_character = {
			template = pool_repopulate_stewardship
			gender_female_chance = 100
			culture = scope:patron.culture
			faith = scope:patron.faith
			dynasty = none
			location = scope:patron.location
			save_scope_as = negotiator
			after_creation = { add_character_flag = gdp_1200_temp_negotiator trigger_race_giving_no_gene_effect = yes } # Warcraft
		}
	}
	else_if = {
		limit = {
			scope:patron ?= {
				faith_dominant_gender_male_or_equal = yes
				faith_dominant_gender_female_or_equal = no
			}
		}
		create_character = {
			template = pool_repopulate_stewardship
			gender_female_chance = 0
			culture = scope:patron.culture
			faith = scope:patron.faith
			dynasty = none
			location = scope:patron.location
			save_scope_as = negotiator
			after_creation = { add_character_flag = gdp_1200_temp_negotiator trigger_race_giving_no_gene_effect = yes } # Warcraft
		}
	}
	else = {
		create_character = {
			template = pool_repopulate_stewardship
			gender_female_chance = 50
			culture = scope:patron.culture
			faith = scope:patron.faith
			dynasty = none
			location = scope:patron.location
			save_scope_as = negotiator
			after_creation = { add_character_flag = gdp_1200_temp_negotiator trigger_race_giving_no_gene_effect = yes } # Warcraft
		}
	}
}

#Actual patron duel effect
scripted_effect gdp_1200_duel_effect_a = {
	duel = {
		skill = $SKILL$
		target = scope:negotiator
		#Success: Get what you wanted
		50 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = 3.5
				min = -49
			}
			desc = gdp_visit_holding_decision_event.1200.a_$SKILL$.success
			send_interface_toast = {
				title = gdp_visit_holding_decision_event.1200.a_$SKILL$.success
				left_icon = root
				right_icon = scope:negotiator
				add_character_modifier = {
					modifier = gdp_patron_competitiveness_modifier
					years = 5
				}
				#Cooldown
				set_variable = {
					name = 1200_patron_competitiveness
					value = flag:success
					days = visit_holding_cooldown_days
				}
				visit_holding_refresh_patron_area_tracker_effect = yes
				trigger_event = gdp_visit_holding_decision_event.1200
			}
		}
		#Failure: you get nothing/get a worse deal, patron/steward gets mad
		45 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = -3.5
				min = -20
			}	
			desc = gdp_visit_holding_decision_event.1200.a_$SKILL$.failure
			send_interface_toast = {
				title = gdp_visit_holding_decision_event.1200.a_$SKILL$.failure
				left_icon = root
				right_icon = scope:negotiator
				visit_holding_set_option_cooldown_patron_effect = { VAR = 1200_patron_competitiveness }
				# they get mad
				progress_towards_rival_effect = {
					REASON = rival_merchant_asked_too_much
					CHARACTER = scope:negotiator
					OPINION = default_rival_opinion
				}
				#Go back to the main event
				trigger_event = gdp_visit_holding_decision_event.1200
			}
		}
		#Critical Failure: The local guards are summoned
		5 = {
			trigger = {
				root = {
					OR = {
						highest_skill = martial 
						highest_skill_including_prowess = prowess
					}
				}			
			}
			compare_modifier = {
				value = scope:duel_value
				multiplier = -3.5
			}	
			min = 5
			desc = gdp_visit_holding_decision_event.1200.a_$SKILL$.critical_failure
			send_interface_toast = {
				title = gdp_visit_holding_decision_event.1200.a_$SKILL$.critical_failure
				left_icon = root
				right_icon = scope:negotiator
				#Set up the cooldown for the convince-ction
				visit_holding_set_option_cooldown_patron_effect = { VAR = 1200_patron_competitiveness }
				#Quite the tussle
				increase_wounds_effect = { REASON = fight }
				#This patron doesn't want to see you again
				progress_towards_rival_effect = {
					REASON = rival_merchant_tried_to_intimidate
					CHARACTER = scope:negotiator
					OPINION = default_rival_opinion
				}
				#And you will not be going back to this patron for a while, but feel free to look at the greyed out option
				visit_holding_set_option_cooldown_patron_effect = { VAR = 1200_guards_summoned }
				#Go back to the main event
				trigger_event = gdp_visit_holding_decision_event.1000
			}
		}
	}
}

#Visit Patron
gdp_visit_holding_decision_event.1200 = { 
	type = character_event 
	title = gdp_visit_holding_decision_event.1200.t 
	window = visit_settlement_window
	desc = {
		desc = gdp_visit_holding_decision_event.1200.intro_desc
		first_valid = {
			triggered_desc = {
				trigger = { 
					AND = {
						scope:patron = scope:negotiator
						scope:steward ?= { is_available = yes } 
					}
				}
				desc = gdp_visit_holding_decision_event.1200.all_available_desc
			}
			triggered_desc = {
				trigger = { 
					AND = {
						scope:patron = scope:negotiator
						OR = {
							NOT = { exists = scope:steward }
							scope:steward = { is_available = no } 
						}
					}
				}
				desc = gdp_visit_holding_decision_event.1200.patron_available_desc
			}
			triggered_desc = {
				trigger = { scope:steward ?= scope:negotiator }
				desc = gdp_visit_holding_decision_event.1200.steward_available_desc
			}
			triggered_desc = {
				trigger = {
					exists = scope:negotiator
					NOR = {
						scope:negotiator = scope:patron
						scope:steward ?= { this = scope:negotiator }
					}
				}
				desc = gdp_visit_holding_decision_event.1200.representative_desc
			}
			desc = gdp_visit_holding_decision_event.1200.none_available_desc
		}
	}

	theme = stewardship 
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_generic" }
	override_background = throne_room

	left_portrait = {
		character = root 
		animation = holding_scrolls
		camera = camera_event_scheme_center
	}

	center_portrait = {
		character = scope:steward
		animation = chancellor
		camera = camera_event_scheme_far_right
		trigger = {
			scope:steward ?= {
				this != scope:negotiator
				is_available = yes 
			}
		}
	}

	right_portrait = {
		character = scope:negotiator
		animation = pondering
	}

	immediate = {
		suzerain = { save_scope_as = patron }
		scope:patron.cp:councillor_steward ?= { save_scope_as = steward }
		if = {
			limit = { NOT = { exists = scope:negotiator } }
			if = {
				limit = {
					scope:patron = { is_available = yes }
				}
				scope:patron = { save_scope_as = negotiator }
			}
			else_if = {
				limit = {
					scope:steward ?= { is_available = yes }
				}
				scope:steward = { save_scope_as = negotiator }
			}
			else_if = {
				limit = {
					scope:patron.cp:councillor_chancellor ?= { is_available = yes }
				}
				scope:patron.cp:councillor_chancellor ?= { save_scope_as = negotiator }
			}
			else_if = {
				limit = {
					scope:patron = {
						any_councillor = { is_available = yes }
					}
				}
				scope:patron = {
					random_councillor = {
						limit = { is_available = yes }
						save_scope_as = negotiator
					}
				}
			}
			else_if = {
				limit = {
					scope:patron = {
						any_courtier = {
							is_available = yes
							is_adult = yes
						}
					}
				}
				scope:patron = {
					random_courtier = {
						limit = {
							is_available = yes
							is_adult = yes
						}
						save_scope_as = negotiator
					}
				}
			}
			else_if = {
				limit = {
					scope:patron = {
						any_vassal = {
							is_available = yes
							is_adult = yes
						}
					}
				}
				scope:patron = {
					random_vassal = {
						limit = {
							is_available = yes
							is_adult = yes
						}
						save_scope_as = negotiator
					}
				}
			}
			else = { gdp_1200_create_fallback_negotiator_effect = yes }
		}
	}

	option = { # Ask for Trade Privileges
		trigger = {
			custom_tooltip = {
				text = gdp_visit_holding_decision_event.1200.a.negotiator.tt
				exists = scope:negotiator
			}
			custom_tooltip = {
				text = gdp_visit_holding_decision_event.1200.a.competitiveness_cooldown.tt
				NOT = { has_variable = 1200_patron_competitiveness }
			}
		}
		show_as_unavailable = { always = yes }
		name = gdp_visit_holding_decision_event.1200.a
		custom_tooltip = gdp_visit_holding_decision_event.1200.a.tt 
		trigger_event = gdp_visit_holding_decision_event.1201
	}
	option = { # Look for Investment Opportunities
		trigger = {
			custom_tooltip = {
				text = gdp_visit_holding_decision_event.1200.a.negotiator.tt
				exists = scope:negotiator
			}
			custom_tooltip = {
				text = gdp_visit_holding_decision_event.1200.b.prestige_cooldown.tt
				NOT = { has_variable = 1200_patron_prestige }
			}
			short_term_gold >= medium_gold_value
		}
		reason = add_prestige
		show_as_unavailable = { always = yes }
		name = gdp_visit_holding_decision_event.1200.b
		custom_tooltip = gdp_visit_holding_decision_event.1200.b.tt
		trigger_event = gdp_visit_holding_decision_event.1202
	}

	option = { # Back Out
		name = gdp_visit_holding_decision_event.1200.c
		custom_tooltip = gdp_visit_holding_decision_event.1200.c.tt
		if = {
			limit = { scope:negotiator ?= { has_character_flag = gdp_1200_temp_negotiator } }
			scope:negotiator = { silent_disappearance_effect = yes }
		}
		trigger_event = gdp_visit_holding_decision_event.1000	 	
	}
}

			NOT = { exists = scope:$SCENE$_loc_tracker }	#e.g. scope:weaponsmith_loc_tracker
		}
		save_scope_value_as = {
			name = $SCENE$_loc_tracker
			value = 0
		}
	}
	else_if = {
		limit = { scope:$SCENE$_loc_tracker ?= 1 }
		save_scope_value_as = {
			name = $SCENE$_loc_tracker
			value = 0
		}
	}
	else = {
		save_scope_value_as = {
			name = $SCENE$_loc_tracker
			value = 1
		}			
	}
}

		#Or not

			modifier = {
				has_trait = rakish
				add = 500
			}
			modifier = {
				has_trait = chaste
				factor = 0
			}
		}
	}

	option = {
		name = gdp_visit_holding_decision_event.1505.d

		if = {

				has_variable = 1505_contemplated_garden_recently

					limit = { 1505_could_unwind_in_garden_trigger = yes }
					has_variable = 1505_garden_unwinded_recently
				}
			}

		}
		else = {
			#Sets variables and returns us to church grounds screen
			1501_been_to_effect = { GUILD = garden }
			#For desc flavor
			if = {
				limit = {
					NOT = { exists = scope:been_to_garden }
				}
				save_scope_as = been_to_garden
			}
			1501_return_to_holy_site_effect = yes
		}

	}
	after = {

		1500_update_last_location_effect = { LOCATION_FLAG = flag:garden }
	}
