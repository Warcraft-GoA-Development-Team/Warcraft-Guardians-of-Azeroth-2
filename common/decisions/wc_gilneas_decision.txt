gilneas_build_wall_decision = {
    ai_check_interval = 12
    decision_group_type = major
    picture = { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" }

    cooldown = {years = 1}

    is_shown = {
		is_landed = yes
        completely_controls = title:d_ambermill
        completely_controls = title:d_crowvale
        has_title  = title:k_gilneas
        exists = cp:councillor_chancellor

        is_liege_or_above_of = title:c_southpoint.holder
        is_liege_or_above_of = title:c_western_strand.holder

        any_realm_province = {
            NOR = {
                has_building = greymane_wall_01
                has_building = greymane_wall_02
            }
        }
        
    }

    is_valid = {
        custom_tooltip = {
			text = horde_title_not_exist
			NOT = {exists = title:e_horde.holder}
		}
		custom_tooltip = {
			text = dark_portal_closed
			title:b_dark_portal.title_province = {has_province_modifier = closed_from_another_side}
		}
    }

    is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

    effect = {
        custom_tooltip = gilneas_build_wall_decision_tt
        trigger_event = {
           id = wc_greymane_wall_events.0001
        }
    }
    ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
