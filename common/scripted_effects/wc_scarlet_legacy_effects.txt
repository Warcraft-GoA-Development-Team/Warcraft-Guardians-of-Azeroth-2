# Save compatibility only. Ordinary Scarlet mechanics never read a personal story.
wc_scarlet_migrate_crusade_effect = {
	if = {
		limit = { NOT = { exists = global_var:wc_scarlet_crusade_title } }
		if = {
			limit = {
				OR = {
					character:60021 = { has_character_flag = has_formed_the_scarlet_crusade }
					AND = {
						exists = global_var:balnazzar_dreadlord_story
						global_var:balnazzar_dreadlord_story = { exists = var:wc_balnazzar_scarlet_founded }
					}
				}
			}
			set_global_variable = { name = wc_scarlet_crusade_title value = title:k_order_of_the_silver_hand }
			title:k_order_of_the_silver_hand = { set_variable = wc_scarlet_prefer_isillien }
		}
	}
	if = {
		limit = {
			exists = global_var:wc_scarlet_crusade_title
			# Personal history belongs only to the Order, never a generic founder's title.
			global_var:wc_scarlet_crusade_title = title:k_order_of_the_silver_hand
			global_var:wc_scarlet_crusade_title = { NOT = { exists = var:wc_scarlet_legacy_migrated } }
		}
		global_var:wc_scarlet_crusade_title = { set_variable = wc_scarlet_legacy_migrated }
		if = {
			limit = { exists = global_var:balnazzar_dreadlord_story }
			global_var:balnazzar_dreadlord_story = {
				# The council and Renault still use this private, historical marker.
				if = {
					limit = {
						NOT = { exists = var:wc_balnazzar_scarlet_founded }
						OR = {
							story_owner = { has_character_flag = has_formed_the_scarlet_crusade }
							character:60021 = { has_character_flag = has_formed_the_scarlet_crusade }
						}
					}
					set_variable = wc_balnazzar_scarlet_founded
				}
				if = {
					limit = {
						OR = {
							exists = var:wc_balnazzar_stratholme_resolved
							story_owner = { has_character_flag = wc_balnazzar_stratholme_taken }
							AND = {
								NOT = { exists = var:wc_balnazzar_stratholme_migrated }
								story_owner = { has_character_flag = wc_balnazzar_disguise_active wc_scarlet_controls_stratholme_trigger = yes }
							}
						}
					}
					global_var:wc_scarlet_crusade_title = { set_variable = wc_scarlet_stratholme_resolved }
				}
				if = {
					limit = { OR = { exists = var:wc_balnazzar_purification_resolved story_owner = { has_character_flag = wc_balnazzar_stratholme_purified } } }
					global_var:wc_scarlet_crusade_title = { set_variable = wc_scarlet_purification_resolved }
				}
				if = {
					limit = { exists = var:wc_balnazzar_scarlet_head_migrated }
					global_var:wc_scarlet_crusade_title = { set_variable = wc_scarlet_head_initialized }
				}
				if = {
					limit = { exists = var:wc_balnazzar_stratholme_open }
					global_var:wc_scarlet_crusade_title = {
						set_variable = { name = wc_scarlet_stratholme_open value = global_var:balnazzar_dreadlord_story.var:wc_balnazzar_stratholme_open }
					}
				}
				if = {
					limit = { exists = var:wc_balnazzar_purification_open }
					global_var:wc_scarlet_crusade_title = {
						set_variable = { name = wc_scarlet_purification_open value = global_var:balnazzar_dreadlord_story.var:wc_balnazzar_purification_open }
					}
				}
			}
		}
		if = {
			limit = { character:60021 = { has_character_flag = wc_balnazzar_stratholme_taken } }
			global_var:wc_scarlet_crusade_title = { set_variable = wc_scarlet_stratholme_resolved }
		}
		if = {
			limit = { character:60021 = { has_character_flag = wc_balnazzar_stratholme_purified } }
			global_var:wc_scarlet_crusade_title = { set_variable = wc_scarlet_purification_resolved }
		}
	}
}

# Both old open windows and newly delivered windows use the same retained IDs.
wc_scarlet_clear_legacy_windows_effect = {
	if = {
		limit = { exists = global_var:balnazzar_dreadlord_story }
		global_var:balnazzar_dreadlord_story = {
			if = {
				limit = { exists = var:wc_balnazzar_stratholme_open var:wc_balnazzar_stratholme_open = root }
				remove_variable = wc_balnazzar_stratholme_open
			}
			if = {
				limit = { exists = var:wc_balnazzar_purification_open var:wc_balnazzar_purification_open = root }
				remove_variable = wc_balnazzar_purification_open
			}
		}
	}
}
