namespace = liberate_lordaeron

liberate_lordaeron.1000 = { # Event for one who takes the decision
    type = character_event
	trigger = { wc_lordaeron_liberation_valid_trigger = yes }
	content_source = dlc_GOA
	title = liberate_lordaeron.1000.title
	desc = liberate_lordaeron.1000.desc
	theme = realm
    override_background = {
		reference = wc_lordaeron_throne_room
	}

	left_portrait = {
		character = root
		animation = emotion_thinking_scepter
	}

	lower_center_portrait = {
		trigger = { exists = scope:menethil_family_leader }
		character = scope:menethil_family_leader
	}

	right_portrait = {
		trigger = { exists = scope:my_head_of_faith }
		character = scope:my_head_of_faith
		animation = war_defender
	}

	immediate = {
		play_music_cue = "mx_cue_sacredrite"
		hidden_effect = {
			title:d_lordaeron = { save_scope_as = liberated_lordaeron }
			title:k_lordaeron = { save_scope_as = new_lordaeron_kingdom }

			if = {
                limit = {
                    exists = faith.religious_head
                }
                faith.religious_head = {
                    save_scope_as = my_head_of_faith
                }
            }

			if = { # Calia herself, then her oldest eligible child (not root's children).
				limit = {
					exists = character:60004
					character:60004 = { wc_lordaeron_beneficiary_valid_trigger = yes }
				}
				character:60004 = { save_scope_as = menethil_family_leader }
			}
			else_if = {
				limit = { exists = character:60004 }
				character:60004 = {
					ordered_child = {
						limit = {
							wc_lordaeron_beneficiary_valid_trigger = yes
							dynasty = dynasty:30000 # Menethil restoration, as promised by option B.
						}
						order_by = age
						max = 1
						save_scope_as = menethil_family_leader
					}
				}
			}
		}
	}

	option = { # I Put myself on the Throne of Lordaeron
		trigger = { wc_lordaeron_liberation_valid_trigger = yes }
		name = liberate_lordaeron.1000.A
		if = {
			limit = { wc_lordaeron_liberation_valid_trigger = yes }
			give_land_to_settler = { TO_WHO = root COUNTY_NAME = scope:new_lordaeron_kingdom GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
			if = {
				limit = { title:k_lordaeron.holder = root }
				add_character_modifier = saviour_of_lordaeron
				root = { save_scope_as = new_ruler_of_lordaeron }
			}
		}

		ai_chance = {
			base = 0
			modifier = {
				add = 50
				has_trait = greedy
			}
			modifier = {
				add = 15
				has_trait = diligent
			}
			modifier = {
				add = 50
				has_trait = arrogant
			}
			modifier = {
				add = 25
				has_trait = deceitful
			}
			modifier = {
				add = 1000
				has_trait = ambitious
			}
			modifier = {
				add = 50
				has_trait = callous
			}
			modifier = {
				add = 15
				has_trait = stubborn
			}
		}
	}

	option = { # I Look for a living Menethil and put them on the Throne of Lordaeron
		trigger = {
			exists = scope:menethil_family_leader
			scope:menethil_family_leader = { wc_lordaeron_beneficiary_valid_trigger = yes }
			wc_lordaeron_liberation_valid_trigger = yes
		}
		name = liberate_lordaeron.1000.B
		if = {
			limit = { wc_lordaeron_liberation_valid_trigger = yes
				exists = scope:menethil_family_leader
				scope:menethil_family_leader = { wc_lordaeron_beneficiary_valid_trigger = yes } }
			give_land_to_settler = { TO_WHO = scope:menethil_family_leader COUNTY_NAME = title:c_lordaeron GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
			give_land_to_settler = { TO_WHO = scope:menethil_family_leader COUNTY_NAME = scope:liberated_lordaeron GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
			give_land_to_settler = { TO_WHO = scope:menethil_family_leader COUNTY_NAME = scope:new_lordaeron_kingdom GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
			if = {
				limit = { title:k_lordaeron.holder = scope:menethil_family_leader }
				add_character_modifier = servant_of_lordaeron
				scope:menethil_family_leader = { save_scope_as = new_ruler_of_lordaeron }
			}
		}

		ai_chance = {
			base = 0
			modifier = {
				add = 50
				has_trait = temperate
			}
			modifier = {
				add = 25
				has_trait = generous
			}
			modifier = {
				add = 25
				has_trait = humble
			}
			modifier = {
				add = 50
				has_trait = content
			}
			modifier = {
				add = 25
				has_trait = just
			}
			modifier = {
				add = 100
				has_trait = trusting
			}
			modifier = {
				add = 100
				has_trait = compassionate
			}
		}
	}

	option = { # I Put the Spiritual Head of Faith on the Throne of Lordaeron
		trigger = {
			wc_lordaeron_liberation_valid_trigger = yes
			exists = scope:my_head_of_faith
			exists = faith.religious_head
			scope:my_head_of_faith = faith.religious_head
			scope:my_head_of_faith = { is_alive = yes is_adult = yes is_imprisoned = no }
			faith = { has_doctrine_parameter = spiritual_head_of_faith }
		}
		name = liberate_lordaeron.1000.C
		if = {
			limit = { wc_lordaeron_liberation_valid_trigger = yes
				exists = scope:my_head_of_faith
				exists = faith.religious_head
				scope:my_head_of_faith = faith.religious_head
				faith = { has_doctrine_parameter = spiritual_head_of_faith }
				scope:my_head_of_faith = { is_alive = yes is_adult = yes is_imprisoned = no } }
			give_land_to_settler = { TO_WHO = scope:my_head_of_faith COUNTY_NAME = title:c_lordaeron GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
			give_land_to_settler = { TO_WHO = scope:my_head_of_faith COUNTY_NAME = scope:liberated_lordaeron GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
			give_land_to_settler = { TO_WHO = scope:my_head_of_faith COUNTY_NAME = scope:new_lordaeron_kingdom GIVE_INDEPENDENCY = yes ADD_CLAIM_ON_LOSS = no }
			if = {
				limit = { title:k_lordaeron.holder = scope:my_head_of_faith }
				add_character_modifier = warrior_of_lordaeron
				scope:my_head_of_faith = { save_scope_as = new_ruler_of_lordaeron }
			}
		}

		ai_chance = {
			base = 0
			modifier = {
				add = 25
				has_trait = temperate
			}
			modifier = {
				add = 50
				has_trait = generous
			}
			modifier = {
				add = 25
				has_trait = content
			}
			modifier = {
				add = -100
				has_trait = cynical
			}
			modifier = {
				add = 100
				has_trait = zealous
			}
			modifier = {
				add = 50
				has_trait = compassionate
			}
		}
	}
	option = {
		name = liberate_lordaeron.1001.A
		trigger = { wc_lordaeron_liberation_valid_trigger = no }
	}
	after = {
		if = {
			limit = {
				exists = scope:new_ruler_of_lordaeron
				title:k_lordaeron.holder = scope:new_ruler_of_lordaeron
				scope:new_ruler_of_lordaeron = { is_alive = yes completely_controls = title:d_lordaeron }
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:liberate_lordaeron_from_the_scourge_decision
					}
				}
			}
			title:d_lordaeron = {
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_county }
					set_county_faith = root.faith
					set_county_culture = culture:lordaeronian
				}
			}
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:liberate_lordaeron_from_the_scourge_decision
			}
			every_ruler = {
				limit = {
					is_alive = yes
					has_trait = creature_human
					NOT = { this = scope:new_ruler_of_lordaeron }
					exists = capital_province
					capital_province = { geographical_region = world_eastern_kingdoms_lordaeron }
				}
				trigger_event = { id = liberate_lordaeron.1001 }
			}
		}
	}
}

liberate_lordaeron.1001 = { # Notification event for all Humans in Lordaeron Eastern Kingdoms
    type = character_event
	content_source = dlc_GOA
	title = liberate_lordaeron.1001.title
	desc = liberate_lordaeron.1001.desc
	theme = realm
    override_background = {
		reference = wc_lordaeron_throne_room
	}

	trigger = {
		exists = scope:new_ruler_of_lordaeron
		scope:new_ruler_of_lordaeron = { is_alive = yes }
		title:k_lordaeron.holder = scope:new_ruler_of_lordaeron
		NOT = {
			this = scope:new_ruler_of_lordaeron
		}
	}

	immediate = {
		play_music_cue = "mx_cue_sacredrite"
	}

	left_portrait = {
		character = scope:new_ruler_of_lordaeron
		animation = toast
	}

	right_portrait = {
		character = root
		animation = thinking
	}

	option = {
		name = liberate_lordaeron.1001.A
	}
}
