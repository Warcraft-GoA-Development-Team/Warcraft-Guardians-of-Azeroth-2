is_criminal_in_faith_trigger = { #determine if any of their traits are criminal
	any_character_trait = {
		trait_is_criminal_in_faith_trigger = {
			TRAIT = this
			FAITH = $FAITH$
			GENDER_CHARACTER = root
		}
	}
}

		faith.religion = { is_in_family = rf_disorder }
		faith.religion = religion:deathbringer
		is_of_forsaken_like_culture = yes
	}
}
