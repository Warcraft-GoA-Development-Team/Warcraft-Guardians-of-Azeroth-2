# Starting Holy Site Relics #
#
# Helpers used by historical_artifacts.0050 to seed enshrined relics at
# holy sites. The event runs two passes:
#   Pass 1: for every faith, at each of its eminent/regular holy sites, if the
#           county controller shares that faith, create one relic.
#   Pass 2: for every faith with at least 3 county followers, at each of
#           its eminent holy sites that still has no artifact, if the
#           controller's faith rates the diaspora faith astray-or-better,
#           create one relic anyway.
#
# Expected scopes on entry to any create_starting_relic_*_effect:
#   scope:relic_faith = the faith the relic is for (may not be the
#                       controller's faith in pass 2)
#   scope:relic_site  = the holy site scope
#   scope:relic_owner = the temporary artifact-creation-scope owner
#                       (the finalizer's set_holy_site_owner transfers
#                       ownership to the holy site itself)
#
# Dispatchers route on (relic_site, relic_faith.religion) to either a
# curated named-relic block or a per-religion generic block.

# Regular-site dispatcher #

create_starting_holy_relic_regular_effect = {
	# Curated relics at regular sites
	if = {
		limit = {
			scope:relic_faith.religion = religion:christianity_religion
			scope:relic_site = holy_site:kent
		}
		create_starting_relic_kent_christian_effect = yes
	}
	else_if = {
		# Antioch took Aksum's eminent slot on Miaphysitism
		limit = {
			scope:relic_faith.religion = religion:christianity_religion
			scope:relic_site = holy_site:aksum
		}
		create_starting_relic_aksum_christian_effect = yes
	}
	else_if = {
		limit = {
			scope:relic_faith.religion = religion:christianity_religion
			scope:relic_site = holy_site:cologne
		}
		create_starting_relic_cologne_christian_effect = yes
	}
	else_if = {
		limit = {
			scope:relic_faith.religion = religion:islam_religion
			scope:relic_site = holy_site:baghdad
		}
		create_starting_relic_baghdad_islamic_effect = yes
	}
	else = {
		# Generic at common rarity for everything else.
		create_starting_generic_relic_effect = { RARITY = common }
	}
}

create_starting_relic_antioch_christian_effect = {
	set_artifact_rarity_illustrious = yes
	scope:relic_owner = {
		# The Holy Lance of Antioch was discovered during the First
		# Crusade siege of Antioch in 1098. Before that, the great local
		# relic is the bones of St. Ignatius of Antioch (patriarch and
		# martyr, d. ~110 AD).
		if = {
			limit = { current_date >= 1098.6.14 }
			create_artifact = {
				name = artifact_relic_lance_antioch_name

				modifier = artifact_placeholder_modifier
				save_scope_as = newly_created_artifact
			}
		}
		else = {
			create_artifact = {
				name = artifact_relic_ignatius_name
				description = artifact_relic_ignatius_desc

# Per-religion helpers vary names so sites do not clone the same Branch of the Tree of Life
create_starting_generic_relic_effect = {
	if = {
		limit = { scope:relic_faith.religion = religion:christianity_religion }
		create_starting_generic_christian_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:islam_religion }
		create_starting_generic_islamic_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:buddhism_religion }
		create_starting_generic_buddhist_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:hinduism_religion }
		create_starting_generic_hindu_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:zoroastrianism_religion }
		create_starting_generic_zoroastr_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:germanic_religion }
		create_starting_generic_germanic_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		limit = { scope:relic_faith.religion = religion:slavic_religion }
		create_starting_generic_slavic_relic_effect = { RARITY = $RARITY$ }
	}
	else_if = {
		# Only Roog passes branch_boog_template is_holy_relic
		limit = { scope:relic_faith.religion = religion:west_african_roog_religion }
		create_starting_generic_west_african_relic_effect = { RARITY = $RARITY$ }
	}
	else = {
		# No dedicated template: misc helper sets var:relic_religion on the artifact
		create_starting_generic_misc_relic_effect = { RARITY = $RARITY$ }
	}
}

# Shared finalizer: modifiers + enshrinement
finalize_starting_relic_effect = {
	# A holy site only has the three holy_relic_* slots, and a relic past that would sit unenshrined and assert on equip
	if = {
		limit = { scope:relic_site = { holy_site_artifact_count >= 3 } }
		destroy_artifact = scope:newly_created_artifact
	}
	else = {
		scope:newly_created_artifact = {
			scope:relic_owner = { save_scope_as = owner } # Required for the 'add_scaled_artifact_' scripted effects
			add_scaled_artifact_modifier_piety_effect = yes
			add_scaled_artifact_modifier_grandeur_small_effect = yes
			if = {
				limit = {
					OR = {
						rarity = famed
						rarity = illustrious
					}
				}
				add_2_scaled_artifact_modifier_devotion_effect = yes
			}
			else = {
				add_scaled_artifact_modifier_devotion_effect = yes
			}
			add_artifact_modifier = artifact_monthly_prestige_penalty_modifier # overwrite default creation modifier
		}

		scope:newly_created_artifact = {
			# Enshrine without history (setting owner & equipping = enshrining), skipping 'can_enshrine_artifact' rules)
			set_holy_site_owner = {
				target = scope:relic_site
				generate_history = no
			}
			equip_artifact_to_owner = yes
		}
	}
}
