# Agent types for wc_mograine_assassination - fixed named slots for specific
# loyalists in Dathrohan's household, not open courtier-pool recruitment.
# Renault stays a scripted event participant instead of an agent slot since
# his own employer switch lands mid-story, too unreliable to gate a slot on.

wc_mograine_whitemane_agent = {
	valid_agent_for_slot = {
		root = character:60572 # Sally Whitemane
	}

	contribution_type = success_chance_max

	contribution = {
		value = scheme_agent_general_bonuses_contribution_score_value

		add = {
			value = intrigue
			multiply = agent_3to1_1_skill_mult_value
			ceiling = yes
			max = agent_max_skill_value
			desc = scheme_agent_aptitude.skill.intrigue
		}
		add = {
			value = learning
			multiply = agent_3to1_1_skill_mult_value
			ceiling = yes
			max = agent_max_skill_value
			desc = scheme_agent_aptitude.skill.learning
		}

		if = {
			limit = { has_trait = zealous }
			add = {
				value = agent_trait_bonus_t3_value
				desc = scheme_agent_aptitude.trait.zealous
			}
		}
		if = {
			limit = { has_trait = wrathful }
			add = {
				value = agent_trait_bonus_t2_value
				desc = scheme_agent_aptitude.trait.wrathful
			}
		}
		if = {
			limit = { has_trait = arbitrary }
			add = {
				value = agent_trait_bonus_t1_value
				desc = scheme_agent_aptitude.trait.arbitrary
			}
		}
		if = {
			limit = { has_trait = compassionate }
			add = {
				value = agent_trait_malus_t3_value
				desc = scheme_agent_aptitude.trait.compassionate
			}
		}

		max = agent_scheme_contributor_max
		min = agent_scheme_contributor_min
	}
}

# James and Brigitte, two Scarlet Crusade founders, share one OR slot (James
# dies 606.1.1, so it's Brigitte-only after that) - same narrative weight,
# whoever's alive fills it.
# contribution_type = secrecy, not success_chance_max like the other two:
# every non-secrecy agent slot drains -10 secrecy, and three would floor it
# at 0%. Diplomacy-weighted (mirrors vanilla agent_alibi) since both outclass
# their own intrigue there.
wc_mograine_founders_agent = {
	valid_agent_for_slot = {
		OR = {
			root = character:60560 # James
			root = character:60561 # Brigitte
