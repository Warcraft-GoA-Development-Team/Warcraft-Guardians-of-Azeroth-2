# Fired when you make a faith. Temporal rel head setup
# currently it can be fired by HoF who kick a rite out as a heresy, so always check correct scope. TIT-74780
faith_creation.0001 = {
	hidden = yes
	
	trigger = {
		faith = {
			NOT = { exists = religious_head }
			trigger_if = {
				limit = { exists = scope:rite }
				main_rite = scope:rite
			}
			has_doctrine = doctrine_temporal_head
			can_create_temporal_head_of_faith_title_trigger = yes
		}
	}
	
	immediate = {
		set_up_dynamic_temporal_hof_title_effect = { NEW_HOLDER = root }
	}
}

# Fired when you make a faith.
faith_creation.0003 = {
	hidden = yes
	
	trigger = {
		is_ai = no
	}

	immediate = {
		# Set original faith this one was based on
		faith = {
			if = {
				limit = { exists = scope:old_faith.var:foundational_faith }
				set_variable = {
					name = foundational_faith
					value = scope:old_faith.var:foundational_faith
				}
			}
			else = {
				set_variable = {
					name = foundational_faith
					value = scope:old_faith
				}
			}
		}
	}
}

# apply correct succession law for hof title
faith_creation.0004 = {
	hidden = yes
	trigger = {
		exists = root
		faith.religious_head_title.holder ?= this
		faith = {
			OR = {
				AND = {
					has_doctrine = doctrine_spiritual_head
					religious_head_title = { has_title_law = temporal_head_of_faith_succession_law }
				}
				AND = {
					has_doctrine = doctrine_temporal_head
					NOT = { religious_head_title = { has_title_law = temporal_head_of_faith_succession_law } }
				}
			}
		}
	}
	immediate = {
		faith = {
			if = {
				limit = { has_doctrine = doctrine_spiritual_head }
				religious_head_title ?= {
					remove_title_law = temporal_head_of_faith_succession_law
				}
			}
			else = {
				religious_head_title ?= {
					add_title_law = temporal_head_of_faith_succession_law
				}
			}
		}
	}
}

# Temporal Head of Faith re-creation event
faith_creation.0011 = {
	hidden = yes

	trigger = {
		faith = {
			trigger_if = {
				limit = { exists = scope:rite }
				main_rite = scope:rite
			}
			has_doctrine = doctrine_temporal_head
			can_create_temporal_head_of_faith_title_trigger = yes

			# Title must exist but be uncreated.
			exists = religious_head_title
			NOT = { exists = religious_head_title.holder }
		}
	}

	immediate = {
		faith = {	
			religious_head_title = {
				save_scope_as = my_hof_title
			}
		}

		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:my_hof_title = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:my_hof_title = {
			add_title_law = temporal_head_of_faith_succession_law
		}
	}
}

# Spiritual Head of Faith re-creation event
faith_creation.0012 = {
	hidden = yes

	trigger = {
		faith = {
			trigger_if = {
				limit = { exists = scope:rite }
				main_rite = scope:rite
			}
			has_doctrine = doctrine_spiritual_head
			can_create_spiritual_head_of_faith_title_trigger = yes

			# Title must exist but be uncreated.
			exists = religious_head_title
			NOT = { exists = religious_head_title.holder }			
		}
	}

	immediate = {
		faith = {
			religious_head_title = {
				save_scope_as = my_hof_title
			}
		}

		if = {
			limit = {
				root = { is_clergy = yes }
			}
			root = { save_scope_as = new_religious_head }
		}
		else_if = {
			limit = {
				root.puppet:theological_agent_puppet ?= { rite = root.rite }
			}
			root.puppet:theological_agent_puppet = {
				save_scope_as = new_religious_head
			}
		}
		else_if = {
			limit = {
				root.puppet:external_theological_agent_puppet ?= { rite = root.rite }
			}
			root.puppet:external_theological_agent_puppet = {
				save_scope_as = new_religious_head
			}
		}
		else_if = {
			limit = {
				root.cp:councillor_court_chaplain ?= { rite = root.rite }
			}
			root.cp:councillor_court_chaplain = {
				save_scope_as = new_religious_head
			}
		}
		else_if = {
			limit = {
				any_theocratic_vassal = {
					faith = root.faith
				}
				is_ai = yes # Warcraft
			}
			ordered_theocratic_vassal = {
				limit = {
					faith = root.faith
				}
				order_by = head_of_faith_selection_weight
				save_scope_as = new_religious_head
			}
		}
		else = {
			create_character = {
				location = root.capital_province
				# Warcraft
				age = { 30 50 }
				gender_female_chance = root_faith_clergy_gender_female_chance
				trait = education_learning_4
				faith = root.faith
				rite = root.rite
				culture = root.culture
				learning = { 14 22 }
				save_scope_as = new_religious_head
			}
			
			# Warcraft
			scope:new_religious_head = {
				trigger_race_giving_no_gene_effect = yes			#Assigns race trait
			}
		}

		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:my_hof_title = {
			change_title_holder = {
				holder = scope:new_religious_head
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		# The establishing rite becomes the mainline rite of the faith.
		if = {
			limit = {
				exists = rite
				NOT = { rite = faith.main_rite }
			}
			rite = { set_main_rite = yes }
		}
	}
}

# Fired when a new Faith is created by a player, or when a Pagan faith is reformed. Fired off an informative event for all players.
faith_creation.1000 = {
	hidden = yes

	trigger = {
		# declaring a rite a heresy branches it off into a new faith, which makes this particular subsequent event contextually unnecessary
		NOT = { exists = scope:diverged }
	}

	immediate = {
		save_scope_as = reformer
		faith = {
			save_scope_as = reformed_faith
		}
		if = {
			limit = {
				faith = {
					exists = religious_head
				}
			}
			faith = {
				religious_head = {
					save_scope_as = reformed_hof
				}
			}
		}

		if = {
			limit = {
				exists = scope:old_faith.religious_head
			}
			scope:old_faith = {
				religious_head = {
					save_scope_as = old_hof
				}
			}
		}

		every_player = {
			trigger_event = {
				id = faith_creation.1001
				days = 1
			}
		}
	}
}

								rite_has_tenet = tenet_natural_primitivism
								rite_has_tenet = tenet_mountain_worship
								rite_has_tenet = tenet_sun_worship
							}
						}
					}
				}
				desc = faith_creation.1001.desc.belief.nature
			}
			#Worldly Pragmatism
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {
						main_rite = {
							OR = {
								rite_has_tenet = tenet_simony
								rite_has_tenet = tenet_tax_nonbelievers
								rite_has_tenet = tenet_false_conversion_sanction
								rite_has_tenet = tenet_sacred_shadows
							}
						}
					}
				}
				desc = faith_creation.1001.desc.belief.worldly
			}
			#Syncretic
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {
						main_rite = {
							OR = {
								rite_has_tenet = tenet_adaptive
								# Warcraft - real-world syncretism tenets are not in GoA
								#rite_has_tenet = tenet_sinitic_syncretism
								#rite_has_tenet = tenet_eastern_syncretism
								#rite_has_tenet = tenet_unreformed_syncretism
								#rite_has_tenet = tenet_christian_syncretism
								#rite_has_tenet = tenet_islamic_syncretism
								#rite_has_tenet = tenet_jewish_syncretism
								has_syncretic_doctrine_trigger = yes # Warcraft
							}
						}
					}
				}
				desc = faith_creation.1001.desc.belief.syncretic
			}
			#Community
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {
						main_rite = {
							OR = {
								rite_has_tenet = tenet_communal_identity
								rite_has_tenet = tenet_ritual_hospitality
								rite_has_tenet = tenet_ritual_celebrations
							}
						}
					}
				}
				desc = faith_creation.1001.desc.belief.community
			}
			desc = faith_creation.1001.desc.belief.fallback
		}

		#Conclude with the new faith's religious leadership form.
		first_valid = {
			#Temporal Head and I am it!
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {
						has_doctrine = doctrine_temporal_head
						religious_head ?= root
					}
				}
				desc = faith_creation.1001.desc.temporal.personal
			}
			#Temporal Head and it is someone else...
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {
						has_doctrine = doctrine_temporal_head
						exists = religious_head
					}
				}
				desc = faith_creation.1001.desc.temporal
			}
			#Spiritual Head
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {
						has_doctrine = doctrine_spiritual_head
						exists = religious_head
					}
				}
				desc = {
					first_valid = {
						triggered_desc = {
							trigger = { scope:reformed_faith.religious_head = scope:old_hof } # Prevents the old pope from being labeled as new
							desc = faith_creation.1001.desc.spiritual.sameold
						}
						desc = faith_creation.1001.desc.spiritual
					}
				}
			}
			#No Head
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {
						has_doctrine = doctrine_no_head
					}
				}
				desc = faith_creation.1001.desc.no_head
			}
			triggered_desc = {
				trigger = {
					scope:reformed_faith = {				
						OR = {
							has_doctrine = doctrine_temporal_head
							has_doctrine = doctrine_spiritual_head
						}
						NOT = { exists = religious_head }
					}
				}
				desc = faith_creation.1001.desc.no_holy_sites
			}
		}
	}
	
	left_portrait = {
		character = scope:reformer
	}
	right_portrait = {
		trigger = {
			scope:reformed_hof != scope:reformer
		}
		character = scope:reformed_hof
		animation = personality_zealous
	}
	theme = faith
	override_background = {
		reference = temple_scope
	}

	immediate = {
		scope:reformer = {
			save_scope_as = background_temple_scope
		}
		play_music_cue = "mx_cue_epic_sacral_moment"
	}

	option = {
		name = faith_creation.1001.o.getreaction	

		# Granted in on_faith_created; shown here so the faithful learn about it.
		if = {
			limit = {
				faith = scope:reformed_faith
				scope:reformed_faith = {
					faith_has_free_holy_site_action_trigger = yes
				}
			}
			custom_tooltip = faith_creation.1001.o.free_holy_site_actions
		}
	}
}

# Fired when a rite has diverged so far it is detached into a new faith (scope:diverged set by code).
# Root is the head of the detached rite; players of the old and the new faith are notified.
faith_creation.1002 = {
	type = character_event
	window = big_event_window
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { faith = scope:old_rite.faith }
				desc = faith_creation.1002.t.new_faith
			}
			desc = faith_creation.1002.t
		}
	}
	desc = {
		# Who broke away, from whose side of the divide.
		first_valid = {
			triggered_desc = {
				trigger = { this = scope:founder }
				desc = faith_creation.1002.desc.intro.founder
			}
			triggered_desc = {
				trigger = { faith = scope:new_faith }
				desc = faith_creation.1002.desc.intro.member
			}
			desc = faith_creation.1002.desc.intro.observer
		}
		# The old faith's answer.
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:old_hof }
				desc = faith_creation.1002.desc.anathema.hof
			}
			desc = faith_creation.1002.desc.anathema.no_hof
		}
		# Leadership of the new faith.
		first_valid = {
			triggered_desc = {
				trigger = { scope:new_hof ?= this }
				desc = faith_creation.1002.desc.new_head.personal
			}
			triggered_desc = {
				trigger = { scope:new_hof ?= scope:founder }
				desc = faith_creation.1002.desc.new_head.founder
			}
			triggered_desc = {
				trigger = { exists = scope:new_hof }
				desc = faith_creation.1002.desc.new_head.other
			}
			desc = faith_creation.1002.desc.new_head.none
		}
		# What the break means, for us or for them.
		first_valid = {
			triggered_desc = {
				trigger = { faith = scope:new_faith }
				desc = faith_creation.1002.desc.outro.new_faith
			}
			desc = faith_creation.1002.desc.outro.old_faith
		}
	}

	theme = faith
	override_background = { reference = temple_scope }

	major = yes
	major_trigger = {
		OR = {
			faith = scope:old_faith
			faith = scope:old_rite.faith
		}
	}

	widget = {
		gui = "event_window_widget_rite_founder"
		container = "custom_widgets_container"
	}

	left_portrait = {
		character = scope:founder
		animation = preaching
		camera = camera_event_scheme_left
	}
	right_portrait = {
		trigger = {
			exists = scope:old_hof
			NOT = { scope:old_hof = scope:founder }
		}
		character = scope:old_hof
		animation = condemnation
		camera = camera_event_scheme_far_right
	}

	trigger = {
		exists = scope:diverged
		scope:diverged = yes
	}

	immediate = {
		save_scope_as = founder
		save_scope_as = background_temple_scope
		faith = { save_scope_as = new_faith }
		scope:old_faith.religious_head ?= { save_scope_as = old_hof }
		faith.religious_head ?= { save_scope_as = new_hof }
	}

	# The new faith embraces the break
	option = {
		name = {
			trigger = { this = scope:founder }
			text = faith_creation.1002.a.founder
		}
		name = faith_creation.1002.a
		trigger = { faith = scope:new_faith }
		custom_tooltip = faith_creation.1002.consequences.tt
		# Granted in on_faith_created; shown here so the faithful learn about it.
		if = {
			limit = {
				scope:new_faith = {
					faith_has_free_holy_site_action_trigger = yes
				}
			}
			custom_tooltip = faith_creation.1002.a.free_holy_site_actions
		}
		# Invisible ecumenism is stripped without any warning in the rite window
		if = {
			limit = {
				scope:old_faith = { has_doctrine = special_doctrine_ecumenical_christian }
				scope:new_faith = {
					NOT = { has_doctrine = special_doctrine_ecumenical_christian }
				}
			}
			custom_tooltip = faith_creation.1002.a.lost_ecumenism
		}
		change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		add_piety = medium_piety_gain
	}

	# The old faith condemns it
	option = {
		name = {
			trigger = { scope:old_hof ?= this }
			text = faith_creation.1002.b.hof
		}
		name = faith_creation.1002.b
		trigger = { faith = scope:old_faith }
		custom_tooltip = faith_creation.1002.consequences.tt
		add_piety = minor_piety_gain
	}
}

	}
