# Voluntary entry point: stage the canonical defeat before Sylvanas attacks.
wc_balnazzar_stage_death_decision = {
	ai_check_interval = 0
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = major

	desc = wc_balnazzar_stage_death_decision_desc

	is_shown = {
		this = character:430400 # Balnazzar
		is_alive = yes
		is_ai = no
		is_landed = yes
		has_title = title:k_lordaeron
		exists = global_var:balnazzar_dreadlord_story
		global_var:balnazzar_dreadlord_story = { story_owner = root }
		NOT = { has_character_flag = wc_balnazzar_staged_death }
		NOT = { has_character_flag = wc_balnazzar_disguise_active }
		NOT = { has_character_flag = balnazzar_became_dathrohan }
		exists = character:430100 # Varimathras
		character:430100 = { is_alive = yes }
		exists = character:42076 # Sylvanas Windrunner
		character:42076 = { is_alive = yes }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
		is_at_war = no
	}

	effect = {
		custom_tooltip = wc_balnazzar_stage_death_decision_effect_tooltip
		hidden_effect = {
			trigger_event = wc_balnazzar_story.0002
		}
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

# Player-only adventurer decision after losing Lordaeron; the player chooses
# when to take Dathrohan's place.

	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }

	# with the other adventurer-only major decisions instead of the generic major list.

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12

	desc = balnazzar_become_dathrohan_decision_desc

	is_shown = {

		NOT = { has_character_flag = balnazzar_became_dathrohan }

		base = 10
	}
}

balnazzar_move_against_alexandros_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	# Major decision - starts assassination scheme, point-of-no-return
	decision_group_type = major

	# Dathrohan holds Tyr's Hand; Order seized at .1007. AI check 0 = skip.
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	desc = balnazzar_move_against_alexandros_decision_desc

	is_shown = {
		has_character_flag = wc_balnazzar_disguise_active
		has_character_flag = balnazzar_kelthuzad_pact_sealed
		exists = character:60017 # Alexandros Mograine
		character:60017 = { is_alive = yes }
		# Hide only once Balnazzar's murder scheme against Alexandros is already running.
		# An unrelated murder scheme must not block this story decision.
		NOT = {
			any_scheme = {
				type = murder
				scheme_target_character = character:60017
				count > 0
			}
		}
	}

	# CK3 bug: bare is_alive=yes doesn't render in Requirements panel - wrap in custom_tooltip
	is_valid = {
		custom_tooltip = {
			text = balnazzar_move_against_alexandros_decision_requirement_alive
			exists = character:60017
			character:60017 = { is_alive = yes }
			exists = character:60019 # Renault Mograine
			character:60019 = { is_alive = yes }
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		custom_tooltip = balnazzar_move_against_alexandros_decision_effect_tooltip
		trigger_event = {
			id = wc_balnazzar_story.1005
			days = 3
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 50
	}
}

# Player-triggered reveal - gated on high stress
balnazzar_shed_the_mask_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = decisions

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	desc = balnazzar_shed_the_mask_decision_desc

	is_shown = {
		has_character_flag = wc_balnazzar_disguise_active
		OR = {
			stress_level >= 2
			has_character_modifier = wc_balnazzar_suspicion_inquiry_modifier
		}
		NOT = { has_character_flag = balnazzar_shed_the_mask }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_incapable_trigger = no
	}

	# CK3 quirk: bare trigger_event blanks Effects tooltip - wrap in custom_tooltip
	effect = {
		custom_tooltip = balnazzar_shed_the_mask_decision_effect_tooltip

		trigger_event = {
			id = wc_balnazzar_story.2000
			days = 3
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 10

		modifier = {
			add = 40
			has_trait = wrathful
		}
		modifier = {
			add = 40
			has_trait = paranoid
		}
	}
}
