# Legion attention / Kirin Tor standing API. Mirrors change_dark_frenzy_effect (wc_law_effects.txt).
# Both values are hidden: the tooltip shows vague prose, never the number.

wc_change_legion_attention_effect = {
	custom_description = {
		text = wc_change_legion_attention_effect
		value = $VALUE$

		# Only eligible characters start a meter, and a decrease on someone the Legion never noticed is a no-op, so no stray variable is left without its story
		if = {
			limit = {
				OR = {
					has_variable = wc_legion_attention
					AND = {
						$VALUE$ > 0
						wc_legion_attention_eligible_trigger = yes
					}
				}
			}
			if = {
				limit = { NOT = { has_variable = wc_legion_attention } }
				set_variable = { name = wc_legion_attention value = 0 }
			}
			change_variable = { name = wc_legion_attention add = $VALUE$ }
			clamp_variable = { name = wc_legion_attention min = wc_legion_attention_min_value max = wc_legion_attention_max_value }
		}
	}
	hidden_effect = { wc_legion_attention_start_story_effect = yes }
}

# Start the invisible story cycle on the first gain (it runs decay and the stage events)
wc_legion_attention_start_story_effect = {
	if = {
		limit = {
			wc_legion_attention_current_value > 0
			wc_legion_attention_eligible_trigger = yes
			NOT = { any_owned_story = { type = story_cycle_wc_legion_attention } }
		}
		create_story = { type = story_cycle_wc_legion_attention }
	}
}

# Final reset for the Dreadlord's Offer Refuse / Call-for-help options only (Accept ends the story instead); direct set so nothing afterwards changes it
wc_reset_legion_attention_effect = {
	custom_tooltip = wc_reset_legion_attention_effect_tt
	hidden_effect = {
		set_variable = { name = wc_legion_attention value = wc_legion_attention_reset_value }
	}
}

# Called from wc_legion_attention_on_cast; root is the caster
wc_legion_attention_on_cast_effect = {
	if = {
		limit = { wc_legion_attention_cast_gain_value > 0 }
		wc_change_legion_attention_effect = { VALUE = wc_legion_attention_cast_gain_value }
	}
}

			set_variable = { name = wc_kirin_tor_standing value = 0 }

	}
