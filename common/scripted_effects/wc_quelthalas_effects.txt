destroy_corrupted_sunwell_effect = {
    title:c_sunwell_plateau.title_province = {
        remove_building = sunwell_corrupted_death
        add_special_building = sunwell_inert
    }
}

create_blood_knight_order_effect = {
	if = { 
		limit = {
			character:42305 = { 
				is_alive = yes
				employer = title:e_quelthalas.holder
			}
		}
		character:42305 = { 
			save_scope_as = liadrin
		}
	}
	else = { 
		create_character = {
			template = liadrin_substitute
			save_scope_as = liadrin
		}
	}
	set_variable = {
		name = liadrin
		value = scope:liadrin
	}

	# Set up Titles
	create_title_and_vassal_change = {
		type = returned
		save_scope_as = change
		#add_claim_on_loss = no
	}
	title:d_blood_knights_order = {
		holder = {
			if = {
				limit = { NOT = { this = scope:liadrin } }
				add_opinion = {
					target = scope:liadrin
					modifier = revoked_title
					opinion = -30
				}
			}
		}
		change_title_holder = {
			holder = scope:liadrin
			change = scope:change
		}
	}
	title:c_suns_reach = {
		holder = {
			if = {
				limit = { NOT = { this = scope:liadrin } }
				add_opinion = {
					target = scope:liadrin
					modifier = revoked_title
					opinion = -30
				}
			}
		}
		change_title_holder = {
			holder = scope:liadrin
			change = scope:change
		}
	}
	title:c_dawnstar = {
		holder = {
			if = {
				limit = { NOT = { this = scope:liadrin } }
				add_opinion = {
					target = scope:liadrin
					modifier = revoked_title
					opinion = -30
				}
			}
		}
		change_title_holder = {
			holder = scope:liadrin
			change = scope:change
		}
	}

	scope:liadrin = {
		change_liege = {
			liege = title:e_quelthalas.holder
			change = scope:change
		}
	}

	resolve_title_and_vassal_change = scope:change

	if = {
		limit = {
			has_title = title:d_blood_knights_order
		}
		set_primary_title_to = title:d_blood_knights_order
	}
}
