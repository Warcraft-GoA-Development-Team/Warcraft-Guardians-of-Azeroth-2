#Give characters money to improve their opinion of you
gift_interaction = {
	icon = icon_gold
	category = interaction_category_friendly
	common_interaction = yes
	interface_priority = 65
	desc = gift_interaction_desc
	use_diplomatic_range = { # Warcraft
		trigger_if = {
			limit = {
				scope:puppet_or_actor = { has_variable = this_union }
				scope:recipient ?= { has_variable = this_union }
			}
			NOT = {
				scope:puppet_or_actor.var:this_union = scope:recipient.var:this_union
			}
		}
	}

	greeting = positive
	notification_text = SEND_GIFT_PROPOSAL

	answer_accept_key = SEND_GIFT_ACCEPT
	answer_reject_key = SEND_GIFT_REJECT

	ai_targets = {
		ai_recipients = scripted_relations
		ai_recipients = liege
		ai_recipients = head_of_faith
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_targets = {
		ai_recipients = vassals
		max = 10
	}
	ai_target_quick_trigger = {
		adult = yes
	}
	ai_frequency_by_tier = {
		barony = 120
		county = 60
		duchy = 60
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			ai_greed < medium_positive_ai_value
			short_term_gold >= gift_interaction_cutoff
			ai_should_focus_on_building_in_their_capital = no
			is_available_at_peace_ai_adult = yes
			NOR = {
				has_trait = greedy
				has_variable = conqueror
			}
		}
	}

	is_shown = {
		# Prioritize paying back loans first
		scope:puppet_or_actor ?= {
			this != scope:recipient
			trigger_if = {
				limit = {
					exists = var:loan_amount_owed
					exists = var:loan_holder
				}
				NOT = { var:loan_holder = scope:recipient }
			}
			# A landless puppet has nothing to gift that is not already the puppeteer's
			trigger_if = {
				limit = { scope:recipient = scope:actor }
				is_ruler = yes
			}
		}
	}

	is_highlighted = {
		scope:recipient = {
			OR = {
				house ?= {
					OR = {
						is_dominant_family = yes
						is_powerful_family = yes
					}
				}
			}
		}
	}

	highlighted_reason = HIGHLIGHTED_FAMILY_RATING

	is_valid_showing_failures_only = {
		scope:puppet_or_actor.gold >= gift_value
		scope:recipient = {
			NOT = { is_imprisoned_by = scope:puppet_or_actor }
		}
		# puppet_or_actor only differs from the actor on puppet actions, where the puppet's own purse pays
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			custom_tooltip = {
				text = puppet_gift_cooldown_tt
				NOT = { scope:puppet_or_actor = { has_variable = puppet_gift_cooldown } }
			}
		}
	}

	on_auto_accept = {
		scope:recipient = {
			trigger_event = char_interaction.0100
		}
	}

	on_accept = {
		scope:recipient = {
			# Verify that they could become friend
			if = {
				limit = {
					NAND = {
						has_relation_friend = scope:puppet_or_actor
						has_relation_lover = scope:puppet_or_actor
						has_relation_soulmate = scope:puppet_or_actor
						has_relation_best_friend = scope:puppet_or_actor
					}
				}
				gifting_leads_towards_friendship_effect = yes
			}
		}
		# HOUSE RELATIONS
		if = {
			limit = {
				scope:puppet_or_actor = {
					is_house_head = yes
				}
				scope:recipient = {
					is_house_head = yes
					NOT = {
						has_opinion_modifier = {
							target = scope:puppet_or_actor
							modifier = gift_opinion
						}
					}
				}
			}
			scope:recipient.house = {
				change_house_relation_effect = {
					HOUSE = scope:puppet_or_actor.house
					VALUE = house_relation_improve_medium_value
					REASON = sent_a_gift
					CHAR = scope:puppet_or_actor
					TARGET_CHAR = scope:recipient
					TITLE = scope:dummy_gender
				}
			}
		}
		scope:puppet_or_actor = {
			# Warning for multiple gifts
			if = {
				limit = {
					scope:actor != scope:recipient
					scope:recipient = {
						has_opinion_modifier = {
							target = scope:puppet_or_actor
							modifier = gift_opinion
						}
					}
				}
				custom_tooltip = ALREADY_SENT_GIFT_WARNING
			}

			send_interface_message = {
				type = event_gold_neutral
				title = gift_interaction_notification
				right_icon = scope:recipient
				pay_short_term_gold = {
					gold = gift_value
					target = scope:recipient
				}
				stress_and_fulfillment_impact = {
					greedy = medium_stress_impact_gain
				}
				if = {
					limit = {
						scope:recipient = {
							NOT = {
								is_heir_of = scope:puppet_or_actor
							}
						}
					}
					stress_and_fulfillment_impact = {
						generous = medium_stress_impact_loss
					}
				}
				if = {
					limit = {
						scope:puppet_or_actor = {
							has_personal_tenet_flag = tenet_simony_personal_gifts
						}
					}
					# Doubled: personal pick has stress_loss_mult -0.5
					add_stress = medium_stress_loss
					change_spiritual_fulfillment = miniscule_spiritual_fulfillment_value
				}

				#FP3 Tenet Communal Possessions Perk - piety gain for gift giving.
				if = {
					limit = {
						scope:puppet_or_actor = {
							rite = {
					 			rite_has_parameter = piety_from_gifts_active
					 		}
					 	}
						NOT = {
							scope:recipient = {
								has_opinion_modifier = {
									target = scope:puppet_or_actor
									modifier = gift_opinion
								}
							}
						}
					}
					scope:puppet_or_actor = {
						add_piety = minor_piety_gain
					}
				}

				if = {
					limit = {
						scope:puppet_or_actor = {
							has_personal_tenet_flag = tenet_philanthropia_alms_to_the_destitute
						}
						scope:recipient = { is_in_debt = yes }
						NOT = {
							scope:recipient = {
								has_opinion_modifier = {
									target = scope:puppet_or_actor
									modifier = gift_opinion
								}
							}
						}
					}
					scope:puppet_or_actor = {
						add_prestige = minor_prestige_gain
						add_piety = minor_piety_gain
					}
				}

				# Check if the target already has been sent a gift or not. If not, apply the following effects
				if = {
					limit = {
						NOT = {
							scope:recipient = {
								has_opinion_modifier = {
									target = scope:puppet_or_actor
									modifier = gift_opinion
								}
							}
						}
					}
					
					# Warcraft - Remove Struggle Catalyst
					
					# FP2 Urbanism Legacy Perk 2: increase cultural acceptance when gifting a republican vassal
					if = {
						limit = {
							scope:puppet_or_actor = {
								dynasty ?= { has_dynasty_perk = fp2_urbanism_legacy_4 }
							}
							scope:recipient = {
								AND = {
									is_vassal_of = scope:puppet_or_actor
									government_has_flag = government_is_republic
								}
							}
							scope:puppet_or_actor.culture != scope:recipient.capital_province.county.culture
						}
						scope:recipient.capital_province.county.culture = {
							change_cultural_acceptance = {
								target = scope:puppet_or_actor.culture
								value = {
									value = 20
									multiply = scope:recipient.capital_province.county.development_level
									divide = 100
								}
								desc = cultural_acceptance_gain_gift_dynasty_perk
							}
						}
					}
				}

				# Let's apply the opinion modifier last, as to apply everything else correctly first
				# Don't grant the puppeteer opinion of their own puppet when they are the gift recipient
				if = {
					limit = {
						scope:actor != scope:recipient
						scope:puppet_or_actor = {
							NOT = { has_personal_tenet_flag = tenet_simony_personal_gifts }
						}
					}
					scope:recipient = {
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = gift_opinion
							opinion = send_gift_opinion
						}
					}
				}
				if = {
					limit = {
						scope:actor != scope:recipient
						scope:puppet_or_actor = {
							has_personal_tenet_flag = tenet_simony_personal_gifts
						}
					}
					scope:recipient = {
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = friendliness_opinion
							opinion = send_gift_opinion
						}
					}
				}
			}
			if = {
				limit = {
					scope:actor != scope:recipient
					scope:puppet_or_actor = {
						has_personal_tenet_flag = tenet_simony_personal_gifts
					}
					scope:recipient = {
						NOT = {
							has_opinion_modifier = {
								target = scope:puppet_or_actor
								modifier = friendliness_opinion
							}
						}
					}
				}
				progress_towards_friend_effect = {
					REASON = friend_simony_gift
					CHARACTER = scope:recipient
					OPINION = 0
				}
			}

			if = { # for tutorial purposes
				limit = {
					is_ai = no
					scope:recipient = { is_child_of = scope:puppet_or_actor }
				}
				add_character_flag = {
					flag = tutorial_sent_gift
					days = 200
				}
			}
			#Influence gain between admin rulers
			if = {
				limit = {
					is_ruler = yes
					government_has_flag = government_has_influence
					scope:recipient = {
						is_ruler = yes
						government_has_flag = government_has_influence
						is_governor_or_landless_admin = yes
					}
				}
				if = {
					limit = {
						scope:recipient = {
							OR = {
								house = {
									is_dominant_family = yes
								}
								this = scope:puppet_or_actor.top_liege
							}
							influence_level >= 5
						}
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 8
					}
				}
				else_if = {
					limit = {
						scope:recipient = {
							OR = {
								house = {
									is_dominant_family = yes
								}
								this = scope:puppet_or_actor.top_liege
							}
							influence_level >= 3
						}
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 7
					}
				}
				else_if = {
					limit = {
						scope:recipient = {
							OR = {
								house = {
									is_dominant_family = yes
								}
								this = scope:puppet_or_actor.top_liege
							}
						}
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 6
					}
				}
				else_if = {
					limit = {
						scope:recipient = {
							OR = {
								house = {
									is_powerful_family = yes
								}
								this = scope:puppet_or_actor.top_liege
							}
							influence_level >= 5
						}
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 6
					}
				}
				else_if = {
					limit = {
						scope:recipient = {
							OR = {
								house = {
									is_powerful_family = yes
								}
								this = scope:puppet_or_actor.top_liege
							}
							influence_level >= 3
						}
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 5
					}
				}
				else_if = {
					limit = {
						scope:recipient = {
							OR = {
								house = {
									is_powerful_family = yes
								}
								this = scope:puppet_or_actor.top_liege
							}
						}
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 4
					}
				}
				else_if = {
					limit = {
						scope:recipient = { influence_level >= 5 }
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 3
					}
				}
				else_if = {
					limit = {
						scope:recipient = { influence_level >= 3 }
					}
					change_influence = {
						value = miniscule_influence_gain
						multiply = 2
					}
				}
				else = {
					change_influence = miniscule_influence_gain
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor = { is_ai = no }
			}
			scope:puppet_or_actor = {
				set_variable = {
					name = puppet_gift_cooldown
					years = 5
				}
			}
		}
	}

	ai_accept = {
		base = 0
		modifier = {
			add = 100
			desc = GOLD_REASON
		}

		# Struggle motive
		

		# Warcraft
		scourge_interaction_modifiers = yes

	}

	auto_accept = {
		custom_description = {
			text = auto_accept_interaction_ai
			object = scope:recipient
			scope:recipient = {
				is_ai = yes
			}
		}
	}

	ai_min_reply_days = 1
	ai_max_reply_days = 1

	ai_will_do = {
		base = 100

		modifier = { # Do not send double-gifts
			factor = 0
			scope:recipient = {
				has_opinion_modifier = {
					target = scope:puppet_or_actor
					modifier = gift_opinion
				}
			}
		}
		modifier = { # Do not repeat Simony gifts
			factor = 0
			scope:puppet_or_actor = {
				has_personal_tenet_flag = tenet_simony_personal_gifts
			}
			scope:recipient = {
				has_opinion_modifier = {
					target = scope:puppet_or_actor
					modifier = friendliness_opinion
				}
			}
		}

		modifier = { # Basic Filtering
			factor = 0
			scope:recipient = {
				NOR = {
					AND = { # Bankrupt lovers, friends and a bankrupt liege should be considered
						OR = {
							scope:puppet_or_actor = {
								any_liege_or_above = {
									this = scope:recipient
								}
							}
							has_secret_relation_lover = scope:puppet_or_actor
							has_relation_lover = scope:puppet_or_actor
							has_relation_soulmate = scope:puppet_or_actor
							has_relation_friend = scope:puppet_or_actor
							has_relation_best_friend = scope:puppet_or_actor
						}
						gold < 0
					}
					AND = { # Generous characters will give gifts to their friends/lovers
						is_ruler = yes
						OR = {
							has_secret_relation_lover = scope:puppet_or_actor
							has_relation_lover = scope:puppet_or_actor
							has_relation_soulmate = scope:puppet_or_actor
							has_relation_friend = scope:puppet_or_actor
							has_relation_best_friend = scope:puppet_or_actor
						}
						scope:puppet_or_actor = {
							OR = {
								ai_greed <= high_negative_ai_value
								AND = {
									ai_greed < 0
									culture = {
										has_cultural_parameter = gives_more_gifts
									}
								}
							}
						}
					}
					AND = { # Characters with the Generous Cultural Tradition will give gifts to more people
						is_ruler = yes
						OR = {
							is_allied_to = scope:puppet_or_actor
							is_close_or_extended_family_of = scope:puppet_or_actor
						}
						scope:puppet_or_actor = {
							ai_greed < 0
							culture = {
								has_cultural_parameter = gives_more_gifts
							}
						}
					}
					AND = { # Characters with the loyal trait more likely to give gifts to friends/lieges
						OR = {
							scope:puppet_or_actor.liege ?= this
							has_relation_friend = scope:puppet_or_actor
							has_relation_best_friend = scope:puppet_or_actor
							is_allied_to = scope:puppet_or_actor
							is_close_or_extended_family_of = scope:puppet_or_actor
						}
						scope:puppet_or_actor = {
							ai_greed <= 0
							has_trait = loyal
						}
					}
					AND = { # Zealous characters want to support the defending side in GHW's
						gold < 200
						any_character_war = {
							OR = {
								using_cb = undirected_great_holy_war
								using_cb = directed_great_holy_war
							}
							primary_defender = scope:recipient
						}
						scope:puppet_or_actor = {
							ai_zeal >= 50
						}
					}
					AND = { # Powerful vassals should be considered
						is_powerful_vassal_of = scope:puppet_or_actor
						opinion = {
							target = scope:puppet_or_actor
							value < 0
						}
						NOT ={
							has_opinion_modifier = {
								target = scope:puppet_or_actor
								modifier = gift_opinion
							}
						}
					}
					AND = { # Factioneering vassals should be considered
						is_vassal_of = scope:puppet_or_actor
						is_a_faction_member = yes
						NOT ={
							has_opinion_modifier = {
								target = scope:puppet_or_actor
								modifier = gift_opinion
							}
						}
					}
					AND = { # Realm Priests should be considered
						scope:puppet_or_actor = {
							rite = { rite_has_doctrine = doctrine_theocracy_temporal }
							exists = cp:councillor_court_chaplain
							cp:councillor_court_chaplain = scope:recipient
						}
						opinion = {
							target = scope:puppet_or_actor
							value <= 25
						}
						NOT = {
							has_opinion_modifier = {
								target = scope:puppet_or_actor
								modifier = gift_opinion
							}
						}
					}
					AND = { # Rulers at war with your rivals deserve gold if they're bankrupt
						gold < 0
						is_at_war = yes
						any_war_enemy = {
							has_relation_rival = scope:puppet_or_actor
						}
					}
					AND = { # Independent ruler within the struggle should be considered
					 	any_character_struggle = {
					 		phase_has_catalyst = catalyst_gift_independent_ruler
					 	}
					}
				}
			}
		}

		# Struggle
		
		
		modifier = {
			factor = 0.1
			scope:recipient = {
				opinion = {
					target = scope:puppet_or_actor
					value < 0
				}
				NAND = {
					is_vassal_of = scope:puppet_or_actor
					scope:puppet_or_actor = {
						ai_rationality > 50
					}
				}
			}
		}
		
		# Nomads are slightly less likely to do this since they'd prefer to send Herd instead
		modifier = {
			add = -10
			scope:puppet_or_actor = {
				government_has_flag = government_is_nomadic
			}
			scope:recipient = {
				government_has_flag = government_is_nomadic
			}
		}

		modifier = {
			factor = 0.1
			scope:recipient = {
				has_relation_rival = scope:puppet_or_actor
			}
		}

		modifier = {
			factor = 2
			scope:puppet_or_actor.house.house_confederation ?= {
				has_cohesion_level_parameter = bloc_members_send_leader_gifts
				leading_house.house_head ?= scope:recipient
			}
		}
	}
}
