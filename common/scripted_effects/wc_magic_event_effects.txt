    if = {
        limit = {
            NOT = {
                has_character_modifier = wc_studying_leyline_crystals_character
            }
        }
        send_interface_toast = {
            title = wc_studying_leyline_crystals_title
            type = event_toast_effect_good
            add_character_modifier = {
                modifier = wc_studying_leyline_crystals_character
            }
        }
        trigger_event = {
            on_action = wc_studying_leyline_removal_pulse
            months = 1
        }
    }
}
