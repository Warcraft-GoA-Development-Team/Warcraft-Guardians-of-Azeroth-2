# A Face Not His Own - Balnazzar fakes death, takes Saidan Dathrohan's identity (character:60021).
# Player transfer, seizes Order of the Silver Hand from within.
wc_balnazzar_story.1002 = {
	type = character_event
	title = wc_balnazzar_story.1002.title
	desc = wc_balnazzar_story.1002.desc
	theme = intrigue

	override_background = wc_undercity_throne_room

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60021 # Saidan Dathrohan
		animation = worry
	}

	immediate = {
		character:60021 = { save_scope_as = dathrohan }
		global_var:balnazzar_dreadlord_story = {
			set_variable = { name = story_stage value = 3 }
		}
	}

	# .1002.b: fallback if Dathrohan is dead/unplayable - avoids a soft-lock.
	option = {
		name = wc_balnazzar_story.1002.a

		trigger = {
			scope:dathrohan = {
				is_alive = yes
				is_playable_character = yes
			}
		}

		show_as_tooltip = {
			death = { death_reason = death_vanished }
		}
		hidden_effect = {
			scope:dathrohan = {
				add_character_flag = balnazzar_wearing_dathrohan
				add_trait = balnazzar_disguise

				# Possession burns off Dathrohan's age-related decline.
				# Ongoing protection while trait held: wc_story_cycle_balnazzar.txt effect_group.
				remove_trait = clouded_eyes
				remove_trait = faltering_heart
				remove_trait = fragile_bones
				remove_trait = infirm
				remove_trait = withering_mind

				# The truth itself, tracked as a real CK3 secret - exposure
				# (however it happens) resolves the story via .2000.
				add_dreadlord_disguise_secret_effect = yes
			}

			# Keep story attached to whoever plays Balnazzar's arc.
			# Without this, killing root fires on_owner_death and ends story early.
			global_var:balnazzar_dreadlord_story = {
				make_story_owner = scope:dathrohan
			}

			# Move player off root before killing it, or engine aborts effect at death screen.
			# Call unscoped (root=player), not inside scope:dathrohan - engine requires current scope = player.
			set_player_character = scope:dathrohan

			death = { death_reason = death_vanished } # Balnazzar's old face "dies" - root, now an NPC

			scope:dathrohan = {
				trigger_event = {
					id = wc_balnazzar_story.1003
					days = { 10 30 }
				}
			}
		}
	}

	option = {
		name = wc_balnazzar_story.1002.b

		trigger = {
			NOT = {
				scope:dathrohan = {
					is_alive = yes
					is_playable_character = yes
				}
			}
		}

		hidden_effect = {
			global_var:balnazzar_dreadlord_story = {
				end_story = yes
			}
		}
	}
}

# Blood on Silver - Alexandros dead, Renault broken; Dathrohan-Balnazzar takes the Order.
# Soon reforged as the Scarlet Crusade.
wc_balnazzar_story.1006 = {
	type = character_event
	title = wc_balnazzar_story.1006.title
	desc = wc_balnazzar_story.1006.desc
	theme = intrigue

	override_background = sm_pa_scarlet_monastery_sm

	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = character:60019 # Renault Mograine
		animation = shame # Hollow, guilt-ridden - days removed from the kill itself
	}

	immediate = {
		character:60019 = { save_scope_as = renault }
		global_var:balnazzar_dreadlord_story = {
			set_variable = { name = story_stage value = 7 }
		}
	}

	# .1006.b is the fallback if Renault is dead/missing by this point.
	option = {
		name = wc_balnazzar_story.1006.a

		trigger = {
			scope:renault = { is_alive = yes }
		}

		hidden_effect = {
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = balnazzar_silver_hand_change
				add_claim_on_loss = yes
			}
			title:k_order_of_the_silver_hand = {
				change_title_holder_include_vassals = {
					holder = root
					change = scope:balnazzar_silver_hand_change
				}
			}
			resolve_title_and_vassal_change = scope:balnazzar_silver_hand_change

			# Succession can briefly make Renault our liege before we reseize the Order.
			# Force him back into vassalage, or his duchy stays outside the realm.
			if = {
				limit = {
					scope:renault = { NOR = { top_liege = root this = root } }
				}
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = balnazzar_renault_vassalize_change
				}
				scope:renault = {
					change_liege = {
						liege = root
						change = scope:balnazzar_renault_vassalize_change
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_renault_vassalize_change
			}

			# The Order becomes the Scarlet Crusade - crown, color and creed all at once.
			title:k_order_of_the_silver_hand = {
				set_title_name = scarlet_crusade_custom_name
				generate_coa = yes
				set_color_from_title = title:k_scarlet_monastery # The faith's own head title - authoritative Scarlet red
			}

			# Scoped to Silver Hand realm only (root + vassals), not every Holy Light believer on the map.
			# Whole-world conversion would be a much bigger swing.
			if = {
				limit = { has_faith = faith:holy_light }
				set_character_faith_with_conversion = faith:scarletism
			}
			every_vassal_or_below = {
				limit = { has_faith = faith:holy_light }
				set_character_faith_with_conversion = faith:scarletism
			}

			# This IS forming the Scarlet Crusade - grant same reward as form_the_scarlet_crusade_decision, mark consumed.
			# Otherwise that generic decision keeps showing (its own is_shown check can't self-suppress).
			add_character_modifier = wc_scarlet_crusader
			add_character_flag = has_formed_the_scarlet_crusade
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:form_the_scarlet_crusade_decision
			}
			spawn_army = {
				uses_supply = no
				inheritable = yes
				name = scarlet_crusade
				men_at_arms = { type = trebuchet stacks = 5 }
				men_at_arms = { type = archmagi stacks = 5 }
				men_at_arms = { type = armored_horsemen stacks = 5 }
				men_at_arms = { type = armored_footmen stacks = 5 }
				men_at_arms = { type = light_footmen stacks = 5 }
				men_at_arms = { type = bowmen stacks = 5 }
				men_at_arms = { type = pikemen_unit stacks = 5 }
				location = capital_province
				origin = capital_province
			}

			# TODO: Varimathras arc (fake loyalty to Sylvanas, secretly still
			# Legion-bound) deliberately deferred - see conversation notes.
		}
	}

	option = {
		name = wc_balnazzar_story.1006.b

		trigger = {
			NOT = { scope:renault = { is_alive = yes } }
		}

		hidden_effect = {
			global_var:balnazzar_dreadlord_story = {
				end_story = yes
			}
		}
	}
}

# The Mask Comes Off - shared resolution for all "reject the mask" triggers (old age, exposure, paranoia).
# Disguise ends, Crusade burns, the Risen rise at Stratholme.
wc_balnazzar_story.2000 = {
	type = character_event
	title = wc_balnazzar_story.2000.title
	desc = wc_balnazzar_story.2000.desc
	theme = death

	# Self-guard against double-firing - the flag is only set once .2000
	# actually runs, not when it's queued, so two callers could race.
	trigger = {
		NOT = { has_character_flag = balnazzar_shed_the_mask }
	}

	override_background = wc_stratholme

	left_portrait = {
		character = root
		animation = rage
	}

	immediate = {
		add_character_flag = balnazzar_shed_the_mask
		remove_trait = balnazzar_disguise
	}

	option = {
		name = wc_balnazzar_story.2000.a

		hidden_effect = {
			# Courtiers convert if physically in Hasic - excludes guests/
			# hostages who happen to be present but aren't his household.
			every_courtier = {
				limit = {
					is_alive = yes
					NOT = { has_trait = being_undead }
					location.county = root.capital_county
				}
				raise_undead_faith_only_effect = { MASTER = root }
				add_opinion = { modifier = loyal_undead target = root }
			}

			# Resurrect Balnazzar's nathrezim body (character:430400, "died" at .1002) - one-shot, purges source.
			# Falls through to the shell surviving if unavailable, instead of an unrecoverable Game Over.
			if = {
				limit = {
					exists = character:430400
					is_landed = yes
					exists = capital_county
				}

				wc_resurrect_CHARACTER_with_STATUS_effect = {
					CHARACTER = character:430400
					STATUS = alive
				}

				# Capital county + every held duchy break away with Balnazzar.
				# type = granted, not "independence" (invalid, fails "No type set" at read-time).
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = balnazzar_reveal_title_change
					add_claim_on_loss = no
				}
				root.capital_county = {
					change_title_holder_include_vassals = {
						holder = scope:resurrected_character
						change = scope:balnazzar_reveal_title_change
					}
				}
				every_held_title = {
					limit = { tier = tier_duchy }
					save_scope_as = balnazzar_duchy_grant
					change_title_holder_include_vassals = {
						holder = scope:resurrected_character
						change = scope:balnazzar_reveal_title_change
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_reveal_title_change

				# swear_fealty's tier check compares primary_title, not highest held title.
				# Without this, Balnazzar stays count-tier and vassalizing fails "same or higher tier".
				if = {
					limit = { exists = scope:balnazzar_duchy_grant }
					scope:resurrected_character = {
						set_primary_title_to = scope:balnazzar_duchy_grant
					}
				}

				# Grant leaves him a vassal of the de jure liege - sever explicitly.
				# type = independency, not "independence" (invalid value).
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = balnazzar_independence_change
				}
				scope:resurrected_character = {
					change_liege_or_become_independent = {
						CHANGE = scope:balnazzar_independence_change
						VASSAL = this
					}
				}
				resolve_title_and_vassal_change = scope:balnazzar_independence_change

				# include_vassals above already re-vassalized everyone in the duchy - just convert them.
				# Army composition mirrors this mod's Scourge pattern, not .1006's living Crusade.
				scope:resurrected_character = {
					every_vassal_or_below = {
						limit = {
							is_alive = yes
							NOT = { has_trait = being_undead }
						}
						raise_undead_faith_only_effect = { MASTER = liege }
						add_opinion = { modifier = loyal_undead target = liege }
					}
					spawn_army = {
						uses_supply = no
						inheritable = yes
						name = the_risen
						men_at_arms = { type = ghoul stacks = 8 }
						men_at_arms = { type = deathcaller stacks = 4 }
						men_at_arms = { type = banshee stacks = 4 }
						men_at_arms = { type = frost_wyrm stacks = 1 }
						location = capital_province
						origin = capital_province
					}
				}

				# Move the player off root before killing it - killing the
				# still-played root aborts the rest of the effect otherwise.
				set_player_character = scope:resurrected_character

				# Fires on_owner_death (story_balnazzar_dreadlord), ending
				# the story - no separate end_story call needed.
				death = { death_reason = death_vanished }
			}
			else = {
				# No valid handoff body - root survives, exposed but alive,
				# rather than risk an unrecoverable Game Over.
				global_var:balnazzar_dreadlord_story = {
