# County scope; only the ruler's directly owned holdings count.
wc_titanforged_local_forge_level = {
	value = 0
	every_county_province = {
		limit = { province_owner = scope:councillor_liege }
		if = { limit = { has_building_or_higher = titan_forge_01 } min = 1 }
		if = { limit = { has_building_or_higher = titan_forge_02 } min = 2 }
		if = { limit = { has_building_or_higher = titan_forge_03 } min = 3 }
		if = { limit = { has_building_or_higher = titan_forge_04 } min = 4 }
		if = { limit = { has_building_or_higher = titan_forge_05 } min = 5 }
		if = { limit = { has_building_or_higher = titan_forge_06 } min = 6 }
		if = { limit = { has_building_or_higher = titan_forge_07 } min = 7 }
		if = { limit = { has_building_or_higher = titan_forge_08 } min = 8 }
	}
}

# Native county adjacency is terrestrial; no realm/coastal reach or recursion.
wc_titanforged_forge_service_level = {
	value = 0
	if = {
		limit = { scope:councillor_liege = { completely_controls = prev } }
		if = {
			limit = { wc_titanforged_local_forge_level > 0 }
			value = wc_titanforged_local_forge_level
		}
		else = {
			every_neighboring_county = {
				limit = { scope:councillor_liege = { completely_controls = prev } }
				min = wc_titanforged_local_forge_level
			}
		}
	}
}

wc_titanforged_forge_speed = {
	value = scope:county.wc_titanforged_forge_service_level
	subtract = 1
	multiply = 0.05
	add = 1
}
