ai_spell_modifier_value = { # Which spell should I pick?
    if = {
        limit = {
            exists = var:current_spell_name
            has_variable = real_caster
        }

        ### RELATIONS ###
        if = {
            limit = {
                spell_is_harmful_trigger = { SPELL = var:current_spell_name }
                var:real_caster = {
                    OR = {
                        has_any_negative_relation_trigger = yes
                        is_at_war = yes
                    }
                }
            }
            add = 50
        }
        if = {
            limit = {
                spell_is_helpful_trigger = { SPELL = var:current_spell_name }
                var:real_caster = { has_any_positive_relation_trigger = yes }
            }
            add = 50
        }
        if = {
            limit = {
                spell_is_harmful_trigger = { SPELL = var:current_spell_name }
                var:real_caster = {
                    any_in_list = {
                        variable = recently_casted_on_by_harmful_targets
                        count >= 1
                    }
                }
            }
            add = 25
        }
        ### PERSONALITY ###
        if = {
            limit = {
                spell_is_harmful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_personality_malicious_trigger = yes }
            }
            add = 100
        }
        if = {
            limit = {
                spell_is_helpful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_personality_benevolent_trigger = yes }
            }
            add = 100
        }
        if = {
            limit = {
                spell_is_harmful_trigger = { SPELL = var:current_spell_name }
                var:real_caster = { has_trait = sadistic }
            }
            add = 25
        }
        ### MAGIC MODIFIERS ###
        if = {
            limit = {
                spell_is_ritual_trigger = { SPELL = var:current_spell_name }
            }
            add = 100
        }
        if = { # Any spells that summon troops
            limit = {
                spell_summons_maa_trigger = { SPELL = var:current_spell_name }
                is_at_war = yes
            }
            add = 100
        }
        if = { # Any spells that affect knights or troops
            limit = {
                OR = {
                    spell_target_is_friendly_army_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
                    spell_target_is_enemy_army_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
                }
                is_at_war = yes
            }
            add = 50
        }
        if = {
            limit = {
                spell_restores_mana_trigger = { SPELL = var:current_spell_name }
                var:wc_current_mana_variable <= wc_thirty_percent_of_max_mana_value
            }
            add = 25
        }
        if = {
            limit = {
                spell_is_spell_type_enchantment_trigger = { SPELL = var:current_spell_name }
            }
            add = 10
        }

        ### EVENT MODIFIERS ###

        ### SPELL SPECIFIC MODIFIERS ###
        # subtract = {
        #     value = var:current_spell_mana_cost
        #     multiply = 0.5
        # }

        ## ORDER
        ## DISORDER
        ## LIFE
        ## DEATH
        ## LIGHT
        if = {
            limit = {
                var:current_spell_name = flag:salvation
                NOT = { has_good_salvation_target_trigger = yes }
            }
            add = -250
        }
        ## ELEMENTAL FIRE
        ## ELEMENTAL WATER
        ## ELEMENTAL EARTH
        ## ELEMENTAL AIR
        # Shouldn't choose raise dead unless someone actually died (funeral activity?)
    }
}

ai_spell_target_character_modifier_value = {
    if = {
        limit = {
            exists = var:spell_recipient
            exists = var:real_caster
            exists = var:real_caster.var:spell_recipient
        }

        ### RELATIONS ###
        if = {
            limit = {
                spell_is_harmful_trigger = { SPELL = var:current_spell_name }
                spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
            }
            add = {
                value = "opinion(var:spell_recipient)"
                multiply = -5.5
            }
        }
        if = {
            limit = {
                spell_is_helpful_trigger = { SPELL = var:current_spell_name }
                spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
            }
            add = {
                value = "opinion(var:spell_recipient)"
                multiply = 5.5
            }
        }
        if = {
            limit = {
                spell_is_harmful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:spell_recipient = { is_vassal_or_below_of = var:real_caster }
            }
            add = -75
        }
        if = {
            limit = {
                spell_is_helpful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:spell_recipient = { is_vassal_or_below_of = var:real_caster }
            }
            add = 75
        }
        if = {
            limit = {
                var:real_caster = { is_of_major_interest_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 50
        }
        if = {
            limit = {
                var:real_caster = { is_of_minor_interest_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 25
        }
        if = {
            limit = {
                var:spell_recipient = { has_no_real_status_trigger = yes }
            }
            add = -75
        }

        if = {
            limit = {
                spell_is_helpful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_any_mild_good_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 10
        }
        if = {
            limit = {
                spell_is_helpful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 25
        }
        if = {
            limit = {
                spell_is_helpful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_any_best_good_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 50
        }

        if = {
            limit = {
                spell_is_harmful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_any_mild_bad_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 10
        }
        if = {
            limit = {
                spell_is_harmful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_any_moderate_bad_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 25
        }
        if = {
            limit = {
                spell_is_harmful_trigger = {
                    SPELL = var:current_spell_name
                }
                var:real_caster = { has_any_worst_bad_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
            }
            add = 50
        }

        if = {
            limit = {
                spell_is_harmful_trigger = {
                    SPELL = var:current_spell_name
                }
                OR = {
                    var:real_caster = { is_at_war_with = var:spell_recipient }
                    var:real_caster = { is_in_coalition_against_trigger = { CRISIS_HOLDER = var:spell_recipient } }
                }
            }
            add = 100
        }

        if = {
            limit = {
                spell_is_harmful_trigger = { SPELL = var:current_spell_name }
                var:real_caster = {
                    any_in_list = {
                        variable = recently_casted_on_by_harmful_targets
                        this = var:spell_recipient
                    }
                }
            }
            add = 250
        }

        if = {
            limit = {
                spell_is_helpful_trigger = { SPELL = var:current_spell_name }
                var:real_caster = {
                    any_in_list = {
                        variable = recently_casted_on_by_helpful_targets
                        this = var:spell_recipient
                    }
                }
            }
            add = 250
        }

        # TODO_MAGIC_AI: Don't cast on people more powerful than you, scaling, depending on your personality
        if = { # Higher rank than me
            limit = {
                is_playable_character = yes
                var:real_caster.highest_held_title_tier < var:spell_recipient.highest_held_title_tier
            }
            add = {
                value = var:spell_recipient.highest_held_title_tier
                subtract = var:real_caster.highest_held_title_tier
                multiply = 5
            }
        }
        if = { # Lower rank than me
            limit = {
                var:spell_recipient ?= { is_playable_character = yes }
                var:real_caster.highest_held_title_tier > var:spell_recipient.highest_held_title_tier
            }
            add = {
                value = var:real_caster.highest_held_title_tier
                subtract = var:spell_recipient.highest_held_title_tier
                multiply = -5
            }
        }
        if = {
            limit = {
                var:spell_recipient.wc_mp_value > var:real_caster.wc_mp_value
                spell_is_harmful_trigger = { SPELL = var:current_spell_name }
            }
            add = {
                value = var:spell_recipient.wc_mp_value
                subtract = var:real_caster.wc_mp_value
                multiply = -10
            }
        }
        # TODO_MAGIC_AI: Scaling per relation & general modifier - war targets/allies as well
        # TODO_MAGIC_AI: More likely to cast rituals on themselves or their vassals
        # TODO_MAGIC_AI: Less likely to on people outside of realm in general, scaling with distance
        # TODO_MAGIC_AI: Don't cast harmful spells on yourself unless you have flag trait

        ### PERSONALITY ###
        if = {
            limit = {
                spell_is_harmful_trigger = { SPELL = $SPELL$ }
                var:real_caster = { has_trait = flagellant }
                var:real_caster = var:spell_recipient
            }
            add = 100
        }

        ### MAGIC MODIFIERS ###
        # If someone has a bunch of traits or magic modifiers try not to cast too often on them

        ### EVENT MODIFIERS ###

        ### SPELL SPECIFIC MODIFIERS ###
        ## ORDER
        ## DISORDER
        ## LIFE
        ## DEATH
        ## LIGHT
        if = {
            limit = {
                exists = var:spell_recipient
                is_good_salvation_target_trigger = { TARGET = var:spell_recipient }
                var:current_spell_name = flag:salvation
                spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
            }
            add = 250
        }
        ## SHADOW
        if = {
            limit = {
                var:spell_recipient ?= {
                    OR = {
                        is_magic_type_criminal_trigger = { MAGICTYPE = shadow }
                        is_magic_type_shunned_trigger = { MAGICTYPE = shadow }
                    }
                }
                var:current_spell_name = flag:void_leader
                spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
            }
            add = -250

        }
        ## ELEMENTAL FIRE
        ## ELEMENTAL WATER
        ## ELEMENTAL EARTH
        ## ELEMENTAL AIR
    }
