is_from_the_drust_trigger = {
	OR = {
		AND = {
			has_title = title:k_drustvar
			owns_story_of_type = story_drust_incursion
		}
		OR = {
			is_vassal_or_below_of = title:k_drustvar.holder
			employer ?= title:k_drustvar.holder
			is_courtier_of = title:k_drustvar.holder
		}
	}
}

title_in_drust_warpath_trigger = {
	OR = {
		empire = title:e_kul_tiras
		empire = title:e_arathor
		empire = title:e_khaz_modan
		empire = title:e_lordaeron
	}
}
gorak_tul_can_war_trigger = { 
	top_liege = this
	save_temporary_scope_as = possible_enemy
	scope:gorak_tul = {
		can_declare_war = {
			casus_belli = mongol_invasion_war
			defender = scope:possible_enemy
			target_titles = { scope:possible_target.kingdom }
		}
	}
	scope:gorak_tul.current_military_strength > scope:possible_enemy.current_military_strength
}
