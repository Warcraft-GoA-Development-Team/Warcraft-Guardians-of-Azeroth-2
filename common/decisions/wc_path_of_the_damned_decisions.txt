digging_up_the_dead_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/digging_up_the_dead.dds" }
	sort_order = 10

	title = digging_up_the_dead_decision
	desc = digging_up_the_dead_decision_desc

	ai_check_interval = 1
	decision_group_type = major
	selection_tooltip = digging_up_the_dead_decision_tooltip

	is_shown = {
		game_start_date < 605.6.6

		any_owned_story = {
			type = story_path_of_the_damned
			var:story_stage = 2
		}

		NOT = { # You can't do that as lich king
			has_trait = lich_king
			has_title = title:h_scourge
		}

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:digging_up_the_dead_decision
			}
		}
	}

	is_valid = {
		AND = {
			is_landed = yes
			scourge_quelthalas_frostmourne_check_trigger = yes
			has_trait = champion_of_the_scourge
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	cost = {}

	effect = {
		arthas_scourge_spawn_troops_effect = yes

		global_var:path_of_the_damned_story.var:enchanted_urn = {
			# Save scopes
			save_scope_as = enchanted_urn
			artifact_owner = { save_scope_as = war_target }

			# Digging up the Dead events
			root = {
				add_personal_artifact_claim = scope:enchanted_urn # Claim
				start_war = { # War
					cb = artifact_war
					target = scope:war_target
				}
			}

			# Holy orders join the war
			custom_tooltip = digging_up_the_dead_decision_holy_orders_join_effect.tt
			scope:war_target = {
				trigger_event = {
					id = wc_path_of_the_damned.1903
					days = 1
				}
			}
		}

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:digging_up_the_dead_decision
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}

invade_scourge_quelthalas_decision = { # Triggers the next phase of the Third War
	picture = { reference = "gfx/interface/illustrations/decisions/wc_undercity_throne_room_decision.dds" }
	sort_order = 10

	# By Raged
	title = {
		first_valid = {
			# Everyone (Arthas, or arthas equivalent)
			desc = invade_scourge_quelthalas_decision
		}
	}
	desc = {
		first_valid = {
			# Everyone (Arthas, or Arthas equivalent)
			desc = invade_scourge_quelthalas_decision.desc
		}
	}
	
	ai_check_interval = 12
	decision_group_type = major
	selection_tooltip = invade_scourge_quelthalas_decision_tt

	is_shown = {
		game_start_date < 605.6.6

		any_owned_story = {
			type = story_path_of_the_damned
			var:story_stage = 3
		}

		NOT = { # You can't do that as lich king
			has_trait = lich_king
			has_title = title:h_scourge
		}

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_scourge_quelthalas_decision
			}
		}
	}
	

	cost = {}
	
	is_valid = {
		AND = {
			is_landed = yes
			scourge_quelthalas_frostmourne_check_trigger = yes
			scourge_quelthalas_phylactery_check_trigger = yes
			has_trait = champion_of_the_scourge
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = scourge_quelthalas_decision_effect.tt
		save_scope_as = invaderquelthalas

		if = {
			limit = {
				OR = {
					title:c_sunwell_plateau.holder = { has_title = title:h_scourge }
					title:c_sunwell_plateau.holder = { is_vassal_of = title:h_scourge.holder }
				}
			}
			trigger_event = { id = scourgequel.1011 } # Jump to summoning Kel'thuzad
		}
		else = { trigger_event = { id = scourgequel.0001 } } # Start a war if nor held by liege or fellow vassal

		hidden_effect = {
			if = {
				limit = { any_living_character = { is_kelthuzad_trigger = yes } }
				character:59100 = {
					override_death_effect = { death_reason = death_disappearance }
				}
			}

			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_scourge_quelthalas_decision
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}

resurrect_kelthuzad_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/wc_undercity_throne_room_decision.dds" }
	sort_order = 10

	ai_check_interval = 1
	decision_group_type = major
	selection_tooltip = resurrect_kelthuzad_decision_tt

	is_shown = {
		NOT = { # You can't do that as lich king
			has_trait = lich_king
			has_title = title:h_scourge
		}
		
		AND = { # For arthas
			scourge_quelthalas_frostmourne_check_trigger = yes
			is_landed = yes
		}
		game_start_date < 605.6.6

		is_target_in_global_variable_list = { # Only after Quelthalas is defeated
			name = unavailable_unique_decisions
			target = flag:invade_scourge_quelthalas_decision
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:resurrect_kelthuzad_decision
			}
		}
	}


	cost = {}

	is_valid = {
		AND = { # For arthas
			scourge_quelthalas_frostmourne_check_trigger = yes
			is_landed = yes
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = resurrect_kelthuzad_decision_effect_tt

		hidden_effect = { 
			wc_resurrect_CHARACTER_with_STATUS_effect = { CHARACTER = character:59100 STATUS = lich }
			scope:resurrected_character = {
				root = { add_courtier = prev }
				add_character_flag = kel_thuzad_lich

				set_character_faith = faith:death_god
				set_culture = culture:scourge
			}
		}

		trigger_event = {
			id = scourgequel.1011
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}
