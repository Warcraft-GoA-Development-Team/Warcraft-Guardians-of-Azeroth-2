##########################################################################################
# 0002 - Pikuach Nefesh (rescue a traveler on the Sabbath)
##########################################################################################
scripted_effect pam_secular_faith_shabbat_rescue_success_effect = {
	if = {
		limit = { can_add_hook = { target = scope:traveler_in_danger type = saved_my_life_hook } }
		add_hook = { type = saved_my_life_hook target = scope:traveler_in_danger }
	}
	scope:traveler_in_danger = {
		add_opinion = { target = root modifier = grateful_opinion opinion = 30 }
		if = {
			limit = { exists = current_travel_plan }
			current_travel_plan = { add_travel_plan_modifier = pam_sf_safe_passage_modifier }
		}
	}
}

scripted_effect pam_secular_faith_shabbat_rescue_failure_effect = {
	scope:traveler_in_danger = {
		increase_wounds_no_death_effect = { REASON = fight }
		if = {
			limit = { exists = current_travel_plan }
			current_travel_plan = { add_travel_plan_modifier = pam_sf_waylaid_road_modifier }
		}
	}
}

			add_known_tenet = tenet_sacred_shadows
			pam_secular_faith_seed_tenet_message_effect = { TENET = tenet:tenet_sacred_shadows }
		}
		10 = {
			trigger = { NOT = { knows_tenet = tenet_human_sacrifice } }
			add_known_tenet = tenet_human_sacrifice
			pam_secular_faith_seed_tenet_message_effect = { TENET = tenet:tenet_human_sacrifice }
		}
		10 = {
			trigger = { NOT = { knows_tenet = tenet_pursuit_of_power } }
			add_known_tenet = tenet_pursuit_of_power
			pam_secular_faith_seed_tenet_message_effect = { TENET = tenet:tenet_pursuit_of_power }
		}
		10 = {
			trigger = { NOT = { knows_tenet = tenet_carnal_exaltation } }
			add_known_tenet = tenet_carnal_exaltation
			pam_secular_faith_seed_tenet_message_effect = { TENET = tenet:tenet_carnal_exaltation }
		}
		10 = {
			trigger = { NOT = { knows_tenet = tenet_hedonistic } }
			add_known_tenet = tenet_hedonistic
			pam_secular_faith_seed_tenet_message_effect = { TENET = tenet:tenet_hedonistic }

		# Safety fallback - only reachable once every notable tenet above is already known, so the list is never empty.

				knows_tenet = tenet_sacred_shadows

				knows_tenet = tenet_pursuit_of_power

			base = massive_stress_impact_loss
			zealous = medium_stress_impact_gain

			base = 25
			modifier = {
				factor = 0
				OR = {
					has_trait = zealous

			}
		}

##########################################################################################
# 0005 - By Candlelight (evening study reveals a foreign tenet or doctrine)
##########################################################################################
scripted_effect pam_secular_faith_study_revelation_select_effect = {
	random_list = {
		50 = {
			trigger = {
				any_character_tenet = {
					knowledge_filter = unknown
					count >= 1
				}
			}
			random_character_tenet = {
				knowledge_filter = unknown
				save_scope_as = pam_sf_revelation_tenet
			}
		}
		50 = {
			trigger = {
				any_character_doctrine = {
					knowledge_filter = unknown
					passive_rite_learning_doctrine_is_learnable_trigger = yes
					count >= 1
				}
			}
			random_character_doctrine = {
				knowledge_filter = unknown
				limit = { passive_rite_learning_doctrine_is_learnable_trigger = yes }
				save_scope_as = pam_sf_revelation_doctrine
			}
		}
	}
}

# Grant knowledge of whichever subject pam_secular_faith_study_revelation_select_effect picked.
scripted_effect pam_secular_faith_study_revelation_learn_effect = {
	if = {
		limit = { exists = scope:pam_sf_revelation_tenet }
		add_known_tenet = scope:pam_sf_revelation_tenet
	}
	else_if = {
		limit = { exists = scope:pam_sf_revelation_doctrine }
		add_known_doctrine = scope:pam_sf_revelation_doctrine
	}
}

# 0006 - Bones of a Saint (the relic-seller)
##########################################################################################
scripted_trigger pam_sf_relic_seller_valid_trigger = {
	is_available_healthy_ai_adult = yes
	has_no_particular_noble_roots_trigger = yes
	faith = root.faith
}

pam_secular_faith_events.0006 = {
	type = character_event
	title = pam_secular_faith_events.0006.t
	desc = pam_secular_faith_events.0006.desc
	theme = christian
	override_background = market

	left_portrait = {
		character = root
		animation = contemplating
	}
	right_portrait = {
		character = scope:relic_seller
		animation = beg
	}

	cooldown = { years = 50 }

	trigger = {
		pam_secular_faith_christian_actor_trigger = yes
		gold > 0
	}

	immediate = {
		if = {
			limit = {
				any_pool_character = {
					province = root.location
					pam_sf_relic_seller_valid_trigger = yes
				}
			}
			random_pool_character = {
				province = root.location
				limit = { pam_sf_relic_seller_valid_trigger = yes }
				save_scope_as = relic_seller
			}
		}
		else = {
			create_character = {
				location = root.capital_province
				template = monk_character_template
				faith = root.faith
				rite = root.rite
				culture = root.culture
				save_scope_as = relic_seller
				after_creation = {
					add_character_flag = created
				}
			}
		}
	}

	option = { # Appraise it as a scholar - a learned eye may judge it a far finer piece
		name = pam_secular_faith_events.0006.a
		trigger = {
			has_trait = lifestyle_scholar
		}
		show_as_unavailable = { always = yes }
		custom_tooltip = pam_sf_examine_relic_tt
		pay_short_term_gold = {
			target = scope:relic_seller
			gold = minor_gold_value
		}
		add_scholar_trait_or_xp_effect = { VALUE = 20 }
		hidden_effect = {
