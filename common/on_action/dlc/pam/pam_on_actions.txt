pam_cathedral_great_project_yearly_on_action = {
	effect = {
		every_great_project = {
			limit = {
				OR = {
					great_project_type = pam_cathedral_02
					great_project_type = pam_cathedral_03
				}
			}
			great_project_founder = {
				save_scope_as = founder
				trigger_event = {
					on_action = pam_cathedral_great_project_yearly_founder_events_on_action
					years = 5
				}
			}
			great_project_owner = {
				if = {
					limit = {
						this = scope:founder
					}
					trigger_event = {
						on_action = pam_cathedral_great_project_yearly_owner_events_on_action
						years = 10
					}
				}
				else = {
					trigger_event = {
						on_action = pam_cathedral_great_project_yearly_owner_events_on_action
						years = 5
					}
				}
			}
		}
	}
}

pam_personal_tenet_status_change_fulfillment_on_action = {
	trigger = {
		OR = {
			any_in_list = { list = changed_tenet_status count >= 1 }
			any_in_list = { list = added_tenets count >= 1 }
			any_in_list = { list = removed_tenets count >= 1 }
		}
	}
	effect = {
		root.faith = {
			every_faith_character = {
				pam_personal_tenet_rite_update_fulfillment_effect = yes
			}
		}
	}
}

pam_study_faith_personal_tenet_on_action = {
	effect = {
		if = {
			limit = {
				scope:owner = {
					has_personal_tenet_flag = adaptive_study_rite_cultural_acceptance
					culture != scope:target.culture
				}
			}
			scope:owner = {
				culture = {
					change_cultural_acceptance = {
						target = scope:target.culture
						value = minor_cultural_acceptance_gain
						desc = pam_adaptive_study_rite_cultural_acceptance
					}
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_adaptive_study_rite_cultural_acceptance_toast
					left_icon = scope:owner
					right_icon = scope:target
				}
			}
		}
	}
}

pam_mountain_travel_personal_tenet_on_action = {
	effect = {
		if = {
			limit = {
				has_personal_tenet_flag = cthonic_redoubts_mountain_travel_fulfillment

				set_variable = pam_mountain_travel_fulfillment_awarded
			}

				type = event_toast_effect_good

				left_icon = root
				add_piety = minor_piety_gain
				change_spiritual_fulfillment = minor_spiritual_fulfillment_value
			}
		}
	}
}

pam_sanctity_of_nature_travel_on_action = {
	effect = {
		if = {
			limit = {
				current_travel_plan.current_location = {
					has_province_modifier = preserved_wilderness_modifier
				}
			}
			current_travel_plan = {
				if = {
					limit = {
						NOT = { has_travel_plan_modifier = preserved_wilderness_travel_modifier }
					}
					add_travel_plan_modifier = preserved_wilderness_travel_modifier
				}
			}
		}
		else_if = {
			limit = {
				current_travel_plan = {
					has_travel_plan_modifier = preserved_wilderness_travel_modifier
				}
			}
			current_travel_plan = {
				remove_travel_plan_modifier = preserved_wilderness_travel_modifier
			}
		}

		if = {
			limit = {
				has_personal_tenet_flag = tenet_sanctity_of_nature_wilderness_fulfillment
				current_travel_plan.current_location = {
					has_holding = no
					is_sea_province = no
				}
				current_travel_plan = {
					OR = {
						NOT = { has_variable_list = pam_sanctity_of_nature_visited_wilderness }
						AND = {
							has_variable_list = pam_sanctity_of_nature_visited_wilderness
							NOT = {
								is_target_in_variable_list = {
									name = pam_sanctity_of_nature_visited_wilderness
									target = root.current_travel_plan.current_location
								}
							}
						}
					}
				}
			}
			change_spiritual_fulfillment = miniscule_spiritual_fulfillment_value
			current_travel_plan = {
				add_to_variable_list = {
					name = pam_sanctity_of_nature_visited_wilderness
					target = root.current_travel_plan.current_location
				}
			}
		}
	}
}

		if = {
			limit = {
				OR = {
					scope:mother = {

					}

						has_personal_tenet_flag = mystical_birthright_spiritual_leader
					}
				}
			}
			# Base 5% chance for later children; the first-child modifiers shift the chance to 20%.
			random_list = {
				95 = {
					modifier = {
						add = -15
						OR = {
							scope:mother = {
								has_personal_tenet_flag = mystical_birthright_spiritual_leader
								NOR = {
									any_child = {
										even_if_dead = yes
										this != scope:child
									}
								}
							}
							scope:real_father ?= {
								has_personal_tenet_flag = mystical_birthright_spiritual_leader
								NOR = {
									any_child = {
										even_if_dead = yes
										this != scope:child
									}
								}
							}
						}
					}

						}

# scope:county = capital county
# scope:reason = reason that triggered this mismatch. flag can be:
#	- county_converted (county is now of different faith, holder stayed old faith)
#	- character_converted (character converted away from county faith)
#	- changed_tenets_or_doctrines (county and character are still matching faith but the faith doesn't support clerical regions anymore)
on_clerical_region_faith_mismatch = {
	# TIT-75963 player choice when capital converts; TIT-75964 other succession cases
	events = {
		# Capital converted; both Faiths have Clerical Regions (player)
		pam_clergy_events.0051
		# AI / other mismatch reasons / new Faith lacks Clerical Regions
		pam_clergy_events.0052
	}
}
