namespace = wc_holy_war

# Great Holy War proclamation (issue #1973), fired by wc_proclaim_holy_war_decision
# wc_holy_war.0001 - The head of faith proclaims the holy war (applies the tenet swap)
# wc_holy_war.0002 - Letter to every other ruler of the faith

# The Head of Faith Proclaims the Holy War
wc_holy_war.0001 = {
	type = character_event
	window = big_event_window
	theme = faith
	content_source = dlc_GOA
	title = wc_holy_war.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { faith = faith:holy_light }
				desc = wc_holy_war.0001.desc.light
			}
			triggered_desc = {
				trigger = { faith = faith:aldor }
				desc = wc_holy_war.0001.desc.aldor
			}
			triggered_desc = {
				trigger = { faith = faith:cult_of_sethraliss }
				desc = wc_holy_war.0001.desc.sethraliss
			}
			triggered_desc = {
				trigger = { faith = faith:spider_philosophy }
				desc = wc_holy_war.0001.desc.nerubian
			}
			triggered_desc = {
				trigger = { faith = faith:forgotten_shadow }
				desc = wc_holy_war.0001.desc.forsaken
			}
			triggered_desc = {
				trigger = { faith = faith:cult_of_sunwell }
				desc = wc_holy_war.0001.desc.sunwell
			}
			desc = wc_holy_war.0001.desc.zandalari # the remaining roster faiths are the Zandalari loa
		}
	}

	widget = { # Shows the changed rite, as vanilla pam_decision_events.0019
		gui = "event_window_widget_rite"
		container = "custom_widgets_container"
	}

	left_portrait = {
		character = root
		animation = preaching
	}

	immediate = {
		save_scope_as = hof_scope
		wc_proclaim_holy_war_swap_effect = yes
	}

	# So it shall be
	option = {
		name = wc_holy_war.0001.a
	}

	after = {
		faith = {
			every_faith_character = {
				limit = {
					is_ruler = yes
					this != scope:hof_scope
				}
				trigger_event = {
					id = wc_holy_war.0002
					days = 1
				}
			}
		}
	}
}

# Letter: The Holy War Is Proclaimed

	sender = scope:hof_scope
	opening = wc_holy_war.0002.opening
	desc = {

			triggered_desc = {
				trigger = { scope:hof_scope.faith = faith:aldor }

			}
			triggered_desc = {
				trigger = {
					scope:hof_scope.faith = { wc_holy_war_struggle_faith_trigger = yes }

				desc = wc_holy_war.0002.desc.struggle_submission
			}
			desc = wc_holy_war.0002.desc.armed_pilgrimages
		}
	}

	# So be it
	option = {
		name = wc_holy_war.0002.a
	}
}
