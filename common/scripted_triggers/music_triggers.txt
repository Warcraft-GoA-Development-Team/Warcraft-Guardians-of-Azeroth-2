# Various music triggers

# Trigger that blocks generic music groups for specific cultures/governments that should only use their own
should_not_use_bespoke_music_trigger = {
	trigger_if = {
		limit = {
			has_tgp_dlc_trigger = yes
			should_use_chinese_music_trigger = yes
		}
		always = no
	}
	trigger_else_if = {
		limit = {
			has_mpo_dlc_trigger = yes
			should_use_nomadic_music_trigger = yes
		}
		always = no
	}
	trigger_else = {
		always = yes
	}
}

should_not_be_valid_if_asian_music_is_valid_trigger = {
	trigger_if = {
		limit = {
			has_tgp_dlc_trigger = yes
			should_use_asian_music_trigger = yes
		}
		always = no
	}
	trigger_else = {
		always = yes
	}
}
# Warcrafted
should_use_asian_music_trigger = {
	OR = {
		AND = {
			culture = {
				has_building_gfx = steppe_building_gfx
			}
			OR = {
				culture = { like_heritage_mongolic_trigger = yes }
				culture = { any_parent_culture_or_above = { like_heritage_mongolic_trigger = yes } }
				culture = { like_heritage_tungusic_trigger = yes }
				culture = { any_parent_culture_or_above = { like_heritage_tungusic_trigger = yes } }
			}
			capital_province ?= {
				geographical_region = world_pandaria 
			}
		}
		culture = {
			OR = {
				has_building_gfx = chinese_building_gfx
				has_building_gfx = indian_building_gfx
				has_building_gfx = southeast_asian_building_gfx
				has_building_gfx = tibetan_building_gfx
				has_building_gfx = japanese_building_gfx
			}
		}
		like_sinitic_religion_trigger = yes
		like_buddhism_religion_trigger = yes
		like_hinduism_religion_trigger = yes
		like_shintoism_religion_trigger = yes
		any_character_situation = { situation_type = dynastic_cycle }
	}
}

should_use_chinese_music_trigger = {
	OR = {
		culture = { like_heritage_chinese_trigger = yes }
		culture = { any_parent_culture_or_above = { like_heritage_chinese_trigger = yes } }
		any_character_situation = { situation_type = dynastic_cycle }
		like_sinitic_religion_trigger = yes
	}
}

	like_christianity_religion_trigger = yes
}

should_use_european_christian_music_trigger = {
	NOT = { government_has_flag = government_is_nomadic }
	religion = religion:christianity_religion
	OR = {
		AND = {
			like_christianity_religion_trigger = yes
			like_orthodox_faith_trigger = no
			like_miaphysitism_faith_trigger = no
			like_armenian_apostolic_faith_trigger = no
			like_nestorian_faith_trigger = no
			like_messalian_faith_trigger = no
		}
		AND = {
			exists = capital_province
			capital_province = {
				OR = {
					geographical_region = world_europe_west
					geographical_region = world_europe_north
					geographical_region = world_europe_south
				}
			}
		}
	}
}

					OR = {
						like_heritage_iberian = yes

						like_heritage_west_germanic = yes
