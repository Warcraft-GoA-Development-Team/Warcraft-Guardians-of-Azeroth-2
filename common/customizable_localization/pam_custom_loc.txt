# Situation Participant Groups
GetMyChurchPower = {
	type = character
	text = {
		trigger = {
			top_participant_group:the_christian_church ?= {
				participant_group_type = christian_clerical_main_power_rulers
			}
		}
		localization_key = my_church_power_clerical_main
	}
	text = {
		trigger = {
			top_participant_group:the_christian_church ?= {
				participant_group_type = christian_regular_main_power_rulers
			}
		}
		localization_key = my_church_power_regular_main
	}
	text = {
		trigger = {
			top_participant_group:the_christian_church ?= {
				participant_group_type = christian_clerical_secondary_powers_rulers
			}
		}
		localization_key = my_church_power_clerical_secondary
	}
	text = {
		trigger = {
			top_participant_group:the_christian_church ?= {
				participant_group_type = christian_regular_secondary_powers_rulers
			}
		}
		localization_key = my_church_power_regular_secondary
	}
	text = {
		trigger = {
			top_participant_group:the_christian_church ?= {
				participant_group_type = christian_clerical_heretic_rulers
			}
		}
		localization_key = my_church_power_clerical_heretic
	}
	text = {
		trigger = {
			top_participant_group:the_christian_church ?= {
				participant_group_type = christian_regular_heretic_rulers
			}
		}
		localization_key = my_church_power_regular_heretic
	}
	### FALLBACK
	text = {
		trigger = { always = no }
		localization_key = my_church_power_fallback
		fallback = yes
	}	
}

GetClericalRegionMapModeTooltip = {
	type = province

	text = {
		setup_scope = {
			county.clerical_region_title ?= { save_scope_as = clerical_region_title }
		}
		trigger = {
			county ?= {
				clerical_region_title ?= { exists = holder }
				NOT = { rite = clerical_region_title.holder.rite }
			}
		}
		localization_key = CLERICAL_REGIONS_MAP_MODE_TOOLTIP_HELD_DIVERGENT
	}

	text = {
		setup_scope = {
			county.clerical_region_title ?= { save_scope_as = clerical_region_title }
		}
		trigger = {
			county.clerical_region_title ?= { exists = holder }
		}
		localization_key = CLERICAL_REGIONS_MAP_MODE_TOOLTIP_HELD
	}

	text = {
		setup_scope = {
			county.clerical_region_title ?= { save_scope_as = clerical_region_title }
		}
		trigger = {
			exists = county.clerical_region_title
		}
		localization_key = CLERICAL_REGIONS_MAP_MODE_TOOLTIP_UNHELD
	}

	text = { localization_key = CLERICAL_REGIONS_MAP_MODE_TOOLTIP_NONE fallback = yes }
}

GetTopLiegeRiteLine = {
	type = province

	text = {
		setup_scope = {
			county.holder.top_liege ?= { save_scope_as = county_top_liege }
		}
		trigger = {
			county.holder ?= {
				exists = top_liege
				NOT = { rite = top_liege.rite }
			}
		}
		localization_key = FAITHS_MAP_MODE_TOOLTIP_TOP_LIEGE_RITE
	}
}
