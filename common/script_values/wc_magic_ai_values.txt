minimum_days_before_casting_value = {
    value = 14

    if = {
        limit = {
            has_character_modifier = important_lore_character
        }
        multiply = 0.5
    }

    if = { # Has some buff promoting casting
		limit = {
            any_spell_modifier_trigger = { OPERATOR = OR TRIGGER = has_positive_spell_modifier }
        }
		subtract = 5
	}

    if = { # Someone doing it for them
        limit = {
            OR = {
                has_summoned_demon_trigger = yes
                has_subjugated_trigger = yes
            }
        }
        subtract = 5
    }

    min = 0
}
