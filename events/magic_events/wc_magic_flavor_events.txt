# Someone offers to sell you a mana potion
wc_magic_flavor_events.0010 = {
    type = character_event
	content_source = dlc_GOA
    theme = magic
    title = wc_magic_flavor_events.0010.t
    desc = wc_magic_flavor_events.0010.d

    cooldown = { years = 10 }

    trigger = {
        has_variable = wc_current_mana_variable
        var:wc_current_mana_variable <= wc_fourty_percent_of_max_mana_value
    }

    weight_multiplier = {
		base = 25
		modifier = { 
			add = {
				if = {
					limit = { 
						var:wc_current_mana_variable <=  wc_twenty_percent_of_max_mana_value
					}
					add = 50
				}
			}
        }
    }

    immediate = {
        create_character = {
			template = merchant_template
			location = root.location
			random_traits_list = {
                count = 1
                honest = {}
                deceitful = {}
            }
			save_scope_as = merchant
			after_creation = {
				add_character_flag = created
			}
		}
    }

    left_portrait = {
 		character = root
        animation = interested
	}
	right_portrait = {
 		character = scope:merchant
		animation = steward
	}

    # Ok how much is it 
    option = {
        name = wc_magic_flavor_events.0010.a
        trigger = {
            short_term_gold >= medium_gold_value
        }
        show_as_unavailable = { always = yes }
        
        pay_short_term_gold = { gold = medium_gold_value target = scope:merchant }

        get_potion_effect_0010 = yes

        stress_impact = {
            trusting = minor_stress_impact_loss   
            greedy = minor_stress_impact_gain
        }  

        ai_chance = {
            base = 75
            ai_value_modifier = {
                ai_greed = -1
            }
        }
    }

    # Let me examine the potion
    option = {
        name = wc_magic_flavor_events.0010.b

        duel = {
            skill = learning
            value = high_skill_rating
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = 3.5
                }
                custom_tooltip = wc_magic_flavor_events.0010.b_tt
                send_interface_toast = {
                    title = wc_magic_flavor_events.0010.b_tt
                    type = event_toast_effect_good
                    left_icon = root
                    right_icon = scope:merchant
                    wc_change_mana_by_value_effect = {
                        CHANGE = add
                        VALUE = minor_mana_value
                    }
                }
            }
            50 = {
                compare_modifier = {
                    value = scope:duel_value
                    multiplier = -3.5
                }
                send_interface_toast = {
                    title = wc_magic_flavor_events.0010.b_bad
                    type = event_toast_effect_neutral
                    left_icon = root
                    right_icon = scope:merchant
                    add_stress = minor_stress_gain
                }
            }
        }

        stress_impact = {
            paranoid = minor_stress_impact_loss  
            stubborn = minor_stress_impact_loss 
        }  

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_rationality = 1
            }
        }
    }

    # Just give it to me or die
    option = {
        name = wc_magic_flavor_events.0010.c 
        custom_tooltip = wc_magic_flavor_events.0010.c_tt
        add_dread = minor_dread_gain
        add_tyranny = minor_tyranny_gain

        get_potion_effect_0010 = yes

        stress_impact = {
            compassionate = medium_stress_impact_gain
            just = medium_stress_impact_gain
            callous = minor_stress_impact_loss   
            arbitrary = minor_stress_impact_loss   
            sadistic = minor_stress_impact_loss  
        }  

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_sociability = -1
                ai_boldness = 0.5
            }
        } 
    }

    after = {
        hidden_effect = {
            scope:merchant = { silent_disappearance_effect = yes }
        }
    }
}

# You come across ley-crystals 
wc_magic_flavor_events.0020 = {
    type = character_event
	content_source = dlc_GOA
    theme = magic
    title = wc_magic_flavor_events.0020.t
    desc = wc_magic_flavor_events.0020.desc

    override_background = {
		reference = wilderness
	}

    cooldown = { years = 10 }

    trigger = {
        location = {
            OR = {
                geographical_region = world_broken_isles
                geographical_region = world_eastern_kingdoms_quelthalas
                geographical_region = world_eastern_kingdoms_azeroth_black_morass
                geographical_region = world_northrend_storm_peaks
                geographical_region = world_northrend_borean_tundra
                geographical_region = world_northrend_howling_fjord
                geographical_region = world_pandaria_vale_of_eternal_blossoms
                geographical_region = world_kalimdor_south_tanaris
                geographical_region = world_kalimdor_south_silithus
                geographical_region = world_kalimdor_north_hyjal
                # dragon isles
            }
            OR = {
                county.holder = root
                county.holder.top_liege = root
                county.holder.liege = root
            }
        }
    }

    weight_multiplier = {
		base = 25
		modifier = { 
			add = {
				if = {
					limit = { 
						var:wc_current_mana_variable <= wc_twenty_percent_of_max_mana_value
					}
					add = 50
				}
			}
        }
        modifier = { 
			add = {
				if = {
					limit = { 
						var:wc_current_mana_variable <= wc_fourty_percent_of_max_mana_value
					}
					add = 25
				}
			}
        }
    }

    left_portrait = {
 		character = root
        animation = lantern
	}

    # Harvest every crystal we can find
    option = {
        name = wc_magic_flavor_events.0020.a

        add_short_term_gold = massive_gold_value

        stress_impact = {
            greedy = minor_stress_impact_loss
            avaricious = minor_stress_impact_loss
        }

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_greed = 1.5
            }
            modifier = { # I lowkey need money
                factor = 2
                short_term_gold <= medium_gold_value
            }
            modifier = { # I lowkey need money
                factor = 3
                short_term_gold <= minor_gold_value
            }
        }
    }

    # Mine them over time
    option = {
        name = wc_magic_flavor_events.0020.b

        stress_impact = {
            administrator = minor_stress_impact_loss
            architect = minor_stress_impact_loss
        }

        location.county = {
            add_county_modifier = {
                modifier = wc_mining_leyline_crystals
                years = 5
            }
        }

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_rationality = 1
                ai_greed = 1
            }
            modifier = {
                factor = 2
                has_lifestyle = stewardship_lifestyle
            }
            modifier = {
                factor = 2
                OR = {
                    has_trait = administrator
                    has_trait = architect
                }
            }
        }
    }

    # Study them
    option = {  
        name = wc_magic_flavor_events.0020.c

        stress_impact = {
            scholar = tiny_stress_impact_loss
        }

        location.county = {
            add_county_modifier = {
                modifier = wc_studying_leyline_crystals
                years = 3
            }
            custom_tooltip = wc_magic_flavor_events.0020.c_tt

            trigger_event = {
                on_action = wc_studying_leyline_county_pulse
            }
        }

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_rationality = 1.5
            }
            modifier = {
                factor = 2
                has_lifestyle = learning_lifestyle
            }
            modifier = {
                factor = 2
                has_trait = scholar
            }
        }
    }

    # Absorb the crystals myself
    option = {
        name = wc_magic_flavor_events.0020.d

        wc_change_mana_by_value_effect = {
            CHANGE = add
            VALUE = minor_mana_value
        }

        wc_change_amount_mp_effect = {
            CHANGE = add
            VALUE = 5
        }

        add_character_modifier = {
            modifier = wc_absorbed_leyline_crystals
            years = 3
        }

        stress_impact = {
            arrogant = tiny_stress_impact_loss
            paranoid = minor_stress_impact_gain
        }

        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_greed = 1
                ai_boldness = 1
            }
            modifier = {
                factor = 2
                has_trait = arrogant
            }
            modifier = {
                factor = 2
                var:wc_current_mana_variable <= wc_twenty_percent_of_max_mana_value
            }
            modifier = {
                factor = 1
                var:wc_current_mana_variable <= wc_fourty_percent_of_max_mana_value
            }
        }
    }
}
