
get_spell_class_event_effect = {
    if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = Order SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:Order
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = disorder SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:disorder
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = light SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:light
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = shadow SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:shadow
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = death SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:death
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = life SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:life
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = elemental_water SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:elemental_water
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = elemental_fire SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:elemental_fire
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = elemental_air SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:elemental_air
        }
    }
    else_if = {
        limit = {
            is_in_spell_class_trigger_scoped = { CLASS = elemental_earth SPELL = $SPELL$ }
        }
        set_variable = {
            name = current_spell_class
            value = flag:elemental_earth
        }
    }
    else = {
        set_variable = {
            name = current_spell_class
            value = flag:unknown
        }
    }
}

cast_spell_event_single_character_effect = {
    set_variable = {
        name = real_caster
        value = root
    }
    # set_variable = { # Already set
    #     name = current_spell_class
    #     value = flag:$CLASS$
    # }
    set_variable = {
        name = current_spell_name
        value = $SPELL$
    }
    set_variable = {
        name = current_spell_rank
        value = $RANK$
    }
    set_variable = {
        name = spell_recipient
        value = $TARGET$
    }
    if = {
        limit = {
            exists = var:current_spell_name
        }
        set_spell_target_type_effect = yes 
        execute_scoped_spell_effect = {
            EFFECT = spell_cast_effect
            SPELL = var:current_spell_name
        }
    }
    #remove_magic_variables = yes
}

cast_spell_event_single_character_effect_elemental = {
    set_variable = {
        name = real_caster
        value = root
    }
    set_variable = {
        name = elemental_type
        value = $ELEMENTAL_TYPE$
    }
    # set_variable = {
    #     name = current_spell_class
    #     value = flag:$CLASS$
    # }
    set_variable = {
        name = current_spell_name
        value = flag:$SPELL$
    }
    set_variable = {
        name = current_spell_rank
        value = $RANK$
    }
    set_variable = {
        name = spell_recipient
        value = $TARGET$
    }
    set_spell_target_type_effect = yes 
    execute_scoped_spell_effect = {
        EFFECT = spell_cast_effect
        SPELL = var:current_spell_name
    }
    remove_magic_variables = yes
}

wc_health_events_get_helpful_spell_effect = {
    if = {
        limit = {
            any_known_spell_can_cast_on_character_trigger = { TARGET = root TRIGGER = spell_is_helpful_against_illness_trigger }
        }
        random_in_list = {
            variable = known_spells
            limit = {
                save_temporary_scope_as = this_spell
                root = {
                    spell_is_helpful_against_illness_trigger = { SPELL = scope:this_spell }
                    can_cast_spell_on_character_trigger = { TARGET = root SPELL = scope:this_spell }
                }
            }
            save_scope_as = helpful_spell
        }
        set_variable = {
            name = helpful_spell
            value = scope:helpful_spell
        }
    }
}

wc_health_events_cast_helpful_spell_effect = {
    get_spell_class_event_effect = { SPELL = scope:helpful_spell }
    if = {
        limit = {
            var:helpful_spell != flag:conjure_refreshment
        }
        cast_spell_event_single_character_effect = {
            SPELL = scope:helpful_spell
            RANK = 0
            TARGET = root
        }
    }
    else_if = {
        limit = {
            var:helpful_spell = flag:conjure_refreshment
        }
        cast_spell_event_single_character_effect = {
            SPELL = scope:helpful_spell
            RANK = 1
            TARGET = root
        }
    }
    remove_variable = helpful_spell
}
