# Legion attention (road of power) and Kirin Tor standing (road of discipline)
# Used by the wandering arcanist pack: events/magic_events/wc_wandering_arcanist_events.txt

wc_legion_attention_min_value = { value = 0 }
wc_legion_attention_max_value = { value = 100 }
wc_legion_attention_noticed_value = { value = 25 }
wc_legion_attention_hunted_value = { value = 50 }
wc_legion_attention_marked_value = { value = 80 }
wc_legion_attention_reset_value = { value = 40 }
wc_legion_attention_decay_value = { value = -5 }

# Safe reader: a missing variable spams error.log, so default to 0
wc_legion_attention_current_value = {
	value = 0
	if = {
		limit = { has_variable = wc_legion_attention }
		add = var:wc_legion_attention
	}
}

# Gain from one completed Order-class cast; root is the caster inside wc_on_spell_cast_finish
wc_legion_attention_cast_gain_value = {
	value = 0
	if = {
		limit = { has_variable = casted_rank }
		add = var:casted_rank
	}
	max = 3
	if = {
		limit = {
			has_variable = casted_spell
			spell_is_ritual_trigger = { SPELL = var:casted_spell }
		}
		add = 2
	}
	if = {
		limit = { is_ai = yes }
		multiply = 0.5
		floor = yes
	}
}

	}
}
