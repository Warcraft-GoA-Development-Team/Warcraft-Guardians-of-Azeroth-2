
# Base gate shared by every Secular Faith event.
pam_secular_faith_base_trigger = {
	has_pam_dlc_trigger = yes
	is_available_adult = yes
}

# The default ambient actor.
pam_secular_faith_landed_actor_trigger = {
	pam_secular_faith_base_trigger = yes
	is_landed_or_landless_administrative = yes
}

# As above, but also open to clergy rulers (visions, scripture, hermits, etc. make sense for them too).
pam_secular_faith_clergy_ok_actor_trigger = {
	pam_secular_faith_base_trigger = yes
	is_landed_or_landless_administrative_or_theocratic = yes
}

# Christian-primary convenience gate.
pam_secular_faith_christian_actor_trigger = {
	pam_secular_faith_landed_actor_trigger = yes
	religion = religion:christianity_religion
}

pam_can_adopt_forbidden_personal_tenet_trigger = {
	any_desired_tenet = {
		only_positive = yes
		threshold = 1
		selection_count = 97
		save_temporary_scope_as = pam_sf_forbidden_candidate
		NOT = { root = { any_personal_tenet = { this = scope:pam_sf_forbidden_candidate } } }
		root.faith.main_rite = {
			has_tenet_status = {
				tenet = scope:pam_sf_forbidden_candidate
				status = prohibited
			}
		}
	}
}


pam_sf_carnal_art_indian_setting_trigger = {
	OR = {
		like_heritage_indo_aryan_trigger = yes
		like_heritage_dravidian_trigger = yes
		location = { geographical_region = world_india }
	}
}

pam_poverty_tenets_trigger = {
	OR = {
		this = tenet:tenet_mendicant_preachers
		this = tenet:tenet_vows_of_poverty
		this = tenet:tenet_asceticism
		this = tenet:tenet_sanctity_of_nature
		this = tenet:tenet_communal_possessions
		this = tenet:tenet_philanthropia
	}
}
