education_light_magic = {
    type = education

    is_valid = { 
        MAGIC_is_accepted_or_approved_trigger = { MAGIC = light }
    }

    desc = {
		desc = education_light_magic_desc
		first_valid = {
			triggered_desc = {
				trigger = { has_magic_education_affinity_childhood_trait_trigger = yes }
				desc = education_magic_tt.affinity
			}
			triggered_desc = {
				trigger = { has_magic_education_disaffinity_childhood_trait_trigger = yes }
				desc = education_magic_tt.disaffinity
			}
		}
        first_valid = {
            triggered_desc = {
				trigger = { has_light_magic_education_affinity_childhood_trait_trigger = yes }
				desc = education_light_magic_tt.affinity
			}
			triggered_desc = {
				trigger = { has_light_magic_education_disaffinity_childhood_trait_trigger = yes }
				desc = education_light_magic_tt.disaffinity
			}
        }
		triggered_desc = {
			trigger = {
				liege = { 
					OR = { 
						employs_court_position = court_tutor_court_position
						employs_court_position = court_guru_court_position
					}
				}
			}
			desc = education_any_tt.court_tutor
		}
        triggered_desc = {
			trigger = {
				liege = { 
					employs_court_position = court_caster_court_position
				}
			}
			desc = education_any_tt.court_caster
		}
		desc = line_break
	}

	is_good_for = {
		has_light_magic_education_affinity_childhood_trait_trigger = yes
	}

	is_bad_for = {
		has_light_magic_education_disaffinity_childhood_trait_trigger = yes
	}
	
	on_change_to = {
		if = {
			limit = {
				exists = var:education_point_gain_variable
			}
			change_variable = {
				name = education_point_gain_variable
				multiply = point_conversion_factor
			}
			set_variable = {
				name = education_magic_variable
				value = var:education_point_gain_variable
			}
			remove_variable = education_point_gain_variable
		}
	}
	
	on_change_from = {
		if = {
			limit = {
				is_alive = yes
				is_adult = no
			}
			add_character_flag = light_magic_previous_education
			if = {
				limit = {
					has_variable = education_magic_variable
				}
				set_variable = {
					name = education_point_gain_variable
					value = var:education_magic_variable
				}
				remove_variable = education_magic_variable
			}
			else = {
				set_variable = {
					name = education_point_gain_variable
					value = 0
				}
			}
			if = {
				limit = {
					any_relation = {
						type = guardian
						NOT = { is_parent_of = root }
					}
				}
				save_scope_as = ward
				random_relation = {
					type = guardian
					trigger_event = childhood_education.0004
				}
			}
		}
	}

	auto_selection_weight = {
		value = 200
		if = {
			limit = { has_light_magic_education_affinity_childhood_trait_trigger = yes }
			add = 900
		}
		if = {
			limit = { has_light_magic_education_disaffinity_childhood_trait_trigger = yes }
			add = -900
		}
	}
		
	modifier = {
	}
}
