single_combat_death_effect = {
	if = {
		limit = { scope:sc_victor = { can_use_frostmourne_to_twist_soul_trigger = yes } }
		raise_undead_by_depose_effect = { MASTER = scope:sc_victor }
	}
	else = {
		override_death_killer_effect = { death_reason = death_duel killer = scope:sc_victor }
	}
}

demon_hunter_slays_demon_effect = {
	if = {
		limit = {
			scope:sc_loser = {
				has_trait = being_demon
			}
		}
		if = {
			limit = {
				scope:sc_victor.has_trait = demon_hunter
				scope:sc_loser.has_trait = being_demon
			}
			add_trait_xp = {
				trait = demon_hunter
				value = demon_hunter_xp_gain_minor_value
			}
		}
	}
}
