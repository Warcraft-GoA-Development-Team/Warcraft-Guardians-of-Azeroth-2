can_be_councillor_basics_trigger = {
	is_available_quick = {
		adult = yes
		incapable = no
		imprisoned = no
	}
	is_at_war_with_liege = no
	trigger_if = {
		limit = { has_variable = block_hire_councillor }
		NOT = { var:block_hire_councillor = liege }
	}
	custom_tooltip = {
		text = not_travel_option_character
		NOT = { has_character_flag = travel_option_added_character }
	}
}

can_be_councillor_gender_trigger = {
	OR = {
		is_ruler = yes
		has_variable = can_be_councillor_override
		AND = {
			is_female = yes
			$COURT_OWNER$.rite = { NOT = { rite_has_parameter = male_dominated_council } }
		}
		AND = {
			is_female = yes
			$COURT_OWNER$ = {
				has_personal_tenet_flag = tenet_hyperdulia_women_can_be_councillors
			}
		}
		AND = {
			is_male = yes
			$COURT_OWNER$.rite = { NOT = { rite_has_parameter = female_dominated_council } }
		}
		# Warcraft - TODO: Find a better workaround to this
		# Exception so Katrana Prestor can be councilor
		AND = {
			exists = character:58050
			this = character:58050
		}
	}
}

can_be_chancellor_trigger = {
	can_be_councillor_basics_trigger = yes

		rite = { rite_has_parameter = clerical_appointment_fixed }

			this = liege.diarch

		}
	}
	can_be_councillor_gender_trigger = { COURT_OWNER = $COURT_OWNER$ }

can_be_court_chaplain_trigger = {
	save_temporary_scope_as = court_chaplain_check
	can_be_councillor_basics_trigger = yes
	$COURT_OWNER$ ?= {
		NOT = { government_has_flag = government_is_nomadic }
	}
	NOR = {
		has_council_position = councillor_spouse
		AND = {
			this = liege.diarch
			liege = { has_diarchy_active_parameter = diarchy_is_vizierate }
		}
	}
	$COURT_OWNER$.rite = { rite_has_allowed_gender_for_clergy = scope:court_chaplain_check }
	rite = $COURT_OWNER$.rite
	trigger_if = {
		limit = {
			rite = { rite_has_doctrine = doctrine_theocracy_temporal }
		}
		OR = {
			is_ruler = no
			government_has_flag = government_is_theocracy
		}
	}
	trigger_if = {
		limit = {
			exists = var:escaped_imprisonment_from
		}
		NOT = { var:escaped_imprisonment_from = $COURT_OWNER$ }
	}
	trigger_if = { # An excommunicated ruler tolerate excommunicated bishops
		limit = {
			$COURT_OWNER$ ?= { NOT = { has_trait = excommunicated } }
		}
		NOT = { has_trait = excommunicated }
	}
	trigger_if = { # Married characters are excluded if the doctrine is not there
		limit = {
			rite = {
				NOT = { rite_has_parameter = clergy_can_marry }
			}
		}
		is_married = no
	}
}

	weighted_calc_true_if = {

		20 = {
			NOT = { has_trait = ambitious }
		}
		20 = {
			opinion = {
				target = liege
				value > 0
			}
		}
		20 = {
			opinion = {
				target = liege
				value > 20
			}
		}
		20 = {
			opinion = {
				target = liege
				value > 40
			}
		}
		20 = {
			opinion = {
				target = liege
				value > 60
			}
		}
	}
}

	AND = {
		NOT = {
			has_diarchy_type = vizierate
		}
		any_spouse = {
			OR = {
				court_owner = root
				AND = {
					is_ruler = yes

				}
			}
		}
	}
	NOT = { government_has_flag = government_is_landless_adventurer }
	# Celestial hegemons have access to the ministry instead, so no spouse
	tgp_has_access_to_ministry_trigger = no
}

valid_spouse_councillor_trigger = {
	scope:councillor = { is_available = yes }
	cp:councillor_spouse ?= scope:councillor

is_blocked_from_being_fired_from_council_trigger = {
	custom_description = {
		text = "is_blocked_from_being_fired_from_council"
		subject = this
		object = $COURT_OWNER$

		OR = {
			# Check in code if the court owner is prevented from firing the councilor
			# can_fire_position must take a council task as input, but checks the councilor behind the task.
			council_task ?= { can_fire_position = no }
			# Check for variable preventing the firing of a councilor set by script
			var:block_fire_councillor ?= $COURT_OWNER$
		}
	}
}

can_be_fired_from_council_trigger = {
	NOT = {
		is_blocked_from_being_fired_from_council_trigger = { COURT_OWNER = $COURT_OWNER$ }
	}
	trigger_if = {
		limit = {
			has_council_position = councillor_court_chaplain
			liege = { tgp_has_access_to_ministry_trigger = no }
		}
		rite = {
			OR = {
				rite_has_doctrine = doctrine_clerical_succession_temporal_appointment
				rite_has_doctrine = doctrine_clerical_succession_spiritual_appointment
			}
		}
	}
}

		}
