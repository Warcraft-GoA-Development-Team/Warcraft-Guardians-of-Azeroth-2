ai_spell_modifier_value = { # Which spell should I pick?
    ### RELATIONS ###
    if = {
        limit = {
            spell_is_harmful_trigger = { SPELL = var:current_spell_name }
            var:real_caster = { 
                OR = {
                    has_any_negative_relation_trigger = yes
                    is_at_war = yes
                }
            }
        }
        add = 50
    }
    if = {
        limit = {
            spell_is_helpful_trigger = { SPELL = var:current_spell_name }
            var:real_caster = { has_any_positive_relation_trigger = yes }
        }
        add = 50
    }
    ### PERSONALITY ###
    if = {
        limit = {
            spell_is_harmful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_personality_malicious_trigger = yes } 
        }
        add = 100
    }
    if = {
        limit = {
            spell_is_helpful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_personality_benevolent_trigger = yes } 
        }
        add = 100
    }
    if = {
        limit = {
            spell_is_harmful_trigger = { SPELL = var:current_spell_name }
            var:real_caster = { has_trait = sadistic } 
        }
        add = 25
    }
    ### MAGIC MODIFIERS ###

    ### EVENT MODIFIERS ###

    ### SPELL SPECIFIC MODIFIERS ###
    if = {
        limit = {
            var:current_spell_name = flag:salvation
            NOT = { has_good_salvation_target_trigger = yes }
        }
        add = -250
    }
    # Shouldn't choose raise dead unless someone actually died (funeral activity?)
}

ai_spell_target_character_modifier_value = {
    ### RELATIONS ###
    if = {
        limit = {
            spell_is_harmful_trigger = { SPELL = var:current_spell_name }
            spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
        }
        add = {
            value = "opinion(var:spell_recipient)"
            multiply = -5.5
        }
    }
    if = {
        limit = {
            spell_is_helpful_trigger = { SPELL = var:current_spell_name }
            spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
        }
        add = {
            value = "opinion(var:spell_recipient)"
            multiply = 5.5
        }
    }
    if = {
        limit = {
            spell_is_harmful_trigger = {
                SPELL = var:current_spell_name
            }
            var:spell_recipient = { is_vassal_or_below_of = var:real_caster }
        }
        add = -75
    }
    if = {
        limit = {
            spell_is_helpful_trigger = {
                SPELL = var:current_spell_name
            }
            var:spell_recipient = { is_vassal_or_below_of = var:real_caster }
        }
        add = 75
    }
    if = {
        limit = {
            var:real_caster = { is_of_major_interest_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 50
    }
    if = {
        limit = {
            var:real_caster = { is_of_minor_interest_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 25
    }
    if = {
        limit = {
            var:spell_recipient = { has_no_real_status_trigger = yes }
        }
        add = -75
    }

    if = {
        limit = {
            spell_is_helpful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_any_mild_good_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 10
    }
    if = {
        limit = {
            spell_is_helpful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 25
    }
    if = {
        limit = {
            spell_is_helpful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_any_best_good_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 50
    }

    if = {
        limit = {
            spell_is_harmful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_any_mild_bad_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 10
    }
    if = {
        limit = {
            spell_is_harmful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_any_moderate_bad_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 25
    }
    if = {
        limit = {
            spell_is_harmful_trigger = {
                SPELL = var:current_spell_name
            }
            var:real_caster = { has_any_worst_bad_relationship_with_character_trigger = { CHARACTER = var:spell_recipient } }
        }
        add = 50
    }

    # If the spell is helpful and they've targeted you with a helpful spell before, more likely to cast on them and vise versa
    # TODO_MAGIC_AI: Don't cast on people more powerful than you, scaling, depending on your personality
    # TODO_MAGIC_AI: Scaling per relation & general modifier - war targets/allies as well
    # TODO_MAGIC_AI: More likely to cast rituals on themselves or their vassals
    # TODO_MAGIC_AI: Less likely to on people outside of realm in general, scaling with distance
    # TODO_MAGIC_AI: Don't cast harmful spells on yourself unless you have flag trait

    ### PERSONALITY ###
    if = {
        limit = {
            spell_is_harmful_trigger = { SPELL = $SPELL$ }
            var:real_caster = { has_trait = flagellant } 
            var:real_caster = var:spell_recipient
        }
        add = 100
    }
    
    ### MAGIC MODIFIERS ###
    # If someone has a bunch of traits or magic modifiers try not to cast too often on them

    ### EVENT MODIFIERS ###

    ### SPELL SPECIFIC MODIFIERS ###
    if = {
        limit = {
            var:spell_recipient = {
                OR = {
                    is_magic_type_criminal_trigger = { MAGICTYPE = shadow }
                    is_magic_type_shunned_trigger = { MAGICTYPE = shadow }
                }
            }
            var:current_spell_name = flag:void_leader
            spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
        }
        add = -250
        
    }
    if = {
        limit = {
            is_good_salvation_target_trigger = { TARGET = var:spell_recipient }
            var:current_spell_name = flag:salvation
            spell_target_is_character_trigger = { SPELL = var:current_spell_name RANK_VALUE = var:current_spell_rank }
        }
        add = 250
    }
}
