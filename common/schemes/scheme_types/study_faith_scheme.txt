	# Base Chances
	base_success_chance = {
		base = 0
		
		modifier = {
			add = {
				value = scope:owner.learning
				multiply = 2
			}
			desc = SCHEME_LEARNING_MODIFIER
			scope:owner.learning > 0
		}
		
		modifier = {
			add = {
				value = scope:owner.piety_level
				multiply = 3
			}
			desc = SCHEME_PIETY_LEVEL_MODIFIER
			scope:owner.piety_level > 0
		}

		#Trait and Perk modifiers 

		#Intellect good
		modifier = {
			desc = "scheme_intellect_good_1"
			scope:owner = { has_trait = intellect_good_1 }
			add = 10
		}
		
		modifier = {
			desc = "scheme_intellect_good_2"
			scope:owner = { has_trait = intellect_good_2 }
			add = 20
		}
		
		modifier = {
			desc = "scheme_intellect_good_3"
			scope:owner = { has_trait = intellect_good_3 }
			add = 30
		}
		#Ambitious
		modifier = {
			scope:owner = { has_trait = ambitious }
			desc = "SCHEME_TRAIT_AMBITIOUS"
			add = 5
		}
		#Theologian
		modifier = {
			scope:owner = { has_trait = theologian }
			desc = "STUDY_FAITH_THEOLOGIAN_PERK_DESC"
			add = 10
		}
		#Erudite
		modifier = {
			scope:owner = { has_trait = erudite }
			desc = "SCHEME_TRAIT_ERUDITE"
			add = 10
		}
		#Witch
		modifier = {
			scope:owner = { has_trait = witch }
			desc = "SCHEME_TRAIT_WITCH"
			add = 5
		}
		#Mystic
		modifier = {
			scope:owner = { has_trait = lifestyle_mystic }
			desc = "SCHEME_TRAIT_MYSTIC"
			add = 10
		}
		#Devoted (monk)
		modifier = {
			scope:owner = { has_trait = devoted }
			desc = "SCHEME_TRAIT_DEVOTED"
			add = 10
		}
		#Scholar (leveled trait)
		modifier = {
			scope:owner = { has_trait = lifestyle_scholar }
			desc = "SCHEME_TRAIT_SCHOLAR"
			add = {
				value = "scope:owner.has_trait_xp(lifestyle_scholar)"
				divide = 2
			}
		}
		#Open Minded
		modifier = {
			scope:owner = { has_perk = open_minded_perk }
			desc = "SCHEME_PERK_OPEN_MINDED"
			add = 10
		}
		#More difficult if you don't know their language
		modifier = {
			NOT = {
				scope:owner = { knows_language_of_character = scope:target }
			}
			desc = "SCHEME_DIFFERENT_LANGUAGE_MODIFIER"
			add = -15

		#Paranoid value is added automatically because of trait	parameter enemy_personal_scheme_success_chance_add

	# On Actions
	on_phase_completed = {
		# Grab our scopes.
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = { save_scope_as = owner }
		scope:scheme = {
			random = {
				chance = scope:scheme.scheme_success_chance

				save_scope_value_as = {
					name = scheme_successful
					value = yes
				}
			}
			scope:owner = {
				if = {
					limit = {
						exists = scope:scheme_successful
					}
					trigger_event = {
						on_action = study_faith_success
					}
				}
				else = {
					trigger_event = {
						on_action = study_faith_failure
					}	
				}
			}
		}
	}
	on_monthly = {
		save_scope_as = scheme
		scheme_target_character = { save_scope_as = target }
		scheme_owner = {
			save_scope_as = owner
			trigger_event = { on_action = study_faith_ongoing }
		}
	}
	on_invalidated = {
		scheme_target_character = {
			save_scope_as = target
		}
		scheme_owner = {
			save_scope_as = owner
		}

		if = {
			limit = {
				scope:target = { is_alive = no }
			}
			scope:owner = {
				send_interface_toast = {
					title = study_faith_invalidated
					custom_tooltip = study_faith_invalidated_dead_desc
					left_icon = scope:target
				}
			}
		}
		else_if = {
			limit = {
				scope:owner.rite = scope:target.rite
			}
			scope:owner = {
				send_interface_toast = {
					title = study_faith_invalidated
					custom_tooltip = study_faith_invalidated_cannot_study_faith_desc
					left_icon = scope:target
				}
			}
		}

		if = {
			limit = {
				scope:target = { 
					NOT = { in_diplomatic_range = scope:owner }
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = study_faith_invalidated_title
					left_icon = scope:target
					custom_description_no_bullet = {
						object = scope:target
						text = scheme_target_not_in_diplomatic_range
					}
				}
			}
		}
		if = {
			limit = {
				scope:target = { is_imprisoned = yes }
			}
			scope:target.imprisoner = {
				save_scope_as = other_imprisoner
			}
			scope:owner = {
				send_interface_toast = {
					title = study_faith_invalidated_title
					left_icon = scope:target
					right_icon = scope:other_imprisoner
					custom_description_no_bullet = {
						subject = scope:other_imprisoner
						object = scope:target
						text = scheme_target_imprisoned_by_other
					}
				}
			}
		}
		
		# invalidated by invalid target but can still study the rite
		scope:owner = {
			if = {
				limit = {
					is_alive = yes
					scope:target = {
						OR = {
							is_alive = no
							NOT = { in_diplomatic_range = scope:owner }
							is_imprisoned = yes
						}
					}
					rite != scope:target.rite
				}
				set_variable = {
					name = study_faith_progress
					value = prev.scheme_progress
				}
				trigger_event = study_faith_outcome.3000
			}
		}
	}
