wc_on_ai_spellcasting = {
    trigger = { 
        is_magic_user_trigger = yes
    }
    effect = { # Stage 1
        # Can we even cast a spell?
        if = {
            limit = {
                is_ai = yes
                is_any_spell_castable_trigger = yes 
            }
            decide_to_cast_spell_effect = yes
            set_variable = ai_cast_1
            if = {
                limit = {
                    debug_only = yes 
                }
                change_global_variable = {
                    name = ai_spells_pre_cast_stage_1
                    add = 1
                }
            }
        }
        else_if = {
            limit = {
                is_ai = yes
            }
            set_variable = ai_cast_1
            # Come back later
            trigger_event = {
                on_action = wc_on_ai_spellcasting
                days = { 1 30 } # So it will not cause a major lag on every 1st of each month
            }
        }
        else_if = { # Character is a player now
            limit = {
                is_ai = no
            }
            remove_variable = ai_cast_1
            trigger_event = {
                on_action = wc_on_ai_spellcasting
                days = 500 # in case they become an AI 
            }
        }
    }
}
