# Authority over realm-wide faith - vassals convert or pay; real opinion penalties

	picture = { reference = "gfx/interface/illustrations/decisions/the_scarlet_crusade.dds" }
	decision_group_type = decisions

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 12
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	desc = balnazzar_demand_purity_tribute_decision_desc

	is_shown = {
		has_character_flag = wc_balnazzar_disguise_active
		# Only after Crusade founded (.1006) - can't demand tribute for a non-existent creed
		has_character_flag = has_formed_the_scarlet_crusade
		any_vassal = {
			count > 0
			NOT = { faith = faith:scarletism }
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = balnazzar_demand_purity_tribute_decision.cooldown_desc
			NOT = { has_character_flag = balnazzar_demanded_purity_tribute_cooldown }
		}
		is_imprisoned = no
		is_incapable_trigger = no
	}

	effect = {
		add_gold = medium_gold_value
		add_piety = minor_piety_value
		add_character_flag = { flag = balnazzar_demanded_purity_tribute_cooldown years = 3 }

		every_vassal = {
			limit = { NOT = { faith = faith:scarletism } }
			add_opinion = { modifier = wc_purity_tribute_demanded target = root }
			random = {
				chance = 30
				set_character_faith_with_conversion = faith:scarletism
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 20
	}
}
