spiritual_counsel_interaction = {
	category = interaction_category_religion
	icon = icon_request_priest
	interface_priority = 35
	common_interaction = no

	use_diplomatic_range = yes
	ai_maybe = no
	popup_on_receive = yes

	desc = spiritual_counsel_interaction_desc
	greeting = positive
	notification_text = spiritual_counsel_notification_text

	cost = {
		piety = {
			value = scope:actor.medium_piety_value
			if = {
				limit = {
					scope:actor = {
						domicile ?= { has_domicile_parameter = estate_cheaper_spiritual_counsel }
					}
				}
				multiply = ecclesiastical_domicile_cheaper_spiritual_counsel_mult
			}
		}
		treasury_or_gold = {
			value = scope:actor.medium_gold_value
			if = {
				limit = {
					scope:actor = {
						domicile ?= { has_domicile_parameter = estate_cheaper_spiritual_counsel }
					}
				}
				multiply = ecclesiastical_domicile_cheaper_spiritual_counsel_mult
			}
		}
	}

	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 10 }

	is_available = {
		OR = {
			government_has_flag = government_is_theocracy
			rite ?= {
				rite_has_parameter = gnosticism_cleric_interactions
			}
		}
		trigger_if = {
			limit = { is_ai = yes }
			piety >= medium_piety_value
			short_term_gold >= medium_gold_value
			NOT = { has_trait = cynical }
		}
	}

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = {
			NOT = { government_has_flag = government_is_theocracy }
			faith ?= scope:actor.faith
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			piety_level >= 1
			is_available_adult = yes
			NOR = {
				is_at_war_with = scope:recipient
				has_usable_hook = scope:recipient
			}
			#Gnostic seculars can do this with high learning
			trigger_if = {
				limit = {
					NOT = {
						government_has_flag = government_is_theocracy
					}
				}
				learning >= very_high_skill_rating
			}
		}
		scope:recipient = {
			is_available_adult = yes
		}
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = spiritual_counsel_interaction_accept_actor_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				add_hook = {
					type = spiritual_counsel_hook
					target = scope:recipient
				}
			}
		}
		scope:recipient = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = spiritual_counsel_interaction_accept_recipient_toast
				left_icon = scope:recipient
				right_icon = scope:actor
				add_piety = {
					value = minor_piety_value
					add = {
						value = miniscule_piety_value
						multiply = scope:actor.piety_level
					}
					if = {
						limit = { religious_head_or_challenger ?= scope:actor }
						add = minor_piety_value
					}
					if = {
						limit = { scope:actor.rite = { rite_has_parameter = tenet_confession_spiritual_counsel_bonus } }
						add = miniscule_piety_value
					}
					if = {
						limit = { scope:actor.rite = { rite_has_tenet = tenet_philanthropia } }
						add = miniscule_piety_value
					}
					if = {
						limit = { scope:actor.rite = { rite_has_tenet = tenet_love_thy_neighbour } }
						add = miniscule_piety_value
					}
					if = {
						limit = { scope:actor.rite = { rite_has_tenet = tenet_communion } }
						add = miniscule_piety_value
					}
					if = {
						limit = { scope:actor.rite = { rite_has_tenet = tenet_scholasticism } }
						add = miniscule_piety_value
					}
					if = {
						limit = { scope:actor.rite = { rite_has_parameter = tenet_hesychasm_spiritual_counsel_bonus } }
						add = miniscule_piety_value
					}
				}
				if = {
					limit = { is_playable_character = yes }	# change_spiritual_fulfillment errors on non-playable characters
					change_spiritual_fulfillment = {
						value = miniscule_spiritual_fulfillment_value
						add = scope:actor.piety_level
						if = {
							limit = { religious_head_or_challenger ?= scope:actor }
							add = 2
						}
						if = {
							limit = { scope:actor.rite = { rite_has_parameter = tenet_confession_spiritual_counsel_bonus } }
							add = 2
						}
						if = {
							limit = { scope:actor.rite = { rite_has_tenet = tenet_philanthropia } }
							add = 1
						}
						if = {
							limit = { scope:actor.rite = { rite_has_tenet = tenet_love_thy_neighbour } }
							add = 1
						}
						if = {
							limit = { scope:actor.rite = { rite_has_tenet = tenet_communion } }
							add = 2
						}
						if = {
							limit = { scope:actor.rite = { rite_has_tenet = tenet_scholasticism } }
							add = 1
						}
						if = {
							limit = { scope:actor.rite = { rite_has_parameter = tenet_hesychasm_spiritual_counsel_bonus } }
							add = 1
						}
					}
				}
				stress_impact = {
					base = medium_stress_loss
					cynical = minor_stress_gain
					zealous = minor_stress_loss
					humble = minor_stress_loss
				}
				add_opinion = {
					target = scope:actor
					modifier = received_spiritual_counsel_opinion
				}
			}
		}
	}

	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = spiritual_counsel_interaction_decline_actor_toast
				right_icon = scope:recipient
				add_opinion = {
					target = scope:recipient
					modifier = spurned_spiritual_counsel_opinion
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = neighboring_rulers_including_tributary_borders
		ai_recipients = rulers_in_clerical_region
		ai_recipients = scripted_relations
	}

	ai_targets = {
		ai_recipients = vassals
		max = 5
	}

	ai_targets = {
		ai_recipients = peer_vassals
		max = 10
	}

	ai_target_quick_trigger = {
		adult = yes
	}

	ai_frequency_by_tier = {
		barony = 0
		county = 200
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	ai_will_do = {
		base = 25
		
		modifier = {
			add = 100
			scope:actor = {
				has_trait = greedy
			}
			scope:recipient = {
				OR = {
					has_character_modifier = clerical_chastity_vow_modifier
					NOT = {
						any_child = {
							is_alive = yes
						}
					}
				}
			}
		}
		
		modifier = {
			add = 25
			recipient_in_actor_clerical_region_trigger = yes
		}

		modifier = {
			add = 25
			scope:actor = { has_trait = zealous }
		}
		ai_value_modifier = {
			who = scope:actor
			ai_zeal = 0.25
		}
		modifier = {
			add = 15
			scope:actor = { piety_level >= 3 }
		}
		modifier = {
			add = 15
			scope:actor = { piety_level >= 5 }
		}
		modifier = {
			add = 50
			scope:recipient = {
				OR = {
					is_close_or_extended_family_of = scope:actor
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = { has_relation_rival = scope:actor }
		}
	}

	ai_accept = {
		base = 0

		# Warcraft
		master_thrall_modifier = yes
		# End Warcraft

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:actor
			RECIPIENT = scope:recipient
		}

		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}

		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_1
			desc = ESTATE_SPIRITUAL_COUNSEL_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_easier_spiritual_counsel_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_2
			desc = ESTATE_SPIRITUAL_COUNSEL_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_easier_spiritual_counsel_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_3
			desc = ESTATE_SPIRITUAL_COUNSEL_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_easier_spiritual_counsel_3
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_1
			desc = ESTATE_COUNSEL_CONFESSIONAL_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_counsel_confessional_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_2
			desc = ESTATE_COUNSEL_CONFESSIONAL_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_counsel_confessional_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_minor_value_3
			desc = ESTATE_COUNSEL_CONFESSIONAL_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_counsel_confessional_3
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_1
			desc = ESTATE_CHANCERY_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_chancery_counsel_1
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_2
			desc = ESTATE_CHANCERY_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_chancery_counsel_2
				}
			}
		}
		modifier = {
			add = ecclesiastical_domicile_acceptance_micro_value_3
			desc = ESTATE_CHANCERY_REASON
			scope:actor = {
				domicile ?= {
					has_domicile_parameter = estate_chancery_counsel_3
				}
			}
		}
		ecclesiastical_domicile_cathedral_awe_modifier = yes

		ai_value_modifier = {
			who = scope:recipient
			ai_zeal = 0.5
		}

		modifier = {
			add = 25
			scope:recipient = { has_trait = zealous }
			desc = spiritual_counsel_ai_recipient_zealous
		}
		modifier = {
			add = -40
			scope:recipient = { has_trait = cynical }
			desc = spiritual_counsel_ai_recipient_cynical
		}

		modifier = {
			add = 15
			scope:actor = {
				piety_level >= 3
				piety_level < 5
			}
			desc = spiritual_counsel_ai_priest_pious
		}
		modifier = {
			add = 30
			scope:actor = { piety_level >= 5 }
			desc = spiritual_counsel_ai_priest_very_pious
		}
		modifier = {
			add = -50
			scope:actor = { piety_level <= 1 }
			desc = spiritual_counsel_ai_priest_impious
		}

		modifier = {
			add = {
				value = 5
				multiply = scope:actor.num_virtuous_traits
			}
			scope:actor = { num_virtuous_traits > 0 }
			desc = spiritual_counsel_ai_priest_virtuous
		}
		modifier = {
			add = {
				value = -30
				multiply = scope:actor.num_sinful_traits
			}
			scope:actor = { num_sinful_traits > 0 }
			desc = spiritual_counsel_ai_priest_sinful
		}

		modifier = {
			add = 30
			religious_head_or_challenger ?= scope:actor
			desc = spiritual_counsel_ai_priest_hof
		}

		modifier = {
			add = 5
			scope:actor.rite = scope:recipient.rite
			desc = spiritual_counsel_ai_same_rite
		}
		modifier = {
			add = -100
			scope:actor.rite != scope:recipient.rite
			desc = spiritual_counsel_ai_different_rite
		}

		modifier = {
			add = -20
			scope:recipient = { highest_held_title_tier = tier_duchy }
			desc = spiritual_counsel_ai_recipient_tier
		}
		modifier = {
			add = -50
			scope:recipient = { highest_held_title_tier = tier_kingdom }
			desc = spiritual_counsel_ai_recipient_tier
		}
		modifier = {
			add = -75
			scope:recipient = { highest_held_title_tier = tier_empire }
			desc = spiritual_counsel_ai_recipient_tier
		}
		modifier = {
			add = -100
			scope:recipient = { highest_held_title_tier = tier_hegemony }
			desc = spiritual_counsel_ai_recipient_tier
		}

		modifier = {
			add = 15
			scope:actor.rite = { rite_has_parameter = tenet_confession_spiritual_counsel_bonus }
			desc = spiritual_counsel_ai_priest_tenet_confession
		}
		modifier = {
			add = 10
			scope:actor.rite = { rite_has_tenet = tenet_philanthropia }
			desc = spiritual_counsel_ai_priest_tenet_philanthropia
		}
		modifier = {
			add = 10
			scope:actor.rite = { rite_has_tenet = tenet_love_thy_neighbour }
			desc = spiritual_counsel_ai_priest_tenet_love_thy_neighbour
		}
		modifier = {
			add = 10
			scope:actor.rite = { rite_has_tenet = tenet_communion }
			desc = spiritual_counsel_ai_priest_tenet_communion
		}
		modifier = {
			add = 10
			scope:actor.rite = { rite_has_tenet = tenet_scholasticism }
			desc = spiritual_counsel_ai_priest_tenet_scholasticism
		}
		modifier = {
			add = 10
			scope:actor.rite = { rite_has_parameter = tenet_hesychasm_spiritual_counsel_bonus }
			desc = spiritual_counsel_ai_priest_tenet_hesychasm
		}

		modifier = {
			add = 200
			scope:recipient = {
				has_relation_friend = scope:actor
			}
			desc = spiritual_counsel_ai_friend
		}
		modifier = {
			add = -200
			scope:recipient = { has_relation_rival = scope:actor }
			desc = spiritual_counsel_ai_rival
		}
	}
}
