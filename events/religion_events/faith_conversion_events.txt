# Fired when you convert to a new faith. Convert capital county.
faith_conversion.0001 = {
	hidden = yes

	trigger = {
		is_landed = yes
		primary_title.tier > tier_barony
		exists = capital_county
		exists = scope:old_faith
		capital_county.faith = scope:old_faith
	}

	immediate = {
		capital_county ?= {
			save_scope_as = converted_ruler_capital
		}

		# set_county_rite also sets the parent faith, so a faith change would convert the capital for free
		if = {
			limit = { capital_county.faith = root.faith }

			# if their capital and ours coincide region holder's rite wins - try to convert adjacent county root owns
			if = {
				limit = {
					capital_county.clerical_region_title ?= {
						holder ?= {
							capital_county = scope:converted_ruler_capital
							faith = root.faith
							rite != root.rite
							NOR = {
								root.puppet:theological_agent_puppet ?= this
								root.puppet:external_theological_agent_puppet ?= this
							}
						}
					}
				}

				capital_county = {
					random_neighboring_county = {
						limit = {
							holder = root
							faith = root.faith
							clerical_region_title ?= {
								OR = {
									not = { exists = holder }
									holder.rite = root.rite
									holder.capital_county != prev
								}
							}
						}

						set_county_rite = root.rite
					}
				}
			}
			else = {
				capital_county = { set_county_rite = root.rite }
			}
		}

		root.rite = {
			force_refresh_head_of_rite = yes
		}
	}
}

scripted_trigger faith_conversion_0003_valid_character = {
	faith = scope:old_faith
	# If conversion is within the same religion, check the rite is the same
	trigger_if = {
		limit = { scope:new_faith.religion = scope:old_faith.religion }
		rite = scope:old_rite
	}
	is_alive = yes
	is_ai = yes
	# A relative's conversion must not drag along characters the rest of the conversion systems protect
	NOR = {
		has_character_flag = ai_will_not_convert
		this = faith.religious_head
		rite.head_of_rite ?= this
	}
}

# Fired when you convert to a new faith. Convert close family.
faith_conversion.0003 = {
	hidden = yes

	immediate = {
		faith = { save_scope_as = new_faith }
		rite = { save_scope_as = new_rite }
		every_spouse = {
			limit = {
				faith_conversion_0003_valid_character = yes
				is_landed = no
			}
			# Warcraft
			add_to_list = family_to_convert
		}
		every_close_family_member = {
			limit = {
				faith_conversion_0003_valid_character = yes
				OR = {
					# A minor fostered outside the realm keeps their guardian's rite
					AND = {
						is_child_of = root
						OR = {
							is_adult = no
							is_landed = no
						}
						trigger_if = {
							limit = {
								is_adult = no
								any_relation = {
									type = guardian
								}
							}
							any_relation = {
								type = guardian
								count = all
								target_is_liege_or_above = root
							}
						}
					}
					# Cascade travels down the hierarchy, not up to a parent or sibling
					target_is_liege_or_above = root
				}
			}
			# Warcraft
			add_to_list = family_to_convert
		}

		# Warcraft
		save_scope_as = owner # For plague_agent_join_chance_modifier

		every_in_list = {
			list = family_to_convert

			random_list = {
				1 = {
					plague_convert_chance_modifier = yes

					set_character_rite = root.rite
				}
				# May run from evil family if doesn't support virtues
				10 = {
					trigger = {
						is_imprisoned = no
						is_ruler = no
						root = { evil_public_religion_trigger = yes }
					}
					move_to_pool = yes
				}
			}
		}
	}
}

# For softening opinion penalties, e.g., I believed in monogamous faith but was in a polygamous marriage, but now I've been converted to polygamous faith.
faith_conversion.0005 = {
	hidden = yes

	trigger = {
		any_consort = { }
	}

	immediate = {
		every_consort = {
			save_scope_as = this_consort
			root = {
				update_active_consort_opinion_effect = { PARTNER = scope:this_consort }
			}
		}
	}
}

					override_death_effect = { death_reason = death_disappearance } # Warcraft

			}
		}

				}
			}
			scope:template_priest = {
				remove_character_flag = is_naked
			}
		}
		

		add_character_flag  = {
			flag = recent_convert
			years = 20
		}
	}
}
