## pam_theocratic_ruler_events.0001 — A Bequest for the Church
############################

# Same faith landed theocrat in your archdiocese clerical region (may not be your feudal vassal).
# Scope: county holder. root: the theocratic ruler.
scripted_trigger pam_theocratic_ruler_0001_clerical_region_priest_trigger = {
	this != root
	is_available = yes
	faith = root.faith
	government_has_flag = government_is_theocracy
	is_landed = yes
}

# Suffragans use plain theocracy, not government_is_ecclesiastical
# Scope: candidate character. root: the theocratic ruler
scripted_trigger pam_theocratic_ruler_0001_subordinate_priest_trigger = {
	this != root
    is_available = yes
	faith = root.faith
	government_has_flag = government_is_theocracy
	is_landed = yes
}

pam_theocratic_ruler_events.0001 = {
	type = character_event
	title = pam_theocratic_ruler_events.0001.t
	desc = pam_theocratic_ruler_events.0001.desc
	theme = christian
	override_background = { reference = council_chamber }

	left_portrait = {
		character = root
		animation = priest_corrupt
	}
	right_portrait = {
		character = scope:subordinate_priest
		animation = reading
	}

	cooldown = { years = 15 }

	trigger = {
		pam_government_is_religious_playable_trigger = yes # In pam_scripted_triggers
		is_landed = yes
		is_available_adult = yes
		OR = {
			any_vassal_or_below = {
				pam_theocratic_ruler_0001_subordinate_priest_trigger = yes
			}
			AND = {
				exists = capital_county
				capital_county = {
					exists = clerical_region_title
					clerical_region_title = {
						any_county_in_clerical_region = {
							holder = {
								pam_theocratic_ruler_0001_clerical_region_priest_trigger = yes
							}
						}
					}
				}
			}
			cp:councillor_court_chaplain ?= {
				is_imprisoned = no
				faith = root.faith
			}
		}
	}

	immediate = {
		every_vassal_or_below = {
			limit = {
				pam_theocratic_ruler_0001_subordinate_priest_trigger = yes
			}
			add_to_list = pam_theocratic_ruler_0001_priest_candidates
		}
		if = {
			limit = {
				exists = capital_county
				capital_county = {
					exists = clerical_region_title
				}
			}
			capital_county.clerical_region_title = {
				every_county_in_clerical_region = {
					limit = {
						holder = {
							pam_theocratic_ruler_0001_clerical_region_priest_trigger = yes
						}
					}
					holder = {
						add_to_list = pam_theocratic_ruler_0001_priest_candidates
					}
				}
			}
		}
		if = {
			limit = {
				list_size = {
					name = pam_theocratic_ruler_0001_priest_candidates
					value > 0
				}
			}
			random_in_list = {
				list = pam_theocratic_ruler_0001_priest_candidates
				save_scope_as = subordinate_priest
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:subordinate_priest }
				exists = cp:councillor_court_chaplain
				cp:councillor_court_chaplain = {
					this != root
					is_alive = yes
					is_imprisoned = no
					faith = root.faith
				}
			}
			cp:councillor_court_chaplain = {
				save_scope_as = subordinate_priest
			}
		}
	}

	# Frame the priest: I find evidence he mishandled sacred funds…
	option = {
		name = pam_theocratic_ruler_events.0001.a
		skill = intrigue
		trigger = {
			intrigue >= high_skill_rating
		}
		show_as_unavailable = {
			always = yes
		}
		add_short_term_gold = medium_gold_value
		add_piety = medium_piety_loss
		rightfully_imprison_character_less_verbose_effect = {
			TARGET = scope:subordinate_priest
			IMPRISONER = root
		}
		if = {
			limit = {
				scope:subordinate_priest = {
					is_councillor_of = root
				}
			}
			fire_councillor = scope:subordinate_priest
		}
		if = {
			limit = {
				scope:subordinate_priest = {
					is_courtier_of = root
				}
			}
			remove_courtier_or_guest = scope:subordinate_priest
		}
		scope:subordinate_priest = {
			add_opinion = {
				modifier = blamed_me_unjustly
				target = root
			}
			add_stress = major_stress_gain
		}
		stress_and_fulfillment_impact = {
			honest = medium_stress_impact_gain
			just = minor_stress_impact_gain
			deceitful = medium_stress_impact_loss
			callous = minor_stress_impact_loss
		}
		ai_chance = {
			base = 5
			ai_value_modifier = {
				ai_honor = -0.5
				ai_greed = 0.35
				ai_rationality = 0.2
			}
		}
	}

	# "Lets reason together as clergy."
	option = {
		name = pam_theocratic_ruler_events.0001.b
		duel = {
			skills = { diplomacy intrigue }
			target = scope:subordinate_priest
			50 = {
				desc = pam_theocratic_ruler_events.0001.b.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				add_short_term_gold = medium_gold_value
				add_diplomacy_lifestyle_xp = minor_lifestyle_xp
				add_intrigue_lifestyle_xp = minor_lifestyle_xp
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_theocratic_ruler_events.0001.b.toast.success
					left_icon = scope:subordinate_priest
					right_icon = root
					scope:subordinate_priest = {
						if = {
							limit = {
								can_add_hook = {
									target = root
									type = favor_hook
								}
							}
							add_hook = {
								type = favor_hook
								target = root
							}
						}
					}
				}
			}
			50 = {
				desc = pam_theocratic_ruler_events.0001.b.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = pam_theocratic_ruler_events.0001.b.toast.failure
					left_icon = scope:subordinate_priest
					right_icon = root
					add_stress = minor_stress_gain
					scope:subordinate_priest = {
						add_opinion = {
							modifier = upset_opinion
							target = root
							opinion = -25
						}
					}
				}
			}
		}
		stress_and_fulfillment_impact = {
			gregarious = minor_stress_impact_loss
			shy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 35
			ai_value_modifier = {
				ai_rationality = 0.35
				ai_sociability = 0.25
				ai_honor = 0.15
			}
		}
	}

	# Into the treasury, with thanksgiving and care.
	option = {
		name = pam_theocratic_ruler_events.0001.c
		add_treasury_or_gold = minor_treasury_or_gold_value
		add_piety = medium_piety_gain
		stress_and_fulfillment_impact = {
			zealous = minor_stress_impact_loss
			humble = minor_stress_impact_loss
			greedy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 55
			ai_value_modifier = {
				ai_zeal = 0.45
				ai_compassion = 0.2
				ai_greed = -0.35
			}
		}
	}
}

## pam_theocratic_ruler_events.0002 — A Traveler from Across the Border
############################

scripted_trigger pam_theocratic_ruler_0002_foreign_neighbor_trigger = {
	top_liege != root.top_liege
	is_available = yes
	NOR = {
		is_allied_to = root
		has_truce = root
	}
	faith = {
		has_doctrine_parameter = unreformed
		this != root.faith
	}
	culture != root.culture
}

pam_theocratic_ruler_events.0002 = {
	type = character_event
	title = pam_theocratic_ruler_events.0002.t
	desc = pam_theocratic_ruler_events.0002.desc
	theme = christian
	override_background = { reference = activity_tour } 

	left_portrait = {
		character = root
		animation = personality_compassionate
	}
	right_portrait = {
		character = scope:foreign_ruler
		animation = personality_honorable
	}

	cooldown = { years = 10 }

	trigger = {
		pam_government_is_religious_playable_trigger = yes
		is_landed = yes
		is_available_adult = yes
		trigger_if = {
			limit = {
				top_liege != this
			}
			any_neighboring_realm_same_rank_owner = {
				culture != root.culture
				pam_theocratic_ruler_0002_foreign_neighbor_trigger = yes
			}
		}
		trigger_else = {
			any_neighboring_top_liege_realm_owner = {
				culture != root.culture
				pam_theocratic_ruler_0002_foreign_neighbor_trigger = yes
			}
		}
	}

	immediate = {
		if = {
			limit = {
				top_liege != this
			}
			random_neighboring_realm_same_rank_owner = {
				limit = {
					pam_theocratic_ruler_0002_foreign_neighbor_trigger = yes
				}
				save_scope_as = foreign_ruler
			}
		}
		else = {
			random_neighboring_top_liege_realm_owner = {
				limit = {
					pam_theocratic_ruler_0002_foreign_neighbor_trigger = yes
				}
				save_scope_as = foreign_ruler
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			add = 0.5
			diplomacy >= medium_skill_rating
		}
		modifier = {
			add = 0.25
			has_trait = zealous
		}
	}

	# A modest gift and offer of peace.
	option = {
		name = pam_theocratic_ruler_events.0002.a
		trigger = {
			short_term_gold >= minor_gold_value
		}
		add_truce_both_ways = {
			character = scope:foreign_ruler
			years = 5
			name = TRUCE_PAM_NEIGHBOR_GIFT
		}
		pay_short_term_gold = {
			target = scope:foreign_ruler
			gold = minor_gold_value
		}
		scope:foreign_ruler = {
			add_opinion = {
				modifier = grateful_opinion
				target = root
				opinion = 15
			}
		}
		ai_chance = {
			base = 35
			ai_value_modifier = {
				ai_greed = -0.25
				ai_zeal = 0.5
				ai_rationality = 0.35
			}
		}
	}

	# I seek honest understanding between our peoples.
	option = {
		name = pam_theocratic_ruler_events.0002.b
		skill = diplomacy
		duel = {
			skill = diplomacy
			target = scope:foreign_ruler
			45 = {
				desc = pam_theocratic_ruler_events.0002.b.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = pam_theocratic_ruler_events.0002.b.toast.success
					left_icon = scope:foreign_ruler
					right_icon = root
					root.culture = {
						change_cultural_acceptance = {
							target = scope:foreign_ruler.culture
							value = medium_cultural_acceptance_gain
							desc = pam_theocratic_ruler_events.0002.b.acceptance
						}
					}
				}
			}
			45 = {
				desc = pam_theocratic_ruler_events.0002.b.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = pam_theocratic_ruler_events.0002.b.toast.failure
					left_icon = scope:foreign_ruler
					right_icon = root
					scope:foreign_ruler = {
						add_opinion = {
							modifier = insult_opinion
							target = root
							opinion = low_negative_opinion
						}
					}
				}
			}
		}
		stress_and_fulfillment_impact = {
			shy = minor_stress_impact_gain
			gregarious = minor_stress_impact_loss
		}
		ai_chance = {
			base = 45
			ai_value_modifier = {
				ai_sociability = 0.45
				ai_rationality = 0.25
				ai_boldness = 0.15
			}
		}
	}

	# Heathens have no place in our lands.
	option = {
		name = pam_theocratic_ruler_events.0002.c
		add_piety = medium_piety_gain
		scope:foreign_ruler = {
			progress_towards_rival_effect = {
				CHARACTER = root
				REASON = rival_insulted_me
				OPINION = 0
			}
		}
		stress_and_fulfillment_impact = {
			zealous = minor_stress_impact_loss
			forgiving = minor_stress_impact_gain
			compassionate = minor_stress_impact_gain
		}
		ai_chance = {
			base = 25
			ai_value_modifier = {
				ai_zeal = 0.65
				ai_compassion = -0.35
				ai_honor = -0.25
			}
		}
	}
}


###########################
## pam_theocratic_ruler_events.0003 - Counsel for a Hardened Border
############################


scripted_trigger pam_theocratic_ruler_0003_alliance_target_trigger = {
	this != root
	highest_held_title_tier >= tier_county
	government_has_flag = government_is_theocracy
	faith = root.faith
	NOR = {
		is_allied_to = root
		has_truce = root
		is_at_war_with = root
	}
	in_diplomatic_range = root
}

pam_theocratic_ruler_events.0003 = {
	type = character_event
	title = pam_theocratic_ruler_events.0003.t
	desc = pam_theocratic_ruler_events.0003.desc
	theme = christian
	override_background = { reference = council_chamber }

	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:advisor
		animation = personality_honorable
	}
	lower_center_portrait = {
		character = scope:alliance_target
	}

	cooldown = { years = 8 }

	trigger = {
		pam_government_is_religious_playable_trigger = yes
		is_landed = yes
		is_available_adult = yes
		exists = scope:advisor
		any_ruler = {
			pam_theocratic_ruler_0003_alliance_target_trigger = yes
		}
		OR = {
			any_memory = {
				OR = {
					has_memory_type = battle_won_memory
					has_memory_type = battle_lost_memory
				}
			}
			any_sub_realm_county = {
				title_province = {
					OR = {
						has_province_modifier = recently_looted_modifier
						has_province_modifier = recently_bartered_modifier
					}
				}
			}
		}
	}

	immediate = {
		if = {
			limit = {
				exists = cp:councillor_marshal
			}
			cp:councillor_marshal = { save_scope_as = advisor }
		}
		else_if = {
			limit = {
				exists = cp:councillor_chancellor
			}
			cp:councillor_chancellor = { save_scope_as = advisor }
		}
		else_if = {
			limit = {
				exists = cp:councillor_steward
			}
			cp:councillor_steward = { save_scope_as = advisor }
		}

		random_ruler = {
			limit = {
				pam_theocratic_ruler_0003_alliance_target_trigger = yes
			}
			weight = {
				base = 1
				modifier = {
					add = 8
					is_independent_ruler = no
				}
				modifier = {
					add = 4
					highest_held_title_tier = tier_county
				}
				modifier = {
					add = 3
					highest_held_title_tier = tier_duchy
				}
				modifier = {
					add = -20
					primary_title = {
						is_head_of_faith = yes
					}
				}
				modifier = {
					add = -8
					highest_held_title_tier >= tier_kingdom
				}
			}
			save_scope_as = alliance_target
		}

		if = {
			limit = {
				any_sub_realm_county = {
					title_province = {
						OR = {
							has_province_modifier = recently_looted_modifier
							has_province_modifier = recently_bartered_modifier
						}
					}
				}
			}
			random_sub_realm_county = {
				limit = {
					title_province = {
						OR = {
							has_province_modifier = recently_looted_modifier
							has_province_modifier = recently_bartered_modifier
						}
					}
				}
				save_scope_as = raided_county
			}
		}
	}

	# Send envoys. We must stand together.
	option = {
		name = pam_theocratic_ruler_events.0003.a
		custom_tooltip = pam_theocratic_ruler_events.0003.a.tt
		create_alliance = {
			target = scope:alliance_target
			allied_through_owner = root
			allied_through_target = scope:alliance_target
		}
		add_prestige = minor_prestige_loss
		ai_chance = {
			base = 45
			ai_value_modifier = {
				ai_rationality = 0.6
				ai_honor = 0.3
			}
		}
	}

	# Raise a retinue from faithful peasants for border defense.
	option = {
		name = pam_theocratic_ruler_events.0003.b
		spawn_army = {
			levies = 150
			location = root.capital_province
			name = pam_theocratic_ruler_events.0003.b.army
			inheritable = no
		}
		add_piety = minor_piety_loss
		ai_chance = {
			base = 30
			ai_value_modifier = {
				ai_boldness = 0.35
				ai_zeal = 0.2
			}
		}
	}

	# Send aid and restore one recently raided county.
	option = {
		name = pam_theocratic_ruler_events.0003.c
		trigger = {
			exists = scope:raided_county
		}
		scope:raided_county.title_province = {
			remove_province_modifier = recently_looted_modifier
			remove_province_modifier = recently_bartered_modifier
		}
		ai_chance = {
			base = 15
			ai_value_modifier = {
				ai_compassion = 0.2
				ai_rationality = 0.2
			}
		}
	}

	# I place this in God's hands.
	option = {
		name = pam_theocratic_ruler_events.0003.d
		add_piety = minor_piety_gain
		stress_and_fulfillment_impact = {
			zealous = miniscule_stress_impact_loss
		}
		ai_chance = {
			base = 10
			ai_value_modifier = {
				ai_zeal = 0.4
			}
		}
	}
}

###########################
## pam_theocratic_ruler_events.0004 — Brethern as Neighbors
############################

pam_theocratic_ruler_events.0004 = {
	type = character_event
	title = pam_theocratic_ruler_events.0004.t
	desc = pam_theocratic_ruler_events.0004.desc
	theme = christian

	left_portrait = {
		character = root
		animation = chaplain
	}
	right_portrait = {
		character = scope:monastic_order_head
		animation = prayer_crossing
	}
	lower_center_portrait = {
		character = scope:monastic_monk
	}

	cooldown = { years = 8 }

	trigger = {
	pam_government_is_religious_playable_trigger = yes
		is_available_adult = yes
		any_sub_realm_county = {
			any_county_province = {
				barony = {
					is_leased_out = yes
					exists = lessee
					lessee = {
						government_has_flag = government_is_monastic_holy_order
						is_alive = yes
					}
				}
			}
		}
	}

	immediate = {
		random_sub_realm_county = {
			limit = {
				any_county_province = {
					barony = {
						is_leased_out = yes
						exists = lessee
						lessee = {
							government_has_flag = government_is_monastic_holy_order
							is_alive = yes
						}
					}
				}
			}
			save_scope_as = pam_monastic_lease_county
			random_county_province = {
				limit = {
					barony = {
						is_leased_out = yes
						lessee ?= {
							government_has_flag = government_is_monastic_holy_order
						}
					}
				}
				barony = {
					lessee = { save_scope_as = monastic_order_head }
				}
			}
		}

		if = {
			limit = {
				scope:pam_monastic_lease_county = {
					any_neighboring_county = {
						holder = {
							top_liege = root.top_liege
						}
					}
				}
			}
			scope:pam_monastic_lease_county = {
				random_neighboring_county = {
					limit = {
						holder = {
							top_liege = root.top_liege
						}
					}
					save_scope_as = aided_neighbor_county
				}
			}
		}
		else_if = {
			limit = {
				capital_county = {
					any_neighboring_county = {
						holder = {
							top_liege = root.top_liege
						}
					}
				}
			}
			capital_county = {
				random_neighboring_county = {
					limit = {
						holder = {
							top_liege = root.top_liege
						}
					}
					save_scope_as = aided_neighbor_county
				}
			}
		}

		if = {
			limit = {
				scope:monastic_order_head = {
