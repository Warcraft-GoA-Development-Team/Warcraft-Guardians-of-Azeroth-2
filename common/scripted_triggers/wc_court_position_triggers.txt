# camp officers valid character triggers

light_aerial_captain_validity_trigger = {
	custom_tooltip = {
		text = none_does_not_have_a_full_time_job
		trigger_if = {
			limit = { NOT = { has_court_position = light_aerial_captain_camp_officer } }
			court_position_does_not_already_have_a_job_trigger = yes
		}
	}
	base_court_position_validity_trigger = { EMPLOYER = $EMPLOYER$ }
}
