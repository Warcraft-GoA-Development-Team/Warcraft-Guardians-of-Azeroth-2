# A supporter must still belong to the living human Light/Scarlet milieu.
wc_balnazzar_scarlet_support_trigger = {
	is_alive = yes
	is_imprisoned = no
	has_trait = creature_human
	has_being_trait_trigger = no
	OR = { has_faith = faith:holy_light has_faith = faith:scarletism }
	OR = {
		NOT = { exists = liege }
		top_liege = {
			OR = { has_faith = faith:holy_light has_faith = faith:scarletism }
		}
	}
}

# Positive evidence from pre-flag saves: only the scripted parricide raised him.
wc_balnazzar_parricide_trigger = {
	OR = {
		has_character_flag = wc_balnazzar_parricide_committed
		# Resurrection creates a new character and reassigns both sons' father.
		character:60019 = {
			exists = father
			father = { has_character_flag = wc_balnazzar_raised_alexandros }
		}
		character:60020 = {
			exists = father
			father = { has_character_flag = wc_balnazzar_raised_alexandros }
		}
	}
}

# Same check at event delivery and immediately before the irreversible handoff.
wc_balnazzar_can_take_next_vessel_trigger = {
	is_alive = yes
	has_character_flag = wc_balnazzar_disguise_active
	NOT = { has_character_flag = balnazzar_shed_the_mask }
	exists = global_var:balnazzar_dreadlord_story
	global_var:balnazzar_dreadlord_story = { story_owner = root }
	exists = scope:next_vessel
	scope:next_vessel = {
		is_alive = yes
		is_ai = yes
		is_adult = yes
		is_landed = yes
		is_playable_character = yes
		has_trait = creature_human
		has_being_trait_trigger = no
		NOT = { has_character_flag = wc_balnazzar_disguise_active }
	}
}

wc_balnazzar_can_found_scarlet_trigger = {
	this = character:60021
	is_alive = yes
	has_character_flag = wc_balnazzar_disguise_active
	NOT = { has_character_flag = has_formed_the_scarlet_crusade }
	# The takeover stays inside the surviving Order, even after succession.
	title:k_order_of_the_silver_hand = {
		exists = holder
		holder = {
			wc_balnazzar_scarlet_support_trigger = yes
			OR = {
				this = root
				is_vassal_or_below_of = root
				root = { is_vassal_or_below_of = prev }
			}
		}
	}
	exists = global_var:balnazzar_dreadlord_story
	global_var:balnazzar_dreadlord_story = {
		story_owner = root
		NOT = { has_variable = wc_balnazzar_scarlet_founded }
	}
	OR = {
		character:60017 = { is_alive = no }
		wc_balnazzar_parricide_trigger = yes
	}
	OR = {
		AND = {
			wc_balnazzar_parricide_trigger = yes
			character:60019 = { wc_balnazzar_scarlet_support_trigger = yes }
		}
		character:60550 = { wc_balnazzar_scarlet_support_trigger = yes }
		character:60560 = { wc_balnazzar_scarlet_support_trigger = yes }
	}
}
