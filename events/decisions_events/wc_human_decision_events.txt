namespace = wc_defias_brotherhood

### 0001 -> After the Defias Brotherhood conquers Northshire and Stormwind

wc_defias_brotherhood.0001 = { # By Ikarimonogatari
    type = character_event
    content_source = dlc_GOA
    title = wc_defias_brotherhood.0001.title
    desc = wc_defias_brotherhood.0001.desc
    theme = war

    left_portrait = {
        character = root
        animation = celebrate_sword
    }

    right_portrait = {
        character = title:k_stormwind.holder
        animation = beg
    }

    option = { # Extort the Money in exchange for sparing the nobles
        name = wc_defias_brotherhood.0001.a
        title:k_stormwind.holder = {
            pay_short_term_gold = {
                target = root
                gold = 2000
            }
        }
    }
    option = { # Execute them all and decide what to do with the lands
        name = wc_defias_brotherhood.0001.b
        trigger_event = {
			id = wc_defias_brotherhood.0002
		}
        title:k_stormwind.holder = {
            override_death_killer_effect = {
                death_reason = death_execution
	    		killer = root
            }
            every_close_or_extended_family_member = {
                override_death_killer_effect = {
                    death_reason = death_execution
	    			killer = root
                }
            }
        }
    }
}

wc_defias_brotherhood.0002 = { # By Ikarimonogatari
    type = character_event
    content_source = dlc_GOA
    title = wc_defias_brotherhood.0002.title
    desc = wc_defias_brotherhood.0002.desc
    theme = realm

    right_portrait = {
        character = root
        animation = thinking
    }

    option = { # Declare ourselves as the new Noble leaders of the Kingdom!
        name = wc_defias_brotherhood.0002.a
        get_title = title:k_stormwind
        if = {
            limit = { has_trait = gallowsbait }
            remove_trait = gallowsbait
        }
        if = {
            limit = { has_trait = peasant_leader }
            remove_trait = peasant_leader
        }
        add_legitimacy = 500
        add_prestige = 2000
    }
    option = { # Burn down the Noble's systems and continue our crusades
        name = wc_defias_brotherhood.0002.b
        destroy_title = title:k_stormwind
        spawn_army = { # Fellow peasents gather to liberate other lowborns!
            inheritable = yes
            uses_supply = no
            name = defias_league
            men_at_arms = {
                type = bombard
                stacks = 1
            }
            levies = 5000
            location = capital_province
			origin = capital_province
        }
    }
}
