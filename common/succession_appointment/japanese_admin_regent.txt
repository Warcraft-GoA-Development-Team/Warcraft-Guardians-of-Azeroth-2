japanese_regent = {
	default_candidates = { holder_close_family landed_vassal unlanded_noble_house_head }

	candidate_score = {
		value = {
			# Base - Based on skills
			add = {
				value = martial
				add = diplomacy
				add = stewardship
				add = intrigue
				add = learning
				multiply = 1
				desc = sum_of_skills_desc
			}
			
			add = appointment_score_base
			add = appointment_score_non_military
			add = appointment_score_non_merit
			add = appointment_score_regent

			### JAPAN
			# Dynasty Renown
			if = {
				limit = {
					dynasty ?= { dynasty_prestige_level >= 1 }
				}
				add = {
					value = {
						value = dynasty.dynasty_prestige_level
						multiply = {
							value = 5
							if = {
								limit = {
									dynasty.dynast.house = root.house
								}
								add = 5
							}
						}
					}
					desc = japanese_kampaku_dynasty_splendor_desc
				}
			}

			add = appointment_score_final_factors
		}
	}
}
