wc_embrace_vampire_interaction = {
	icon = being_undead
	interface_priority = 30
	category = interaction_category_hostile
	popup_on_receive = yes
	pause_on_receive = yes
	desc = wc_embrace_vampire_interaction_desc

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = vampire
				has_perk = death_magic_tree_3_perk_4
			}
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			wc_can_become_vampire_trigger = yes
			OR = {
				is_imprisoned_by = scope:actor
				is_courtier_of = scope:actor
				liege = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			NOT = { is_isolated_from_trigger = { target = scope:recipient } }	# Warcraft
		}
	}

	cost = { prestige = medium_prestige_value }
	cooldown = { years = 5 }

	auto_accept = {
		scope:recipient = { is_imprisoned_by = scope:actor }
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { has_trait = vampire }
				custom_tooltip = wc_embrace_vampire_interaction.bite.tt
			}
			else = {
				custom_tooltip = wc_embrace_vampire_interaction.ritual.tt
			}
		}
		scope:recipient = {
			if = {
				limit = { is_imprisoned_by = scope:actor }
				save_temporary_scope_as = wc_forced_vampire
			}
			wc_turn_into_vampire_effect = { SIRE = scope:actor }
			if = {
				limit = { exists = scope:wc_forced_vampire }
				add_opinion = {
					target = scope:actor
					modifier = wc_forcibly_embraced
				}
			}
			else = {
				add_opinion = {
					target = scope:actor
					modifier = wc_embraced_by
				}
			}
		}
	}

	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = wc_embrace_vampire_interaction.declined.tt
				left_icon = scope:recipient
			}
			hidden_effect = { remove_interaction_cooldown = wc_embrace_vampire_interaction }
		}
	}

	ai_accept = {
		base = -50

		modifier = {
			add = 50
			scope:recipient = { has_trait = being_undead }
			desc = wc_embrace_vampire_undead_reason
		}
		modifier = {
			add = -100
			scope:recipient = {
				faith = { has_doctrine_parameter = being_undead_illegal }
			}
			desc = wc_embrace_vampire_illegal_reason
		}
		modifier = {
			add = -50
			scope:recipient = {
				faith = { has_doctrine_parameter = being_undead_shunned }
			}
			desc = wc_embrace_vampire_shunned_reason
		}

		modifier = {
			add = 20
			scope:recipient = { has_trait = ambitious }
			desc = wc_embrace_vampire_ambitious_reason
		}
		modifier = {
			add = 20
			scope:recipient = { has_trait = callous }
			desc = wc_embrace_vampire_callous_reason
		}
		modifier = {
			add = 20
			scope:recipient = { has_trait = sadistic }
			desc = wc_embrace_vampire_sadistic_reason
		}
		modifier = {
			add = -30
			scope:recipient = { has_trait = zealous }
			desc = wc_embrace_vampire_zealous_reason
		}
		modifier = {
			add = -30
			scope:recipient = { has_trait = compassionate }
			desc = wc_embrace_vampire_compassionate_reason
		}

		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}

		modifier = {
			add = intimidated_reason_value
			scope:recipient = {
				target_is_liege_or_above = scope:actor
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = cowed_reason_value
			scope:recipient = {
				target_is_liege_or_above = scope:actor
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			add = intimidated_external_reason_value
			scope:recipient = {
				NOT = { target_is_liege_or_above = scope:actor }
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = cowed_external_reason_value
			scope:recipient = {
				NOT = { target_is_liege_or_above = scope:actor }
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
	}

	ai_targets = {
		ai_recipients = prisoners
	}
	ai_targets = {
		ai_recipients = courtiers
		chance = 0.25
	}

	ai_frequency = 60

	ai_potential = {
		is_imprisoned = no

		OR = {
			has_trait = vampire
			AND = {
				has_perk = death_magic_tree_3_perk_4
				has_faith = faith:death_god
			}
		}
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = 50
			scope:recipient = { is_imprisoned_by = scope:actor }
		}
		modifier = {
			add = 25
			scope:actor = {
				OR = {
					has_trait = callous
					has_trait = sadistic
				}
			}
		}
	}
}
