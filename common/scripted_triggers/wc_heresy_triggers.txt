# Historical heresy outbreaks (By God Alone component 3) for GoA.
# Callers: wc_faith_historical_heresy_pulse (common/on_action/wc_heresy_on_actions.txt) and
# events/wc_events/wc_heresy_historical_events.txt.
#
# A heresy is defined by:
#	- wc_heresy_<key>_eligible_trigger and wc_heresy_<key>_region_trigger (this file; the region is written only there)
#	- an entry in the wc_heresy_historical.0001 random_list
#	- a case in each switch on scope:heresy_key: wc_heresy_in_home_region_trigger (this file),
#	  wc_heresy_dispatch_founder_effect and wc_heresy_queue_flavor_effect (wc_heresy_effects.txt)
#	- its .0050 notice entries and loc
# Per-heresy state lives on the seat barony (see wc_heresy_set_cooldown_effect), so two heresies never share a seat.

##### Router eligibility (faith scope) #####

wc_heresy_historical_eligible_trigger = {
	OR = {
		wc_heresy_seline_eligible_trigger = yes
		wc_heresy_damned_eligible_trigger = yes
		wc_heresy_scarlet_eligible_trigger = yes
		wc_heresy_deaths_head_eligible_trigger = yes
		wc_heresy_atalai_eligible_trigger = yes
		wc_heresy_burning_blade_eligible_trigger = yes
	}
}

# ORIGIN: origin faith key. SEAT: seat barony key. START/END: first-outbreak window (END exclusive).
# The origin check comes first: the pulse evaluates this for every faith every month.
wc_heresy_eligible_trigger = {
	this = faith:$ORIGIN$

	title:$SEAT$ = {
		NOT = { has_variable = wc_heresy_cooldown }
	}

	OR = {
		wc_heresy_is_resurgence_trigger = { SEAT = $SEAT$ }
		exists = scope:debug_ignore_windows
		AND = {
			current_date >= $START$
			current_date < $END$
		}
	}
}

##### Resurgence (any scope) #####

	title:$SEAT$ = { has_variable = wc_heresy_first_fired }
}

# Province scope. Dispatches on scope:heresy_key.
wc_heresy_in_home_region_trigger = {
	switch = {
		trigger = scope:heresy_key
		flag:seline = { wc_heresy_seline_region_trigger = yes }
		flag:damned = { wc_heresy_damned_region_trigger = yes }
		flag:scarlet = { wc_heresy_scarlet_region_trigger = yes }
		flag:deaths_head = { wc_heresy_deaths_head_region_trigger = yes }
		flag:atalai = { wc_heresy_atalai_region_trigger = yes }
		flag:burning_blade = { wc_heresy_burning_blade_region_trigger = yes }
		fallback = { always = no }
	}
}

##### Characters #####

# A lore founder can lead the outbreak. TARGET: the heresy faith (faith:x).
# Faith conditions are the caller's: each case of wc_heresy_dispatch_founder_effect adds its own.
wc_heresy_founder_usable_trigger = {
	is_alive = yes

	is_ai = yes
	is_adult = yes
	is_imprisoned = no

	NOT = { scope:origin_faith.religious_head ?= this }
	# A head of the target faith may lead (Jin'do); a head of any other faith may not
	trigger_if = {
		limit = { NOT = { faith = $TARGET$ } }
		NOT = { faith.religious_head ?= this }
	}
}

# Players who get the outbreak notice (.0050)
wc_heresy_notice_audience_trigger = {
	is_ai = no

	NOT = { this = scope:founder }

	faith = scope:origin_faith
}

# Players who get the flavor chain: landed, outside the heresy, near the outbreak.
# v1's Light heresies keep origin-faith players only, because their texts assume it; the others reach any faith.
wc_heresy_audience_trigger = {
	is_ai = no

	is_landed = yes
	exists = capital_county

	NOT = { this = scope:founder }

	trigger_if = {
		limit = {
			OR = {
				scope:heresy_key = flag:seline
				scope:heresy_key = flag:damned
				scope:heresy_key = flag:scarlet
			}
		}
		faith = scope:origin_faith
	}
	NOT = { wc_heresy_player_is_insider_trigger = yes }
	NOT = { scope:heresy_faith.religious_head ?= this }

	OR = {
		capital_county.title_province = { wc_heresy_in_home_region_trigger = yes }
		capital_county = {
			squared_distance = {
				target = scope:epicenter_ruler.capital_county
				value <= squared_distance_medium
			}
		}
	}
}

# Already in the heresy, openly or in secret
wc_heresy_player_is_insider_trigger = {
	OR = {
		faith = scope:heresy_faith
		secret_faith ?= scope:heresy_faith
	}
}
