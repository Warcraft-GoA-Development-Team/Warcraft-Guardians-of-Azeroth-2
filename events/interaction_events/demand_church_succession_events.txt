####################
# DEMAND CHURCH SUCCESSION
# A theocrat has invoked the right to inherit your realm should your line die out.
####################

namespace = demand_church_succession

# Letter to the secular ruler. Layered variants by sender role, sender personality, and recipient circumstance.
demand_church_succession.0001 = {
	type = letter_event
	opening = {
		desc = demand_church_succession.0001.opening
	}
	desc = {
		# Sender-role flavour (which kind of theocrat is making the demand).
		# Christian variants get the canonical Pope/Archbishop/Bishop framing; non-Christian theocracies fall through to the generic variants.
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						faith = { always = no } # Warcraft - christianity_religion does not exist in GoA
						exists = faith.religious_head
						this = faith.religious_head
					}
				}
				desc = demand_church_succession.0001.desc.pope
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						faith = { always = no } # Warcraft - christianity_religion does not exist in GoA
						any_held_title = { has_clerical_region = yes }
					}
				}
				desc = demand_church_succession.0001.desc.archbishop
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = { faith = { always = no } } # Warcraft - christianity_religion does not exist in GoA
				}
				desc = demand_church_succession.0001.desc.bishop
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						exists = faith.religious_head
						this = faith.religious_head
					}
				}
				desc = demand_church_succession.0001.desc.head_of_faith_generic
			}
			triggered_desc = {
				desc = demand_church_succession.0001.desc.theocrat_generic
			}
		}
		# Personality flavour line. First match wins; default catches the rest.
		first_valid = {
			triggered_desc = {
				trigger = { scope:puppet_or_actor = { has_trait = zealous } }
				desc = demand_church_succession.0001.flavour.zealous
			}
			triggered_desc = {
				trigger = { scope:puppet_or_actor = { has_trait = ambitious } }
				desc = demand_church_succession.0001.flavour.ambitious
			}
			triggered_desc = {
				trigger = { scope:puppet_or_actor = { has_trait = deceitful } }
				desc = demand_church_succession.0001.flavour.deceitful
			}
			triggered_desc = {
				trigger = { scope:puppet_or_actor = { has_trait = greedy } }
				desc = demand_church_succession.0001.flavour.greedy
			}
			triggered_desc = {
				trigger = { scope:puppet_or_actor = { has_trait = compassionate } }
				desc = demand_church_succession.0001.flavour.compassionate
			}
			triggered_desc = {
				desc = demand_church_succession.0001.flavour.default
			}
		}
		# Recipient-condition flavour: whether they are aging/childless makes the news read very differently.
		first_valid = {
			triggered_desc = {
				trigger = {
					age >= 50
					NOT = { any_child = { is_alive = yes } }
				}
				desc = demand_church_succession.0001.recipient.elderly_childless
			}
			triggered_desc = {
				trigger = {
					NOT = { any_child = { is_alive = yes } }
				}
				desc = demand_church_succession.0001.recipient.childless
			}
			triggered_desc = {
				trigger = {
					any_child = {
						is_alive = yes
						count = 1
					}
				}
				desc = demand_church_succession.0001.recipient.one_child
			}
			triggered_desc = {
				desc = demand_church_succession.0001.recipient.default
			}
		}
	}

	sender = {
		character = scope:puppet_or_actor
		animation = personality_zealous
	}

	# Accept the arrangement: take the spiritual comfort that comes with it.
	option = {
		name = demand_church_succession.0001.a
		show_as_tooltip = {
			scope:recipient = {
				set_fallback_succession_title = scope:puppet_or_actor.primary_title
			}
		}
		custom_tooltip = church_fallback_succession_promise_recipient_tt
		
		if = {
			limit = {
				exists = scope:puppet_or_actor.faith.religious_head
				scope:puppet_or_actor = scope:puppet_or_actor.faith.religious_head
			}
			add_piety = major_piety_value
			change_spiritual_fulfillment = major_spiritual_fulfillment_value
		}
		else = {
			add_piety = medium_piety_value
			change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		}
		demand_church_succession_side_effects = yes
		ai_chance = {
			base = 100
		}
	}

	# Take the warning to heart and redouble efforts to produce an heir.
	option = {
		name = demand_church_succession.0001.b
		trigger = {
			can_have_children = yes
		}
		show_as_tooltip = {
			scope:recipient = {
				set_fallback_succession_title = scope:puppet_or_actor.primary_title
			}
		}
		custom_tooltip = church_fallback_succession_promise_recipient_tt
		add_character_modifier = {
			modifier = demand_church_succession_fertility_modifier
			years = 10
		}
		demand_church_succession_side_effects = yes
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				NOT = { any_child = { is_alive = yes } }
			}
			modifier = {
				add = 25
				has_trait = lustful
			}
		}
	}

	# Defy the Church: clear the fallback at significant cost. Player-only.
	option = {
		name = demand_church_succession.0001.c
		trigger = {
			is_ai = no
		}
		show_as_unavailable = {
			piety_level <= -3
		}
		add_internal_flag = dangerous
		custom_tooltip = demand_church_succession.0001.c.tt
		clear_fallback_succession = yes
		add_piety_level = -1
		add_character_flag = {
			flag = demand_church_succession_refused_immunity
			years = 10
		}
		scope:puppet_or_actor = {
			add_opinion = {
				target = root
				modifier = defied_church_succession_opinion
			}
			trigger_event = demand_church_succession.0002
		}
	}
}

demand_church_succession.0002 = {
	type = letter_event
	opening = {
		desc = demand_church_succession.0002.opening
	}
	desc = demand_church_succession.0002.desc

	sender = {
		character = scope:recipient
		animation = personality_callous
	}

	option = {
		name = demand_church_succession.0002.a
		show_as_tooltip = {
			scope:recipient = {
				clear_fallback_succession = yes
				add_piety_level = -1
			}
			add_opinion = {
				target = scope:recipient
				modifier = defied_church_succession_opinion
			}
		}
		add_piety = scope:puppet_or_actor.major_piety_value
		if = {
			limit = {
				scope:puppet_or_actor.faith.religious_head ?= scope:puppet_or_actor
			}
			add_treasury_or_gold = {
				value = scope:puppet_or_actor.major_treasury_or_gold_value
				multiply = scope:recipient.primary_title.tier
			}
		}
		else = {
			add_treasury_or_gold = {
				value = scope:puppet_or_actor.medium_treasury_or_gold_value
				multiply = scope:recipient.primary_title.tier
			}
		}
	}
}
