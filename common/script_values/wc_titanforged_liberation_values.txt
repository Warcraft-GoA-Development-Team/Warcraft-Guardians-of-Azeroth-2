wc_titanforged_reorder_county_threshold = 12

wc_titanforged_reorder_county_current_progress = {
	value = 0
	if = {
		limit = {
			scope:county ?= {
				has_variable = wc_titanforged_reorder_current_progress
			}
		}
		add = scope:county.var:wc_titanforged_reorder_current_progress
	}
}

wc_titanforged_reorder_county_base = {
	value = 1
}

wc_titanforged_reorder_county_skill_bonus = {
	value = scope:councillor.martial
	subtract = 10
	divide = 10
}

wc_titanforged_reorder_county_friend_bonus = {
	value = 0
	if = {
		limit = {
			scope:councillor_liege = {
				has_relation_friend = scope:councillor
				NOT = { has_relation_best_friend = scope:councillor }
			}
		}
		add = {
			value = wc_titanforged_reorder_county_base
			multiply = {
				value = council_friend_impact_percentage
				divide = 100
			}
		}
	}
}

wc_titanforged_reorder_county_same_culture_bonus = {
	value = 0
	if = {
		limit = {
			scope:county = {
				culture = scope:councillor_liege.culture
			}
		}
		add = {
			value = wc_titanforged_reorder_county_base
			multiply = {
				value = 50
				divide = 100
			}
		}
	}
}

wc_titanforged_reorder_county_development_penalty = {
	value = 0
	if = {
		limit = { exists = scope:county }
		add = {
			value = wc_titanforged_reorder_county_base
			multiply = {
				subtract = scope:county.development_level
				divide = 50
			}
		}
	}
}

wc_titanforged_reorder_county_unraised_army_bonus = {
	value = 0
	if = {
		limit = {
			scope:councillor_liege = { has_raised_armies = no }
		}
		add = {
			value = wc_titanforged_reorder_county_base
			multiply = {
				value = scope:councillor_liege.current_military_strength
				divide = 10000
			}
			max = 2
		}
	}
}

wc_titanforged_reorder_county_total = {
	value = wc_titanforged_reorder_county_base
	add = wc_titanforged_reorder_county_skill_bonus
	add = wc_titanforged_reorder_county_friend_bonus
	add = wc_titanforged_reorder_county_best_friend_bonus
	add = wc_titanforged_reorder_county_rival_bonus
	add = wc_titanforged_reorder_county_nemesis_bonus
	add = wc_titanforged_reorder_county_same_culture_bonus
	add = wc_titanforged_reorder_county_same_faith_bonus
	add = wc_titanforged_reorder_county_development_penalty
	add = wc_titanforged_reorder_county_unraised_army_bonus
}
