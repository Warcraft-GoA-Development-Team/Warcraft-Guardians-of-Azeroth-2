sanctify_artifact_interaction = {
	category = interaction_category_religion
	icon = sanctify_artifact
	interface_priority = 35

	desc = sanctify_artifact_interaction_desc

	target_type = artifact
	target_filter = actor_artifacts
	show_all_targets = no

	greeting = positive
	notification_text = sanctify_artifact_notification_text
	answer_accept_key = SANCTIFY_ARTIFACT_ACCEPT
	answer_reject_key = SANCTIFY_ARTIFACT_REJECT

	cost = {
		piety = {
			if = {
				limit = {
					scope:target = {
						rarity = illustrious
					}
				}
				value = scope:actor.major_piety_value
			}
			if = {
				limit = {
					scope:target = {
						rarity_level <= famed
					}
				}
				value = scope:actor.medium_piety_value
			}
		}
		treasury_or_gold = {
			if = {
				limit = {
					scope:target = {
						rarity = illustrious
					}
				}
				value = scope:actor.medium_gold_value
			}
			if = {
				limit = {
					scope:target = {
						rarity_level <= famed
					}
				}
				value = scope:actor.minor_gold_value
			}
		}
	}

	cooldown = { years = 5 }

	popup_on_receive = yes
	pause_on_receive = yes
	ai_min_reply_days = 1
	ai_max_reply_days = 5

	is_available = {
		has_any_artifact = yes
	}

	is_shown = {
		# Faith has a head of faith or an antipope -- petition them directly. (a head of faith uses sanctify_artifact_hof_interaction instead)
		trigger_if = {
			limit = { scope:actor.religious_head_or_challenger ?= scope:recipient }
			scope:recipient != scope:actor
		}
		# No head of faith -- petition the actor's court chaplain instead.
		trigger_else = {
			scope:actor = {
				cp:councillor_court_chaplain ?= scope:recipient
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			pam_antipope_clerical_request_trigger = { RECIPIENT = scope:recipient }
			custom_tooltip = {
				text = sanctify_artifact_same_faith_as_clergy_tt
				faith = scope:recipient.faith
			}
			is_imprisoned = no
			piety_level >= 2
			NOT = {
				has_character_modifier = promised_pilgrimage_to_clergy_modifier
			}
			custom_tooltip = {
				text = sanctify_artifact_has_eligible_artifact_tt
				any_character_artifact = {
					NOT = { is_sanctified_for_faith = scope:actor.faith }
					sanctify_artifact_is_eligible_trigger = {
						RITE = scope:actor.rite
					}
				}
			}
		}
	}

	can_be_picked_artifact = {
		scope:target = {
			custom_tooltip = {
				text = sanctify_artifact_not_already_sanctified_tt
				NOT = { is_sanctified_for_faith = scope:actor.faith }
			}
			custom_tooltip = {
				text = sanctify_artifact_tenet_allows_tt
				sanctify_artifact_is_eligible_trigger = {
					RITE = scope:actor.rite
				}
			}
		}
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
		}
	}

	send_option = {
		is_valid = {
			scope:actor = { has_usable_hook = scope:recipient }
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_shown = {
			has_pam_dlc_trigger = yes
			pam_offer_pilgrimage_option_shown_trigger = { ACTOR = scope:actor }
		}
		is_valid = {
			pam_offer_pilgrimage_option_valid_trigger = { ACTOR = scope:actor }
		}
		flag = offer_pilgrimage
		localization = OFFER_PILGRIMAGE_INTERACTION_OPTION
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
	send_options_exclusive = no

	on_accept = {
		scope:target = {
			if = {
				limit = {
					is_holy_relic = no
				}
				# Add a small modifier to artifact if not already a relic potentially
				add_artifact_modifier = sanctified_artifact_modifier
			}
			sanctify_artifact_flavor_effect = { PRIEST = scope:recipient }
		}
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = sanctify_artifact_interaction_accept_toast
				left_icon = scope:recipient
				right_icon = scope:target

				scope:target = {
					add_sanctified_for_faith = scope:actor.faith
				}
			}
		}
		scope:recipient = {
			add_piety = minor_piety_gain
		}
		pam_offer_pilgrimage_effect = {
			ACTOR = scope:actor
			CLERIC = scope:recipient
		}
	}

	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = sanctify_artifact_interaction_decline_toast
				left_icon = scope:recipient
				right_icon = scope:target
			}
		}
	}

	ai_accept = {
		base = 0

		# Warcraft
		master_thrall_modifier = yes
		# End Warcraft

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:actor
			RECIPIENT = scope:recipient
		}

		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}

		modifier = { # Hook spent
			desc = SCHEME_WEAK_HOOK_USED
			scope:hook = yes
			add = 100
		}

		modifier = {
			add = pam_offer_pilgrimage_acceptance_value
			scope:offer_pilgrimage = yes
			desc = OFFER_PILGRIMAGE_INTERACTION_OPTION_REASON
		}

		modifier = { # Different rite -- the clergyman dislikes blessing it
			desc = sanctify_artifact_ai_different_rite
			scope:actor.rite != scope:recipient.rite
			add = -25
		}

		modifier = { # Highly divergent rite -- borderline heretical
			desc = sanctify_artifact_ai_high_divergence
			scope:actor.rite != scope:recipient.rite
			scope:actor.rite = {
				"divergence(scope:recipient.rite)" > faith.heresy_threshold
			}
			add = -50
		}

		modifier = { # Their rite does not consider this a valid sanctification target
			desc = sanctify_artifact_ai_my_rite_not_proper_to_sanctify
			scope:target = {
				NOT = {
					sanctify_artifact_is_eligible_trigger = {
						RITE = scope:recipient.rite
					}
				}
			}

			add = -50
		}

		modifier = { # Excommunicated petitioner -- refused
			desc = sanctify_artifact_ai_excommunicated
			scope:actor = { has_trait = excommunicated }
			add = -500
		}

		modifier = {
			add = {
				value = 5
				multiply = scope:actor.piety_level
			}
			scope:actor.piety_level > 1
			desc = I_AM_PIOUS
		}

		modifier = {
			add = {
				value = 10
				multiply = scope:actor.num_virtuous_traits
			}
			scope:actor.num_virtuous_traits > 0
			desc = I_AM_VIRTUOUS
		}

		modifier = {
			add = {
				value = -10
				multiply = scope:actor.num_sinful_traits
			}
			scope:actor.num_sinful_traits > 0
			desc = I_AM_SINFUL
		}

		modifier = {
			add = 25
			desc = WE_ARE_FRIENDS
			scope:recipient = { has_relation_friend = scope:actor }
		}

		modifier = {
			add = 50
			desc = WE_ARE_BEST_FRIENDS
			scope:recipient = { has_relation_best_friend = scope:actor }
		}

		modifier = {
			add = -100
			desc = WE_ARE_RIVALS
			scope:recipient = { has_relation_rival = scope:actor }
		}

		modifier = {
			add = -500
			desc = WE_ARE_NEMESES
			scope:recipient = { has_relation_nemesis = scope:actor }
		}

		modifier = {
			add = 10
			desc = YOU_ARE_MY_DYNASTY_MEMBER
			scope:recipient = {
				is_lowborn = no
				dynasty = scope:actor.dynasty
			}
		}

		modifier = {
			add = 15
			desc = YOU_ARE_MY_HOUSE_MEMBER
			scope:recipient = {
				is_lowborn = no
				house = scope:actor.house
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}

# The head of faith blesses an artifact of any same-faith character (including themselves) directly -- no petition, no acceptance.
sanctify_artifact_hof_interaction = {
	category = interaction_category_religion
	icon = sanctify_artifact
	interface_priority = 35
	common_interaction = no

	desc = sanctify_artifact_hof_interaction_desc

	target_type = artifact
	target_filter = recipient_artifacts
	show_all_targets = no

	cost = {
		piety = {
			if = {
				limit = {
					scope:target = {
						rarity = illustrious
					}
				}
				value = scope:actor.major_piety_value
			}
			if = {
				limit = {
					scope:target = {
						rarity_level <= famed
					}
				}
				value = scope:actor.medium_piety_value
			}
			multiply = 1.5
		}
		treasury_or_gold = {
			if = {
				limit = {
					scope:target = {
						rarity = illustrious
					}
				}
				value = scope:actor.medium_gold_value
			}
			if = {
				limit = {
					scope:target = {
						rarity_level <= famed
					}
				}
				value = scope:actor.minor_gold_value
			}
			multiply = 1.5
		}
	}

	cooldown_against_recipient = { years = 5 }

	is_shown = {
		scope:recipient = {
			religious_head_or_challenger ?= scope:actor
			faith ?= scope:actor.faith
			has_any_artifact = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
		}
		scope:recipient = {
			custom_tooltip = {
				text = sanctify_artifact_has_eligible_artifact_tt
				any_character_artifact = {
					NOT = { is_sanctified_for_faith = scope:actor.faith }
					sanctify_artifact_is_eligible_trigger = {
						RITE = scope:actor.rite
					}
				}
			}
		}
	}

	can_be_picked_artifact = {
		scope:target = {
			custom_tooltip = {
				text = sanctify_artifact_not_already_sanctified_tt
				NOT = { is_sanctified_for_faith = scope:actor.faith }
			}
			custom_tooltip = {
				text = sanctify_artifact_tenet_allows_tt
				sanctify_artifact_is_eligible_trigger = {
					RITE = scope:actor.rite
				}
			}
		}
	}

	auto_accept = yes

	on_accept = {
		scope:target = {
			if = {
				limit = {
					is_holy_relic = no
				}
				add_artifact_modifier = sanctified_artifact_modifier
			}
			sanctify_artifact_flavor_effect = { PRIEST = scope:actor }
		}
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = sanctify_artifact_interaction_accept_toast
				left_icon = scope:recipient
				right_icon = scope:target

				scope:target = {
					add_sanctified_for_faith = scope:actor.faith
				}
				if = {
					limit = {
						scope:recipient = { this != scope:actor }
					}
					scope:recipient = {
						add_opinion = {
							target = scope:actor
							modifier = sanctified_my_artifact_opinion
						}
					}
				}
			}
		}
		if = {
			limit = {
				scope:recipient = {
					this != scope:actor
					is_ai = no
				}
			}
			scope:recipient = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = sanctify_artifact_hof_interaction_recipient_toast
					left_icon = scope:actor
					right_icon = scope:target

					show_as_tooltip = {
						scope:target = {
							add_sanctified_for_faith = scope:actor.faith
						}
						add_opinion = {
							target = scope:actor
							modifier = sanctified_my_artifact_opinion
						}
					}
				}
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}
