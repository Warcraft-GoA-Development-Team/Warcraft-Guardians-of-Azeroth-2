# Raw effect {} blocks for the same on_action don't merge across files - only the most-recently-loaded survives.
# Use on_actions = { list } to accumulate instead. Demon liegges auto-raise newly-gained vassals as undead thralls.
on_vassal_gained = {
	on_actions = {
		wc_balnazzar_on_vassal_gained_actions
	}
}

wc_balnazzar_on_vassal_gained_actions = {
	effect = {
		if = {
			limit = {
				root = { has_trait = being_demon }
				NOT = { scope:vassal = { has_trait = being_undead } }
			}
			scope:vassal = {
				raise_undead_faith_only_effect = { MASTER = root }
				add_opinion = { modifier = loyal_undead target = root }
				# Use _no_notification variant: become_undead_effect leaks memory, can't cascade on entire vassal trees.
				every_vassal_or_below = {
					limit = {
						is_alive = yes
						NOT = { has_trait = being_undead }
					}
					raise_undead_faith_only_no_notification_effect = { MASTER = liege }
					add_opinion = { modifier = loyal_undead target = liege }
				}
			}
		}
	}
}

# Stratholme taken by conquest while disguised triggers wc_balnazzar_story.1007.
on_title_gain = {
	on_actions = {
		wc_balnazzar_on_title_gain_actions
	}
}

wc_balnazzar_on_title_gain_actions = {
	effect = {
		if = {
			limit = {
				has_character_flag = wc_balnazzar_disguise_active
				scope:title = title:c_stratholme
				OR = {
					scope:transfer_type = flag:conquest
					scope:transfer_type = flag:conquest_holy_war
					scope:transfer_type = flag:conquest_claim
					scope:transfer_type = flag:conquest_populist
				}
			}
			trigger_event = {
				id = wc_balnazzar_story.1007
				days = 3
			}
		}
	}
}

# Redirect pattern avoids clobbering other on_title_lost hooks. Losing Lordaeron triggers chain immediately.
on_title_lost = {
	on_actions = {
		wc_balnazzar_on_title_lost_actions
	}
}

wc_balnazzar_on_title_lost_actions = {
	effect = {
		if = {
			limit = {
				scope:title = title:c_lordaeron
				exists = global_var:balnazzar_dreadlord_story
				global_var:balnazzar_dreadlord_story = { story_owner = root }
				NOT = { has_character_flag = wc_balnazzar_disguise_active }
				NOT = { has_character_flag = balnazzar_went_to_ground }
			}
			trigger_event = {
				id = wc_balnazzar_story.1002
				days = { 1 3 }
			}
		}
	}
}

# on_title_lost misses vassalization at kingdom/empire tier. on_vassal_change's root is the vassal, filtered with exists = liege.
on_vassal_change = {
	on_actions = {
		wc_balnazzar_on_vassal_change_actions
	}
}

wc_balnazzar_on_vassal_change_actions = {
	effect = {
		if = {
			limit = {
				exists = liege
				exists = global_var:balnazzar_dreadlord_story
				global_var:balnazzar_dreadlord_story = { story_owner = root }
				NOT = { has_character_flag = wc_balnazzar_disguise_active }
				NOT = { has_character_flag = balnazzar_went_to_ground }
			}
			# Call effect directly, not via .1001 popup - conquest vassalization must convert him immediately.
			balnazzar_go_to_ground_effect = yes
		}
	}
}

# Alexandros dying while marked (wc_balnazzar_force_renault_as_assassin_effect's
# murder scheme) advances the story to .1006. exists = scope:killer excludes
# natural/disease deaths - only an actual killing should advance the plot.
on_death = {
	on_actions = {
		wc_balnazzar_on_death_actions
	}
}

wc_balnazzar_on_death_actions = {
	effect = {
		if = {
			limit = {
				this = character:60017 # Alexandros Mograine
				has_character_flag = wc_balnazzar_alexandros_marked_for_murder
				exists = scope:killer
			}
			remove_character_flag = wc_balnazzar_alexandros_marked_for_murder
			if = {
				limit = { exists = global_var:balnazzar_dreadlord_story }
				global_var:balnazzar_dreadlord_story.story_owner = {
					trigger_event = {
						id = wc_balnazzar_story.1006
						days = { 3 7 }
					}
				}
			}
		}
	}
}
