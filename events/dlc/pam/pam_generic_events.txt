# Child doubts their goodness
scripted_trigger pam_generic_events_0002_valid_child_trigger = {
	is_alive = yes
	is_courtier_of = root
	NOT = { has_relation_rival = root }
	rite = root.rite
	age > 5
	age < 19
	basic_is_available_ai = yes
	OR = {
		num_sinful_traits > 0
		spiritual_fulfillment < 0
	}
}

# Friend/chaplain comforts you after a family death
scripted_trigger pam_generic_events_0004_recent_family_death_memory_trigger = {
	OR = {
		has_memory_type = relative_died
		has_memory_type = spouse_died
	}
	memory_age_years < 5
	exists = memory_participant:dead_relation
}

pam_generic_events.0004 = {
	type = character_event
	title = pam_generic_events.0004.t
	desc = pam_generic_events.0004.desc
	theme = christian
	override_background = { reference = temple }
	widgets = {
		widget = { gui = event_window_widget_vfx_left_character_flickering_candlelight_dark container = foreground_shader_vfx_container }
		widget = { gui = event_window_widget_vfx_background_night_scene container = foreground_shader_vfx_container }
		widget = { gui = event_window_widget_vfx_fog container = foreground_shader_vfx_container }
		widget = { gui = event_window_widget_vfx_left_character_night_scene container = foreground_shader_vfx_container }
		widget = { gui = event_window_widget_vfx_right_character_night_scene container = foreground_shader_vfx_container }
	}

	left_portrait = {
		character = root
		animation = sadness
	}
	right_portrait = {
		character = scope:comforter
		animation = personality_compassionate
		outfit_tags = { no_headgear }
		camera = camera_event_right_away_3_4
	}
	lower_center_portrait = {
		trigger = { exists = scope:deceased }
		character = scope:deceased
		animate_if_dead = yes
	}

	cooldown = { years = 10 }

	trigger = {
		has_pam_dlc_trigger = yes
		is_available = yes
		age >= 10
		any_memory = {
			pam_generic_events_0004_recent_family_death_memory_trigger = yes
		}
		OR = {
			any_relation = {
				type = friend
				is_available_ai_adult = yes
			}
			cp:councillor_court_chaplain ?= {
				is_available_ai_adult = yes
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			add = 0.5
			any_relation = {
				type = friend
			}
		}
		modifier = {
			add = 0.25
			any_secret = {
				type = secret_murder
				secret_target = {
					is_alive = no
					OR = {
						is_close_family_of = root
						is_spouse_of_even_if_dead = root
					}
				}
			}
		}
	}

	immediate = {
		# Prefer mourning someone you secretly murdered
		if = {
			limit = {
				any_secret = {
					type = secret_murder
					secret_target ?= {
						is_alive = no
						time_since_death = { years <= 5 }
						OR = {
							is_close_family_of = root
							is_spouse_of_even_if_dead = root
						}
					}
				}
			}
			random_secret = {
				type = secret_murder
				limit = {
					secret_target = {
						is_alive = no
						time_since_death = { years <= 5 }
						OR = {
							is_close_family_of = root
							is_spouse_of_even_if_dead = root
						}
					}
				}
				save_scope_as = murder_secret
				secret_target = { save_scope_as = deceased }
			}
		}
		else_if = {
			limit = { NOT = { exists = scope:deceased } }
			random_memory = {
				limit = {
					pam_generic_events_0004_recent_family_death_memory_trigger = yes
				}
				memory_participant:dead_relation = {
					save_scope_as = deceased
				}
			}
		}
		if = {
			limit = {
				any_relation = {
					type = best_friend
					is_available_ai_adult = yes
				}
			}
			random_relation = {
				type = best_friend
				limit = { is_available_ai_adult = yes }
				save_scope_as = comforter
			}
		}
		else_if = {
			limit = {
				any_relation = {
					type = friend
					is_available_ai_adult = yes
				}
			}
			random_relation = {
				type = friend
				limit = { is_available_ai_adult = yes }
				save_scope_as = comforter
			}
		}
		else = {
			cp:councillor_court_chaplain = { save_scope_as = comforter }
		}
	}

	option = { # Confess the murder
		name = pam_generic_events.0004.a
		trigger = {
			exists = scope:murder_secret
		}
		add_internal_flag = special
		scope:murder_secret = { remove_secret = yes }
		stress_and_fulfillment_impact = {
			base = massive_stress_impact_gain
			humble = minor_stress_impact_loss
			callous = minor_stress_impact_gain
			sadistic = minor_stress_impact_gain
		}
		ai_chance = {
			base = 50
			ai_value_modifier = {
				ai_zeal = 0.5
				ai_compassion = 0.25
			}
			modifier = {
				factor = 2
				has_trait = humble
			}
			modifier = {
				factor = 0
				has_trait = callous
			}
			modifier = {
				factor = 0
				has_trait = sadistic
			}
		}
	}

	option = { # Quote scripture
		name = pam_generic_events.0004.b
		trigger = {

			}

		change_spiritual_fulfillment = minor_spiritual_fulfillment_value
		stress_and_fulfillment_impact = {
			zealous = minor_stress_impact_loss
			cynical = minor_stress_impact_gain

			}
