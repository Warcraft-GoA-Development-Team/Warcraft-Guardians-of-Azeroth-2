# Historical heresy outbreaks (By God Alone component 3) for GoA.
# Callers: pam_heresy_c3_eligible_trigger (pam_heresy_triggers.txt) and
# events/wc_events/wc_heresy_historical_events.txt.

##### Router eligibility (faith scope) #####

wc_heresy_historical_eligible_trigger = {
	this = faith:holy_light
	OR = {
		wc_heresy_seline_eligible_trigger = yes
		wc_heresy_damned_eligible_trigger = yes
		wc_heresy_scarlet_eligible_trigger = yes
	}
}

# Seline's Shadow: first outbreak 583-591 (Second War), resurgence any time after
wc_heresy_seline_eligible_trigger = {
	NOT = { exists = global_var:wc_heresy_seline_cooldown }

	OR = {
		wc_heresy_seline_is_resurgence_trigger = yes
		exists = global_var:wc_heresy_debug_ignore_windows
		AND = {
			current_date >= 583.1.1
			current_date < 592.1.1
		}
	}
}

# Cult of the Damned: first outbreak 598-602 (Kel'Thuzad's recruitment), resurgence any time after
wc_heresy_damned_eligible_trigger = {
	NOT = { exists = global_var:wc_heresy_damned_cooldown }

	OR = {
		wc_heresy_damned_is_resurgence_trigger = yes
		exists = global_var:wc_heresy_debug_ignore_windows
		AND = {
			current_date >= 598.1.1
			current_date < 603.1.1
		}
	}
}

# Scarlet zeal: only once the Scarlet Crusade exists (the Balnazzar story converts its first members)
wc_heresy_scarlet_eligible_trigger = {
	NOT = { exists = global_var:wc_heresy_scarlet_cooldown }

	faith:scarletism = {
		any_faith_ruler = { highest_held_title_tier >= tier_county }
	}
}

##### Resurgence (any scope) #####

wc_heresy_seline_is_resurgence_trigger = {
	exists = global_var:wc_heresy_seline_first_fired
}

wc_heresy_damned_is_resurgence_trigger = {
	exists = global_var:wc_heresy_damned_first_fired
}

wc_heresy_scarlet_is_resurgence_trigger = {
	exists = global_var:wc_heresy_scarlet_first_fired
}

##### Home regions (province scope) #####

wc_heresy_seline_region_trigger = {
	geographical_region = world_eastern_kingdoms_azeroth
}

wc_heresy_damned_region_trigger = {
	geographical_region = world_eastern_kingdoms_lordaeron
}

wc_heresy_scarlet_region_trigger = {
	geographical_region = world_eastern_kingdoms_lordaeron
}

# Province scope. Dispatches on scope:heresy_key.
wc_heresy_in_home_region_trigger = {
	trigger_if = {
		limit = { scope:heresy_key = flag:seline }
		wc_heresy_seline_region_trigger = yes
	}
	trigger_else_if = {
		limit = { scope:heresy_key = flag:damned }
		wc_heresy_damned_region_trigger = yes
	}
	trigger_else = {
		wc_heresy_scarlet_region_trigger = yes
	}
}

##### Characters #####

# A lore founder (Seline 1620, Kel'Thuzad 59100, Isillien 60550) can lead the outbreak
wc_heresy_founder_usable_trigger = {
	is_alive = yes

	is_ai = yes
	is_adult = yes
	is_imprisoned = no

	NOT = { faith.religious_head ?= this }

	NOT = { faith = $TARGET$ }
}

# Players who get the flavor chain: landed, origin faith, near the outbreak
wc_heresy_audience_trigger = {
	is_ai = no

	is_landed = yes
	exists = capital_county

	NOT = { this = scope:founder }

	faith = scope:origin_faith
	NOT = { wc_heresy_player_is_insider_trigger = yes }

	OR = {
		capital_county.title_province = { wc_heresy_in_home_region_trigger = yes }
		capital_county = {
			squared_distance = {
				target = scope:epicenter_ruler.capital_county
				value <= squared_distance_medium
			}
		}
	}
}

# Already in the heresy, openly or in secret
wc_heresy_player_is_insider_trigger = {
	OR = {
		faith = scope:heresy_faith
		secret_faith ?= scope:heresy_faith
	}
}

# A courtier the Scarlet inquisitors can accuse (wc_heresy_historical.3002)
wc_heresy_pyre_accused_trigger = {
	is_ai = yes
	is_adult = yes
	is_imprisoned = no

	is_councillor = no
	NOT = { is_close_or_extended_family_of = root }

	NOT = { faith = faith:scarletism }
}
