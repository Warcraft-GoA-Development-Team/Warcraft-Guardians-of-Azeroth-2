drust_crisis.1 = {
	type = character_event
	title = drust_crisis.1.t
	desc = drust_crisis.1.desc
	theme = intrigue
	override_background = wc_waycrest_manor

	left_portrait = {
		character = scope:sick_spouse
		animation = sick
	}

	right_portrait = {
		character = scope:worried_spouse
		animation = grief
	}

	# Not even death will do us part

		name = drust_crisis.1.a

		scope:worried_spouse = {
			add_trait = being_undead
			set_character_faith = faith:throsic
		}
		scope:sick_spouse = {
			add_trait = being_undead
			set_character_faith = faith:throsic
		}

		scope:sick_spouse = { trigger_event = drust_crisis.2 }

		title:d_drustvar.holder = {
			# Gives troops
			heartsbane_spawn_starting_troops_effect = yes
			heartsbane_spawn_starting_troops_effect = yes

            every_child = { banish = yes }
			
			# Turns court undead
			every_courtier_or_guest = {
				become_undead_no_notification_effect = yes
				set_character_faith = faith:throsic
			}
		}

		# Independence stuff goes here
		title:d_drustvar.holder = {
			declare_war_on_kt_heartsbane_effect = yes
		}

		# Notify the Admiral
		title:k_kul_tiras.holder = { trigger_event = drust_crisis.3 }

		ai_chance = {
			base = 1
		}
	}

	# I refuse
	option = {
		name = drust_crisis.1.b

		add_character_flag = heartsbane_coven_declined_flag

		custom_tooltip = {
			text = drust_stops_the_event_chain
			remove_global_variable = heartsbane_coven_event_var
			remove_global_variable = drustvar_crisis_ongoing_var	# Blocks summon_gorak_tul
		}

		ai_chance = {
			base = 0
		}
	}
}

# Hidden Event for AI war declaration
drust_crisis.8 = { 
	hidden = yes

	trigger = {
		is_ai = yes
	}

	immediate = {
		debug_log = "drust_incursion_log - Invasion Event fired"
		save_scope_as = gorak_tul
		
		if = {
			limit = {
				scope:gorak_tul = {
					is_at_war = no
					any_character_to_title_neighboring_county = {
						title_in_drust_warpath_trigger = yes
						save_temporary_scope_as = possible_target
						OR = {
							AND = {
								holder = {
									top_liege != this
								}
								holder.top_liege = {
									gorak_tul_can_war_trigger = yes
								}
							}
							holder = {
								top_liege = this
								gorak_tul_can_war_trigger = yes
							}
						}
					}
				}
			}
			debug_log = "drust_incursion_log - Invasion Succeeded Check"
			scope:gorak_tul = {
				random_character_to_title_neighboring_county = {
					limit = {
						title_in_drust_warpath_trigger = yes
						save_temporary_scope_as = possible_target
						OR = {
							AND = {
								holder = {
									top_liege != this
								}
								holder.top_liege = {
									gorak_tul_can_war_trigger = yes
								}
							}
							holder = {
								top_liege = this
								gorak_tul_can_war_trigger = yes
							}
						}
					}
					save_scope_as = newtarget
					if = {
						limit = {
							holder = {
								top_liege != this
							}
						}
						holder.top_liege = { save_scope_as = newenemy }
						debug_log = "drust_incursion_log - Invasion Target found"
					}
					else = {
						holder = { save_scope_as = newenemy }
						debug_log = "drust_incursion_log - Invasion Target found"
					}
					
					weight = { # Weighting
						base = 100
						modifier = { # Eliminating the hated human settlers comes first
							empire = title:e_kul_tiras
							factor = 1.5
						}
						modifier = { # Moving deeper inland comes next
							empire = title:e_arathor
							factor = 0.8
						}
						modifier = { # Dealing with the rest of the humans
							empire = title:e_lordaeron
							factor = 0.5
						}
						modifier = { # Challenge the Dwarves
							empire = title:e_khaz_modan
							factor = 0.25
						}
						modifier = { # Tribals in general are much less important
							holder = { has_government = tribal_government }
							factor = 0.5
						}
					}
				}
				# Finally, declare the war
				if = {
					limit = {
						can_declare_war = {
							casus_belli = mongol_invasion_war
							defender = scope:newenemy
							target_titles = { scope:newtarget.kingdom }
						}
					}
					start_war = {
						cb = mongol_invasion_war
						target = scope:newenemy
						target_title = scope:newtarget.kingdom
					}
					debug_log = "drust_incursion_log - Invasion begun!"
				}
			}
		}
	}
}

#Gorak Tul invades - Triggered for AI
drust_crisis.9 = {
	hidden = yes
	scope = none

	trigger = {
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:summon_gorak_tul
			}
		}
	}

	immediate = {	
		debug_log = "gorak_tul_log - Spawning Gorak Tul"
		debug_log_date = yes

		set_global_variable = drust_incursion_happened		# Locks the crisis opening event if it didn't happen

		# Summon Gorak Tul
		create_character = {
			template = gorak_tul_character_template
			save_scope_as = gorak_tul
			location = province:6023
		}

		# Set up Gorak Tul's assets
		scope:gorak_tul = {
			gorak_tul_effect_ai = yes
			gorak_tul_create_court_effect = yes
		}
		
		# Start the story
		scope:gorak_tul = {
			create_story = story_drust_incursion
		}
	}
}
