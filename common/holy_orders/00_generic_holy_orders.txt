#MILITARY
ho_generic_place_name_order = {

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = {
					has_variable = {
						name = military_order_game_start_founder
					}
				}
			}
			always = no
		}
		trigger_else = {
			exists = scope:province
			#Less suitable for Islam, which has ghazis
			NOT = {
				scope:founder.religion = religion:islam_religion
			}
			NOT = {
				scope:founder.faith = {
					any_faith_holy_order = {
						title = {
							has_variable = place_name_order
							var:place_name_order = scope:province
						}
					}
				}
			}
		}
	}

	weight = { value = 0.25 }

	effect = {
		set_variable = {
			name = place_name_order
			value = scope:province
		}
	}

	name = holy_order_generic_place_name_order
	government = holy_order_government
	type = military
}

ho_generic_culture_order = {

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = {
					has_variable = {
						name = military_order_game_start_founder
					}
				}
			}
			always = no
		}
		trigger_else = {
			scope:founder.culture = scope:leader.culture
			NOT = {
				scope:founder.religion = religion:islam_religion
			}
			NOT = {
				scope:founder.faith = {
					any_faith_holy_order = {
						title = {
							has_variable = culture_order
							var:culture_order = scope:founder.culture
						}
					}
				}
			}
		}
	}

	weight = {
		value = 1
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							title = {
								has_variable = culture_order
							}
						}
					}
				}
			}
			multiply = 3
		}
	}

	effect = {
		set_variable = {
			name = culture_order
			value = scope:founder.culture
		}
		scope:title = {
			set_variable = {
				name = order_heritage
				value = scope:founder.culture
			}
		}
		scope:leader = {
			if = {
				limit = {
					NOT = {
						has_realm_law = cultural_holy_order_succession_law
					}
				}
				add_realm_law = cultural_holy_order_succession_law
			}
		}
	}

	name = holy_order_culture_order
	government = holy_order_government
	type = military
}

		trigger_if = {
			limit = {
				scope:founder = {
					has_variable = {
						name = military_order_game_start_founder
					}
				}
			}
			always = no
		}
		trigger_else = {
			exists = scope:province
		}
		#These names are a bit unusual for western christians, we have more normal ones
		NOT = {
			scope:founder.rite ?= {
				rite_has_doctrine = special_doctrine_is_western_christian_faith
			}
		}
	}

	weight = {
		value = 1
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							title = {
								has_variable = military_place_an_order
							}
						}
					}
				}
			}
			multiply = 3
		}
	}

	effect = {
		set_variable = {
			name = military_place_an_order
		}
	}

	name = holy_order_place_an_order
	government = holy_order_government
	type = military
}

ho_generic_culture_monks = {

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = {
					has_variable = {
						name = monastic_game_start_founder
					}
				}
			}
			always = no
		}
		trigger_else = {
			scope:leader = { is_male = yes }
			scope:founder.culture = scope:leader.culture
			NOT = {
				scope:founder.faith = {
					any_faith_holy_order = {
						title = {
							has_variable = culture_monks
							var:culture_monks = scope:founder.culture
						}
					}
				}
			}
			scope:founder = {
				OR = {
					religion = religion:buddhism_religion
					religion = religion:christianity_religion
				}
			}
		}
	}

	weight = {
		value = 1
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							title = {
								has_variable = culture_monks
							}
						}
					}
				}
			}
			multiply = 3
		}
	}

	effect = {
		set_variable = {
			name = culture_monks
			value = scope:founder.culture
		}
	}

	name = holy_order_culture_monks
	government = monastic_holy_order_government
	type = monastic
}

ho_generic_house_order = {
	trigger = {
		#Stop game start creation
		trigger_if = {
			limit = {
				scope:founder = {
					has_variable = {
						name = monastic_game_start_founder
					}
				}
			}
			always = no
		}
		trigger_else = {
			exists = scope:founder.house
			NOT = {
				scope:leader.faith = {
					any_faith_holy_order = {
						title = {
							has_variable = house_order
							var:house_order = scope:founder.house
						}
					}
				}
			}
		}
	}

	effect = {
		scope:founder.house = {
			save_scope_as = house_order
		}
		set_variable = {
			name = house_order
			value = scope:house_order
		}
	}
	
	name = "holy_order_house_order"
	government = monastic_holy_order_government
	type = monastic
	weight = 1
}
