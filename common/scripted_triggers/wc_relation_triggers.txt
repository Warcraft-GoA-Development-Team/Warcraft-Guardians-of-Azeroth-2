wc_is_obedient = {
    has_relation_flag = {
        target = $TARGET$
        relation = $RELATION$
        flag = obedient
    }
}

# Add relationships that can force obedience here
is_obedient_any = {
    OR = {
        wc_is_obedient = { TARGET = $TARGET$ RELATION = master }
        wc_is_obedient = { TARGET = $TARGET$ RELATION = summoner }
    }
}

has_any_positive_relation_trigger = {
    OR = {
        any_relation = { type = potential_friend count >= 1 }
        any_relation = { type = best_friend count >= 1 }
        any_relation = { type = soulmate count >= 1 }
        any_relation = { type = friend count >= 1 }
        any_relation = { type = lover count >= 1 }
        any_relation = { type = crush count >= 1 }
        any_relation = { type = ward count >= 1 }
        any_relation = { type = guardian count >= 1 }
        any_relation = { type = mentor count >= 1 }
        any_relation = { type = student count >= 1 }
        any_relation = { type = elder count >= 1 }
        any_relation = { type = disciple count >= 1 }
        any_spouse = { count >= 1 }
        any_child = { count >= 1 }
        any_sibling = { count >= 1 }
        any_parent = { is_alive = yes }
    }
}

has_any_negative_relation_trigger = {
    OR = {
        any_relation = { type = grudge count >= 1 }
        any_relation = { type = rival count >= 1 }
        any_relation = { type = potential_rival count >= 1 }
        any_relation = { type = victim count >= 1 }
        any_relation = { type = nemesis count >= 1 }
        any_relation = { type = bully count >= 1 }
    }
}
