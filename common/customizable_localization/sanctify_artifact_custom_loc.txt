# Scope: the clergyman performing the rite (sanctify_artifact_flavor_effect)
SanctifiedArtifactPrefix = {
	type = character

	text = {
		localization_key = sanctified_artifact_prefix_venerated
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_ancestral } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_sacred
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_natural } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_anointed
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_sacrificial } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_radiant
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_regal } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_sanctified
		trigger = { rite = { rite_has_parameter = sanctify_artifacts_humble } }
	}

	# Zoroastrianism belongs to rf_eastern, so it has to be checked before that family below.
	text = {
		localization_key = sanctified_artifact_prefix_purified
		trigger = {
			faith = {
				OR = {
					is_in_family = rf_light # Warcraft
					is_in_family = rf_life
					is_in_family = rf_order
				}
			}
		}
	}

	text = {
		localization_key = sanctified_artifact_prefix_hallowed
		trigger = { faith = { is_in_family = rf_light } }
	}

	text = {
		localization_key = sanctified_artifact_prefix_auspicious
		trigger = {
			faith = {
				OR = {
					is_in_family = rf_elemental # Warcraft
					is_in_family = rf_life
				}
			}
		}
	}

	text = {
		localization_key = sanctified_artifact_prefix_blessed
		fallback = yes
	}
}
