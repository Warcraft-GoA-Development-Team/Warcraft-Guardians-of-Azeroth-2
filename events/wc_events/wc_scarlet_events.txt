scarlet.1005 = { # This event creates HoF title without requiring the need to have held a Holy Site
	hidden = yes

	trigger = {
		faith = {
			has_doctrine = doctrine_spiritual_head

			# Title must exist but be uncreated.
			exists = religious_head_title
			NOT = { exists = religious_head_title.holder }			
		}
	}

	immediate = {
		faith = {
			religious_head_title = {
				save_scope_as = my_hof_title
			}
		}

		if = {
			limit = {
				any_living_character = { is_sally_whitemane_trigger = yes }
			}
			character:60572 = {
				save_scope_as = new_religious_head
			}
		}
		else = {	
			create_character = {
				location = root.capital_province
				age = { 30 50 }
				gender_female_chance = root_faith_clergy_gender_female_chance
				trait = education_learning_4
				faith = root.faith
				culture = root.culture
				learning = { 14 22 }
				save_scope_as = new_religious_head
				after_creation = {
					trigger_race_giving_no_gene_effect = yes # Warcraft, Assigns race trait
				}
			}
		}

		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:my_hof_title = {
			change_title_holder = {
				holder = scope:new_religious_head
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change		
	}
}
