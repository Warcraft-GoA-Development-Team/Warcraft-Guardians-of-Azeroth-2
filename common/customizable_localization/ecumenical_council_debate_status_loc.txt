# Custom-loc helpers that resolve the ETenetApprovalStatus of a tenet
# the activity host has selected for an Ecumenical Council debate.
#
# Called from localization as:
#   [Activity.GetHost.Custom('GetEcumenicalDebateTenet1Status')]
#
# Returns the matching concept-link label (e.g. [core_tenet_status|E]).

GetEcumenicalDebateTenet1Status = {
	type = character

	text = {
		setup_scope = {
			involved_activity ?= {
				activity_tenet:1 ?= { save_scope_as = ec_debate_tenet }
			}
		}
		trigger = {
			faith.main_rite = {
				has_tenet_status = { tenet = scope:ec_debate_tenet status = core }
			}
		}
		localization_key = faith_tenet_status_core_name
	}

	text = {
		setup_scope = {
			involved_activity ?= {
				activity_tenet:1 ?= { save_scope_as = ec_debate_tenet }
			}
		}
		trigger = {
			faith.main_rite = {
				has_tenet_status = { tenet = scope:ec_debate_tenet status = permitted }
			}
		}
		localization_key = faith_tenet_status_permitted_name
	}

	text = {
		setup_scope = {
			involved_activity ?= {
				activity_tenet:1 ?= { save_scope_as = ec_debate_tenet }
			}
		}
		trigger = {
			faith.main_rite = {
				has_tenet_status = { tenet = scope:ec_debate_tenet status = prohibited }
			}
		}
		localization_key = faith_tenet_status_prohibited_name
	}

	text = {
		setup_scope = {
			involved_activity ?= {
				activity_tenet:1 ?= { save_scope_as = ec_debate_tenet }
			}
		}
		trigger = {
			faith.main_rite = {
				has_tenet_status = { tenet = scope:ec_debate_tenet status = known }
			}
		}
		localization_key = faith_tenet_status_known_name
	}

	# Fallback: anything that isn't core/permitted/prohibited/known is unknown.
	text = {
		localization_key = faith_tenet_status_unknown_name
		fallback = yes
	}
}

	type = activity

	# Tenet debates (first & second): current_debate_proposal holds the tenet-status flag.
	text = {
		trigger = {
			always = no # Warcraft - ecumenical council emptied in GoA, never set
		}
		localization_key = ecumenical_council_proposal_make_core
	}
	text = {
		trigger = {
			always = no # Warcraft - ecumenical council emptied in GoA, never set
		}
		localization_key = ecumenical_council_proposal_permit
	}
	text = {
		trigger = {
			always = no # Warcraft - ecumenical council emptied in GoA, never set
		}
		localization_key = ecumenical_council_proposal_prohibit
	}
	text = {
		trigger = {
			always = no # Warcraft - ecumenical council emptied in GoA, never set
		}
		localization_key = ecumenical_council_proposal_learn
	}
	# The doctrine is stored in var:third_debate_proposal, not current_debate_proposal
	text = {
		trigger = {
			always = no # Warcraft - ecumenical council emptied in GoA, never set
		}
