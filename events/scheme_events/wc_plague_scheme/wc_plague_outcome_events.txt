plague_outcome.0001 = {
	type = character_event
	content_source = dlc_GOA
	hidden = yes

	immediate = {
		scope:target.capital_county = { save_scope_as = plague_center }

		scope:scheme = {
			every_scheme_agent_character = {
				trigger_event_plague_outcome_0002_effect = { OWNER = scope:owner TARGET = scope:target PLAGUE_CENTER = scope:plague_center }
			}
		}
		trigger_event_plague_outcome_0002_effect = { OWNER = scope:owner TARGET = scope:target PLAGUE_CENTER = scope:plague_center }
	}
}

# Owner POV — Success
plague_outcome.0002 = {
	type = character_event
	content_source = dlc_GOA
	window = scheme_successful_event
	title = plague_outcome.0002.title
	desc = {
		desc = plague_outcome.0002.desc
		
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:scheme_discovered
				}
				desc = plague_outcome.0002.desc.exposed_scheme
			}
			desc = plague_outcome.0002.desc.not_exposed_scheme
		}
	}
	theme = plague
	right_portrait = scope:target

	immediate = {
		# General owner success effects.
		scheme_owner_plague_success_effect = yes

		# Launch the plague
		plague_success_effect = yes
	}

	option = {
		name = plague_outcome.0002.a
	}

	# Fire the rest of the outcome.
	after = {
		scope:target = { trigger_event = plague_outcome.0003 }
	}
}

# Owner POV — Failure
plague_outcome.0004 = {
	type = character_event
	content_source = dlc_GOA
	window = scheme_failed_event
	title = plague_outcome.0004.title
	desc = {
		# Opening
		random_valid = {
			desc = plague_outcome.0004.desc.failure_declaration_1
			desc = plague_outcome.0004.desc.failure_declaration_2
			desc = plague_outcome.0004.desc.failure_declaration_3
		}
		# What was our plan?
		desc = plague_outcome.0004.desc
		# Secrecy.
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:scheme_discovered
				}
				desc = plague_outcome.0004.desc.exposed_scheme
			}
			desc = plague_outcome.0004.desc.not_exposed_scheme
		}
	}
	theme = plague
	left_portrait = {
		character = scope:owner
		animation = rage
	}
	right_portrait = {
		character = scope:target
		animation = shock
	}

	immediate = {
		show_as_tooltip = { plague_failure_effect = yes }
	}

	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { exists = scope:scheme_discovered }
						desc = plague_outcome.0004.a.discovered
					}
					desc = plague_outcome.0004.a
				}
			}
		}
		scope:scheme = { end_scheme = yes }
	}

	option = {
		name = plague_outcome.0004.b
		trigger = {
			NOT = { exists = scope:scheme_discovered }
		}
		restart_plague_scheme_effect = yes
	}

	# Fire the rest of the outcome.
	after = {
		scope:target = { trigger_event = plague_outcome.0005 }
	}
}
