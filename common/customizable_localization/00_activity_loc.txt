GetCharacterContestStatus = {
	type = character

	text = {
		trigger = {
			is_in_guest_subset = {
				name = spectator
			}
		}
		localization_key = locale_window_status_spectating
	}

	text = {
		trigger = {
			involved_activity = {
				is_current_phase_active = yes
			}
			is_in_guest_subset = {
				name = qualified
			}
			exists = var:progress_to_victory
		}
		localization_key = locale_window_status_participating
	}

	text = {
		trigger = {
			involved_activity = {
				is_current_phase_active = yes
			}
			is_in_guest_subset = {
				name = contestant
			}
			not = {
				is_in_guest_subset = {
					name = qualified
				}
			}
		}
		localization_key = locale_window_status_not_qualify
	}

	text = {
		trigger = {
			involved_activity = {
				is_current_phase_active = yes
			}
			is_in_guest_subset = {
				name = qualified
			}
			NOR = {
				exists = var:progress_to_victory 
				has_character_flag = resigned_contest_flag
				has_character_flag = disqualified_contest_flag 
			}
		}
		localization_key = locale_window_status_knocked_out
	}

	text = {
		trigger = {
			involved_activity = {
				is_current_phase_active = yes
			}
			is_in_guest_subset = {
				name = qualified
			}
			has_character_flag = disqualified_contest_flag
		}
		localization_key = locale_window_status_discalified
	}

	text = {
		trigger = {
			involved_activity = {
				is_current_phase_active = yes
			}
			is_in_guest_subset = {
				name = qualified
			}
			has_character_flag = resigned_contest_flag
		}
		localization_key = locale_window_status_resign
	}

	text = {
		trigger = {
			involved_activity = {
				is_current_phase_active = yes
			}
			exists = var:progress_to_victory
		}
		localization_key = locale_window_status_participating
	}
}

GetIsolationReason = {
	type = character

	text = {
		localization_key = game_concept_earthquake
		trigger = {
			any_character_situation = {
				situation_type = natural_disaster_earthquake
				tpg_character_in_any_situation_sub_region_trigger = yes
				NOT = { situation_current_phase = recovery }
			}
		}
	}

	text = {
		localization_key = game_concept_flood
		trigger = {
			any_character_situation = {
				situation_type = natural_disaster_flood
				tpg_character_in_any_situation_sub_region_trigger = yes
				NOT = { situation_current_phase = recovery }
			}
		}
	}

	# we don't allow this for now, but keep this entry for full consistency in case someday it used.
	text = {
		localization_key = game_concept_epidemic
		trigger = {
			OR = {
				capital_county ?= {
					OR = {
						# Own capital
						has_province_with_epidemic = { intensity = any }
						# Neighboring own capital
						any_neighboring_county = {
							has_province_with_epidemic = { intensity = any }
						}
					}
				}
				# Somewhere in realm
				any_sub_realm_county = {
					has_province_with_epidemic = { intensity = any }
				}
			}
		}
	}

	text = {
		localization_key = game_concept_disaster
		fallback = yes
	}
}
