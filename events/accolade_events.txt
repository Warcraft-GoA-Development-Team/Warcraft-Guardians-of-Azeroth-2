#Accolade rank gain notification
# by Jason Cantalini
accolade.0001 = {
	hidden = yes

	trigger = {
		exists = acclaimed_knight
		has_dlc_feature = accolades
	}

	immediate = {
		scope:changing_accolade = {
			acclaimed_knight = {
				save_scope_as = acclaimed_knight
			}
		}
		if = {
			limit = {
				scope:changing_accolade = {
					accolade_rank >= 5
				}
			}
			send_interface_message = {
				type = msg_accolade_gained_rank
				title = accolade.0001.notification_high
				desc = accolade_rank_unlock_notification_tooltip_high
				left_icon = scope:acclaimed_knight
			}
		}
		else = {
			send_interface_message = {
				type = msg_accolade_gained_rank
				title = accolade.0001.notification_low
				desc = accolade_rank_unlock_notification_tooltip_low
				left_icon = scope:acclaimed_knight
			}
		}
	}
}

#Accolade rank loss notification
# by Jason Cantalini
accolade.0002 = {
	hidden = yes

	trigger = {
		has_dlc_feature = accolades
	}

	immediate = {
		if = {
			limit = {
				exists = scope:changing_accolade.acclaimed_knight
			}
			scope:changing_accolade = {
				acclaimed_knight = {
					save_scope_as = acclaimed_knight
				}
			}
			if = {
				limit = {
					scope:changing_accolade = {
						accolade_rank >= 5
					}
				}
				send_interface_message = {
					type = msg_accolade_lost_rank
					title = accolade.0002.notification_high
					left_icon = scope:acclaimed_knight
					desc = {
						first_valid = {
							triggered_desc = {
								trigger = {
									scope:acclaimed_knight = {
										is_alive = yes
									}
								}
								desc = accolade_rank_loss_notification_tooltip_high
							}
							desc = accolade_rank_loss_notification_tooltip_high_dead
						}
					}
				}
			}
			else_if = {
				limit = {
					scope:changing_accolade = {
						accolade_rank >= 1
					}
				}
				send_interface_message = {
					type = msg_accolade_lost_rank
					title = accolade.0002.notification_low
					left_icon = scope:acclaimed_knight
					desc = {
						first_valid = {
							triggered_desc = {
								trigger = {
									scope:acclaimed_knight = {
										is_alive = yes
									}
								}
								desc = accolade_rank_loss_notification_tooltip_low
							}
							desc = accolade_rank_loss_notification_tooltip_low_dead
						}
					}
				}
			}
			else = {
				send_interface_message = {
					type = msg_accolade_lost_rank
					title = accolade.0002.notification_low
					left_icon = scope:acclaimed_knight
					desc = {
						first_valid = {
							triggered_desc = {
								trigger = {
									scope:acclaimed_knight = {
										is_alive = yes
									}
								}
								desc = accolade_rank_loss_notification_tooltip_no_rank
							}
							desc = accolade_rank_loss_notification_tooltip_no_rank_dead
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				NOT = { exists = scope:changing_accolade.acclaimed_knight }
				scope:changing_accolade = { accolade_rank >= 1 }
			}
			send_interface_message = {
				type = msg_accolade_lost_rank
				title = accolade.0002.notification_empty
				desc = accolade_rank_loss_notification_tooltip_empty
			}
		}
		else = {
			send_interface_message = {
				type = msg_accolade_lost_rank
				title = accolade.0002.notification_low
				desc = accolade_rank_loss_notification_tooltip_empty_no_rank
			}
		}
	}
}

#Create a sellsword band led by your Acclaimed Knight
accolade.0100 = {
	type = character_event
	title = accolade.0100.t
	desc = accolade.0100.desc
	theme = war

	left_portrait = {
		character = root
		animation = frontend_left_idle
		camera = camera_event_left_to_the_left
	}
	right_portrait = {
		character = scope:recipient
		animation = sword_coup_degrace
	}

	#NAMING WIDGET
	widget = {
		gui = "event_window_widget_enter_text"
		container = "dynamic_birth_name"
		controller = {
			type = text
			data = {
				key = new_laamp_title_name
				### Accolade's Name
				default = {
					localization_key = accolade_mercenary_accolade_name
					trigger = {
						exists = scope:accolade
						root = { is_ai = no }
					}
				}
				default = {
					localization_key = accolade_mercenary_kings_men_name
					trigger = {
						root = {
							prestige_level >= 3
						}
						scope:acclaimed_knight = {
							is_male = yes
						}
					}
				}
				default = {
					localization_key = accolade_mercenary_knightly_company_name
					trigger = {
						root = {
							culture = {
								OR = {
									has_graphical_western_culture_group_trigger = yes
									has_graphical_iberian_culture_group_trigger = yes
								}
							}
							NOR = {
								government_has_flag = government_is_tribal
								government_has_flag = government_is_nomadic
								AND = {
									NOT = {
										government_has_mechanic = administrative
									}
									culture = {
										NOT = { has_cultural_era_or_later = culture_era_high_medieval }
									}
								}
							}
						}
					}
				}
				default = {
					localization_key = accolade_honor_band_name
					trigger = {
						root = {
							OR = {
								government_has_flag = government_is_tribal
								government_has_flag = government_is_nomadic
								AND = {
									NOT = {
										government_has_mechanic = administrative
									}
									culture = {
										NOT = { has_cultural_era_or_later = culture_era_high_medieval }
									}
								}
							}
						}
					}
				}
				default = {
					localization_key = accolade_rulers_swords_name
					trigger = {
						root = {
							OR = {
								government_has_flag = government_is_tribal
								AND = {
									NOT = {
										government_has_mechanic = administrative
									}
									culture = {
										NOT = { has_cultural_era_or_later = culture_era_high_medieval }
									}
								}
							}
							NOT = { government_has_flag = government_is_nomadic }
						}
					}
				}
				default = {
					localization_key = accolade_culture_riders_name
					trigger = {
						root = {
							government_has_flag = government_is_nomadic
						}
					}
				}
				default = {
					localization_key = accolade_culture_company_name
					trigger = {
						root = {
							NOR = {
								government_has_flag = government_is_tribal
								government_has_flag = government_is_nomadic
								AND = {
									NOT = {
										government_has_mechanic = administrative
									}
									culture = {
										NOT = { has_cultural_era_or_later = culture_era_high_medieval }
									}
								}
							}
						}
					}
				}
				default = {
					localization_key = accolade_free_knights_name
					#Only triggers when King X's Men doesn't
					trigger = {
						OR = {
							root = {
								prestige_level < 3
							}
							scope:acclaimed_knight = {
								is_male = no
							}
						}
					}
				}
			}
		}
		setup_scope = {
			scope:acclaimed_knight = { save_scope_as = text_target }
		}
	}

	immediate = {
		scope:acclaimed_knight = { save_scope_as = acclaimed_mercenary }
		play_music_cue = "mx_cue_combat_2"
	}
	option = {
		trigger = {
			is_ai = no
			has_dlc_feature = landless_playable
		}
		name = accolade.0100.a_special
		add_internal_flag = dangerous
		custom_tooltip = switch_to_acclaimed_mercenary_tt
		if = {
			limit = {
				NOT = {
					scope:recipient.dynasty ?= scope:actor.dynasty
				}
			}
			custom_tooltip = danger_dynasty_abandon_tt
		}
		save_scope_as = switch
		add_character_flag = {
			flag = switching_to_knight
			days = 20
		}
		
		
		ai_chance = {
			base = 0
		}
	}
	option = {
		name = accolade.0100.a

		scope:accolade_owner = {
			custom_tooltip = accolade_mercenary_gold_tt
			scope:acclaimed_knight = {
				set_variable = {
					name = acc_merc_patron
					value = scope:accolade_owner
				}
			}
		}
	}

	after = {
		# clear_variable_list = realm_title_names # Warcraft - the list is never filled in GoA

		scope:acclaimed_knight = {
			if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_mercenary_accolade_name
				}
				add_character_flag = {
					flag = accolade_mercenary_accolade_name
					days = 20
				}
			}
			else_if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_mercenary_kings_men_name
				}
				add_character_flag = {
					flag = accolade_mercenary_kings_men_name
					days = 20
				}
			}
			else_if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_mercenary_knightly_company_name
				}
				add_character_flag = {
					flag = accolade_mercenary_knightly_company_name
					days = 20
				}
			}
			else_if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_honor_band_name
				}
				add_character_flag = {
					flag = accolade_honor_band_name
					days = 20
				}
			}
			else_if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_rulers_swords_name
				}
				add_character_flag = {
					flag = accolade_rulers_swords_name
					days = 20
				}
			}
			else_if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_culture_riders_name
				}
				add_character_flag = {
					flag = accolade_culture_riders_name
					days = 20
				}
			}
			else_if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_free_knights_name
				}
				add_character_flag = {
					flag = accolade_free_knights_name
					days = 20
				}
			}
			else_if = {
				limit = {
					var:new_laamp_title_name ?= flag:accolade_culture_company_name
				}
				add_character_flag = {
					flag = accolade_culture_company_name
					days = 20
				}
			}
			else = {
				add_character_flag = {
					flag = new_laamp_title_name_entered_text
					days = 20
				}
			}
			accolade_to_mercenary_effect = yes
		}
	}
}
#Sellsword band sends you gold
accolade.0110 = {
	hidden = yes
	immediate = {
		send_interface_message = {
			type = msg_accolade_merc
			title = accolade_merc_gold.t
			left_icon = scope:accolade_merc
			desc = accolade_merc_gold.desc
			scope:accolade_merc = {
				#Send a little bit of gold
				if = {
					limit = {
						gold > 0
					}
					if = {
						limit = {
							gold >= five_times_monumental_value
						}
						pay_short_term_gold = {
							target = root
							gold = massive_gold_value
						}
					}
					else_if = {
						limit = {
							gold >= monumental_gold_value
						}
						pay_short_term_gold = {
							target = root
							gold = medium_gold_value
						}
					}
					else = {
						pay_short_term_gold = {
							target = root
							gold = minor_gold_value
						}
					}
				}
				#If you have no gold, secretly gain it and then send it
				else = {
					hidden_effect = {
						add_gold = minor_gold_value
					}
					pay_short_term_gold = {
						target = root
						gold = minor_gold_value
					}
				}
			}
		}
	}
}
