init_life_spells_effect = {
    add_to_global_variable_list = { name = life_spells target = flag:natures_cure }
    add_to_global_variable_list = { name = life_spells target = flag:hotw }
    add_to_global_variable_list = { name = life_spells target = flag:wildgrowth }
    add_to_global_variable_list = { name = life_spells target = flag:regrowth }
    add_to_global_variable_list = { name = life_spells target = flag:invigorate }
    add_to_global_variable_list = { name = life_spells target = flag:dreamwalk }
    add_to_global_variable_list = { name = life_spells target = flag:druid_forms }
    add_to_global_variable_list = { name = life_spells target = flag:shred }
    add_to_global_variable_list = { name = life_spells target = flag:thrash }
    add_to_global_variable_list = { name = life_spells target = flag:ironbark }
}


cast_natures_cure_effect = {
    get_recipient_effect = { SPELL = natures_cure }

    if = {
        limit = {
            exists = scope:this_recipient
        }
        # Look at ranks first
        if = {
            limit = {
                OR = { # current_spell_rank is for tooltips
                    AND = {
                        exists = var:natures_cure_rank
                        var:natures_cure_rank = 3
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank = 3
                    }
                }
            }
            scope:this_recipient = {
                if = {
                    limit = {
                        target_has_magic_illness_trigger = yes
                    }
                    remove_magic_illness_effect =  yes
                }
                else = {
                    custom_tooltip = wc_dispel_no_effect_tooltip
                }
            }
        }
        else_if = {
            limit = {
                OR = {
                    AND = {
                        exists = var:natures_cure_rank
                        var:natures_cure_rank = 2
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank = 2
                    }
                }
            }
            scope:this_recipient = {
                if = {
                    limit = {
                        target_has_major_illness_trigger = yes
                    }
                    remove_major_illness_effect = yes
                }
                else = {
                    custom_tooltip = wc_dispel_no_effect_tooltip
                }
            }
        }
        else_if = {
            limit = {
                OR = {
                    AND = {
                        exists = var:natures_cure_rank
                        var:natures_cure_rank = 1
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank = 1
                    }
                }
            }
            scope:this_recipient = {
                if = {
                    limit = {
                        target_has_illness_trigger = yes
                    }
                    remove_illness_effect =  yes
                }
                else_if = {
                    limit = {
                        target_has_poison_trigger = yes
                    }
                    remove_poison_effect = yes
                }
                else = {
                    custom_tooltip = wc_dispel_no_effect_tooltip
                }
            }
        }
    }
}

cast_hotw_effect = {
    get_recipient_effect = { SPELL = hotw }

    scope:this_recipient ?= {
        root = {
            save_temporary_scope_value_as = {
                name = duration
                value = wc_spell_hotw_duration_days_value
            }
        }
        add_character_modifier = {
            modifier = wc_hotw_modifier
            days = wc_current_spell_duration
        }
        set_variable = {
            name = spell_removed
            value = 0
        }
        custom_tooltip = {
            text = wc_spell_hotw_tt
            every_in_list = {
                variable = spells_cd
                save_scope_as = cd_spell
                
                if = {
                    limit = {
                        scope:this_recipient = {
                            var:spell_removed < 5
                            NOT = {
                                spell_is_ritual_trigger = { SPELL = scope:cd_spell }
                            }
                        }
                    }
                    scope:this_recipient = {
                        change_variable = {
                            name = spell_removed
                            add = 1
                        }
                        execute_scoped_spell_effect = { SPELL = scope:cd_spell EFFECT = remove_spell_cooldown_effect }
                    }
                }
            }
        }
        remove_variable = spell_removed
    }

}

cast_wildgrowth_effect = {
    get_recipient_effect = { SPELL = wildgrowth }
    save_temporary_scope_value_as = {
        name = duration
        value = wc_spell_wildgrowth_duration_days_value
    }

    scope:this_recipient ?= {
        if = {
            limit = {
                tier = tier_barony
            }
            title_province = {
                add_province_modifier = {
                    modifier = wc_wildgrowth_province_modifier
                    days = wc_current_spell_duration
                }
            }
        }
        else_if = {
            limit = {
                tier = tier_county
            }
            add_county_modifier = {
                modifier = wc_wildgrowth_county_modifier
                days = wc_current_spell_duration
            }
        }
        else_if = {
            limit = {
                tier = tier_duchy
            }
            every_de_jure_county = {
                add_county_modifier = {
                    modifier = wc_wildgrowth_county_modifier
                    days = wc_current_spell_duration
                }
            }
        }
    }
}

cast_regrowth_effect = {
    if = {
        limit = {
            NOT = { exists = scope:cancel }
        }
        get_recipient_effect = { SPELL = regrowth }
    
        scope:this_recipient ?= {
            add_character_modifier = {
                modifier = wc_regrowth_modifier
            }
        }
    }
    else = {
        var:regrowth_recipient = {
            remove_character_modifier = wc_regrowth_modifier
        }
    }
}

cast_invigorate_effect = {
    get_recipient_effect = { SPELL = invigorate }

    scope:this_recipient ?= {
        remove_injury_effect = yes
    }
}

cast_dreamwalk_effect = {
    save_temporary_scope_value_as = {
        name = duration
        value = wc_spell_dreamwalk_duration_days_value
    }
    
    add_character_modifier = {
        modifier = wc_dreamwalk_modifier
        days = wc_current_spell_duration
    }

    custom_tooltip = {
        text = wc_dreamwalk_tt
        every_held_title ?= {
            every_in_de_jure_hierarchy = {
                if = {
                    limit = {
                        tier = tier_county
                    }
                    add_county_modifier = {
                        modifier = wc_dreamwalk_county_modifier
                        days = wc_current_spell_duration
                    }
                }
                else_if = {
                    limit = {
                        tier = tier_barony
                    }
                    title_province = {
                        every_character_in_location = {
                            limit = {
                                NOT = {
                                    has_character_modifier = wc_dreamwalk_modifier
                                }
                            }
                            apply_character_dreamwalk_effect = yes
                        }
                    }
                }
            }
        }
    }

    if = {
        limit = {
            is_landless_adventurer = yes 
        }
        every_courtier = {
            limit = {
                NOT = {
                    has_character_modifier = wc_dreamwalk_modifier
                }
            }
            apply_character_dreamwalk_effect = yes
        }
    }
}

apply_character_dreamwalk_effect = {
    if = {
        limit = {
            NOT = {
                has_character_modifier = wc_dreamwalk_modifier
            }
        }
        send_interface_toast = {
            title = wc_dreamwalk_title
            type = event_toast_effect_good
            add_character_modifier = {
                modifier = wc_dreamwalk_modifier
            }
        }
        trigger_event = {
            on_action = wc_dreamwalk_removal_pulse
            months = 1
        }
    }
}

cast_druid_forms_effect = {
    if = {
        limit = {
            exist = scope:cancel
        }
        remove_druid_forms_effect = yes
    }
    else = {
        custom_tooltip = cast_druid_forms_effect_tooltip
    }
}

change_druid_form_effect = {
    wc_change_mana_by_value_effect = {
		CHANGE = subtract
		VALUE = 10
	}
    remove_druid_forms_effect = yes 
    switch = {
        trigger = $FORM$
        flag:cat = {
            add_trait = cat_form
        }
        flag:bear = {
            add_trait = bear_form
        }
        flag:stag = {
            add_trait = stag_form
        }
        flag:treant = {
            add_trait = treant_form
        }
    }
}

remove_druid_forms_effect = {
    if = {
        limit = {
            has_trait = cat_form
        }
        remove_trait = cat_form
    }
    if = {
        limit = {
            has_trait = bear_form
        }
        remove_trait = bear_form
        if = {
            limit = {
                has_character_modifier = wc_ironbark_modifier 
            }
            cancel_channeling_spell_effect = { SPELL = ironbark RANK = 0 TYPE = flag:poop } 
        }
    }
    if = {
        limit = {
            has_trait = stag_form
        }
        remove_trait = stag_form
    }
    if = {
        limit = {
            has_trait = treant_form
        }
        remove_trait = treant_form
    }
}

cast_shred_effect = {
    get_recipient_effect = { SPELL = shred }

    scope:this_recipient ?= {
        apply_poison_effect = { POISON = raked }
    }
}

cast_thrash_effect = {
    root = {
        save_temporary_scope_value_as = {
            name = duration
            value = wc_spell_thrash_duration_days_value
        }
    }
    if = {
        limit = {
            root = {
                OR = {
                    AND = {
                        exists = var:thrash_rank
                        var:thrash_rank <= 2
                    }
                    AND = {
                        exists = var:current_spell_rank
                        var:current_spell_rank <= 2
                    }
                }
            }
        }
        get_recipient_effect = { SPELL = thrash }
        scope:this_recipient ?= {
            apply_thrash_character_effect = yes
        }
    }
    else = {
        scope:this_recipient ?= {
            army_commander = { apply_thrash_character_effect ?= yes }
            deplete_army_by_percentage = root.thrash_decrease_percent
        }
    }
}

apply_thrash_character_effect = {
    add_character_modifier = {
        modifier = wc_thrashed_modifier
        days = wc_current_spell_duration
    }

    random = {
        chance = 50 
        random_list = {
            50 = {
                trigger = { 
                    NOT = { has_trait = wounded_3 }
                }
                increase_wounds_effect = { REASON = magic }
            }
            50 = {
                trigger = { 
                    NOT = { has_trait = raked }
                }
                apply_poison_effect = { POISON = raked }
            }
        }
    }
}

cast_ironbark_effect = {
