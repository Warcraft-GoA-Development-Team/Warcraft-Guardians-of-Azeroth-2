digging_up_the_dead_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/digging_up_the_dead.dds" }
	sort_order = 10

	title = digging_up_the_dead_decision
	desc = digging_up_the_dead_decision_desc

	ai_check_interval = 1
	decision_group_type = major
	selection_tooltip = digging_up_the_dead_decision_tooltip

	is_shown = {
		game_start_date < 605.6.6

		any_owned_story = {
			type = story_path_of_the_damned
			var:story_stage = 2
		}

		# The urn should exists
		exists = global_var:path_of_the_damned_story.var:enchanted_urn

		NOT = { # You can't do that as lich king
			has_trait = lich_king
			has_title = title:h_scourge
		}

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:digging_up_the_dead_decision
			}
		}
	}

	is_valid = {
		AND = {
			is_landed = yes
			scourge_quelthalas_frostmourne_check_trigger = yes
			has_trait = champion_of_the_scourge
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	cost = {}

	effect = {
		global_var:path_of_the_damned_story.var:enchanted_urn = {
			# Save scopes
			save_scope_as = enchanted_urn
			artifact_owner = { save_scope_as = war_target }

			# Digging up the Dead events
			root = {
				add_personal_artifact_claim = scope:enchanted_urn # Claim
				start_war = { # War
					cb = artifact_war
					target = scope:war_target
				}
			}

			# Holy orders join the war
			custom_tooltip = digging_up_the_dead_decision_holy_orders_join_effect.tt
			scope:war_target = {
				trigger_event = {
					id = wc_path_of_the_damned.1903
					days = 1
				}
			}
		}

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:digging_up_the_dead_decision
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}

		hidden_effect = {
			if = {
				limit = { any_living_character = { is_kelthuzad_trigger = yes } }

				}
			}

			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:invade_scourge_quelthalas_decision
			}
		}
	}

resurrect_kelthuzad_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/resurrection_as_lich.dds" }
	sort_order = 10

	ai_check_interval = 1
	decision_group_type = major
	selection_tooltip = resurrect_kelthuzad_decision_tt

	is_shown = {
		game_start_date < 605.6.6 # Not triggered in Frozen Throne

		any_owned_story = {
			type = story_path_of_the_damned
			var:story_stage = 3
		}

		NOT = { # You can't do that as lich king
			has_trait = lich_king
			has_title = title:h_scourge
		}

		is_target_in_global_variable_list = { # Only after Quelthalas is defeated
			name = unavailable_unique_decisions
			target = flag:invade_scourge_quelthalas_decision
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:resurrect_kelthuzad_decision
			}
		}
	}


	cost = {}

	is_valid = {
		AND = {
			is_landed = yes
			OR = {
				has_title = title:c_sunwell_plateau
				title:h_scourge.holder = { has_title = title:c_sunwell_plateau }
				title:h_scourge.holder = { any_vassal_or_below = { has_title = title:c_sunwell_plateau } }
			}
			title:c_sunwell_plateau.title_province = { has_building_or_higher = sunwell_01 }

			scourge_quelthalas_frostmourne_check_trigger = yes
			scourge_quelthalas_phylactery_check_trigger = yes
			has_trait = champion_of_the_scourge
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	effect = {
		global_var:path_of_the_damned_story.var:kelthuzad = { save_scope_as = necromancer }

		custom_tooltip = resurrect_kelthuzad_decision_effect_tt
		show_as_tooltip = { wc_resurrect_CHARACTER_with_STATUS_effect = { CHARACTER = scope:necromancer STATUS = lich } } # Resurrect Kel'Thuzad
		show_as_tooltip = { root = { add_courtier = scope:necromancer } }
		show_as_tooltip = { global_var:path_of_the_damned_story.var:enchanted_urn = { set_owner = scope:necromancer } }

		trigger_event = scourgequel.1011

		add_to_global_variable_list = { # So you cannot spam it
			name = unavailable_unique_decisions
			target = flag:resurrect_kelthuzad_decision
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}

summon_burning_legion_decision = { # By Ikarimonogatari
	picture = { reference = "gfx/interface/illustrations/decisions/wc_summon_burning_legion.dds" }
	sort_order = 10

	ai_check_interval = 1
	decision_group_type = major
	desc = summon_burning_legion_decision_desc

	is_shown = {
		game_start_date < 605.6.6 # Not triggered in Frozen Throne

		any_owned_story = {
			type = story_path_of_the_damned
			var:story_stage = 4
		}

		is_target_in_global_variable_list = { # Only make this available before kelthuzad is spawned
			name = unavailable_unique_decisions
			target = flag:resurrect_kelthuzad_decision
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:summon_burning_legion_decision
			}
		}
	}

	is_valid = {
		AND = {
			is_landed = yes
			scourge_quelthalas_frostmourne_check_trigger = yes
			has_trait = champion_of_the_scourge
		}

		OR = {
			any_courtier = { has_trait = creature_lich }
			any_vassal = { has_trait = creature_lich }
		}
		NOT = { # Don't allow this if Nordrassil has already been destroyed
			title:b_nordrassil.title_province = { has_building = nordrassil_destroyed }
		}
	}

	cost = {}

	cooldown = { years = 2 }

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = {
			text = summon_burning_legion_decision.tt
			if = { # If the Lich King or a fellow vassal holds it, give the title to Arthas
				limit = {
					OR = {
						title:c_dalaran.holder = { has_title = title:h_scourge }
						title:c_dalaran.holder = { is_vassal_of = title:h_scourge.holder }
					}
				}
				trigger_event = burning_legion.0001
			}
			else_if = { # Invade Dalaran
				limit = { NOT = { title:c_dalaran.holder = root } }
				trigger_event = burning_legion.0002
			}
			else = { # Start searching for the Book of Medivh and Summon the Legion
				limit = { title:c_dalaran.holder = root }
				trigger_event = burning_legion.0003
			}
		}
	}

	ai_potential = {
		is_imprisoned = no
		is_capable_adult = yes
		is_at_war = no
	}

	ai_will_do = {
		base = 100
	}
}
