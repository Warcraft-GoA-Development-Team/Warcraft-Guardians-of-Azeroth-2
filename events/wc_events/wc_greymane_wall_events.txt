# 0001 - Discuss building the wall with Godfrey (Chancellor)
wc_greymane_wall_events.0001 = {
    type = character_event
    content_source = dlc_GOA
    title = wc_greymane_wall_events.0001.t
    desc = wc_greymane_wall_events.0001.desc
    theme = martial

    left_portrait = {
        character = root
        animation = personality_cynical
    }
    right_portrait = {
        character  = cp:councillor_chancellor
        animation = disbelief

    }
    trigger = {
        exists = cp:councillor_chancellor
    }

    #Continue the event
    option = {
        name = wc_greymane_wall_events.0001.a
        trigger = {
            exists = cp:councillor_chancellor
            is_liege_or_above_of = title:c_western_strand.holder
            is_liege_or_above_of = title:c_crowley.holder
        }
        trigger_event = {
            id = wc_greymane_wall_events.0002
        }
        ai_chance = {
            base = 100
        }
    }

    #No nobles version (ambermill is a republic county)
    option = {
        name = wc_greymane_wall_events.0001.a
        trigger = {
            exists = cp:councillor_chancellor
            has_title = title:c_crowley
            has_title = title:c_northwood
            has_title = title:c_pyrewood
            has_title = title:c_southpoint
            has_title = title:c_western_strand
        }
        trigger_event = {
            id = wc_greymane_wall_events.0102
        }

        ai_chance = {
            base = 100
        }
    }

    #Change your mind
    option = {
        name = wc_greymane_wall_events.0001.b
        ai_chance = {
            base = 0
       }
    } 
}

# 0002 - Discuss location of the wall with Godfrey (Chancellor)
wc_greymane_wall_events.0002 = {
    type = character_event
    content_source = dlc_GOA
    title = wc_greymane_wall_events.0002.t
    desc = wc_greymane_wall_events.0002.desc
    theme = martial

    left_portrait = {
        character = root
        animation = personality_cynical
    }
    right_portrait = {
        character  = cp:councillor_chancellor
        animation = scheme
    }

    #Build the wall in the north (Lord Marley)
    option = {
        name = wc_greymane_wall_events.0002.a
        trigger = {
            completely_controls = title:c_western_strand
            completely_controls = title:c_southpoint
            completely_controls = title:c_ambermill
            exists = title:c_western_strand.holder
        }
        trigger_event = {
            id = wc_greymane_wall_events.0003
            days = {2 5}
        }
        ai_chance = {
            base = 10
        }
    }

    #Cut Crowley's territory in half
    option = {
        name = wc_greymane_wall_events.0002.b
        trigger = {
            exists = title:c_crowley.holder
        }

        trigger_event = {
            id = wc_greymane_wall_events.0010
            days = {1 3}
       }
        ai_chance = {
            base = 90
        }
    }
      
}

# 0003 - Discuss the wall with Lord Marley (Western Strand holder)
wc_greymane_wall_events.0003 = {
    type = character_event
    content_source = dlc_GOA
    title = wc_greymane_wall_events.0003.t
    desc = wc_greymane_wall_events.0003.desc
    theme = vassal

    left_portrait = {
        character = root
        animation = thinking
    }
    right_portrait = {
        character  = title:c_western_strand.holder
        animation = personality_rational
    }

    #Continue with Marley
    option = {
        name = wc_greymane_wall_events.0003.a
        trigger = {
            is_liege_or_above_of = title:c_western_strand.holder
            is_liege_or_above_of = title:c_southpoint.holder
            is_liege_or_above_of = title:c_ambermill.holder
        }
        show_as_unavailable = {
            always = yes
        }
        pay_short_term_gold = { target = title:c_western_strand.holder gold = 600}
        pay_short_term_gold = { target = title:c_southpoint.holder gold = 600}
        pay_short_term_gold = { target = title:c_ambermill.holder gold = 1200}
        
        title:b_cursewich.title_province = {
            set_holding_type = city_holding
            add_province_modifier = {
                modifier = building_greymane_wall_modifier
                years = 5
            }
        }

        title:b_western_strand.title_province = {
            add_province_modifier = {
                modifier = building_greymane_wall_modifier
                years = 5
            }
        }

        title:b_ambermill.title_province = {
            add_province_modifier = {
                modifier = building_greymane_wall_modifier
                years = 5
            }
        }

        title:b_bridgeview.title_province = {
            add_province_modifier = {
                modifier = building_greymane_wall_modifier
                years = 5
            }
        }
        trigger_event = {
            id = wc_greymane_wall_events.0004
            years = 5
        }

       ai_chance = {
            base = 100
            modifier = {
                add = -101
                gold <= 2400
            }
        }
    }

    #Reconsider Crowley
    option = {
        name = wc_greymane_wall_events.0003.b
        trigger_event = {
            id = wc_greymane_wall_events.0010
            days = {1 3}
       }
        ai_chance = {
            base = 0
            modifier = {
                add = 100
                gold <= 1800
            }
        }
    }
}

#0011 - The Wall is finished, Crowley edition
wc_greymane_wall_events.0011 = {

    type = character_event
    content_source = dlc_GOA
    title = wc_greymane_wall_events.0004.t
    desc = wc_greymane_wall_events.0011.desc
    theme = vassal

    left_portrait = {
        character = root
        animation = personality_cynical
    }
    right_portrait = {
        character  = title:c_crowley.holder
        animation = anger
    }

    immediate = {
        title:c_crowley.holder = {
            add_character_modifier = {
                modifier = dislikes_greymane_wall_modifier
            }   
        } 

        title:c_aderic.title_province = {
            add_province_modifier = {
                modifier = strategic_gerymane_wall_modifier
            }
        }

        title:c_headlands.title_province = {
            add_province_modifier = {
                modifier = strategic_gerymane_wall_modifier
            }
        }

        title:c_galewatch.title_province = {
            add_province_modifier = {
                modifier = strategic_gerymane_wall_modifier
            }
        }

        title:b_crowley.title_province = {
            add_special_building = greymane_wall_02
        }

        title:b_kirk.title_province = {
            add_special_building = greymane_wall_02
        }
    }

    #The wall is done!
    option = {
        name = wc_greymane_wall_events.0011.a
        ai_chance = {
            base = 100
        }
    }
    #Tear down this wall
    option = {
        name = wc_greymane_wall_events.0011.b

        title:c_crowley.holder = {
            remove_character_modifier = dislikes_greymane_wall_modifier
             
        }

        title:c_aderic.title_province = {
            remove_province_modifier = strategic_gerymane_wall_modifier
        }

        title:c_headlands.title_province = {
            remove_province_modifier = strategic_gerymane_wall_modifier
        }

        title:c_galewatch.title_province = {
            remove_province_modifier = strategic_gerymane_wall_modifier
        }

        title:b_crowley.title_province = {
            remove_building = greymane_wall_02
            add_special_building = greymane_wall_01
        }

        title:b_kirk.title_province = {
            remove_building = greymane_wall_02
            add_special_building = greymane_wall_01
        }

        ai_chance = {
            base = 0
        }
    }

}
