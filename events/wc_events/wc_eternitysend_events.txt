eternitysend.0001 = {
    scope = none
    hidden = yes

    immediate = {
        if = {
            limit = { any_living_character = { is_cenarius_trigger = yes } }
            character:410002 = { save_scope_as = cenarius }
        }

        if = {
            limit = { scope:cenarius = { is_alive = yes } }
            trigger_event = { id = eternitysend.0100 }
        }
        else = {
            title:h_burning_legion.holder = {
                start_war = {
                    casus_belli = mongol_invasion_war
                    target = title:e_kaldorei.holder
                    target_title = title:k_hyjal
                }
            }
        }

        hidden_effect = {
            add_to_global_variable_list = {
                name = warcraft_invaders_story
                target = flag:eternitysend_story_started
            }
        }
    }
}

eternitysend.0100 = {
    scope = none
    hidden = yes

    immediate = {
        hidden_effect = {
            if = {
                limit = {
                    any_ruler = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:warsong
                        this = culture.culture_head
                    }
                }
                random_ruler = {
                    limit = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:warsong
                        this = culture.culture_head
                    }
                    save_scope_as = warsong_or_alternative
                }

                scope:warsong_or_alternative = {
                    trigger_event = { id = eternitysend.0200 }
                }
            }
            else_if = {
                limit = {
                    NOT = { exists = scope:warsong_or_alternative }
                    any_ruler = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:durotaric
                        this = culture.culture_head
                    }
                }
                random_ruler = {
                    limit = {
                        is_landed = yes
                        capital_province = { geographical_region = world_kalimdor }
                        has_culture = culture:durotaric
                        this = culture.culture_head
                    }
                    save_scope_as = frostwolf_or_alternative
                }

                scope:frostwolf_or_alternative = {
                    trigger_event = { id = eternitysend.0300 }
                }
            }
            else = {
                trigger_event = { id = eternitysend.0400 }
            }
        }
    }
}
