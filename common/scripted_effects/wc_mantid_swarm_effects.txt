# Mantid Swarming cycle effects. Events: events/wc_events/wc_mantid_swarm_events.txt

# Rolls the next Swarm year from the last one.
wc_swarm_roll_next_effect = {
	set_global_variable = {
		name = wc_next_swarming_year
		value = {
			value = global_var:wc_last_swarming_year
			add = { integer_range = { min = 95 max = 105 } }
		}
	}
}

# Sets up the clock once (game start, or the first yearly pulse of an older save). The last Swarm was ~510.
wc_swarm_init_clock_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = wc_last_swarming_year }
		}
		set_global_variable = {
			name = wc_last_swarming_year
			value = 510
		}
		wc_swarm_roll_next_effect = yes
	}
}

# Called on a successful launch.
wc_swarm_reset_clock_effect = {
	set_global_variable = {
		name = wc_last_swarming_year
		value = current_year
	}
	wc_swarm_roll_next_effect = yes
	remove_global_variable = wc_swarm_delayed_this_cycle
}

# Player head holds the Swarm back 5 years (once per cycle).
wc_swarm_delay_effect = {
	set_global_variable = {
		name = wc_swarm_delayed_this_cycle
		value = yes
	}
	change_global_variable = {
		name = wc_next_swarming_year
		add = 5
	}
	remove_global_variable = wc_swarm_decision_pending
}

# Saves every shared scope the Swarm events use. Re-run in each event: years pass between launch and war end.
# scope:empress = the aqir holder of e_shekmag, else the head of faith.
wc_swarm_save_scopes_effect = {
	faith:yshaarjism = { save_scope_as = swarm_faith }
	title:d_seven_prophets.holder ?= { save_scope_as = swarm_head }
	title:k_shado_pan.holder ?= { save_scope_as = shado_pan_leader }
	title:d_klaxxi.holder ?= { save_scope_as = klaxxi_leader }
	if = {
		limit = {
			exists = title:e_shekmag.holder
			title:e_shekmag.holder = {
				is_alive = yes
				wc_swarm_is_aqir_trigger = yes
			}
		}
		title:e_shekmag.holder = { save_scope_as = empress }
	}
	else_if = {
		limit = { exists = title:d_seven_prophets.holder }
		title:d_seven_prophets.holder = { save_scope_as = empress }
	}
}

# Player picked a target in wc_mantid_swarm.0011 (scope:target_kingdom saved by the option).
wc_swarm_player_launch_effect = {
	hidden_effect = {
		remove_global_variable = wc_swarm_decision_pending
		faith:yshaarjism = { trigger_event = wc_mantid_swarm.0012 }
	}
}

# Downgrades every serpents_spine_wall_02 inside the final target to _01 (vanilla swap pattern: ep3 Hasan decisions).
wc_swarm_breach_wall_effect = {
	global_var:wc_swarm_final_target = {
		every_in_de_jure_hierarchy = {
			continue = { tier > tier_county }
			limit = { tier = tier_county }
			every_county_province = {
				limit = { has_building = serpents_spine_wall_02 }
				remove_building = serpents_spine_wall_02
				add_special_building = serpents_spine_wall_01
			}
		}
	}
}

# Same tick as the war end, while scope:war still exists. OUTCOME = attacker_won | defender_won.
wc_swarm_record_war_end_effect = {
	set_global_variable = {
		name = wc_swarm_outcome
		value = flag:$OUTCOME$
	}
	if = {
		limit = {
			faith:yshaarjism = { exists = great_holy_war.ghw_target_title }
		}
		set_global_variable = {
			name = wc_swarm_final_target
			value = faith:yshaarjism.great_holy_war.ghw_target_title
		}
	}
	else = {
		set_global_variable = {
			name = wc_swarm_final_target
			value = global_var:wc_swarm_target
		}
	}
	if = {
		limit = {
			global_var:wc_swarm_outcome = flag:attacker_won
			global_var:wc_swarm_final_target = { wc_swarm_wall_kingdom_trigger = yes }
		}
		wc_swarm_breach_wall_effect = yes
	}
	scope:defender = { save_scope_as = swarm_defender }
	scope:attacker = { trigger_event = wc_mantid_swarm.0090 }
}

# Ends a Swarm: removes its temporary modifiers and every per-Swarm global.
wc_swarm_cleanup_effect = {
	every_ruler = {
		limit = { has_character_modifier = wc_swarm_stirring }
		remove_character_modifier = wc_swarm_stirring
	}
	every_ruler = {
		limit = { has_character_modifier = wc_wall_rallied }
		remove_character_modifier = wc_wall_rallied
	}
	wc_swarm_wall_counties_to_list_effect = yes
	every_in_list = {
		list = swarm_wall_counties
		limit = { has_county_modifier = wc_wall_reinforced }
		remove_county_modifier = wc_wall_reinforced
	}
	remove_global_variable = wc_swarm_active
	remove_global_variable = wc_swarm_target
	remove_global_variable = wc_swarm_culls
	remove_global_variable = wc_swarm_final_target
	remove_global_variable = wc_swarm_outcome
	remove_global_variable = wc_swarm_decision_pending
}

# After (or instead of) a Culling at the Wall: count it and queue the next one, at most 2 per Swarm.
wc_swarm_cull_requeue_effect = {
	if = {
		limit = { has_global_variable = wc_swarm_active }
		change_global_variable = {
			name = wc_swarm_culls
			add = 1
		}
		if = {
			limit = { global_var:wc_swarm_culls < 2 }
			wc_swarm_save_scopes_effect = yes
			scope:empress ?= {
				trigger_event = {
					id = wc_mantid_swarm.0030
					days = { 180 365 }
				}
			}
		}
	}
}
