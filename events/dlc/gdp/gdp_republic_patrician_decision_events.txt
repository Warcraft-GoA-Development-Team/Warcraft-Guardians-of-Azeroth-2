#Come across some left behind cargo
gdp_republic_patrician_decision.1000 = {
	type = character_event
	window = big_event_window
	title = gdp_republic_patrician_decision.1000.t
	desc = gdp_republic_patrician_decision.1000.desc
	theme = republic
	left_portrait = {
		character = root
		animation = interested
	}
	center_portrait = {
		character = scope:local_character
		animation = storyteller
		camera = camera_event_scheme_center_look_right
		hide_info = yes
	}
	right_portrait = {
		character = scope:local_character_2
		animation = contemplating
		camera = camera_event_scheme_vs_extra_right_look_left
		hide_info = yes
	}
	lower_right_portrait = {
		trigger = { exists = scope:liege_scope }
		character = scope:liege_scope 
	}
	cooldown = { years = 5 }
	trigger = {
		is_available_adult = yes
	}
	immediate = {
		capital_province = { 
			save_scope_as = city_location
			barony.trade_market = {
				if = {
					limit = { 
						any_trade_good_with_need = { count >= 1 }
					}
					random_trade_good_with_need = { 
						limit = {
							save_temporary_scope_as = trade_good_check
							is_breakable_trade_good_trigger = { TRADE_GOOD = scope:trade_good_check }
						}
						save_scope_as = abandoned_trade_good
					}
					save_scope_as = is_needs_trade_good
				}
				else = {
					random_trade_good_type = {
						limit = {
							save_temporary_scope_as = trade_good_check
							is_breakable_trade_good_trigger = { TRADE_GOOD = scope:trade_good_check }
						}
						save_scope_as = abandoned_trade_good
					}
				}
			}
		}
		if = {
			limit = { is_independent_ruler = no }
			liege = { save_scope_as = liege_scope }
		}
		if = {
			limit = {
				any_pool_character = {
					province = scope:city_location
					is_available_healthy_ai_adult = yes
					has_no_particular_noble_roots_trigger = yes
				}
			}
			random_pool_character = {
				province = scope:city_location
				limit = {
					is_available_healthy_ai_adult = yes
					has_no_particular_noble_roots_trigger = yes
				}
				weight = {
					base = 1
					modifier = {
						has_character_flag = created
						add = 1
					}
				}
				save_scope_as = local_character
			}
		}
		else = {
			create_character = {
				template = generic_peasant_character
				location = scope:city_location
				culture = scope:city_location.culture
				faith = scope:city_location.faith
				save_scope_as = local_character
				after_creation = { add_character_flag = created trigger_race_giving_no_gene_effect = yes } # Warcraft
			}
		}
		if = {
			limit = {
				any_pool_character = {
					province = scope:city_location
					is_available_healthy_ai_adult = yes
					has_no_particular_noble_roots_trigger = yes
					this != scope:local_character
				}
			}
			random_pool_character = {
				province = scope:city_location
				limit = {
					is_available_healthy_ai_adult = yes
					has_no_particular_noble_roots_trigger = yes
					this != scope:local_character
				}
				save_scope_as = local_character_2
			}
		}
		else = {
			create_character = {
				template = generic_peasant_character
				location = scope:city_location
				culture = scope:city_location.culture
				faith = scope:city_location.faith
				save_scope_as = local_character_2
				after_creation = { add_character_flag = created trigger_race_giving_no_gene_effect = yes } # Warcraft
			}
		}	
	}
	#Break it to create a deficit?
	option = {
		name = gdp_republic_patrician_decision.1000.a
		flavor = gdp_republic_patrician_decision.1000.a.flavor
		trigger = { exists = scope:is_needs_trade_good }
		patrician_decision_1000_apply_deficit_modifier_effect = yes
		ai_chance = {
			base = 100
		}
	}
	#Sell it!
	option = {
		name = gdp_republic_patrician_decision.1000.b
		add_gold = major_gold_value
		ai_chance = {
			base = 100
		}
	}
	#Gift it to the local liege
	option = {
		name = gdp_republic_patrician_decision.1000.c
		trigger = { exists = scope:liege_scope }
		if = {
			limit = {
				NOT = { has_relation_friend = scope:liege_scope }
			}
			scope:liege_scope = {
				progress_towards_friend_effect = {
					REASON = friend_gifted_trade_goods
					CHARACTER = root
					OPINION = 30
				}
			}
		}
		else = {
			reverse_add_opinion = {
		 		target = scope:liege_scope
		 		modifier = grateful_opinion
		 		opinion = 30
		 	}
		}
		ai_chance = {
			base = 100
			modifier = {
				has_relation_rival = scope:liege_scope
				factor = 0
			}
		}
	}
	#Let's not engage
	option = {
		name = gdp_republic_patrician_decision.1000.d
		add_character_modifier = {
			modifier = gdp_city_stroll_modifier
			years = 5
		}
		stress_impact = {
			base = miniscule_stress_impact_loss
			craven = minor_stress_loss
			paranoid = medium_stress_loss
		}
		ai_chance = {
			base = 200
		}
	}
	after = {
		scope:local_character = {
			silent_disappearance_ai_if_created_effect = yes
		}
		scope:local_character_2 = {
			silent_disappearance_ai_if_created_effect = yes
		}
	}
}
