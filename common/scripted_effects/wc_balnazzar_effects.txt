# Shared by wc_balnazzar_story.1001.a (auto-fired on_title_lost when c_lordaeron actually falls)
# and balnazzar_go_to_ground_decision (manual fallback for players who never lose it in a war -
# the story used to have no way in for them at all, silently softlocking the rest of the chain).
# change_government to landless_adventurer_government alone silently fails - that government
# needs a landless-type duchy title, which create_landless_adventurer_title_effect mints but
# doesn't give up any titles still held. Hand everything to his liege first (same title-transfer
# pattern as vanilla's become_landless_adventurer_decision), so he genuinely walks out with
# nothing, matching .1010's framing.
balnazzar_go_to_ground_effect = {
	# Guards every entry point (on_title_lost, on_vassal_change, .1001's own trigger,
	# balnazzar_go_to_ground_decision) against firing this twice - conquest can trip more than
	# one of those hooks in the same update.
	add_character_flag = balnazzar_went_to_ground

	if = {
		limit = { top_liege != this }
		liege = { save_scope_as = balnazzar_land_recipient }

		every_held_title = {
			title_tier > barony
			limit = {
				OR = {
					is_landless_type_title = no
					is_nomad_title = yes
				}
			}
			add_to_list = balnazzar_relinquished_titles
		}

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = balnazzar_land_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = balnazzar_relinquished_titles
			change_title_holder_include_vassals = {
				holder = scope:balnazzar_land_recipient
				change = scope:balnazzar_land_change
			}
		}
		resolve_title_and_vassal_change = scope:balnazzar_land_change
	}

	create_landless_adventurer_title_effect = {
		REASON = flag:voluntary
		FLAVOR_CHAR = root
	}

	trigger_event = {
		id = wc_balnazzar_story.1010
		days = { 1 3 }
	}
}

# Call on the character wearing nathrezim_disguise (scope:dathrohan in wc_balnazzar_story.1002).
add_dreadlord_disguise_secret_effect = {
	add_secret = {
		type = secret_dreadlord_disguise
		target = this
	}
}

# Mirrors vanilla murder_scheme_prep_effect: routes through the native
# scheme_critical_moments.0002 event (the "spend Advantage points" screen)
# instead of jumping straight to the outcome roll, with follow_up_event
# pointed at our own resolution event instead of murder's.
wc_mograine_assassination_scheme_prep_effect = {
	save_scope_as = scheme
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:wc_mograine_assassination_outcome.0001
	}
	if = {
		limit = { NOT = { exists = scope:suppress_next_event } }
		scheme_owner = {
			trigger_event = scheme_critical_moments.0002
		}
	}
}
