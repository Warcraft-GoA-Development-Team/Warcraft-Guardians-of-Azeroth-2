has_crisis_on_same_continent_trigger = {
    trigger_if = {
        limit = { exists = title:e_horde.holder NOT = { this = title:e_horde.holder } }
        ruler_realm_is_on_same_continent_trigger = { CHARACTER = title:e_horde.holder }
        title:c_stormwind.holder.top_liege = title:e_horde.holder
    }
    trigger_else_if = {
        limit = { exists = title:e_scourge.holder NOT = { this = title:e_scourge.holder } }
        ruler_realm_is_on_same_continent_trigger = { CHARACTER = title:e_scourge.holder }
    }
    trigger_else_if = {
        limit = { exists = title:e_burning_legion.holder NOT = { this = title:e_burning_legion.holder } }
        ruler_realm_is_on_same_continent_trigger = { CHARACTER = title:e_burning_legion.holder }
    }
    trigger_else_if = {
        limit = { exists = title:e_black_empire.holder NOT = { this = title:e_black_empire.holder } }
        ruler_realm_is_on_same_continent_trigger = { CHARACTER = title:e_black_empire.holder }
    }
    trigger_else_if = {
        limit = {
            exists = title:e_pandaria_empire.holder
            NOT = { this = title:e_pandaria_empire.holder }
            title:e_pandaria_empire.holder = {
                is_mogu_thunder_king_trigger = yes
            }
        }
        ruler_realm_is_on_same_continent_trigger = { CHARACTER = title:e_pandaria_empire.holder }
    }
    trigger_else_if = {
        limit = {
            has_global_variable = current_player_crisis
        }
        ruler_realm_is_on_same_continent_trigger = { CHARACTER = global_var:current_player_crisis }
    }
    trigger_else = {
        always = no
    }
}

is_crisis_trigger = {
	OR = {
		is_horde_invader_trigger = yes
		is_mogu_thunder_king_trigger = yes
		is_existential_crisis_trigger = yes
	}
}

# Scourge, Legion, Black Empire
is_existential_crisis_trigger = {
	OR = {
		has_title = title:e_scourge
		has_title = title:e_burning_legion
		has_title = title:e_black_empire
	}
}

is_crisis_title = {
    OR = {
        this = title:e_horde
        this = title:e_scourge
        this = title:e_burning_legion
        this = title:e_black_empire 
        holder = { is_mogu_thunder_king_trigger = yes }
        global_var:current_player_crisis ?= holder
    }
}

can_be_in_coalition_trigger = {
	is_independent_ruler = yes
	OR = {
		is_landed = yes 
		is_landless_adventurer = yes
	}
	is_crisis_trigger = no
    is_incapable_trigger = no 
}

is_in_coalition_against_trigger = {
	custom_description = {
		text = is_in_coalition_against_trigger_loc
		object = $CRISIS_HOLDER$
		
		is_target_in_variable_list = { name = coalition_targets target = $CRISIS_HOLDER$ }
	}
}

is_in_any_coalition_trigger = {
	has_variable_list = coalition_targets
}

has_crisis_coalition_cooldown_trigger = {
	custom_description = {
		text = has_crisis_coalition_cooldown_trigger_loc
		has_character_flag = crisis_coalition_cooldown_flag
	}
}

is_crisis_active_trigger = {
	trigger_if = {
        limit = {
            $TITLE$ = title:e_pandaria_empire
            exists = title:e_pandaria_empire.holder
        }
        title:e_pandaria_empire.holder = {
            is_mogu_thunder_king_trigger = yes
        }
    }
    trigger_else = {
        exists = $TITLE$.holder
    }
}

can_be_in_coalition_against_trigger = {
    NOR = { 
        is_allied_to = $CRISIS_HOLDER$ 
        is_in_same_union_trigger = { TARGET = $CRISIS_HOLDER$ }
    }
}

is_in_same_coalition_trigger = {
    any_in_list = {
        variable = coalition_targets
        any_in_list = {
            variable = coalition_members

            this = $ALLY$
        }
    }
}
