# Confirmed via error.log: title_on_actions.txt (vanilla) declares its own raw
# on_vassal_gained/on_title_gain effect blocks. Jomini does NOT merge multiple
# raw `effect = {}` blocks for the same on_action across files - it keeps only
# the most-recent-loaded one and silently drops the other ("There is more than
# one 'effect' defined using most recent" in error.log). Alphabetically
# title_on_actions.txt < wc_balnazzar_on_actions.txt, so this file's raw block
# used to win and silently dropped vanilla's realm_maintenance.1000 vassal
# notification, confederation handling, and the entire county-tier title_gain
# block. `on_actions = { list }` (the redirect pattern) does NOT have this
# problem - that list accumulates across files instead of being overwritten.
# Any demon liege (Balnazzar included) auto-raises newly gained vassals as undead thralls.
# Reuses raise_undead_faith_only_effect (faith conversion, culture untouched) + loyal_undead opinion, same as wc_balnazzar_story.2000.
on_vassal_gained = {
	on_actions = {
		wc_balnazzar_on_vassal_gained_actions
	}
}

wc_balnazzar_on_vassal_gained_actions = {
	effect = {
		if = {
			limit = {
				root = { has_trait = being_demon }
				NOT = { scope:vassal = { has_trait = being_undead } }
			}
			scope:vassal = {
				raise_undead_faith_only_effect = { MASTER = root }
				add_opinion = { modifier = loyal_undead target = root }
				# _no_notification variant - on_vassal_gained fires constantly across every
				# demon-trait character in the mod (162+, not just Balnazzar), and a big
				# conquest can hand over an entire vassal tree in one pulse. The notifying
				# become_undead_effect leaks memory per call (see wc_being_effects.txt) -
				# fine for the single newly-gained vassal above, not for a bulk cascade.
				every_vassal_or_below = {
					limit = {
						is_alive = yes
						NOT = { has_trait = being_undead }
					}
					raise_undead_faith_only_no_notification_effect = { MASTER = liege }
					add_opinion = { modifier = loyal_undead target = liege }
				}
			}
		}
	}
}

# Stratholme taken by conquest (not grant/inheritance) while still disguised - fires wc_balnazzar_story.1007.
on_title_gain = {
	on_actions = {
		wc_balnazzar_on_title_gain_actions
	}
}

wc_balnazzar_on_title_gain_actions = {
	effect = {
		if = {
			limit = {
				has_trait = nathrezim_disguise
				scope:title = title:c_stratholme
				OR = {
					scope:transfer_type = flag:conquest
					scope:transfer_type = flag:conquest_holy_war
					scope:transfer_type = flag:conquest_claim
					scope:transfer_type = flag:conquest_populist
				}
			}
			trigger_event = {
				id = wc_balnazzar_story.1007
				days = 3
			}
		}
	}
}

# Redirect, not a raw on_title_lost effect block, so it doesn't clobber the other
# on_title_lost hooks in this mod. Losing Lordaeron kicks off the .1001->.1009->.1002
# chain that ends with him killing and possessing Saidan Dathrohan. .1001 itself applies
# the landless-adventurer conversion in its own immediate block, so no delay here either -
# the player is an adventurer within days of losing the county, not weeks.
on_title_lost = {
	on_actions = {
		wc_balnazzar_on_title_lost_actions
	}
}

wc_balnazzar_on_title_lost_actions = {
	effect = {
		if = {
			limit = {
				scope:title = title:c_lordaeron
				exists = global_var:balnazzar_dreadlord_story
				global_var:balnazzar_dreadlord_story = { story_owner = root }
				NOT = { has_trait = nathrezim_disguise }
				NOT = { has_character_flag = balnazzar_went_to_ground }
			}
			trigger_event = {
				id = wc_balnazzar_story.1001
				days = { 1 3 }
			}
		}
	}
}

# A ruler playing this arc at kingdom/empire tier can be conquered down into someone else's
# vassalage without c_lordaeron itself ever changing hands - a lesser vassal can hold that one
# county while the overlordship/duchies above it are what actually gets conquered. on_title_lost
# alone missed that case entirely. root here is the character whose liege status just changed
# (confirmed via game files: on_vassal_change's root is the vassal, not the new liege) - fires for
# both gaining a liege and gaining independence, filtered to the former via exists = liege.
on_vassal_change = {
	on_actions = {
		wc_balnazzar_on_vassal_change_actions
	}
}

wc_balnazzar_on_vassal_change_actions = {
	effect = {
		if = {
			limit = {
				exists = liege
				exists = global_var:balnazzar_dreadlord_story
				global_var:balnazzar_dreadlord_story = { story_owner = root }
				NOT = { has_trait = nathrezim_disguise }
				NOT = { has_character_flag = balnazzar_went_to_ground }
			}
			# Called directly, not through wc_balnazzar_story.1001's popup - a forced conquest
			# vassalization should never leave him sitting as someone's vassal even briefly.
			# .1001 stays the on_title_lost path's own entry point for players who lose
			# c_lordaeron while otherwise still independent.
			balnazzar_go_to_ground_effect = yes
		}
	}
}
