demand_child_as_monk_interaction = {
	category = interaction_category_religion
	icon = devoted
	interface_priority = 30
	common_interaction = no

	use_diplomatic_range = yes
	ai_maybe = no
	popup_on_receive = yes

	send_name = SEND_DEMAND
	desc = demand_child_as_monk_interaction_desc
	notification_text = demand_child_as_monk_notification
	prompt = demand_child_as_monk_prompt

	cost = {
		piety = {
			value = scope:actor.medium_piety_value
			if = {
				limit = {
					scope:actor = {
						domicile ?= { has_domicile_parameter = estate_cheaper_oblation }
					}
				}
				multiply = ecclesiastical_domicile_cheaper_oblation_mult
			}
			if = {
				limit = {
					scope:actor = { has_character_modifier = pam_oblate_patron_modifier }
				}
				multiply = 0.5
			}
		}
		treasury_or_gold = {
			value = scope:actor.minor_gold_value
			if = {
				limit = {
					scope:actor = {
						domicile ?= { has_domicile_parameter = estate_cheaper_oblation }
					}
				}
				multiply = ecclesiastical_domicile_cheaper_oblation_mult
			}
			if = {
				limit = {
					scope:actor = { has_character_modifier = pam_oblate_patron_modifier }
				}
				multiply = 0.5
			}
		}
	}

	cooldown_against_recipient = { years = 3 }
	cooldown = { years = 1 }

	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}

	send_options_exclusive = no

	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	# Hook forces auto-accept only against AI recipients, player recipients keep the choice but pay a piety-level cost in on_decline if they defy the hook.
	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
			scope:recipient = { is_ai = yes }
		}
	}

	redirect = {
		if = {
			limit = {
				scope:actor = { is_ai = no }
			}
			scope:recipient = {
				if = {
					limit = {
						is_ruler = no
						exists = employer
						employer != scope:recipient
					}
					save_scope_as = secondary_recipient
					employer = {
						save_scope_as = recipient
					}
				}
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = {
			every_courtier = {
				limit = {
					is_available_child = yes
					faith = scope:actor.faith
					NOR = {
						has_trait = devoted
						has_trait = order_member
					}
				}
				add_to_list = characters
			}
			every_child = {
				limit = {
					is_available = yes
					faith = scope:actor.faith
					is_courtier_of = scope:recipient
					NOR = {
						has_trait = devoted
						has_trait = order_member
					}
				}
				add_to_list = characters
				every_child = {
					limit = {
						is_available = yes
						faith = scope:actor.faith
						is_courtier_of = scope:recipient
						NOR = {
							has_trait = devoted
							has_trait = order_member
						}
					}
					add_to_list = characters
					every_child = {
						limit = {
							is_available = yes
							faith = scope:actor.faith
							is_courtier_of = scope:recipient
							NOR = {
								has_trait = devoted
								has_trait = order_member
							}
						}
						add_to_list = characters
					}
				}
			}
		}
	}

	is_available = {
		OR = {
			government_has_flag = government_is_theocracy
			government_has_flag = government_is_monastic_holy_order
		}
		trigger_if = {
			limit = { is_ai = yes }
			piety >= medium_piety_value
			short_term_gold >= minor_gold_value
			NOT = { has_trait = cynical }
		}
	}

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = {
			NOR = {
				government_has_flag = government_is_theocracy
				government_has_flag = government_is_monastic_holy_order
			}
			faith ?= scope:actor.faith
			exists = liege
			trigger_if = {
				limit = { scope:actor = { is_ai = yes } }
				NOT = { has_variable = demand_child_as_monk_ai_cooldown }
			}
		}
		trigger_if = {
			limit = { exists = scope:secondary_recipient }
			scope:secondary_recipient = {
				faith ?= scope:actor.faith
				NOR = {
					has_trait = devoted
					has_trait = order_member
				}
				OR = {
					is_adult = no
					OR = {
						is_child_of = scope:recipient
						is_grandchild_of = scope:recipient
						is_great_grandchild_of = scope:recipient
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_ruler = yes
			is_adult = yes
			is_available_quick = {
				imprisoned = no
				incapable = no
				alive = yes
			}
			NOT = { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			is_physically_able = yes
			custom_tooltip = {
				text = demand_child_as_monk_eligible_target_tt
				OR = {
					any_courtier = {
						is_available_child = yes
						faith = scope:actor.faith
						NOR = {
							has_trait = devoted
							has_trait = order_member
						}
					}
					any_child = {
						is_available_child = yes
						faith = scope:actor.faith
						NOR = {
							has_trait = devoted
							has_trait = order_member
						}
					}
					any_child = {
						any_child = {
							is_available_child = yes
							faith = scope:actor.faith
							NOR = {
								has_trait = devoted
								has_trait = order_member
							}
						}
					}
					any_child = {
						any_child = {
							any_child = {
								is_available_child = yes
								faith = scope:actor.faith
								NOR = {
									has_trait = devoted
									has_trait = order_member
								}
							}
						}
					}
				}
			}
		}
	}

	has_valid_target_showing_failures_only = {
		exists = scope:secondary_recipient
		scope:secondary_recipient = {
			faith = scope:actor.faith
			is_courtier_of = scope:recipient
			is_available = yes
			NOR = {
				has_trait = devoted
				has_trait = order_member
			}
			trigger_if = {
				limit = {
					is_adult = yes
				}
				custom_tooltip = {
					text = demand_child_as_monk_valid_secondary_target_tt
					OR = {
						is_child_of = scope:recipient
						is_grandchild_of = scope:recipient
						is_great_grandchild_of = scope:recipient
					}
				}
			}
		}
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = demand_child_as_monk_accept_actor_toast
				left_icon = scope:actor
				right_icon = scope:secondary_recipient
				if = {
					limit = { scope:hook = yes }
					use_hook = scope:recipient
				}
				add_courtier = scope:secondary_recipient
				add_opinion = {
					target = scope:recipient
					modifier = grateful_opinion
					opinion = 20
				}
				scope:secondary_recipient = {
					add_piety = major_piety_value
					if = {
						limit = { is_married = yes }
						every_spouse = {
							divorce = scope:secondary_recipient
						}
					}
					if = {
						limit = { exists = betrothed }
						break_betrothal = betrothed
						if = {
							limit = { has_been_promised_grand_wedding = yes }
							break_grand_wedding_betrothal_effect = yes
						}
					}
					if = {
						limit = {
							rite != scope:actor.rite
						}
						set_character_rite = scope:actor.rite
					}
					add_trait = devoted

					# Personality-driven opinion of the priest who claimed them
					if = {
						limit = {
							OR = {
								has_trait = humble
								has_trait = zealous
								has_trait = content
								has_trait = shy
								has_trait = patient
								AND = {
									faith = { trait_is_virtue = compassionate }
									has_trait = compassionate
								}
							}
						}
						add_opinion = {
							target = scope:actor
							modifier = grateful_for_calling_opinion
						}
					}
					else_if = {
						limit = {
							OR = {
								has_trait = ambitious
								has_trait = cynical
								has_trait = arrogant
								has_trait = lustful
								has_trait = brave
								has_trait = impatient
							}
						}
						add_opinion = {
							target = scope:actor
							modifier = forced_into_monastery_opinion
						}
					}
				}
			}
		}
		hidden_effect = {
			scope:recipient = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = demand_child_as_monk_accept_recipient_toast
					left_icon = scope:secondary_recipient
					right_icon = scope:actor
					show_as_tooltip = {
						scope:actor = {
							if = {
								limit = { scope:hook = yes }
								use_hook = scope:recipient
							}
							add_courtier = scope:secondary_recipient
							add_opinion = {
								target = scope:recipient
								modifier = grateful_opinion
								opinion = 20
							}
						}
						scope:secondary_recipient = {
							add_piety = major_piety_value
							if = {
								limit = { is_married = yes }
								every_spouse = {
									divorce = scope:secondary_recipient
								}
							}
							if = {
								limit = { exists = betrothed }
								break_betrothal = betrothed
							}
							add_trait = devoted

							# Personality-driven opinion of the priest who claimed them
							if = {
								limit = {
									OR = {
										has_trait = humble
										has_trait = zealous
										has_trait = content
										has_trait = shy
										has_trait = patient
										AND = {
											faith = { trait_is_virtue = compassionate }
											has_trait = compassionate
										}
									}
								}
								add_opinion = {
									target = scope:actor
									modifier = grateful_for_calling_opinion
								}
							}
							else_if = {
								limit = {
									OR = {
										has_trait = ambitious
										has_trait = cynical
										has_trait = arrogant
										has_trait = lustful
										has_trait = brave
										has_trait = impatient
									}
								}
								add_opinion = {
									target = scope:actor
									modifier = forced_into_monastery_opinion
								}
							}
						}
					}
				}
				if = {
					limit = { is_ai = no }
					set_variable = {
						name = demand_child_as_monk_ai_cooldown
						years = 3
					}
				}
			}
		}
	}

	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = demand_child_as_monk_decline_actor_toast
				right_icon = scope:recipient
				if = {
					limit = { scope:hook = yes }
					use_hook = scope:recipient
				}
				scope:recipient = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = demand_child_as_monk_decline_recipient_toast
						right_icon = scope:actor
						if = {
							limit = { scope:hook = yes }
							add_piety_level = -1
						}
						else = {
							add_piety = miniscule_piety_loss
						}
						reverse_add_opinion = {
							target = scope:actor
							modifier = refused_monk_demand_opinion
						}
					}
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { is_ai = no }
				set_variable = {
					name = demand_child_as_monk_ai_cooldown
					years = 3
				}
			}
		}
	}

				NOT = { is_primary_heir_of = scope:recipient }

					count >= 2
					age > prev.age
				}
			}
		}
		modifier = {
			add = 50

			scope:actor.rite = { rite_has_preferred_gender_for_clergy = scope:secondary_recipient }
			scope:secondary_recipient = {
				NOT = { is_primary_heir_of = scope:recipient }
				OR = {
					learning >= 12
					has_trait = humble
					has_trait = zealous
					has_trait = content
					has_trait = shy

			add = 100

				has_character_modifier = clerical_chastity_vow_modifier
			}
			scope:secondary_recipient = { is_heir_of = scope:recipient }
		}

		modifier = {
			add = 100
			scope:recipient = { has_relation_rival = scope:actor }
		}

		modifier = {
			factor = 0
			scope:recipient = {
				OR = {
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
				}
			}
		}
		modifier = {
			factor = 0
			scope:secondary_recipient.age < 10
		}
	}

		}

		# Parent eagerly gives up a distant child of their own to the cloister
		modifier = {
			add = 60
			scope:secondary_recipient = {
				is_child_of = scope:recipient
				NOT = { is_primary_heir_of = scope:recipient }
				OR = {
					AND = {
						is_male = yes
						any_sibling = {
							count >= 2

							age > prev.age
						}
					}
					AND = {
						is_female = yes
						any_sibling = {
							count >= 2
							is_female = yes
							age > prev.age
						}
					}
				}
			}
			desc = demand_child_as_monk_ai_distant_child
		}

		# Parent does not eagerly give up non distant children of their own to the cloister
		modifier = {
			add = -40
			scope:secondary_recipient = {
				is_child_of = scope:recipient
				OR = {
					AND = {
						is_male = yes
						any_sibling = {
							count < 2

							age > prev.age

			desc = demand_child_as_monk_ai_non_distant_child

		# Parent will fight tooth and nail to keep an heir from the cloister.
		# is_primary_heir_of can fire for a grandchild standing in line, so leaving it ungated.

		opinion_modifier = {
