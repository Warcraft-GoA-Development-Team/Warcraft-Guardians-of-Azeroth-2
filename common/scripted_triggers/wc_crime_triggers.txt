wc_killing_is_crime_trigger = { # takes two arguments: KILLER,  and VICTIM
	wc_hostile_act_is_crime_trigger = yes
	$KILLER$.culture = { 
		culture_not_pacifistic_trigger = yes
	}
	$KILLER$.faith = { like_pacifism_faith_trigger = no }
	NAND = { 
		$KILLER$ = { 
			OR = { 
				has_trait = demon_hunter
				has_character_flag = in_training_demon_hunter
			}
		}
		$VICTIM$.culture = { has_cultural_pillar = heritage_demonic }
	}
}

wc_hostile_act_is_crime_trigger = { # takes two arguments: ACTOR,  and VICTIM
	OR = { 
		$ACTOR$.faith = { 
			faith_hostility_level = { 
				target = $VICTIM
				value = faith_evil_level
			}
		}
		# maybe add something for "eye for an eye" and other vengeful culture traits here?
	}
}
