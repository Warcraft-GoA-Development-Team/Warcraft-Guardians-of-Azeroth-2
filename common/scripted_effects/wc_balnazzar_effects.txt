# Called on losing/being forced to swear away c_lordaeron (wc_balnazzar_on_actions.txt's
# on_title_lost/on_vassal_change). Only gets Balnazzar to is_landless_ruler = yes, the
# precondition balnazzar_become_dathrohan_decision checks - the story flags stay fully
# gated behind the player clicking the decision, untouched here.
balnazzar_become_landless_adventurer_effect = {
	if = {
		limit = { top_liege != this }
		liege = { save_scope_as = balnazzar_land_recipient }
		every_held_title = {
			title_tier > barony
			limit = {
				OR = {
					is_landless_type_title = no
					is_nomad_title = yes
				}
			}
			add_to_list = balnazzar_relinquished_titles
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = balnazzar_land_change
			add_claim_on_loss = no
		}
		every_in_list = {
			list = balnazzar_relinquished_titles
			change_title_holder_include_vassals = {
				holder = scope:balnazzar_land_recipient
				change = scope:balnazzar_land_change
			}
		}
		resolve_title_and_vassal_change = scope:balnazzar_land_change
	}
	create_landless_adventurer_title_effect = {
		REASON = flag:voluntary
		FLAVOR_CHAR = root
	}
}

# Called from balnazzar_become_dathrohan_decision, player-triggered only.
# is_landless_ruler = yes is already required to see the decision, so there's
# no land left to relinquish - just mark it done and move to the kill itself.
balnazzar_become_dathrohan_effect = {
	# Guards against firing twice - conquest can trip multiple hooks in one update.
	add_character_flag = balnazzar_became_dathrohan

	trigger_event = {
		id = wc_balnazzar_story.1001
		days = { 1 3 }
	}
}

# Call on the character possessing Dathrohan's body. Grants the secret, the
# companion query flag every gate checks, and the mechanical bonuses that
# used to live on a visible nathrezim_disguise trait.
add_dreadlord_disguise_secret_effect = {
	add_character_flag = wc_balnazzar_disguise_active
	add_character_modifier = {
		modifier = wc_nathrezim_disguise_modifier
	}
	add_secret = {
		type = secret_dreadlord_disguise
		target = this
	}
}

# Call at the reveal (wc_balnazzar_story.2000) on the character shedding the disguise.
remove_dreadlord_disguise_secret_effect = {
	remove_character_flag = wc_balnazzar_disguise_active
	remove_character_modifier = wc_nathrezim_disguise_modifier
}

# Called right after murder_scheme starts on Alexandros with scope:renault and
# scope:balnazzar_alexandros_murder_scheme already set. grab_best_agent_murderer_effect
# picks a random FILLED agent_assassin slot - leaving only one filled removes the
# randomness, so Renault (not a randomly recruited agent) ends up doing the killing.
wc_balnazzar_force_renault_as_assassin_effect = {
	scope:balnazzar_alexandros_murder_scheme = {
		random_scheme_agent_slot = {
			limit = {
				is_agent_slot_type = agent_assassin
				is_filled = no
				save_temporary_scope_as = balnazzar_slot_temp
				scope:renault = { is_valid_as_agent_in_slot = scope:balnazzar_slot_temp }
			}
			save_scope_as = balnazzar_renault_slot
		}
	}
	if = {
		limit = { exists = scope:balnazzar_renault_slot }
		scope:renault = {
			force_add_to_agent_slot = {
				agent_slot = scope:balnazzar_renault_slot
				years = 5
			}
		}
		# Remove the other default assassin slot so no other recruited agent can
		# out-compete Renault for the kill.
		scope:balnazzar_alexandros_murder_scheme = {
			random_scheme_agent_slot = {
				limit = {
					is_agent_slot_type = agent_assassin
					is_filled = no
				}
				save_scope_as = balnazzar_spare_assassin_slot
			}
		}
		if = {
			limit = { exists = scope:balnazzar_spare_assassin_slot }
			scope:balnazzar_alexandros_murder_scheme = {
				remove_agent_slot = scope:balnazzar_spare_assassin_slot
			}
		}
	}
}
