action_join_coalition = {
	priority = 1000
	combine_into_one = yes
	check_create_action = {
		if = {
            limit = {
                can_be_in_coalition_trigger = yes 
                has_crisis_on_same_continent_trigger = yes
            }
            every_ruler = {
                limit = {
                    is_crisis_trigger = yes
                }
                save_scope_as = crisis_holder
                if = {
                    limit = {
                        root = {
                            is_character_interaction_valid = {
								recipient = scope:crisis_holder
								interaction = join_coalition_interaction
							}
                        }
                    }
                    try_create_important_action = {
                        important_action_type = action_join_coalition
                        actor = root
                        recipient = scope:crisis_holder
                    }
                }
            }
        }
	}
	
	effect = {
		open_interaction_window = {
			interaction = join_coalition_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
