# Called from code when a player refunds their perks
# Root is the player
on_perks_refunded = {
	effect = {
		refund_all_perks = yes
		add_stress = refund_perks_stress_impact
		add_character_flag = {
			flag = has_refunded_perks
		}
		remove_all_perk_lifestyle_traits_effect = yes

		# Warcraft
		wc_remove_all_wc_lifestyle_traits_effect = yes
		clear_variable_list = known_spells
		set_variable = {
			name = known_spells_count value = 0
		}
		clear_variable_list = known_rank_2_spells
		clear_variable_list = known_rank_3_spells
	}
}
