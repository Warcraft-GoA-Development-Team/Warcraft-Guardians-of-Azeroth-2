# Historical heresy outbreaks: values. See wc_heresy_effects.txt.

# Converts the spread sweep may make (vanilla pam_heresy_c3_spread_budget_base_value)
wc_heresy_spread_budget_value = {
	value = 20
}

# Years between two outbreaks of the same heresy
wc_heresy_cooldown_years_value = {
	value = 50
}

# Anonymous preachers follow the heresy faith's gender doctrine
wc_heresy_preacher_female_chance_value = {
	value = 50
	if = {
		limit = { scope:heresy_faith.main_rite = { rite_has_doctrine = doctrine_gender_male_dominated } }
		value = 0
	}
	else_if = {
		limit = { scope:heresy_faith.main_rite = { rite_has_doctrine = doctrine_gender_female_dominated } }
		value = 100
	}
}

# Monthly chance of no outbreak for wc_faith_historical_heresy_pulse (common/on_action/wc_heresy_on_actions.txt).
# Copy of vanilla pam_heresy_pulse_no_event_value (pam_heresy_values.txt) without its small-faith multiplier:
# GoA's historical heresies start in small faiths on purpose.
wc_heresy_pulse_no_event_value = {
	if = {
		limit = { fervor > 30 }
		value = 99
	}
	else_if = {
		limit = { fervor > 20 }
		value = 32
	}
	else_if = {
		limit = { fervor > 10 }
		value = 11
	}
	else = { value = 2 }
	if = {
		limit = { has_game_rule = much_less_heresy_outbreak_frequency }
		multiply = 4
	}
	else_if = {
		limit = { has_game_rule = less_heresy_outbreak_frequency }
		multiply = 2
	}
	else_if = {
		limit = { has_game_rule = more_heresy_outbreak_frequency }
		multiply = 0.5
	}
	else_if = {
		limit = { has_game_rule = much_more_heresy_outbreak_frequency }
		multiply = 0.25
	}
	min = 1
}
