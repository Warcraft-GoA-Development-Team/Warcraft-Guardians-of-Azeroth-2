story_path_of_the_damned = {
	visible = yes
	icon = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/icons/story_cycles/story_icon_path_of_the_damned.dds"
	}
	background = {
		reference = "gfx/interface/illustrations/event_scenes/wc_frostmourne_cave.dds"
	}
	visualization = {
		custom_string_key = "PATH_OF_THE_DAMNED_CUSTOM_STRING_KEY"
		traits = { champion_of_the_scourge }

		basic_counter = {
			variable_name = "story_stage"
			min = 1
			max = 4
			label = "PATH_OF_THE_DAMNED_STAGE_LABEL"
			min_label = "PATH_OF_THE_DAMNED_STAGE_MIN_LABEL"
			max_label = "PATH_OF_THE_DAMNED_STAGE_MAX_LABEL"
		}
	}

	on_setup = {
		set_global_variable = { # Save the story globally
			name = path_of_the_damned_story
			value = this 
		}

		set_variable = {
			name = story_stage
			value = 1
		}

		# Save variables
		if = { # Kel'Thuzad
			limit = { title:h_scourge.holder = { has_variable = necromancer } }
			set_variable = { name = necromancer value = title:h_scourge.holder.var:necromancer }
			title:h_scourge.holder = { remove_variable = necromancer }
		}

		if = { # Mal'ganis
			limit = { story_owner = { has_variable = dreadlord } }
			set_variable = { name = dreadlord value = story_owner.var:dreadlord }
			story_owner = { remove_variable = dreadlord }
		}

		if = { # Frostmourne
			limit = { story_owner = { has_variable = frostmourne } }
			set_variable = { name = frostmourne value = story_owner.var:frostmourne }
			story_owner = { remove_variable = frostmourne }
		}

		set_variable = { name = plagued_king value = global_var:plagued_king } # Terenas

		if = { # Trudging through the Ashes
			limit = {
				exists = var:plagued_king
				story_owner = {
					NOT = { this = title:h_scourge.holder }
					any_killed_character = { this = scope:story.var:plagued_king }
				}
			}

			story_owner = {
				trigger_event = { # Only if you accomplished the activity
					id = wc_path_of_the_damned.1001
					days = 14
				}
			}
		}
		else_if = { # Natural death of Terenas
			limit = {
				exists = var:plagued_king
				story_owner = { this = title:h_scourge.holder }
			}

			var:plagued_king = {
				trigger_event = {
					id = wc_path_of_the_damned.0003
					months = { 2 12 }
				}
			}
		}

		if = { # Arthas things
			limit = { story_owner = character:60003 }

			story_owner = { # Arthas raise invincible
				add_character_flag = is_in_naming_event
				trigger_event = {
					id = wc_path_of_the_damned.0001
					days = 2
				}
			}
		}
	}

	on_end = {
		remove_global_variable = path_of_the_damned_story
	}

	on_owner_death = {
		# Save scopes
		story_owner = { save_scope_as = old_champion }
		title:h_scourge.holder = { save_scope_as = lich_king }

		if = {
			limit = { exists = scope:lich_king }
			defeat_of_the_champion_effect = yes
		}
		else = { end_story = yes }
	}

	effect_group = { # You lost Frostmourne
		days = { 15 30 }

		trigger = {
			story_owner = { NOT = { any_character_artifact = { has_variable = frostmourne } } }
		}

		triggered_effect = {
			effect = {
				resolve_lost_artifact_effect = {
					ARTIFACT_VARIABLE = frostmourne
					CLAIM_MSG = msg_frostmourne_claim
				}
			}
		}
	}

	effect_group = { # You lost the Urn/Phylactery
		days = { 15 30 }

		trigger = {
			var:story_stage = 3 # Only between Digging up the Dead and Kel'Thuzad's resurrection
			story_owner = { NOT = { any_character_artifact = { has_variable = urn_of_king_of_lordaeron } } }
		}

		triggered_effect = {
			effect = {
				resolve_lost_artifact_effect = {
					ARTIFACT_VARIABLE = urn_of_king_of_lordaeron
					CLAIM_MSG = msg_urn_claim
				}
			}
		}
	}

	effect_group = { # A new Champion
		days = { 30 60 }

		trigger = {
			story_owner = {
				# Is the Lich King
				any_held_title = { this = title:h_scourge }
				has_trait = lich_king

				# Frostmourne
				any_character_artifact = { has_variable = frostmourne }
			}
		}

		triggered_effect = {
			effect = {
				story_owner = {
					save_scope_as = lich_king # Save scope

					ordered_vassal_or_below = { # Find a new champion
						limit = {
							is_adult = yes
							has_religion = prev.religion
						}
						order_by = martial
						max = 1

						save_scope_as = new_champion
					}

					if = {
						limit = { exists = scope:new_champion }

						every_character_artifact = {
							limit = { has_variable = frostmourne }
							save_scope_as = frostmourne
						}

						# Notification and effect
						trigger_event = wc_path_of_the_damned.0101
						every_vassal_or_below = { trigger_event = wc_path_of_the_damned.0101 }
					}
				}

				if = {
					limit = { exists = scope:new_champion }

					make_story_owner = scope:new_champion
				}
			}
		}
	}

	effect_group = { # Nerubian reinforcements
		months = 2
		chance = 50

		trigger = {
			var:story_stage >= 2
			story_owner = { NOT = { has_character_flag = nerubian_troops_flag } }
		}

		triggered_effect = {
			effect = {
				story_owner = {
					add_character_flag = nerubian_troops_flag
					trigger_event = wc_path_of_the_damned.0200
				}
			}
		}
	}

	effect_group = { # Banshee reinforcements
		months = 2
		chance = 50

		trigger = {
			var:story_stage >= 3
			story_owner = { NOT = { has_character_flag = banshee_troops_flag } }
		}

		triggered_effect = {
			effect = {
				story_owner = {
					add_character_flag = banshee_troops_flag
					trigger_event = wc_path_of_the_damned.0202
				}
			}
		}
	}

	effect_group = { # Frost Wyrm reinforcements
		months = 2
		chance = 50

		trigger = {
			var:story_stage >= 4
			story_owner = { NOT = { has_character_flag = frost_wyrm_troops_flag } }
		}

		triggered_effect = {
			effect = {
				story_owner = {
					add_character_flag = frost_wyrm_troops_flag
					trigger_event = wc_path_of_the_damned.0203
				}
			}
		}
	}
}
