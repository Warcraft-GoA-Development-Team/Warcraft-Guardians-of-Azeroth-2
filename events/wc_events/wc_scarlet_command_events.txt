namespace = wc_balnazzar_story

# Historical event IDs retained for pending saves; these scenes now belong to the Crusade.

wc_balnazzar_story.1008 = {
	type = character_event
	trigger = {
		wc_scarlet_stratholme_owner_valid_trigger = yes
		global_var:wc_scarlet_crusade_title = {
			NOT = { exists = var:wc_scarlet_stratholme_open }
		}
	}
	title = wc_balnazzar_story.1008.title
	desc = wc_balnazzar_story.1008.desc
	theme = intrigue

	override_background = wc_stratholme

	left_portrait = {
		character = root
		animation = schadenfreude
	}

	option = {
		name = wc_balnazzar_story.1008.a
		trigger = { wc_scarlet_stratholme_choice_valid_trigger = yes }
		if = {
			limit = { wc_scarlet_stratholme_choice_valid_trigger = yes }
			hidden_effect = {
				global_var:wc_scarlet_crusade_title = {
					set_variable = { name = wc_scarlet_stratholme_resolved value = yes }
				}
			}
			add_prestige = major_prestige_value
			wc_scarlet_grant_stratholme_effect = yes
		}
	}
	option = {
		name = wc_balnazzar_story.0101.close
		trigger = { wc_scarlet_stratholme_choice_valid_trigger = no }
	}
	immediate = {
		wc_scarlet_migrate_crusade_effect = yes
		global_var:wc_scarlet_crusade_title = {
			set_variable = { name = wc_scarlet_stratholme_open value = root }
		}
	}
	after = {
		wc_scarlet_clear_legacy_windows_effect = yes
		if = {
			limit = {
				exists = global_var:wc_scarlet_crusade_title
				global_var:wc_scarlet_crusade_title = {
					exists = var:wc_scarlet_stratholme_open
					var:wc_scarlet_stratholme_open = root
				}
			}
			global_var:wc_scarlet_crusade_title = { remove_variable = wc_scarlet_stratholme_open }
		}
	}
}
