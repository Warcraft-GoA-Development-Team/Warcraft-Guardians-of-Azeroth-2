pam_adorcism_events.0001 = {
	type = character_event
	title = pam_adorcism_events.0001.t
	desc = pam_adorcism_events.0001.desc
	theme = faith
	override_background = { reference = fp3_cave }
	override_effect_2d = { reference = fog }

	left_portrait = {
		character = root
		animation = divine_ecstasy
	}
	lower_right_portrait = {
		character = scope:pam_adorcism_curse_target
		animation = worry
	}

	trigger = {
		has_personal_tenet_flag = tenet_adorcism_curse_enemy
		pam_adorcism_curse_target_is_valid_trigger = {
			TARGET = scope:pam_adorcism_curse_target
			ACTOR = root
		}
	}

	option = {
		name = pam_adorcism_events.0001.a
		add_character_modifier = {
			modifier = pam_adorcism_curse_focus_modifier
			years = 3
		}
		add_stress = major_stress_loss
		change_spiritual_fulfillment = minor_spiritual_fulfillment_value
		save_scope_as = pam_adorcism_curser

		if = {
			limit = {
				pam_adorcism_curse_target_is_valid_trigger = {
					TARGET = scope:pam_adorcism_curse_target
					ACTOR = root
				}
			}
			scope:pam_adorcism_curse_target = {
				if = {
					limit = {
						rite = root.rite
					}
					add_character_modifier = {
						modifier = pam_adorcism_ill_omen_modifier
						years = 1
					}
				}
				if = {
					limit = {
						OR = {
							rite = root.rite
							probably_superstitious = yes
						}
					}
					add_character_modifier = {
						modifier = pam_adorcism_curse_anxiety_modifier
						years = 1
					}
					trigger_event = pam_adorcism_events.0002
				}
			}
		}
	}
}
