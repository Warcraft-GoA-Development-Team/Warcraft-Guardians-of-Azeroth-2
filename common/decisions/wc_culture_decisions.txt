build_cosmic_observatory_decision = { 
	picture = { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" } #TODO
	decision_group_type = major

	ai_check_interval = 36

	desc = build_cosmic_observatory_decision_desc

	is_shown = {
		culture = { has_cultural_pillar = heritage_highborne }
		is_landed = yes
		highest_held_title_tier >= tier_duchy
		OR = { # Show it earlier so it's something to work towards
			has_trait = lifestyle_mystic
			has_trait = scholar
			has_education_learning_trigger = yes
		}
		NOR = {
			top_liege = { 
				any_sub_realm_county = {
					has_variable = has_cosmic_observatory
				}
			}
			has_trait = demon_hunter # Demon Hunters shouldn't become void being. Canonically not a thing.
		}
		
		# MAGIC 2024 TODO: void magic pathway
	}

	is_valid = {
		has_trait = lifestyle_mystic
		OR = { 
			has_trait = education_learning_5
			AND = { 
				OR = { 
					has_trait = education_learning_3
					has_trait = education_learning_4
				}
				has_trait = scholar
			}
			# MAGIC 2024 TODO: void magic pathway
		}
	}

	is_valid_showing_failures_only = { 
		custom_tooltip = {
			text = cant_be_hof
			NOT = {
				any_held_title = {
					is_head_of_faith = yes
				}
			}
		}
	}

	widget = {
		controller = create_holy_order
		gui = "decision_view_widget_place_special_building"
		barony_valid = {
			OR = {
				holder = scope:ruler
				AND = { 
					holder.primary_title.tier = tier_barony
					exists = holder.liege
					holder.liege = scope:ruler
				}
			}
			title_province = {
				has_special_building = no
				has_special_building_slot = no
			}
		}
	}

	effect = {
		scope:barony ?= {
			title_province = {
				add_special_building_slot = cosmic_observatory
			}
			county = {
				set_variable = has_cosmic_observatory
			}
		}
		
		add_character_flag = can_build_observatory
	}
}

become_a_demon_hunter_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/becoming_a_demon_hunter.dds" }
	decision_group_type = major

	ai_check_interval = 60

	desc = become_a_demon_hunter_decision_desc

	is_shown = {
		OR = {
			has_trait = creature_night_elf
			has_trait = creature_high_elf
		}
		NOR = {
			has_trait = demon_hunter
			has_trait = being_void # Because of event: void_elf.0130
		}
	}

	is_valid = {
		OR = { # Only certain personalities can take such a drastic decision
			has_trait = vengeful
			has_trait = callous
			has_trait = cynical # Since you don't trust your old religion to defeat the demons
			has_trait = wrathful
		}
		NOR = {
			has_trait = forgiving
			has_trait = compassionate
			has_trait = trusting
			has_trait = loyal
			has_trait = zealous # Why would you abandon your old religion so easily?
			has_trait = content
			has_trait = craven
			has_trait = calm
		}
	}

	cost = {
		piety = 2000
        prestige = 2000
    }

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}

	effect = {
		add_trait = demon_hunter
		add_trait = blind        # Should instead be an event, when the Demon Hunter trait track reaches XP = 66
		add_trait = being_demon  # Should instead be an event, when the Demon Hunter trait track reaches XP = 100
		show_as_tooltip = { set_character_faith = faith:illidari }
		hidden_effect = {
			set_character_faith_with_conversion = faith:illidari
		}
	}

	ai_potential = {
		is_available_adult = yes
		is_at_war = no
	}
	ai_will_do = {
		base = 5
		modifier = {
			add = 995
			has_faith = faith:illidari # If you are already a Vindicatory faith elf but haven't became a Demon Hunter yet
		}
		modifier = {
			add = 20
			current_date >= 618.1.1 # Lore-wise, this is when Demon Hunters canonically start appearing
		}
	}
}
